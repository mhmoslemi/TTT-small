import numpy as np
from scipy.optimize import minimize
import time


def sort_key_score(item):

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size), mode="edge")
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(250):
        s = float(np.sum(h, dtype=np.float64))
        diff = target - s
        if abs(diff) <= 1e-13 * max(1.0, n):
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if not np.any(mask):
                break
            tot = float(room[mask].sum(dtype=np.float64))
            if tot <= 1e-18:
                break
            inc = np.zeros(n, dtype=float)
            inc[mask] = (diff / tot) * room[mask]
            h += inc
        else:
            mass = h.copy()
            mask = mass > 1e-14
            if not np.any(mask):
                break
            tot = float(mass[mask].sum(dtype=np.float64))
            if tot <= 1e-18:
                break
            dec = np.zeros(n, dtype=float)
            dec[mask] = (diff / tot) * mass[mask]
            h += dec
    np.clip(h, 0.0, 1.0, out=h)
    return h


def make_exact_sum(h, n):
    h = project_feasible(h, n)
    target = n / 2.0
    for _ in range(30):
        s = float(np.sum(h, dtype=np.float64))
        diff = target - s
        if diff == 0.0 or abs(diff) <= 1e-15:
            break
        idx = int(np.argmin(np.abs(h - 0.5)))
        if diff > 0:
            if h[idx] < 1.0:
                h[idx] = min(1.0, h[idx] + diff)
        else:
            if h[idx] > 0.0:
                h[idx] = max(0.0, h[idx] + diff)
        s = float(np.sum(h, dtype=np.float64))
        diff = target - s
        if diff > 0:
            idx = int(np.argmin(np.abs(h - 0.5)))
            if h[idx] < 1.0:
                h[idx] = np.nextafter(h[idx], 1.0)
        elif diff < 0:
            idx = int(np.argmin(np.abs(h - 0.5)))

                h[idx] = np.nextafter(h[idx], 0.0)
    return h


def add_unique_candidate(pool, h, n, tol=1e-12, max_len=60):
    hp = project_feasible(h, n)
    if len(pool) >= max_len:
        return pool
    for old in pool:
        if np.linalg.norm(hp - old) <= tol:
            return pool
    pool.append(hp)
    return pool


def compute_c_vec_fft(x, n, nfft=None):
    x = np.asarray(x, dtype=float)
    n = int(n)
    dx = 2.0 / float(n)
    if nfft is None:
        nfft = next_pow2(2 * n - 1)
    F = np.fft.rfft(x, nfft)
    auto = np.fft.irfft(F * np.conj(F), n=nfft).real[:n]
    auto = np.maximum(auto, 0.0)
    P = np.empty(n + 1, dtype=float)
    P[0] = 0.0
    P[1:] = np.cumsum(x)
    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1] = P[n] - auto[0]
    if n > 1:
        r = np.arange(1, n)
        c[n:] = P[n - r] - auto[r]
        c[:n - 1] = (P[n] - P[r][::-1]) - auto[r][::-1]
    c *= dx
    return c


def compute_c(h, n, nfft=None):
    return float(compute_c_vec_fft(h, n, nfft).max())


def softmax_c(c, alpha):
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 1e-300:
        e = np.zeros_like(c, dtype=float)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s
    logsum = mx + float(np.log(s))
    return w, m, logsum


def full_gradient_from_weights(x, n, w, dx, nfft3=None):
    x = np.asarray(x, dtype=float)
    n = int(n)
    w = np.asarray(w, dtype=float)
    if n == 1:
        return dx * w[0] * (1.0 - 2.0 * x)
    c0 = n - 1
    w_pos = w[c0:]
    w_neg = w[:c0][::-1]
    cum_pos = np.cumsum(w_pos)
    pos_vals = cum_pos[n - 1::-1]
    neg_pre = np.zeros(n, dtype=float)
    neg_pre[1:] = np.cumsum(w_neg)
    g = dx * (pos_vals + neg_pre)
    diag = 2.0 * dx * w_pos[0]
    if diag != 0.0:
        g = g - diag * x
    mu_kernel = dx * (w_pos[1:] + w_neg)
    if np.any(mu_kernel > 1e-300):
        if nfft3 is None:
            nfft3 = next_pow2(3 * n - 2)
        K = np.zeros(2 * n - 1, dtype=float)
        K[:n - 1] = mu_kernel[::-1]
        K[n:] = mu_kernel
        Kp = np.zeros(nfft3, dtype=float)
        Kp[:len(K)] = K
        conv = np.fft.irfft(np.fft.rfft(x, nfft3) * np.fft.rfft(Kp, nfft3), n=nfft3).real
        g = g - conv[c0:c0 + n]
    return g


def full_smooth_obj(x, n, alpha, mu, nfft2=None):
    c = compute_c_vec_fft(x, n, nfft2)
    w, m, logsum = softmax_c(c, alpha)
    val = m + logsum / float(alpha)
    if mu > 0.0:
        s = float(np.sum(x))
        val += mu * (s - n / 2.0) ** 2
    return float(val)


def full_smooth_grad(x, n, alpha, mu, nfft2=None, nfft3=None):
    x = np.asarray(x, dtype=float)
    c = compute_c_vec_fft(x, n, nfft2)
    w, m, logsum = softmax_c(c, alpha)
    dx = 2.0 / float(n)
    g = full_gradient_from_weights(x, n, w, dx, nfft3)
    if mu > 0.0:
        s = float(np.sum(x))
        g += 2.0 * mu * (s - n / 2.0)
    return g


def estimate_times(x, n, alpha, mu, nfft2, nfft3):
    try:
        t1 = time.time()
        full_smooth_obj(x, n, alpha, mu, nfft2)
        t1 = max(1e-5, time.time() - t1)
        t2 = time.time()
        full_smooth_grad(x, n, alpha, mu, nfft2, nfft3)
        t2 = max(1e-5, time.time() - t2)
        return float(t1), float(t2)
    except Exception:
        return 0.005, 0.010


def optimize_full(h, n, end, alphas=None, mu=0.02, maxiters=None):
    x = project_feasible(h, n)
    best_x = x.copy()
    best_c = compute_c(x, n)
    if alphas is None or len(alphas) == 0:
        return best_x, best_c
    nfft2 = next_pow2(2 * n - 1)
    nfft3 = next_pow2(3 * n - 2)
    bounds = [(0.0, 1.0)] * n
    t_obj, t_grad = estimate_times(x, n, min(max(10.0, float(alphas[0])), 200000.0), mu, nfft2, nfft3)
    fg = max(1e-5, t_obj + t_grad)
    per_iter = 5.0 * fg
    for i, a in enumerate(alphas):
        rem = end - time.time()
        if rem < 8.0:
            break
        slab = rem / float(max(1, len(alphas) - i))
        if maxiters is not None and i < len(maxiters):
            maxit = int(maxiters[i])
        else:
            maxit = int(slab / max(1e-6, per_iter))
        maxit = max(20, min(2500, int(maxit)))
        maxfun = max(3 * maxit, int(slab / max(1e-6, fg)))
        maxfun = min(maxfun, 25 * maxit + 200)
        try:
            res = minimize(
                full_smooth_obj,
                x,
                jac=full_smooth_grad,
                args=(n, float(a), float(mu), nfft2, nfft3),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxit),
                    "ftol": 1e-20,
                    "gtol": 1e-12,
                    "maxfun": int(maxfun),
                    "maxcor": 30,
                    "maxls": 50,
                },
            )
            y = project_feasible(res.x, n)
            cy = compute_c(y, n)
            if cy < best_c:
                best_c = cy
                best_x = y
            if cy <= best_c + 1e-10:
                x = y
        except Exception:
            pass
    return best_x, best_c


def active_obj(x, n, active, alpha, mu, nfft2=None):

    vals = c[active]
    if len(vals) == 0:

    w, m, logsum = softmax_c(vals, alpha)

    if mu > 0.0:
        s = float(np.sum(x))

    return float(val)


def active_grad(x, n, active, alpha, mu, nfft2=None):
    x = np.asarray(x, dtype=float)
    n = int(n)
    c = compute_c_vec_fft(x, n, nfft2)
    vals = c[active]
    if len(vals) == 0:
        idx = int(np.argmax(c))
        vals = c[idx:idx + 1]
        active = np.array([idx], dtype=int)
    w, m, logsum = softmax_c(vals, alpha)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    c0 = n - 1
    for wi, idx in zip(w, active):
        q = int(idx) - c0
        fac = float(wi) * dx
        if fac == 0.0:
            continue
        if q >= 0:
            nv = n - q
            g[:nv] += fac * (1.0 - x[q:])
            g[q:] += fac * (-x[:nv])
        else:
            r = -q
            nv = n - r
            g[r:] += fac * (1.0 - x[:nv])
            g[:nv] += fac * (-x[r:])
    if mu > 0.0:
        s = float(np.sum(x))
        g += 2.0 * mu * (s - n / 2.0)
    return g


def run_active_lbfgs(h, n, end, alphas=None, top_k=4, rounds=3, mu=0.002):

            m = float(c.max())
            tol = max(1e-9, 1e-3 * m)
            act = np.where(c >= m - tol)[0]
            if len(act) > top_k:
                order = np.argsort(-c[act])
                act = act[order[:top_k]]
            if len(act) == 0:
                act = np.array([int(np.argmax(c))], dtype=int)
            slab = rem / float(max(1, total_runs - done))
            per_iter = 0.004 + 0.000016 * float(n)
            maxit = int(slab / max(1e-6, per_iter))
            maxit = max(30, min(600, maxit))
            maxfun = max(3 * maxit, int(slab / max(1e-6, 0.5 + 0.00005 * float(n))))
            maxfun = min(maxfun, 25 * maxit + 200)
            try:
                res = minimize(
                    active_obj,
                    x,
                    jac=active_grad,
                    args=(n, act, float(a), float(mu), nfft2),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": int(maxit),
                        "ftol": 1e-20,
                        "gtol": 1e-12,
                        "maxfun": int(maxfun),
                        "maxcor": 25,
                        "maxls": 40,
                    },
                )
                y = project_feasible(res.x, n)
                cy = compute_c(y, n)
                if cy < best_c:
                    best_c = cy
                    best_x = y
                if cy <= best_c + 1e-10:
                    x = y
            except Exception:
                pass
            done += 1
        if time.time() >= end - 5.0:
            break
    return best_x, best_c


def resample_h(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)


def resample_nearest(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    idx = np.searchsorted(x_old, x_new, side="right") - 1
    idx = np.clip(idx, 0, n_old - 1)
    h_new = h_old[idx]
    return project_feasible(h_new, n_new)


def lowpass_h(h, n, keep=None):
    h = np.asarray(h, dtype=float)
    n = int(n)
    f = 2.0 * h - 1.0
    F = np.fft.rfft(f)
    if keep is None:
        keep = max(3, n // 50)
    keep = min(int(keep), F.size)
    F = F.copy()
    F[keep:] = 0.0
    fl = np.fft.irfft(F, n=n).real
    mx = float(np.max(np.abs(fl)))
    if mx > 1e-14:
        fl = fl / mx
    h2 = 0.5 + 0.5 * np.clip(fl, -1.0, 1.0)
    return project_feasible(h2, n)


def make_parent_variants(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    out = []
    h = project_feasible(h, n)
    add_unique_candidate(out, h, n)
    add_unique_candidate(out, 1.0 - h, n)
    add_unique_candidate(out, h[::-1], n)
    add_unique_candidate(out, 1.0 - h[::-1], n)
    if n > 16:
        shifts = [n // 16, n // 8, n // 4, -n // 16, -n // 8, -n // 4]
        for sh in shifts:
            add_unique_candidate(out, np.roll(h, sh), n)
    sym = 0.5 * (h + h[::-1])
    add_unique_candidate(out, sym, n)
    f = 2.0 * h - 1.0
    f_ant = 0.5 * (f - f[::-1])
    add_unique_candidate(out, 0.5 + 0.5 * np.clip(f_ant, -1.0, 1.0), n)
    lm = max(3, n // 50)
    add_unique_candidate(out, lowpass_h(h, n, lm), n)
    mx = float(np.max(np.abs(f)))
    if mx > 1e-14:
        add_unique_candidate(out, 0.5 + 0.5 * np.tanh(3.0 * f / mx), n)
        add_unique_candidate(out, 0.5 + 0.25 * (f / mx), n)
    return out


def make_basis(n, max_mode=8):
    if n <= 0:
        return []
    x = 2.0 * np.arange(n) / float(n)
    bases = []
    for m in range(1, max_mode + 1):
        bases.append(np.cos(np.pi * float(m) * x))
        bases.append(np.sin(np.pi * float(m) * x))
    return bases


def make_fourier_candidates(n, rng, count, bases=None, max_mode=8):
    n = int(max(2, n))
    out = []
    if count <= 0:
        return out
    if bases is None:
        bases = make_basis(n, max_mode)
    mn = len(bases) // 2
    for _ in range(count):
        raw = np.zeros(n, dtype=float)
        for m in range(1, mn + 1):
            idx = 2 * (m - 1)
            amp = 1.0 / np.sqrt(float(m))
            a = float(rng.standard_normal()) * amp
            b = float(rng.standard_normal()) * amp
            raw += a * bases[idx] + b * bases[idx + 1]
        mx = float(np.max(np.abs(raw)))
        if mx <= 1e-12:
            continue
        f = raw / mx
        add_unique_candidate(out, 0.5 + 0.5 * f, n)
        add_unique_candidate(out, 0.5 + 0.25 * f, n)
        order = np.argsort(raw)
        hi = np.zeros(n, dtype=float)
        lo = np.zeros(n, dtype=float)
        half = n // 2
        if half > 0:
            hi[order[half:]] = 1.0
            lo[order[:half]] = 1.0
            add_unique_candidate(out, hi, n)
            add_unique_candidate(out, lo, n)
    return out


def make_beatty_candidates(n, alphas, rng, max_betas=2):
    n = int(max(2, n))
    out = []
    alphas = list(alphas)
    if len(alphas) == 0 or max_betas <= 0:
        return out
    for a in alphas:
        betas = [0.0, 0.5] if max_betas >= 2 else [0.5]
        if rng is not None:
            betas.append(float(rng.uniform(0.0, 1.0)))
        for beta in betas:
            raw = np.mod(np.arange(n) * float(a) + float(beta), 1.0)
            order = np.argsort(raw)
            hi = np.zeros(n, dtype=float)
            lo = np.zeros(n, dtype=float)
            half = n // 2
            if half > 0:
                hi[order[half:]] = 1.0
                lo[order[:half]] = 1.0
                add_unique_candidate(out, hi, n)
                add_unique_candidate(out, lo, n)
            f = 1.0 - 2.0 * np.abs(raw - 0.5)
            add_unique_candidate(out, 0.5 + 0.5 * f, n)
    return out


def make_candidate_pool(base, n, rng, random_count=3, max_mode=8):
    n = int(n)
    pool = []
    add_unique_candidate(pool, base, n)
    for v in make_parent_variants(base, n):
        add_unique_candidate(pool, v, n)
    if random_count > 0 and rng is not None:
        bases = make_basis(n, max_mode)
        for v in make_fourier_candidates(n, rng, random_count, bases, max_mode):
            add_unique_candidate(pool, v, n)
    alphas = [0.6180339887498949, 0.4142135623730951]
    if rng is not None:
        alphas.append(float(rng.uniform(0.15, 0.85)))
    for v in make_beatty_candidates(n, alphas, rng, max_betas=2):
        add_unique_candidate(pool, v, n)
    if not pool:
        pool.append(project_feasible(np.full(n, 0.5, dtype=float), n))
    return pool


def screen_candidates(cands, n, end, force_first=False, max_keep=2, alphas=None, mu=0.02, maxiters=None):
    n = int(n)
    scored = []
    for i, h in enumerate(cands):
        hp = project_feasible(h, n)
        c = compute_c(hp, n)
        scored.append([c, hp, i])
    if not scored:
        h = project_feasible(np.full(n, 0.5, dtype=float), n)
        return h, compute_c(h, n)
    scored.sort(key=sort_key_score)
    best_h = scored[0][1].copy()
    best_c = float(scored[0][0])
    if alphas is None or max_keep <= 0 or end - time.time() < 25.0:
        return best_h, best_c
    keep_list_indices = []
    for item in scored:
        if len(keep_list_indices) >= max_keep:
            break
        keep_list_indices.append(item[2])
    if force_first and 0 not in keep_list_indices:
        if keep_list_indices:
            keep_list_indices.pop()
        keep_list_indices.append(0)
    keep_set = set(keep_list_indices)
    keep_list = []
    for item in scored:
        if item[2] in keep_set:
            keep_list.append(item[1])
    if not keep_list:
        keep_list = [best_h]
    total = max(1.0, end - time.time())
    per = total / float(len(keep_list))
    for h in keep_list:
        t_start = time.time()
        cand_end = min(end, t_start + per * 1.25)
        hopt, c_opt = optimize_full(h, n, cand_end, alphas=alphas, mu=mu, maxiters=maxiters)
        if c_opt < best_c:
            best_c = float(c_opt)
            best_h = np.asarray(hopt, dtype=float).copy()
        if time.time() >= end - 5.0:
            break
    return best_h, best_c


def gradient_local_search(h, n, end):
    n = int(n)
    x = project_feasible(h, n)
    best_c = compute_c(x, n)
    if n <= 1 or time.time() >= end:
        return x, best_c
    k = 8 if n <= 800 else 5
    max_evals_per_pass = 260 if n <= 800 else 110
    dx = 2.0 / float(n)
    c0 = n - 1
    while time.time() < end - 2.0:
        c = compute_c_vec_fft(x, n)
        cm = float(c.max())
        if cm < best_c:
            best_c = cm
        order = np.argsort(-c)
        act = order[:min(6, len(c))]
        g = np.zeros(n, dtype=float)
        fac = dx / float(max(1, len(act)))
        for idx in act:
            q = int(idx) - c0
            if q >= 0:
                nv = n - q
                g[:nv] += fac * (1.0 - x[q:])
                g[q:] += fac * (-x[:nv])
            else:
                r = -q
                nv = n - r
                g[r:] += fac * (1.0 - x[:nv])
                g[:nv] += fac * (-x[r:])
        low = np.argsort(g)[:k]
        high = np.argsort(g)[-k:][::-1]
        improved = False
        ev = 0
        for i in high:
            if ev >= max_evals_per_pass or time.time() >= end - 2.0:
                break
            for j in low:
                if i == j:
                    continue
                d = min(0.02, float(x[i]), 1.0 - float(x[j]))
                if d <= 1e-10:
                    continue
                for dd in (d, 0.5 * d, 0.25 * d):
                    if dd <= 1e-10:
                        continue
                    x[j] += dd
                    x[i] -= dd
                    cc = compute_c(x, n)
                    ev += 1
                    if cc < best_c - 1e-16:
                        best_c = float(cc)
                        improved = True
                        break
                    x[j] -= dd
                    x[i] += dd
                    if ev >= max_evals_per_pass:
                        break
                if improved or ev >= max_evals_per_pass:
                    break
            if improved or time.time() >= end - 2.0:
                break
        if improved:
            x = project_feasible(x, n)
        else:
            break
    x = make_exact_sum(x, n)
    best_c = compute_c(x, n)
    return x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    main_end = t0 + max(60.0, budget_s - 8.0)

    try:
        h_input = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h_input = np.full(512, 0.5, dtype=float)

    if h_input.size == 0:
        h_input = np.full(512, 0.5, dtype=float)
    n_input = int(len(h_input))
    if n_input % 2 != 0:
        n_initial = n_input + 1
        h_initial = resample_h(h_input, n_input, n_initial)
    else:
        n_initial = n_input
        h_initial = project_feasible(h_input, n_initial)

    best_h = np.asarray(h_initial, dtype=float).copy()
    best_n = int(n_initial)
    best_c = compute_c(best_h, best_n)
    rng = np.random.RandomState(seed)

    total = max(60.0, main_end - t0)
    end_parent = t0 + total * 0.24
    end_explore = end_parent + total * 0.10
    end_hi1 = end_explore + total * 0.26
    end_hi2 = end_hi1 + total * 0.14

    if time.time() < end_parent - 8.0:
        hopt, c_opt = optimize_full(
            best_h, best_n, end_parent,
            alphas=(50, 200, 1000, 5000, 20000, 100000, 500000, 2000000),
            mu=0.02
        )
        if c_opt < best_c:
            best_c = float(c_opt)
            best_n = int(len(hopt))
            best_h = np.asarray(hopt, dtype=float).copy()

    if time.time() < end_explore - 8.0:
        pool = make_candidate_pool(best_h, best_n, rng, random_count=3, max_mode=8)
        hopt, c_opt = screen_candidates(
            pool, best_n, end_explore,
            force_first=True, max_keep=2,
            alphas=(100, 1000, 10000, 100000, 1000000),
            mu=0.02
        )
        if c_opt < best_c:
            best_c = float(c_opt)
            best_n = int(len(hopt))
            best_h = np.asarray(hopt, dtype=float).copy()

    if time.time() < end_hi1 - 10.0:
        if budget_s < 500:
            n_hi1 = 512
        elif budget_s < 800:
            n_hi1 = 768
        else:
            n_hi1 = 1024
        if n_hi1 <= best_n and best_n < 1024:
            n_hi1 = best_n + 256
        if n_hi1 % 2 != 0:
            n_hi1 += 1
        n_hi1 = int(min(n_hi1, 1280 if budget_s < 900 else 1536))
        if n_hi1 > best_n:
            h_start = resample_h(best_h, best_n, n_hi1)
            c_start = compute_c(h_start, n_hi1)
            if c_start <= best_c + 0.025:
                pool = make_candidate_pool(h_start, n_hi1, rng, random_count=2, max_mode=8)
                h_nearest = resample_nearest(best_h, best_n, n_hi1)
                add_unique_candidate(pool, h_nearest, n_hi1)
                hopt, c_opt = screen_candidates(
                    pool, n_hi1, end_hi1,
                    force_first=True, max_keep=2,
                    alphas=(20, 200, 2000, 20000, 200000, 1000000),
                    mu=0.015
                )
                if c_opt < best_c:
                    best_c = float(c_opt)
                    best_n = int(len(hopt))
                    best_h = np.asarray(hopt, dtype=float).copy()

    if time.time() < end_hi2 - 10.0 and budget_s >= 800:
        if budget_s >= 1000:
            n_hi2 = 1536
        else:
            n_hi2 = 1280
        if n_hi2 <= best_n and best_n < 1536:
            n_hi2 = best_n + 384
        if n_hi2 % 2 != 0:
            n_hi2 += 1
        if n_hi2 > best_n:
            h_start = resample_h(best_h, best_n, n_hi2)
            c_start = compute_c(h_start, n_hi2)
            if c_start <= best_c + 0.035:
                pool = make_candidate_pool(h_start, n_hi2, rng, random_count=1, max_mode=6)
                h_nearest = resample_nearest(best_h, best_n, n_hi2)
                add_unique_candidate(pool, h_nearest, n_hi2)
                hopt, c_opt = screen_candidates(
                    pool, n_hi2, end_hi2,
                    force_first=True, max_keep=1,
                    alphas=(50, 500, 5000, 50000, 500000),
                    mu=0.01
                )
                if c_opt < best_c:
                    best_c = float(c_opt)
                    best_n = int(len(hopt))
                    best_h = np.asarray(hopt, dtype=float).copy()

    if time.time() < main_end - 45.0:
        hfin = np.asarray(best_h, dtype=float).copy()
        hopt, c_opt = run_active_lbfgs(
            hfin, best_n, main_end - 25.0,
            alphas=(100000, 500000, 2000000, 10000000),
            top_k=4, rounds=4, mu=0.002
        )
        if c_opt < best_c:
            best_c = float(c_opt)
            best_n = int(len(hopt))
            best_h = np.asarray(hopt, dtype=float).copy()

    if time.time() < main_end - 4.0:
        hfin = np.asarray(best_h, dtype=float).copy()
        hopt, c_opt = gradient_local_search(hfin, best_n, main_end - 2.0)
        if c_opt < best_c:
            best_c = float(c_opt)
            best_n = int(len(hopt))
            best_h = np.asarray(hopt, dtype=float).copy()

    best_h = make_exact_sum(best_h, best_n)
    best_c = compute_c(best_h, best_n)
    best_n = int(len(best_h))
    return best_h, float(best_c), best_n
