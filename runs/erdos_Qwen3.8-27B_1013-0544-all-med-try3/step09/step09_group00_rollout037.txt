from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def make_beatty(n, alpha):
    h = np.zeros(n)
    seen = set()
    count = 0
    k = 1
    while count < n // 2 and k < 1000 * n:
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
            count += 1
        k += 1
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def sa_binary(h_init, n, deadline, rng, t0=0.003, t1=0.00005):
    """Simulated annealing for binary optimization."""
    best = h_init.copy()
    best_c = compute_c5(best, n)
    cur = best.copy()
    cur_c = best_c
    
    ones = np.where(cur == 1)[0].tolist()
    zeros = np.where(cur == 0)[0].tolist()
    
    step = 0
    max_steps = 5000000
    
    while time.time() < deadline and step < max_steps:
        # Temperature schedule: exponential cooling
        frac = step / max_steps
        t = t0 * (t1 / t0) ** frac
        
        # Random swap
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur, n)
        
        delta = new_c - cur_c
        if delta < 0 or (t > 1e-10 and rng.rand() < np.exp(-delta / (t + 1e-15))):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
            # Update lists
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            cur[i] = 1
            cur[j] = 0
        
        step += 1
    
    return best, best_c

def local_search_binary(h_init, n, deadline, rng):
    """Hill climbing: only accept improvements."""
    best = h_init.copy()
    best_c = compute_c5(best, n)
    cur = best.copy()
    cur_c = best_c
    
    ones = np.where(cur == 1)[0].tolist()
    zeros = np.where(cur == 0)[0].tolist()
    
    no_improve_count = 0
    max_no_improve = 5000
    
    while time.time() < deadline and no_improve_count < max_no_improve:
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur, n)
        
        if new_c < cur_c - 1e-15:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
            ones[i_idx] = j
            zeros[j_idx] = i
            no_improve_count = 0
        else:
            cur[i] = 1
            cur[j] = 0
            no_improve_count += 1
    
    return best, best_c

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = len(c)
    for mm in range(nn):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def continuous_optimize(h_init, n, deadline):
    """L-BFGS-B continuous optimization."""
    best = project_feasible(h_init, n)
    best_c = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            def obj(x):
                return smooth_max_obj(x, n, alpha)
            def grad(x):
                c = compute_c5_vec(x, n)
                return smooth_max_grad(c, alpha, x, n)
            res = minimize(obj, best, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit*10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best = h_new
        except Exception:
            pass
    return best, best_c

def grad_descent_polish(h_init, n, deadline):
    """Gradient descent with pair swaps for final polish."""
    best = h_init.copy()
    best_c = compute_c5(best, n)
    c_vec = compute_c5_vec(best, n)
    
    dx_val = 2.0 / n
    step_sizes = [0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005, 0.00001]
    
    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best, n)
            max_val = c_vec.max()
            # Find active lags
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            # Compute gradient
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best[k:n]) * dx_val
                    grad[k:n] += (-best[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best[0:nv]) * dx_val
                    grad[0:nv] += (-best[kk:n]) * dx_val
            grad /= len(active)
            
            # Find best pair swap
            # We want to decrease h where grad > 0 and increase where grad < 0
            pos = np.where((grad > 1e-8) & (best > 1e-8))[0]
            neg = np.where((grad < -1e-8) & (best < 1.0 - 1e-8))[0]
            
            if len(pos) == 0 or len(neg) == 0:
                break
            
            best_gain = -1
            best_p = -1
            best_q = -1
            for p in pos[:50]:
                for q in neg[:50]:
                    gain = grad[p] - grad[q]
                    if gain > best_gain:
                        best_gain = gain
                        best_p = p
                        best_q = q
            
            if best_p < 0:
                break
            
            amt = min(step, best[best_p], 1.0 - best[best_q])
            if amt < 1e-12:
                break
            
            best[best_p] -= amt
            best[best_q] += amt
            best = project_feasible(best, n)
            c_new = compute_c5(best, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
            else:
                best[best_p] += amt
                best[best_q] -= amt
                best = project_feasible(best, n)
    
    return best, best_c

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def make_beatty(n, alpha):
    h = np.zeros(n)
    seen = set()
    count = 0
    k = 1
    while count < n // 2 and k < 10000 * n:
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
            count += 1
        k += 1
    return h

def make_quadratic_residue(n, rng):
    p = n
    while True:
        is_prime = all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)) if p > 1 else False
        if is_prime:
            break
        p += 1
    residues = sorted(set((i * i) % p for i in range(1, p)))
    h = np.zeros(n)
    count = 0
    for r in residues:
        if count >= n // 2:
            break
        if r < n:
            h[r] = 1.0
            count += 1
    if count < n // 2:
        remaining = n // 2 - count
        avail = np.where(h == 0)[0]
        idx = rng.choice(avail, remaining, replace=False)
        h[idx] = 1.0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def sa_binary(h_init, n, deadline, rng, t_start=0.004, t_min=1e-5, cooling=0.9999, kick_size=8, max_kicks=40):
    best = h_init.copy().astype(float)
    best_c = compute_c5(best, n)
    cur = best.copy()
    cur_c = best_c

    ones_list = np.where(cur > 0.5)[0].tolist()
    zeros_list = np.where(cur < 0.5)[0].tolist()

    t = t_start
    no_improve = 0
    kicks = 0
    step_count = 0

    while time.time() < deadline:
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        cur[i] = 0.0
        cur[j] = 1.0
        new_c = compute_c5(cur, n)

        delta = new_c - cur_c
        accept = delta < 0 or (t > t_min and rng.rand() < np.exp(-delta / t))

        if accept:
            cur_c = new_c
            if new_c < best_c - 1e-15:
                best_c = new_c
                best = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            ii = ones_list.index(i)
            jj = zeros_list.index(j)
            ones_list[ii] = j
            zeros_list[jj] = i
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1

        t = max(t_min, t * cooling)
        step_count += 1

        if no_improve > 50000 and kicks < max_kicks:
            for _ in range(kick_size):
                ii = rng.randint(len(ones_list))
                jj = rng.randint(len(zeros_list))
                a = ones_list[ii]
                b = zeros_list[jj]
                cur[a] = 0.0
                cur[b] = 1.0
                ones_list[ii] = b
                zeros_list[jj] = a
            cur_c = compute_c5(cur, n)
            if cur_c < best_c:
                best_c = cur_c
                best = cur.copy()
            t = t_start * 0.4
            no_improve = 0
            kicks += 1

    return best, best_c

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = len(c)
    for mm in range(nn):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def continuous_optimize(h_init, n, deadline):
    best = project_feasible(h_init.astype(float), n)
    best_c = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n

    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        maxit = max(100, int((deadline - time.time()) * 4))
        try:
            def obj(x):
                c = compute_c5_vec(x, n)
                return smooth_max_val(c, alpha)
            def grad(x):
                c = compute_c5_vec(x, n)
                return smooth_max_grad(c, alpha, x, n)
            res = minimize(obj, best, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best = h_new
        except Exception:
            pass
    return best, best_c

def grad_pair_polish(h_init, n, deadline):
    best = h_init.copy().astype(float)
    best_c = compute_c5(best, n)
    dx_val = 2.0 / n
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]

            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best[k:n]) * dx_val
                    grad[k:n] += (-best[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best[0:nv]) * dx_val
                    grad[0:nv] += (-best[kk:n]) * dx_val
            grad /= len(active)

            pos = np.where((grad > 1e-9) & (best > 1e-9))[0]
            neg = np.where((grad < -1e-9) & (best < 1.0 - 1e-9))[0]

            if len(pos) == 0 or len(neg) == 0:
                break

            best_gain = -1
            best_p = -1
            best_q = -1
            for p in pos[:60]:
                for q in neg[:60]:
                    gain = grad[p] - grad[q]
                    if gain > best_gain:
                        best_gain = gain
                        best_p = p
                        best_q = q

            if best_p < 0 or best_gain <= 0:
                break

            amt = min(step, best[best_p], 1.0 - best[best_q])
            if amt < 1e-12:
                break

            best[best_p] -= amt
            best[best_q] += amt
            best = project_feasible(best, n)
            c_new = compute_c5(best, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
            else:
                best[best_p] += amt
                best[best_q] -= amt
                best = project_feasible(best, n)

    return best, best_c
