    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    return w

def grad_c5_vec(h, n, w):
    """Compute gradient of weighted sum of correlation values w^T c_vec wrt h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_obj_and_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        w = np.exp(sh - mx)
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
        w = np.exp(sh)
    w /= w.sum()
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    grad = grad_c5_vec(h, n, w)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return float(val), grad

def lbfgs_adaptive(h, n, deadline, alpha_list=None, mu=100.0, max_iter_total=8000):
    if alpha_list is None:
        alpha_list = [200, 500, 1500, 4000, 10000, 25000]
    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    iters_used = 0
    for alpha in alpha_list:
        if time.time() > deadline - 10:
            break
        rem_iter = max_iter_total - iters_used
        if rem_iter <= 0:
            break
        this_iter = min(rem_iter, max_iter_total // len(alpha_list))
        
        def obj_fn(x):
            v, _ = smooth_obj_and_grad(x, n, alpha, mu)
            return v
        def jac_fn(x):
            _, g = smooth_obj_and_grad(x, n, alpha, mu)
            return g
        
        t_start = time.time()
        try:
            res = minimize(obj_fn, h, jac=jac_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': this_iter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': this_iter * 12,
                                   'maxls': 40, 'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
            iters_used += res.nit if res.nit else this_iter
        except Exception:
            h_new = project_feasible(h, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
    return h, best_c

def sqp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        k = idx - (n - 1)
        c_val = c_vec[idx]
        if k >= 0:
            nv = n - k
            A_ub[row, 0:nv] += (1.0 - h[k:n]) * dx_tmp
            A_ub[row, k:n] += (-h[0:nv]) * dx_tmp
        else:
            kk = -k
            nv = n - kk
            A_ub[row, kk:n] += (1.0 - h[0:nv]) * dx_tmp
            A_ub[row, 0:nv] += (-h[kk:n]) * dx_tmp
        A_ub[row, n] = -1.0
        b_ub[row] = -c_val
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

dx_tmp = 0.0

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.03):
    global dx_tmp
    dx_tmp = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 5e-4 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 30:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:30]]
        delta, t_opt = sqp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.4
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            if rho < 0.06:
                rho = min(rho * 1.2, 0.08)
        else:
            stall += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if stall > 30:
                break
    return best_h, best_c

def coord_descent(h, n, deadline, max_rounds=50):
    best_c = compute_c5(h, n)
    target = n / 2.0
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline - 5:
            break
        for rnd in range(max_rounds):
            if time.time() > deadline - 5:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 1e-3 * max_val)
            active = np.where(c_vec >= thresh)[0]
            # Compute aggregated gradient for prioritization
            w = np.zeros(2 * n - 1)
            for a in active:
                w[a] = c_vec[a] / max_val
            grad = grad_c5_vec(h, n, w)
            # Find top candidates
            n_cand = min(200, n)
            top_idx = np.argsort(-np.abs(grad))[:n_cand]
            improved = False
            for i in top_idx:
                if time.time() > deadline - 5:
                    break
                i = int(i)
                # Try adjusting h[i] and compensating elsewhere
                if h[i] + step <= 1.0 + 1e-12:
                    # Find j to decrease
                    candidates = np.where(h > step + 1e-9)[0]
                    if len(candidates) > 0:
                        # Pick j closest to i for locality
                        dists = np.abs(candidates - i)
                        order = np.argsort(dists)[:6]
                        for j_idx in order:
                            j = int(candidates[j_idx])
                            if i == j:
                                continue
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= step
                                h[j] += step
                if h[i] - step >= -1e-12:
                    candidates = np.where(h < 1.0 - step - 1e-9)[0]
                    if len(candidates) > 0:
                        dists = np.abs(candidates - i)
                        order = np.argsort(dists)[:6]
                        for j_idx in order:
                            j = int(candidates[j_idx])
                            if i == j:
                                continue
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += step
                                h[j] -= step
        if not improved:
            continue
    return h, best_c

def random_perturb(h, n, sigma, deadline):
    """Add small random perturbation maintaining constraints."""
    rng = np.random.RandomState(12345)
    h_pert = h.copy()
    h_pert += rng.randn(n) * sigma
    h_pert = project_feasible(h_pert, n)
    return h_pert

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Optimize at original resolution (n_init ~ 500)
    # L-BFGS with adaptive temperature
    t1 = min(t0 + 60, deadline - 700)
    if t1 > time.time() + 5:
        h1, c1 = lbfgs_adaptive(best_h, best_n, t1,
            alpha_list=[300, 1000, 3000, 8000], mu=200.0)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # SQP at original resolution
    t2 = min(t0 + 120, deadline - 600)
    if t2 > time.time() + 5:
        h2, c2 = sqp_optimize(best_h, best_n, t2, max_iter=300, rho_init=0.03)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 2: Upsample to n=1000 and optimize
    n2 = 1000
    h_med = upsample(best_h, best_n, n2)
    c_med = compute_c5(h_med, n2)
    if c_med < best_c:
        best_c = c_med
        best_h = h_med
        best_n = n2

    t3 = min(t0 + 120 + 80, deadline - 400)
    if t3 > time.time() + 5:
        h3, c3 = lbfgs_adaptive(h_med, n2, t3,
            alpha_list=[500, 2000, 6000, 15000], mu=500.0)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n2

    # SQP at medium resolution
    t4 = min(t3 + 80, deadline - 300)
    if t4 > time.time() > 5:
        h4, c4 = sqp_optimize(best_h, best_n, t4, max_iter=250, rho_init=0.025)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Coordinate descent at medium resolution
    t5 = min(t4 + 60, deadline - 200)
    if t5 > time.time() + 5:
        h5, c5 = coord_descent(best_h, best_n, t5, max_rounds=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 3: Try n=1500
    n3 = 1500
    h_hi = upsample(best_h, best_n, n3)
    c_hi = compute_c5(h_hi, n3)
    
    t6 = min(t5 + 70, deadline - 150)
    if t6 > time.time() + 10:
        h6, c6 = lbfgs_adaptive(h_hi, n3, t6,
            alpha_list=[500, 2000, 8000], mu=1000.0)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n3
        elif c6 < c_hi:
            c_hi = c6
            h_hi = h6

    # SQP at high resolution
    t7 = min(t6 + 50, deadline - 80)
    if t7 > time.time() + 5:
        h7, c7 = sqp_optimize(best_h, best_n, t7, max_iter=200, rho_init=0.02)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Coordinate descent at high res
    t8 = min(t7 + 50, deadline - 30)
    if t8 > time.time() + 5:
        h8, c8 = coord_descent(best_h, best_n, t8, max_rounds=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 4: Final polish - adaptive L-BFGS with very high alpha
    if time.time() < deadline - 15:
        h9, c9 = lbfgs_adaptive(best_h, best_n, deadline - 3,
            alpha_list=[20000, 50000, 100000], mu=5000.0)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final feasibility projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
