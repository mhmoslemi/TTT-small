import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 25:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:25]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            rho *= 0.7
            no_improve += 1
            if no_improve > 15:
                break
            continue
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 25:
                break
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, alphas, maxiter=15000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-22,
                                   'gtol': 1e-20, 'maxfun': 300000, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def directed_local_search(h, n, deadline, alpha=50000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            grad = smooth_grad(best_h, n, alpha)
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            active_grad = np.zeros(n)
            for idx in active:
                active_grad += np.abs(shift_grad_single(best_h, n, idx))
            active_grad /= len(active)
            top_count = min(30, n // 10)
            high_idx = np.argsort(-active_grad)[:top_count]
            low_idx = np.argsort(active_grad)[:top_count]
            for i_val in high_idx:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                for j_val in low_idx:
                    if time.time() > deadline:
                        return best_h, best_c
                    j = int(j_val)
                    if i == j:
                        continue
                    for direction in (1, -1):
                        hi_new = best_h[i] + direction * step
                        hj_new = best_h[j] - direction * step
                        if hi_new < -1e-12 or hi_new > 1.0 + 1e-12:
                            continue
                        if hj_new < -1e-12 or hj_new > 1.0 + 1e-12:
                            continue
                        best_h[i] = hi_new
                        best_h[j] = hj_new
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            best_h[i] -= direction * step
                            best_h[j] += direction * step
    return best_h, best_c

def gradient_step_polish(h, n, deadline, alpha=100000, max_iter=5000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_size = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_grad(best_h, n, alpha)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-14:
            break
        grad = -grad / norm_g
        grad -= grad.mean()
        trial = best_h + step_size * grad
        trial = project_feasible(trial, n)
        c_trial = compute_c5(trial, n)
        if c_trial < best_c:
            best_c = c_trial
            best_h = trial
            step_size = min(step_size * 1.2, 0.05)
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=5000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    t_end = min(deadline, time.time() + 60)
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'eps': 1e-12})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

    t0 = time.time()

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS at original resolution (generous time)
    t1 = min(t0 + 150, deadline - 700)
    if t1 > time.time() + 10:
        best_h, best_c = lbfgs_optimize(best_h, n_orig, t1,
            alphas=[200, 1000, 5000, 20000, 80000], maxiter=15000)

    # Phase 2: SQP refinement
    t2 = min(t0 + 280, deadline - 550)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=500, rho_init=0.05)

    # Phase 3: Gradient step polish
    t3 = min(t0 + 340, deadline - 500)
    if t3 > time.time() + 5:
        best_h, best_c = gradient_step_polish(best_h, n_orig, t3, alpha=50000, max_iter=8000)

    # Phase 4: Directed local search
    t4 = min(t0 + 440, deadline - 400)
    if t4 > time.time() + 5:
        best_h, best_c = directed_local_search(best_h, n_orig, t4, alpha=50000)

    # Phase 5: Another SQP round
    t5 = min(t0 + 520, deadline - 350)
    if t5 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t5, max_iter=300, rho_init=0.02)

    # Phase 6: SLSQP polish
    t6 = min(t0 + 600, deadline - 250)
    if t6 > time.time() + 10:
        h6, c6 = slqp_polish(best_h, n_orig, t6, alpha=5000000, maxiter=8000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Upsample to higher resolution
    n_high = min(n_orig * 2, 1000)
    if n_high > n_orig and time.time() < deadline - 280:
        x_old = np.linspace(0, 2, n_orig, endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        hh = h_high.copy()

        t7 = min(time.time() + 120, deadline - 150)
        if t7 > time.time() + 10:
            hh, c_high = lbfgs_optimize(hh, n_high, t7,
                alphas=[1000, 5000, 20000], maxiter=10000)

        t8 = min(time.time() + 80, deadline - 70)
        if t8 > time.time() + 10:
            hh, c_high = sqp_optimize(hh, n_high, t8, max_iter=200, rho_init=0.03)

        t9 = min(time.time() + 50, deadline - 30)
        if t9 > time.time() + 5:
            hh, c_high = gradient_step_polish(hh, n_high, t9, alpha=50000, maxiter=5000)

        if c_high < best_c:
            best_c = c_high
            best_h = hh

    # Phase 8: Final polish on best
    if time.time() < deadline - 10:
        hf, cf = slqp_polish(best_h, len(best_h), deadline - 5, alpha=10000000, maxiter=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
