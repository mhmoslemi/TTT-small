        diff = target - h.sum()
        if abs(diff) < 1e-14:

            if tot < 1e-14:

            if tot < 1e-14:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()

        else:

            if tot < 1e-14:

        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def lse_smooth(x, n, alpha, mu=0.0):

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for mm in range(len(c)):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

                                   'gtol': 1e-16, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)

            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass

        np.cos(np.linspace(0, 6*np.pi, n_parent)) * 0.03,
    ]
    
    for p in perturbations:
        if time.time() > deadline - 60:
            break
        h_pert = project_feasible(h_parent + p, n_parent)
        c_pert = compute_c5(h_pert, n_parent)
        
        if c_pert < best_c + 0.01:
            t_sub = min(time.time() + 50, deadline - 5)
            h_opt, c_opt = run_lbfgs(h_pert, n_parent, t_sub,
                alphas=[5000, 50000, 500000], maxiter=3000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at original resolution
    if time.time() < deadline - 700:
        t1 = min(time.time() + 200, deadline - 700)
        best_h, best_c = run_lbfgs(best_h, best_n, t1,
            alphas=[200, 1000, 5000, 20000, 100000, 500000, 2000000],
            maxiter=4000, mu=0.01)
    
    # Phase 2: Multi-start at original resolution
    if time.time() < deadline - 500:
        t2 = min(time.time() + 100, deadline - 500)
        h_ms, c_ms = multi_start_optimize(best_h, best_n, t2)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms
    
    # Phase 3: Upsample to 2x and optimize
    n_up = best_n * 2
    if n_up <= 2000 and time.time() < deadline - 400:
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        c_up = compute_c5(h_up, n_up)
        
        if c_up < best_c or (deadline - time.time() > 300):
            t3 = min(time.time() + 250, deadline - 300)
            h_up, c_up = run_lbfgs(h_up, n_up, t3,
                alphas=[500, 2000, 10000, 50000, 200000, 1000000, 5000000],
                maxiter=3000, mu=0.01)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n_up
    
    # Phase 4: SLSQP polish on best
    if time.time() < deadline - 250:
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 230:
                break
            t4 = min(time.time() + 60, deadline - 230)
            h4, c4 = run_slqp(best_h, best_n, t4, alpha=alpha, maxiter=3000)
            if c4 < best_c:
                best_c = c4
                best_h = h4
    
    # Phase 5: Active set refinement
    if time.time() < deadline - 150:
        t5 = min(time.time() + 80, deadline - 150)
        best_h, best_c = active_set_refine(best_h, best_n, t5)
    
    # Phase 6: Projected gradient descent
    if time.time() < deadline - 100:
        t6 = min(time.time() + 60, deadline - 100)
        h6, c6 = proj_grad_descent(best_h, best_n, 10000000, t6,
                                    init_lr=0.005, momentum=0.8, decay=0.9998)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Another SLSQP round
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        h7, c7 = run_slqp(best_h, best_n, t7, alpha=20000000, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Local 1D search
    if time.time() < deadline - 20:
        t8 = min(time.time() + 25, deadline - 20)
        best_h, best_c = local_1d_search(best_h, best_n, t8)
    
    # Phase 9: Final SLSQP
    if time.time() < deadline - 3:
        h9, c9 = run_slqp(best_h, best_n, deadline - 1, alpha=50000000, maxiter=1000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
