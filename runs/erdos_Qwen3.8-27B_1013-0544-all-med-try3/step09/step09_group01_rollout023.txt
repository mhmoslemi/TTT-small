        for beta in betas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 3))
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, beta, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    active = np.where(compute_c5_vec(best_h, n) >= best_c - 0.0002 * best_c)[0]
                    if len(active) > top_k:
                        active = np.array(np.argsort(-compute_c5_vec(best_h, n)[active])[:top_k])
            except Exception:
                pass
    return best_h, best_c

def perturbed_reoptimize(h, n, deadline, rng, n_perturb=5, noise_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_perturb):
        if time.time() > deadline - 100:
            break
        h_pert = best_h + rng.normal(0, noise_scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        p_end = min(time.time() + 80, deadline - 50)
        h_opt, c_opt = lbfgs_optimize(h_pert, n, p_end,
            alphas=[1000, 10000, 100000], mu=0.01, maxit_per=[1000, 2000, 3000])
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def binary_start_optimize(h_ref, n, deadline, rng, n_starts=8):
    best_h = h_ref.copy()
    best_c = compute_c5(h_ref, n)
    n_ones = n // 2
    for s in range(n_starts):
        if time.time() > deadline - 150:
            break
        h_bin = np.zeros(n)
        idx = rng.permutation(n)
        h_bin[idx[:n_ones]] = 1.0
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin >= best_c:
            continue
        p_end = min(time.time() + 60, deadline - 80)
        h_opt, c_opt = lbfgs_optimize(h_bin, n, p_end,
            alphas=[50, 500, 5000, 50000], mu=0.01, maxit_per=[1000, 1500, 2000, 2000])
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def pair_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in step_sizes:
        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            order = np.random.permutation(n)
            for i in order:
                if time.time() > deadline:
                    break
                lo = max(0, i - 60)
                hi = min(n, i + 61)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            if not improved:
                break
    return best_h, best_c

def subgrad_descent(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.008
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.03)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Upscale to n=1000 and optimize
    n_hi = 1000
    x_fine = np.linspace(0, 2, n_hi)
    x_coarse = np.linspace(0, 2, n_orig)
    h_fine = np.interp(x_fine, x_coarse, h0)
    h_fine = project_feasible(h_fine, n_hi)
    c_fine = compute_c5(h_fine, n_hi)

    p1_end = min(t0 + 320, deadline - 200)
    h1, c1 = lbfgs_optimize(h_fine, n_hi, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
        mu=0.005, maxit_per=[1500, 2000, 2500, 3000, 3000, 3500, 3500, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = n_hi

    # Also optimize at original n
    p1b_end = min(time.time() + 80, deadline - 180)
    h1b, c1b = lbfgs_optimize(h0, n_orig, p1b_end,
        alphas=[500, 5000, 50000, 500000],
        mu=0.01, maxit_per=[2000, 3000, 3000, 3000])
    if c1b < best_c:
        best_c = c1b
        best_h = h1b
        best_n = n_orig

    # Phase 2: Active-shift focused optimization
    if time.time() < deadline - 250:
        p2_end = min(time.time() + 150, deadline - 220)
        h2, c2 = active_shift_optimize(best_h, best_n, p2_end, top_k=8,
            betas=[10000, 50000, 200000, 1000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = best_n

    # Phase 3: Perturbation + re-optimization (escape local optima)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 150, deadline - 160)
        h3, c3 = perturbed_reoptimize(best_h, best_n, p3_end, rng,
            n_perturb=4, noise_scale=0.015)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n

    # Phase 4: Active shifts again after perturbation
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 80, deadline - 120)
        h4, c4 = active_shift_optimize(best_h, best_n, p4_end, top_k=6,
            betas=[50000, 200000, 1000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n

    # Phase 5: Try n=1500 (if current best is at n=1000 or less)
    n_hi2 = 1500
    if time.time() < deadline - 180:
        x_f2 = np.linspace(0, 2, n_hi2)
        x_c2 = np.linspace(0, 2, best_n)
        h_f2 = np.interp(x_f2, x_c2, best_h)
        h_f2 = project_feasible(h_f2, n_hi2)
        p5_end = min(time.time() + 120, deadline - 100)
        h5, c5 = lbfgs_optimize(h_f2, n_hi2, p5_end,
            alphas=[20, 200, 2000, 20000, 200000],
            mu=0.003, maxit_per=[1000, 1500, 2000, 2000, 2500])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_hi2

    # Phase 6: Binary restart optimization
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 100, deadline - 100)
        h6, c6 = binary_start_optimize(best_h, best_n, p6_end, rng, n_starts=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # Phase 7: Subgradient refinement
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 50, deadline - 40)
        h7, c7 = subgrad_descent(best_h, best_n, p7_end, n_iter=8000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n

    # Phase 8: Pair descent polishing
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 40, deadline - 5)
        h8, c8 = pair_descent(best_h, best_n, p8_end, max_rounds=100)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n

    # Phase 9: Final active shifts
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        active = np.where(compute_c5_vec(best_h, best_n) >= best_c - 0.0001 * best_c)[0]
        if len(active) > 4:
            active = np.array(np.argsort(-compute_c5_vec(best_h, best_n)[active])[:4])
        h9 = best_h.copy()
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(topk_obj, h9, jac=topk_grad,
                          args=(best_n, active, 5000000, 0.001), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14})
            h9 = project_feasible(res.x, best_n)
            c9 = compute_c5(h9, best_n)
            if c9 < best_c:
                best_c = c9
                best_h = h9
        except Exception:
            pass

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
