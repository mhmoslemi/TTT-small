                    best_h[i] -= step
                    best_h[j] += step
                    current_step_idx = min(len(step_sizes) - 1, current_step_idx + 1)
            elif best_h[i] - step >= 0.0 and best_h[j] + step <= 1.0:
                best_h[i] -= step
                best_h[j] += step
                c = compute_c5(best_h, n)
                if c < best_c - 1e-17:
                    best_c = c
                    current_step_idx = max(0, current_step_idx - 1)
                else:
                    best_h[i] += step
                    best_h[j] -= step
                    current_step_idx = min(len(step_sizes) - 1, current_step_idx + 1)
    
    return best_h, best_c

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def slqp_run(h, n, deadline, alpha, maxiter=1000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        if smx > 500:
            return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def jac(x):
        return lse_grad(x, n, alpha, 0.0)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        maxit = min(maxiter, max(200, int((deadline - time.time()) * 3)))
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 180
    h1, c1 = lbfgs_run(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, maxit_list=[1500, 2000, 2000, 2500, 2500, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active-lag L-BFGS at original resolution
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 120, deadline - 500)
        h2, c2 = active_lbfgs_run(best_h, best_n, p2_end, top_k=8,
            alphas=[5000, 20000, 100000, 500000, 2000000], mu=0.005)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n_fine = 1000
    if time.time() < deadline - 400:
        h_fine = upsample(best_h, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p3a_end = min(time.time() + 200, deadline - 250)
        h3a, c3a = lbfgs_run(h_fine, n_fine, p3a_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
            mu=0.005, maxit_list=[1000, 1500, 1500, 2000, 2000, 2000, 2500])
        
        if c3a < c_fine:
            c_fine = c3a
            h_fine = h3a
        
        if time.time() < deadline - 200:
            p3b_end = min(time.time() + 100, deadline - 200)
            h3b, c3b = active_lbfgs_run(h_fine, n_fine, p3b_end, top_k=10,
                alphas=[10000, 50000, 200000, 1000000], mu=0.003)
            if c3b < c_fine:
                c_fine = c3b
                h_fine = h3b
        
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
            best_n = n_fine
    
    # Phase 4: Gradient local search at best resolution
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 100, deadline - 150)
        h4, c4 = gradient_local_search(best_h, best_n, p4_end, max_rounds=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Pair swap local search (most important for fine improvement)
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 120, deadline - 100)
        h5, c5 = pair_swap_search(best_h, best_n, p5_end, max_rounds=50)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Try upsampling to n=1500 if time permits
    n_fine2 = 1500
    if time.time() < deadline - 200:
        h_fine2 = upsample(best_h, n_fine2)
        c_fine2 = compute_c5(h_fine2, n_fine2)
        
        p6a_end = min(time.time() + 120, deadline - 100)
        h6a, c6a = lbfgs_run(h_fine2, n_fine2, p6a_end,
            alphas=[20, 100, 500, 2000, 10000, 50000],
            mu=0.003, maxit_list=[800, 1000, 1000, 1200, 1200, 1500])
        
        if c6a < c_fine2:
            c_fine2 = c6a
            h_fine2 = h6a
        
        if time.time() < deadline - 60:
            p6b_end = min(time.time() + 60, deadline - 50)
            h6b, c6b = gradient_local_search(h_fine2, n_fine2, p6b_end, max_rounds=200)
            if c6b < c_fine2:
                c_fine2 = c6b
                h_fine2 = h6b
        
        if c_fine2 < best_c:
            best_c = c_fine2
            best_h = h_fine2
            best_n = n_fine2
    
    # Phase 7: More pair swaps at best resolution
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 50, deadline - 30)
        h7, c7 = pair_swap_search(best_h, best_n, p7_end, max_rounds=30)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: SLSQP polish
    if time.time() < deadline - 30:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 25:
                break
            p8_end = min(time.time() + 20, deadline - 20)
            h8, c8 = slqp_run(best_h, best_n, p8_end, alpha=alpha, maxiter=800)
            if c8 < best_c:
                best_c = c8
                best_h = h8
    
    # Phase 9: Final gradient local search
    if time.time() < deadline - 15:
        p9_end = deadline - 8
        h9, c9 = gradient_local_search(best_h, best_n, p9_end, max_rounds=200)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final pair swaps
    if time.time() < deadline - 3:
        p10_end = deadline - 1
        h10, c10 = pair_swap_search(best_h, best_n, p10_end, max_rounds=10)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
