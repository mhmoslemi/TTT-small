```python
def build_basis(n):
    basis = np.eye(n)
    # Fourier modes
    for k in range(1, min(6, n//4+1)):
        basis = np.vstack([basis, np.sin(2*np.pi*k*np.arange(n)/n),
                            np.cos(2*np.pi*k*np.arange(n)/n)])
    # Sawtooth
    basis = np.vstack([basis, np.r_[np.min(basis)**2, np.arange(1, n)+np.linalg.norm(basis)**2]])
    return basis.T
```
