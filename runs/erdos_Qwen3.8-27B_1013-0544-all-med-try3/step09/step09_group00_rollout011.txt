def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

            w_new = active_scores / active_scores.sum()
            w = 0.3 * w + 0.7 * w_new

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 2 * n + 1):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10 and len(positions) < n:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            step_scale *= 0.95
        c = compute_c5_vec(best_h, n)

        active_scores = np.maximum(0.0, c - (c_max - margin))
        if active_scores.sum() > 1e-15:
            w_new = active_scores / active_scores.sum()
            w = 0.2 * w + 0.8 * w_new
        grad = np.zeros(n)

            if w[mm] < 1e-12:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx

                grad[k:n] += wm * (-best_h[0:nv])

                grad[0:nv] += wm * (-best_h[kk:n])
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)

            grad /= float(len(active))

        p2_end = min(time.time() + 130, deadline - 300)
        h2, c2 = lbfgs_phase(best_h, n, p2_end,
            alphas=[50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Multi-top active set
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 130, deadline - 200)
        h3, c3 = multi_top_optimize(best_h, n, p3_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Subgradient minimax
    if time.time() < deadline - 180:
        p4_end = min(time.time() + 80, deadline - 140)
        h4, c4 = subgradient_minimax(best_h, n, p4_end, max_steps=4000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient coordinate descent
    if time.time() < deadline - 140:
        p5_end = min(time.time() + 70, deadline - 100)
        h5, c5 = grad_coord_descent(best_h, n, p5_end,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Refined active set (very high alpha, small top_k)
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 50, deadline - 70)
        bounds = [(0.0, 1.0)] * n
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-10, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 4:
            active = np.array(np.argsort(-c_vec[active])[:4])
        if len(active) > 0:
            for alpha in [20000000, 100000000]:
                if time.time() > deadline - 60:
                    break
                maxit = max(300, int((deadline - time.time()) * 2))
                try:
                    res = minimize(obj_active, best_h, jac=grad_active,
                                  args=(n, active, alpha, 0.001), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-25,
                                           'gtol': 1e-20, 'maxfun': maxit * 6,
                                           'maxls': 40, 'maxcor': 40})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass

    # Phase 7: Perturbation restarts
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 35, deadline - 40)
        h7, c7 = perturb_and_optimize(best_h, n, p7_end, rng, n_tries=3)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine local pair search
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 25, deadline - 20)
        h8, c8 = fine_local_search(best_h, n, p8_end, radius=20,
            deltas=[0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final coordinate descent
    if time.time() < deadline - 10:
        h9, c9 = grad_coord_descent(best_h, n, deadline - 3,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
