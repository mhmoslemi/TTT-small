    c = compute_c5_vec(x, n)
    return smooth_max_grad(c, alpha, x, n)

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_offsets = n - 1
    for wi, idx in zip(w, active):
        k = idx - n_offsets
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi * dx * (1.0 - x[k:n])
            grad[k:n] += wi * dx * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * dx * (1.0 - x[0:nv])
            grad[0:nv] += wi * dx * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alpha, maxit=3000, mu=0.01):
    bounds = [(0.0, 1.0)] * n
    remaining = max(5, deadline - time.time())
    maxit = min(maxit, max(100, int(remaining * 5)))
    try:
        res = minimize(obj_smooth, h, jac=grad_smooth,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxit * 12,
                               'maxls': 50, 'maxcor': 80})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def active_set_optimize(h, n, deadline, top_k=12, alphas=None, rounds=30):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100000, 500000, 2000000, 10000000, 50000000]
    for rnd in range(rounds):
        if time.time() > deadline - 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(200, int((deadline - time.time()) * 2))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def adam_optimize(h, n, deadline, alpha=1000000, lr=0.001, beta1=0.8, beta2=0.999,
                  eps=1e-8, top_k=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr_decay = 0.9995
    while time.time() < deadline:
        t += 1
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        g = active_grad(best_h, n, active)
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g * g
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        step = lr * (lr_decay ** t) * m_hat / (np.sqrt(v_hat) + eps)
        new_h = best_h - step
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = new_h
        else:
            m_v = m_v  # keep momentum
            lr *= 0.9
            if lr < 1e-10:
                break
    return best_h, best_c

def coord_pair_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    n_offsets = n - 1
    imps = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 30:
            active = np.array(np.argsort(-c_vec[active])[:30])
        grad = np.zeros(n)
        for idx in active:
            k = idx - n_offsets
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        pos_mask = (grad > 5e-8) & (best_h > 5e-8)
        neg_mask = (grad < -5e-8) & (best_h < 1.0 - 5e-8)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        found = False
        best_gain = 0
        for i in pos_idx[:40]:
            gi = grad[i]
            for j in neg_idx[:40]:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_gain < 1e-12:
            break
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                     0.00002, 0.00001, 0.000005, 0.000002]:
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                found = True
                imps += 1
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
        if not found and rnd > 10:
            break
    return best_h, best_c

def perturbation_search(h, n, deadline, n_perturb=30, n_coords=4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for p in range(n_perturb):
        if time.time() > deadline - 30:
            break
        h_pert = best_h.copy()
        idxs = rng.choice(n, size=n_coords, replace=False)
        for idx in idxs:
            h_pert[idx] += rng.uniform(-0.05, 0.05)
        h_pert = project_feasible(h_pert, n)
        h_new, c_new = active_set_optimize(h_pert, n, deadline - 25,
                                            top_k=10,
                                            alphas=[500000, 5000000],
                                            rounds=5)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 3 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def upsample_h(h, n_old, n_new):
    ratio = n_new / n_old
    if ratio == 2.0:
        h_new = np.zeros(n_new)
        h_new[0::2] = h
        h_new[1::2] = h
        return project_feasible(h_new, n_new)
    else:
        x_old = np.linspace(0, 2, n_old)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, h)
        return project_feasible(h_new, n_new)

def fine_sweep(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def newton_active(h, n, active, max_iter=100):
    """Solve for equalization of active shifts using Newton's method."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    n_offsets = n - 1
    m = len(active)
    for it in range(max_iter):
        c_vec = compute_c5_vec(best_h, n)
        vals = c_vec[active]
        target = vals.max()
        diff = vals - target
        if np.max(np.abs(diff)) < 1e-12:
            break
        # Build Jacobian: J[i,j] = dc[active[i]]/dh[j]
        # Approximate with finite differences for the active set
        J = np.zeros((m, n))
        for i, idx in enumerate(active):
            k = idx - n_offsets
            if k >= 0:
                nv = n - k
                J[i, 0:nv] = (1.0 - best_h[k:n]) * dx_val
                J[i, k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                J[i, kk:n] = (1.0 - best_h[0:nv]) * dx_val
                J[i, 0:nv] += (-best_h[kk:n]) * dx_val
        # Add constraint: sum(h) = n/2
        J_aug = np.vstack([J, np.ones(n)])
        b_aug = np.concatenate([-(vals - target), best_h.sum() - n/2.0])
        # Least squares solve
        try:
            step, _, _, _ = np.linalg.lstsq(J_aug, b_aug, rcond=None)
        except Exception:
            break
        step = np.clip(step, -best_h, 1.0 - best_h)
        lr = 1.0
        for _ in range(20):
            new_h = project_feasible(best_h + lr * step, n)
            c_new = compute_c5(new_h, n)
            if c_new < best_c:
                best_h = new_h
                best_c = c_new
                break
            lr *= 0.5
        if lr < 0.5:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Try a few alternative starts at n_orig
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        make_beatty_start(n_orig, golden / 2.0),
        make_beatty_start(n_orig, np.sqrt(2.0) / 2.0),
        np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_orig)]),
    ]
    for s in starts:
        s = project_feasible(s, n_orig)
        c_s = compute_c5(s, n_orig)
        if c_s < best_c:
            best_c = c_s
            best_h = s.copy()

    # Phase 1: L-BFGS warmup at original resolution
    p1_end = t0 + 120
    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    h1, c1 = best_h, best_c
    for alpha in alphas_p1:
        if time.time() > p1_end - 5:
            break
        h1, c1 = lbfgs_optimize(h1, best_n, p1_end - 3, alpha=alpha, maxit=2000)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization
    p2_end = min(t0 + 300, deadline - 400)
    h2, c2 = active_set_optimize(best_h, best_n, p2_end,
                                  top_k=15,
                                  alphas=[200000, 1000000, 5000000, 20000000],
                                  rounds=40)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Coordinate pair descent
    p3_end = min(t0 + 420, deadline - 300)
    h3, c3 = coord_pair_descent(best_h, best_n, p3_end, max_rounds=500)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Adam optimizer
    if time.time() < deadline - 250:
        p4_end = min(t0 + 500, deadline - 220)
        h4, c4 = adam_optimize(best_h, best_n, p4_end,
                                alpha=5000000, lr=0.0005, top_k=20)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try at higher resolution (n=1000)
    n_hi = 1000
    if time.time() < deadline - 200 and n_orig < n_hi:
        h_hi = upsample_h(best_h, n_orig, n_hi)
        # Quick L-BFGS at high res
        p5_end = min(time.time() + 80, deadline - 150)
        h5, c5 = lbfgs_optimize(h_hi, n_hi, p5_end, alpha=10000, maxit=1500)
        if c5 < best_c:
            # Continue optimizing at high res
            p5b_end = min(time.time() + 60, deadline - 80)
            h5b, c5b = active_set_optimize(h5, n_hi, p5b_end,
                                            top_k=15,
                                            alphas=[1000000, 10000000],
                                            rounds=10)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
                best_n = n_hi
            else:
                best_h = h5
                best_n = n_hi
            # Coord descent at high res
            if time.time() < deadline - 60:
                p5c_end = min(time.time() + 40, deadline - 50)
                h5c, c5c = coord_pair_descent(best_h, best_n, p5c_end, max_rounds=200)
                if c5c < best_c:
                    best_c = c5c
                    best_h = h5c

    # Phase 6: Perturbation search at current best resolution
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 60, deadline - 60)
        h6, c6 = perturbation_search(best_h, best_n, p6_end,
                                      n_perturb=15, n_coords=3)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = len(best_h)

    # Phase 7: Final active set + Newton equalization
    if time.time() < deadline - 30:
        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 20:
            active = np.array(np.argsort(-c_vec[active])[:20])
        h7, c7 = newton_active(best_h, best_n, active, max_iter=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # One more active set optimization round
        if time.time() < deadline - 15:
            p7_end = deadline - 10
            h7b, c7b = active_set_optimize(best_h, best_n, p7_end,
                                            top_k=20,
                                            alphas=[10000000, 50000000],
                                            rounds=10)
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Phase 8: Ultra-fine coordinate sweep
    if time.time() < deadline - 5:
        p8_end = deadline - 3
        h8, c8 = coord_pair_descent(best_h, best_n, p8_end, max_rounds=300)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: very fine pair search
    if time.time() < deadline - 3:
        h9, c9 = fine_sweep(best_h, best_n, deadline - 2, radius=30,
                            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
