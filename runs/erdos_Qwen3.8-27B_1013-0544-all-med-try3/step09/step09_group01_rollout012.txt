import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_per=None, bounds=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxit_per is not None and i < len(maxit_per):
            maxit = maxit_per[i]
        else:
            maxit = max(1000, int(remaining * 3))
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-12:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=6, alphas=None, n_rounds=15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000]
    
    for round_idx in range(n_rounds):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Active set: lags within a tolerance of max
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            top_idx = np.argsort(-c_vec[active])[:top_k]
            active = active[top_idx]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(500, int((deadline - time.time()) * 2))

                                       'maxls': 50, 'maxcor': 50})

                    best_h = h_new
            except Exception:

def local_refine(h, n, deadline, radius=50, n_paths=4):
    """Very local refinement with small moves"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for path in range(n_paths):
        h_work = best_h.copy()
        c_work = best_c
        step = 0.0005
        for _ in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(0, n, 3):  # Skip some for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    # Try increasing i, decreasing j
                    if h_work[i] + step <= 1.0 and h_work[j] - step >= 0.0:
                        h_work[i] += step
                        h_work[j] -= step
                        c = compute_c5(h_work, n)
                        if c < c_work - 1e-16:
                            c_work = c
                            improved = True
                        else:
                            h_work[i] -= step
                            h_work[j] += step
                    # Try decreasing i, increasing j
                    if h_work[i] - step >= 0.0 and h_work[j] + step <= 1.0:
                        h_work[i] -= step
                        h_work[j] += step
                        c = compute_c5(h_work, n)
                        if c < c_work - 1e-16:
                            c_work = c
                            improved = True
                        else:
                            h_work[i] += step
                            h_work[j] -= step
                if not improved and _ > 50:
                    step *= 0.5
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
        # Reset step for next path
        step = 0.0005
    
    return best_h, best_c

def make_binary_from_h(h, n):
    h_bin = np.zeros(n)
    n_ones = int(n / 2)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def upsamples(h, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_to = np.interp(x_to, x_from, h)
    return project_feasible(h_to, n_to)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()

        wm = w[mm] * dx
        if k >= 0:

            grad[k:n] += wm * (-x[0:nv])

    best_n = n0

    # Phase 1: Quick polish at original resolution
    p1_end = min(t0 + 150, deadline - 450)
    if p1_end > t0 + 30:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[100, 1000, 10000, 100000, 500000],
            mu=0.01, maxit_per=[2000, 3000, 3000, 4000, 4000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=1000
    n_up = 1000
    if time.time() < deadline - 350:
        h_up = upsample(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n_up
        else:
            best_h = h_up
            best_n = n_up
            best_c = compute_c5(best_h, best_n)

        # Phase 3: L-BFGS at n=1000
        p3_end = min(time.time() + 250, deadline - 250)
        h3, c3 = run_lbfgs(best_h, best_n, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 3000000],
            mu=0.005, maxit_per=[1500, 2000, 2000, 2500, 3000, 3500, 3500, 4000, 4000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n

    # Phase 4: Active set at best resolution
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 120, deadline - 120)
        h4, c4 = run_active(best_h, best_n, p4_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000], n_rounds=10)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Cyclic subgradient on true max
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 60, deadline - 60)
        h5, c5 = cyclic_subgrad(best_h, best_n, p5_end, n_iter=10000, eta0=0.002)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active set again (with sharper focus)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 40, deadline - 45)
        h6, c6 = run_active(best_h, best_n, p6_end, top_k=3,
            alphas=[500000, 2000000, 5000000], n_rounds=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish with hard constraint
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 20, deadline - 20)
        h7, c7 = slqp_polish(best_h, best_n, p7_end, alpha=2000000, maxiter=1000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very fine local refinement
    if time.time() < deadline - 10:
        p8_end = deadline - 3
        h8, c8 = local_refine(best_h, best_n, p8_end, radius=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
