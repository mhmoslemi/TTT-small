def lag_grad(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += -h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += -h[m:]
    return grad

def softmax_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    return w

    c = np.correlate(h, 1 - h, mode='full') * dx
    w = softmax_weights(c, alpha)
    active = np.where(w > 1e-12)[0]
    if len(active) == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        grad += w[idx] * lag_grad(h, n, lag)
    return grad * dx

For each h[j], the objective as a function of h[j] (holding all others fixed) is:
C5(h) = max_k f_k(h[j]) where f_k is the k-th overlap as a function of h[j].

Each f_k is linear in h[j] (for the relevant terms) or constant.

Actually, for a fixed lag m:
overlap(m) as a function of h[j]:
- If m > 0 and j < n-m: h[j](1-h[j+m]) + const → linear in h[j] with slope (1-h[j+m])
- If m > 0 and j ≥ m: h[j-m](1-h[j]) + const → linear in h[j] with slope (-h[j-m])
- If m < 0, similarly
- If m = 0: h[j](1-h[j]) + const → quadratic in h[j]

So for most lags, the dependence on h[j] is linear, and for lag 0 it's quadratic.

The max of linear functions is convex, so for each j, we can find the optimal h[j] by minimizing a convex piecewise-linear (plus one quadratic) function. This can be done in O(n log n) per coordinate (or O(n) with the right data structure).

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        wt = w[idx]
        lag = idx - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-m] += wt * (1.0 - h[m:])
            grad[m:] += -wt * h[:n-m]
        else:
            grad[m:] += wt * (1.0 - h[:n-m])
            grad[:n-m] += -wt * h[m:]
    return grad * dx, cmax + np.log(w_sum) / alpha

def momentum_descent(h_init, n, deadline, alpha, lr, momentum, alpha_multiplier=1.0, alpha_iter=1000):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    patience = 0
    lr_curr = lr
    alpha_curr = alpha
    
    while time.time() < deadline:
        g, _ = compute_smooth_max_grad(h, n, alpha_curr)
        v = momentum * v + lr_curr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h = h_new
        
        # Schedule adjustments
        if patience % alpha_iter == 0 and patience > 0:
            alpha_curr = min(5000000.0, alpha_curr * alpha_multiplier)
        
        if patience > 1000:
            lr_curr *= 0.8
            v *= 0.5
            patience = 0
        
        if patience > 10000:
            break
    
    return best_h, best_c

def _sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_constraint_jac(x, n):
    return np.ones(n)

def _slsqp_obj(x, n, alpha, top_m):
    c = compute_all_lags(x, n)
    top_idx = np.argsort(c)[-top_m:]
    top_vals = c[top_idx]
    m = top_vals.max()
    shifted = alpha * (top_vals - m)
    mx = shifted.max()
    if mx > 500:
        e = np.exp(shifted - mx)
        return float(m + (mx + np.log(e.sum())) / alpha)
    else:
        return float(m + np.log(np.exp(shifted).sum()) / alpha)

def _slsqp_obj_grad(x, n, alpha, top_m):
    c = compute_all_lags(x, n)
    top_idx = np.argsort(c)[-top_m:]
    top_vals = c[top_idx]
    m_t = top_vals.max()
    shifted = alpha * (top_vals - m_t)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] < 1e-12:
            continue
        wt = w[i]
        lag = idx - (n - 1)
        m_abs = abs(lag)
        if m_abs >= n:
            continue
        if lag == 0:
            grad += wt * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n-m_abs] += wt * (1.0 - x[m_abs:])
            grad[m_abs:] += -wt * x[:n-m_abs]
        else:
            grad[m_abs:] += wt * (1.0 - x[:n-m_abs])
            grad[:n-m_abs] += -wt * x[m_abs:]
    return grad * dx

def slsqp_refine(h, n, alpha, maxiter, top_m):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h.copy(), jac=_slsqp_obj_grad, args=(n, alpha, top_m),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        x_new = project_feasible(res.x, n)
        c_new = compute_c5(x_new, n)
        return x_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_swap_search(h_init, n, deadline, rng):
    # Make it binary
    h = h_init.copy()
    # Keep top n/2 values as 1
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    h = h_bin.copy()
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    one_set = sorted_idx[:n//2].copy()
    zero_set = sorted_idx[n//2:].copy()
    stagnant = 0
    
    while time.time() < deadline:
        a = one_set[rng.randint(len(one_set))]
        b = zero_set[rng.randint(len(zero_set))]
        
        h[a] = 0.0
        h[b] = 1.0
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            stagnant = 0
            # Update sets
            one_set[a] = b  # b is now 1, a is now 0
            zero_set[b] = a
        else:
            h[a] = 1.0
            h[b] = 0.0
            stagnant += 1
        
        if stagnant > 50000:
            break
    
    return best_h, best_c
