def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def coord_descent_sweep(h, n, mu, rng):
    """One sweep of coordinate descent."""
    target = n / 2.0
    order = rng.permutation(n)
    current_sum = h.sum()
    
    for m in order:
        old = h[m]
        # Compute current c_full
        c_full = np.correlate(h, 1 - h, mode='full').astype(float)
        
        # Determine range of i affected by h[m]
        start = max(0, n - 1 - m)
        end = min(2 * n - 2, 2 * n - 2 - m)
        
        a = c_full.copy()
        b = np.zeros(2 * n - 1)
        
        for i in range(start, end + 1):
            idx_A = m + i - (n - 1)
            j_B = m - (n - 1) + i
            a[i] -= old * (1.0 - h[idx_A])
            b[i] += (1.0 - h[idx_A])
            a[i] -= h[j_B] * (1.0 - old)
            b[i] -= h[j_B]
        
        S = current_sum - old - target
        
        # Ternary search for v in [0, 1]
        lo, hi = 0.0, 1.0
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(a + b * m1)) + mu * (S + m1) ** 2
            f2 = float(np.max(a + b * m2)) + mu * (S + m2) ** 2
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        v_star = (lo + hi) / 2.0
        h[m] = v_star
        current_sum += v_star - old
    
    return h
