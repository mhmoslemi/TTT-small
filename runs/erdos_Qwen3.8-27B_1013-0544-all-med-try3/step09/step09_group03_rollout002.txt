import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, mu):
    """Log-sum-exp smoothed max of correlations + penalty."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        wm = w[idx] * dx
        if k >= 0:

                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def power_sum_obj(x, n, p, mu):
    """Power-sum approximation: (sum c_k^p)^(1/p) + penalty."""
    c = compute_c5_vec(x, n)
    c_clipped = np.maximum(c, 0)  # correlations should be non-negative for h in [0,1]
    val = (np.sum(c_clipped ** p)) ** (1.0 / p)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def power_sum_grad(x, n, p, mu):

    c_clipped = np.maximum(c, 0)
    if p == 1:
        weights = (c_clipped > 0).astype(float)
        s = weights.sum()

            return np.zeros(n) + 2.0 * mu * (x.sum() - n / 2.0)
