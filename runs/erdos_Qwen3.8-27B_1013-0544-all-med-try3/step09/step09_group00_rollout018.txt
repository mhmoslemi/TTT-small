def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

    k_end2 = i + n - 1
    s_idx2 = i - np.arange(k_start2, k_end2 + 1) + n - 1  # = np.arange(n-1, -1, -1)
    delta[k_start2:k_end2+1] += h[s_idx2]
    
    # h[j]: 0->1, source gain: k in [n-1-j, 2n-2-j]
    k_start3 = n - 1 - j
    k_end3 = 2 * n - 2 - j
    t_idx3 = j + np.arange(k_start3, k_end3 + 1) - n + 1
    delta[k_start3:k_end3+1] += (1.0 - h[t_idx3])
    
    # h[j]: 0->1, target loss: k in [j, j+n-1]
    k_start4 = j
    k_end4 = j + n - 1
    s_idx4 = j - np.arange(k_start4, k_end4 + 1) + n - 1
    delta[k_start4:k_end4+1] -= h[s_idx4]
    
    return delta

def local_search_binary(h, n, deadline, max_sweeps=200):
    """Local search on binary h to minimize max overlap"""
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    o_vec = np.correlate(h, 1 - h, mode='full').astype(float)
    best_max = o_vec.max()
    best_c = best_max * dx
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        best_improvement = 0
        best_flip = None
        
        for i in ones:
            for j in zeros:
                d = flip_delta(h, i, j, n)
                new_max = np.maximum(o_vec + d, 0).max()
                improvement = best_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_flip = (i, j)
        
        if best_flip is None:
            break
        
        i, j = best_flip
        d = flip_delta(h, i, j, n)
        o_vec = np.maximum(o_vec + d, 0)
        h[i] = 0
        h[j] = 1
        ones.remove(i)
        zeros.append(j)
        ones.append(j)
        zeros.remove(i)
        best_max = o_vec.max()
        best_c = best_max * dx
    
    return h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary_from_sorted(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def make_sturmian(n, alpha):
    """Sturmian word of length n with slope alpha (exactly floor(n*alpha) or ceil)"""
    h = np.zeros(n)
    for i in range(n):
        if np.floor((i + 1) * alpha) - np.floor(i * alpha) == 1:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Turn off some ones
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in ones[:n_ones - target]:
            h[i] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target - n_ones]:
            h[i] = 1
    return h

def flip_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    # h[i]: 1->0, source loss
    ks = n - 1 - i
    ke = 2 * n - 2 - i
    if ks <= ke:
        t_idx = i + np.arange(ks, ke + 1) - n + 1
        delta[ks:ke+1] -= (1.0 - h[t_idx])
    
    # h[i]: 1->0, target gain
    ks2 = i
    ke2 = i + n - 1
    if ks2 <= ke2:
        s_idx = i - np.arange(ks2, ke2 + 1) + n - 1
        delta[ks2:ke2+1] += h[s_idx]
    
    # h[j]: 0->1, source gain
    ks3 = n - 1 - j
    ke3 = 2 * n - 2 - j
    if ks3 <= ke3:
        t_idx3 = j + np.arange(ks3, ke3 + 1) - n + 1
        delta[ks3:ke3+1] += (1.0 - h[t_idx3])
    
    # h[j]: 0->1, target loss
    ks4 = j
    ke4 = j + n - 1
    if ks4 <= ke4:
        s_idx4 = j - np.arange(ks4, ke4 + 1) + n - 1
        delta[ks4:ke4+1] -= h[s_idx4]
    
    return delta

def local_search_binary(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    o_vec = np.correlate(h, 1 - h, mode='full').astype(float)
    best_max = o_vec.max()
    best_c = best_max * dx
    
    sweeps = 0
    while sweeps < max_sweeps and time.time() < deadline:
        sweeps += 1
        best_improvement = 0
        best_i = -1
        best_j = -1
        best_delta = None
        
        for i in ones:
            for j in zeros:
                d = flip_delta(h, i, j, n)
                new_max = (o_vec + d).max()
                improvement = best_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
                    best_delta = d
        
        if best_i < 0:
            break
        
        o_vec = o_vec + best_delta
        h[best_i] = 0
        h[best_j] = 1
        ones.remove(best_i)
        ones.append(best_j)
        zeros.append(best_i)
        zeros.remove(best_j)
        best_max = o_vec.max()
        best_c = best_max * dx
    
    return h, best_c

        for _ in range(100):

            # Try reducing a 1

                h[i] -= eps
                h = project_feasible(h, n)

                    best_c = c
                    improved = True
                else:
                    h[j] -= eps
                    h = project_feasible(h, n)
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 300:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)
