def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:

            if tot < 1e-14:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def grad_search_full(h, n, deadline, top_pos=80, top_neg=80, max_iters=50000):

    best_c = compute_c5(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    dx_val = 2.0 / n
    count = 0
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        max_val = c_vec.max()
        tol = max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]

        for idx in active:

            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        pos_mask = (grad > 1e-12) & (best_h > 0.1)

    return np.ones(n)

def run_slqp(h, n, deadline, alpha=1000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(10, int(deadline - time.time()))
        maxit = min(maxiter, max(100, rem * 5))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Try alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        beatty_start(n_orig, golden / 2.0),
        beatty_start(n_orig, np.sqrt(2.0) / 2.0),
        beatty_start(n_orig, 0.38196601125),
        random_binary(n_orig, rng),
        random_binary(n_orig, rng),
    ]
    for hs in starts:
        c_alt = compute_c5(hs, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = hs.copy()

    # Phase 1: L-BFGS warmup
    p1_end = t0 + 120
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
        alphas=[20, 50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, maxit_list=[800, 1000, 1500, 1500, 2000, 2000, 2500, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Multi-active set optimization
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 180, deadline - 300)
        h2, c2 = multi_active_optimize(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient-guided search (larger candidate set)
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 150, deadline - 250)
        h3, c3 = grad_search_full(best_h, best_n, p3_end, top_pos=80, top_neg=80, max_iters=50000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Binary swap search (round to binary first)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 120, deadline - 200)
        h4b, c4b = binary_swap_search(best_h, best_n, p4_end, max_passes=100)
        if c4b < best_c:
            best_c = c4b
            best_h = h4b

    # Phase 5: Fine pair search (continuous)
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = fine_pair_search(best_h, best_n, p5_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another round of active set + gradient
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 80)
        h6, c6 = multi_active_optimize(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        if time.time() < deadline - 40:
            p6b_end = min(time.time() + 40, deadline - 35)
            h6b, c6b = grad_search_full(best_h, best_n, p6b_end, top_pos=100, top_neg=100, max_iters=30000)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b

    # Phase 7: Binary swap polish
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 45)
        h7b, c7b = binary_swap_search(best_h, best_n, p7_end, max_passes=50)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b

    # Phase 8: Very fine pair search
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 22, deadline - 20)
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: SLSQP polish
    if time.time() < deadline - 12:
        for alpha in [5000000, 20000000, 100000000]:
            if time.time() > deadline - 5:
                break
            h9, c9 = run_slqp(best_h, best_n, deadline - 2, alpha=alpha, maxiter=500)
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Final: very fine adjustment
    if time.time() < deadline - 2:
        hF, cF = fine_pair_search(best_h, best_n, deadline - 1, radius=20,
            deltas=[0.00001, 0.000005, 0.000002])
        if cF < best_c:
            best_c = cF
            best_h = hF

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
