import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def get_gradient(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            if n - m > 0:
                grad[0:n-m] += wk * (1.0 - h[m:n])
            if m < n:
                grad[m:n] += wk * (-h[0:n-m])
        else:
            m = -lag
            if n - m > 0:
                grad[m:n] += wk * (1.0 - h[0:n-m])
            if m < n:
                grad[0:n-m] += wk * (-h[m:n])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e10):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = get_gradient(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'maxls': 40,
                                'iprint': -1, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

def directed_local_search(h, n, deadline, rng, alpha=1000000):

    best_c = compute_c5(h, n)
    no_improve = 0

    while time.time() < deadline:

        top_k = np.argsort(c_all)[-5:]
        k_star = top_k[rng.randint(len(top_k))]
        lag = k_star - (n - 1)
        if lag == 0:
            no_improve += 1
            if no_improve > 5000:
                break
            continue
        m = abs(lag)
        if m >= n:
            no_improve += 1
            continue

        if lag > 0:
            contrib = h[0:n-m] * (1.0 - h[m:n])
        else:
            contrib = h[m:n] * (1.0 - h[0:n-m])

        if contrib.max() < 1e-8:
            no_improve += 1
            if no_improve > 10000:
                break
            continue

        top_contrib = np.argsort(contrib)[-20:]
        i = top_contrib[rng.randint(len(top_contrib))]

        if lag > 0:
            a, b = i, i + m
        else:
            a, b = i + m, i

        if h[a] < h[b]:
            a, b = b, a

        if h[a] - h[b] < 1e-8:
            no_improve += 1
            continue

        step = min(0.02, (h[a] - h[b]) * 0.15, 0.05)
        step = max(step, 0.001)

        h[a] -= step
        h[b] += step
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[a] += step
            h[b] -= step
            h = project_feasible(h, n)
            no_improve += 1
            if no_improve > 10000:
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                no_improve = 0
                if time.time() > deadline - 10:
                    break

    return best_h, best_c

def gradient_descent_finish(h, n, deadline, alpha=5000000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.001
    no_improve = 0

    while time.time() < deadline:
        grad = get_gradient(h, n, alpha)
        target_sum = n / 2.0
        diff = h.sum() - target_sum
        grad += 0.1 * diff * np.ones(n)

        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break

        direction = -grad / norm
        h_new = h + lr * direction
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            lr = min(lr * 1.1, 0.01)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 200:
                break

    return best_h, best_c

def multi_alpha_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    alpha_list = [500, 5000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]

            best_h = h_opt
