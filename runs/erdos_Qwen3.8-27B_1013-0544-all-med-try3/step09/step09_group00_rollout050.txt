    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_smooth(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(topK_obj, best_h,
                          args=(n, np.arange(len(compute_c5_vec(best_h, n))), alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, top_k=5, alphas=None, rounds=30, mu=0.001):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100000, 500000, 2000000, 10000000, 50000000, 200000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 5e-4 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def binary_local_search(h, n, deadline, max_radius=80, rounds=10):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 2:
            break
        radius = max_radius if rnd == 0 else max(max(5, max_radius // (rnd + 1)))
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            if time.time() > deadline - 2:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            candidates = zeros[(zeros >= lo) & (zeros < hi)]
            for j in candidates:
                if i == j:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_h = project_feasible(best_h, n)
        if not improved and rnd >= 2:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            n_iter = 0
            for i in range(n):
                if time.time() > deadline:
                    break
                n_iter += 1
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        old_i = best_h[i]
                        old_j = best_h[j]
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        old_i = best_h[i]
                        old_j = best_h[j]
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
    return best_h, best_c

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in step_sizes:
        stalled = 0
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-14:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stalled = 0
            else:
                best_h[best_i] = old_i
                best_h[best_j] = old_j
                best_h = project_feasible(best_h, n)
                stalled += 1
                if stalled > 100:
                    break
    return best_h, best_c

def perturb_reoptimize(h, n, deadline, n_trials=5, noise_scale=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for trial in range(n_trials):
        if time.time() > deadline - 30:
            break
        h_pert = best_h + rng.normal(0, noise_scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert > best_c * 1.001:
            continue
        h_opt, c_opt = active_set_lbfgs(h_pert, n, deadline - 30 - trial * 20,
                                         top_k=5,
                                         alphas=[100000, 1000000, 10000000],
                                         rounds=4, mu=0.001)
        if c_opt < best_c - 1e-16:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def interp_to_resolution(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_to)

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def multi_coord_fine(h, n, deadline, max_pairs_per_round=5000, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    for d in deltas:
        for _ in range(3):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-13) & (best_h > 1e-10)
            neg_mask = (grad < -1e-13) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            rng_local = np.random.RandomState(0)
            pairs = 0
            improved = False
            while pairs < max_pairs_per_round and time.time() < deadline:
                i = pos_idx[rng_local.randint(len(pos_idx))]
                j = neg_idx[rng_local.randint(len(neg_idx))]
                if i == j:
                    pairs += 1
                    continue
                amt = min(d, best_h[i], 1.0 - best_h[j])
                if amt < 1e-13:
                    pairs += 1
                    continue
                old_i = best_h[i]
                old_j = best_h[j]
                best_h[i] -= amt
                best_h[j] += amt
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
                pairs += 1
            if not improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Try alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty_start(n0, golden / 2.0),
        make_beatty_start(n0, np.sqrt(2.0) / 2.0),
        make_beatty_start(n0, 0.41421356),
    ]
    for _ in range(3):
        alt_starts.append(make_beatty_start(n0, 0.2 + rng.random() * 0.3))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n0)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup
    p1_end = t0 + 60
    if time.time() < p1_end:
        h1, c1 = lbfgs_smooth(best_h, best_n, p1_end,
            alphas=[10, 50, 200, 1000, 5000, 50000],
            mu=0.01, maxit_list=[800, 1000, 1500, 2000, 2500, 3000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization
    p2_end = min(time.time() + 150, deadline - 400)
    if time.time() < p2_end:
        h2, c2 = active_set_lbfgs(best_h, best_n, p2_end,
            top_k=6, alphas=[50000, 200000, 1000000, 5000000, 20000000, 100000000],
            rounds=20, mu=0.001)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient-guided search
    p3_end = min(time.time() + 120, deadline - 300)
    if time.time() < p3_end:
        h3, c3 = grad_guided_search(best_h, best_n, p3_end,
            step_sizes=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=80000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Active set refinement
    p4_end = min(time.time() + 100, deadline - 250)
    if time.time() < p4_end:
        h4, c4 = active_set_lbfgs(best_h, best_n, p4_end,
            top_k=4, alphas=[500000, 5000000, 50000000, 200000000],
            rounds=15, mu=0.0005)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Binary local search
    p5_end = min(time.time() + 100, deadline - 200)
    if time.time() < p5_end:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n)
        h5, c5v = binary_local_search(h_bin, best_n, p5_end, max_radius=60, rounds=8)
        if c5v < best_c:
            best_c = c5v
            best_h = h5
        h5b, c5b = binary_local_search(best_h, best_n, p5_end, max_radius=30, rounds=5)
        if c5b < best_c:
            best_c = c5b
            best_h = h5b

    # Phase 6: Fine pair search
    p6_end = min(time.time() + 80, deadline - 150)
    if time.time() < p6_end:
        h6, c6 = fine_pair_search(best_h, best_n, p6_end, radius=40,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Multi-coord fine search
    p7_end = min(time.time() + 60, deadline - 100)
    if time.time() < p7_end:
        h7, c7 = multi_coord_fine(best_h, best_n, p7_end,
            max_pairs_per_round=3000,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Perturb and re-optimize
    p8_end = min(time.time() + 60, deadline - 60)
    if time.time() < p8_end:
        h8, c8 = perturb_reoptimize(best_h, best_n, p8_end,
            n_trials=4, noise_scale=0.03)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final active set + fine tuning
    p9_end = min(time.time() + 40, deadline - 25)
    if time.time() < p9_end:
        h9, c9 = active_set_lbfgs(best_h, best_n, p9_end,
            top_k=3, alphas=[10000000, 100000000, 500000000],
            rounds=10, mu=0.0001)
        if c9 < best_c:
            best_c = c9
            best_h = h9
        if time.time() < deadline - 15:
            h9b, c9b = multi_coord_fine(best_h, best_n, deadline - 10,
                max_pairs_per_round=2000,
                deltas=[0.00001, 0.000005, 0.000002, 0.000001])
            if c9b < best_c:
                best_c = c9b
                best_h = h9b

    # Phase 10: Try different resolution
    if time.time() < deadline - 30:
        for n_alt in [400, 600]:
            if time.time() > deadline - 20:
                break
            h_alt_res = interp_to_resolution(best_h, best_n, n_alt)
            c_alt_res = compute_c5(h_alt_res, n_alt)
            if c_alt_res < best_c * 0.999:
                h_opt, c_opt = active_set_lbfgs(h_alt_res, n_alt, deadline - 15,
                    top_k=5, alphas=[100000, 1000000, 10000000],
                    rounds=6, mu=0.001)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    best_n = n_alt

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    # Ensure exact sum
    diff = final_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        adjust = np.where(best_h < 1.0 - 1e-10, 1, 0).astype(float) if diff > 0 else np.where(best_h > 1e-10, 1, 0).astype(float)
        tot = adjust.sum()
        if tot > 0:
            best_h += (diff / tot) * adjust
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, final_n)

    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
