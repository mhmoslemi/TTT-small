import numpy as np
from scipy.optimize import minimize
import time

def fft_corr(h, n):
    """FFT-based cross-correlation of h and (1-h). Returns 2n-1 values matching np.correlate(h, 1-h, 'full')."""
    if n == 1:
        return np.array([h[0] * (1.0 - h[0])])
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    r = np.fft.irfft(H * G.conj(), N)
    return np.concatenate([r[N - n + 1:], r[:n]])

def c5_vec(h, n):
    return fft_corr(h, n) * (2.0 / n)

def c5_val(h, n):
    return float(c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def smo_obj(x, n, alpha, mu, active):
    c = c5_vec(x, n)
    v = c[active]
    m = v.max()
    s = alpha * (v - m)
    mx = s.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smo_grad(x, n, alpha, mu, active):
    c = c5_vec(x, n)
    v = c[active]
    m = v.max()
    s = alpha * (v - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_lbfgs(h, n, deadline, alpha, mu, top_k, maxiters):
    """Run L-BFGS-B on active set of top_k worst constraints."""
    c = c5_vec(h, n)
    m = c.max()
    threshold = m - max(1e-9, 0.0005 * m)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    elif len(active) == 0:
        active = np.array([np.argmax(c)])
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smo_obj, h, jac=smo_grad,
                       args=(n, alpha, mu, active), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiters, 'ftol': 1e-24,
                                'gtol': 1e-16, 'maxfun': maxiters * max(10, min(50, n // 100)),
                                'maxls': 50, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = c5_val(h_new, n)
        return h_new, c_new, active
    except Exception:
        return h, c5_val(h, n), active

def upsample(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def iterative_active_opt(h, n, deadline, max_rounds=200):
    """Iterative active-set optimization with adaptive parameters."""
    best_h = project_feasible(h, n)
    best_c = c5_val(best_h, n)
    alphas = [1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    mus = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for rnd in range(max_rounds):
        rem = deadline - time.time()
        if rem < 30:
            break
        
        c = c5_vec(best_h, n)
        m = c.max()
        
        # Find active set: top worst + near-worst
        threshold = m - max(1e-9, 0.001 * m)
        near = np.where(c >= threshold)[0]
        top_k = min(20, max(5, len(near)))
        order = np.argsort(-c)
        active = order[:top_k]
        
        # Choose alpha and mu based on round progression
        frac = rnd / max(1, max_rounds)
        ai = min(int(frac * len(alphas)), len(alphas) - 1)
        # Allow some variation
        alpha = alphas[ai]
        mu = mus[ai]
        
        maxit = max(200, int(rem * 0.8))
        
        h_new, c_new, _ = active_lbfgs(best_h, n, deadline, alpha, mu, top_k, maxit)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
        elif c_new < best_c + 1e-10:
            # Try with slightly different active set (exclude the absolute worst)
            active2 = order[1:min(top_k + 3, len(order))]
            h_new2, c_new2, _ = active_lbfgs(best_h, n, deadline, alpha * 2, mu, len(active2), maxit // 2)
            if c_new2 < best_c - 1e-12:
                best_c = c_new2
                best_h = h_new2
    
    return best_h, best_c

def greedy_peel(h, n, deadline, max_rounds=50):
    """Peel off the worst constraint one at a time."""
    best_h = project_feasible(h, n)
    best_c = c5_val(best_h, n)
    
    for rnd in range(max_rounds):
        rem = deadline - time.time()
        if rem < 20:
            break
        c = c5_vec(best_h, n)
        worst = np.argmax(c)
        active = np.array([worst])
        alpha = 10000000
        mu = 0.0001
        maxit = max(100, int(rem * 0.5))
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smo_obj, best_h, jac=smo_grad,
                           args=(n, alpha, mu, active), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-24,
                                    'gtol': 1e-16, 'maxfun': maxit * 20,
                                    'maxls': 40, 'maxcor': 20})
            h_new = project_feasible(res.x, n)
            c_new = c5_val(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def local_search(h, n, deadline):
    """Efficient local search: try moving mass from worst-contributing positions."""
    best_h = h.copy()
    best_c = c5_val(h, n)
    dx = 2.0 / n
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                # Try increasing h[i] and decreasing h[j] for nearby j
                lo = max(0, i - 50)
                hi = min(n, i + 51)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def random_binary_start(n, seed_val):
    """Create a random binary-like starting point."""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def periodic_start(n, period, duty):
    """Create a periodic starting point with given period and duty cycle."""
    h = np.zeros(n)
    x = np.arange(n) * (2.0 / n)
    phase = (x % (2.0 / period)) * period
    h[phase < 2.0 * duty / period] = 1.0
    return project_feasible(h, n)
