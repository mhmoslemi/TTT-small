            h += (diff / tot) * h

def binarize(h, n):
    """Convert to binary, maintaining sum constraint"""
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    if h_bin.sum() != target:
        # Adjust: sort by distance to 0.5, flip the closest ones
        diff = target - h_bin.sum()
        if diff > 0:
            # Need more 1s: flip 0s closest to 1
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(diff)]:
                h_bin[zeros[idx]] = 1.0
        else:
            # Need fewer 1s: flip 1s closest to 0
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(-diff)]:
                h_bin[ones[idx]] = 0.0
    return h_bin

    """Compute new C5 after swapping h[i] (was 1) and h[j] (was 0)"""
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return compute_c5(h2, n)

    if time.time() < deadline - 2:
        h9, c9 = fine_coord_descent(best_h, best_n, deadline - 1,
                                   steps=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                   max_rounds=100)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
