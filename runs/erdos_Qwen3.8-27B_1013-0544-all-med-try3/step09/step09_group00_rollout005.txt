            step *= 0.5
            if step < 1e-10:
                break
            no_improve += 1
            if no_improve > 20:
                break
            continue
        # Try multiple step sizes via line search
        moves = [step, step * 0.5, step * 0.25, step * 0.1]
        found = False
        for mv in moves:
            amt = min(mv, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                continue
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                steps_done += 1
                no_improve = 0
                found = True
                break
            else:
                best_h[best_i] = old_i
                best_h[best_j] = old_j
                best_h = project_feasible(best_h, n)
        if not found:
            step *= 0.5
            no_improve += 1
            if step < 1e-10 or no_improve > 30:
                break
        steps_done += 1
    return best_h, best_c

def pair_local_search(h, n, deadline, radius=60, deltas=None):
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] >= 1e-6 and best_h[j] <= 1.0 - 1e-6:
                        if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] += d
                                best_h[j] -= d
                                best_h = project_feasible(best_h, n)
                    if best_h[i] <= 1.0 - 1e-6 and best_h[j] >= 1e-6:
                        if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] -= d
                                best_h[j] += d
                                best_h = project_feasible(best_h, n)
    return best_h, best_c

def binary_swap_search(h, n, deadline, max_rounds=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng_local = np.random.RandomState(12345 + rnd)
        rng_local.shuffle(ones)
        rng_local.shuffle(zeros)
        for i in ones[:min(len(ones), 80)]:
            if time.time() > deadline - 3:
                break
            for j in zeros[:min(len(zeros), 80)]:
                old_i = best_h[i]
                old_j = best_h[j]
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
            if improved:
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h < 0.5)[0]
        if not improved:
            break
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def perturb_and_reopt(h, n, deadline, rng, num_perturbs=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(num_perturbs):
        if time.time() > deadline - 30:
            break
        h_p = h.copy()
        # Random perturbation
        n_perturb = min(n // 10, 30)
        idx = rng.choice(n, size=n_perturb, replace=False)
        h_p[idx] = np.clip(h_p[idx] + rng.uniform(-0.1, 0.1, n_perturb), 0, 1)
        h_p = project_feasible(h_p, n)
        # Short optimization
        try:
            c_vec = compute_c5_vec(h_p, n)
            active = get_active_lags(c_vec, frac=0.01, max_k=10)
            bounds = [(0.0, 1.0)] * n
            maxit = max(100, int((deadline - time.time()) * 0.5))
            res = minimize(active_set_objective, h_p, jac=active_set_gradient,
                          args=(n, active, 1000000, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    n_orig = len(h0)
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Try alternative initializations at same n
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    irrationals = [golden / 2.0, np.sqrt(2.0) / 2.0, np.sqrt(3.0) / 2.0,
                   np.pi / 4.0, np.e / 3.0, np.sqrt(5.0) / 3.0]
    alt_starts = []
    for irr in irrationals:
        alt_starts.append(make_beatty_start(n_orig, irr))
    for _ in range(4):
        alt_starts.append(make_random_binary(n_orig, rng))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Also try 1-h symmetry
    h_flip = project_feasible(1.0 - h0, n_orig)
    c_flip = compute_c5(h_flip, n_orig)
    if c_flip < best_c:
        best_c = c_flip
        best_h = h_flip.copy()

    # Phase 1: L-BFGS warmup (0 to ~150s)
    p1_end = t0 + 150
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
        alphas=[10, 50, 200, 1000, 5000, 50000, 500000],
        mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization (~150s to ~350s)
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 200, deadline - 300)
        h2, c2 = phase_active(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: L-BFGS with high alpha (~350s to ~450s)
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 100, deadline - 280)
        h3, c3 = phase_lbfgs(best_h, best_n, p3_end,
            alphas=[100000, 1000000, 10000000], mu=0.001)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Active set again with larger k (~450s to ~550s)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 100, deadline - 230)
        h4, c4 = phase_active(best_h, best_n, p4_end,
            configs=[(10, 5000000, 4), (16, 50000000, 3), (24, 200000000, 3)])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient local search (~550s to ~750s)
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 180, deadline - 180)
        h5, c5 = grad_local_search(best_h, best_n, p5_end, max_steps=300000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Pair local search (~750s to ~900s)
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 120, deadline - 130)
        h6, c6 = pair_local_search(best_h, best_n, p6_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Binary swap search (~900s to ~950s)
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 50, deadline - 90)
        h7, c7 = binary_swap_search(best_h, best_n, p7_end, max_rounds=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Perturb and reoptimize (~950s to ~1030s)
    if time.time() < deadline - 50:
        p8_end = min(time.time() + 70, deadline - 40)
        h8, c8 = perturb_and_reopt(best_h, best_n, p8_end, rng, num_perturbs=4)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final gradient polish
    if time.time() < deadline - 30:
        p9_end = min(time.time() + 25, deadline - 25)
        h9, c9 = grad_local_search(best_h, best_n, p9_end, max_steps=100000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Very fine pair search
    if time.time() < deadline - 10:
        h10, c10 = pair_local_search(best_h, best_n, deadline - 3, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    # Ensure exact constraint
    diff = final_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(1.0 - best_h)
            best_h[idx] = min(1.0, best_h[idx] + diff)
        else:
            idx = np.argmax(best_h)
            best_h[idx] = max(0.0, best_h[idx] + diff)
        best_h = project_feasible(best_h, final_n)
        best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
