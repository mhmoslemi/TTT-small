        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slqp_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active_shifts(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000]
    
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(300, int((deadline - time.time()) * 2.5))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def run_slqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, jac=lambda x: slqp_grad(x, n, alpha),
                      args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d_idx, d in enumerate(deltas):
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def vectorized_pair_search(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                js = list(range(lo, hi))
                if i in js:
                    js.remove(i)
                
                # Try all j at once using vectorized check
                valid_plus = [j for j in js if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0]
                valid_minus = [j for j in js if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0]
                
                for j in valid_plus:
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        c_vec = compute_c5_vec(best_h, n)
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                
                if improved:
                    continue
                    
                for j in valid_minus:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        c_vec = compute_c5_vec(best_h, n)
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
    
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.0008
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.99
        
        h = h_new
    
    return best_h, best_c

def make_periodic_start(n, period, frac_ones=0.5):
    h = np.zeros(n)
    ones_per_period = max(1, int(period * frac_ones))
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def upsample(h_coarse, n_coarse, n_fine):
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + 1.0/n_coarse
    x_fine = np.linspace(0, 2, n_fine, endpoint=False) + 1.0/n_fine
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return project_feasible(h_fine, n_fine)

def random_perturb(h, n, rng, scale=0.01):
    h_new = h + rng.randn(n) * scale
    return project_feasible(h_new, n)

def targeted_lag_reduction(h, n, deadline, max_iters=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.002
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-8, 0.0002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            step = min(step * 1.002, 0.01)
        else:
            step *= 0.995
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at original resolution with multi-alpha annealing
    p1_end = min(t0 + 280, deadline - 600)
    h1, c1 = run_lbfgs_phase(best_h, best_n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.005, maxiter_per=[1500, 2000, 2000, 2500, 3000, 3000, 3500, 3500, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active-shifts at original resolution
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 120, deadline - 500)
        h2, c2 = run_active_shifts(best_h, best_n, p2_end, top_k=6,
            alphas=[50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n_fine = 1000
    if time.time() < deadline - 400:
        x_fine = np.linspace(0, 2, n_fine, endpoint=False) + 1.0/n_fine
        x_coarse = np.linspace(0, 2, best_n, endpoint=False) + 1.0/best_n
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p3_end = min(time.time() + 250, deadline - 250)
        h3, c3 = run_lbfgs_phase(h_fine, n_fine, p3_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.003, maxiter_per=[1500, 2000, 2000, 2500, 2500, 3000, 3000, 3000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_fine
    
    # Phase 4: Active shifts at current best resolution
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = run_active_shifts(best_h, best_n, p4_end, top_k=5,
            alphas=[100000, 500000, 2000000, 5000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Targeted lag reduction
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 80, deadline - 150)
        h5, c5_val = targeted_lag_reduction(best_h, best_n, p5_end, max_iters=8000)
        if c5_val < best_c:
            best_c = c5_val
            best_h = h5
    
    # Phase 6: Random perturbation + re-optimize (escape local minima)
    if time.time() < deadline - 150:
        for pert_scale in [0.02, 0.005]:
            if time.time() > deadline - 120:
                break
            h_pert = random_perturb(best_h, best_n, rng, scale=pert_scale)
            p6_end = min(time.time() + 50, deadline - 100)
            h6, c6 = run_lbfgs_phase(h_pert, best_n, p6_end,
                alphas=[10000, 100000, 1000000],
                mu=0.005, maxiter_per=[2000, 2000, 2000])
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 80:
        for alpha in [500000, 2000000, 5000000]:
            if time.time() > deadline - 60:
                break
            p7_end = min(time.time() + 20, deadline - 50)
            h7, c7 = run_slqp(best_h, best_n, p7_end, alpha=alpha, maxiter=1000)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Subgradient refinement
    if time.time() < deadline - 50:
        p8_end = min(time.time() + 25, deadline - 35)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=3000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Fine pair search (most critical for final squeeze)
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=40)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final SLSQP
    if time.time() < deadline - 1:
        h10, c10 = run_slqp(best_h, best_n, deadline - 0.5, alpha=5000000, maxiter=500)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
