            break

    for ai, m in enumerate(active):
        k = m - (n-1)
        if k >= 0:
            nv = n - k
            # ∂c_k/∂h[i] = dx*(1-h[i+k]) for i < nv
            G[ai, :nv] += dx * (1.0 - h[k:])
            # ∂c_k/∂h[i] -= dx*h[i-k] for i >= k
            G[ai, k:] -= dx * h[:nv]
        else:
            kk = -k
            nv = n - kk
            G[ai, kk:] += dx * (1.0 - h[:nv])
            G[ai, :nv] -= dx * h[kk:]
    return G  # shape (n_act, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_smoothed(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 8,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_local_search(h, n, deadline, top_k=4):

    n_sorted = np.argsort(best_h)[::-1]

        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        best_move = None
        best_move_val = best_c
        interior = np.where((best_h > 1e-6) & (best_h < 1 - 1e-6))[0]
        if len(interior) == 0:
            break
        np.random.shuffle(interior)
        interior = interior[:min(80, len(interior))]
        for i in interior:
            if time.time() > deadline:
                break
            for j in np.random.choice(n, min(60, n), replace=False):
                if i == j:
                    continue
                grad_diff = np.zeros(len(active))
                for ai, m in enumerate(active):
                    k = m - (n - 1)
                    if k >= 0:
                        g_i = 0.0
                        if i < n - k:
                            g_i += (1.0 - best_h[i + k])
                        if i >= k:
                            g_i -= best_h[i - k]
                        g_j = 0.0
                        if j < n - k:
                            g_j += (1.0 - best_h[j + k])
                        if j >= k:

                    else:
                        kk = -k

                        g_j = 0.0
                        if j >= kk:
                            g_j += (1.0 - best_h[j - kk])
                        if j < n - kk:
                            g_j -= best_h[j + kk]
                    grad_diff[ai] = dx * (g_i - g_j)
                cn = c_vec[active]
                for delta in [0.002, -0.002, 0.005, -0.005, 0.01, -0.01,

                    if best_h[j] - delta < 0 or best_h[j] - delta > 1:
                        continue
                    approx_max = np.max(cn + grad_diff * delta)

        if best_move is not None:

            best_h[i] += delta
            best_h[j] -= delta
            best_h = project_feasible(best_h, n)
            actual_c = compute_c5(best_h, n)
            if actual_c < best_c:

            else:

def stoc_gradient_steps(h, n, deadline, n_iter=50000):

    eta = 0.003

        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-10, 1e-4 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 4:
            order = np.argsort(-c_vec[active])
            active = active[order[:4]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        bounds = [(0.0, 1.0)] * best_n
        for a in [10000000, 50000000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(best_n, active, a, 0.002),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-22,
                                        'gtol': 1e-16, 'maxfun': 800,
                                        'maxls': 40, 'maxcor': 30})
                h_final = project_feasible(res.x, best_n)
                c_final = compute_c5(h_final, best_n)
                if c_final < best_c:
                    best_c = c_final
                    best_h = h_final
            except Exception:
                pass

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
