import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_alpha(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def logsumexp_alpha_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def focused_gradient(x, n, top_k=20):
    """Compute gradient using only the top-k worst correlations."""
    c = compute_c5_vec(x, n)
    # Get indices of top-k worst
    top_idx = np.argsort(c)[-top_k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]  # sorted worst first
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1

        wm = dx  # equal weight for top-k

def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

def binary_project(h, n):
    """Project to nearest binary solution maintaining sum constraint."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0 and diff > 0:
                h_b[i] = 1.0
                diff -= 1.0
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0 and diff < 0:
                h_b[i] = 0.0
                diff += 1.0
    return h_b

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def binary_project(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)

                diff -= 1.0

        for i in order:

        h_p = project_feasible(h_p, n)
        try:
            res = minimize(lse_obj, h_p, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=800000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def coordinate_descent(h, n, deadline, n_rounds=500):
    """Coordinate descent focusing on worst correlations."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(555)
    steps = [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]
    
    for rd in range(n_rounds):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Focus on worst 100 lags
        worst_lags = np.argsort(c_vec)[-100:]
        improved = False
        for lag_idx in worst_lags:
            if time.time() > deadline - 3:
                break
            k = lag_idx - (n - 1)
            if k >= 0:
                affected = list(range(k, n))
            else:
                affected = list(range(-k, n))
            # Try a few random positions
            trial_count = min(10, len(affected))
            for _ in range(trial_count):
                if time.time() > deadline - 3:
                    break
                j = affected[rng.randint(len(affected))]
                for ds in steps:
                    nv = best_h[j] + ds
                    if nv < -1e-9 or nv > 1.0 + 1e-9:
                        continue
                    old = best_h[j]
                    best_h[j] = nv
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        best_h[j] = old
                if improved:
                    break
            if improved:
                break
        if not improved and rd > 50:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_base = 400
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick optimize at base resolution (20s)
    t1 = min(t0 + 25, deadline - 950)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            [(100, 20.0, 5000), (1000, 200.0, 10000), (5000, 2000.0, 15000)])
    
    # Phase 2: Main optimization at n=2000 (most of the time)
    n_main = 2000
    h_main = interpolate_to(best_h, n_base, n_main)
    t2_start = time.time()
    t2_end = deadline - 220
    if t2_end > t2_start + 30:
        configs_main = [
            (50, 10.0, 8000),
            (200, 50.0, 10000),
            (800, 200.0, 15000),
            (3000, 1000.0, 20000),
            (10000, 5000.0, 25000),
            (40000, 20000.0, 30000),
            (150000, 80000.0, 30000),
            (500000, 200000.0, 20000),
            (2000000, 500000.0, 15000),
        ]
        h_main, c_main = run_lbfgs(h_main, n_main, t2_end, configs_main)
        if c_main < best_c:
            best_c = c_main
            best_h = h_main
    
    # Phase 3: Binary local search (80s)
    t3 = min(time.time() + 80, deadline - 130)
    if t3 > time.time() + 10:
        h_bin = binary_project(best_h, len(best_h))
        h_bl, c_bl = binary_local_search(h_bin, len(best_h), t3, max_steps=20000)
        if c_bl < best_c:
            best_c = c_bl
            best_h = h_bl
        # Re-optimize around binary pattern
        t3b = min(time.time() + 40, deadline - 80)
        if t3b > time.time() + 5:
            h_re, c_re = run_lbfgs(h_bl, len(best_h), t3b,
                [(5000, 2000.0, 10000), (50000, 20000.0, 15000)])
            if c_re < best_c:
                best_c = c_re
                best_h = h_re
    
    # Phase 4: Perturbation search (60s)
    t4 = min(time.time() + 60, deadline - 70)
    if t4 > time.time() + 5:
        best_h, best_c = perturb_optimize(best_h, len(best_h), t4,
            n_trials=5, alpha=20000, mu=10000.0)
    
    # Phase 5: SLSQP polish (50s)
    t5 = min(time.time() + 50, deadline - 30)
    n_cur = len(best_h)
    if t5 > time.time() + 5 and n_cur <= 2000:
        h5, c5 = slqp_polish(best_h, n_cur, t5, alpha=1000000, maxiter=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    elif t5 > time.time() + 5:
        # For higher n, do a short L-BFGS instead
        h5, c5 = run_lbfgs(best_h, n_cur, t5,
            [(100000, 50000.0, 10000)])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Coordinate descent refinement (remaining time)
    if time.time() < deadline - 5:
        best_h, best_c = coordinate_descent(best_h, len(best_h), deadline - 2)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
