def projected_gradient_step(h, n, grad, step_size):
    """Take a projected gradient step with sum constraint."""
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project onto sum constraint
    target = n / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:

            tot = room.sum()

                h_new += (diff / tot) * room
        else:
            tot = h_new.sum()
            if tot > 1e-14:
                h_new += (diff / tot) * h_new
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val

        # To reduce max correlation, we want to decrease h where grad > 0 and increase where grad < 0
        pos_mask = (grad > 1e-8) & (best_h > 1e-8)
        neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
        
        if not pos_mask.any() or not neg_mask.any():
            break
        
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]

        top_pos = pos_idx[np.argsort(-grad[pos_mask])[:10]]
        top_neg = neg_idx[np.argsort(grad[neg_mask])[:10]]
        
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in top_pos:
            for j in top_neg:
                gain = grad[i] - grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        
        if best_i < 0 or best_gain < 1e-8:
            break
        
        # Try multiple step sizes
        amounts = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        improved = False
        for amt in amounts:
            if amt > best_h[best_i] or amt > 1.0 - best_h[best_j]:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
        
        if not improved:
            # Decay step and try again
            if step > 100 and not improved:
                break
    
    return best_h, best_c
