    dx = 2.0 / n
    if k >= 0:

    best_c = compute_c5(h, n)

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new

def make_beatty(ns, alpha_param):
    h = np.zeros(ns)
    n_ones = ns // 2
    positions = set()
    for k in range(1, 4 * ns):
        pos = int(k * alpha_param) % ns
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = ns / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, ns)

def run_subgradient(h, n, deadline, init_step=0.02, decay=0.9997, tol_frac=0.003, max_iter=300000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    step = init_step
    plateau = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_cur, n)
        max_val = c_vec.max()
        tol = max(1e-9, tol_frac * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            dx = 2.0 / n
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - h_cur[k:n]) * dx
                g[k:n] += (-h_cur[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - h_cur[0:nv]) * dx
                g[0:nv] += (-h_cur[kk:n]) * dx
        g /= len(active)
        h_new = h_cur - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h_cur = h_new
            plateau = 0
        else:
            plateau += 1
            if plateau > 500:
                step = max(step * 0.5, 1e-8)
                h_cur = best_h.copy()
                plateau = 0
        step *= decay
    return best_h, best_c

def run_coord_descent(h, n, deadline, step_sizes=None, max_steps=200000):

    best_c = compute_c5(h, n)
    if step_sizes is None:

        while time.time() < deadline and steps_done < max_steps:

            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val

    # Phase 3: Try higher resolution n=1000 (380-550s)
    n_hi = 1000
    p3_end = min(t0 + 550, deadline - 250)
    if p3_end > time.time() + 30:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            best_n = n_hi
        h3, c3 = run_lbfgs_phase(h_hi, n_hi, p3_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005,
            maxit_list=[500, 800, 1000, 1200, 1500, 1800, 2000, 2000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_hi

    # Phase 4: Subgradient descent at best resolution (550-720s)
    p4_end = min(t0 + 720, deadline - 200)
    if p4_end > time.time() + 20:
        h4, c4 = run_subgradient(best_h, best_n, p4_end,
            init_step=0.01, decay=0.9998, tol_frac=0.002, max_iter=200000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Coordinate descent (720-850s)
    p5_end = min(t0 + 850, deadline - 150)
    if p5_end > time.time() + 20:
        h5, c5 = run_coord_descent(best_h, best_n, p5_end,
            step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            max_steps=150000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active set refinement (850-950s)
    p6_end = min(t0 + 950, deadline - 80)
    if p6_end > time.time() + 15:
        h6, c6 = run_active_phase(best_h, best_n, p6_end,
            top_k=12, alphas=[1000000, 5000000, 20000000],
            rounds=10, mu=0.0005)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine pair search (950-1030s)
    p7_end = min(t0 + 1030, deadline - 40)
    if p7_end > time.time() + 15:
        h7, c7 = run_fine_pairs(best_h, best_n, p7_end,
            radius=35,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very fine final polish (1030 to deadline)
    p8_end = deadline - 3
    if p8_end > time.time() + 5:
        h8, c8 = run_coord_descent(best_h, best_n, p8_end,
            step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001],
            max_steps=80000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        # One more fine pair pass
        if time.time() < deadline - 3:
            h8b, c8b = run_fine_pairs(best_h, best_n, deadline - 2,
                radius=20,
                deltas=[0.00002, 0.00001, 0.000005, 0.000002])
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
