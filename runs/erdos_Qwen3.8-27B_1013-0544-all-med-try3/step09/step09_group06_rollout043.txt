def smooth_max_grad(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    
    if top_k is not None:
        # Focus on top_k
        top_idx = np.argsort(w)[::-1][:top_k]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()
        active = list(zip(top_idx, w_top))

    for k_idx, w_k in active:
        lag = k_idx - (n - 1)
        grad += w_k * lag_gradient(h, n, lag)

def efficient_local_search(h, n, deadline, max_rounds=100):
    """
    For the top lags, systematically try pairwise swaps.
    Uses the D-vector trick for fast evaluation.
    """
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Compute current lag values
        S = compute_all_lags(h, n)
        current_max = S.max()

        K = 5
        top_lags = np.argsort(S)[::-1][:K]
        
        improved = False
        for k_idx in top_lags:
            lag = k_idx - (n - 1)
            m = abs(lag)

            for i in range(n - m):

                # (moving delta from j to i, i.e., h[i]+=delta, h[j]-=delta)

                if max_delta < 1e-10:

                # And h[i]+delta >= 0 (always true if h[i] >= 0 and delta >= 0)
                # h[j]-delta <= 1 (always true if h[j] <= 1 and delta >= 0)
                
                # Try moving mass from j to i (delta > 0: h[i] increases, h[j] decreases)
                # Or from i to j (delta < 0)
                
                for delta_frac in [0.1, 0.3, 0.5, 0.8]:
                    delta = max_delta * delta_frac

                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
                        break  # If this delta doesn't help, smaller ones might
                        # Actually, let's try the other direction too
                    break  # Don't try smaller deltas in same direction

                        h[j] -= delta

                    break
        
        if not improved:
            break
    
    return h, best_c

I need the D-vector trick to avoid recomputing the full correlation. Let me implement it properly.

The key insight: when I change h[i] by +δ and h[j] by -δ, the change in S(k) for ALL k is:
ΔS(k) = δ * (∂S(k)/∂h[i] - ∂S(k)/∂h[j])

So if I precompute G[i] = ∂S(k)/∂h[i] for all k (a vector of length 2n-1), then:
ΔS(k) = δ * (G[i][k] - G[j][k])

And the new C5 = max_k (S(k) + δ * (G[i][k] - G[j][k]))

To find the best δ, I need to minimize max_k (S(k) + δ * D[k]) where D[k] = G[i][k] - G[j][k].

This is a 1D convex optimization problem (minimum of max of linear functions). The optimal δ is at a breakpoint or at a boundary.

For practical purposes, I can:
1. Compute D = G[i] - G[j] (vector of length 2n-1)
2. Find the valid range of δ: [δ_min, δ_max]
3. Try a few δ values in that range and pick the best

The number of δ values to try: 5-10 should suffice since the function is piecewise linear with at most 2n-1 pieces.

Computing G[i] for all i and all k: this is the "Jacobian" of S with respect to h. It's a (2n-1) × n matrix. Computing it fully takes O(n²) (for each of the 2n-1 lags, computing the gradient takes O(n)).

But I don't need the full Jacobian. For the local search, I need G[i] and G[j] for specific pairs. I can compute them on the fly:

G[i][k] = ∂S(k)/∂h[i] for a given i and all k.

For a fixed i, ∂S(k)/∂h[i]:
- S(k) = sum_m h[m](1-h[m+k])
- h[i] appears when m=i (contribution: 1-h[i+k]) or when m+k=i (contribution: -h[i-k])
- So ∂S(k)/∂h[i] = (1-h[i+k]) * [i+k valid] - h[i-k] * [i-k valid]

For all k at once:
G[i][k] = (1-h[i+k]) * [0≤i+k<n] - h[i-k] * [0≤i-k<n]

This can be computed in O(n) for a given i (just look at h[i+k] and h[i-k] for all valid k).

So for a pair (i, j):
D[k] = G[i][k] - G[j][k]
     = [(1-h[i+k]) - h[i-k]] - [(1-h[j+k]) - h[j-k]]  (with validity conditions)
     = h[j+k] - h[i+k] - h[i-k] + h[j-k]  (where valid indices have the value, invalid have 0)

Actually, with the convention that h[m] = 0 for m < 0 or m ≥ n:
G[i][k] = (1-h[i+k]) * [0≤i+k<n] - h[i-k] * [0≤i-k<n]

So D[k] = G[i][k] - G[j][k]

This is O(n) to compute for each pair.

Then finding the best δ: I evaluate f(δ) = max_k (S[k] + δ * D[k]) at several points.

Total per pair: O(n) for D + O(n) for evaluation = O(n).
Total per sweep (n-m pairs for one lag): O(n * n) = O(n²).
For K lags: O(K * n²).
For n=500, K=5: 5 * 250,000 = 1.25M operations per sweep. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    else:
        grad[0:n-m] += -h[m:n]
        grad[m:n] += (1.0 - h[0:n-m])
    return grad

def smooth_max_grad(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w_sum = w.sum()
    w = w / w_sum
    if top_k is not None and top_k < len(w):
        top_idx = np.argsort(w)[::-1][:top_k]
        w_top = w[top_idx].copy()
        w_top = w_top / w_top.sum()
        active_pairs = list(zip(top_idx, w_top))
    else:
        active_pairs = [(k, w[k]) for k in range(2 * n - 1) if w[k] > 1e-10]
    grad = np.zeros(n)
    for k_idx, w_k in active_pairs:
        lag = k_idx - (n - 1)
        grad += w_k * lag_gradient(h, n, lag)
    return grad * dx

def compute_d_vector(h, n, i, j):
    """Compute D[k] = G_i[k] - G_j[k] for all k (lag values from -(n-1) to n-1)."""
    k_vals = np.arange(-(n - 1), n)
    ik = i + k_vals
    jk = j + k_vals
    ik_neg = i - k_vals
    jk_neg = j - k_vals
    
    mask_ik = (ik >= 0) & (ik < n)
    mask_jk = (jk >= 0) & (jk < n)
    mask_ikn = (ik_neg >= 0) & (ik_neg < n)
    mask_jkn = (jk_neg >= 0) & (jk_neg < n)
    
    Gi = np.zeros(2 * n - 1)
    Gj = np.zeros(2 * n - 1)
    Gi[mask_ik] = 1.0 - h[ik[mask_ik]]
    Gi[mask_ikn] -= h[ik_neg[mask_ikn]]
    Gj[mask_jk] = 1.0 - h[jk[mask_jk]]
    Gj[mask_jkn] -= h[jk_neg[mask_jkn]]
    
    return Gi - Gj

def dvector_local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    S = compute_all_lags(h, n)
    best_c = float(np.max(S))
    best_h = h.copy()
    
    delta_fracs = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        S = compute_all_lags(h, n)
        current_max = np.max(S)
        
        # Get top-K lags
        K = 7
        top_lags_idx = np.argsort(S)[::-1][:K]
        
        improved = False
        for k_idx in top_lags_idx:
            lag = k_idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            
            for i in range(n - m):
                j = i + m
                D = compute_d_vector(h, n, i, j)
                
                avail_pos = min(1.0 - h[i], h[j])
                avail_neg = min(h[i], 1.0 - h[j])
                
                if avail_pos < 1e-10 and avail_neg < 1e-10:
                    continue
                
                best_delta = 0.0
                best_val = current_max
                
                if avail_pos > 1e-10:
                    for d in delta_fracs:
                        delta = avail_pos * d
                        if delta < 1e-12:
                            continue
                        new_max = np.max(S + delta * D)
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_delta = delta
                
                if avail_neg > 1e-10:
                    for d in delta_fracs:
                        delta = avail_neg * d
                        if delta < 1e-12:
                            continue
                        new_max = np.max(S - delta * D)
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_delta = -delta

                    h[j] -= best_delta
                    S += best_delta * D
                    improved = True

            h = project_feasible(h, n)
            S = compute_all_lags(h, n)

def slsqp_polish(h, n, alpha, maxiter=2000, ftol=1e-15, top_m=5):
    dx = 2.0 / n
    
    def _sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def _sum_con_jac(x):
        return np.ones(n)
    
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def _obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
