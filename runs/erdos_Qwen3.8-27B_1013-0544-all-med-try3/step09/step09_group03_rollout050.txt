import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    threshold = 1e-13
    for idx in range(n2m1):
        if w[idx] < threshold:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, top_passes=2):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(30, 5.0), (100, 20.0), (300, 100.0),
                   (1000, 500.0), (3000, 2000.0),
                   (8000, 8000.0), (20000, 30000.0),
                   (50000, 100000.0)]
    for ci, (alpha, mu) in enumerate(configs):
        for pass_i in range(top_passes):
            if time.time() > deadline - 20:
                return h, best_c
            try:
                maxiter = 5000 if n <= 1000 else 2000
                maxfun = 50000 if n <= 1000 else 20000
                res = minimize(smooth_obj, h, jac=smooth_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': maxfun, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new.copy()
            except Exception:
                pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=8, mags=None, seed=99):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)

        mags = [0.02, 0.04, 0.06, 0.01, 0.08, 0.03, 0.05, 0.10]

    for t in range(n_trials):
        if time.time() > deadline - 25:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)

            h_new = project_feasible(res.x, n)

            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def structured_init(n, seed=0):
    rng = np.random.RandomState(seed)
    candidates = []
    # Alternating
    h1 = np.where(np.arange(n) % 2 == 0, 0.5, 0.5)
    candidates.append(h1.copy())
    # 2-on 2-off
    h2 = np.zeros(n)
    for i in range(n):
        if i % 4 < 2:
            h2[i] = 1.0
    candidates.append(project_feasible(h2, n))
    # Random binary
    h3 = (rng.rand(n) > 0.5).astype(float)
    candidates.append(project_feasible(h3, n))
    # Smooth random
    h4 = np.tanh(3.0 * (rng.rand(n) - 0.5))
    h4 = project_feasible(np.clip(h4, 0, 1) * 0.8 + 0.1, n)
    candidates.append(h4)
    # Block structure
    h5 = np.zeros(n)
    block_len = max(4, n // 15)
    for i in range(0, n, block_len * 2):
        h5[i:i+block_len] = 1.0
    candidates.append(project_feasible(h5, n))
    # Smooth blocks
    h6 = np.zeros(n)
    nblocks = 5
    bs = n // nblocks
    for i in range(nblocks * 2):
        start = i * bs // 2
        end = start + bs // 2
        if i % 2 == 0:
            h6[start:min(end, n)] = 1.0
    candidates.append(project_feasible(h6, n))
    best_c = 1e9
    best_h = candidates[0]
    for hc in candidates:
        hc = project_feasible(hc, n)
        c = compute_c5(hc, n)
        if c < best_c:
            best_c = c
            best_h = hc.copy()
    return best_h

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(80):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        dx = 2.0 / n
        if k_worst >= 0:
            affected = range(0, n - k_worst)
            affected2 = range(k_worst, n)
        else:
            kk = -k_worst
            affected = range(kk, n)
            affected2 = range(0, n - kk)
        all_affected = set(list(affected) + list(affected2))
        found = False
        for i in all_affected:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = best_h[i]
            if found:
                h = best_h.copy()
        if not found and round_num > 15:
            break
    return best_h, best_c

def coarsen_optimize(h_fine, n_fine, n_coarse, deadline, configs_coarse):

def binarize_and_refine(h, n, deadline):
    h_bin = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_bin.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            for i in order:
                if h_bin[i] < 1.0:
                    h_bin[i] = 1.0
                    diff -= 1.0
                    if abs(diff) < 1e-12:
                        break
        else:
            order = np.argsort(h)
            for i in order:
                if h_bin[i] > 0.0:
                    h_bin[i] = 0.0
                    diff += 1.0
                    if abs(diff) < 1e-12:
                        break
    h_bin = project_feasible(h_bin, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_bin, jac=smooth_grad,
                      args=(n, 5000, 5000.0), method='L-BFGS-B',
