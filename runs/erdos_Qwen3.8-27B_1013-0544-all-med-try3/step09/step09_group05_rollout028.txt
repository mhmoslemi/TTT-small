            if time.time() > deadline - 100:
                break
            try:
                res = minimize(obj_active, h, jac=grad_active,
                             args=(n, active, alpha), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-15,
                                      'maxfun': 100000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    
    # Phase 3: Very high alpha on top shifts (~150s)
    if time.time() < deadline - 60:
        c_vec = compute_c5_vec(h, n)
        top = np.argsort(-c_vec)[:6]
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 60:
                break
            try:
                res = minimize(obj_active, h, jac=grad_active,
                             args=(n, top, alpha), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 20000, 'ftol': 1e-18, 'gtol': 1e-16,
                                      'maxfun': 200000, 'maxls': 100})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    
    # Phase 4: Perturbation restarts (~80s)
    if time.time() < deadline - 40:
        h, best_c = try_perturb_restart(h, n, best_c, deadline - 40, seed)
    
    # Phase 5: Another round of active-set after perturbation
    if time.time() < deadline - 30:
        c_vec = compute_c5_vec(h, n)
        max_val = float(c_vec.max())
        tol = max(1e-5, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 30:
                break
            try:
                res = minimize(obj_active, h, jac=grad_active,
                             args=(n, active, alpha), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-15,
                                      'maxfun': 100000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    
    # Phase 6: Local search (remaining time)
    if time.time() < deadline - 3:
        h, best_c = local_search(h, n, deadline, max_rounds=30)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Final: project and evaluate
    best_h = project_feasible(best_h, n)
    final_c = compute_c5(best_h, n)
    
    return best_h.tolist(), final_c, n
