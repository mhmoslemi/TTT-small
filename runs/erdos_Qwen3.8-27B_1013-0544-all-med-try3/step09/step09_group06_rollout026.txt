```python
        if g_norm < 1e-12:
            break

        if prev_g is not None:
            s = h_curr - prev_h
            y = g - prev_g
            s_norm = np.dot(s, s)
            y_norm = np.dot(y, y)
            sy = np.dot(s, y)
            if abs(sy) > 1e-12:
                lr_y = s_norm / sy  # BB1
                lr_s = sy / y_norm  # BB2
                lr = 0.5 * (lr_y + lr_s)
                lr = max(0.001, min(lr, 1.0))
        
        prev_g = g.copy()
        prev_h = h_curr.copy()
        
        v = momentum * v + (1 - momentum) * g
        h_new = project_feasible(h_curr - lr * v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience % 100 == 0:
                lr *= 0.8
            
        h_curr = h_new
        
        if patience > 2000:
            break
    
    return best_h, best_c
```

```python
import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    
    if top_m is not None:
        # Only optimize over top-m lags
        top_idx = np.argsort(c)[-top_m:]
        def obj(x):
            cc = np.correlate(x, 1 - x, mode='full') * dx
            active = cc[top_idx]
            m = active.max()
            shifted = alpha * (active - m)
            mx = shifted.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
            return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    else:
        def obj(x):
            cc = np.correlate(x, 1 - x, mode='full') * dx
            m = cc.max()
            shifted = alpha * (cc - m)
            mx = shifted.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
            return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

        return h.copy()
```

I need a faster way to evaluate the effect of a single move. When I change h[i] by -delta and h[j] by +delta (where j = i + m for lag m), the change in c_lag for a specific lag k is:

delta_c_k = dx * [effect on terms involving h[i] and h[j]]

For the lag k = m (the one we're targeting):
c_m = sum_{i=0}^{n-1-m} h[i+m](1-h[i]) dx

The terms involving h[i] are: h[i+m](1-h[i]) [as the (1-h) part] and h[i](1-h[i-m]) [as the h part, if i >= m]

Instead of computing the full c5 after each move, I can compute the change in c_k for the relevant lags analytically.

For a move: h[i] -= delta, h[j] += delta (where j = i + m for lag m):

The change in c_k = sum_i h[i](1-h[i+k]) dx is:
For each term h[i](1-h[i+k]):
- If i = i (the decreased one): change = -delta * (1 - h[i+k])
- If i + k = i (i.e., k = 0): already counted
- If i + k = j (i.e., k = j - i = m): the term h[j-k](1-h[j]) = h[i](1-h[j]) changes by -delta * (1-h[j]) + h[i] * delta...

```python
def efficient_local_search(h, n, deadline, top_k=5):
    """More efficient local search using gradient information."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    max_rounds = 1000
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        
        # Compute gradient of the max (or smooth max of top-k)
        if top_k == 1:
            k_star = top_idx[0]
            lag = k_star - (n - 1)
            if lag == 0:
                break
            g = _lag_gradient(h, n, lag)
        else:
            # Weighted gradient using softmax of top-k values
            top_vals = c_all[top_idx]
            alpha_ls = 100.0
            m_top = top_vals.max()
            w_top = np.exp(alpha_ls * (top_vals - m_top))
            w_top = w_top / w_top.sum()
            g = np.zeros(n)
            for t, k_idx in enumerate(top_idx):
                lag = k_idx - (n - 1)
                g += w_top[t] * _lag_gradient(h, n, lag)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Line search: try a few step sizes
        best_step = 0
        best_new_c = best_c
        for lr in [0.05, 0.03, 0.01, 0.005, 0.002]:
            h_trial = project_feasible(h - lr * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_new_c:
                best_new_c = c_trial
                best_step = lr
        
        if best_step > 0 and best_new_c < best_c - 1e-14:
            h = project_feasible(h - best_step * g, n)
            best_c = best_new_c
        else:
            break
    
    return h, best_c
```

I need a faster way to evaluate swaps. For a binary h, the change in c_k due to swapping h[i] and h[j] (where h[i]=1, h[j]=0) is:

delta_c_k = dx * [(new contribution) - (old contribution)]

The old contribution to c_k from position i is: h[i] * (1-h[i+k]) = 1 * (1-h[i+k])
The old contribution from position j is: h[j] * (1-h[j+k]) = 0 * (1-h[j+k]) = 0
The old contribution from position i-k (in the h part): h[i-k] * (1-h[i]) = h[i-k] * 0 = 0 (since h[i]=1)
The old contribution from position j-k: h[j-k] * (1-h[j]) = h[j-k] * 1

After swap (h[i]=0, h[j]=1):
New from i: 0 * (1-h[i+k]) = 0
New from j: 1 * (1-h[j+k]) = 1-h[j+k]
New from i-k: h[i-k] * (1-0) = h[i-k]
New from j-k: h[j-k] * (1-1) = 0

delta_c_k ≈ dx * [h[i-k] + (1-h[j+k]) - h[i+k] - h[j-k]]
