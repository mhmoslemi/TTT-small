def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of the correlation at a specific lag."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:

        # d(c)/d(h[j]) for j in [0, n-1-lag]: -h[j+lag]

        grad[0:n-lag] += -h[lag:n]
    else:
        m = -lag
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max(c, alpha):

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
