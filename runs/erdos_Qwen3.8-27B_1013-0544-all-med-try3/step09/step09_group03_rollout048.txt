        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_phase(h, n, deadline, alpha_list, mu_list, maxiter_per=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i in range(len(alpha_list)):
        if time.time() > deadline - 20:
            break
        alpha = alpha_list[i]
        mu = mu_list[i]
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxiter_per * 10, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_sa(h_start, n, deadline, max_iters=40000, seed=999):
    rng = np.random.RandomState(seed)
    h = np.where(h_start > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    # Fix sum to be exactly target
    attempts = 0
    while abs(h.sum() - target) > 1e-10 and attempts < 10000:
        if h.sum() > target:
            ones = np.where(h == 1)[0]
            if len(ones) == 0:
                break
            idx = ones[rng.randint(len(ones))]
            h[idx] = 0.0
        else:
            zeros = np.where(h == 0)[0]
            if len(zeros) == 0:
                break
            idx = zeros[rng.randint(len(zeros))]
            h[idx] = 1.0
        attempts += 1
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = 0.008
    T_min = 1e-9
    cool_rate = (T_min / T) ** (1.0 / max(max_iters, 1))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = max(T_min, T * cool_rate)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def binary_2exchange(h_bin, n, deadline, max_rounds=500, seed=42):
    """Systematic 2-exchange local search on binary pattern."""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    target = n / 2.0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_trials = 2000 + 100 * rnd
        for t in range(n_trials):
            if time.time() > deadline - 5:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h, n)
            if new_c < cur_c - 1e-14:
                cur_c = new_c
                improved = True
                if new_c < best_c:
                    best_c = new_c
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            # Try structured moves: flip consecutive pairs
            done = False
            for i in range(n - 1):
                if time.time() > deadline - 5:
                    break
                if h[i] == 1.0 and h[i+1] == 0.0:
                    for j in range(i + 2, n):
                        if h[j] == 0.0 and h[j+1] == 1.0 if j+1 < n else False:
                            break
                        if j < n and h[j] == 0.0:
                            # Try moving the 1 from i to j
                            h[i] = 0.0
                            h[j] = 1.0
                            new_c = compute_c5(h, n)
                            if new_c < cur_c - 1e-14:
                                cur_c = new_c
                                improved = True
                                if new_c < best_c:
                                    best_c = new_c
                            else:
                                h[i] = 1.0
                                h[j] = 0.0
            if not improved:
                break
    return h, best_c

def perturbation_search(h, n, deadline, n_trials=12, seed=777):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mags = [0.015, 0.03, 0.05, 0.01, 0.07, 0.02, 0.04, 0.09, 0.025, 0.06, 0.12, 0.035]
    for t in range(n_trials):
        if time.time() > deadline - 25:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        
        # Quick L-BFGS
        try:
            res = minimize(smooth_max_obj, h_p, jac=smooth_max_grad,
                          args=(n, 2000, 3000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 2000, 'ftol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
        
        # Also try from binary version
        if time.time() < deadline - 30:
            h_bin = np.where(h_p > 0.5, 1.0, 0.0).astype(float)
            if abs(h_bin.sum() - n/2.0) < 1e-8:
                try:
                    res2 = minimize(smooth_max_obj, h_bin, jac=smooth_max_grad,
                                   args=(n, 800, 1500.0), method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n,
                                   options={'maxiter': 1500, 'ftol': 1e-18, 'maxls': 50})
                    h_new2 = project_feasible(res2.x, n)
                    c_new2 = compute_c5(h_new2, n)
                    if c_new2 < best_c:
                        best_c = c_new2
                        best_h = h_new2.copy()
                except Exception:
                    pass
    return best_h, best_c

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(80):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        found = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005,
                         0.005, -0.005, 0.002, -0.002]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = best_h[i]
            if found:
                h = best_h.copy()
                break
        if not found and round_num > 20:
            # Try paired moves
            c_vec = compute_c5_vec(h, n)
            top3 = np.argsort(c_vec)[-3:]
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i+1, n):
                    if time.time() > deadline:
                        return best_h, best_c
                    for di in [0.002, -0.002]:
                        for dj in [-di]:
                            nv_i = h[i] + di
                            nv_j = h[j] + dj
                            if nv_i < 0 or nv_i > 1 or nv_j < 0 or nv_j > 1:
                                continue
                            h[i] = nv_i
                            h[j] = nv_j
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                di = None
                                break
                            h[i] = best_h[i]
                            h[j] = best_h[j]
                    if di is None:
                        break
                if di is None:
                    break
            if di is None:
                h = best_h.copy()
            elif round_num > 40:
                break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init(n, pattern_type=0, seed=42):
    """Generate structured initial patterns."""
    rng = np.random.RandomState(seed)
    h = np.empty(n)
    if pattern_type == 0:
        # Alternating blocks of size 3: 1,1,1,0,0,0
        block = np.array([1,1,1,0,0,0], dtype=float)
        for i in range(n):
            h[i] = block[i % 6]
    elif pattern_type == 1:
        # Sinusoidal + noise
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x / 0.8 + rng.rand() * 6.28)
    elif pattern_type == 2:
        # Random with slow variation
        n_coarse = 20
        coarse = 0.5 + 0.4 * (rng.rand(n_coarse) - 0.5)
        h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), coarse)
        h += 0.1 * rng.rand(n)
    elif pattern_type == 3:
        # Triangular wave
        x = np.linspace(0, 2, n)
        period = 0.6
        frac = (x / period) % 1.0
        h = np.where(frac < 0.5, 1.0, 0.0)
    else:
        h = 0.5 + 0.3 * rng.rand(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick warmup L-BFGS at base resolution
    # Use a smoother starting alpha for better global optimization
    t1 = min(t0 + 120, deadline - 600)
    if t1 > time.time() + 15:
        best_h, best_c = run_lbfgs_phase(best_h, n_base, t1,
            alpha_list=[5, 20, 80, 300, 1000],
            mu_list=[1.0, 5.0, 50.0, 200.0, 500.0],
            maxiter_per=4000)
    
    # Phase 2: SLSQP polish
    t2 = min(t0 + 200, deadline - 500)
    if t2 > time.time() + 15:
        h2, c2 = slqp_polish(best_h, n_base, t2, alpha=2000000, maxiter=4000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Binary simulated annealing (explores different basin)
    t3 = min(t0 + 350, deadline - 400)
    if t3 > time.time() + 20:
        h_bin_sa, c_bin_sa = binary_sa(best_h, n_base, t3, max_iters=45000, seed=seed % 10000)
        # Optimize from binary result
        if t3 - 10 > time.time() + 15:
            h_after_bin, c_after_bin = run_lbfgs_phase(h_bin_sa, n_base, t3,
                alpha_list=[50, 200, 1000, 5000],
                mu_list=[20.0, 100.0, 500.0, 2000.0],
                maxiter_per=2000)
            if c_after_bin < best_c:
                best_c = c_after_bin
                best_h = h_after_bin
        elif c_bin_sa < best_c:
            best_c = c_bin_sa
            best_h = h_bin_sa
    
    # Phase 4: Perturbation search for basin hopping
    t4 = min(t0 + 500, deadline - 300)
    if t4 > time.time() + 20:
        best_h, best_c = perturbation_search(best_h, n_base, t4, n_trials=10, seed=seed+1)
    
    # Phase 5: SLSQP after perturbation
    t5 = min(t0 + 560, deadline - 250)
    if t5 > time.time() + 10:
        h5, c5 = slqp_polish(best_h, n_base, t5, alpha=4000000, maxiter=3000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Upsample to n=1000 and optimize
    n1 = 1000
    if time.time() < deadline - 250:
        h1000 = interpolate_to(best_h, len(best_h), n1)
        t6 = min(time.time() + 100, deadline - 140)
        if t6 > time.time() + 15:
            h1000, c1000 = run_lbfgs_phase(h1000, n1, t6,
                alpha_list=[50, 200, 800, 3000, 8000],
                mu_list=[20.0, 100.0, 500.0, 2000.0, 8000.0],
                maxiter_per=2000)
            if c1000 < best_c:
                best_c = c1000
                best_h = h1000
        
        t6b = min(time.time() + 60, deadline - 70)
        if t6b > time.time() + 10:
            h6b, c6b = slqp_polish(best_h, len(best_h), t6b, alpha=6000000, maxiter=2500)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Binary SA at higher resolution (if time permits)
    if time.time() < deadline - 150:
        n_hi = len(best_h)
        t7 = min(time.time() + 60, deadline - 80)
        if t7 > time.time() + 20:
            h_bin2, c_bin2 = binary_sa(best_h, n_hi, t7, max_iters=20000, seed=seed+100)
            if t7 - 10 > time.time() + 10:
                h_bin2, c_bin2 = run_lbfgs_phase(h_bin2, n_hi, t7,
                    alpha_list=[200, 1000, 5000],
                    mu_list=[100.0, 500.0, 3000.0],
                    maxiter_per=1000)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = h_bin2
    
    # Phase 8: Try a structured alternative at base resolution
    if time.time() < deadline - 100:
        n_struct = n_base
        t8 = min(time.time() + 50, deadline - 40)
        for pt in range(3):
            if time.time() > deadline - 30:
                break
            h_struct = structured_init(n_struct, pattern_type=pt, seed=seed + pt * 111)
            h_s, c_s = run_lbfgs_phase(h_struct, n_struct, t8,
                alpha_list=[10, 100, 500, 2000],
                mu_list=[5.0, 50.0, 200.0, 1000.0],
                maxiter_per=1500)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 9: Final targeted refinement
    if time.time() < deadline - 15:
        n_fin = len(best_h)
        best_h, best_c = targeted_refine(best_h, n_fin, deadline - 5)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
