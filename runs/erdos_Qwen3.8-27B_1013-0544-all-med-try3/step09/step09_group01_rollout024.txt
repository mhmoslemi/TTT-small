import cvxpy as cp

def slp_iteration(h, n, n_active=20, solver=None):
    c_vec = compute_c5_vec(h, n)
    # Get top n_active lags
    idx = np.argsort(-c_vec)[:n_active]
    
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    for lag_idx in idx:
        k = lag_idx - (n - 1)
        # C(k) = dx * sum_i h[i] * (1 - h[i+k]) for valid i
        # grad C(k) / grad h[j] = dx * (sum over valid i of d/dh[j][h[i]*(1-h[i+k])])
        # = dx * ((1-h[j+k]) if j+k valid, as the "first" factor
        #         + -(h[j-k]) if j-k valid, as the "second" factor)
        # More carefully:
        # C(k) = dx * sum_{i: 0≤i<n, 0≤i+k<n} h[i] * (1 - h[i+k])
        # dC(k)/dh[j] = dx * [(1-h[j+k]) * 1_{j+k < n} - h[j-k] * 1_{j-k >= 0}]
        # (the first term from h[j] being the "first" factor with h[j+k] as second)
        # (the second term from h[j] being the "second" factor, i.e., i+k=j so i=j-k)
        
        grad_k = np.zeros(n)
        if k >= 0:
            # h[i] * (1-h[i+k]), valid i: 0 ≤ i < n, 0 ≤ i+k < n → 0 ≤ i < n-k
            # d/dh[j]: (1-h[j+k]) for j < n-k (first factor)
            #          -h[j-k] for j-k ≥ 0 and j-k < n-k, i.e., k ≤ j < 2k... wait
            # Let me redo: term i is h[i]*(1-h[i+k])
            # d/dh[j] of term i: if i=j, gives (1-h[j+k]); if i+k=j (i=j-k), gives -h[j-k]
            # Valid i range: 0 ≤ i < n-k
            # So j (as first factor): valid if 0 ≤ j < n-k
            # j (as second factor, j=i+k): valid if 0 ≤ j-k < n-k, i.e., k ≤ j < n
            for j in range(n - k):
                grad_k[j] += (1.0 - h[j + k])
            for j in range(k, n):
                grad_k[j] -= h[j - k]
            grad_k *= dx
        else:
            kk = -k
            # h[i] * (1-h[i-kk]), valid i: kk ≤ i < n
            # d/dh[j]: (1-h[j-kk]) if j-kk ≥ 0 and j-kk < n (i.e., kk ≤ j < n) as first factor
            #          -h[j+kk] if j+kk ≥ kk and j+kk < n (i.e., 0 ≤ j < n-kk) as second factor
            for j in range(kk, n):
                grad_k[j] += (1.0 - h[j - kk])
            for j in range(n - kk):
                grad_k[j] -= h[j + kk]
            grad_k *= dx
        
        # Constraint: grad_k · h_new - t ≤ grad_k · h - c_vec[lag_idx]
        constraints.append(grad_k @ h_var - t_var <= grad_k @ h - c_vec[lag_idx])
    
    problem = cp.Problem(cp.Minimize(t_var), constraints)
    problem.solve(solver=solver)
    
    if h_var.value is not None:
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5_lag(h, n, k):
    """Gradient of C(k) w.r.t. h, where k is the shift (can be negative)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:n])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad * dx

    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()

    n_var = n + 1  # h (n vars) + t (1 var)

    # s.t. grad_k @ h - t <= grad_k @ h_old - c_vec[k]  for active k
    #      0 <= h <= 1
    #      sum(h) = n/2
    #      t free (unbounded)
    
    # Build constraint matrix
    # For linprog: c @ x (objective), A_ub @ x <= b_ub, A_eq @ x == b_eq
    # x = [h(0), ..., h(n-1), t]

    obj[n] = 1.0  # minimize t
    
    A_list = []
    b_list = []
    
    for idx in top_idx:
        k = idx - (n - 1)
        g = grad_c5_lag(h, n, k)
        # g @ h_new - t <= g @ h - c_vec[idx]
        # g @ h_new - t <= g @ h - c_vec[idx]
        row = np.zeros(n_var)
        row[:n] = g
        row[n] = -1.0
        A_list.append(row)
        b_list.append(g @ h - c_vec[idx])
    
    # Box constraints: 0 <= h_i <= 1 → handled by bounds
    # Sum constraint: sum(h) = n/2
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]  # h_i in [0,1], t free
    
    if A_list:
        A_ub = np.array(A_list)
        b_ub = np.array(b_list)
    else:
        A_ub = np.zeros((0, n_var))
        b_ub = np.zeros(0)
    
    res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=solver)
    
    if res.success:
        h_new = res.x[:n]
        h_new = project_feasible(h_new, n)
        return h_new, c_vec
    return h, c_vec

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def grad_c5_lag(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] += (1.0 - h[k:n])
            grad[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:] += (1.0 - h[:nv])
            grad[:nv] -= h[kk:]
    return grad * dx


def slp_optimize(h_init, n, deadline, max_iters=100000, n_active=100):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_var = n + 1
    num_lags = 2 * n - 1
    lam = 1.0
    no_improve_count = 0

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break

        c_vec = compute_c5_vec(h, n)
        current_max = c_vec.max()

        # Select active lags (top n_active)
        if n_active < num_lags:
            top_idx = np.argsort(-c_vec)[:n_active]
        else:
            top_idx = np.arange(num_lags)

        # Build LP
        obj = np.zeros(n_var)
        obj[n] = 1.0

        n_cons = len(top_idx)
        A_ub = np.zeros((n_cons, n_var))
        b_ub = np.zeros(n_cons)

        for ii, idx in enumerate(top_idx):
            k = idx - (n - 1)
            g = grad_c5_lag(h, n, k)
            A_ub[ii, :n] = g
            A_ub[ii, n] = -1.0
            b_ub[ii] = g @ h - c_vec[idx]

        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]

        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            try:
                res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='interior-point')
            except Exception:
                continue

        if not res.success or res.x is None:
            continue

        h_lp = res.x[:n]
        h_lp = np.clip(h_lp, 0.0, 1.0)

        # Damped step
        h_try = (1.0 - lam) * h + lam * h_lp
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)

        if c_try < best_c - 1e-16:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
            no_improve_count = 0
            lam = min(1.0, lam * 1.05)
        elif c_try < current_max + 1e-14:
            h = h_try
            no_improve_count += 1
            if no_improve_count > 20:
                lam = max(0.3, lam * 0.9)
                no_improve_count = 0
        else:
            no_improve_count += 1
            lam = max(0.2, lam * 0.95)
            if no_improve_count > 50:
                # Try a smaller step
                h_try2 = (1.0 - 0.5 * lam) * h + 0.5 * lam * h_lp
                h_try2 = project_feasible(h_try2, n)
                c_try2 = compute_c5(h_try2, n)
                if c_try2 < c_try:
                    h = h_try2
                    if c_try2 < best_c - 1e-16:
                        best_c = c_try2
                        best_h = h_try2.copy()
                no_improve_count = 0
                lam = max(0.1, lam * 0.9)

    return best_h, best_c


def coord_descent(h_init, n, deadline, max_sweeps=200):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    num_lags = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() > deadline - 3:
            break

        improved = False
        # Compute current C values
        c_vec = compute_c5_vec(h, n)

        for i in range(n):
            if time.time() > deadline - 3:
                break
            # Pair with neighbors and a distant point
            for j in [i + 1, i - 1, (n - 1) - i, i + 7 if i + 7 < n else n - 1]:
                if j < 0 or j >= n or j == i:
                    continue

                # Compute a_k = g_k[i] - g_k[j] for all k
                # g_k[i] = dx * [(1-h[i+k]) * 1{k>=0, i+k<n} - h[i-k] * 1{k<0, i-k>=0}]
                # This gets complex; use direct computation
                # Instead: evaluate C at h and h with small perturbation

                # Try moving mass from j to i and i to j
                for direction in [1, -1]:
                    if direction == 1:
                        delta_max = min(1.0 - h[i], h[j])
                    else:
                        delta_max = min(h[i], 1.0 - h[j])

                    if delta_max < 1e-15:
                        continue

                    # Binary search for optimal delta
                    lo = 0.0
                    hi = delta_max
                    best_delta = 0.0
                    best_val = best_c

                    for _ in range(15):
                        mid = 0.5 * (lo + hi)
                        if direction == 1:
                            h[i] += mid
                            h[j] -= mid
                        else:
                            h[i] -= mid
                            h[j] += mid

                        c_test = compute_c5(h, n)
                        if c_test < best_val:
                            best_val = c_test
                            best_delta = mid
                            lo = mid
                        else:
                            hi = mid

                        h[i] -= direction * mid
                        h[j] += direction * mid

                    if best_delta > 1e-15 and best_val < best_c - 1e-16:
                        if direction == 1:
                            h[i] += best_delta
                            h[j] -= best_delta
                        else:
                            h[i] -= best_delta
                            h[j] += best_delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True

        if not improved:
            break

    return best_h, best_c


def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)


def pair_search(h_init, n, deadline, radius=20):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 3:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline - 3:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        h = best_h.copy()

    return best_h, best_c


def soft_max_lbfgs(h_init, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 500, 5000, 50000]
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def sm_obj(x, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += 0.005 * (x.sum() - target) ** 2
        return float(val)

    def sm_grad(x, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx_ = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx_
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - x[k:n])
                grad[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
        grad += 0.01 * (x.sum() - target)
        return grad

    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        maxit = max(300, int((deadline - time.time()) * 2))
        try:
            res = minimize(sm_obj, h, jac=sm_grad, args=(alpha,), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-18,
                                                   'gtol': 1e-14, 'maxfun': maxit * 6})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            h = project_feasible(h_new, n)
        except Exception:
            pass

    return best_h, best_c
