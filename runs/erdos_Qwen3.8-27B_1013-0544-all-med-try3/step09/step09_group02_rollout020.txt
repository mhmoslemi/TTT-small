    # Find d that minimizes max_j g_j^T d
    # subject to ||d||_inf <= rho, sum(d) = 0 (to preserve feasibility)
    # This is an LP
    m = len(active_grads)
    # Variables: d_0, ..., d_{n-1}, t
    # Minimize t
    # Subject to: g_j^T d <= t for j=1,...,m
    #             sum(d) = 0
    #             |d_i| <= rho for i=0,...,n-1
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((m + 2*n, n + 1))
    b_ub = np.zeros(m + 2*n)
    
    for j in range(m):
        A_ub[j, 0:n] = active_grads[j]
        A_ub[j, n] = -1.0
        # b_ub[j] = 0 (we want g_j^T d - t <= 0)
    
    for i in range(n):
        A_ub[m + i, i] = 1.0
        b_ub[m + i] = rho
    for i in range(n):
        A_ub[m + n + i, i] = -1.0
        b_ub[m + n + i] = rho
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[0:n], res.x[n]
    return None, None
