def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:

    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

        if time.time() > deadline:
            break
        t += 1
        g = smooth_max_grad(h, n, alpha)

        h = h_new

        x_sum += 1.0

    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(-h)

    h_bin[idx[:n//2]] = 1.0
    # Handle odd n

        h_bin[idx[:n//2]] = 1.0
        h_bin[idx[n//2]] = 0.5  # Actually this might not preserve sum exactly
    # Re-project to ensure exact constraint
    return project_feasible(h_bin, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

        m_t = beta1 * m_t + (1.0 - beta1) * g
        v_t = beta2 * v_t + (1.0 - beta2) * (g * g)

        h9, c9 = slqp_polish(best_h, n_cur, t9, 2000000, 5000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Very fine pairwise local search (180s)
    t10 = min(t0 + 940, deadline - 20)
    if t10 > time.time() + 10:
        n_cur = len(best_h)
        best_h, best_c = fine_pairwise_search(
            best_h, n_cur, t10, radius=40,
            steps_dec=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
            n_steps=5)

    # Phase 11: Final micro-SQP (60s)
    t11 = min(t0 + 1000, deadline - 10)
    n_cur = len(best_h)
    if t11 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_cur, t11, 300, 0.02)

    # Phase 12: Ultimate SLSQP (40s)
    t12 = deadline
    n_cur = len(best_h)
    if t12 > time.time() + 5:
        h12, c12 = slqp_polish(best_h, n_cur, t12, 5000000, 3000)
        if c12 < best_c:
            best_c = c12
            best_h = h12

    # Final feasibility and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
