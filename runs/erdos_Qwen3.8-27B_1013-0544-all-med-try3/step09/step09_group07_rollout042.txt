import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):

def smooth_max_grad(h, n, alpha):

    grad = np.zeros(n)

        else:

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e6):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 80,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def remez_equalize(h, n, deadline, K=5, max_rounds=20000):
    """Remez-like algorithm to equalize the top K lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        # Find top K lags
        top_K = np.argsort(c_all)[-K:][::-1]  # descending
        c_top = c_all[top_K]
        c_max = c_top[0]
        c_mean = c_top.mean()
        
        if c_max < best_c - 1e-12:
            best_c = float(c_max)
            best_h = h.copy()
        
        # Check convergence
        spread = c_max - c_mean
        if spread < 1e-8:
            # Lags are well equalized, try to reduce the overall max
            # Just do a gradient step to reduce the max
            pass
        
        # Compute gradients of the top K lags
        # We want to find a direction that reduces c_max (the worst lag)
        # while not increasing the others
        # Strategy: gradient descent on sum of squared deviations from mean
        # + gradient descent on the mean itself
        
        # Simpler: just do gradient descent on the max (using smooth max with high alpha)
        alpha_remez = 100000
        grad = smooth_max_grad(h, n, alpha_remez)
        
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)
        
        # Line search
        best_step = 0
        best_obj = c_max
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_try = h - step * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_all_lags(h_try, n)
            max_try = c_try.max()
            if max_try < best_obj:
                best_obj = float(max_try)
                best_step = step
        
        if best_step == 0:
            break
        
        h = project_feasible(h - best_step * grad, n)
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = float(c_now)
            best_h = h.copy()
    
    return best_h, best_c

def binary_round(h, n):
    """Round h to the nearest binary with exactly n/2 ones."""
    h_sorted = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted[:n//2]] = 1.0
    return h_bin

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):

    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag

    return grad * dx

        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
