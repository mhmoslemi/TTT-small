def slsqqpolish(h, n, alpha, maxiter=2000, top_m=5):

    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        shifted = shifted - shifted.max()
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        shifted = shifted - shifted.max()
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            if wi < 1e-10:
                continue
            lag = int(idx) - (n - 1)
            grad += wi * lag_gradient(x, n, lag)
        return grad * dx
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)

        return grad

        grad[m:n] += (-h[:n-m])
    else:
        grad[m:n] += (1.0 - h[:n-m])
        grad[:n-m] += (-h[m:n])
    return grad
