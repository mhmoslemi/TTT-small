def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sa_optimize(h, n, deadline, rng, T_init=0.005, T_final=0.0001, 
                n_restarts=5, moves_per_restart=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    current_c = best_c
    
    total_budget = moves_per_restart * n_restarts
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        T = T_init * (T_final / T_init) ** (restart / max(1, n_restarts - 1))
        
        n_iter = moves_per_restart
        for it in range(n_iter):
            if time.time() > deadline - 1:
                break
            
            # Choose move type
            r = rng.random()
            if r < 0.4:
                # Gradient-guided 2-point move
                c_vec = np.correlate(current_h, 1 - current_h, mode='full')
                max_val = c_vec.max()
                threshold = max_val - max(1e-6, 0.005 * max_val)
                active = np.where(c_vec >= threshold)[0]
                
                # Compute combined gradient for active shifts
                dx = 2.0 / n
                grad = np.zeros(n)
                for idx in active:
                    k = idx - (n - 1)
                    if k >= 0:
                        nv = n - k
                        grad[0:nv] += (1.0 - current_h[k:n]) * dx
                        grad[k:n] += (-current_h[0:nv]) * dx
                    else:
                        kk = -k
                        nv = n - kk
                        grad[kk:n] += (1.0 - current_h[0:nv]) * dx
                        grad[0:nv] += (-current_h[kk:n]) * dx
                
                # Pick i from high gradient, j from low gradient
                g_abs = np.abs(grad)
                # Weighted random selection
                w_i = g_abs ** 2 + 1e-10
                w_i /= w_i.sum()
                w_j = 1.0 / (g_abs + 1e-10)
                w_j /= w_j.sum()
                
                i = rng.choice(n, p=w_i)
                j = rng.choice(n, p=w_j)
                if i == j:
                    j = (j + 1) % n
                if i == j:
                    continue
                
                # Determine direction
                if grad[i] > grad[j]:
                    delta = rng.uniform(0.005, 0.05)
                else:
                    delta = -rng.uniform(0.005, 0.05)
                
                if current_h[i] + delta < 0 or current_h[i] + delta > 1:
                    delta = -delta
                if current_h[j] - delta < 0 or current_h[j] - delta > 1:
                    continue
                
                current_h[i] += delta
                current_h[j] -= delta
                
            elif r < 0.7:

                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:

                delta = rng.uniform(-0.05, 0.05)

def smooth_max_obj(x, n, alpha, mu, n_over_2):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + math.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + math.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n_over_2) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu, n_over_2):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n_over_2)
    return grad

def lbfgs_optimize(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    n_over_2 = n / 2.0
    
    configs = [(200, 10.0), (1000, 50.0), (5000, 500.0), 
               (20000, 5000.0), (100000, 50000.0)]
    
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, cur, jac=smooth_max_grad,
                           args=(n, alpha, mu, n_over_2),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxls': 50})
            cur = project_feasible(res.x, n)
            c = compute_c5(cur, n)
            if c < best_c:
                best_c = c
        except Exception:
            pass
    
    return cur, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

def compute_grad_for_shifts(h, n, active):
    dx = 2.0 / n
    nm1 = n - 1

        k = int(idx) - nm1
        if k >= 0:
            nv = n - k

            grad[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + math.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + math.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    nm1 = n - 1
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        k = mm - nm1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline):

    best_c = compute_c5(h, n)
    cur = h.copy()
    bounds = [(0.0, 1.0)] * n

                           args=(n, alpha, mu),

                                    'gtol': 1e-14, 'maxls': 50})
            cur = project_feasible(res.x, n)
            c = compute_c5(cur, n)
            if c < best_c:

        except Exception:
            pass
    
    return cur, best_c

                if cur_h[j] - delta < 0 or cur_h[j] - delta > 1:
                    did_skip = True

        elif r < 0.75:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                did_skip = True
            else:
                delta = rng.choice([-1.0, 1.0]) * rng.uniform(0.005, 0.04)
                if cur_h[i] + delta < 0 or cur_h[i] + delta > 1:
                    delta = -delta
                if cur_h[j] - delta < 0 or cur_h[j] - delta > 1:
                    did_skip = True

        elif r < 0.9:
            i = int(rng.integers(0, n - 1))
            j = i + 1
            delta = rng.choice([-1.0, 1.0]) * rng.uniform(0.003, 0.03)
            if cur_h[i] + delta < 0 or cur_h[i] + delta > 1:
                delta = -delta
            if cur_h[j] - delta < 0 or cur_h[j] - delta > 1:
                did_skip = True

        else:
            # Block move
            bs = int(rng.integers(3, min(10, n // 5)))
            if bs < 3:
                bs = 3
            i0 = int(rng.integers(0, max(1, n - bs)))
            j0 = int(rng.integers(0, max(1, n - bs)))
            if abs(i0 - j0) < bs:
                did_skip = True
            else:
                dt = rng.choice([-1.0, 1.0]) * rng.uniform(0.01, 0.05) / bs
                blk_i = cur_h[i0:i0+bs]
                blk_j = cur_h[j0:j0+bs]
                if blk_i.min() + dt < 0 or blk_i.max() + dt > 1:
                    dt = -dt
                if blk_j.min() - dt < 0 or blk_j.max() - dt > 1:
                    did_skip = True
                else:
                    cur_h[i0:i0+bs] += dt
                    cur_h[j0:j0+bs] -= dt
                    new_c = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx)
                    dc = new_c - cur_c
                    if dc <= 0 or rng.random() < math.exp(-max(dc, 0) / max(T, 1e-12)):
                        cur_c = new_c
                        if new_c < best_c:
                            best_c = new_c
                            best_h = cur_h.copy()
                            no_imp = 0
                    else:
                        cur_h[i0:i0+bs] -= dt
                        cur_h[j0:j0+bs] += dt
                    no_imp += 1
                    T = max(T * cool, T_min)
                    if no_imp > restart_at:
                        no_imp = 0
                        T = float(rng.uniform(0.002, 0.008))
                        cur_h = best_h + rng.normal(0, 0.012, n)
                        cur_h = project_feasible(cur_h, n)
                        cur_c = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx)
                    continue

        if did_skip:
            T = max(T * cool, T_min)
            continue

        # 2-point evaluation
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx)
        dc = new_c - cur_c

        if dc <= 0:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                no_imp = 0
            T = max(T * 0.999, T_min)
        elif rng.random() < math.exp(-dc / max(T, 1e-12)):
            cur_c = new_c
        else:
            cur_h[i] -= delta
            cur_h[j] += delta

        no_imp += 1
        T = max(T * cool, T_min)

        if no_imp > restart_at:
            no_imp = 0
            T = float(rng.uniform(0.002, 0.008))
            cur_h = best_h + rng.normal(0, 0.012, n)
            cur_h = project_feasible(cur_h, n)
            cur_c = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx)

    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + math.log(float(np.sum(np.exp(sh - smx))))) / alpha
    else:
        val = m + math.log(float(np.sum(np.exp(sh)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    nm1 = n - 1
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        k = mm - nm1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 10.0), (1000, 50.0), (5000, 500.0),
               (20000, 5000.0), (100000, 50000.0), (500000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, cur, jac=smooth_max_grad,
                           args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxls': 50})
            cur = project_feasible(res.x, n)
            c = compute_c5(cur, n)
            if c < best_c:
                best_c = c
        except Exception:
            pass
    return cur, best_c

def pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    h = h.copy()
    window = min(15, n // 10)
    step_sizes = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        max_sweeps = 3
        for _ in range(max_sweeps):
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + ss <= 1.0 and h[j] - ss >= 0.0:
                        h[i] += ss; h[j] -= ss
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= ss; h[j] += ss
                    if h[i] - ss >= 0.0 and h[j] + ss <= 1.0:
                        h[i] -= ss; h[j] += ss
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += ss; h[j] -= ss
            if not improved:
                break
    return h, best_c
