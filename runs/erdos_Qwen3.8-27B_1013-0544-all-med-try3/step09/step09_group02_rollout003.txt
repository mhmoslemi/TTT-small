            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_overlap(h, n, weights):
    """Gradient of weighted sum of overlap integrals."""
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(weights > 1e-14)[0]
    for k_idx in active:
        k = int(k_idx) - (n - 1)
        wk = weights[k_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wk * (1.0 - h[k:n])
            grad[k:n] += -wk * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wk * (1.0 - h[0:nv])
            grad[0:nv] += -wk * h[kk:n]
    return grad

def nesterov_optimize(h, n, alpha, deadline, max_iter=5000, lr_init=0.01):
    """Nesterov accelerated gradient with projection."""
    h_curr = project_feasible(h.copy(), n)
    h_next = h_curr.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    theta = 1.0
    lr = lr_init
    prev_grad = None

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_next, n)
        m = c_vec.max()
        w = softmax_weights(c_vec, alpha)
        grad = grad_overlap(h_next, n, w)

        # Adaptive learning rate (simple BB-like)
        if prev_grad is not None:
            dh = h_curr - h_next
            s = h_curr - h_next
            y = grad - prev_grad
            y_norm_sq = np.dot(y, y)
            s_dot_y = np.dot(s, y)
            if y_norm_sq > 1e-20 and s_dot_y > 1e-20:
                lr_new = s_dot_y / y_norm_sq
                lr = 0.5 * (lr + np.clip(lr_new, 0.01 * lr, 5.0 * lr))
        prev_grad = grad.copy()

        h_proposed = h_curr - lr * grad
        h_proposed = project_feasible(h_proposed, n)

        c_proposed = compute_c5(h_proposed, n)
        if c_proposed < best_c:
            best_c = c_proposed
            best_h = h_proposed.copy()

        theta_new = (1.0 + np.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
        h_next = h_proposed + (theta / theta_new - 1.0) * (h_proposed - h_curr)
        h_next = project_feasible(h_next, n)
        h_curr = h_proposed
        theta = theta_new

        if it % 100 == 0 and it > 0:
            c_check = compute_c5(h_curr, n)
            if c_check < best_c:
                best_c = c_check
                best_h = h_curr.copy()

    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    """L-BFGS-B with smooth max objective."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    if configs is None:
        configs = [(500, 50.0), (2000, 200.0), (8000, 1000.0), (20000, 5000.0)]

    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def _lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = softmax_weights(c, alpha)
    g = grad_overlap(x, n, w)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def frank_wolfe(h, n, deadline, alpha=1000, max_iter=2000, gamma=0.1):
    """Frank-Wolfe (conditional gradient) for the smooth max problem."""
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    target_sum = n / 2.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_curr, n)
        w = softmax_weights(c_vec, alpha)
        grad = grad_overlap(h_curr, n, w)

        # Linear minimization over polytope: min g^T s s.t. 0<=s<=1, sum(s)=target_sum
        # Solution: set s=1 where g is smallest, s=0 where largest, one fractional
        order = np.argsort(grad)
        s = np.zeros(n)
        count_full = int(target_sum - 0.5)
        if count_full < 0:
            count_full = 0
        if count_full > n:
            count_full = n
        for i in range(count_full):
            s[order[i]] = 1.0
        rem = target_sum - count_full
        if 0 < rem < 1 and count_full < n:
            s[order[count_full]] = rem
        # If remaining is 1, we need another full
        elif rem == 1.0 and count_full < n:
            s[order[count_full]] = 1.0

        # FW step with adaptive step size
        gamma_it = 2.0 / (it + 2.0 + gamma * 100)
        h_new = (1.0 - gamma_it) * h_curr + gamma_it * s
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new

    return best_h, best_c

def sqp_active(h, n, deadline, max_iter=200, rho_init=0.03):
    """Active-set SQP: minimize max over active shifts."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    rho = rho_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]

        # Build LP for direction
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        c_vals = [c_vec[idx] for idx in active]
        total_ineq = len(active) + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for idx in active:
            k = idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[0:nv] = (1.0 - best_h[k:n])
                g[k:n] += -best_h[0:nv]
            else:
                kk = -k
                nv = n - kk
                g[kk:n] = (1.0 - best_h[0:nv])
                g[0:nv] += -best_h[kk:n]
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[0] if row == 0 else -c_vec[idx]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = best_h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - best_h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        bounds = [(None, None)] * n + [(0.0, 1.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                if np.max(np.abs(delta)) < 1e-13:
                    no_improve += 1
                    if no_improve > 15:
                        break
                    rho *= 0.7
                    continue
                h_new = project_feasible(best_h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    no_improve = 0
                    rho = min(rho * 1.2, 0.08)
                else:
                    no_improve += 1
                    rho *= 0.5
                    if rho < 1e-10 or no_improve > 20:
                        break
            else:
                rho *= 0.5
                if rho < 1e-10:
                    break
        except Exception:
            break
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=30):
    """Very fine single-coordinate and pair local search."""
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    target = n / 2.0

    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            # Compute sensitivity gradient
            w = softmax_weights(c_vec, 5000.0)
            grad = grad_overlap(h, n, w)
            # Focus on indices where perturbation most helps
            for i in np.argsort(np.abs(grad))[-min(radius * 2, n) :: -1]:
                if time.time() > deadline:
                    return h, best_c
                i = int(i)
                # Try moving mass from i to neighbors
                for j_off in range(1, min(radius, n)):
                    j1 = i + j_off
                    j2 = i - j_off
                    for j, direction in [(j1, +1), (j2, +1), (j1, -1), (j2, -1)]:
                        if j < 0 or j >= n:
                            continue
                        if direction > 0:
                            if h[i] - step < -1e-12 or h[j] + step > 1.0 + 1e-12:
                                continue
                            h[i] -= step
                            h[j] += step
                        else:
                            if h[i] + step > 1.0 + 1e-12 or h[j] - step < -1e-12:
                                continue
                            h[i] += step
                            h[j] -= step
                        if h[i] < -1e-12:
                            h[i] = 0.0
                        if h[j] < -1e-12:
                            h[j] = 0.0
                        c_val = compute_c5(h, n)
                        if c_val < best_c - 1e-15:
                            best_c = c_val
                            improved = True
                        else:
                            if direction > 0:
                                h[i] += step
                                h[j] -= step
                            else:
                                h[i] -= step
                                h[j] += step
                # Single point nudge
                if h[i] > step and h[i] - step >= 0:
                    h[i] -= step
                    # Redistribute
                    h[i] += step  # undo
                if h[i] < 1.0 - step:
                    h[i] += step
                    # Find where to put the mass - skip for simplicity in tight budget

    return h, best_c

def upsample_and_optimize(h_low, n_low, n_high, deadline):
    """Upsample and run optimization pipeline."""
    x_old = np.linspace(0, 2, n_low)
    x_new = np.linspace(0, 2, n_high)
    h_high = np.interp(x_new, x_old, h_low)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)

    t_remaining = deadline - time.time()
    # L-BFGS
    t1 = time.time() + min(60, t_remaining * 0.3)
    if t1 > time.time() + 5:
        h_high, c_high = lbfgs_optimize(h_high, n_high, t1,
            configs=[(500, 100.0), (2000, 500.0), (6000, 2000.0)])
    # Nesterov
    t2 = time.time() + min(50, deadline - time.time() - 30)
    if t2 > time.time() + 5 and time.time() < deadline - 30:
        h2, c2 = nesterov_optimize(h_high, n_high, 3000, t2, max_iter=2000, lr_init=0.005)
        if c2 < c_high:
            c_high = c2
            h_high = h2
    # SQP
    t3 = time.time() + min(40, deadline - time.time() - 20)
    if t3 > time.time() + 5 and time.time() < deadline - 20:
        h3, c3 = sqp_active(h_high, n_high, t3, max_iter=100, rho_init=0.03)
        if c3 < c_high:
            c_high = c3
            h_high = h3
    # Local search
    if time.time() < deadline - 5:
        h4, c4 = fine_local_search(h_high, n_high, deadline - 3, radius=20)
        if c4 < c_high:
            c_high = c4
            h_high = h4
    return h_high, c_high

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS warmup at initial resolution (40s)
    t1 = min(t0 + 40, deadline - 800)
    if t1 > time.time() + 3:
        best_h, best_c = lbfgs_optimize(best_h, n_init, t1,
            configs=[(300, 50.0), (1500, 300.0), (5000, 2000.0)])

    # Phase 2: Nesterov (40s) - different trajectory
    t2 = min(t0 + 80, deadline - 700)
    if t2 > time.time() + 3:
        h2, c2 = nesterov_optimize(best_h, n_init, 2000, t2, max_iter=3000, lr_init=0.008)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Frank-Wolfe (30s) - explores different region
    t_fw = min(t0 + 110, deadline - 650)
    if t_fw > time.time() + 5:
        h_fw, c_fw = frank_wolfe(best_h, n_init, t_fw, alpha=5000, max_iter=500)
        if c_fw < best_c:
            best_c = c_fw
            best_h = h_fw

    # Phase 4: SQP (60s)
    t3 = min(t0 + 170, deadline - 550)
    if t3 > time.time() + 5:
        best_h, best_c = sqp_active(best_h, n_init, t3, max_iter=150, rho_init=0.04)

    # Phase 5: Second L-BFGS with different params (40s)
    t4 = min(t0 + 210, deadline - 500)
    if t4 > time.time() + 3:
        h4, c4 = lbfgs_optimize(best_h, n_init, t4,
            configs=[(10000, 5000.0), (30000, 20000.0), (60000, 50000.0)])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 6: SQP again (40s)
    t5 = min(t0 + 250, deadline - 450)
    if t5 > time.time() + 3:
        best_h, best_c = sqp_active(best_h, n_init, t5, max_iter=100, rho_init=0.02)

    # Phase 7: Fine local search at initial res (30s)
    t6 = min(t0 + 280, deadline - 400)
    if t6 > time.time() + 3:
        best_h, best_c = fine_local_search(best_h, n_init, t6, radius=40)

    # Phase 8: Upsample to n=1000 and optimize (300s)
    n_high = 1000
    t7 = deadline - 120
    if time.time() < t7 and t7 > time.time() + 60:
        h_high, c_high = upsample_and_optimize(best_h, n_init, n_high, t7)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_cur = n_high
        else:
            n_cur = n_init
    else:
        n_cur = n_init

    # Phase 9: Try n=1200 if time remains (80s)
    if time.time() < deadline - 40:
        n_1200 = 1200
        x_cur = np.linspace(0, 2, n_cur)
        x_1200 = np.linspace(0, 2, n_1200)
        h_1200 = np.interp(x_1200, x_cur, best_h)
        h_1200 = project_feasible(h_1200, n_1200)
        c_1200 = compute_c5(h_1200, n_1200)
        t_1200 = deadline - 15
        if t_1200 > time.time() + 30:
            h_1200, c_1200 = lbfgs_optimize(h_1200, n_1200, min(time.time() + 50, t_1200 - 20),
                configs=[(2000, 500.0), (8000, 3000.0)])
            if time.time() < t_1200 - 10:
                h_1200, c_1200 = sqp_active(h_1200, n_1200, t_1200 - 5, max_iter=50, rho_init=0.03)
            if c_1200 < best_c:
                best_c = c_1200
                best_h = h_1200

    # Final check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    # Verify sum constraint
    assert abs(best_h.sum() - n_final / 2.0) < 1e-8, f"Sum constraint violated: {best_h.sum()} vs {n_final/2.0}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bounds violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
