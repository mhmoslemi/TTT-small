        if abs(diff) < 1e-15:
            break

            if tot < 1e-15:

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:

def lsq_obj(x, n, alpha, mu):
    """Log-sum-exp smoothed max with penalty."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    st_max = sh.max()
    if st_max > 500:
        val = m + (st_max + np.log(np.sum(np.exp(sh - st_max)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lsq_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    st = alpha * (c - m)
    st_max = st.max()
    w = np.exp(st - st_max)

    dx = 2.0 / n

        wm = w[mm] * dx
        if k >= 0:

            if nv > 0:

                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk

                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_step(h, n, active_idx, rho):
    """One SQP step: solve LP to find best descent direction."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    # Constraints: for each active shift, grad_k^T d - t <= -c_k
    # Box: -h_i <= d_i <= 1-h_i (approximate via rho)
    # |d_i| <= rho
    # sum(d) = 0
    total_ineq = n_act + 2 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

    best_c = compute_c5(h, n)
    rho = rho_init

            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:n_active_max]]
        if len(active) == 0:

        delta, t_opt = sqp_step(h, n, active, rho)

            if rho < 1e-10:

        if np.max(np.abs(delta)) < 1e-14:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def lsq_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    st_max = sh.max()
    if st_max > 500:
        val = m + (st_max + np.log(np.sum(np.exp(sh - st_max)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lsq_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    st = alpha * (c - m)
    st_max = st.max()
    w = np.exp(st - st_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    threshold = 1e-12
    for mm in range(2 * n - 1):
        if w[mm] < threshold:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 2 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def run_sqp(h, n, deadline, max_iter=400, rho_init=0.05, n_active_max=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        mv = c_vec.max()
        thresh = mv - max(1e-10, 0.0003 * mv)
        active_mask = c_vec >= thresh
        n_active = int(active_mask.sum())
        if n_active == 0:
            active = [int(np.argmax(c_vec))]
        elif n_active > n_active_max:
            active_all = np.where(active_mask)[0]
            order = np.argsort(-c_vec[active_all])
            active = [int(active_all[i]) for i in order[:n_active_max]]
        else:
            active = [int(x) for x in np.where(active_mask)[0]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-12:

            stall += 1
            if stall > 30:
                break
            continue
        max_delta = float(np.max(np.abs(delta)))

def gradient_local_search(h, n, deadline, n_trials=300, radius=25):

    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:

            for idx in active[:20]:
