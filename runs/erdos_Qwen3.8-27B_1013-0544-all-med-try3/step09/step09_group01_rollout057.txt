        sh_clipped = np.clip(sh, -700, 700)
        val = m + np.log(np.sum(np.exp(sh_clipped))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _sm_grad(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    if active is None:
        idxs = np.arange(len(c))
        vals = c
    else:
        idxs = active
        vals = c[active]
    m = float(vals.max())
    sh = alpha * (vals - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w = w / wsum
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for ii in range(len(w)):
        wi = w[ii]
        if wi < 1e-15:
            continue
        idx = int(idxs[ii])
        k = idx - offset
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h_in, n, deadline, alphas, active, mu, maxit_cap=5000):
    h_cur = project_feasible(h_in, n)
    c_cur = compute_c5(h_cur, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 30:
            break
        maxit = max(500, min(maxit_cap, int(rem * 12)))
        try:
            res = minimize(_sm_obj, h_cur, jac=_sm_grad,
                           args=(n, alpha, active, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': maxit * 12,
                                    'maxls': 60, 'maxcor': 60,
                                    'iprint': -1})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur - 1e-18:
                c_cur = c_new
                h_cur = h_new
            elif c_new <= c_cur + 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c_cur

def get_active_set(h, n, top_k, frac=0.0005):
    c_vec = compute_c5_vec(h, n)
    c_max = float(c_vec.max())
    thresh = c_max - max(1e-10, frac * c_max)
    act = np.where(c_vec >= thresh)[0]
    if len(act) > top_k:
        act = np.array(np.argsort(-c_vec[act])[:top_k])
    if len(act) == 0:
        act = np.array([int(np.argmax(c_vec))])
    return act

def slqp_optimize(h_in, n, deadline, alpha, maxiter):
    h_cur = project_feasible(h_in, n)
    c_cur = compute_c5(h_cur, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_sm_obj, h_cur, args=(n, alpha, None, 0.0),
                       jac=_sm_grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_cur:
            return h_new, c_new
    except Exception:
        pass
    return h_cur, c_cur

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
