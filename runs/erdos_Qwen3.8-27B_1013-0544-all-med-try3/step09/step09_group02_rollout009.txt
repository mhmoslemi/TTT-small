        return None, None
    total_ineq = n_act + 4 * n

        c_val = c_vec[int(idx)]

        A_ub[row, n] = -1.0
        b_ub[row] = -c_val
        row += 1

        b_ub[row] = h[i]
        row += 1

        row += 1

        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj_topk(h, n, alpha, K, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    k_use = min(K, len(c))
    top_idx = np.argsort(-c)[:k_use]
    sh = alpha * (c[top_idx] - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_topk(h, n, alpha, K, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    k_use = min(K, len(c))
    top_idx = np.argsort(-c)[:k_use]
    sh = alpha * (c[top_idx] - m)
    mx = sh.max()
    w = np.exp(sh - mx)

            continue
        g = shift_grad_single(h, n, top_idx[i])
        grad += (w[i] * alpha) * g
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    if n_act == 0:
        return None, None
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, int(idx))
        c_val = c_vec[int(idx)]
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_val
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0

    # Phase 1: Fast top-K gradient descent at original resolution (70s)
    t1 = min(t0 + 70, deadline - 900)
    if t1 > time.time() + 5:
        best_h, best_c = gradient_descent_topk(best_h, n_orig, t1,
            configs=[(200, 12, 100.0), (1000, 20, 500.0), (3000, 30, 2000.0), (8000, 40, 5000.0)])

    # Phase 2: SQP optimization (130s)
    t2 = min(t0 + 200, deadline - 700)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=500, rho_init=0.05, top_k=8)

    # Phase 3: Focused local search (100s)
    t3 = min(t0 + 300, deadline - 550)
    if t3 > time.time() + 5:
        best_h, best_c = local_search_focused(best_h, n_orig, t3, radius=50)

    # Phase 4: Another SQP round (70s)
    t4 = min(t0 + 370, deadline - 450)
    if t4 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t4, max_iter=300, rho_init=0.02, top_k=6)

    # Phase 5: SLSQP polish (50s)
    t5 = min(t0 + 420, deadline - 380)
    if t5 > time.time() + 5:
        h5, c5 = slqp_polish(best_h, n_orig, t5, alpha=3000000, K=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Upsample to n=1000 and optimize (200s)
    n_high = 1000
    if time.time() < deadline - 250:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t6 = min(time.time() + 70, deadline - 150)
        if t6 > time.time() + 5:
            h_high, c_high = gradient_descent_topk(h_high, n_high, t6,
                configs=[(200, 12, 100.0), (1000, 20, 500.0), (3000, 30, 2000.0)])

        t7 = min(time.time() + 60, deadline - 80)
        if t7 > time.time() + 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t7, max_iter=200, rho_init=0.04, top_k=8)

        t8 = min(time.time() + 50, deadline - 40)
        if t8 > time.time() + 5:
            h_high, c_high = local_search_focused(h_high, n_high, t8, radius=30)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Phase 7: Multi-start perturbation (80s)
    if time.time() < deadline - 60:
        n_try = len(best_h)
        for s in [123, 456, 789]:
            if time.time() > deadline - 60:
                break
            allocated = min(25, deadline - 60 - time.time())
            if allocated < 10:
                break
            h_p, c_p = perturb_and_optimize(best_h, n_try, s, allocated)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 8: Binarization attempt
    if time.time() < deadline - 20:
        hb, cb = try_binarize(best_h, len(best_h))
        if cb < best_c:
            best_c = cb
            best_h = hb

    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 8:
        n_final = len(best_h)
        h_final, c_final = slqp_polish(best_h, n_final, deadline - 3, alpha=5000000, K=30)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
