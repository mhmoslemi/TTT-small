import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    val = m + logsum / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_step(h, n, active_idx, rho):
    """Solve LP: min t s.t. g_j·δ ≤ -f_j, -rho ≤ δ_i ≤ rho, sum(δ)=0"""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 2 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        g = shift_gradient(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

        if len(active) > max_active:

            active = active[top_order[:max_active]]
        
        delta, t_opt = sqp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:

            stall += 1
            if stall > 30:

        c_new = compute_c5(h_new, n)

        else:

def lbfgs_optimize(h, n, deadline, configs):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

        except Exception:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    val = m + logsum / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    w_thresh = 1e-12
    for mm in range(2 * n - 1):
        if w[mm] < w_thresh:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_solve(h, n, active_idx, rho):

    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 2 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        g = shift_gradient(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1

            if time.time() < t_sub2 - 3:
                h_lp, c_lp = lbfgs_optimize(h_new, n, t_sub2,
                    configs=[(20000, 50000.0)])
                if c_lp < best_c:
                    best_c = c_lp
                    best_h = h_lp
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS warmup (25s)
    t1 = min(t0 + 25, deadline - 900)
    if time.time() < t1 - 3:
        h, c = lbfgs_optimize(h, n, t1, configs=[(300, 100.0), (2000, 2000.0)])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 2: Primary SQP (200s)
    t2 = min(t0 + 225, deadline - 700)
    if time.time() < t2 - 5:
        h, c = sqp_optimize(h, n, t2, max_iter=800, rho_init=0.08, max_active=25)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 3: High-alpha L-BFGS (100s)
    t3 = min(t0 + 325, deadline - 550)
    if time.time() < t3 - 3:
        h, c = lbfgs_optimize(h, n, t3,
            configs=[(10000, 10000.0), (50000, 100000.0)])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 4: SQP with more active (130s)
    t4 = min(t0 + 455, deadline - 400)
    if time.time() < t4 - 5:
        h, c = sqp_optimize(h, n, t4, max_iter=500, rho_init=0.04, max_active=35)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 5: Coordinate search (100s)
    t5 = min(t0 + 555, deadline - 300)
    if time.time() < t5 - 3:
        h, c = fast_coordinate_search(h, n, t5)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 6: Very high alpha L-BFGS (80s)
    t6 = min(t0 + 635, deadline - 200)
    if time.time() < t6 - 3:
        h, c = lbfgs_optimize(h, n, t6,
            configs=[(100000, 500000.0), (300000, 2000000.0)])
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 7: SQP polish (80s)
    t7 = min(t0 + 715, deadline - 150)
    if time.time() < t7 - 5:
        h, c = sqp_optimize(h, n, t7, max_iter=400, rho_init=0.02, max_active=40)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 8: Perturbation restarts (120s)
    t8 = min(t0 + 835, deadline - 60)
    if time.time() < t8 - 30:
        h, c = perturbation_restart(h, n, t8, n_restarts=3)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 9: Final SQP (60s)
    t9 = min(t0 + 895, deadline - 20)
    if time.time() < t9 - 5:
        h, c = sqp_optimize(h, n, t9, max_iter=200, rho_init=0.01, max_active=45)
        if c < best_c:
            best_c = c
            best_h = h

    # Phase 10: Final L-BFGS polish
    if time.time() < deadline - 8:
        h, c = lbfgs_optimize(h, n, deadline - 3,
            configs=[(500000, 5000000.0)])
        if c < best_c:
            best_c = c
            best_h = h

    # Final feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
