from scipy.optimize import minimize

_CORR_ORIENT = None


def _orient_flag():
    global _CORR_ORIENT
    if _CORR_ORIENT is None:
        a = np.array([0.1, 0.9, 0.2, 0.7, 0.35], dtype=float)
        v = 1.0 - a
        m = 2 * a.size - 1
        L = 1 << (m - 1).bit_length()
        conv = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(v[::-1], L), L)[:m]
        corr = np.correlate(a, v, mode='full')
        ok1 = bool(np.allclose(conv, corr, rtol=1e-8, atol=1e-8))
        ok2 = bool(np.allclose(conv, corr[::-1], rtol=1e-8, atol=1e-8))
        if ok1 and not ok2:
            _CORR_ORIENT = 0
        elif ok2 and not ok1:
            _CORR_ORIENT = 1
        else:
            _CORR_ORIENT = 0
    return _CORR_ORIENT


def fft_c_vec(h):
    n = int(h.shape[0])
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()
    v = 1.0 - h
    c = np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(v[::-1], L), L)[:m]
    if _orient_flag():
        c = c[::-1]
    return c


def c5_fft(h):
    n = len(h)
    c = fft_c_vec(h)
    return float(np.max(c) * (2.0 / float(n)))


def c5_exact(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / float(n)))


def enforce_sum(h, n):
    target = 0.5 * float(n)
    for _ in range(4):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-15:
            break
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            idx = free[int(np.argmin(np.abs(h[free] - 0.5)))]
        elif diff > 0.0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:

            idx = free[int(np.argmin(1.0 - h[free]))]

            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            idx = free[int(np.argmax(h[free]))]

        h = np.clip(h, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-14:
        return enforce_sum(h, n)

    diff = target - s
    h2 = h + diff / float(n)
    if h2.min() >= -1e-15 and h2.max() <= 1.0 + 1e-15:
        h2 = np.clip(h2, 0.0, 1.0)
        if abs(float(h2.sum()) - target) <= 1e-12:
            return enforce_sum(h2, n)

    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(h - mid, 0.0, 1.0).sum())
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - lo, 0.0, 1.0)
    return enforce_sum(h, n)


def make_sturmian(n, alpha):
    i = np.arange(int(n), dtype=float)
    r = np.mod(i * float(alpha), 1.0)
    h = (r < 0.5).astype(float)
    return project_feasible(h, n)


def make_random_binary(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    k = int(n // 2)
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)


def upsample_linear(h, n):
    n0 = int(len(h))
    n = int(n)
    if n0 == n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / float(n0)
    x1 = (np.arange(n) + 0.5) / float(n)
    return np.interp(x1, x0, h)


def upsample_nearest(h, n):
    n0 = int(len(h))
    n = int(n)
    if n0 == n:
        return h.copy()
    if n0 == 1:
        return np.full(n, float(h[0]))
    idx = np.round((n - 1) * np.arange(n) / float(n0 - 1)).astype(int)
    idx = np.clip(idx, 0, n0 - 1)
    return h[idx].copy()


def upsample_candidates(source_h, n):
    out = []
    n0 = int(len(source_h))
    n = int(n)
    if n0 == n:
        out.append(source_h.copy())
    elif n % n0 == 0:
        q = n // n0
        if q <= 256:
            out.append(np.repeat(source_h, q).copy())
    out.append(upsample_linear(source_h, n))
    out.append(upsample_nearest(source_h, n))
    return out


def select_active(c, max_count):
    max_count = int(max(1, int(max_count)))
    c = np.asarray(c, dtype=float)
    maxv = float(np.max(c))
    tol = max(1e-12, 3.0e-4 * maxv)
    idx = np.where(c >= maxv - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    elif idx.size > max_count:
        order = np.argsort(-c[idx])
        idx = idx[order[:max_count]]
    return idx.astype(int)


def smooth_value(c, active, alpha):
    vals = c[active]
    if vals.size == 0:
        return 0.0
    if vals.size == 1 or alpha <= 1e-12:
        return float(vals.max())
    base = float(vals.max())
    sh = alpha * (vals - base)
    z = sh - float(sh.max())
    logsum = float(np.log(np.sum(np.exp(z))))
    return base + logsum / alpha


def softmax_weights(c, active, alpha):
    vals = c[active]
    m = vals.size
    if m == 0:
        return np.zeros(0)
    if m == 1 or alpha <= 1e-12:
        return np.ones(m)
    base = float(vals.max())
    sh = alpha * (vals - base)
    z = sh - float(sh.max())
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(m)
        w[int(np.argmax(vals))] = 1.0
    else:
        w /= s
    return w


def grad_weights(h, n, active, w):
    n = int(n)
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    for wi, idx in zip(w, active):
        wi = float(wi)
        if wi <= 1e-16:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[k:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[k:n])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    return grad


def topK_obj(x, n, active, alpha, mu):
    n = int(n)
    c = fft_c_vec(x)
    val = smooth_value(c, active, alpha) * (2.0 / float(n))
    if mu > 0.0:
        s = float(x.sum())
        val += mu * (s - 0.5 * float(n)) ** 2
    return float(val)


def topK_grad(x, n, active, alpha, mu):
    n = int(n)
    c = fft_c_vec(x)
    w = softmax_weights(c, active, alpha)
    grad = grad_weights(x, n, active, w)
    if mu > 0.0:
        grad += 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return grad


def sum_eq_fun(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_eq_jac(x, n):
    return np.ones(len(x), dtype=float)


def lbfgs_once(h, n, active, alpha, mu, deadline, bounds):
    rem = deadline - time.time()
    if rem < 8.0:
        return None
    if n <= 1024:
        est = 0.006
        cap = 360
    elif n <= 2048:
        est = 0.014
        cap = 220
    else:
        est = 0.035
        cap = 120
    maxit = int(rem / est)
    if maxit > cap:
        maxit = cap
    if maxit < 20:
        return None
    try:
        res = minimize(
            topK_obj,
            h,
            jac=topK_grad,
            args=(n, active, alpha, mu),
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': int(maxit),
                'ftol': 1e-18,
                'gtol': 1e-12,
                'maxfun': int(maxit * 12),
                'maxls': 20,
                'maxcor': 20,
            }
        )
        return np.asarray(res.x, dtype=float)
    except Exception:
        return None


def slqp_once(h, n, active, alpha, deadline, bounds, maxit):
    rem = deadline - time.time()
    if rem < 12.0 or maxit < 5:
        return None
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            topK_obj,
            h,
            jac=topK_grad,
            args=(n, active, alpha, 0.0),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxit), 'ftol': 1e-16}
        )
        return np.asarray(res.x, dtype=float)
    except Exception:
        return None


def active_lbfgs(h, n, deadline, bounds=None):
    n = int(n)
    best = project_feasible(h, n)
    bestc = c5_fft(best)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n

    if n <= 1024:
        configs = [
            (1, [1.0]),
            (3, [10.0, 100.0, 1000.0]),
            (6, [100.0, 1000.0, 5000.0]),
            (12, [1000.0, 5000.0, 50000.0]),
            (24, [5000.0, 50000.0]),
        ]
        max_cycles = 8
    elif n <= 2048:
        configs = [
            (1, [1.0]),
            (3, [10.0, 100.0]),
            (6, [100.0, 1000.0]),
            (12, [1000.0, 5000.0]),
            (24, [5000.0, 50000.0]),
        ]
        max_cycles = 6
    else:
        configs = [
            (1, [1.0]),
            (3, [10.0, 100.0]),
            (8, [100.0, 1000.0]),
            (16, [1000.0, 5000.0]),
        ]
        max_cycles = 4

    stagnant = 0
    cycle = 0
    while time.time() < deadline - 12.0 and cycle < max_cycles:
        cycle += 1
        improved = False
        for top_k, alphas in configs:
            if time.time() > deadline - 20.0:
                break
            kcap = min(int(top_k), max(1, int(0.02 * n) + 1))
            c = fft_c_vec(best)
            active = select_active(c, kcap)
            for alpha in alphas:
                if time.time() > deadline - 20.0:
                    break
                xnew = lbfgs_once(best, n, active, float(alpha), 0.01, deadline, bounds)
                if xnew is None:
                    continue
                hnew = project_feasible(xnew, n)
                cnew = c5_fft(hnew)
                if cnew < bestc - 1e-14:
                    best = hnew
                    bestc = float(cnew)
                    improved = True
                    break
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break
    return best, bestc


def slqp_polish(h, n, deadline, bounds=None):
    n = int(n)
    if n > 800:
        return project_feasible(h, n), c5_fft(h)
    best = project_feasible(h, n)
    bestc = c5_fft(best)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n

    configs = [
        (3, [100.0, 1000.0]),
        (8, [1000.0, 10000.0]),
        (16, [5000.0]),
    ]
    for top_k, alphas in configs:
        if time.time() > deadline - 15.0:
            break
        kcap = min(int(top_k), max(1, int(0.02 * n) + 1))
        c = fft_c_vec(best)
        active = select_active(c, kcap)
        for alpha in alphas:
            if time.time() > deadline - 15.0:
                break
            rem = deadline - time.time()
            maxit = int(min(80.0, max(8.0, rem / 0.35)))
            if maxit < 8:
                break
            xnew = slqp_once(best, n, active, float(alpha), deadline, bounds, maxit)
            if xnew is None:
                continue
            hnew = project_feasible(xnew, n)
            cnew = c5_fft(hnew)
            if cnew < bestc - 1e-14:
                best = hnew
                bestc = float(cnew)
    return best, bestc


def proj_grad_polish(h, n, deadline, max_steps=1000):
    n = int(n)
    h = project_feasible(h, n)
    val = c5_fft(h)
    top_k = 16 if n <= 2048 else 24
    alpha = 100.0
    step_scale = 1.0
    no_accept = 0

    for _ in range(max_steps):
        if time.time() > deadline - 2.0:
            break
        c = fft_c_vec(h)
        active = select_active(c, top_k)
        w = softmax_weights(c, active, alpha)
        grad = grad_weights(h, n, active, w)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break

        t = min(2.0, max(1e-8, step_scale / (gnorm + 1e-12)))
        accepted = False
        for _ in range(28):
            cand = project_feasible(h - t * grad, n)
            v = c5_fft(cand)
            if v < val - 1e-15:
                h = cand
                val = float(v)
                accepted = True
                break
            t *= 0.5
        if accepted:
            no_accept = 0
            step_scale = min(2.0, step_scale * 1.1)
        else:
            no_accept += 1
            step_scale *= 0.5
            if no_accept >= 8 or step_scale < 1e-8:
                break
    return h, val


def pair_gradient_polish(h, n, deadline, max_evals=1200):
    n = int(n)
    h = project_feasible(h, n)
    bestc = c5_fft(h)
    evals = 0
    base_steps = [
        0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5.0e-5, 2.0e-5, 1.0e-5
    ]
    for base in base_steps:
        if time.time() + 2.0 > deadline or evals >= max_evals:
            break
        improved_this_base = False
        while time.time() + 2.0 < deadline and evals < max_evals:
            c = fft_c_vec(h)
            active = select_active(c, 32)
            w = np.full(active.size, 1.0 / max(1, active.size), dtype=float)
            grad = grad_weights(h, n, active, w)

            pos = np.where(grad > 1e-10)[0]
            neg = np.where(grad < -1e-10)[0]
            if pos.size == 0 or neg.size == 0:
                break

            pos_top = pos[np.argsort(-grad[pos])[:20]]
            neg_top = neg[np.argsort(grad[neg])[:20]]

            best_gain = -1.0
            bi = -1
            bj = -1
            for i in pos_top:
                gi = float(grad[i])
                for j in neg_top:
                    g = gi - float(grad[j])
                    if g > best_gain:
                        best_gain = g
                        bi = int(i)
                        bj = int(j)

            if bi < 0 or best_gain < 1e-12:
                break

            maxamt = min(float(base), float(h[bi]), float(1.0 - h[bj]))
            if maxamt < 1e-12:
                break

            moved = False
            for frac in (1.0, 0.5, 0.25):
                amt = maxamt * frac
                if amt < 1e-12:
                    break
                cand = h.copy()
                cand[bi] -= amt
                cand[bj] += amt
                cand = project_feasible(cand, n)
                v = c5_fft(cand)
                evals += 1
                if v < bestc - 1e-15:
                    h = cand
                    bestc = float(v)
                    moved = True
                    improved_this_base = True
                    break
            if not moved:
                break
        if not improved_this_base:
            continue
    return h, bestc


def optimize_n(h, n, deadline):
    n = int(n)
    h = project_feasible(h, n)
    best = h.copy()
    bestc = c5_fft(best)
    bounds = [(0.0, 1.0)] * n

    if time.time() < deadline - 30.0:
        lb_dl = min(deadline - 30.0, time.time() + max(20.0, (deadline - time.time()) * 0.55))
        h1, c1 = active_lbfgs(best, n, lb_dl, bounds)
        if c1 < bestc:
            best = h1
            bestc = float(c1)

    if n <= 800 and time.time() < deadline - 25.0:
        sl_dl = min(deadline - 25.0, time.time() + max(10.0, (deadline - time.time()) * 0.25))
        h2, c2 = slqp_polish(best, n, sl_dl, bounds)
        if c2 < bestc:
            best = h2
            bestc = float(c2)

    if time.time() < deadline - 15.0:
        pg_dl = min(deadline - 12.0, time.time() + max(10.0, (deadline - time.time()) * 0.25))
        h3, c3 = proj_grad_polish(best, n, pg_dl, max_steps=1200)
        if c3 < bestc:
            best = h3
            bestc = float(c3)

    if time.time() < deadline - 4.0:
        max_evals = 700 if n > 2048 else 1400
        h4, c4 = pair_gradient_polish(best, n, deadline - 3.0, max_evals=max_evals)
        if c4 < bestc:
            best = h4
            bestc = float(c4)

    return best, bestc


def initial_best_at_n(n, source_h, rng):
    n = int(n)
    best = make_sturmian(n, 0.6180339887)
    bestc = c5_fft(best)

    cands = []
    if source_h is not None:
        cands.extend(upsample_candidates(source_h, n))

    for alpha in (0.3819660113, 0.2360679775, 0.7236067977,
                  0.4142135624, 0.5360679775, 0.1545084972):
        cands.append(make_sturmian(n, alpha))

    if n <= 2048:
        cands.append(make_random_binary(n, rng))

    for hh in cands:
        try:
            hh = project_feasible(hh, n)
            v = c5_fft(hh)
            if v < bestc:
                best = hh
                bestc = float(v)
        except Exception:
            pass
    return best, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(float(budget_s), 60.0) - 15.0

    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        n0 = int(len(parent))
        h0 = project_feasible(parent, n0)
    else:
        n0 = 1024
        h0 = make_sturmian(n0, 0.6180339887)

    best_h = h0.copy()
    best_c = c5_fft(best_h)
    best_n = n0

    def maybe_update(cond):
        pass

    if best_n <= 1200 and time.time() + 70.0 < deadline - 260.0:
        qdl = min(deadline - 240.0, t0 + 75.0)
        hq, cq = active_lbfgs(best_h, best_n, qdl, [(0.0, 1.0)] * best_n)
        if cq < best_c:
            best_h = hq.copy()
            best_c = float(cq)
            best_n = best_n

    if best_n < 1024 and time.time() + 120.0 < deadline - 260.0:
        target_n = 1024
        phase_dl = min(deadline - 240.0, time.time() + 220.0)
        start, cstart = initial_best_at_n(target_n, best_h, rng)
        if cstart < best_c:
            best_h = start.copy()
            best_c = float(cstart)
            best_n = target_n
        if cstart < best_c + 0.003:
            hopt, copt = optimize_n(start, target_n, phase_dl)
            if copt < best_c:
                best_h = hopt.copy()
                best_c = float(copt)
                best_n = target_n

    if best_n < 2048 and time.time() + 150.0 < deadline - 180.0:
        target_n = 2048
        phase_dl = min(deadline - 160.0, time.time() + 320.0)
        start, cstart = initial_best_at_n(target_n, best_h, rng)
        if cstart < best_c:
            best_h = start.copy()
            best_c = float(cstart)
            best_n = target_n
        if cstart < best_c + 0.003:
            hopt, copt = optimize_n(start, target_n, phase_dl)
            if copt < best_c:
                best_h = hopt.copy()
                best_c = float(copt)
                best_n = target_n

    if best_n < 4096 and time.time() + 180.0 < deadline - 70.0:
        target_n = 4096
        phase_dl = min(deadline - 50.0, time.time() + 240.0)
        start, cstart = initial_best_at_n(target_n, best_h, rng)
        if cstart < best_c:
            best_h = start.copy()
            best_c = float(cstart)
            best_n = target_n
        if cstart < best_c + 0.003:
            hopt, copt = optimize_n(start, target_n, phase_dl)
            if copt < best_c:
                best_h = hopt.copy()
                best_c = float(copt)
                best_n = target_n

    reserve = 10.0 if best_n > 2048 else 6.0
    if time.time() + 5.0 < deadline - reserve:
        fdin = deadline - reserve
        hf, cf = proj_grad_polish(best_h, best_n, fdin, max_steps=700)
        if cf < best_c:
            best_h = hf.copy()
            best_c = float(cf)
        if time.time() < deadline - max(3.0, reserve - 4.0):
            hp, cp = pair_gradient_polish(best_h, best_n, deadline - max(2.0, reserve - 6.0),
                                          max_evals=600)
            if cp < best_c:
                best_h = hp.copy()
                best_c = float(cp)

    best_h = project_feasible(best_h, best_n)
    best_n = int(len(best_h))

    do_exact = best_n <= 4096 and time.time() + 3.0 < deadline
    if do_exact:
        try:
            exact = c5_exact(best_h)
            if np.isfinite(exact):
                best_c = float(exact)
            else:
                best_c = c5_fft(best_h) + 1e-12
        except Exception:
            best_c = c5_fft(best_h) + 1e-12
    else:
        best_c = c5_fft(best_h) + 1e-12

    return best_h, float(best_c), best_n
