    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-15
            if not mask.any(): break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_half = n // 2
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n_half]] = 1.0
    return result
