def compute_c5_vec_fft(h, n):

    dx = 2.0 / n

    # We need: sum_i h[i] * (1-h[i+k]) for various k
    # This is the cross-correlation of h and (1-h)
    # Use FFT: corr = ifft(fft(h) * conj(fft(1-h, 2n)))

    H = fft(h, N)
    O = fft(1 - h, N)

    # np.correlate(h, 1-h, 'full') gives 2n-1 values

    # In FFT: position i gives lag i (for i < N/2) or lag i-N (for i >= N/2)
    # We need lags -(n-1) to (n-1)
    # Full corr[k] = sum_i a[i] b[i+k] for k from -(n-1) to (n-1)
    # Actually np.correlate(a, b, 'full')[k] = sum_i a[i] * b[i - (k - (n-1))]
    # Let me just use the FFT result directly
    # ifft(fft(a, N) * conj(fft(b, N))) gives circular cross-correlation
    # For linear cross-correlation with N >= 2n-1, we get:
    # result[0:n-1] = corr for lags 0, 1, ..., n-1 (positive lags)
    # result[N-n+1:N] = corr for lags -(n-1), ..., -1 (negative lags)
    # Wait, let me be more careful.
    # np.correlate(a, b, 'full')[k] = sum_{j} a[j] * conj(b[j - k + (n-1)])

    # Let m = k - (n-1), then it's sum_j a[j] * b[j - m]
    # This is the cross-correlation at lag m
    # FFT: ifft(fft(a, N) * conj(fft(b, N))) gives circular cross-correlation
    # For N >= 2n-1, this equals the linear cross-correlation
    # The result is: r[m] for m = 0, 1, ..., N-1 (circular)
    # r[m] = sum_j a[j] * b[(j-m) mod N]
    # For the linear cross-correlation, we want:
    # c[m] = sum_j a[j] * b[j-m] for valid j
    # If we zero-pad to length N >= 2n-1:
    # r[0] = c[0] (zero lag)
    # r[m] = c[m] for m = 1, ..., n-1 (positive lag)
    # r[N-m] = c[-m] for m = 1, ..., n-1 (negative lag)
    # So the full correlation vector (matching np.correlate output) is:
    # full[k] = c[k-(n-1)] for k = 0, ..., 2n-2
    # full[0] = c[-(n-1)] = r[N-(n-1)]
    # full[n-1] = c[0] = r[0]
    # full[2n-2] = c[n-1] = r[n-1]
    full = np.zeros(2*n-1)
    full[0:n-1] = r[N-(n-1):N][::-1]  # negative lags
    full[n-1] = r[0]  # zero lag
    full[n:2*n-1] = r[1:n]  # positive lags
    return full * dx

def upsample_cubic(h_old, n_old, n_new):
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 2.0, n_old)
    try:
        cs = CubicSpline(x_old, h_old, bc_type='clamped')
        x_new = np.linspace(0, 2.0, n_new)
        h_new = cs(x_new)
    except Exception:
        x_new = np.linspace(0, 2.0, n_new)
        h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new)

def structured_perturbations(h, n, rng, num_pert=8):
    results = []
    for _ in range(num_pert):
        h_pert = h.copy()
        pert_type = rng.integers(0, 5)
        if pert_type == 0:
            shift = rng.integers(2, max(3, n // 20))
            h_pert = np.roll(h_pert, shift)
        elif pert_type == 1:
            center = rng.integers(0, n)
            width = rng.integers(5, max(6, n // 10))
            amp = rng.uniform(0.02, 0.08)
            for i in range(n):
                d = min((i - center) % n, n - (i - center) % n)
                if d < width:
                    h_pert[i] += amp * np.cos(np.pi * d / width)
        elif pert_type == 2:
            for _ in range(3):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 != i2:
                    swap = (h_pert[i1] + h_pert[i2]) / 2.0
                    h_pert[i1] = swap + rng.uniform(-0.05, 0.05)
                    h_pert[i2] = swap - rng.uniform(-0.05, 0.05)
        elif pert_type == 3:
            mirror_start = rng.integers(0, n // 2)
            mirror_len = rng.integers(n // 8, n // 3)
            mirror_len = min(mirror_len, n - mirror_start)
            seg = h_pert[mirror_start:mirror_start + mirror_len].copy()
            h_pert[mirror_start:mirror_start + mirror_len] = seg[::-1]
        else:
            h_pert += rng.normal(0, 0.02, n)
        h_pert = project_feasible(h_pert)
        results.append(h_pert)
    return results

def short_lbfgs(h, deadline, max_seconds=20):
    n = len(h)
    best_h = project_feasible(h)
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    alphas = [100000, 500000, 2000000]
    c_vec = compute_c5_vec(best_h)
    active = np.argsort(-c_vec)[:3]
    for alpha in alphas:
        if time.time() > deadline - max_seconds / 3:
            break
        maxit = max(100, int((deadline - time.time()) * 2))
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(n, active, alpha, 0.002), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': maxit * 5,
                                    'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                c_vec = compute_c5_vec(best_h)
                active = np.argsort(-c_vec)[:3]
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_c = compute_c5(h0)
    best_n = n_orig

    # Phase 1: Quick L-BFGS polish at original resolution (40s)
    p1_deadline = t0 + 40
    h1, c1 = run_lbfgs(best_h, p1_deadline,
                        alphas=[50, 200, 1000, 5000, 50000],
                        mu=0.01, maxiters_list=[500, 800, 1000, 1200, 1500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Cubic upsample to n=750 and optimize (220s)
    n_mid = 750
    p2_deadline = t0 + 260
    if time.time() < p2_deadline - 60:
        h_mid = upsample_cubic(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid)
        if c_mid < best_c + 0.002:
            h2, c2 = run_lbfgs(h_mid, p2_deadline,
                                alphas=[10, 50, 200, 1000, 5000, 50000, 200000, 1000000, 5000000],
                                mu=0.005, maxiters_list=[400, 600, 800, 1000, 1200, 1500, 1800, 2000, 2500])
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_mid

    # Phase 3: Try n=1000 (150s)
    n_high = 1000
    p3_deadline = t0 + 410
    if time.time() < p3_deadline - 50 and best_n < n_high:
        h_high = upsample_cubic(best_h, best_n, n_high)
        c_high = compute_c5(h_high)
        if c_high < best_c + 0.003:
            h3, c3 = run_lbfgs(h_high, p3_deadline,
                                alphas=[10, 50, 200, 1000, 5000, 50000, 500000],
                                mu=0.003, maxiters_list=[400, 600, 800, 1000, 1200, 1500, 2000])
            if c3 < best_c:
                best_c = c3
                best_h = h3
                best_n = n_high

    # Phase 4: Aggressive active set optimization (180s)
    p4_deadline = t0 + 590
    h4, c4 = run_active_opt(best_h, p4_deadline, top_k=4, n_rounds=40)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Block coordinate descent (100s)
    p5_deadline = t0 + 690
    h5, c5 = block_coordinate_descent(best_h, p5_deadline, block_size=10)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Active set again after block descent (80s)
    p6_deadline = t0 + 770
    h6, c6 = run_active_opt(best_h, p6_deadline, top_k=3, n_rounds=25)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Structured restarts (120s)
    p7_deadline = t0 + 890
    perturbations = structured_perturbations(best_h, best_n, rng, num_pert=10)
    for hp in perturbations:
        if time.time() > p7_deadline - 15:
            break
        c_p = compute_c5(hp)
        if c_p < best_c + 0.005:
            p7_sub = min(time.time() + 10, p7_deadline - 10)
            h7, c7 = short_lbfgs(hp, p7_sub, max_seconds=10)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Final active set push (60s)
    p8_deadline = t0 + 950
    h8, c8 = run_active_opt(best_h, p8_deadline, top_k=2, n_rounds=20)
    if c8 < best_c:
        best_c = c8
        best_h = h8

    # Phase 9: Final block descent (30s)
    p9_deadline = t0 + 980
    h9, c9 = block_coordinate_descent(best_h, p9_deadline, block_size=5)
    if c9 < best_c:
        best_c = c9
        best_h = h9

    # Phase 10: Ultra-fine active set (remaining)
    p10_deadline = deadline - 2
    c_vec = compute_c5_vec(best_h)
    active = np.argsort(-c_vec)[:2]
    n = len(best_h)
    bounds = [(0.0, 1.0)] * n
    for alpha in [10000000, 50000000]:
        if time.time() > p10_deadline - 3:
            break
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(n, active, alpha, 0.0005),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-23,
                                    'gtol': 1e-19, 'maxfun': 2000,
                                    'maxls': 30, 'maxcor': 30})
            h10 = project_feasible(res.x)
            c10 = compute_c5(h10)
            if c10 < best_c:
                best_c = c10
                best_h = h10
        except Exception:
            pass

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    return best_h, best_c, final_n
