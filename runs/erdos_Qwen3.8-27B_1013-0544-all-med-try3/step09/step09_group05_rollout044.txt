```python
def solve_linearized_qp(h, n, active_shifts, max_iter=20):
    """Solve the linearized QCQP at current point."""
    from scipy.optimize import linprog
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        c_vec = compute_c5_vec(h, n)
        # Recompute active set
        max_c = c_vec.max()
        thresh = max_c - max(1e-8, 0.0001 * max_c)
        active = np.where(c_vec >= thresh)[0]
        
        # Build linearized constraints
        # Variables: h[0],...,h[n-1], λ
        # c_k(h) ≈ c_k(h*) + Σ_i ∂c_k/∂h[i] * (h[i] - h*[i])
        # We want: c_k(h*) + Σ_i grad_k[i] * (h[i] - h*[i]) ≤ λ
        # i.e.: Σ_i grad_k[i] * h[i] - λ ≤ -c_k(h*) + Σ_i grad_k[i] * h*[i]
        
        n_vars = n + 1  # h[0..n-1] and λ
        # Objective: minimize λ = variable n
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0
        
        A_ub = []
        b_ub = []
        
        for k_idx in active:
            k = k_idx - (n - 1)
            # Compute gradient of c_k at h
            grad_k = np.zeros(n)
            if k >= 0:
                nv = n - k
                grad_k[0:nv] = (1.0 - h[k:n])
                grad_k[k:n] += (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad_k[kk:n] = (1.0 - h[0:nv])
                grad_k[0:nv] += (-h[kk:n])
            
            # c_k(h) ≈ c_k(h*) + grad_k · (h - h*) ≤ λ
            # grad_k · h - λ ≤ c_k(h*) - grad_k · h*
            row = np.zeros(n_vars)
            row[0:n] = grad_k * dx
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(c_vec[k_idx] - np.dot(grad_k, h) * dx)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds: 0 ≤ h[i] ≤ 1, λ ≥ 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        # Equality constraint: sum(h) = n/2
        A_eq = np.zeros((1, n_vars))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([n / 2.0])
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.status == 0:
            h_new = res.x[0:n]
            lambda_new = res.x[n]
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
            else:
                break
        else:
            break
    
    return h, best_c
```

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h

                break
            h += (diff / tot) * room

            if tot < 1e-16:
