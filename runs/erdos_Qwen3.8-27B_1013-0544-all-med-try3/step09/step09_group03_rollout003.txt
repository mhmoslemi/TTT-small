# Power-sum objective
def psum_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    val = (np.sum(c**p))**(1.0/p)
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def psum_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    S = np.sum(c**p)
    cp = c**(p-1)
    # d(S^(1/p))/dx = (1/p) * S^(1/p - 1) * p * cp * dc/dx = S^(1/p-1) * cp * dc/dx
    factor = S**(1.0/p - 1.0)
    # dc/dx is like the gradient of the linear part
    dx_val = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2*n - 1
    center = n - 1
    for idx in range(n2m1):
        w = factor * cp[idx] * dx_val
        if w < 1e-15:
            continue
        k = idx - center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += w * (1.0 - x[k:n])
                grad[k:n] += w * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += w * (1.0 - x[0:nv])
                grad[0:nv] += w * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def psum_obj_fn(x, n, p, mu):
    c = compute_c5_vec(x, n)
    S = np.sum(c**p)
    val = S**(1.0/p)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def psum_grad_fn(x, n, p, mu):
    c = compute_c5_vec(x, n)
    S = np.sum(c**p)
    cp = c**(p-1)
    factor = S**(1.0/p - 1.0)
    dx_val = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2*n - 1
    center = n - 1
    for idx in range(n2m1):
        w = factor * cp[idx] * dx_val
        if w < 1e-16:
            continue
        k = idx - center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += w * (1.0 - x[k:n])
                grad[k:n] += w * (-x[0:nv])
        else:

            nv = n - kk
            if nv > 0:
                grad[kk:n] += w * (1.0 - x[0:nv])
                grad[0:nv] += w * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def lse_obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def lse_grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2*n - 1
    center = n - 1
    for idx in range(n2m1):

def run_lbfgs_psum(h, n, deadline, p_list=None, mu=1.0):
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128, 256]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if time.time() > deadline - 20:

def slqp_polish(h, n, deadline, p=64, mu=5.0, maxiter=3000):
    def obj_fn(x, na, pp, mm):
        c = compute_c5_vec(x, na)
        S = np.sum(c**pp)
        return float(S**(1.0/pp) + mm*(x.sum()-na/2.0)**2)
    def jac_fn(x, na, pp, mm):
        return psum_grad_fn(x, na, pp, mm)
    def sq_fn(x, na):
        return float(x.sum() - na/2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, args=(n, p, mu),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def local_search_coord(h, n, deadline, n_rounds=2000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(777)
    step_sizes = [0.001, 0.0005, 0.002, 0.0001, 0.005, 0.0002]
    
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        if r % 100 == 0:
            best_c = compute_c5(best_h, n)
        idx_pick = rng.randint(n)
        delta = step_sizes[r % len(step_sizes)]
        # Try increasing
        nv = best_h[idx_pick] + delta
        if nv <= 1.0 + 1e-12:
            best_h[idx_pick] = min(nv, 1.0)
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
            else:
                best_h[idx_pick] = nv
        # Try decreasing
        nv2 = best_h[idx_pick] - delta
        if nv2 >= -1e-12:
            best_h[idx_pick] = max(nv2, 0.0)
            c2 = compute_c5(best_h, n)
            if c2 < best_c - 1e-15:
                best_c = c2
            else:
                best_h[idx_pick] = nv2
        # Pair move: increase one, decrease another
        if r % 3 == 1:
            idx2 = rng.randint(n)
            if idx2 != idx_pick:
                d = 0.002
                best_h[idx_pick] = min(best_h[idx_pick] + d, 1.0)
                best_h[idx2] = max(best_h[idx2] - d, 0.0)
                hp = best_h.copy()
                c3 = compute_c5(best_h, n)
                if c3 < best_c - 1e-15:
                    best_c = c3
                else:
                    best_h = hp
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_push(h, n, threshold=0.5):
    h_b = np.where(h > threshold, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if diff > 0:
        frac = diff
        order = np.argsort(-(h_b + 0.1*(h - h_b)))
        count = int(frac)
        for i in range(count):
            h_b[order[i]] = 1.0
        rem = frac - count
        if rem > 1e-12 and count < len(order):
            h_b[order[count]] = rem
    elif diff < 0:
        frac = -diff
        order = np.argsort(h_b + 0.1*(h - h_b))
        count = int(frac)
        for i in range(count):
            h_b[order[i]] = 0.0
        rem = frac - count
        if rem > 1e-12 and count < len(order):
            h_b[order[count]] = 1.0 - rem
    return project_feasible(h_b, n)

    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
