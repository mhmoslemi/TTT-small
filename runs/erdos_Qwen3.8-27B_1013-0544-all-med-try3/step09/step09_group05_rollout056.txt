```python
import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda for sum constraint
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Need to decrease; reduce values proportionally to their "headroom"
            room = h  # can decrease by h[i]
            if room.sum() < 1e-15:
                break
            h -= (diff / room.sum()) * room
        else:
            # Need to increase; increase values proportionally to their "room"
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += (-diff / room.sum()) * room
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Compute gradient efficiently
    # For each shift index mm, the gradient contribution is w[mm] * dx * grad_c[mm]
    # grad_c[mm][j] = (1-x[j+shift]) - x[j-shift] (with bounds)
    for mm in range(2*n-1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def nesterov_optimize(h, n, deadline, alpha_start=1000, n_iter_max=200000):
    """Nesterov accelerated projected gradient method."""
    h = project_feasible(h, n).copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    t_k = 1.0
    alpha = alpha_start
    gamma = 1.0 / (n * alpha * 2.0)  # initial step size estimate
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        
        # Nesterov momentum
        mu = (t_k - 1) / (t_k + 1)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0, 1)
        
        # Gradient at y
        g = smooth_max_grad(y, n, alpha)
        
        # Step
        x_new = h - gamma * g
        x_new = project_feasible(x_new, n)
        
        # Check improvement
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        
        # Backtracking line search
        if c_new >= compute_c5(h, n) * (1 - 1e-10):
            gamma *= 0.5
            x_new = h - gamma * g
            x_new = project_feasible(x_new, n)
        
        h_prev = h
        h = x_new
        t_k = (1 + math.sqrt(1 + 4*t_k*t_k)) / 2
        
        # Adaptive alpha: increase slowly
        if it % 10000 == 0 and it > 0:
            alpha *= 1.5
            gamma /= 1.5
    
    return best_h, best_c
```

```python
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = m + np.log(np.sum(np.exp(sh))) / alpha
            return float(val)
        
        def sum_con(x, n):
            return x.sum() - n / 2.0
        
        def sum_jac(x, n):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        
        try:
            res = minimize(obj_eq, best_h, args=(n, 500000, active), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def nesterov_optimize(h, n, deadline, alpha=200000, lr_init=None):
    h = project_feasible(h.copy(), n)
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    t_k = 1.0
    if lr_init is None:
        lr = 1.0 / (n * 2.0 * alpha * 0.01)
    else:
        lr = lr_init
    
    stall_count = 0
    for it in range(500000):
        if time.time() > deadline:
            break
        
        mu_mom = (t_k - 1.0) / (t_k + 1.0)
        y = h + mu_mom * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        g = smooth_grad(y, n, alpha, 0.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        
        x_new = h - lr * g
        x_new = project_feasible(x_new, n)
        
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 500:
                lr *= 0.7
                stall_count = 0
        
        h_prev = h
        h = x_new
        t_k = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        if it % 20000 == 19999:
            g = smooth_grad(h, n, alpha, 0.0)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-10:
                break
    
    return best_h, best_c

def efficient_pair_search(h, n, deadline, top_shifts_k=8, top_coords=40):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(0.001 * max_val, 0.0002)
        active_shifts = np.where(c_vec >= threshold)[0]
        if len(active_shifts) > top_shifts_k:
            order = np.argsort(-c_vec[active_shifts])
            active_shifts = active_shifts[order[:top_shifts_k]]
        
        # Compute gradient for sensitivity
        g = smooth_grad(best_h, n, 10000, 0.0)
        # Top coordinates by |gradient|
        coord_idx = np.argsort(-np.abs(g))[:top_coords]
        
        improved = False
        for step in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() > deadline:
                return best_h, best_c
            for ii in range(len(coord_idx)):
                i = coord_idx[ii]
                for jj in range(len(coord_idx)):
                    if ii == jj:
                        continue
                    j = coord_idx[jj]
                    
                    # Try h[i]+step, h[j]-step
                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        best_h[i] += step
                        best_h[j] -= step
                        c_new = max(compute_c5_vec(best_h, n)[active_shifts])
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    
                    # Try h[i]-step, h[j]+step
                    if best_h[i] - step >= 0.0 and best_h[j] + step <= 1.0:
                        best_h[i] -= step
                        best_h[j] += step
                        c_new = max(compute_c5_vec(best_h, n)[active_shifts])
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
        
        if not improved:
            break
    
    return best_h, best_c

def block_shift_search(h, n, deadline, block_sizes=[2, 3, 5]):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for bs in block_sizes:
        for _ in range(5):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for i in range(0, n - bs, 1):
                if time.time() > deadline:
                    return best_h, best_c
                # Try shifting block [i, i+bs) right by 1
                if i + bs < n:
                    old_i = best_h[i]
                    # Shift: h[i:i+bs+1] -> h[i+1:i+bs+1], h[i] stays
                    best_h[i+bs] += best_h[i+bs] - best_h[i+bs]  # no change needed for sum
                    # Actually just rotate: move h[i] to h[i+bs] and shift others? No.
                    # Simpler: just try swapping h[i] and h[i+bs]
                    best_h[i], best_h[i+bs] = best_h[i+bs], best_h[i]
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[i], best_h[i+bs] = best_h[i+bs], best_h[i]
            
            if not improved:
                break
    
    return best_h, best_c

def perturbed_restart(h, n, deadline, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(12345)
    
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        noise_scale = 0.02 / (r + 1)
        h_pert = best_h + rng.normal(0, noise_scale, size=n)
        h_pert = project_feasible(h_pert, n)
        
        t_sub = min(120, deadline - time.time() - 30)
        if t_sub < 20:
            break
        
        # Quick L-BFGS on perturbed
        h_new, c_new = lbfgs_stage(h_pert, n, 50000, 1.0, time.time() + t_sub, maxiter=5000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # Active set refinement
        if time.time() < deadline - 30:
            h_new, c_new = active_set_lbfgs(h_new, n, time.time() + 30, top_k=5)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Progressive L-BFGS with increasing alpha
    configs_p1 = [(500, 1.0), (2000, 10.0), (10000, 100.0),
                  (100000, 1000.0), (500000, 5000.0), (2000000, 20000.0),
                  (10000000, 100000.0)]
    t_per = min(60, (deadline - t0) / (len(configs_p1) + 4))
    for alpha, mu in configs_p1:
        if time.time() > deadline - 300:
            break
        t_end = min(time.time() + t_per, deadline - 280)
        h_new, c_new = lbfgs_stage(best_h, n, alpha, mu, t_end, maxiter=10000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Active set focused optimization
    if time.time() < deadline - 150:
        t_end = min(time.time() + 120, deadline - 130)
        h_new, c_new = active_set_lbfgs(best_h, n, t_end, top_k=10)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: SLSQP equalization
    if time.time() < deadline - 120:
        t_end = min(time.time() + 90, deadline - 100)
        h_new, c_new = slqp_equalize(best_h, n, t_end, top_k=6)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Nesterov with high alpha
    if time.time() < deadline - 100:
        t_end = min(time.time() + 80, deadline - 80)
        h_new, c_new = nesterov_optimize(best_h, n, t_end, alpha=1000000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Another active set round with tighter threshold
    if time.time() < deadline - 80:
        t_end = min(time.time() + 60, deadline - 60)
        h_new, c_new = active_set_lbfgs(best_h, n, t_end, top_k=5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 6: Efficient pair search
    if time.time() < deadline - 50:
        t_end = min(time.time() + 45, deadline - 30)
        h_new, c_new = efficient_pair_search(best_h, n, t_end, top_shifts_k=10, top_coords=50)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 7: Block shift search
    if time.time() < deadline - 30:
        t_end = min(time.time() + 20, deadline - 15)
        h_new, c_new = block_shift_search(best_h, n, t_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 8: Perturbed restarts
    if time.time() < deadline - 40:
        t_end = min(time.time() + 35, deadline - 10)
        h_new, c_new = perturbed_restart(best_h, n, t_end, n_restarts=2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 9: Final L-BFGS polish at very high alpha
    if time.time() < deadline - 10:
        t_end = deadline - 3
        h_new, c_new = lbfgs_stage(best_h, n, 5000000, 5000.0, t_end, maxiter=20000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Final pair search with very small steps
    if time.time() < deadline - 5:
        t_end = deadline - 2
        h_new, c_new = efficient_pair_search(best_h, n, t_end, top_shifts_k=15, top_coords=60)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Final projection and evaluation
    final_h = project_feasible(best_h, n)
    final_c = compute_c5(final_h, n)
    
    return final_h, final_c, n
```
