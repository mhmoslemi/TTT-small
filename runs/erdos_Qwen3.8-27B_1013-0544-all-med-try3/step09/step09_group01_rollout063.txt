```python
def solve_1d_minmax(A, B):
    """Minimize max_k (A[k]*t + B[k]) over t in [0,1]."""
    K = len(A)
    
    # Check endpoints
    f0 = np.max(B)  # t=0
    f1 = np.max(A + B)  # t=1
    
    # If all slopes are the same sign, the answer is at an endpoint
    if np.all(A >= 0):
        return 0.0
    if np.all(A <= 0):
        return 1.0
    
    # Find breakpoints in [0,1]
    # Breakpoint between lines k and j: t = (B[j]-B[k])/(A[k]-A[j])
    # Collect all breakpoints in [0,1]
    candidates = [0.0, 1.0]
    for k in range(K):
        for j in range(k+1, K):
            denom = A[k] - A[j]
            if abs(denom) > 1e-14:
                t = (B[j] - B[k]) / denom
                if -1e-10 <= t <= 1 + 1e-10:
                    candidates.append(max(0.0, min(1.0, t)))
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        val = np.max(A * t + B)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t
```

```python
    candidates = T[mask]
    candidates = np.concatenate([[0.0, 1.0], candidates])
    
    # Evaluate
    vals = np.max(A[:, None] * candidates[None, :] + B[:, None], axis=0)
    idx = np.argmin(vals)
    
    return float(candidates[idx])
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c_raw(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def solve_1d_minmax(A, B):
    K = len(A)
    if K == 0:
        return 0.5
    if K == 1:
        if A[0] > 1e-12:
            return 0.0
        if A[0] < -1e-12:
            return 1.0
        return 0.5
    ad = A[:, None] - A[None, :]
    bd = B[None, :] - B[:, None]
    safe = np.where(np.abs(ad) > 1e-14, ad, 1.0)
    T = bd / safe
    T[np.abs(ad) <= 1e-14] = np.nan
    T = T[~np.isnan(T)]
    T = T[(T >= -1e-12) & (T <= 1.0 + 1e-12)]
    T = np.clip(T, 0.0, 1.0)
    cands = np.concatenate(([0.0, 1.0], T))
    vals = np.max(A[:, None] * cands[None, :] + B[:, None], axis=0)
    return float(cands[np.argmin(vals)])


def run_coord_descend(h, n, deadline, max_sweeps=50000, top_k=30,
                      recompute_every=5, project_every=10, damping=1.0):
    h = project_feasible(h.astype(np.float64).copy(), n)
    c_full = compute_c_raw(h, n)
    dx = 2.0 / n
    best_c = float(c_full.max()) * dx
    best_h = h.copy()
    q_base = n - 1
    n_c = len(c_full)

    kn = min(top_k, n_c)
    active = np.argpartition(-c_full, kn - 1)[:kn]

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        if sweep % recompute_every == 0:
            c_full = compute_c_raw(h, n)
            cmax = c_full.max()
            thresh = cmax - max(1e-10, 5e-5 * cmax)
            act = np.where(c_full >= thresh)[0]
            if len(act) > top_k:
                act = act[np.argsort(-c_full[act])[:top_k]]
            if len(act) == 0:
                act = np.array([int(np.argmax(c_full))])
            active = act

        q_arr = active - q_base
        na = len(q_arr)

        for i in range(n):
            ov = h[i]
            m1 = i + q_arr
            m2 = i - q_arr
            v1 = (m1 >= 0) & (m1 < n)
            v2 = (m2 >= 0) & (m2 < n)

            A = np.zeros(na)
            np.add.at(A, np.where(v1)[0], 1.0 - h[m1[v1]])
            np.add.at(A, np.where(v2)[0], -h[m2[v2]])
            
            B = c_full[active] - A * ov
            t = solve_1d_minmax(A, B)

            d = damping * (t - ov)
            if abs(d) < 1e-15:
                continue
            nv = max(0.0, min(1.0, ov + d))
            ad = nv - ov
            if abs(ad) < 1e-15:
                continue
            h[i] = nv
            c_full[active] += ad * A

        if (sweep + 1) % project_every == 0:
            h = project_feasible(h, n)
            c_val = compute_c5(h, n)
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()

    h = project_feasible(h, n)
    c_val = compute_c5(h, n)
    if c_val < best_c:
        best_c = c_val
        best_h = h.copy()

    return best_h, best_c
```
