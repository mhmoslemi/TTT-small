    #   d/dh[i] = -dx*h[i+kk] for i < n-kk  (as the (1-h[i]) term)
    #   d/dh[i] = dx*(1-h[i-kk]) for i >= kk  (as the h[i+kk] term)
    
    grad = np.zeros(n)
    # Process positive lags (k >= 0)
    for k in range(0, n):
        if w[n-1+k] < 1e-15:
            continue
        nv = n - k
        wm = w[n-1+k] * dx
        grad[0:nv] += wm * (1.0 - x[k:n])
        grad[k:n] += wm * (-x[0:nv])
    # Process negative lags (k < 0, i.e., kk > 0)
    for kk in range(1, n):
        if w[n-1-kk] < 1e-15:
            continue
        nv = n - kk
        wm = w[n-1-kk] * dx
        grad[kk:n] += wm * (1.0 - x[0:nv])
        grad[0:nv] += wm * (-x[kk:n])
    return grad
```

This is O(n²) per gradient call. For n=1000, that's 10^6 operations (vectorized), which should take a few ms.

But wait, with the log-sum-exp weights, in practice only a few lags will have significant weight (the ones near the max). So we can skip the loop for most k.

Let me add that optimization:

```python
def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    dx = 2.0 / n
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    active = np.where(w > 1e-12)[0]
    for j in active:
        k = j - (n - 1)
        wm = w[j] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad
```

This is much faster when alpha is large (few active lags).

Now, for the optimization strategy, let me think about what would be most effective:

1. **Phase 1**: Start from initial_h_values (n=500), do a quick L-BFGS to get a good starting point.

2. **Phase 2**: Interpolate to n=1000, do L-BFGS with increasing alpha.

3. **Phase 3**: Do a "subgradient" refinement: identify the active lags (where c_j is close to max), compute their gradients, and take a step in the negative gradient direction with line search on the exact objective.

4. **Phase 4**: Coordinate descent / local search: for each coordinate i, try small adjustments and keep the best.

5. **Phase 5**: Try to push further with higher alpha and more iterations.

Let me also think about a "push-pull" strategy:
- For the active lags, we want to reduce c_j.
- The gradient of c_j w.r.t. h[i] tells us which direction reduces c_j.
- Taking a step in the negative gradient direction (weighted by the active lags) should reduce the max.

But we need to maintain feasibility (0 ≤ h ≤ 1, sum(h) = n/2).

A simple approach: 
- Compute the average gradient of active lags
- Take a step: h_new = h - step_size * avg_grad
- Project back to feasible set
- Accept if C₅ improved

This is essentially a subgradient method with projection.

Let me code this up properly. I'll also add a "local coordinate descent" that, for each position i, tries to adjust h[i] by a small amount (transferring mass to/from another position j) to reduce C₅.

Actually, for the constraint sum(h) = n/2, we can't just change h[i] independently. We need to compensate by changing another position. So a "move" is: h[i] += δ, h[j] -= δ (or vice versa).

The local search would be:
For each i, for each j ≠ i:
  Try h[i] += δ, h[j] -= δ (if feasible)
  Try h[i] -= δ, h[j] += δ (if feasible)
  Keep the move if C₅ decreased

This is O(n³) per full pass (n² pairs, each requiring O(n²) to compute C₅). Too slow for n=1000.

Better: for a fixed i, find the best j to transfer to/from. And use the gradient to guide the search.

Even better: compute the gradient of C₅ (or the smooth approximation), and do a few steps of projected gradient descent.

Let me write the final code:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 700:
        val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    return float(val)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for j in active:
        k = j - (n - 1)
        wm = w[j] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def subgradient_step(h, n, step_sizes, deadline):
    """Take subgradient steps on the exact max objective."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    
    for step in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            m = c_vec.max()
            # Active lags: those within 1% of max
            thresh = m - max(1e-10, 0.005 * m)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            
            # Compute average gradient of active lags
            dx = 2.0 / n
            grad = np.zeros(n)
            for j in active:
                k = j - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
            grad /= len(active)
            
            # Projected gradient step
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                improved = True
    
    return best_h, best_c

def local_search(h, n, deadline, top_k=100):
    """Coordinate descent: try exchanging mass between pairs of positions."""
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            m = c_vec.max()
            thresh = m - max(1e-8, 0.003 * m)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute gradient magnitude to prioritize positions
            dx = 2.0 / n
            grad_mag = np.zeros(n)
            for j in active:
                k = j - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad_mag[0:nv] += np.abs((1.0 - h[k:n])) * dx
                    grad_mag[k:n] += np.abs(h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad_mag[kk:n] += np.abs((1.0 - h[0:nv])) * dx
                    grad_mag[0:nv] += np.abs(h[kk:n]) * dx
            
            # Focus on top positions by gradient magnitude
            top_idx = np.argsort(-grad_mag)[:top_k]
            
            for i in top_idx:
                if time.time() > deadline:
                    return h, best_c
                # Try transferring to/from nearby positions
                lo = max(0, int(i) - 50)
                hi = min(n, int(i) + 51)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 + 1e-10 and h[j] - step >= -1e-10:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # h[i] -= step, h[j] += step
                    if h[i] - step >= -1e-10 and h[j] + step <= 1.0 + 1e-10:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    
    return h, best_c

def sqp_step(h, n, active_idx, rho, deadline):
    """One SQP step using projected gradient."""
    from scipy.optimize import linprog
    c_vec = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    # Build gradient for active constraints
    grads = np.zeros((len(active_idx), n))
    c_vals = np.zeros(len(active_idx))
    for j, idx in enumerate(active_idx):
        k = idx - (n - 1)
        c_vals[j] = c_vec[idx]
        if k >= 0:
            nv = n - k
            grads[j, 0:nv] += (1.0 - h[k:n]) * dx
            grads[j, k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grads[j, kk:n] += (1.0 - h[0:nv]) * dx
            grads[j, 0:nv] += (-h[kk:n]) * dx
    
    # LP: minimize t subject to:
    #   grad_j · delta + c_j ≤ t  for each active j
    #   -rho ≤ delta_i ≤ rho
    #   sum(delta) = 0
    #   0 ≤ t ≤ 2
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active_idx) + 2*n, n_var))
    b_ub = np.zeros(len(active_idx) + 2*n)
    
    for j in range(len(active_idx)):
        A_ub[j, 0:n] = grads[j]
        A_ub[j, n] = -1.0
        b_ub[j] = -c_vals[j]
    
    for i in range(n):
        A_ub[len(active_idx) + i, i] = -1.0
        b_ub[len(active_idx) + i] = h[i]
        A_ub[len(active_idx) + n + i, i] = 1.0
        b_ub[len(active_idx) + n + i] = 1.0 - h[i]
    
    # Add step size constraint
    # Actually let's keep it simple: -rho ≤ delta_i ≤ rho
    # Replace the box constraints above
    for i in range(n):
        A_ub[len(active_idx) + i, i] = -1.0
        b_ub[len(active_idx) + i] = rho
        A_ub[len(active_idx) + n + i, i] = 1.0
        b_ub[len(active_idx) + n + i] = rho
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=200, rho_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        thresh = m - max(1e-10, 0.002 * m)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        # Limit active set for LP size
        if len(active) > 30:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:30]]
        
        delta, t_opt = sqp_step(h, n, active, rho, deadline)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.5, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-10 or no_improve > 30:
                break
    
    return best_h, best_c
