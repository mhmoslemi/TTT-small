def corr_change_single_vec(h, n, i, delta):

    # k >= 0 part: indices n-1 to 2n-2
    ks = np.arange(n)  # k = 0, 1, ..., n-1
    mask1 = ks <= i  # k <= i
    mask2 = ks <= n-1-i  # k <= n-1-i

    part1 = np.zeros(n)
    part1[mask1] = (1.0 - h[i - ks[mask1]])
    part2 = np.zeros(n)
    part2[mask2] = h[i + ks[mask2]]
    dc[n-1:] = delta * (part1 - part2)
    
    # k < 0 part: kk = 1, ..., n-1, indices n-1-kk = 0 to n-2
    kks = np.arange(1, n)

    part3 = np.zeros(n-1)

    part4 = np.zeros(n-1)
    part4[mask4] = h[i - kks[mask4]]
    dc[:n-1] = delta * (part3 - part4)
    
    return dc

            if tot < 1e-15:

                         bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-20,
                                  'gtol': 1e-15, 'maxfun': 20000, 'maxls': 50})
        else:
            res = minimize(active_smooth_obj, h, jac=active_smooth_grad,
                         args=(n, active, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-20,
                                  'gtol': 1e-15, 'maxfun': 20000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def get_active_set(h, n, top_k=10, threshold_frac=0.001):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, threshold_frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def perturb_h(h, n, sigma, rng):
    """Add small noise and project back to feasible."""
    noise = rng.normal(0, sigma, n)
    h_pert = h + noise
    h_pert = project_feasible(h_pert, n)
    return h_pert

def slqp_optimize(h, n, deadline, alpha=200000, maxiter=2000):
    def sq_obj(x):
        return float(x.sum() - n / 2.0)
    def sq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_obj, 'jac': sq_jac}]
    try:
        res = minimize(full_smooth_obj, h, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(300, 0.5)
        n_orig = 300
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS at original resolution
    t1 = min(t0 + 60, deadline - 500)
    if t1 > t0 + 5:
        h0, best_c = lbfgs_optimize(h0, n_orig, t1,
            alpha_list=[100, 1000, 10000],
            mu_list=[10, 100, 1000])
        best_h = h0.copy()
    
    # Work at n=500
    n = 500
    if n_orig != n:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h500 = np.interp(x_new, x_old, best_h)
        h500 = project_feasible(h500, n)
    else:
        h500 = best_h.copy()
    
    c500 = compute_c5(h500, n)
    if c500 <= best_c:
        best_h = h500.copy()
        best_c = c500
    h_work = h500.copy()
    
    # Phase 2: L-BFGS at n=500
    t2 = min(t0 + 120, deadline - 400)
    if time.time() < t2:
        h_work, c_work = lbfgs_optimize(h_work, n, t2,
            alpha_list=[500, 5000, 50000, 200000],
            mu_list=[50, 500, 5000, 50000])
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # Phase 3: Active set optimization
    if time.time() < deadline - 300:
        for _ in range(3):
            if time.time() > deadline - 300:
                break
            active = get_active_set(h_work, n, top_k=10, threshold_frac=0.002)
            t3 = min(time.time() + 60, deadline - 300)
            h_work, c_work = lbfgs_optimize(h_work, n, t3,
                alpha_list=[5000, 50000],
                mu_list=[100, 1000],
                active=active)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
            # Update active set
            active = get_active_set(h_work, n, top_k=8, threshold_frac=0.001)
            t3b = min(time.time() + 40, deadline - 300)
            h_work, c_work = lbfgs_optimize(h_work, n, t3b,
                alpha_list=[1000, 10000, 100000],
                mu_list=[50, 500, 5000],
                active=active)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 200:
        t4 = min(time.time() + 60, deadline - 200)
        h_work, c_work = slqp_optimize(h_work, n, t4, alpha=200000, maxiter=2000)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # Phase 5: Fast local search (MAIN IMPROVEMENT)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 500, deadline - 80)
        h_work, c_work = fast_local_search_v3(h_work, n, t5, radius=20)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # Phase 6: Another L-BFGS round + local search
    if time.time() < deadline - 50:
        active = get_active_set(best_h, n, top_k=8, threshold_frac=0.001)
        t6 = min(time.time() + 30, deadline - 50)
        h_work, c_work = lbfgs_optimize(best_h, n, t6,
            alpha_list=[10000, 100000],
            mu_list=[1000, 10000],
            active=active)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
        
        if time.time() < deadline - 15:
            t7 = deadline - 10
            h_work, c_work = fast_local_search_v3(best_h, n, t7, radius=15)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
    
    # Phase 7: Try perturbation + re-optimize
    if time.time() < deadline - 30:
        for sigma in [0.001, 0.0005]:
            if time.time() > deadline - 25:
                break
            h_pert = perturb_h(best_h, n, sigma, rng)
            tp = min(time.time() + 15, deadline - 20)
            h_pert, c_pert = lbfgs_optimize(h_pert, n, tp,
                alpha_list=[10000], mu_list=[1000])
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()
                if time.time() < deadline - 10:
                    h_pert, c_pert = fast_local_search_v3(h_pert, n, deadline - 5, radius=10)
                    if c_pert < best_c:
                        best_c = c_pert
                        best_h = h_pert.copy()
    
    # Final: verify and return
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Also check at original resolution if different
    if n_orig != n:
        x_new = np.linspace(0, 2, n)
        x_old = np.linspace(0, 2, n_orig)
        h_orig_res = np.interp(x_old, x_new, best_h)
        h_orig_res = project_feasible(h_orig_res, n_orig)
        c_orig = compute_c5(h_orig_res, n_orig)
        if c_orig < best_c:
            best_h = h_orig_res
            best_c = c_orig
            n = n_orig
    
    return best_h, best_c, n
