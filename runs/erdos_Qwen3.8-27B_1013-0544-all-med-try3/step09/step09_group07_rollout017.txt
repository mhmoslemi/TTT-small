    c = np.correlate(h, 1.0 - h, mode='full')

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def smooth_max_val(c, alpha):
    mc = c.max()
    shifted = alpha * (c - mc)
    mx = shifted.max()

        return mc + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:

    else:

def subgradient_step(h, n, step_size, deadline):
    """One step of projected subgradient method on the exact max."""
    c = compute_all_lags(h, n)
    max_idx = np.argmax(c)
    # Check how many lags are close to max
    threshold = c[max_idx] - 1e-8
    active = np.where(c >= threshold)[0]
    
    # Average gradient over active lags
    g_total = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag == 0:
            g_total += (1.0 - 2.0 * h)
        elif lag > 0:
            g_total[lag:] += (1.0 - h[:n-lag])
            g_total[:n-lag] += (-h[lag:])
        else:
            m = -lag
            g_total[:n-m] += (1.0 - h[m:])
            g_total[m:] += (-h[:n-m])
    
    g_total /= len(active)
    g_total *= (2.0 / n)

    h_new = h - step_size * g_total

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    mc = c.max()
    shifted = alpha * (c - mc)
    mx = shifted.max()
    if mx > 500:
        return float(mc + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    else:
        return float(mc + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    mc = c.max()
    shifted = alpha * (c - mc)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c_raw * dx, alpha)
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        wm = w[n - 1 + m]
        if wm > 1e-14:
            grad[m:] += wm * (1.0 - h[:n - m])
            grad[:n - m] += wm * (-h[m:])
        wm2 = w[n - 1 - m]
        if wm2 > 1e-14:
            grad[:n - m] += wm2 * (1.0 - h[m:])
            grad[m:] += wm2 * (-h[:n - m])
    return grad * dx

def grad_single_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0 and lag < n:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] += (-h[lag:])
    elif lag < 0:
        m = -lag
        if m < n:
            g[:n - m] += (1.0 - h[m:])
            g[m:] += (-h[:n - m])
    return g * dx

def lbfgsb_opt(h, n, alpha, maxiter=3000, ftol=1e-20):
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        val = smooth_max_val(np.correlate(x, 1.0 - x, mode='full') * (2.0 / n), alpha)
        g = grad_smooth_max(x, n, alpha)
        return val, g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-15,
                                'maxls': 100, 'iprint': -1, 'maxfun': maxiter * 15})
    except Exception:
        return h
    return project_feasible(np.clip(res.x, 0, 1), n)

def subgradient_optimize(h, n, deadline, n_iter=20000):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    base_step = 0.002
    for k in range(n_iter):
        if time.time() > deadline:
            break
        t_n = (1.0 + np.sqrt(1.0 + 4.0 * (k + 1) * (k + 1))) / 2.0
        h_mom = h_curr + ((k + 1.0) / t_n) * (h_curr - h_prev)
        h_mom = project_feasible(h_mom, n)
        c = compute_all_lags(h_mom, n)
        max_idx = np.argmax(c)
        lag = max_idx - (n - 1)
        g = grad_single_lag(h_mom, n, lag)
        step = base_step / (1.0 + 0.01 * k)
        h_next = h_mom - step * g
        h_next = project_feasible(h_next, n)
        h_prev = h_curr
        h_curr = h_next
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c - 1e-15:
            best_c = c_curr
            best_h = h_curr.copy()
    return best_h, best_c

    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)

def make_binary_init(n, rng, pattern='random'):
    h = np.zeros(n)
    target = n // 2
    if pattern == 'random':
        idx = rng.choice(n, target, replace=False)
        h[idx] = 1.0
    elif pattern == 'block':
        h[:target] = 1.0
    elif pattern == 'alternating':
        h[0:n:2] = 1.0
        if h.sum() < target:
            h[n - 1] = 1.0
        elif h.sum() > target:
            h[0] = 0.0
    else:
        h[:target] = 1.0
    return project_feasible(h, n)

def top_lag_direct_opt(h, n, deadline, max_iter=100000):
    """Directly minimize the maximum lag overlap using a Dantzig-Wolfe style approach."""
    h_curr = h.copy()
    best_h = h_curr.copy()

    step = 0.001
    for it in range(max_iter):
        if time.time() > deadline:
            break

        max_val = c_all.max()
        # Find all lags within epsilon of max
        eps = max(1e-6, max_val * 1e-4)
        active = np.where(c_all >= max_val - eps)[0]
        # Average gradient over active set
        g_avg = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            g_avg += grad_single_lag(h_curr, n, lag)
        g_avg /= len(active)
        # Adaptive step
        h_try = h_curr - step * g_avg
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h_curr = h_try
            step = min(step * 1.1, 0.01)
        else:
            step *= 0.95
            if step < 1e-7:
                step = 0.0005
                h_curr = best_h + np.random.RandomState(it).normal(0, 0.001, len(h_curr))
                h_curr = project_feasible(h_curr, n)

        h_opt = lbfgsb_opt(best_h, len(best_h), 100000000, maxiter=2000)
        c_opt = compute_c5(h_opt, len(h_opt))
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation and return
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
