import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_single_lag_grad(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h)
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] += (-h[:n - lag])
    else:
        m = -lag
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] += (-h[m:])
    return g * dx

def compute_all_grads(h, n):
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        grads[k] = compute_single_lag_grad(h, n, lag)
    return grads

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def slp_single_step(h, n, delta=None):
    """One SLP iteration: solve linearized LP with optional trust region."""
    f_vals = compute_all_lags(h, n)
    n_lags = 2 * n - 1
    grads = compute_all_grads(h, n)

    A_ineq = np.hstack([grads, -np.ones((n_lags, 1))])
    b_ineq = -f_vals + grads @ h

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    if delta is not None:
        lower = np.maximum(0.0, h - delta)
        upper = np.minimum(1.0, h + delta)
        bounds = list(zip(lower, upper)) + [(0.0, None)]
    else:
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None

    if res.success:
        return res.x[:n]
    return None

def slp_optimize(h, n, deadline, max_iters=400, delta_init=0.15):
    """SLP optimization with adaptive trust region and line search."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    delta = delta_init
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Try with trust region first
        new_h = slp_single_step(h, n, delta=delta)
        if new_h is None:
            new_h = slp_single_step(h, n, delta=None)
        if new_h is None:
            break

        # Evaluate and line search
        h_trial = project_feasible(new_h, n)
        c_trial = compute_c5(h_trial, n)

        if c_trial < best_c - 1e-12:
            h = h_trial
            best_c = c_trial
            best_h = h.copy()
            no_improve = 0
            delta = min(0.4, delta * 1.03)
        else:
            # Try partial steps
            improved = False
            for alpha in [0.7, 0.5, 0.3, 0.2, 0.1]:

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    """L-BFGS-B optimization with smooth max."""
    target_sum = n / 2.0
    penalty_coeff = 1e6

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > h[i] + 1e-7:
                        d = min(0.002, (h[j] - h[i]) * 0.2)
                        if d < 1e-10:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            any_improve = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
            if not any_improve:
                break

    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, alpha_sa=50000, t_init=0.005, t_min=0.0001, cool=0.9999):
    """Simulated annealing with structure-aware moves."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.7:
            # Pair move targeting worst lags
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-5:]
            k_star = top_k[rng.randint(5)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-6:
                d = rng.uniform(0.001, min(0.05, 0.005 + T * 5))
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-6:
                d = -rng.uniform(0.001, min(0.05, 0.005 + T * 5))
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue
        else:
            # Random pair move
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.08, 0.003 + T * 6)
            if h[i] > h[j]:
                d = rng.uniform(0.001, max_step)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i]:
                d = -rng.uniform(0.001, max_step)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool

        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = project_feasible(np.full(n, 0.5), n)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: SLP optimization from initial (250s)
    slp_deadline = min(time.time() + 250, deadline - 400)
    h, c = slp_optimize(h, n, slp_deadline, max_iters=500, delta_init=0.12)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: L-BFGS-B polish (80s)
    if time.time() < deadline - 300:
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 280:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 25, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()

    # Phase 3: SLP from perturbations (120s)
    for pert_iter in range(8):
        if time.time() > deadline - 250:
            break
        noise_scale = 0.02 + 0.01 * pert_iter
        h_pert = project_feasible(best_h + rng.normal(0, noise_scale, n), n)
        slp_dl = min(time.time() + 15, deadline - 220)
        h, c = slp_optimize(h_pert, n, slp_dl, max_iters=100, delta_init=0.08)
        if c < best_c:
            best_c, best_h = c, h.copy()
            # Quick polish
            h_opt = lbfgsb_optimize(best_h, n, 200000, time.time() + 8, maxiter=1000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()

    # Phase 4: Targeted local search (150s)
    if time.time() < deadline - 100:
        ls_deadline = min(time.time() + 150, deadline - 80)
        h, c = targeted_local_search(best_h, n, ls_deadline, rng)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: SLP + polish on best (60s)
    if time.time() < deadline - 80:
        slp_dl = min(time.time() + 30, deadline - 60)
        h, c = slp_optimize(best_h, n, slp_dl, max_iters=200, delta_init=0.05)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h_opt = lbfgsb_optimize(best_h, n, 10000000, min(time.time() + 15, deadline - 50), maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # Phase 6: Simulated annealing (80s)
    if time.time() < deadline - 60:
        sa_deadline = min(time.time() + 80, deadline - 40)
        h, c = simulated_annealing(best_h, n, sa_deadline, rng,
                                   alpha_sa=100000, t_init=0.003, t_min=0.00005, cool=0.99992)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Higher resolution attempt (80s)
    if time.time() < deadline - 50 and n < 1000:
        n_high = min(2 * n, 1000)
        if n_high % 2 == 0:
            h_high = expand_to_resolution(best_h, n, n_high)
            c_high = compute_c5(h_high, n_high)
            if c_high < best_c:
                best_c, best_h = c_high, h_high
            slp_dl = min(time.time() + 50, deadline - 30)
            h, c = slp_optimize(h_high, n_high, slp_dl, max_iters=100, delta_init=0.08)
            if c < best_c:
                best_c, best_h = c, h.copy()
                n = n_high

    # Phase 8: Final local search (30s)
    if time.time() < deadline - 5:
        n_final = len(best_h)
        ls_dl = min(time.time() + 25, deadline - 2)
        h, c = targeted_local_search(best_h, n_final, ls_dl, rng)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 9: Final SLP polish
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h, c = slp_optimize(best_h, n_final, deadline - 1, max_iters=50, delta_init=0.03)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
