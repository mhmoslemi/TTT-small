                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def perturb(h, n, seed, scale=0.01):
    rng = np.random.RandomState(seed)
    p = h.copy() + rng.randn(n) * scale
    return project_feasible(p, n)

def make_periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(min(ones_per_period, period)):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick optimization at original resolution
    p1_end = min(t0 + 80, deadline - 200)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 1000000],
        mu=0.01, maxit_per=[2000, 2000, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000 and optimize aggressively
    n2 = 1000
    if time.time() < deadline - 150:
        h_up = upsample(best_h, n2)
        p2_end = min(time.time() + 200, deadline - 100)
        h2, c2 = run_lbfgs(h_up, n2, p2_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.005, maxit_per=[2000, 2000, 2500, 3000, 3000, 3500, 4000, 4000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2

    # Phase 3: Active-set optimization at best resolution
    if time.time() < deadline - 120:
        p3_end = min(time.time() + 150, deadline - 80)
        h3, c3 = run_active(best_h, best_n, p3_end, top_k=12, rel_tol=0.001)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n

    # Phase 4: Try perturbed starts and re-optimize
    if time.time() < deadline - 100:
        for seed_p in [seed + 1, seed + 2, seed + 3]:
            if time.time() > deadline - 80:
                break
            h_pert = perturb(best_h, best_n, seed_p, scale=0.02)
            c_pert = compute_c5(h_pert, best_n)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert
            p4_end = min(time.time() + 60, deadline - 60)
            h4, c4 = run_lbfgs(h_pert, best_n, p4_end,
                alphas=[10000, 50000, 200000, 1000000, 5000000],
                mu=0.005, maxit_per=[2000, 2500, 3000, 3000, 3000])
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = best_n

    # Phase 5: Coordinate descent
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 50, deadline - 40)
        h5, c5 = coordinate_descent(best_h, best_n, p5_end, max_rounds=100)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = best_n

    # Phase 6: Local pair search (focused)
    if time.time() < deadline - 30:
        p6_end = min(time.time() + 40, deadline - 10)
        h6, c6 = local_pair_search(best_h, best_n, p6_end, radius=40)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # Phase 7: Final active-set polish
    if time.time() < deadline - 10:
        p7_end = deadline - 3
        h7, c7 = run_active(best_h, best_n, p7_end, top_k=8, rel_tol=0.0005)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n

    # Final: upsample to even higher resolution and do one more pass
    if time.time() < deadline - 15 and best_n < 2000:
        n_hi = min(2000, best_n * 2)
        h_hi = upsample(best_h, n_hi)
        p8_end = deadline - 3
        h8, c8 = run_lbfgs(h_hi, n_hi, p8_end,
            alphas=[10000, 50000, 500000, 5000000],
            mu=0.001, maxit_per=[2000, 3000, 3000, 2000])
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = n_hi

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
