    if abs(remaining) > 1e-10:
        if n_ones < n:
            h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
    return best_h, best_c

def fine_local_search(h, n, deadline, n_sweeps=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for delta in deltas:
        for sweep in range(n_sweeps):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                partner = (i + 1) % n
                if i == partner:
                    partner = (i + 7) % n
                for sign in [1, -1]:
                    new_i = best_h[i] + sign * delta
                    new_p = best_h[partner] - sign * delta
                    if 0.0 <= new_i <= 1.0 and 0.0 <= new_p <= 1.0:
                        old_i, old_p = best_h[i], best_h[partner]
                        best_h[i], best_h[partner] = new_i, new_p
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[partner] = old_i, old_p
            if not improved:
                break
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(400, 0.5)
        n_orig = 400
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Optimize at original resolution
    p1_end = min(t0 + 120, deadline - 500)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000],
        mu=0.01, maxit_per=[1500, 2000, 2000, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set at original resolution
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 100, deadline - 400)
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=5,
            alphas=[50000, 200000, 1000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n2 = 1000
    if time.time() < deadline - 300:
        h_up = upsample(best_h, best_n, n2)
        c_up = compute_c5(h_up, n2)
        
        p3_end = min(time.time() + 250, deadline - 250)
        h3, c3 = run_lbfgs(h_up, n2, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 500000],
            mu=0.003, maxit_per=[1000, 1500, 2000, 2500, 3000, 3000, 3000, 3000])
        
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n2
        else:
            # Keep the upsampled version for active set even if not better yet
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n2
    
    # Phase 4: Active set refinement at best resolution
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 150, deadline - 180)
        h4, c4 = run_active(best_h, best_n, p4_end, top_k=4,
            alphas=[100000, 500000, 2000000, 5000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Binarization track
    if time.time() < deadline - 200:
        h_bin = binarize(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = run_lbfgs(h_bin, best_n, p5_end,
            alphas=[100, 1000, 5000, 50000, 200000],
            mu=0.01, maxit_per=[2000, 2000, 2500, 3000, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Active set on best
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 80, deadline - 100)
        h6, c6 = run_active(best_h, best_n, p6_end, top_k=3,
            alphas=[500000, 2000000, 10000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Subgradient refinement
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: SLSQP polish (multiple alphas)
    if time.time() < deadline - 30:
        for alpha in [200000, 1000000, 5000000]:
            if time.time() > deadline - 20:
                break
            p8_end = min(time.time() + 15, deadline - 15)
            h8, c8 = run_slqp(best_h, best_n, p8_end, alpha=alpha, maxiter=800)
            if c8 < best_c:
                best_c = c8
                best_h = h8
    
    # Phase 9: Fine local search
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = fine_local_search(best_h, best_n, p9_end, n_sweeps=10)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final: subgradient + SLSQP
    if time.time() < deadline - 2:
        h_fin, c_fin = subgrad_refine(best_h, best_n, deadline - 1, n_iter=3000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
