def subgradient_max(h, n, active_indices, tol=1e-6):
    """Compute subgradient of max_k f_k(h) using active set."""
    dx = 2.0 / n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    
    # Find active shifts
    active = [i for i in active_indices if c_vec[i] > max_val - tol]
    if not active:
        active = [np.argmax(c_vec)]
    
    grad = np.zeros(n)
    for idx in active:
        kk = abs(idx - (n - 1))
        if kk == 0:
            grad[:] += (1.0 - 2.0 * h) * dx
        else:
            grad[:kk] += (-h[kk:n]) * dx
            mid_start = kk
            mid_end = n - kk
            if mid_end > mid_start:
                grad[mid_start:mid_end] += ((1.0 - h[:mid_end-kk]) - h[kk:mid_end]) * dx
            grad[n-kk:] += (1.0 - h[:kk]) * dx
    
    grad /= len(active)
    return grad, max_val, len(active)

    dx = 2.0 / n
    if kk == 0:
        return (1.0 - 2.0 * h) * dx
    grad = np.zeros(n)
    grad[kk:] += (1.0 - h[:n - kk]) * dx
    grad[:n - kk] += -h[kk:n] * dx
    return grad

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        kk = abs(mm - (n - 1))
        grad += w[mm] * gradient_fk(x, n, kk)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, deadline, alpha, mu, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def active_grad(h, n, tol_frac=0.002):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    thresh = max_val - tol_frac * max_val
    active = np.where(c_vec >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    grad = np.zeros(n)
    for idx in active:
        kk = abs(idx - (n - 1))
        grad += gradient_fk(h, n, kk)
    grad /= len(active)
    return grad, max_val, active

def subgradient_opt(h, n, deadline, tol_frac=0.002, lr_init=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    while time.time() < deadline:
        grad, max_val, active = active_grad(best_h, n, tol_frac)
        v = beta * v - lr * grad
        h_trial = best_h + v
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-14:
            best_c = c_trial
            best_h = h_trial
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            lr *= 0.92
            if lr < 1e-9:
                lr = lr_init * 0.5
                v = np.zeros(n)
            if stall > 2000:
                break
    return best_h, best_c

def pair_polish_v2(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for d in step_schedule:
        if time.time() > deadline:
            break
        rounds = 0
        while time.time() < deadline:
            rounds += 1
            if rounds > 200:
                break
            grad, max_val, active = active_grad(best_h, n, 0.003)
            sorted_desc = np.argsort(-grad)[:40]
            sorted_asc = np.argsort(grad)[:40]
            improved = False

            for i in sorted_desc:
                if time.time() > deadline:
                    return best_h, best_c

                    if i == j:
                        continue
                    if grad[i] <= grad[j]:
                        continue

                    hj = best_h[j]
                    if hi < d * 0.1 or hj > 1 - d * 0.1:
                        continue

                    best_delta = 0
                    best_trial_c = best_c

                        if hi - delta < 0 or hj + delta > 1:
                            continue
                        best_h[i] -= delta
                        best_h[j] += delta
                        c = compute_c5(best_h, n)
                        if c < best_trial_c:
                            best_trial_c = c
                            best_delta
