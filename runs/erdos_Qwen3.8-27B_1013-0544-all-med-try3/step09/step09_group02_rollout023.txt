                        h[j] -= step

                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def random_restarts_sqp(h, n, deadline, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(123)
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        pert = rng.normal(0, 0.02, n)
        h_pert = project_feasible(h + pert, n)
        sub_deadline = min(time.time() + 60, deadline - 10)
        h_r, c_r = sqp_optimize(h_pert, n, sub_deadline, max_iter=100, rho_init=0.03, top_k=20)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS warmup at original resolution (30s)
    t1 = min(t0 + 30, deadline - 900)
    if t1 > time.time() + 5:
        best_h, best_c = lbfgs_run(best_h, n_orig, t1,
            alphas=[1000, 5000, 20000])

    # Phase 2: SQP at original resolution (60s)
    t2 = min(t0 + 90, deadline - 750)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=300, rho_init=0.05, top_k=25)

    # Phase 3: Upsample to n=1000
    n_mid = 1000
    if time.time() < deadline - 500:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_mid)
        h_mid = np.interp(x_new, x_old, best_h)
        h_mid = project_feasible(h_mid, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
            n_orig = n_mid

    # Phase 4: L-BFGS at n=1000 with increasing alpha (80s)
    t3 = min(time.time() + 80, deadline - 500)
    if t3 > time.time() + 10:
        best_h, best_c = lbfgs_run(best_h, n_orig, t3,
            alphas=[2000, 8000, 20000, 50000, 100000])

    # Phase 5: Main SQP at n=1000 (200s)
    t4 = min(time.time() + 200, deadline - 300)
    if t4 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t4, max_iter=400, rho_init=0.06, top_k=30)

    # Phase 6: Subgradient with momentum (60s)
    t5 = min(time.time() + 60, deadline - 200)
    if t5 > time.time() + 10:
        h_sg, c_sg = subgradient_optimize(best_h, n_orig, t5, lr_init=0.005)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 7: Another SQP round (80s)
    t6 = min(time.time() + 80, deadline - 150)
    if t6 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t6, max_iter=200, rho_init=0.03, top_k=40)

    # Phase 8: Fine local search (50s)
    t7 = min(time.time() + 50, deadline - 60)
    if t7 > time.time() + 5:
        best_h, best_c = fine_local_search(best_h, n_orig, t7)

    # Phase 9: Random restarts + SQP (60s)
    t8 = min(time.time() + 60, deadline - 40)
    if t8 > time.time() + 10:
        best_h, best_c = random_restarts_sqp(best_h, n_orig, t8, n_restarts=2)

    # Phase 10: Upsample to n=2000 for final refinement
    n_high = 2000
    if time.time() < deadline - 80:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t9 = min(time.time() + 50, deadline - 40)
        if t9 > time.time() + 5:
            h_high, c_high = lbfgs_run(h_high, n_high, t9,
                alphas=[10000, 30000])

        t10 = min(time.time() + 25, deadline - 15)
        if t10 > time.time() + 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t10, max_iter=100, rho_init=0.04, top_k=25)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
