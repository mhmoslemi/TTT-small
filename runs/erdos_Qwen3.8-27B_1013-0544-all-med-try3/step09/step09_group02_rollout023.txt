def do_local_search(h, n, deadline, radius=40):

    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            cmax = c_vec.max()

                break

            grads = np.stack([single_shift_grad(h, n, int(idx)) for idx in active])
            grad_mag = np.sum(np.abs(grads), axis=0) / len(active)
            top = np.argsort(-grad_mag)[:min(2 * radius, n)]
            for i_val in top:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Direction 1: h[i] += step, h[j] -= step
                    if h[i] + step > 1.0 + 1e-12 or h[j] - step < -1e-12:
                        continue
                    dc = step * (grads[:, i] - grads[:, j])
                    if np.all(c_vec[active] + dc <= cmax + 1e-10):
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Direction 2: h[i] -= step, h[j] += step
                    if h[i] - step < -1e-12 or h[j] + step > 1.0 + 1e-12:
                        continue
                    dc = -step * (grads[:, i] - grads[:, j])
                    if np.all(c_vec[active] + dc <= cmax + 1e-10):
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

# === Core ===

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

# === Gradients ===

def single_shift_grad(h, n, idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] = (1.0 - h[k:k+nv]) * dx
            grad[k:k+nv] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:kk+nv] = (1.0 - h[:nv]) * dx
            grad[:nv] += (-h[kk:kk+nv]) * dx
    return grad

# === Smoothed objective for L-BFGS ===

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def do_lbfgs(h, n, deadline, alphas, mus):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 5:
            break
        mu = mus[i] if i < len(mus) else 0.0
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxfun': 15000})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                best_h = hn
        except Exception:
            pass
    return best_h, best_c

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = [single_shift_grad(h, n, idx) for idx in active_idx]
    c_vals = [c_vec[idx] for idx in active_idx]
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0

        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1

        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1

    A_eq[0, 0:n] = 1.0
