def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, top_idx=None):
    c = compute_c5_vec(x, n)
    if top_idx is not None and len(top_idx) > 0:
        vals = c[top_idx]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha, top_idx=None):
    c = compute_c5_vec(x, n)
    if top_idx is not None and len(top_idx) > 0:
        indices = list(top_idx)
        vals = c[indices]
    else:
        indices = list(range(len(c)))
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_mid = n - 1
    for wi, mm in zip(w, indices):
        if wi < 1e-14:
            continue
        k = mm - n_mid
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def get_active_shifts(h, n, frac=0.01):

    max_val = c.max()
    threshold = max_val - max(1e-10, frac * max_val)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active, c

    if time.time() < deadline_global - 80:
        t6 = min(time.time() + 50, deadline_global - 80)
        h, c = subgradient_line_search(h, n, max_iter=300, step_scale=0.0005)
        if c < best_c:
            best_c = c
            best_h = h
        h = best_h
    
    # Phase 7: Fine local search
    if time.time() < deadline_global - 40:
        t7 = min(time.time() + 30, deadline_global - 40)
        h, c = fine_local_search(h, n, t7, max_rounds=2000)
        if c < best_c:
            best_c = c
            best_h = h
        h = best_h
    
    # Phase 8: Final SLSQP polish
    if time.time() < deadline_global - 5:
        h, c = slqp_optimize(h, n, deadline_global - 2, alpha=50000000)
        if c < best_c:
            best_c = c
            best_h = h
    
    # Final projection and evaluation
    n_final = len(best_h)
    n_use = n_orig if n_orig <= 500 else 500
    if len(best_h) != n_use:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_use)
        best_h = np.interp(x_new, x_old, best_h)
        best_h = project_feasible(best_h, n_use)
        n_final = n_use
    
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Also check at original resolution
    if n_orig != n_final:
        x_old = np.linspace(0, 2, n_final)
        x_new = np.linspace(0, 2, n_orig)
        h_check = np.interp(x_new, x_old, best_h)
        h_check = project_feasible(h_check, n_orig)
        c_check = compute_c5(h_check, n_orig)
        if c_check < best_c:
            best_c = c_check
            best_h = h_check
            n_final = n_orig
    
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
