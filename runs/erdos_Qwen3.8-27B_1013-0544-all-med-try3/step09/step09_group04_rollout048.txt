            mutant = de_mutation(pop[0], pop[idxs[0]], pop[idxs[1]], pop[idxs[2]], 
                                  F=0.7 + 0.3 * rng.random(), n=n)
            crossovered = de_crossover(mutant, pop[i], CR=0.8, rng=rng)
            c_new = compute_c5(crossovered, n)
            if c_new <= pop_c[i]:
                new_pop.append(crossovered)
            else:
                new_pop.append(pop[i].copy())
        pop = new_pop
        pop_c = np.array([compute_c5(p, n) for p in pop])
        
        if pop_c[0] < best_c:
            best_c = pop_c[0]
            best_h = pop[0].copy()
    
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters_list is not None and i < len(maxiters_list):
            maxit = maxiters_list[i]
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k_list=None, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if top_k_list is None:
        top_k_list = [8, 6, 4, 3, 2, 1]
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    
    total_rounds = 30
    for rnd in range(total_rounds):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        top_k = top_k_list[min(rnd, len(top_k_list) - 1)]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(50, int(rem2 * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-25,
                                        'gtol': 1e-20, 'maxfun': maxit * 8,
                                        'maxls': 50, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def smart_pair_search(h, n, deadline, max_iters=20000):
    """Gradient-guided pair search: move mass from high-gradient to low-gradient positions"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        # Compute gradient of max correlation
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        
        # Find best pair: max grad_src, min grad_dst
        # We want to decrease the objective, so move mass in -grad direction
        # Increasing h[i] changes obj by grad[i]*delta, so we want grad[i] < 0 for increase
        # Decreasing h[j] changes obj by -grad[j]*delta, so we want grad[j] > 0 for decrease
        
        # Try several step sizes
        deltas = [0.001, 0.002, 0.005, 0.001, 0.002]
        d_val = deltas[it % len(deltas)]
        
        best_improvement = 0
        best_move = None
        
        # Sample candidate pairs (too expensive to check all)
        n_samples = min(200, n)
        idx_src = np.argsort(grad)[:n_samples]  # lowest gradient (good to increase)
        idx_dst = np.argsort(-grad)[:n_samples]  # highest gradient (good to decrease)
        
        found = False
        for s in idx_src[:50]:
            if best_h[s] >= 1.0 - 1e-10:
                continue
            for d in idx_dst[:50]:
                if best_h[d] <= 1e-10 or s == d:
                    continue
                # Try moving d_val from d to s
                if best_h[d] - d_val < 0 or best_h[s] + d_val > 1:
                    continue
                best_h[s] += d_val
                best_h[d] -= d_val
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    improvement = best_c - c
                    best_h[s] -= d_val
                    best_h[d] += d_val
                    # Apply the move
                    best_h[s] += d_val
                    best_h[d] -= d_val
                    best_c = c
                    found = True
                    break
                else:
                    best_h[s] -= d_val
                    best_h[d] += d_val
            if found:
                break
        if not found:
            # Try smaller step
            d_val = 0.0005
            for s in idx_src[:20]:
                if best_h[s] >= 1.0 - 1e-10:
                    continue
                for d in idx_dst[:20]:
                    if best_h[d] <= 1e-10 or s == d:
                        continue
                    if best_h[d] - d_val < 0 or best_h[s] + d_val > 1:
                        continue
                    best_h[s] += d_val
                    best_h[d] -= d_val
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        found = True
                        break
                    else:
                        best_h[s] -= d_val
                        best_h[d] += d_val
                if found:
                    break
    return best_h, best_c

def subgrad_dive(h, n, deadline, n_iter=5000):
    """Pure subgradient on the max constraint with adaptive step"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.001
    dx_val = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        worst = np.argmax(c_vec)
        k = worst - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
            grad[k:n] += (-best_h[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
            grad[0:nv] += (-best_h[kk:n]) * dx_val
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.02, 0.005)
        else:
            eta *= 0.95
    return best_h, best_c

def structured_starts(n, rng):
    """Generate diverse structured starting points"""
    starts = []
    # Sine-based
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(np.pi * freq * x)
        starts.append(project_feasible(h, n))
    # Block-based
    for block in [4, 8, 16, 32]:
        n_blocks = n // block
        h = np.zeros(n)
        for b in range(n_blocks):
            if b % 2 == 0:
                h[b*block:(b+1)*block] = 1.0
            else:
                h[b*block:(b+1)*block] = 0.0
        starts.append(project_feasible(h + rng.random(n) * 0.05, n))
    # Smooth random
    for _ in range(3):
        h = np.cumsum(rng.standard_normal(n))
        h = (h - h.min()) / (h.max() - h.min())
        starts.append(project_feasible(h, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(200, 0.5)
        n0 = 200
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: Quick L-BFGS polish at native resolution (60s)
    p1_end = t0 + 60
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 500, 5000, 50000, 200000],
        mu=0.01, maxiters_list=[800, 1000, 1500, 2000, 2500])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: DE global search at low resolution (40s)
    if deadline - time.time() > 200:
        n_de = 40
        p2_end = min(time.time() + 40, deadline - 250)
        
        # Generate diverse starts for DE
        de_starts = structured_starts(n_de, rng)
        # Also upsample current best
        de_starts.append(upsample(best_h, best_n, n_de))
        
        best_de_h = de_starts[0]
        best_de_c = compute_c5(de_starts[0], n_de)
        for s in de_starts[1:]:
            c_s = compute_c5(s, n_de)
            if c_s < best_de_c:
                best_de_c = c_s
                best_de_h = s
        
        h_de, c_de = run_de_global(best_de_h, n_de, p2_end, pop_size=50, max_gen=150, seed=seed)
        
        # Upsample DE result and compare
        h_de_up = upsample(h_de, n_de, best_n)
        c_de_up = compute_c5(h_de_up, best_n)
        if c_de_up < best_c:
            best_c = c_de_up
            best_h = h_de_up
    
    # Phase 3: Try n=200 fresh optimization (80s) if time allows
    if deadline - time.time() > 250:
        n_fresh = 200
        p3_end = min(time.time() + 60, deadline - 220)
        h_fresh = upsample(best_h, best_n, n_fresh)
        c_fresh = compute_c5(h_fresh, n_fresh)
        h_f2, c_f2 = run_lbfgs(h_fresh, n_fresh, p3_end,
            alphas=[100, 1000, 10000, 100000],
            mu=0.01, maxiters_list=[800, 1200, 1500, 2000])
        if c_f2 < best_c:
            best_c = c_f2
            best_h = h_f2
            best_n = n_fresh
    
    # Phase 4: Upsample to n=500 and optimize (150s)
    if deadline - time.time() > 200 and best_n != 500:
        n_mid = 500
        p4_end = min(time.time() + 120, deadline - 160)
        h_mid = upsample(best_h, best_n, n_mid)
        h_m, c_m = run_lbfgs(h_mid, n_mid, p4_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.005, maxiters_list=[500, 700, 1000, 1200, 1500, 1800, 2200, 2500])
        if c_m < best_c:
            best_c = c_m
            best_h = h_m
            best_n = n_mid
    
    # Phase 5: Active set optimization (120s)
    if deadline - time.time() > 150:
        p5_end = min(time.time() + 100, deadline - 120)
        h5, c5 = run_active_opt(best_h, best_n, p5_end,
            top_k_list=[10, 8, 6, 5, 4, 3, 2, 1, 1, 1],
            alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Try n=1000 (150s)
    if deadline - time.time() > 180:
        n_high = 1000
        p6_end = min(time.time() + 120, deadline - 140)
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        h_h, c_h = run_lbfgs(h_high, n_high, p6_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.003, maxiters_list=[400, 600, 800, 1000, 1200, 1500, 1800, 2000])
        if c_h < best_c:
            best_c = c_h
            best_h = h_h
            best_n = n_high
        
        # Active set at high res
        if deadline - time.time() > 60:
            p6b_end = min(time.time() + 50, deadline - 50)
            h6b, c6b = run_active_opt(best_h, best_n, p6b_end,
                top_k_list=[5, 4, 3, 2, 1],
                alphas=[500000, 2000000, 10000000, 50000000])
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Try n=1500 if time (100s)
    if deadline - time.time() > 120 and best_n < 1500:
        n_vhigh = 1500
        p7_end = min(time.time() + 80, deadline - 80)
        h_vh = upsample(best_h, best_n, n_vhigh)
        h_vh2, c_vh2 = run_lbfgs(h_vh, n_vhigh, p7_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
            mu=0.003, maxiters_list=[300, 500, 700, 900, 1100, 1300, 1500])
        if c_vh2 < best_c:
            best_c = c_vh2
            best_h = h_vh2
            best_n = n_vhigh
    
    # Phase 8: Active set at current best resolution (60s)
    if deadline - time.time() > 80:
        p8_end = min(time.time() + 55, deadline - 50)
        h8, c8 = run_active_opt(best_h, best_n, p8_end,
            top_k_list=[4, 3, 2, 1, 1, 1, 1],
            alphas=[2000000, 10000000, 50000000, 200000000])
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Subgradient dive (30s)
    if deadline - time.time() > 50:
        p9_end = min(time.time() + 30, deadline - 30)
        h9, c9 = subgrad_dive(best_h, best_n, p9_end, n_iter=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Smart pair search (40s)
    if deadline - time.time() > 45:
        p10_end = min(time.time() + 35, deadline - 10)
        h10, c10 = smart_pair_search(best_h, best_n, p10_end, max_iters=15000)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Phase 11: Final active set squeeze (15s)
    if deadline - time.time() > 20:
        p11_end = deadline - 5
        h11, c11 = run_active_opt(best_h, best_n, p11_end,
            top_k_list=[2, 1, 1, 1],
            alphas=[50000000, 200000000, 1000000000])
        if c11 < best_c:
            best_c = c11
            best_h = h11
    
    # Final: one more direct gradient step on worst constraint
    if deadline - time.time() > 3:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, np.array([worst]), 500000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 100, 'ftol': 1e-25,
                                    'gtol': 1e-22, 'maxfun': 300})
            h_fin = project_feasible(res.x, best_n)
            c_fin = compute_c5(h_fin, best_n)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin
        except Exception:
            pass
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
