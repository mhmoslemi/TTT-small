import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, gtol=1e-12):
    target_sum = n / 2.0
    penalty_coeff = 1e6

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': gtol,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)

        return project_feasible(h.copy(), n)

    iter_schedule = [200, 300, 400, 500, 600, 800, 1000]
    
    for alpha, maxit in zip(alpha_schedule, iter_schedule):
        if time.time() > deadline - 400:
            break
        remaining = deadline - 400 - time.time()
        if remaining < 10:
            break
        t_sub = time.time() + min(remaining, 45)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t_sub, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Top-lag gradient descent (fast, many iterations)
    if time.time() < deadline - 250:
        t_sub = min(time.time() + 120, deadline - 250)
        h_opt, c_opt = top_lag_gradient_descent(best_h, n, t_sub, lr_init=0.004, max_iter=300000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: More L-BFGS-B at very high alpha
    if time.time() < deadline - 200:
        for alpha in [10000000, 50000000, 200000000]:
            if time.time() > deadline - 150:
                break
            t_sub = time.time() + min(35, deadline - 150 - time.time())
            h_opt = lbfgsb_optimize(best_h, n, alpha, t_sub, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Targeted pair-sweep on top lags
    if time.time() < deadline - 120:
        t_sub = min(time.time() + 80, deadline - 120)
        h_opt, c_opt = targeted_sweep(best_h, n, t_sub, rng, top_k=5, inner_iters=15000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Simulated annealing (focused)
    if time.time() < deadline - 100:
        t_sub = min(time.time() + 60, deadline - 100)
        h_opt, c_opt = simulate_annealing_focused(best_h, n, t_sub, rng,
                                                   t_init=0.003, t_min=0.00001, cool=0.99992)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Multi-start with perturbations
    if time.time() < deadline - 100:
        t_sub = deadline - 80
        best_h, best_c = multi_start_optimize(best_h, n, best_c, t_sub, rng, n_starts=3)
    
    # Phase 7: Final top-lag descent
    if time.time() < deadline - 60:
        t_sub = min(time.time() + 40, deadline - 60)
        h_opt, c_opt = top_lag_gradient_descent(best_h, n, t_sub, lr_init=0.002, max_iter=200000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Try higher resolution n=1000
    if time.time() < deadline - 120:
        n_high = 1000
        h_high = interpolate_to(best_h, n, n_high)
        c_high = compute_c5(h_high, n_high)
        
        if c_high < best_c + 0.005:  # Worth trying
            t_sub = min(time.time() + 50, deadline - 80)
            h_opt = lbfgsb_optimize(h_high, n_high, 100000, t_sub, maxiter=1500)
            c_opt = compute_c5(h_opt, n_high)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_high
            
            if time.time() < deadline - 60:
                t_sub2 = min(time.time() + 30, deadline - 60)
                h_opt = lbfgsb_optimize(best_h, n, 5000000, t_sub2, maxiter=2000)
                c_opt = compute_c5(h_opt, n)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
            
            if time.time() < deadline - 40:
                t_sub3 = min(time.time() + 25, deadline - 40)
                h_opt, c_opt = top_lag_gradient_descent(best_h, n, t_sub3, lr_init=0.002, max_iter=100000)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    
    # Phase 9: Final polish - L-BFGS-B + targeted sweep
    n_final = len(best_h)
    if time.time() < deadline - 20:
        t_sub = min(time.time() + 15, deadline - 20)
        h_opt = lbfgsb_optimize(best_h, n_final, 100000000, t_sub, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    if time.time() < deadline - 5:
        t_sub = deadline - 3
        h_opt, c_opt = targeted_sweep(best_h, n_final, t_sub, rng, top_k=3, inner_iters=10000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation and return
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Verify feasibility
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    integral = best_h.sum() * (2.0 / n_final)
    assert abs(integral - 1.0) < 0.01, f"Integral constraint violated: {integral}"
    
    return best_h, best_c, n_final
