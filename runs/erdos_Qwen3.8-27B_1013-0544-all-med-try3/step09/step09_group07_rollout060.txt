    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        grad += w[k] * compute_grad_lag(h, n, lag)
    return grad

def lbfgsb_optimize(h_init, n, alpha, deadline, maxiter=5000, penalty=1e6):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full')
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h_init, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'maxls': 100,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h_init.copy(), n)

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def systematic_pair_search(h, n, deadline, rng, top_k=5):
    """Systematic search: for each top lag, try all pairs systematically."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    stall = 0

    while time.time() < deadline and it < 500000:
        it += 1
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False

        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            n_pairs = n - m
            # Try a random subset of pairs each iteration
            n_tries = min(n_pairs, 300)
            for _ in range(n_tries):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n_pairs)
                j = i + m
                diff_val = h[i] - h[j]
                if abs(diff_val) < 1e-7:
                    continue
                # Move from higher to lower
                d = min(0.005, abs(diff_val) * 0.05)
                if diff_val > 0:
                    h[i] -= d
                    h[j] += d
                else:
                    h[i] += d
                    h[j] -= d
                h = project_feasible(h, n)
                c_test = float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    if diff_val > 0:
                        h[i] += d
                        h[j] -= d
                    else:
                        h[i] -= d
                        h[j] += d
                    h = project_feasible(h, n)

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb and retry
                h = best_h + rng.normal(0, 0.003, n)
                h = project_feasible(h, n)
                stall = 0

    return best_h, best_c

def aggressive_smooth_opt(h, n, deadline, rng):
    """Aggressively increase alpha and optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000, 200000000, 1000000000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        t_rem = deadline - 5 - time.time()
        if t_rem < 5:
            break
        maxit = min(5000, max(500, int(t_rem * 50)))
        t_end = time.time() + min(80, t_rem * 0.8)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t_end, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt
        # Small perturbation to escape saddle points
        if rng.random() < 0.3:
            h_opt = best_h + rng.normal(0, 0.002, n)
            h_opt = project_feasible(h_opt, n)
            c_opt2 = compute_c5(h_opt, n)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt
    return best_h, best_c

def subgradient_polish(h, n, deadline, n_iter=50000):
    """Subgradient method on the exact max."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        max_c = c.max()
        c5 = max_c * dx

        # Active set: lags within 1% of max
        thresh = max_c * 0.995
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([c.argmax()])

        # Average subgradient over active lags
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            grad += compute_grad_lag(h, n, lag)
        grad /= len(active)

        # Step size with decay
        step = 0.002 / (1.0 + 0.05 * it)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new, n)
        if c5_new < best_c - 1e-15:
            best_c = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c

def multi_start_optimize(h_base, n, deadline, rng, n_starts=3):
    """Try multiple perturbations of the base solution."""
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n)

    per_start = (deadline - time.time()) / max(1, n_starts + 1)
    if per_start < 10:
        n_starts = 1

    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        # Create perturbation
        scale = 0.01 * (s + 1)
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)

        t_end = min(deadline - 5, time.time() + per_start)
        h_opt, c_opt = aggressive_smooth_opt(h_pert, n, t_end, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n0 = 500

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)
    n = n0
    print(f"Initial: n={n}, C5={best_c:.6f}")

    # Phase 1: Progressive alpha optimization at parent resolution
    if time.time() < deadline - 400:
        alphas_p1 = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
        for alpha in alphas_p1:
            if time.time() > deadline - 350:
                break
            t_rem = deadline - 300 - time.time()
            if t_rem < 10:
                break
            t_end = time.time() + min(60, t_rem * 0.6)
            h_opt = lbfgsb_optimize(best_h, n, alpha, t_end, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        print(f"After Phase 1: C5={best_c:.6f}")

    # Phase 2: Multi-start at parent resolution
    if time.time() < deadline - 250:
        t_rem = deadline - 200 - time.time()
        n_starts = max(2, int(t_rem / 80))
        best_h, best_c = multi_start_optimize(best_h, n, time.time() + min(t_rem, 300), rng, n_starts)
        print(f"After Phase 2: C5={best_c:.6f}")

    # Phase 3: Try other resolutions
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 150:
            break
        t_rem = deadline - 100 - time.time()
        if t_rem < 60:
            break
        h_interp = interpolate_to(best_h, n, n_try)
        c_interp = compute_c5(h_interp, n_try)
        # Optimize at new resolution
        h_opt, c_opt = aggressive_smooth_opt(h_interp, n_try, time.time() + min(80, t_rem * 0.6), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try
            print(f"  n={n_try}: C5={best_c:.6f} (improved)")
        else:
            print(f"  n={n_try}: C5={c_opt:.6f} (no improvement, best stays {best_c:.6f})")

    # Phase 4: Very high alpha optimization on best solution
    if time.time() < deadline - 100:
        t_rem = deadline - 50 - time.time()
        best_h, best_c = aggressive_smooth_opt(best_h, n, time.time() + min(t_rem, 200), rng)
        print(f"After Phase 4: C5={best_c:.6f}")

    # Phase 5: Systematic pair search (local search)
    if time.time() < deadline - 50:
        t_ls = min(deadline - 20, time.time() + 100)
        best_h, best_c = systematic_pair_search(best_h, n, t_ls, rng, top_k=5)
        print(f"After Phase 5: C5={best_c:.6f}")

    # Phase 6: Subgradient polish
    if time.time() < deadline - 20:
        sg_deadline = min(deadline - 5, time.time() + 60)
        best_h, best_c = subgradient_polish(best_h, n, sg_deadline, n_iter=30000)
        print(f"After Phase 6: C5={best_c:.6f}")

    # Phase 7: Final L-BFGS-B with extreme alpha
    if time.time() < deadline - 10:
        h_opt = lbfgsb_optimize(best_h, n, 5000000000, deadline - 3, maxiter=5000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        print(f"After Phase 7: C5={best_c:.6f}")

    # Phase 8: Final tiny local search
    if time.time() < deadline - 3:
        best_h, best_c = systematic_pair_search(best_h, n, deadline - 1, rng, top_k=3)
        print(f"After Phase 8: C5={best_c:.6f}")

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    n_final = len(best_h)
    print(f"Final: n={n_final}, C5={best_c:.6f}")

    return best_h, best_c, n_final
