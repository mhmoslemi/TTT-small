def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (residual / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += (residual / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    n_full = len(c)
    for k in range(n_full):
        if w[k] < 1e-9:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _topk_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15, top_m=None):

def interpolate_to(h, n_from, n_to):
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_new = np.interp(x_to, x_from, h)
    return h_new
