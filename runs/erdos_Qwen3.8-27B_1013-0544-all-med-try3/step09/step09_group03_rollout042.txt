import time
import math
import numpy as np
from scipy.optimize import minimize, differential_evolution


def compute_c5_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def quick_project(h, n, iterations=80, tol=1e-13):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iterations):
        diff = target - float(h.sum())
        if abs(diff) < tol:
            break
        h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(20000):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break

        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-15:
                break
            h += (diff / tot) * h

        h = np.clip(h, 0.0, 1.0)

    # Tiny final repair on interior points, if needed.
    for _ in range(200):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        m = int(interior.sum())
        if m <= 0:
            break
        h[interior] += diff / float(m)
        h = np.clip(h, 0.0, 1.0)

    return h


def interpolate_to(h_old, n_old, n_new):
    x_old = 2.0 * np.arange(n_old) / float(n_old)
    x_new = 2.0 * np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old,
                      left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)


def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    if mx > 500.0:
        val = m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    else:
        val = m + float(np.log(np.sum(np.exp(sh)))) / alpha
    if mu != 0.0:
        ss = float(x.sum() - 0.5 * float(n))
        val += mu * ss * ss
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    wsum = float(w.sum())
    if wsum <= 0.0:
        w = None
    else:
        w /= wsum

    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    center = n - 1

    if w is not None:
        keep = w > 1e-12
        idxs = np.nonzero(keep)[0]
        for pos in idxs:
            wm = float(w[pos]) * dx
            k = int(pos - center)
            if k >= 0:
                L = n - k
                if L > 0:
                    grad[k:] += wm * (1.0 - x[:L])
                    grad[:L] -= wm * x[k:]
            else:
                kk = -k
                L = n - kk
                if L > 0:
                    grad[:L] += wm * (1.0 - x[kk:])
                    grad[kk:] -= wm * x[:L]

    if mu != 0.0:
        ss = float(x.sum() - 0.5 * float(n))
        grad += 2.0 * mu * ss

    return grad


def run_lbfngs(h, n, deadline, configs=None, maxiter=80):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if configs is None:
        configs = [(200.0, 1.0), (2000.0, 4.0),
                   (10000.0, 16.0), (40000.0, 64.0)]

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))

    for alpha, mu in configs:
        if time.time() > deadline - 8.0:
            break

        try:
            res = minimize(
                smooth_obj,
                h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': maxiter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxfun': maxiter * 30,
                    'maxls': 20,
                }
            )
            if res is not None and hasattr(res, 'x'):
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if np.isfinite(c_new) and c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c


def grad_shift(x, n, k):
    g = np.zeros(n, dtype=float)
    if k >= 0:
        L = n - k
        if L > 0:
            g[k:] += 1.0 - x[:L]
            g[:L] -= x[k:]
    else:
        kk = -k
        L = n - kk
        if L > 0:
            g[:L] += 1.0 - x[kk:]
            g[kk:] -= x[:L]
    return g


def line_search_direction(h, n, d, best_c):
    d = np.asarray(d, dtype=float)
    if not np.all(np.isfinite(d)):
        return h.copy(), best_c, False

    d -= d.mean()
    norm = float(np.linalg.norm(d))
    if norm < 1e-15:
        return h, best_c, False

    pos = d > 1e-14
    neg = d < -1e-14
    amax = 1.0
    if pos.any():
        amax = min(amax, float(np.min((1.0 - h[pos]) / d[pos])))
    if neg.any():
        amax = min(amax, float(np.min(h[neg] / (-d[neg]))))
    if amax > 1.0:
        amax = 1.0
    amax *= 0.999
    if amax <= 1e-12 or not np.isfinite(amax):
        return h, best_c, False

    improved = False
    for frac in (1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02, 0.008):
        alpha = amax * frac
        if alpha <= 0.0:
            continue
        trial = h + alpha * d
        trial = np.clip(trial, 0.0, 1.0)
        if abs(float(trial.sum()) - 0.5 * float(n)) > 1e-10:
            trial = quick_project(trial, n, iterations=20, tol=1e-13)
        c = compute_c5(trial, n)
        if np.isfinite(c) and c < best_c - 1e-14:
            best_c = c
            h = trial.copy()
            improved = True
    return h, best_c, improved


def active_line_search(h, n, deadline, active=48, rounds=30, beta=18.0):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stale = 0

    for _ in range(rounds):
        if time.time() > deadline - 3.0:
            break

        c = compute_c5_vec(best_h, n)
        mx = float(c.max())
        if not np.isfinite(mx):
            break

        m = min(active, len(c))
        if m >= len(c):
            idx = np.nonzero(np.ones(len(c), dtype=bool))[0]
        else:
            idx = np.argpartition(c, -m)[-m:]

        active_c = c[idx]
        w = np.exp(beta * (active_c - mx))
        wsum = float(w.sum())
        if wsum <= 0.0:
            w = np.ones_like(active_c) / float(len(active_c))
        else:
            w /= wsum

        g = np.zeros(n, dtype=float)
        for wi, pos in zip(w, idx):
            if wi > 1e-14:
                k = int(pos - (n - 1))
                g += wi * grad_shift(best_h, n, k)

        h, best_c, improved = line_search_direction(best_h, n, -g, best_c)

        # Occasionally use a Frank-Wolfe direction toward a vertex.
        if (len(idx) % 4 == 0) and time.time() < deadline - 8.0:
            order = np.argsort(g)
            v = np.zeros(n, dtype=float)
            v[order[:n // 2]] = 1.0
            h2, c2, improved2 = line_search_direction(best_h, n, v - h, best_c)
            if c2 < best_c - 1e-14:
                best_c = c2
                h = h2
                improved = True
                if c2 < compute_c5(best_h, n) + 1e-14:
                    best_h = h.copy()
            else:
                h = h.copy()

        if time.time() > deadline - 3.0:
            break

        c_now = compute_c5(h, n)
        if c_now < best_c - 1e-14:
            best_c = c_now
            best_h = h.copy()

        if improved:
            stale = 0
            h = best_h.copy()
        else:
            stale += 1
            if stale >= 8:
                break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def make_zero_half(n, U):
    # U is total number of ones and also number of central zeros.
    # Pattern: ones, halves, zeros, halves, ones.
    if U <= 0 or U > n // 2:
        return None
    if (n - 2 * U) < 0:
        return None
    if U % 2 != 0:
        return None

    m = U // 2
    H = n - 2 * U
    left = H // 2
    right = H - left

    parts = [
        np.ones(m, dtype=float),
        np.full(left, 0.5, dtype=float),
        np.zeros(U, dtype=float),
        np.full(right, 0.5, dtype=float),
        np.ones(m, dtype=float),
    ]
    h = np.concatenate(parts)
    if h.shape[0] != n:
        return None
    return quick_project(h, n, iterations=20, tol=1e-13)


def search_zero_half(n, deadline):
    best_h = None
    best_c = float('inf')

    Umin = max(4, int(0.10 * n))
    Umax = min(n // 2 - 2, int(0.49 * n))
    if Umin > Umax:
        return None, float('inf')

    # Even U keeps the pattern exactly feasible.
    if Umin % 2 != 0:
        Umin += 1

    for U in range(Umin, Umax + 1, 4):
        if time.time() > deadline - 2.0:
            break
        h = make_zero_half(n, U)
        if h is None:
            continue
        c = compute_c5(h, n)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()

    if best_h is not None:
        best_c = compute_c5(project_feasible(best_h, n), n)
        best_h = project_feasible(best_h, n)
    return best_h, best_c


def fourier_h(params, n, modes):
    s = float(params[0])
    pre = 2.0 / float(n)
    x = pre * np.arange(n, dtype=float)
    A = np.zeros(n, dtype=float)

    p = 1
    for m in modes:
        a1 = float(params[p])
        a2 = float(params[p + 1])
        p += 2
        mx = math.pi * float(m) * x
        if a1 != 0.0:
            A += a1 * np.cos(mx)
        if a2 != 0.0:
            A += a2 * np.sin(mx)

    A *= s
    h = 0.5 * (1.0 + np.tanh(A))
    h = quick_project(h, n, iterations=30, tol=1e-13)
    return h


def fourier_de_objective(params, n, modes):
    try:
        h = fourier_h(params, n, modes)
        return compute_c5(h, n)
    except Exception:
        return 1e6

            h_hi, c_hi = active_line_search(
                h_hi, n_final,
                min(time.time() + 35.0, deadline - 8.0),
                active=80,
                rounds=12,
                beta=28.0
            )
            add_candidate(h_hi)
        except Exception:
            pass

    # Ensure exact feasibility and report the exact measured objective.
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    if not np.isfinite(best_c):
        best_h = project_feasible(h_start, n_start)
        best_c = compute_c5(best_h, n_start)

    return best_h, float(best_c), int(len(best_h))
