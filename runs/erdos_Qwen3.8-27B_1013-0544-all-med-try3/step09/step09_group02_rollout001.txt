            room = 1.0 - h

            if tot < 1e-15:

            tot = h.sum()
            if tot < 1e-15:

        grad[0:nv] += (1.0 - h[k:n]) * dx

        grad += w[idx] * g
    return float(val), grad

        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in range(n_act):
            g = grad_for_correlation(h, n, active[j])
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vec[active[j]]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        bds = [(None, None)] * n + [(0.0, 2.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bds, method='highs')
            if res.success:
                delta = res.x[0:n]
                if np.max(np.abs(delta)) < 1e-13:
                    break
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    no_improve = 0
                    if rho < 0.06:
                        rho = min(rho * 1.2, 0.06)
                else:
                    no_improve += 1
                    rho *= 0.6
                    if no_improve > 30:
                        break
            else:
                rho *= 0.6
                no_improve += 1
                if no_improve > 30:
                    break
        except Exception:
            rho *= 0.5
            no_improve += 1
            if no_improve > 30:
                break
        if rho < 1e-9:
            break
    return best_h, best_c

def block_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    block_sizes = [3, 5, 8, 12, 20, 30]
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for L in block_sizes:
        for d in deltas:
            if time.time() > deadline:
                break
            improved = True
            while improved and time.time() < deadline:
                improved = False
                c_vec = compute_c5_vec(h, n)
                max_val = c_vec.max()
                active = np.where(c_vec >= max_val - max(1e-8, 0.001 * max_val))[0]
                grad_abs = np.zeros(n)
                for idx in active[:30]:
                    grad_abs += np.abs(grad_for_correlation(h, n, idx))
                if len(active) > 0:
                    grad_abs /= len(active)
                top = np.argsort(-grad_abs)[:min(200, n)]
                np.random.seed(int(time.time() * 1000) % 100000)
                np.random.shuffle(top)
                for i_val in top[:80]:
                    if time.time() > deadline:
                        return best_h, best_c
                    i = int(i_val)
                    for offset in np.random.choice(range(max(1, n-L), min(n, n-L+50)), size=min(5, max(1, n-L)), replace=False):
                        j = int(offset)
                        if abs(i - j) < L:
                            continue
                        # Try: increase block [i, i+L), decrease block [j, j+L)
                        i1 = min(i + L, n)
                        j1 = min(j + L, n)
                        if h[i:i1].max() + d > 1.0 + 1e-12 or h[j:j1].min() - d < -1e-12:
                            # Try partial
                            continue
                        h[i:i1] += d
                        h[j:j1] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i:i1] -= d
                            h[j:j1] += d
                        # Try reverse
                        if h[j:j1].max() + d <= 1.0 + 1e-12 and h[i:i1].min() - d >= -1e-12:
                            h[j:j1] += d
                            h[i:i1] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[j:j1] -= d
                                h[i:i1] += d
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            active = np.where(c_vec >= max_val - max(1e-8, 0.001 * max_val))[0]
            grad_abs = np.zeros(n)
            for idx in active[:25]:
                grad_abs += np.abs(grad_for_correlation(h, n, idx))
            if len(active) > 0:
                grad_abs /= len(active)
            top = np.argsort(-grad_abs)[:min(100, n)]
            for i_val in top:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                # Find best j to pair with
                candidates = np.argsort(-grad_abs)[:50]
                for j_val in candidates:
                    j = int(j_val)
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def slqp_refine(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh)))) / alpha
    def obj_grad(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for idx in range(len(w)):
            if w[idx] < 1e-13:
                continue
            grad += w[idx] * grad_for_correlation(x, n, idx)
        return grad
    def cons_fun(x, n):
        return x.sum() - n / 2.0
    def cons_jac(x, n):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    t_remaining = deadline - time.time()
    n_iter = max(100, int(t_remaining * 10))
    try:
        res = minimize(lambda x: obj_fn(x, n, alpha), h,
                      jac=lambda x: obj_grad(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': min(n_iter, 3000), 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample_and_optimize(h_low, n_low, n_high, deadline):
    x_old = np.linspace(0, 2, n_low)
    x_new = np.linspace(0, 2, n_high)
    h_high = np.interp(x_new, x_old, h_low)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)
    
    t1 = min(time.time() + 60, deadline - 120)
    if t1 > time.time() + 5:
        h_high, c_high = lbfgs_smooth(h_high, n_high, t1, alphas=[5000, 20000, 80000])
    
    t2 = min(time.time() + 50, deadline - 60)
    if t2 > time.time() + 5:
        h_high, c_high = sqp_active_set(h_high, n_high, t2, max_iter=200, rho_init=0.02, active_size=20)
    
    t3 = min(time.time() + 30, deadline - 10)
    if t3 > time.time() + 3:
        h_high, c_high = fine_pair_search(h_high, n_high, t3)
    
    return h_high, c_high

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        n_orig = 500
        h_init = np.full(n_orig, 0.5)
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: L-BFGS with moderate alpha (40s)
    t1 = min(t0 + 40, deadline - 900)
    if t1 > time.time() + 3:
        best_h, best_c = lbfgs_smooth(best_h, n_orig, t1, alphas=[3000, 10000, 30000])
    
    # Phase 2: Proximal gradient (60s)
    t2 = min(t0 + 100, deadline - 750)
    if t2 > time.time() + 5:
        h_pg, c_pg = proximal_gradient(best_h, n_orig, t2, alpha=30000, max_iter=50000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    
    # Phase 3: SQP - main optimization (120s)
    t3 = min(t0 + 220, deadline - 550)
    if t3 > time.time() + 10:
        best_h, best_c = sqp_active_set(best_h, n_orig, t3, max_iter=400, rho_init=0.04, active_size=30)
    
    # Phase 4: L-BFGS with high alpha (60s)
    t4 = min(t0 + 280, deadline - 450)
    if t4 > time.time() + 5:
        best_h, best_c = lbfgs_smooth(best_h, n_orig, t4, alphas=[50000, 150000, 400000])
    
    # Phase 5: SQP with finer steps (80s)
    t5 = min(t0 + 360, deadline - 350)
    if t5 > time.time() + 10:
        best_h, best_c = sqp_active_set(best_h, n_orig, t5, max_iter=300, rho_init=0.015, active_size=35)
    
    # Phase 6: Block local search (80s)
    t6 = min(t0 + 440, deadline - 280)
    if t6 > time.time() + 5:
        best_h, best_c = block_local_search(best_h, n_orig, t6)
    
    # Phase 7: Fine pair search (80s)
    t7 = min(t0 + 520, deadline - 200)
    if t7 > time.time() + 5:
        best_h, best_c = fine_pair_search(best_h, n_orig, t7)
    
    # Phase 8: Another SQP round (60s)
    t8 = min(t0 + 580, deadline - 150)
    if t8 > time.time() + 10:
        best_h, best_c = sqp_active_set(best_h, n_orig, t8, max_iter=250, rho_init=0.01, active_size=40)
    
    # Phase 9: SLSQP polish (80s)
    t9 = min(t0 + 660, deadline - 80)
    if t9 > time.time() + 10:
        h_sl, c_sl = slqp_refine(best_h, n_orig, t9, alpha=800000, maxiter=1000)
        if c_sl < best_c:
            best_c = c_sl
            best_h = h_sl
    
    # Phase 10: Final fine search (40s)
    t10 = min(t0 + 700, deadline - 30)
    if t10 > time.time() + 3:
        best_h, best_c = fine_pair_search(best_h, n_orig, t10)
    
    # Phase 11: Upsample to higher resolution if time permits
    n_u = max(800, n_orig * 2)
    if time.time() < deadline - 150 and n_u <= 2000:
        h_u, c_u = upsample_and_optimize(best_h, n_orig, n_u, deadline - 30)
        if c_u < best_c:
            best_c = c_u
            best_h = h_u
            n_orig = n_u
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
