        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(30, 5.0), (100, 20.0), (300, 100.0),
                   (1000, 500.0), (3000, 2000.0), (10000, 10000.0),
                   (30000, 50000.0), (100000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 8:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 100000, 'ftol': 1e-25,
                                   'gtol': 1e-20, 'maxfun': 500000, 'maxls': 200})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_search_pattern(h, n, deadline, max_flips=2000):
    """Local search on binary pattern: flip bits to reduce max overlap."""
    b = np.where(h > 0.5, 1, 0).astype(int)
    target = n // 2
    if b.sum() != target:
        # Adjust to have exactly n/2 ones
        if b.sum() > target:
            order = np.argsort(b == 0)
            for i in order:
                if b.sum() == target:
                    break
                b[i] = 1
        else:
            order = np.argsort(b == 1)
            for i in order:
                if b.sum() == target:
                    break
                b[i] = 0
    best_b = b.copy()
    best_c = compute_c5(best_b.astype(float), n)
    
    for iteration in range(max_flips):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_b.astype(float), n)
        worst_k = int(np.argmax(c_vec)) - (n - 1)
        
        found_improvement = False
        # Try flipping bits that affect the worst shift
        if worst_k >= 0:
            candidates = list(range(0, n - worst_k)) + list(range(worst_k, n))
        else:
            candidates = list(range(-worst_k, n)) + list(range(0, n + worst_k))
        
        rng_local = np.random.RandomState(iteration + 42)
        rng_local.shuffle(candidates)
        
        for i in candidates[:50]:
            new_b = best_b.copy()
            if new_b[i] == 1:
                # Find another 1 to swap with (maintain count)
                other_candidates = [j for j in range(n) if j != i and new_b[j] == 1]
                if not other_candidates:
                    break
                j = other_candidates[rng_local.randint(len(other_candidates))]
                new_b[i] = 0
                new_b[j] = 1
            else:
                other_candidates = [j for j in range(n) if j != i and new_b[j] == 0]
                if not other_candidates:
                    break
                j = other_candidates[rng_local.randint(len(other_candidates))]
                new_b[i] = 1
                new_b[j] = 0
            c_new = compute_c5(new_b.astype(float), n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_b = new_b
                found_improvement = True
                break
        if not found_improvement:
            # Try single bit flips with compensation
            for i in candidates[:30]:
                new_b = best_b.copy()
                new_b[i] = 1 - new_b[i]
                c_new = compute_c5(new_b.astype(float), n)
                if c_new < best_c - 1e-12:
                    # Fix the count
                    if new_b.sum() != target:
                        if new_b.sum() > target:
                            z = np.where(new_b == 1)[0]
                            new_b[z[rng_local.randint(len(z))]] = 0
                        else:
                            z = np.where(new_b == 0)[0]
                            new_b[z[rng_local.randint(len(z))]] = 1
                    best_c = compute_c5(new_b.astype(float), n)
                    best_b = new_b
                    found_improvement = True
                    break
            if not found_improvement:
                break
    
    return best_b.astype(float), best_c

def targeted_nudge(h, n, deadline, max_rounds=200):
    """Very targeted: adjust individual values to reduce the max."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Get top 3 worst shifts
        top_k = np.argsort(c_vec)[-3:][::-1]
        
        improved = False
        for idx in top_k:
            k = int(idx - (n - 1))
            if k >= 0:
                i_range = range(0, n - k)
                j_range = range(k, n)
            else:
                kk = -k
                i_range = range(kk, n)
                j_range = range(0, n - kk)
            
            # For this shift, overlap = sum h[i]*(1-h[j]) where j = i+k
            # To reduce: increase h where both are high, or make them more binary
            pairs = list(zip(i_range, j_range))
            if len(pairs) > 200:
                idxs = np.random.RandomState(rnd).choice(len(pairs), 200, replace=False)
                pairs = [pairs[i] for i in idxs]
            
            for (i, j) in pairs:
                if time.time() > deadline - 3:
                    return best_h, best_c
                # Try small adjustments
                for di in [0.001, -0.001, 0.0005, -0.0005]:
                    ni = best_h[i] + di
                    if ni < -1e-9 or ni > 1 + 1e-9:
                        continue
                    best_h[i] = ni
                    # Adjust j to maintain sum
                    best_h[j] = best_h[j] - di
                    if best_h[j] < -1e-9 or best_h[j] > 1 + 1e-9:
                        best_h[i] = ni - di  # revert
                        best_h[j] = best_h[j] + di
                        continue
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] = ni - di
                        best_h[j] = best_h[j] + di
                if improved:
                    break
            if improved:
                break
        if not improved and rnd > 20:
            break
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=12, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    if mags is None:
        mags = [0.01, 0.02, 0.03, 0.005, 0.05, 0.015, 0.025, 0.04,
                0.008, 0.012, 0.02, 0.03]
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 10000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': 100000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def oscillation_refine(h, n, deadline):
    """Try small periodic oscillations to break symmetry and reduce max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    
    for period in [3, 4, 5, 7, 8, 11, 13]:
        if time.time() > deadline - 5:
            break
        for amplitude in [0.002, 0.005, 0.01, 0.003, 0.007]:
            if time.time() > deadline - 5:
                break
            for phase in np.linspace(0, 2*np.pi, period, endpoint=False):
                x = np.linspace(0, 2, n)
                osc = amplitude * np.sin(2*np.pi*x/period + phase)
                h_new = project_feasible(h + osc, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
                    # Quick L-BFGS from this point
                    try:
                        res = minimize(smooth_obj, best_h, jac=smooth_grad,
                                      args=(n, 20000, 50000.0), method='L-BFGS-B',
                                      bounds=[(0.0, 1.0)] * n,
                                      options={'maxiter': 5000, 'ftol': 1e-20,
                                               'gtol': 1e-18})
                        h_opt = project_feasible(res.x, n)
                        c_opt = compute_c5(h_opt, n)
                        if c_opt < best_c:
                            best_c = c_opt
                            best_h = h_opt
                    except Exception:
                        pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Thorough L-BFGS at base resolution
    t1 = min(t0 + 120, deadline - 950)
    if t1 > time.time() + 8:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(20, 5.0), (50, 10.0), (150, 50.0), (400, 200.0),
                     (1000, 500.0), (3000, 2000.0), (10000, 10000.0),
                     (30000, 50000.0), (100000, 100000.0)])
    
    # Phase 2: SLSQP polish
    t2 = min(t0 + 200, deadline - 870)
    if t2 > time.time() + 8:
        h2, c2 = slqp_polish(best_h, n_base, t2, alpha=2000000, maxiter=8000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Oscillation refinement
    t3 = min(t0 + 280, deadline - 790)
    if t3 > time.time() + 5:
        best_h, best_c = oscillation_refine(best_h, n_base, t3)
    
    # Phase 4: Perturbation search
    t4 = min(t0 + 420, deadline - 650)
    if t4 > time.time() + 10:
        best_h, best_c = perturbation_search(best_h, n_base, t4, n_trials=15,
            mags=[0.008, 0.015, 0.025, 0.005, 0.035, 0.012, 0.02,
                 0.004, 0.018, 0.028, 0.01, 0.03, 0.014, 0.022, 0.006])
    
    # Phase 5: SLSQP after perturbations
    t5 = min(t0 + 500, deadline - 570)
    if t5 > time.time() + 8:
        h5, c5 = slqp_polish(best_h, n_base, t5, alpha=5000000, maxiter=6000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Binary exploration
    t6 = min(t0 + 600, deadline - 470)
    if t6 > time.time() + 10:
        b_sol, b_c = binary_search_pattern(best_h, n_base, t6, max_flips=3000)
        if b_c < best_c:
            # Relax binary to continuous and re-optimize
            h_relaxed = b_sol.astype(float) + 0.02 * (np.random.RandomState(1).rand(n_base) - 0.5)
            h_relaxed = project_feasible(h_relaxed, n_base)
            h_r, c_r = run_lbfgs(h_relaxed, n_base, min(t6, deadline - 300),
                configs=[(1000, 500.0), (5000, 2000.0), (20000, 50000.0)])
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
            # Also try the binary directly
            if b_c < best_c:
                best_c = b_c
                best_h = b_sol.astype(float)
    
    # Phase 7: Upsample to n=1000
    n1 = 1000
    t7_start = time.time()
    if t7_start < deadline - 350:
        h1000 = interpolate_to(best_h, len(best_h), n1)
        t7 = min(t7_start + 100, deadline - 200)
        if t7 > time.time() + 8:
            h1000, c1000 = run_lbfgs(h1000, n1, t7,
                configs=[(200, 50.0), (500, 200.0), (2000, 1000.0),
                         (5000, 5000.0), (15000, 20000.0), (50000, 50000.0)])
        
        t8 = min(time.time() + 60, deadline - 130)
        if t8 > time.time() + 8:
            h8, c8 = slqp_polish(best_h if c1000 >= best_c else h1000,
                                 n1 if c1000 < best_c else len(best_h),
                                 t8, alpha=10000000, maxiter=5000)
            n_use = n1 if c1000 < best_c else len(best_h)
            if c8 < best_c:
                best_c = c8
                best_h = h8
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
    
    # Phase 8: Perturbation at higher resolution
    if time.time() < deadline - 150:
        n_curr = len(best_h)
        t9 = min(time.time() + 60, deadline - 60)
        if t9 > time.time() + 8:
            best_h, best_c = perturbation_search(best_h, n_curr, t9, n_trials=8,
                mags=[0.005, 0.01, 0.015, 0.003, 0.02, 0.008, 0.012, 0.006])
    
    # Phase 9: Final SLSQP
    if time.time() < deadline - 30:
        n_curr = len(best_h)
        t10 = min(time.time() + 25, deadline - 10)
        if t10 > time.time() + 5:
            h10, c10 = slqp_polish(best_h, n_curr, t10, alpha=20000000, maxiter=3000)
            if c10 < best_c:
                best_c = c10
                best_h = h10
    
    # Phase 10: Targeted nudge
    if time.time() < deadline - 8:
        n_curr = len(best_h)
        best_h, best_c = targeted_nudge(best_h, n_curr, deadline - 4, max_rounds=100)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
