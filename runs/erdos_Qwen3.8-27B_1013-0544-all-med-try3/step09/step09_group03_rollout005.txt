def subgradient_method(h, n, deadline, n_iter=50000):
    """Direct subgradient on the max correlation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.001  # initial learning rate
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
        
        # Find worst shift
        worst_idx = int(np.argmax(c_vec))
        k_shift = worst_idx - (n - 1)
        dx = 2.0 / n
        
        # Compute subgradient for the worst shift
        sg = np.zeros(n)
        if k_shift >= 0:
            nv = n - k_shift
            if nv > 0:
                sg[0:nv] += (1.0 - h[k_shift:n]) * dx
                sg[k_shift:n] += (-h[0:nv]) * dx
        else:
            kk = -k_shift
            nv = n - kk
            if nv > 0:
                sg[kk:n] += (1.0 - h[0:nv]) * dx
                sg[0:nv] += (-h[kk:n]) * dx
        
        # Add constraint gradient
        sg += 2.0 * 10.0 * (h.sum() - n / 2.0)
        
        # Take step
        h_new = h - lr * sg
        h_new = project_feasible(h_new, n)
        
        # Check if it helped
        c_new = compute_c5(h_new, n)
        if c_new < c_max + 1e-10:  # allow small increases
            h = h_new
        else:
            lr *= 0.95  # decrease learning rate
        
        # Adapt learning rate
        if it % 100 == 0:
            lr = 0.001 * (best_c / c_max) ** 0.5
        
        # Periodic restart with noise
        if it % 5000 == 4999:
            h = h + 0.01 * np.random.randn(n)
            h = project_feasible(h, n)
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_rounds=100):

    target = n / 2.0
    n_ones = int(round(target))

    h_bin = h.copy()
    # Sort by value, set top n_ones to 1, rest to 0
    order = np.argsort(-h_bin)

    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        improved = False

        zeros = np.where(h_bin == 0.0)[0]

        for _ in range(n_trials):

            i = ones[rng.randint(len(ones))]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k_shift = idx - center
        wm = w[idx] * dx
        if k_shift >= 0:
            nv = n - k_shift
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k_shift:n])
                grad[k_shift:n] += wm * (-x[0:nv])
        else:
            kk = -k_shift
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (1000, 500.0), (5000, 2000.0),
                   (20000, 10000.0), (100000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def pairwise_cd(h, n, deadline, n_passes=200, n_trials_per_pass=2000, seed=42):
    """Exact pairwise coordinate descent with ternary search on delta."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h_work = h.copy()
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        # Focus on pairs involving indices near worst shift region
        c_vec = compute_c5_vec(h_work, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        # Build candidate index set
        if k_worst >= 0:
            region1 = set(range(0, n - k_worst))
            region2 = set(range(k_worst, n))
        else:
            region1 = set(range(-k_worst, n))
            region2 = set(range(0, n + k_worst))
        region = region1 | region2
        region_list = list(region)
        all_idx = list(range(n))
        
        for t in range(n_trials_per_pass):
            if time.time() > deadline:
                return best_h, best_c
            # 70% chance pick from region, 30% random
            if rng.rand() < 0.7 and len(region_list) > 0:
                i = region_list[rng.randint(len(region_list))]
            else:
                i = all_idx[rng.randint(n)]
            j = all_idx[rng.randint(n)]
            if i == j:
                continue
            
            h_i = h_work[i]
            h_j = h_work[j]
            d_lo = max(-h_i, h_j - 1.0)
            d_hi = min(1.0 - h_i, h_j)
            if d_lo >= d_hi - 1e-14:
                continue
            
            # Ternary search (20 iterations)
            lo = d_lo
            hi = d_hi
            h_work[i] += lo  # save and restore pattern
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h_work[i] = h_i + m1
                h_work[j] = h_j - m1
                c1 = float(np.max(np.correlate(h_work, 1 - h_work, mode='full'))) * dx
                h_work[i] = h_i + m2
                h_work[j] = h_j - m2
                c2 = float(np.max(np.correlate(h_work, 1 - h_work, mode='full'))) * dx
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            
            delta = (lo + hi) / 2.0
            h_work[i] = np.clip(h_i + delta, 0.0, 1.0)
            h_work[j] = np.clip(h_j - delta, 0.0, 1.0)
            c_work = compute_c5(h_work, n)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_rounds=50, n_trials=3000, seed=123):

    n_ones = int(round(target))

    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    no_improve = 0
    for rnd in range(max_rounds):

            break
        improved = False
        for _ in range(n_trials):

                return best_h, best_c
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0

            if c < best_c - 1e-15:
                best_c = c

            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]

def subgradient_method(h, n, deadline, max_iter=30000, lr_init=0.05):
    """Subgradient method targeting worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_work = h.copy()
    lr = lr_init
    dx = 2.0 / n
    center = n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_work, n)
        c_max = c_vec.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_work.copy()
        
        # Find top-3 worst shifts for combined subgradient
        top_idx = np.argsort(-c_vec)[:3]
        sg = np.zeros(n)
        for idx in top_idx:
            k_shift = idx - center
            if k_shift >= 0:
                nv = n - k_shift
                if nv > 0:
                    sg[0:nv] += (1.0 - h_work[k_shift:n]) * dx
                    sg[k_shift:n] += (-h_work[0:nv]) * dx
            else:
                kk = -k_shift
                nv = n - kk
                if nv > 0:
                    sg[kk:n] += (1.0 - h_work[0:nv]) * dx
                    sg[0:nv] += (-h_work[kk:n]) * dx
        sg /= 3.0
        
        # Constraint gradient
        sg += 50.0 * (h_work.sum() - n / 2.0)
        
        # Step
        h_new = h_work - lr * sg
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new <= c_max + lr * 0.1:
            h_work = h_new
            lr = max(lr * 0.999, 0.001)
        else:
            lr = max(lr * 0.95, 0.001)
        
        if it % 5000 == 4999:
            # Periodic perturbation
            h_work = h_work + 0.005 * np.random.randn(n)
            h_work = project_feasible(h_work, n)
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new

        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: More pairwise CD (150s)
    tgt4 = min(t0 + 600, total_deadline - 300)
    if time.time() < tgt4:
        h4, c4 = pairwise_cd(best_h, n_base, tgt4, n_passes=400, n_trials_per_pass=2000, seed=seed+100)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Binary local search (100s)
    tgt5 = min(t0 + 700, total_deadline - 200)
    if time.time() < tgt5:
        h5, c5 = binary_local_search(best_h, n_base, tgt5, max_rounds=30, n_trials=2000, seed=seed+200)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            # Polish binary solution with continuous optimization
            tgt5b = min(t0 + 750, total_deadline - 150)
            if time.time() < tgt5b:
                h5b, c5b = run_lbfgs(best_h, n_base, tgt5b,
                    configs=[(10000, 5000.0), (50000, 20000.0)])
                if c5b < best_c:
                    best_c = c5b
                    best_h = h5b
    
    # Phase 6: Targeted worst-shift optimization (80s)
    tgt6 = min(t0 + 790, total_deadline - 120)
    if time.time() < tgt6:
        h6, c6 = targeted_worst_shift(best_h, n_base, tgt6, n_rounds=150)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish (60s)
    tgt7 = min(t0 + 850, total_deadline - 60)
    if time.time() < tgt7:
        h7, c7 = slqp_polish(best_h, n_base, tgt7, alpha=2000000, maxiter=1500)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Multi-start with different perturbations (50s)
    tgt8 = min(t0 + 900, total_deadline - 40)
    if time.time() < tgt8:
        h8, c8 = multi_start_optimize(best_h, n_base, tgt8, n_starts=3, seed=seed+500)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Final pairwise CD polish (remaining - 20s)
    tgt9 = total_deadline - 20
    if time.time() < tgt9:
        h9, c9 = pairwise_cd(best_h, n_base, tgt9, n_passes=100, n_trials_per_pass=1000, seed=seed+999)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
