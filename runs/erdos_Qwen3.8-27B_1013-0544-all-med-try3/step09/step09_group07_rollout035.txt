def grad_lag(h, n, k_idx):
    """Gradient of the correlation at index k_idx w.r.t. h, multiplied by dx."""

        g[lag:n] += (-h[0:n-lag]) * dx
    else:

        g[m:n] += (1.0 - h[0:n-m]) * dx

                            best_c = float(c_test)
                            best_h_bin = h_bin.copy()
                            improved = True
                            break
                        h_bin[i] = 0
                        h_bin[i+m] = 1
        
        if not improved:
            # Try a random swap
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = rng_local.choice(ones)
                j = rng_local.choice(zeros)
                h_bin[i] = 0
                h_bin[j] = 1
                c_test = compute_c5(h_bin, n)
                if c_test < best_c - 1e-14:
                    best_c = float(c_test)
                    best_h_bin = h_bin.copy()
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
    
    # Relax the best binary solution back to continuous
    h_relaxed = lbfgsb_optimize(best_h_bin.astype(float), n, 5000000, min(deadline, time.time() + 30), maxiter=2000)
    c_relaxed = compute_c5(h_relaxed, n)
    
    if c_relaxed < best_c:
        return h_relaxed, c_relaxed
    return best_h_bin, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    n = len(best_h)
    
    # Phase 1: L-BFGS-B warmup
    for alpha, maxit in [(1000, 500), (10000, 800), (100000, 1000)]:
        if time.time() > deadline - 400:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(30, (deadline - 400 - time.time())), maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Leveling optimization (main workhorse)
    if time.time() < deadline - 200:
        deadline_lv = min(deadline - 150, time.time() + 300)
        h_opt, c_opt = leveling_optimize(best_h, n, deadline_lv, K=10, max_iter=100000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: High-alpha L-BFGS-B
    if time.time() < deadline - 150:
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > deadline - 100:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(25, (deadline - 100 - time.time())), maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Another leveling pass
    if time.time() < deadline - 120:
        deadline_lv2 = min(deadline - 80, time.time() + 100)
        h_opt, c_opt = leveling_optimize(best_h, n, deadline_lv2, K=5, max_iter=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Binary local search + relaxation
    if time.time() < deadline - 100:
        h_bin, c_bin = binary_local_search(best_h, n, min(deadline - 60, time.time() + 80))
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Final L-BFGS-B polish on the result
        if time.time() < deadline - 30:
            h_opt = lbfgsb_optimize(best_h, n, 50000000, deadline - 10, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: Try different resolutions
    for n_try in [n, 750, 1000, 400]:
        if n_try == n:
            continue
        if time.time() > deadline - 80:
            break
        remaining = deadline - 40 - time.time()
        if remaining < 30:
            break
        h_try = expand_resolution(best_h, len(best_h), n_try)
        # Quick optimization at new resolution
        h_opt = lbfgsb_optimize(h_try, n_try, 100000, time.time() + min(15, remaining * 0.4), maxiter=2000)
        h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, time.time() + min(10, remaining * 0.3), maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 7: Final leveling on best
    if time.time() < deadline - 20:
        h_opt, c_opt = leveling_optimize(best_h, len(best_h), deadline - 5, K=8, max_iter=30000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
