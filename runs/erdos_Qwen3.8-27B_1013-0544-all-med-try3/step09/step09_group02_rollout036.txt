def project_box_sum(v, target):
    """Project v onto {0≤x≤1, sum(x)=target}"""
    n = len(v)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(v - tau, 0, 1)

def project_box_sum(v, target):
    n = len(v)
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(v - tau, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
        else:
            mask2 = (result == 0.0) | (result == 1.0)
            if mask2.sum() > 0:
                adj = diff / mask2.sum()
                result[mask2] = np.clip(result[mask2] + adj, 0.0, 1.0)
    return result

        idx = int(idx)
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
            if nv > 0:
                grad[k:n] -= wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
            if nv > 0:
                grad[0:nv] -= wm * h[kk:n]
    return grad

def lbfgs_smooth(h, n, alpha, deadline, maxiter=20000):
    h = project_box_sum(h.copy(), n / 2.0)
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        c = compute_c5_vec(x, n)
        return smooth_obj_value(c, alpha)

    def grad_fn(x):
        c = compute_c5_vec(x, n)
        w = smooth_weights(c, alpha)
        return compute_grad(x, n, w)

    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': 50000, 'maxls': 50})
        h_new = project_box_sum(res.x, n / 2.0)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_box_sum(h, n / 2.0), compute_c5(h, n)

def fista_smooth(h, n, alpha, deadline, max_iter=200000):
    h = project_box_sum(h.copy(), n / 2.0)
    y = h.copy()
    x_prev = h.copy()
    t_param = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)

    c = compute_c5_vec(y, n)
    w = smooth_weights(c, alpha)
    grad = compute_grad(y, n, w)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h, best_c
    L = alpha * 4.0 * (2.0 / n) + 0.01
    step = 1.0 / L

    for it in range(max_iter):
        if time.time() > deadline:
            break

        x_new = project_box_sum(y - step * grad, n / 2.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - x_prev)
        t_param = t_new
        x_prev = x_new

        c = compute_c5_vec(y, n)
        w = smooth_weights(c, alpha)
        new_grad = compute_grad(y, n, w)
        new_gnorm = np.linalg.norm(new_grad)

        c_check = compute_c5_vec(x_new, n)
        val_old = smooth_obj_value(c, alpha)
        val_new = smooth_obj_value(c_check, alpha)

        if val_new > val_old + 1e-12 * max(1.0, abs(val_old)):
            step *= 0.5
            y = x_new.copy()
            grad = new_grad
            gnorm = new_gnorm
            t_param = 1.0
            x_prev = x_new.copy()
            if step < 1e-15:
                break
        else:
            step = min(step * 1.05, 5.0 / L)
            grad = new_grad

        if it % 50 == 0:
            c_val = compute_c5(x_new, n)
            if c_val < best_c:
                best_c = c_val
                best_h = x_new.copy()

    c_val = compute_c5(x_new, n)
    if c_val < best_c:
        best_c = c_val
        best_h = x_new.copy()

    return best_h, best_c

def sqp_step_active(h, n, active_indices, rho=0.03):
    c_vec = compute_c5_vec(h, n)
    n_ineq = len(active_indices) + 2 * n
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    row = 0
    for idx in active_indices:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[0:nv] = (1.0 - h[k:n])
                g[k:n] += -h[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:n] = (1.0 - h[0:nv])
                g[0:nv] += -h[kk:n]
        A_ub[row, :] = g
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i] - 1e-12
        row += 1
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i] - 1e-12
        row += 1

    try:
        from scipy.optimize import linprog
        c_obj = np.zeros(n)
        A_eq = np.zeros((1, n))
        A_eq[0, :] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-rho, rho)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def sqp_optimize(h, n, deadline, max_iter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        if len(active) > 15:
            top = np.argsort(-c_vec[active])[:15]
            active = active[top]
        delta = sqp_step_active(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-8:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_box_sum(h + delta, n / 2.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            if rho < 0.06:
                rho = min(rho * 1.5, 0.1)
        else:
            rho *= 0.5
            if rho < 1e-9:
                break
    return best_h, best_c

def local_search(h, n, deadline, max_rounds=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()

    for step in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
        rounds = 0
        while rounds < max_rounds and time.time() < deadline:
            rounds += 1
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]

            if len(active) == 0:
                break

            grad_sum = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                g = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    if nv > 0:
                        g[0:nv] = (1.0 - h[k:n])
                        g[k:n] += -h[0:nv]
                else:
                    kk = -k
                    nv = n - kk
                    if nv > 0:
                        g[kk:n] = (1.0 - h[0:nv])
                        g[0:nv] += -h[kk:n]
                grad_sum += g

            if len(active) > 0:
                grad_sum /= float(len(active))

            grad_sum += 0.001 * h
            grad_sum -= 0.001 * (1.0 - h)

            pos_candidates = np.where(h > step)[0]
            neg_candidates = np.where(h < 1.0 - step)[0]

            if len(pos_candidates) == 0 or len(neg_candidates) == 0:
                break

            i_sorted = pos_candidates[np.argsort(-grad_sum[pos_candidates])[:min(30, len(pos_candidates))]]
            j_sorted = neg_candidates[np.argsort(grad_sum[neg_candidates])[:min(30, len(neg_candidates))]]

            improved = False
            for i in i_sorted:
                if time.time() > deadline:
                    return h, best_c
                for j in j_sorted:
                    if i == j:
                        continue
                    h[int(i)] += step
                    h[int(j)] -= step
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[int(i)] -= step
                        h[int(j)] += step
                if improved:
                    break

            if not improved:
                break

    return h, best_c

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    num_ones = n // 2
    positions = rng.choice(n, size=num_ones, replace=False)
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_box_sum(h_init, n / 2.0)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS with escalating alpha at base resolution
    alpha_schedule = [50, 200, 1000, 5000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 500:
            break
        t_phase = min(time.time() + 45, deadline - 450)
        h_curr, c_curr = lbfgs_smooth(h, n, alpha, t_phase, maxiter=10000)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
            h = h_curr
        else:
            h = best_h

    # Phase 2: FISTA with high alpha at base resolution
    if time.time() < deadline - 350:
        t_phase = min(time.time() + 60, deadline - 300)
        h_curr, c_curr = fista_smooth(best_h, n, 20000, t_phase)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()

    # Phase 3: SQP refinement
    if time.time() < deadline - 250:
        t_phase = min(time.time() + 60, deadline - 200)
        h_curr, c_curr = sqp_optimize(best_h, n, t_phase, max_iter=100)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()

    # Phase 4: Local search at base resolution
    if time.time() < deadline - 150:
        t_phase = min(time.time() + 50, deadline - 100)
        h_curr, c_curr = local_search(best_h, n, t_phase, max_rounds=3000)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()

    # Phase 5: Try upsample to higher resolution
    n_high = max(n * 2, 1000)
    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_box_sum(h_high, n_high / 2.0)

        for alpha in [200, 2000, 20000]:
            if time.time() > deadline - 80:
                break
            t_phase = min(time.time() + 50, deadline - 70)
            h_high, c_high = lbfgs_smooth(h_high, n_high, alpha, t_phase, maxiter=8000)

        if time.time() < deadline - 60:
            t_phase = min(time.time() + 40, deadline - 50)
            h_high, c_high = fista_smooth(h_high, n_high, 50000, t_phase)

        if time.time() < deadline - 30:
            t_phase = deadline - 15
            h_high, c_high = local_search(h_high, n_high, t_phase, max_rounds=2000)

        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high.copy()
            n = n_high

    # Phase 6: Final SQP polish at best resolution
    if time.time() < deadline - 20:
        t_phase = deadline - 5
        h_final, c_final = sqp_optimize(best_h, len(best_h), t_phase, max_iter=50)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
