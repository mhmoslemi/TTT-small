    m = c.max()

        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(80):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 10:
                break
            maxit = max(150, int(rem2 * 1.2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.002), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 8,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def de_evaluate(h, n):
    return compute_c5(h, n)

def de_generate_population(h_start, n, npop, rng, scale=0.3):
    pop = np.zeros((npop, n))
    pop[0] = h_start.copy()
    for i in range(1, npop):
        pert = rng.normal(0, scale, size=n)
        cand = h_start + pert
        pop[i] = project_feasible(cand, n)
    return pop

def de_generated_fitness(pop, n):
    fitness = np.zeros(len(pop))
    for i in range(len(pop)):
        fitness[i] = compute_c5(pop[i], n)
    return fitness

def de_mutate_crossover(pop, npop, n, F, CR, rng):
    v = pop[rng.randint(npop)] + F * (pop[rng.randint(npop)] - pop[rng.randint(npop)])
    mask = rng.random(n) < CR
    trial = np.where(mask, v, pop[rng.randint(npop)])
    return project_feasible(trial, n)

def run_memetic_de(h_start, n, deadline, npop=35, F=0.7, CR=0.85,
                   refine_every=8, refine_steps=80, scale=0.25):
    rng = np.random.default_rng(42)
    pop = de_generate_population(h_start, n, npop, rng, scale)
    fitness = de_generated_fitness(pop, n)
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]
    
    gen = 0
    stall_count = 0
    F_cur = F
    scale_cur = scale
    
    while time.time() < deadline:
        gen += 1
        for i in range(npop):
            if time.time() > deadline - 5:
                break
            idx_a = rng.integers(0, npop)
            idx_b = rng.integers(0, npop)
            idx_c = rng.integers(0, npop)
            while idx_b == idx_a:
                idx_b = rng.integers(0, npop)
            while idx_c == idx_a or idx_c == idx_b:
                idx_c = rng.integers(0, npop)
            
            v = pop[idx_a] + F_cur * (pop[idx_b] - pop[idx_c])
            mask = rng.random(n) < CR
            trial = np.where(mask, v, pop[i])
            trial = project_feasible(trial, n)
            f_trial = compute_c5(trial, n)
            
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c:
                    best_c = f_trial
                    best_h = trial.copy()
                    stall_count = 0
            else:
                if rng.random() < 0.05:
                    pert = rng.normal(0, scale_cur * 0.1, n)
                    jitter = project_feasible(pop[i] + pert, n)
                    f_j = compute_c5(jitter, n)
                    if f_j < fitness[i]:
                        pop[i] = jitter
                        fitness[i] = f_j
        
        # Memetic refinement
        if gen % refine_every == 0 and time.time() < deadline - 20:
            bi = np.argmin(fitness)
            refine_start = time.time()
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(sm_obj, pop[bi], jac=sm_grad,
                               args=(n, 50000, 0.01), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': refine_steps, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': refine_steps * 5})
                h_ref = project_feasible(res.x, n)
                c_ref = compute_c5(h_ref, n)
                pop[bi] = h_ref
                fitness[bi] = c_ref
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = h_ref.copy()
            except Exception:
                pass
        
        # Adaptive parameters
        if gen % 20 == 0:
            stall_count += 1
            if stall_count > 3:
                F_cur = max(0.3, F_cur * 0.9)
                scale_cur = min(0.5, scale_cur * 1.2)
                stagnation_pop = fitness.copy()
                stagnation_pop[best_idx] = np.inf
                worst = np.argmax(fitness)
                if worst != best_idx:
                    pop[worst] = project_feasible(
                        best_h + rng.normal(0, scale_cur, n), n)
                    fitness[worst] = compute_c5(pop[worst], n)
            stall_count = max(0, stall_count - 1)
    
    return best_h, best_c

def run_de_multi_start(h_start, n, deadline, n_starts=3):
    rng = np.random.default_rng(123)
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    
    time_per_start = (deadline - time.time()) / n_starts - 10
    
    for s in range(n_starts):
        rem = deadline - time.time()
        if rem < 30:
            break
        if s == 0:
            start_h = h_start.copy()
        else:
            pert = rng.normal(0, 0.15, n)
            start_h = project_feasible(h_start + pert, n)
        
        block_deadline = time.time() + min(time_per_start, rem - 15)
        h_s, c_s = run_memetic_de(start_h, n, block_deadline,
                                   npop=30 + 5 * s,
                                   F=0.6 + 0.1 * s,
                                   CR=0.8 + 0.05 * s,
                                   refine_every=6,
                                   refine_steps=60,
                                   scale=0.2 + 0.05 * s)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_sweeps=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for ss in step_sizes:
                old_val = best_h[i]
                new_val = min(1.0, old_val + ss)
                if abs(new_val - old_val) < 1e-12:
                    continue
                best_h[i] = new_val
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
                    new_val = max(0.0, old_val - ss)
                    if abs(new_val - old_val) < 1e-12:
                        continue
                    best_h[i] = new_val
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] = old_val
                        best_h = project_feasible(best_h, n)
        if not improved and sweep > 50:
            break
    return best_h, best_c

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.015)
        else:
            eta *= 0.96
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = min(t0 + 45, deadline - 600)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
                        alphas=[100, 1000, 10000, 50000],
                        mu=0.01, maxiters=[800, 1200, 1500, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Memetic DE for global exploration
    p2_end = min(t0 + 320, deadline - 400)
    if time.time() < p2_end - 30:
        h2, c2 = run_de_multi_start(best_h, best_n, p2_end, n_starts=3)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        else:
            best_h = best_h  # keep current best
    
    # Phase 3: L-BFGS polish after DE
    p3_end = min(time.time() + 60, deadline - 300)
    h3, c3 = run_lbfgs(best_h, best_n, p3_end,
                        alphas=[10000, 50000, 200000, 500000],
                        mu=0.005, maxiters=[1500, 2000, 2500, 3000])
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Upsample to higher resolution
    n_high = 1500
    p4_end = min(time.time() + 150, deadline - 200)
    if time.time() < p4_end - 20 and best_n < n_high:
        h4 = upsample(best_h, best_n, n_high)
        c4 = compute_c5(h4, n_high)
        if c4 < best_c + 0.003:
            h4, c4b = run_lbfgs(h4, n_high, p4_end,
                                 alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
                                 mu=0.003, maxiters=[500, 700, 1000, 1200, 1500, 2000, 2500])
            if c4b < best_c:
                best_c = c4b
                best_h = h4
                best_n = n_high
    
    # Phase 5: Try even higher resolution
    n_vhigh = 2500
    p5_end = min(time.time() + 120, deadline - 130)
    if time.time() < p5_end - 20 and best_n < n_vhigh:
        h5 = upsample(best_h, best_n, n_vhigh)
        c5_init = compute_c5(h5, n_vhigh)
        if c5_init < best_c + 0.004:
            h5, c5v = run_lbfgs(h5, n_vhigh, p5_end,
                                 alphas=[5, 20, 100, 500, 2000, 10000],
                                 mu=0.002, maxiters=[400, 600, 800, 1000, 1500, 2000])
            if c5v < best_c:
                best_c = c5v
                best_h = h5
                best_n = n_vhigh
    
    # Phase 6: Active set optimization
    p6_end = min(time.time() + 100, deadline - 80)
    if time.time() < p6_end - 15:
        h6, c6 = run_active_opt(best_h, best_n, p6_end, top_k=5,
                                alphas=[50000, 200000, 1000000, 5000000, 20000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Coordinate descent
    p7_end = min(time.time() + 80, deadline - 40)
    if time.time() < p7_end - 10:
        h7, c7 = coordinate_descent(best_h, best_n, p7_end, n_sweeps=300)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Pair search
    p8_end = min(time.time() + 50, deadline - 20)
    if time.time() < p8_end - 5:
        h8, c8 = pair_search(best_h, best_n, p8_end, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Subgradient refinement
    p9_end = deadline - 3
    if time.time() < p9_end - 5:
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, n_iter=6000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
