def add_grad_c_k(g,h,n,k,coeff):
    if coeff == 0.0: return
    if k >= 0:
       gv=n-k
       if gv<=0: return
       g[k:n] += coeff*(1.0 - h[0:gv])

def project_feasible(h,n):
   h=np.asarray(h,dtype=float).copy()
   h=np.clip(h,0,1)
   target=n/2.0
   for _ in range(2000):
      diff=target-h.sum()
      if abs(diff)<1e-14: break
      if diff>0:
         room=1.0-h
         tot=room.sum()
         if tot<1e-15: break
         h += (diff/tot)*room
      else:
         tot=h.sum()
         if tot<1e-15: break
         h += (diff/tot)*h
   h=np.clip(h,0,1)
   # final exact distribution
   h=enforce_sum_exact(h,n)
   return h

def enforce_sum_exact(h,n):

def active_obj(x,n,act,w,mu):
   c=fast_corr_counts(x,n)
   dx=2.0/n
   val=dx*float(np.dot(w,c[act]))
   T=n/2.0
   diff=x.sum()-T
   val += mu*(diff/T)**2
   return float(val)

def active_grad(x,n,act,w,mu):
   dx=2.0/n
   g=np.zeros(n)
   for idx,wi in zip(act,w):
      k=idx-(n-1)
      add_grad_c_k(g,x,n,k,wi)
   g *= dx
   T=n/2.0
   diff=x.sum()-T
   g += 2.0*mu*diff/(T*T)
   return g

import time
import math
import numpy as np
from scipy.optimize import minimize


def compute_c5_direct(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def compute_c_counts_direct(h, n):
    return np.asarray(np.correlate(h, 1.0 - h, mode="full"), dtype=float)


def fast_corr_counts(h, n):
    if n <= 0:
        return np.zeros(0, dtype=float)
    if n == 1:
        return np.array([float(h[0] * (1.0 - h[0]))], dtype=float)

    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1

    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    r = np.fft.irfft(A * np.conj(B), N)

    out = np.concatenate((r[-(n - 1):], r[:n]))
    return np.maximum(out, 0.0)


def add_grad_c_k(g, h, n, k, coeff):
    if coeff == 0.0:
        return
    if k >= 0:
        gv = n - k
        if gv <= 0:
            return
        g[k:n] += coeff * (1.0 - h[0:gv])
        g[0:gv] += coeff * (-h[k:n])
    else:
        r = -k
        gv = n - r
        if gv <= 0:
            return
        g[0:gv] += coeff * (1.0 - h[r:n])
        g[r:n] += coeff * (-h[0:gv])


def enforce_sum_exact(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            add = diff
            for idx in order:
                if add <= 1e-15:
                    break
                sp = room[idx]
                if sp > add:
                    sp = add
                if sp > 0.0:
                    h[idx] += sp
                    add -= sp
        else:
            order = np.argsort(h)
            add = -diff
            for idx in order:
                if add <= 1e-15:
                    break
                sp = h[idx]
                if sp > add:
                    sp = add
                if sp > 0.0:
                    h[idx] -= sp
                    add -= sp
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(h, n):

    target = n / 2.0

            if tot < 1e-15:

    if source_n <= 1:

    h = np.interp(x_new, x_old, source_h)

def make_beatty(n, f):

def active_obj(x, n, act, w, mu):
    c = fast_corr_counts(x, n)
    dx = 2.0 / n
    val = dx * float(np.dot(w, c[act]))
    T = n / 2.0
    diff = x.sum() - T
    val += mu * (diff / T) ** 2
    return float(val)


def active_grad(x, n, act, w, mu):
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    for idx, wi in zip(act, w):
        k = int(idx - (n - 1))
        add_grad_c_k(g, x, n, k, float(wi))
    g *= dx
    T = n / 2.0
    diff = x.sum() - T
    g += 2.0 * mu * diff / (T * T)
    return g


def active_gradient_counts(h, n, c_counts, topk=16):

    topk = min(int(topk), len(c_counts))

                        "maxls": 20,
                    },
                )
                x_new = res.x
            except Exception:
                break

            if x_new is None or not np.all(np.isfinite(x_new)):
                break

            h_new = project_feasible(x_new, n)
            c_new = compute_c5_direct(h_new, n)

            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif c_new < c_now - 1e-10:
                h = h_new.copy()
            else:
                break

    h = best_h
    c = compute_c5_direct(h, n)
    if c < best_c:
        best_c = c
    return best_h, best_c


def swap_eval_counts(c, h, z, n, a, b):
    temp = c.copy()

    s = n - 1 - a
    temp[s:s + n] += h
    s = n - 1 - b
    temp[s:s + n] -= h

    revz = z[::-1]
    temp[a:a + n] -= revz
    temp[b:b + n] += revz

    temp[n - 1] -= 2.0
    lag = b - a
    temp[lag + n - 1] += 1.0
    lag = a - b
    temp[lag + n - 1] += 1.0

    return temp


def binary_optimize(h_init, n, deadline, rng=None):
    if n < 4:
        h = project_feasible(np.asarray(h_init, dtype=float).copy(), n)
        return h, compute_c5_direct(h, n)

    if rng is None:
        rng = np.random.default_rng(777)

    h = quantize_binary(h_init, n)
    z = 1.0 - h
    c = compute_c_counts_direct(h, n)
    dx = 2.0 / n

    current_c = float(c.max()) * dx
    best_c = current_c
    best_h = h.copy()

    if n <= 300:
        p_start = 10
        p_max = 22
    elif n <= 800:
        p_start = 10
        p_max = 16
    else:
        p_start = 8
        p_max = 12

    p = p_start
    stall = 0
    accept_count = 0
    temp = 0.005

    while time.time() < deadline - 0.5:
        if current_c > best_c + 0.010:
            h = best_h.copy()
            z = 1.0 - h
            c = compute_c_counts_direct(h, n)
            current_c = best_c

        g, _, _ = active_gradient_counts(h, n, c, topk=16)
        ones = np.nonzero(h > 0.5)[0]
        zeros = np.nonzero(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if len(ones) > p:
            one_pool = ones[np.argsort(-g[ones])[:p]]
        else:
            one_pool = ones
        if len(zeros) > p:
            zero_pool = zeros[np.argsort(g[zeros])[:p]]
        else:
            zero_pool = zeros

        best_C = float("inf")
        best_ab = None
        best_vec = None

        for a in one_pool:
            if time.time() >= deadline - 0.2:
                break
            for b in zero_pool:
                c_new = swap_eval_counts(c, h, z, n, int(a), int(b))
                Cnew = float(c_new.max()) * dx
                if Cnew < best_C - 1e-14:
                    best_C = Cnew
                    best_ab = (int(a), int(b))
                    best_vec = c_new

        if rng.random() < 0.25:
            rcount = min(18, p)
            for _ in range(rcount):
                if time.time() >= deadline - 0.2:
                    break
                a = int(ones[int(rng.integers(0, len(ones)))])
                b = int(zeros[int(rng.integers(0, len(zeros)))])
                c_new = swap_eval_counts(c, h, z, n, a, b)
                Cnew = float(c_new.max()) * dx
                if Cnew < best_C - 1e-14:
                    best_C = Cnew
                    best_ab = (a, b)
                    best_vec = c_new

        if best_vec is None or best_ab is None:
            stall += 1
            continue

        a, b = best_ab

        if best_C < current_c - 1e-12:
            h[a] = 0.0
            h[b] = 1.0
            z = 1.0 - h
            c = best_vec
            current_c = best_C
            if best_C < best_c - 1e-14:
                best_c = best_C
                best_h = h.copy()
            stall = 0
            accept_count += 1
            if accept_count % 60 == 0:
                c = compute_c_counts_direct(h, n)
                current_c = float(c.max()) * dx
                if current_c < best_c:
                    best_c = current_c
                    best_h = h.copy()
        else:
            if stall >= 30 and temp > 1e-9 and best_C <= current_c + temp:
                diff = max(0.0, best_C - current_c)
                accept = False
                if diff <= 1e-13:
                    accept = True
                else:
                    x = diff / temp
                    if x < 50.0:
                        accept = rng.random() < math.exp(-x)
                    else:
                        accept = False
                if accept:
                    h[a] = 0.0
                    h[b] = 1.0
                    z = 1.0 - h
                    c = best_vec
                    current_c = best_C
                    if best_C < best_c - 1e-14:
                        best_c = best_C
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                    temp *= 0.95
                else:
                    temp *= 0.90
                    stall += 1
            else:
                if p < p_max:
                    p += 2
                stall += 1
                if stall >= 100:
                    h = best_h.copy()
                    ones = np.nonzero(h > 0.5)[0]
                    zeros = np.nonzero(h < 0.5)[0]
                    for _ in range(3):
                        if len(ones) == 0 or len(zeros) == 0:
                            break
                        a = int(ones[int(rng.integers(0, len(ones)))])
                        b = int(zeros[int(rng.integers(0, len(zeros)))])
                        h[a] = 0.0
                        h[b] = 1.0
                    z = 1.0 - h
                    c = compute_c_counts_direct(h, n)
                    current_c = float(c.max()) * dx
                    if p > p_start:
                        p = p_start
                    temp = max(temp, 0.003)
                    stall = 0

        if temp > 1e-7:
            temp *= 0.995

    best_c = compute_c5_direct(best_h, n)
    return best_h, best_c


def slqp_sum_eq(x, n):
    return float(x.sum() - n / 2.0)


def slqp_sum_jac(x, n):
    return np.ones(n, dtype=float)


def slqp_obj(x, n, act, w):
    c = fast_corr_counts(x, n)
    return float((2.0 / n) * np.dot(w, c[act]))


def slqp_grad(x, n, act, w):
    g = np.zeros(n, dtype=float)
    for idx, wi in zip(act, w):
        k = int(idx - (n - 1))
        add_grad_c_k(g, x, n, k, float(wi))
    g *= 2.0 / n
    return g


def slqp_polish(h, n, deadline):
    h = project_feasible(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5_direct(best_h, n)
    if n < 4:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": slqp_sum_eq,
            "jac": slqp_sum_jac,
            "args": (n,),
        }
    ]

    for topk in (8, 16):
        if time.time() >= deadline - 2.0:
            break
        c_counts = fast_corr_counts(h, n)
        topk_eff = min(int(topk), max(1, len(c_counts) - 1))
        act = np.argsort(-c_counts)[:topk_eff]
        vals = c_counts[act]
        w = vals - float(vals.min()) + 1e-8
        w = np.maximum(w, 0.0)
        if w.sum() <= 1e-15:
            w = np.ones(len(act), dtype=float)
        else:
            w = w / w.sum()

        try:
            res = minimize(
                slqp_obj,
                h,
                jac=slqp_grad,
                args=(n, act, w),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 220, "ftol": 1e-14},
            )
            x_new = res.x
        except Exception:
            break

        if x_new is None or not np.all(np.isfinite(x_new)):
            continue
        h_new = project_feasible(x_new, n)
        c_new = compute_c5_direct(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()

    best_c = compute_c5_direct(best_h, n)
    return best_h, best_c


def choose_candidate_list(base, current_best_n):
    raw = [
        base // 2,
        base,
        base * 2,
        max(200, base * 3 // 2),
        200,
        250,
        300,
        400,
        600,
        800,
        1000,
        1200,
    ]
    seen = set()
    ordered = []
    for nn in raw:
        nn = int(nn)
        if nn < 80 or nn > 1200:
            continue
        if nn % 2 != 0:
            nn -= 1
        if nn == current_best_n:
            continue
        if nn in seen:
            continue
        seen.add(nn)
        ordered.append(nn)

    pairs = []
    for nn in ordered:
        pairs.append((abs(int(nn) - int(base)), int(nn)))
    pairs.sort()

    out = []
    for _, nn in pairs:
        out.append(nn)
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + max(10.0, budget - 10.0)
    rng = np.random.default_rng(int(seed))

    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel().copy()
        n0 = int(len(h_init))
    except Exception:
        n0 = 500
        h_init = make_random_smooth(n0, rng)

    if n0 < 4:
        n0 = 200
        h_init = make_random_smooth(n0, rng)

    best_h = project_feasible(h_init, n0)
    best_n = n0
    best_c = compute_c5_direct(best_h, best_n)

    rem = deadline - time.time()
    if rem > 30.0:
        cont_time = min(0.18 * budget, max(20.0, rem * 0.25), rem - 5.0)
        cont_end = time.time() + max(10.0, cont_time)
        if cont_end > deadline - 5.0:
            cont_end = deadline - 5.0
        if cont_end > time.time() + 5.0:
            hc, cc = continuous_refine(best_h, best_n, cont_end)
            if cc < best_c:
                best_c = cc
                best_h = hc
                best_n = int(len(hc))

    nb = best_n if (best_n % 2 == 0) else best_n - 1
    if nb >= 4:
        rem = deadline - time.time()
        if rem > 60.0:
            bin_time = min(0.35 * budget, max(40.0, rem * 0.45), rem - 5.0)
            bin_end = time.time() + max(20.0, bin_time)
            if bin_end > deadline - 50.0:
                bin_end = deadline - 50.0
            if bin_end > time.time() + 10.0:
                if len(best_h) != nb:
                    src = make_interpolated(best_h, len(best_h), nb)
                else:
                    src = best_h
                hb, cb = binary_optimize(src, nb, bin_end, rng)
                if cb < best_c:
                    best_c = cb
                    best_h = hb
                    best_n = nb

    base = best_n if (best_n % 2 == 0) else best_n - 1
    candidates = choose_candidate_list(base, best_n)
    for i, nn in enumerate(candidates):
        rem = deadline - time.time()
        if rem < 30.0:
            break
        n_left = len(candidates) - i
        alloc = rem / float(max(1, n_left))
        alloc = min(120.0, alloc)
        if alloc < 20.0:
            continue
        if nn > 900 and alloc < 45.0:
            continue
        end = time.time() + alloc
        if end > deadline - 20.0:
            end = deadline - 20.0
        if end <= time.time() + 10.0:
            continue
        src = make_interpolated(best_h, best_n, nn)
        hb, cb = binary_optimize(src, nn, end, rng)
        if cb < best_c:
            best_c = cb
            best_h = hb
            best_n = nn

    rem = deadline - time.time()
    if rem > 80.0:
        n_try = best_n if (best_n % 2 == 0) else best_n - 1
        if n_try < 100:
            n_try = 200
        if n_try > 800:
            n_try = 600
        for f in (2.0, 2.5, 3.0, 3.2360679, 3.5, 4.0):
            rem = deadline - time.time()
            if rem < 25.0:
                break
            alloc = min(25.0, rem * 0.30)
            end = time.time() + alloc
            if end > deadline - 20.0:
                end = deadline - 20.0
            if end <= time.time() + 8.0:
                break
            start_h = make_beatty(n_try, float(f))
            hb, cb = binary_optimize(start_h, n_try, end, rng)
            if cb < best_c:
                best_c = cb
                best_h = hb
                best_n = n_try

        rem = deadline - time.time()
        if rem > 35.0:
            alloc = min(35.0, rem * 0.6)
            end = time.time() + alloc
            if end > deadline - 10.0:
                end = deadline - 10.0
            if end > time.time() + 8.0:
                start_h = make_random_smooth(n_try, rng)
                hb, cb = binary_optimize(start_h, n_try, end, rng)
                if cb < best_c:
                    best_c = cb
                    best_h = hb
                    best_n = n_try

    rem = deadline - time.time()
    if rem > 80.0:
        end = time.time() + min(90.0, rem * 0.55)
        if end > deadline - 20.0:
            end = deadline - 20.0
        if end > time.time() + 10.0:
            hc, cc = continuous_refine(best_h, best_n, end)
            if cc < best_c:
                best_c = cc
                best_h = hc
                best_n = int(len(hc))

    rem = deadline - time.time()
    if rem > 45.0:
        h_pert = project_feasible(best_h + rng.normal(0.0, 0.05, best_n), best_n)
        end = deadline - 15.0
        if end > time.time() + 10.0:
            hc, cc = continuous_refine(h_pert, best_n, end)
            if cc < best_c:
                best_c = cc
                best_h = hc
                best_n = int(len(hc))

    rem = deadline - time.time()
    if rem > 18.0:
        hs, cs = slqp_polish(best_h, best_n, deadline - 3.0)
        if cs < best_c:
            best_c = cs
            best_h = hs
            best_n = int(len(hs))

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_direct(best_h, best_n)
    return best_h, float(best_c), int(best_n)
