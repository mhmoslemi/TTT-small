def project_to_feasible(h, target_integral=1.0):
    integral = np.sum(h * dx)
    if abs(integral-target_integral) < 1e-15:
        return h
    # Adjust with Gaussian redistribution
    diff = target_integral - integral
    if diff == 0:
        return h
    # Find the most integral in-efficient part and adjust it
