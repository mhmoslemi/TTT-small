        active = c[top_idx]
    else:
        active = c
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def power_mean_obj(x, n, p, top_idx=None):
    c = compute_all_lags(x, n)
    if top_idx is not None:
        active = c[top_idx]
    else:
        active = c
    return float(np.mean(active**p) ** (1.0/p))

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_m is not None:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        def _obj(x):
            return smooth_max_obj(x, n, alpha, top_idx)
        try:
            res = minimize(_obj, h.copy(), method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        def _obj(x):
            return smooth_max_obj(x, n, alpha, None)
        try:
            res = minimize(_obj, h.copy(), method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

def power_mean_slsqp(h, n, p, maxiter=2000, top_m=5):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    def _obj(x):
        return power_mean_obj(x, n, p, top_idx)
    try:
        res = minimize(_obj, h.copy(), method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_momentum_focused(h, n, deadline, top_m=5, lr_init=1.5, momentum=0.9):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr_init
    patience = 0
    
    for it in range(200000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_m:]
        m_max = c_all.max()
        
        # Compute gradient of power mean over top lags
        active = c_all[top_idx]
        p = 10
        w = active ** (p - 1)
        w_norm = w / (w.sum() + 1e-30)
        
        g = np.zeros(n)
        for ti in range(len(top_idx)):
            lag = top_idx[ti] - (n - 1)
            g += w_norm[ti] * lag_gradient_vec(h, n, lag)
        
        # Nesterov-style
        v = momentum * v + lr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                lr *= 0.7
                v *= 0.3
                patience = 0
            elif patience > 200:
                lr *= 0.85
                v *= 0.5
                patience = 0
        
        h = h_new
        if it % 3000 == 2999:
            lr = max(0.05, lr * 0.9)
    
    return best_h, best_c

def analytical_local_search(h, n, deadline, max_rounds=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[-5:]
        improved = 0
        
        for k_idx in top_lags:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            
            if lag > 0:
                # c[m] = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
                # To decrease: decrease h[i] or increase h[i+m]
                # For pair (i, i+m): transfer from h[i] to h[i+m]
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    # Try moving small amount from h[i] to h[j] (decreases c[m])
                    # Also check effect on c[-m] where this pair contributes h[j]*(1-h[i])
                    
                    # Analytical optimal: minimize h[i]*(1-h[j]) subject to h[i]+h[j] = const
                    # Optimal: h[i] = 0 or h[j] = 1 (push to extremes)
                    # But with bounds, use smaller steps
                    if h[i] > 0.01:
                        for delta in [0.05, 0.02, 0.01, 0.005]:
                            if delta >= h[i] * 0.8 or delta >= (1 - h[j]) * 0.8:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved += 1
                                break
                            else:
                                h[i] += delta
                                h[j] -= delta
                        else:
                            h[i] += 0  # no change made
                    # Also try reverse: increase h[i], decrease h[j]
                    # This increases c[m] but might decrease other lags
                    # Skip for now, focus on decreasing max
                    
            else:
                # lag < 0, m = -lag
                # c[-m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
                # To decrease: decrease h[i+m] or increase h[i]
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[j] > 0.01:
                        for delta in [0.05, 0.02, 0.01, 0.005]:
                            if delta >= h[j] * 0.8 or delta >= (1 - h[i]) * 0.8:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                                improved += 1
                                break
                            else:
                                h[j] += delta
                                h[i] -= delta
                        else:
                            pass
                    
        # Broader search: try random swaps at non-top lags too
        c_all = compute_all_lags(h, n)
        for k_idx in np.argsort(c_all)[-15:]:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for _ in range(min(n - m, 100)):
                if time.time() > deadline:
                    break
                i = np.random.randint(0, n - m)
                j = i + m
                if lag > 0:
                    if h[i] > h[j] + 0.05:
                        delta = min(0.03, (h[i] - h[j]) * 0.3)
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved += 1
                        else:
                            h[i] += delta
                            h[j] -= delta
                else:
                    if h[j] > h[i] + 0.05:
                        delta = min(0.03, (h[j] - h[i]) * 0.3)
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved += 1
                        else:
                            h[j] += delta
                            h[i] -= delta
        
        if improved == 0:
            # Try finer perturbations
            c_all = compute_all_lags(h, n)
            k_star = int(np.argmax(c_all))
            lag = k_star - (n - 1)
            m = abs(lag)
            found = False
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    if h[i] > 0.005 and h[j] < 0.995:
                        delta = 0.002
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            found = True
                            improved += 1
                        else:
                            h[i] += delta
                            h[j] -= delta
                else:
                    if h[j] > 0.005 and h[i] < 0.995:
                        delta = 0.002
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            found = True
                            improved += 1
                        else:
                            h[j] += delta
                            h[i] -= delta
                if found:
                    break
            if improved == 0:
                break
    
    return best_h, best_c

def quantize_and_search(h, n, deadline):
    """Try quantizing toward binary and doing combinatorial search"""
    # Quantize: push values toward 0 or 1
    h_q = h.copy()
    # Use threshold at 0.5 with adjustment for sum constraint
    threshold = 0.5
    h_bin = (h_q > threshold).astype(float)
    # Adjust to get exactly n/2 ones
    num_ones = int(h_bin.sum())
    if num_ones > n // 2:
        # Set excess largest-below-threshold to 0
        below = np.where(h_bin == 0)[0]
        above = np.where(h_bin == 1)[0]
        # Flip excess ones to zero (choose ones closest to 0.5)
        excess = num_ones - n // 2
        to_flip = above[np.argsort(h_q[above])[:excess]]
        h_bin[to_flip] = 0
    elif num_ones < n // 2:
        deficit = n // 2 - num_ones
        below = np.where(h_bin == 0)[0]
        to_flip = below[np.argsort(-h_q[below])[:deficit]]
        h_bin[to_flip] = 1
    
    c_bin = compute_c5(h_bin, n)
    if c_bin < compute_c5(h, n):
        print(f"  Binary better: {c_bin:.8f}")
    
    # Also try: keep continuous but push more toward binary
    h_pushed = np.clip(h_q * 3.0 - 1.0, 0, 1)  # stretch toward {0,1}
    h_pushed = project_feasible(h_pushed, n)
    c_pushed = compute_c5(h_pushed, n)
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    if c_pushed < best_c:
        best_c = c_pushed
        best_h = h_pushed.copy()
    
    # Local search on the best
    if time.time() < deadline:
        best_h, best_c = analytical_local_search(best_h, n, deadline)
    
    return best_h, best_c

def multi_resolution_optimize(h_start, n_start, deadline):
    """Try multiple resolutions and return best"""
    t0 = time.time()
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n_start)
    
    # Optimize at starting resolution
    remaining = deadline - t0
    phase_alloc = remaining * 0.3
    
    # Phase 1: Gradient momentum
    h1, c1 = grad_momentum_focused(h_start, n_start, t0 + phase_alloc * 0.7, top_m=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: SLSQP
    if time.time() < deadline - 20:
        h2 = slsqp_optimize(best_h, n_start, 500000, maxiter=2000, ftol=1e-14, top_m=4)
        c2 = compute_c5(h2, n_start)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        
        h3 = power_mean_slsqp(best_h, n_start, p=20, maxiter=1500, top_m=5)
        c3 = compute_c5(h3, n_start)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 3: Local search
    if time.time() < deadline - 10:
        ls_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.5)
        best_h, best_c = analytical_local_search(best_h, n_start, ls_deadline)
    
    # Try higher resolutions
    higher_ns = [n_start + 50, n_start + 100, n_start + 200, n_start + 300]
    for n_new in higher_ns:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 30
        if remaining < 50:
            break
        
        # Interpolate
        x_old = np.linspace(0, 2, n_start)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_new)
        
        alloc = min(100, remaining * 0.4)
        if alloc < 40:
            break
        
        # Quick optimization at new resolution
        h_opt, c_opt = grad_momentum_focused(h_new, n_new, time.time() + alloc * 0.5, top_m=5)
        
        if time.time() < deadline - 40:
            h_opt = slsqp_optimize(h_opt, n_new, 300000, maxiter=1500, ftol=1e-14, top_m=4)
        
        if time.time() < deadline - 20:
            h_opt, c_opt = analytical_local_search(h_opt, n_new, min(deadline - 5, time.time() + 20))
        
        c_opt = compute_c5(h_opt, n_new)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_start = n_new
    
    # Quantization experiment
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        hq, cq = quantize_and_search(best_h, n_cur, min(deadline - 3, time.time() + 25))
        if cq < best_c:
            best_c = cq
            best_h = hq
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Main optimization at starting resolution
    main_alloc = min(500, (deadline - t0) * 0.55)
    best_h, best_c = multi_resolution_optimize(best_h, n_init, t0 + main_alloc)
    
    # Multi-start: try different perturbations at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        n_restarts = max(1, int((deadline - time.time() - 40) / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 45:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            
            alloc = min(55, (deadline - time.time() - 40) * 0.35)
            if alloc < 25:
                break
            
            # Quick gradient + SLSQP + local search
            h_opt, c_opt = grad_momentum_focused(hp, n_best, time.time() + alloc * 0.4, top_m=5)
            if time.time() < deadline - 30:
                h_opt = slsqp_optimize(h_opt, n_best, 200000, maxiter=1000, ftol=1e-13, top_m=4)
            if time.time() < deadline - 15:
                h_opt, c_opt = analytical_local_search(h_opt, n_best, min(deadline - 5, time.time() + 15))
            
            c_opt = compute_c5(h_opt, n_best)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Try a completely different starting point
    if time.time() < deadline - 60:
        # Structured start: alternating blocks
        n_alt = n_best
        h_alt = np.zeros(n_alt)
        block = n_alt // 10
        for i in range(0, n_alt, 2 * block):
            end = min(i + block, n_alt)
            h_alt[i:end] = 1.0
        h_alt = project_feasible(h_alt, n_alt)
        
        alloc = min(50, (deadline - time.time() - 30) * 0.5)
        if alloc > 20:
            h_opt, c_opt = grad_momentum_focused(h_alt, n_alt, time.time() + alloc * 0.5, top_m=5)
            if time.time() < deadline - 20:
                h_opt = slsqp_optimize(h_opt, n_alt, 200000, maxiter=1000, ftol=1e-13, top_m=4)
            if time.time() < deadline - 10:
                h_opt, c_opt = analytical_local_search(h_opt, n_alt, min(deadline - 3, time.time() + 10))
            
            c_opt = compute_c5(h_opt, n_alt)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Final SLSQP with very tight tolerance
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 1000000, maxiter=2000, ftol=1e-16, top_m=3)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = analytical_local_search(best_h, n, deadline - 0.5)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
