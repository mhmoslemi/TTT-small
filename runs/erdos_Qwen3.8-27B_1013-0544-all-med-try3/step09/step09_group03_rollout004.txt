import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, ftol=1e-18):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 2000.0), (10000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        t_start = time.time()
        try:
            maxiter = max(1000, int((deadline - 10 - t_start) * 200))
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': min(maxiter, 50000), 'ftol': ftol,
                                   'gtol': 1e-14, 'maxfun': min(maxiter*10, 200000), 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def generate_binary_pattern(n, rng):
    """Generate a random binary pattern with exactly n/2 ones."""
    idx = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def generate_block_pattern(n, rng):
    """Generate a pattern with alternating blocks."""
    n_blocks = rng.randint(4, 20)
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        if i % 2 == 0:
            h[i*block_size:(i+1)*block_size] = 1.0
    # Add some randomness
    h += 0.3 * (rng.rand(n) - 0.5)
    return np.clip(h, 0, 1)

    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)

def coordinate_descent(h, n, deadline, n_sweeps=100):
    """Paired coordinate descent: adjust pairs (i,j) to reduce max correlation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        
        # Random permutation of coordinates
        perm = rng.permutation(n)
        improved = False
        
        for i_idx in range(0, n, 2):
            if time.time() > deadline:
                return best_h, best_c
            
            i = perm[i_idx]
            j = perm[i_idx + 1] if i_idx + 1 < n else (i + 1) % n
            
            if i == j:
                continue
            
            # Try different deltas
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.001, -0.001]:
                h[i] += delta
                h[j] -= delta
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                    h[i] -= delta
                    h[j] += delta
                    continue
                
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] -= delta
                    h[j] += delta
        
        if not improved and sweep > 20:
            break
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, ftol=1e-18):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 2000.0), (10000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        t_start = time.time()
        try:
            remaining = max(5.0, deadline - 10 - t_start)
            maxiter = min(50000, max(2000, int(remaining * 300)))
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol,
                                   'gtol': 1e-14, 'maxfun': min(maxiter * 10, 300000), 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def gen_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def gen_block(n, rng):
    n_blocks = rng.randint(4, 25)
    bs = max(1, n // n_blocks)
    h = np.zeros(n)
    starts = rng.choice(n - bs, size=min(n_blocks, (n - bs) // bs), replace=False)
    for s in starts:
        h[s:s + bs] = 1.0
    val_start = np.sum(h)
    if val_start > n / 2 + 1:
        excess = int(val_start - n / 2)
        ones = np.where(h > 0.5)[0]
        to_zero = rng.choice(ones, size=min(excess, len(ones)), replace=False)
        h[to_zero] = 0.0
    elif val_start < n / 2 - 1:
        deficit = int(n / 2 - val_start)
        zeros = np.where(h < 0.5)[0]
        to_one = rng.choice(zeros, size=min(deficit, len(zeros)), replace=False)
        h[to_one] = 1.0
    return h

def gen_sine(n, rng):
    freq = rng.randint(1, 6)
    phase = rng.rand() * 2 * np.pi
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return np.clip(h, 0, 1)

def gen_complement(h, n):
    return np.clip(1.0 - h, 0, 1)

def diverse_restart(h, n, deadline, n_trials, rng):

            h_p = gen_binary(n, rng)
        elif ptype == 2:

        elif ptype == 3:

        if time.time() > deadline - 10:
            break

            res = minimize(lse_obj, h_p, jac=lse_grad,

                          bounds=bounds,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def paired_coord_descent(h, n, deadline, n_sweeps=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(7777)
    deltas = [0.005, -0.005, 0.01, -0.01, 0.002, -0.002, 0.02, -0.02]
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        improved = False
        for i_idx in range(0, n - 1, 2):
            if time.time() > deadline:
                return best_h, best_c
            i = perm[i_idx]
            j = perm[i_idx + 1]
            if i == j:
                continue
            for d in deltas:
                hi = best_h[i] + d
                hj = best_h[j] - d
                if hi < -1e-10 or hi > 1.0 + 1e-10 or hj < -1e-10 or hj > 1.0 + 1e-10:
                    continue
                best_h[i] = hi
                best_h[j] = hj
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = hi - d
                    best_h[j] = hj + d
        if not improved and sweep > 50:
            break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
