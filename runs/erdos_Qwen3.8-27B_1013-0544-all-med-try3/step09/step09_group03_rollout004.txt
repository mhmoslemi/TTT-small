        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def make_sturmian(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)

def make_blocks(n, block_len):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // block_len) % 2 == 0 else 0.0
    return project_feasible(h, n)

def interpolate_to(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
