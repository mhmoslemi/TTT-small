def compute_c5(h, n):

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    wc = w * dx
    for a in range(2 * n - 1):
        if wc[a] < 1e-15:
            continue
        k = a - (n - 1)
        wdx = wc[a]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wdx * (-x[k:n])
            grad[k:n] += wdx * (1.0 - x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wdx * (1.0 - x[0:nv])
            grad[0:nv] += wdx * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    val = vals.mean()
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, mu):
    dx = 2.0 / n
    grad = np.zeros(n)
    for a in active:
        k = a - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (-x[k:n])
            grad[k:n] += dx * (1.0 - x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - x[0:nv])
            grad[0:nv] += dx * (-x[kk:n])
    grad /= len(active)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d

                        h[i] -= d

    return best_h, best_c
