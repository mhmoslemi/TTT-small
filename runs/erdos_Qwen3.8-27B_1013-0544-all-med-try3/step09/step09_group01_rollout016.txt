            constraints_const = [constraints_const[i] for i in keep]

    return grad

def topk_obj(x, n, k_active, alpha, mu):
    """Smoothed top-k mean objective."""
    c = compute_c5_vec(x, n)
    mx = c.max()
    sh = alpha * (c - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    ws = w.sum()
    # Top-k approximation: use softmax weights to compute weighted mean of top values
    val = mx + np.log(ws + 1e-300) / alpha
    # Blend with top-k average
    sorted_c = np.sort(c)[::-1]
    topk_mean = sorted_c[:k_active].mean()
    val = 0.7 * val + 0.3 * topk_mean
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_quasirandom_start(n, seed):
    """Create a quasi-random binary-like start using golden ratio."""
    rng = np.random.RandomState(seed)
    golden = (1 + np.sqrt(5)) / 2.0
    h = np.zeros(n)
    positions = set()
    count = 0
    for i in range(n * 4):
        idx = int((i * golden) % n)
        if idx not in positions:
            positions.add(idx)
            if len(positions) > n * 0.6:
                break
    # Fill to get exactly n/2 ones
    h[list(positions)][:n//2] = 1.0
    # Add some randomness
    for i in range(len(positions), n):
        if h[i] == 0:
            h[i] = rng.uniform(0, 0.1)
    # Rescale to get sum = n/2
    h = project_feasible(h, n)
    return h

def make_periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    n_ones_target = n // 2
    n_ones = h.sum()
    if n_ones < n_ones_target:
        extra_needed = int(n_ones_target - n_ones)
        zeros_idx = np.where(h == 0)[0]
        rng = np.random.RandomState(123)
        rng.shuffle(zeros_idx)
        for i in range(min(extra_needed, len(zeros_idx))):
            h[zeros_idx[i]] = 1.0
    elif n_ones > n_ones_target:
        extra_remove = int(n_ones - n_ones_target)
        ones_idx = np.where(h == 1)[0]
        rng = np.random.RandomState(456)
        rng.shuffle(ones_idx)
        for i in range(min(extra_remove, len(ones_idx))):
            h[ones_idx[i]] = 0.0
    return project_feasible(h, n)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiter_per=None, bounds_cache=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-12:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def subgrad_optimize(h, n, deadline, n_iter=8000, eta0=0.003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta0
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        
        h = h_new
    
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=50):
    """Aggressive pair-move local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):  # sample every 3rd for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                # Try a few random j's in range
                candidates = list(range(lo, hi, max(1, (hi-lo)//20)))
                if i not in candidates:
                    candidates.append(i)
                for j in candidates:
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c

def random_local_search(h, n, deadline, n_trials=20000):
    """Random perturbation local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    dx = 2.0 / n
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        # Random pair move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.001, 0.0005, 0.0002, 0.0001, 0.00005]) * rng.choice([1, -1])
        
        if 0 <= best_h[i] + d <= 1 and 0 <= best_h[j] - d <= 1:
            best_h[i] += d
            best_h[j] -= d
            c = compute_c5(best_h, n)
            if c < best_c - 1e-17:
                best_c = c
            else:
                best_h[i] -= d
                best_h[j] += d
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Optimize at original resolution
    p1_end = min(t0 + 180, deadline - 400)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.01, maxiter_per=[1500, 2000, 2000, 2500, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set at original resolution
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 100, deadline - 350)
        h2, c2 = run_active_set(best_h, best_n, p2_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n_hi = 1000
    if time.time() < deadline - 300:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        p3_end = min(time.time() + 250, deadline - 250)
        h3, c3 = run_lbfgs(h_hi, n_hi, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.005, maxiter_per=[1000, 1500, 2000, 2500, 2500, 3000, 3000, 3000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_hi
    
    # Phase 4: Active set at high resolution
    if time.time() < deadline - 200 and best_n >= 1000:
        p4_end = min(time.time() + 120, deadline - 180)
        h4, c4 = run_active_set(best_h, best_n, p4_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try different initialization at n=1000
    if time.time() < deadline - 200:
        n_alt = 1000
        # Quasi-random start
        h_qr = make_quasirandom_start(n_alt, seed + 1)
        # Also try periodic pattern (period 7, 3 ones)
        h_per = make_periodic_start(n_alt, 7, 3)
        # Choose better of the two
        c_qr = compute_c5(h_qr, n_alt)
        c_per = compute_c5(h_per, n_alt)
        h_alt = h_qr if c_qr < c_per else h_per
        c_alt = min(c_qr, c_per)
        
        p5_end = min(time.time() + 150, deadline - 150)
        h5, c5 = run_lbfgs(h_alt, n_alt, p5_end,
            alphas=[10, 50, 200, 1000, 5000, 50000, 500000],
            mu=0.01, maxiter_per=[1500, 2000, 2000, 2500, 2500, 3000, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_alt
    
    # Phase 6: Subgradient refinement
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 40, deadline - 50)
        h6, c6 = subgrad_optimize(best_h, best_n, p6_end, n_iter=6000, eta0=0.002)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Fine local search
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 25, deadline - 25)
        h7, c7 = fine_local_search(best_h, best_n, p7_end, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Random local search
    if time.time() < deadline - 5:
        p8_end = deadline - 3
        h8, c8 = random_local_search(best_h, best_n, p8_end, n_trials=30000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final: one more active-set pass if time
    if time.time() < deadline - 2:
        p9_end = deadline - 1
        h9, c9 = run_active_set(best_h, best_n, p9_end, top_k=4,
            alphas=[1000000, 5000000, 20000000])
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
