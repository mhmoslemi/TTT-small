For fixed h[other], the overlap at shift k involving h[i] and/or h[j] is:
- If k = j-i: O(k) includes the term h[i](1-h[j])dx
- If k = i-j: O(k) includes the term h[j](1-h[i])dx

So the 2D subproblem is: minimize max_k O(k) where only the terms involving i and j change.

This is a convex problem in (h[i], h[j]) (since max of linear functions is convex)!

We can solve it analytically or with a small QP.

**Even better: Use bisection on t**

For a target t, check if there exists h with:
- 0 ≤ h[i] ≤ 1 for all i
- ∑h[i] = n/2
- O(k) ≤ t for all k

This is a linear feasibility problem (LP)! Because O(k) is linear in h.

Wait, is it? O(k) = ∑ h[i](1-h[i+k])dx. This is NOT linear in h because of the h[i]*h[i+k] term!

So it's not an LP. It's a QP (quadratic program) feasibility problem.

Hmm, but for binary h, h[i](1-h[i+k]) is either 0 or 1, and the constraint O(k) ≤ t becomes a constraint on the number of "bad" pairs, which is linear in the binary variables.

So for binary h, the feasibility check IS an LP (or rather, an integer program / combinatorial problem).

**Let me try yet another approach: Iterative reweighting**

Inspired by the multiplicative weights / exponential mechanisms in learning theory:

1. Start with h₀
2. At each iteration, find the shift k* that gives the maximum overlap
3. Update h to reduce the overlap at k* (while trying not to increase others)
4. Use a smooth update rule

Specifically:
- Compute all O(k) values
- Find k* = argmax O(k)
- The gradient of O(k*) w.r.t. h[i] is: (1-h[i+k*])dx - h[i+k*]*(1-h[i])dx ... no, that's not right.

Let me compute the gradient properly.
O(k) = ∑_i h[i]*(1-h[i+k])*dx (for valid i, k≥0)

∂O(k)/∂h[i] = (1-h[i+k])*dx - (if i is used as the "shifted" index) h[i-k]*(1-h[i])*dx... 

Actually, h[i] appears in O(k) in two ways:
1. As h[i] in the term h[i]*(1-h[i+k]): contributes (1-h[i+k])*dx
2. As h[i] in the term h[i-k]*(1-h[i]) (when i-k is valid): contributes -h[i-k]*dx

So ∂O(k)/∂h[i] = (1-h[i+k])*dx - h[i-k]*dx (with out-of-range terms = 0)

= dx * [(1-h[i+k]) - h[i-k]]

(where h[j] = 0 for j outside [0,n-1])

For the max function M(h) = max_k O(k), the subgradient is:
∂M/∂h[i] ∈ {∑_{k∈S} λ_k * dx * [(1-h[i+k]) - h[i-k]] : S ⊆ active, ∑λ_k=1, λ_k≥0}

The steepest descent direction is the minimum norm vector in this set.

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Gradient of smooth max approximation + log-sum-exp"""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    # But this is the gradient of O(k) for a specific k.
    # We need it for each active k.
    pass

def grad_single_overlap(h, n, k_shift):
    """Gradient of O(k) w.r.t. h, where k_shift is the correlation index offset"""
    # k_shift goes from 0 to 2n-2, corresponding to actual shift k = k_shift - (n-1)
    dx = 2.0 / n
    k = k_shift - (n - 1)  # actual shift
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        # O(k) = sum_{i=0}^{nv-1} h[i]*(1-h[i+k])*dx
        # d/dh[i] for i in [0, nv-1]: (1-h[i+k])*dx
        # d/dh[i] for i in [k, n-1]: -h[i-k]*dx (from term where i-k is the "base")
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        # O(-kk) = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk])*dx
        # = sum_{i=0}^{nv-1} h[i+kk]*(1-h[i])*dx
        # d/dh[i] for i in [0, nv-1]: -h[i+kk]*dx (from (1-h[i]) term)
        # d/dh[i] for i in [kk, n-1]: (1-h[i-kk])*dx
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def steepest_descent_max(h, n, active_indices, tol=1e-7):
    """
    Find the steepest descent direction for max_k O(k) given active set.
    Solves: min sum_{i,j} lam_i lam_j <g_i, g_j> s.t. sum lam = 1, lam >= 0
    Returns the direction d = -sum lam_k * g_k
    """
    m = len(active_indices)
    if m == 0:
        m = 1
        active_indices = [np.argmax(compute_c5_vec(h, n))]
    if m == 1:
        return -grad_single_overlap(h, n, active_indices[0])
    
    # Compute Gram matrix
    grads = np.zeros((m, n))
    for idx in range(m):
        grads[idx] = grad_single_overlap(h, n, active_indices[idx])
    
    # Gram matrix G[i,j] = <g_i, g_j>
    G = grads @ grads.T  # (m, m)
    
    # Solve: min 0.5 * lam^T G lam s.t. sum(lam)=1, lam>=0
    # This is a QP. For small m, use a simple projected gradient or cvxpy.
    # For efficiency, use a simple approach:
    # KKT: G @ lam = mu * ones + nu, where nu_i = 0 if lam_i > 0
    # Simple approach: iterate
    
    lam = np.ones(m) / m  # start with uniform
    for it in range(200):
        # Gradient of 0.5 * lam^T G lam is G @ lam
        g = G @ lam
        # Projected gradient step
        step = 2.0 / (np.max(np.abs(G)) + 1e-10)
        lam_new = lam - step * g
        # Project onto simplex {sum=1, lam>=0}
        lam_new = project_simplex(lam_new)
        if np.max(np.abs(lam_new - lam)) < 1e-10:
            lam = lam_new
            break
        lam = lam_new
    
    # Direction
    d = -grads.T @ lam
    return d

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_smooth(h, n, deadline, alphas):

    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    
    y = best_h.copy()
    t = 1.0
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Adaptive step size based on Lipschitz constant estimate
        # The Hessian of O(k) is bounded, so L ≈ 2*max_gradient_component²
        # For safety, use a small step and backtracking
        step = 0.01  # initial step
        
        for it in range(50000):
            if time.time() > deadline:
                break
            
            # Compute gradient at y
            c_y = compute_c5_vec(y, n)
            m_y = c_y.max()
            sh = alpha * (c_y - m_y)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            
            # Gradient of smooth max
            grad = np.zeros(n)
            for idx in range(len(w)):
                if w[idx] < 1e-10:
                    continue
                grad += w[idx] * grad_single_overlap(y, n, idx)
            
            # FISTA update
            x = y - step * grad
            x = project_feasible(x, n)
            
            # Evaluate at x
            c_x = compute_c5_vec(x, n)
            fx = smooth_max_val(c_x, alpha)
            
            # FISTA momentum update

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

def binary_local_search(h, n, deadline, max_rounds=200):
    # Project to binary
    h_bin = np.zeros(n)
    n_ones = n // 2
    if n % 2 != 0:
        n_ones = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        if n_ones < n:
            h_bin[sorted_idx[n_ones]] += remaining
    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = set(np.where(h_bin >= 0.5)[0].tolist())
    zeros = set(np.where(h_bin < 0.5)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Random pairs of (one, zero) to try flipping
        rng = np.random.RandomState(rnd * 1000 + n)
        one_list = list(ones)
        zero_list = list(zeros)
        n_tries = min(2000, len(one_list) * len(zeros) // 10)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = one_list[rng.randint(len(one_list))]
            j = zero_list[rng.randint(len(zero_list))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            h_bin = project_feasible(h_bin, n)
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_bin.copy()
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                h_bin = project_feasible(h_bin, n)
        if not improved:
            # Try structured flips: adjacent or periodic
            for i in range(0, n, 3):
                if time.time() > deadline:
                    break
                for offset in [1, 2, 5, 10, n // 4, n // 2]:
                    j = (i + offset) % n
                    if i in ones and j in zeros:
                        h_bin[i] = 0.0
                        h_bin[j] = 1.0
                        h_bin = project_feasible(h_bin, n)
                        c = compute_c5(h_bin, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h_bin.copy()
                            ones.discard(i)
                            ones.add(j)
                            zeros.discard(j)
                            zeros.add(i)
                            improved = True
                        else:
                            h_bin[i] = 1.0
                            h_bin[j] = 0.0
                            h_bin = project_feasible(h_bin, n)
            if not improved:
                break
    return best_h, best_c

def fista_smooth(h, n, deadline, alphas):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        y = best_h.copy()
        t_mom = 1.0
        step = 0.005
        n_inner = 0
        for it in range(20000):
            if time.time() > deadline:
                break
            c_y = compute_c5_vec(y, n)
            m_y = c_y.max()
            sh = alpha * (c_y - m_y)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            grad = np.zeros(n)
            for idx in range(len(w)):
                if w[idx] < 1e-12:
                    continue
                grad += w[idx] * grad_single_overlap(y, n, idx)
            x = y - step * grad
            x = project_feasible(x, n)
            c_x = compute_c5_vec(x, n)
            t_new = (1 + np.sqrt(1 + 4 * t_mom * t_mom)) / 2
            y = x + ((t_mom - 1) / t_new) * (x - y)
            t_mom = t_new
            c_true = float(c_x.max())
            if c_true < best_c - 1e-15:
                best_c = c_true
                best_h = x.copy()
            n_inner += 1
    return best_h, best_c

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def downsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    eta = 0.008
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        c_max = c.max()
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Average gradient of active
        g_avg = np.zeros(n)
        for idx in active:
            g_avg += grad_single_overlap(best_h, n, idx)
        g_avg /= len(active)
        h_try = project_feasible(best_h - eta * g_avg, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
            eta = min(eta * 1.01, 0.03)
        else:
            eta *= 0.95
            if eta < 1e-7:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 180
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
        alphas=[100, 500, 2000, 10000, 50000],
        mu=0.01, maxit_list=[1500, 2000, 2500, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Exact steepest descent
    if time.time() < deadline - 250:
        p2_end = min(time.time() + 200, deadline - 250)
        h2, c2 = exact_steepest_descent(best_h, best_n, p2_end, tol_frac=1e-5)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Subgradient refinement
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 100, deadline - 200)
        h3, c3 = subgrad_refine(best_h, best_n, p3_end, n_iter=20000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Binary projection + local search
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = binary_local_search(best_h, best_n, p4_end, max_rounds=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n
    
    # Phase 5: Try higher resolution
    n_high = 1000
    if time.time() < deadline - 250:
        h_high = upsample(best_h, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high
        
        p5a_end = min(time.time() + 80, deadline - 180)
        h5a, c5a = lbfgs_phase(best_h, best_n, p5a_end,
            alphas=[50, 200, 1000, 5000],
            mu=0.005, maxit_list=[1000, 1500, 1500, 2000])
        if c5a < best_c:
            best_c = c5a
            best_h = h5a
            best_n = n_high
        
        if time.time() < deadline - 120:
            p5b_end = min(time.time() + 80, deadline - 120)
            h5b, c5b = exact_steepest_descent(best_h, best_n, p5b_end, tol_frac=1e-5)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
                best_n = n_high
    
    # Phase 6: FISTA at best resolution
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 80, deadline - 100)
        h6, c6 = fista_smooth(best_h, best_n, p6_end,
            alphas=[5000, 20000, 100000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final exact steepest descent
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 40, deadline - 50)
        h7, c7 = exact_steepest_descent(best_h, best_n, p7_end, tol_frac=5e-6)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Binary local search at best resolution
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 25, deadline - 25)
        h8, c8 = binary_local_search(best_h, best_n, p8_end, max_rounds=100)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n
    
    # Phase 9: Final subgradient polish
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, n_iter=5000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
