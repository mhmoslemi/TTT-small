import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e10):
    target_sum = n / 2.0

    def obj_and_grad(x):

        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        obj_val = smooth_max_value(c, alpha)

        obj_val += penalty_coeff * diff * diff
        grad = grad_smooth_max(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff

        return project_feasible(h.copy(), n)

def projected_gradient_descent(h, n, deadline, rng, alpha=50000000, init_step=0.02):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = init_step
    stagnation = 0
    n_inner = 80

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        cur_c = float(c_all.max())
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()

        grad = grad_smooth_max(h, n, alpha, n_active=n_inner)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        accepted = False
        for i in range(30):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c - 1e-15:
                h = h_new
                accepted = True
                stagnation = 0
                step = min(step * 1.1, 0.05)
                break
            step *= 0.7
        if not accepted:
            stagnation += 1
            # Random kick when stuck
            if stagnation > 50:
                noise = rng.normal(0, 0.003, n)
                h = project_feasible(h + noise, n)
                step = 0.01
                stagnation = 0

    return best_h, best_c

def binary_local_search(h, n, deadline, rng):
    """Local search on approximately-binary solutions."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Identify "active" positions (where h is between 0.1 and 0.9)
    for round_num in range(2000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[-8:]
        improved = False

        for _ in range(500):
            if time.time() > deadline:
                break
            k_star = top_lags[rng.randint(len(top_lags))]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            diff = h[i] - h[j]
            if abs(diff) < 0.01:
                continue
            # Move towards making h[i] and h[j] more different (reduce overlap)
            d = 0.15 * diff
            d = max(-0.3, min(0.3, d))
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-15:
                best_c = c_test
                best_h = h.copy()
                improved = True
                c_all = compute_all_lags(h, n)
                top_lags = np.argsort(c_all)[-8:]
            else:
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
        if not improved:
            # Try small uniform perturbations
            h = project_feasible(h + rng.normal(0, 0.005, n), n)
            c_test = compute_c5(h, n)
            if c_test < best_c:
                best_c = c_test
                best_h = h.copy()
            elif round_num % 50 == 0:
                h = np.clip(h + rng.normal(0, 0.02, n), 0, 1)
                h = project_feasible(h, n)

    return best_h, best_c

def multi_pair_local_search(h, n, deadline, rng):
    """Try pairs of moves simultaneously."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    K = 6

    for it in range(5000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        improved = False
        for trial in range(200):
            if time.time() > deadline:
                break
            k1 = top_idx[rng.randint(K)]
            k2 = top_idx[rng.randint(K)]
            if k1 == k2:
                continue
            lag1 = k1 - (n - 1)
            lag2 = k2 - (n - 1)
            if lag1 == 0 or lag2 == 0:
                continue
            m1 = abs(lag1)
            m2 = abs(lag2)
            if m1 >= n or m2 >= n:
                continue
            i1 = rng.randint(0, n - m1)
            j1 = i1 + m1
            i2 = rng.randint(0, n - m2)
            j2 = i2 + m2
            if i1 == i2 or i1 == j2 or j1 == i2 or j1 == j2:
                continue
            d1 = 0.03 * (h[i1] - h[j1])
            d2 = 0.03 * (h[i2] - h[j2])
            if abs(d1) < 1e-8 or abs(d2) < 1e-8:
                continue
            h[i1] -= d1
            h[j1] += d1
            h[i2] -= d2
            h[j2] += d2
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-15:
                best_c = c_test
                best_h = h.copy()
                improved = True
                c_all = compute_all_lags(h, n)
                top_idx = np.argsort(c_all)[-K:]
            else:
                h[i1] += d1
                h[j1] -= d1
                h[i2] += d2
                h[j2] -= d2
                h = project_feasible(h, n)
        if not improved:
            h = project_feasible(h + rng.normal(0, 0.008, n), n)

    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def perturb_h(h, n, strategy, rng):
    """Apply various perturbation strategies."""
    h_p = h.copy()
    if strategy == 'noise_small':
        h_p += rng.normal(0, 0.01, n)
    elif strategy == 'noise_medium':
        h_p += rng.normal(0, 0.03, n)
    elif strategy == 'noise_large':
        h_p += rng.normal(0, 0.08, n)
    elif strategy == 'binary':
        h_p = (h_p > 0.5).astype(float)
        num_ones = int(h_p.sum())
        target = n // 2
        if num_ones > target:
            one_idx = np.where(h_p == 1)[0]
            order = np.argsort(h[one_idx])
            for idx in order[:num_ones - target]:
                h_p[one_idx[idx]] = 0
        elif num_ones < target:
            zero_idx = np.where(h_p == 0)[0]
            order = np.argsort(-h[zero_idx])
            for idx in order[:target - num_ones]:
                h_p[zero_idx[idx]] = 1
        h_p += rng.normal(0, 0.005, n)
    elif strategy == 'anti_binary':
        h_p = 1.0 - h_p
        h_p = (h_p > 0.5).astype(float)
        num_ones = int(h_p.sum())
        target = n // 2
        if num_ones > target:
            one_idx = np.where(h_p == 1)[0]
            order = np.argsort(h_p[one_idx])
            for idx in order[:num_ones - target]:
                h_p[one_idx[idx]] = 0
        elif num_ones < target:
            zero_idx = np.where(h_p == 0)[0]
            order = np.argsort(-h_p[zero_idx])
            for idx in order[:target - num_ones]:
                h_p[zero_idx[idx]] = 1
        h_p += rng.normal(0, 0.005, n)
    elif strategy == 'shuffle_blocks':
        block_size = n // 10
        h_p = h.copy()
        for b in range(9):
            if b + 1 < len(h_p) // block_size:
                s1 = b * block_size
                e1 = (b + 1) * block_size
                s2 = (b + 1) * block_size
                e2 = (b + 2) * block_size
                if rng.random() < 0.3:
                    h_p[s1:e1], h_p[s2:e2] = h_p[s2:e2].copy(), h_p[s1:e1].copy()
    else:
        h_p += rng.normal(0, 0.01, n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n = n_init

    # Phase 1: Quick multi-alpha L-BFGS-B
    alphas = [200, 2000, 20000, 200000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 500:
            break
        time_left = deadline - 500 - time.time()
        if time_left < 8:
            break
        maxit = min(3000, int(time_left * 50))
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(time_left, 25),
                                maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Projected gradient descent (main workhorse)
    if time.time() < deadline - 350:
        pg_deadline = min(deadline - 200, time.time() + 250)
        h_opt, c_opt = projected_gradient_descent(
            best_h, n, pg_deadline, rng, alpha=30000000, init_step=0.015)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multiple perturbation restarts
    strategies = ['noise_small', 'binary', 'noise_medium', 'anti_binary',
                  'noise_large', 'shuffle_blocks', 'binary', 'noise_small']
    for strat in strategies:
        if time.time() > deadline - 250:
            break
        time_left = deadline - 250 - time.time()
        if time_left < 20:
            break
        h_pert = perturb_h(best_h, n, strat, rng)
        # Quick L-BFGS-B on perturbed
        h_opt = lbfgsb_optimize(h_pert, n, 100000, time.time() + min(time_left * 0.3, 15),
                                maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        # Follow with projected gradient
        if time.time() < deadline - 220:
            pg_tl = min(deadline - 200, time.time() + min(time_left * 0.4, 20))
            h_opt2, c_opt2 = projected_gradient_descent(
                h_opt, n, pg_tl, rng, alpha=50000000, init_step=0.01)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2

    # Phase 4: Binary local search
    if time.time() < deadline - 200:
        bls_deadline = min(deadline - 120, time.time() + 120)
        h_opt, c_opt = binary_local_search(best_h, n, bls_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        # Polish after binary search
        if time.time() < deadline - 120:
            h_pol = lbfgsb_optimize(best_h, n, 50000000, time.time() + 20, maxiter=2000)
            c_pol = compute_c5(h_pol, n)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol

    # Phase 5: Multi-pair local search
    if time.time() < deadline - 120:
        mps_deadline = min(deadline - 60, time.time() + 80)
        h_opt, c_opt = multi_pair_local_search(best_h, n, mps_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_new in [800, 1200]:
        if time.time() > deadline - 100:
            break
        time_left = deadline - 80 - time.time()
        if time_left < 30:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_new)
        # Optimize at new resolution
        h_opt = lbfgsb_optimize(h_interp, n_new, 50000, time.time() + min(20, time_left * 0.3),
                                maxiter=1500)
        h_opt = lbfgsb_optimize(h_opt, n_new, 5000000, time.time() + min(15, time_left * 0.25),
                                maxiter=2000)
        if time.time() < deadline - 50:
            h_opt, c_opt = projected_gradient_descent(
                h_opt, n_new, min(deadline - 40, time.time() + 15), rng,
                alpha=30000000, init_step=0.01)
            c_opt = compute_c5(h_opt, n_new)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final L-BFGS-B polish with very high alpha
    if time.time() < deadline - 30:
        n_final = len(best_h)
        for alpha in [100000000, 500000000]:
            if time.time() > deadline - 20:
                break
            h_opt = lbfgsb_optimize(best_h, n_final, alpha, time.time() + 20, maxiter=3000)
            c_opt = compute_c5(h_opt, n_final)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 8: Final projected gradient
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt, c_opt = projected_gradient_descent(
            best_h, n_final, deadline - 5, rng, alpha=100000000, init_step=0.008)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation and return
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
