import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def subgradient_h(h, n, active_set):
    """Compute subgradient: average gradient over active set."""
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in active_set:
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx_val
            grad[k:n] += (-h[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx_val
            grad[0:nv] += (-h[kk:n]) * dx_val
    grad /= len(active_set)
    return grad

def lse_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_grad(c, alpha, x, n)

def lse_obj_constrained(x, n, alpha, mu):
    val = lse_obj(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def lse_grad_constrained(x, n, alpha, mu):
    g = lse_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    return g + 2.0 * mu * s

def nesterov_subgradient(h, n, deadline, T=None, top_k=8):
    """Nesterov subgradient method with averaging for min max."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    if T is None:
        T = int((deadline - time.time()) * 20)
        T = max(T, 500)
    
    h_avg = best_h.copy()
    count = 0
    gamma = (best_c - 0.3) / (1.0 / (2.0 * n * n))  # rough initial step
    
    for t in range(T):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_avg, n)  # evaluate at average
        max_val = c_vec.max()
        
        # Update active set based on current h_avg
        tol = max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        
        # Subgradient at h_avg
        g = subgradient_h(h_avg, n, active)
        g_norm_sq = np.dot(g, g)
        
        # Polyak step
        if g_norm_sq > 1e-20:
            eta = max(1e-8, (max_val - 0.35) / g_norm_sq)
        else:
            break
        eta = min(eta, 5.0 / (t + 1))  # cap
        
        # Update
        h_new = h_avg - eta * g
        h_new = project_feasible(h_new, n)
        
        # Nesterov average
        count += 1
        h_avg = h_avg + (1.0 / count) * (h_new - h_avg)
        
        # Track best
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Also check the average
    h_avg_final = project_feasible(h_avg, n)
    c_avg = compute_c5(h_avg_final, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = h_avg_final
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def active_grad(c, x, n, active):
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in active:
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - x[k:n]) * dx_val
            grad[k:n] += (-x[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - x[0:nv]) * dx_val
            grad[0:nv] += (-x[kk:n]) * dx_val
    grad /= len(active)
    return grad

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, alpha, x, n)
    s = x.sum() - n / 2.0
    return g + 2.0 * mu * s

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    mx = c.max()
    vals = c[active]
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n / 2.0
    return val + mu * s * s

def active_grad_fn(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    mx = c.max()
    vals = c[active]
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, active):
        k = mm - (n - 1)
        wm = wi * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxits=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 15:
            break
        if maxits is not None and i < len(maxits):
            maxit = maxits[i]
        else:
            maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def nesterov_avg(h, n, deadline, top_k=6, target=0.35):
    """Subgradient method with Nesterov averaging."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    h_avg = best_h.copy()
    count = 0
    iters = 0
    while time.time() < deadline - 5:
        c_vec = compute_c5_vec(h_avg, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        g = active_grad(c_vec, h_avg, n, active)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-24:
            break
        eta = (max_val - target) / g_norm_sq
        eta = min(eta, 3.0 / (np.sqrt(count + 1) * 0.01))
        eta = max(eta, 1e-10)
        h_new = project_feasible(h_avg - eta * g, n)
        count += 1
        h_avg = h_avg + (1.0 / count) * (h_new - h_avg)
        iters += 1
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
    h_avg = project_feasible(h_avg, n)
    c_avg = compute_c5(h_avg, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = h_avg
    return best_h, best_c

def pairwise_swap_search(h, n, deadline, max_sweeps=50):
    """Greedy pairwise swap local search for near-binary solutions."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    for sweep in range(max_sweeps):
        if time.time() > deadline - 5:
            break
        improved_this_sweep = False
        rng = np.random.RandomState(sweep * 1000 + 7)
        ones_list = ones.tolist()
        zeros_list = zeros.tolist()
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        for i in ones_list:
            if time.time() > deadline - 5:
                break
            for j in zeros_list:
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved_this_sweep = True
                    ones = np.where(best_h > 0.5)[0]
                    zeros = np.where(best_h < 0.5)[0]
                    ones_list = ones.tolist()
                    zeros_list = zeros.tolist()
                    rng.shuffle(ones_list)
                    rng.shuffle(zeros_list)
                    if time.time() > deadline - 5:
                        break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
        if not improved_this_sweep:
            # Try fractional improvements
            improved_this_sweep = False
            for i in range(n):
                if time.time() > deadline - 5:

                if best_h[i] > 0.01 and best_h[i] < 0.99:

                            if abs(diff) > 1e-14:

                                if others:

                                else:

def fine_coord_descent(h, n, deadline, n_rounds=10):
    """Coordinate descent with line search on the max constraint."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d_idx, d in enumerate(deltas):
        if time.time() > deadline - 3:
            break
        round_improved = True
        while round_improved and time.time() < deadline - 3:
            round_improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            # Compute subgradient
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for mm in active:
                k = mm - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            # Best single-coordinate move
            for j in range(n):
                if time.time() > deadline - 3:
                    break
                if grad[j] > 1e-8 and best_h[j] > d:
                    best_h[j] -= d
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        round_improved = True
                    else:
                        best_h[j] += d
                        best_h = project_feasible(best_h, n)
                elif grad[j] < -1e-8 and best_h[j] < 1.0 - d:
                    best_h[j] += d
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        round_improved = True
                    else:
                        best_h[j] -= d
                        best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    h = project_feasible(h, n)
    return h

def interpolate_h(h_orig, n_orig, n_new):

    x_orig = np.linspace(0, 2, n_orig, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_orig, h_orig, period=2.0)
    return project_feasible(h_interp, n_new)

    if time.time() < deadline - 300:

        h2, c2 = phase_active_set_opt(best_h, best_n, p2_end, top_k=4, rounds=50)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Nesterov subgradient averaging (200s)
    if time.time() < deadline - 150:
        p3_end = min(time.time() + 200, deadline - 120)
        h3, c3 = nesterov_avg(best_h, best_n, p3_end, top_k=5, target=0.35)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Another round of active set (150s)
    if time.time() < deadline - 100:
        p4_end = min(time.time() + 150, deadline - 80)
        h4, c4 = phase_active_set_opt(best_h, best_n, p4_end, top_k=3, rounds=40)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Pairwise swap search (150s) - exploit near-binary structure
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 150, deadline - 50)
        # First, push towards binary
        h_bin = best_h.copy()
        h_bin[h_bin > 0.5] = 1.0
        h_bin[h_bin <= 0.5] = 0.0
        h_bin = project_feasible(h_bin, best_n)
        c_bin = compute_c5(h_bin, best_n)
        # Use the better of current and binary as start for swap search
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        h5, c5 = pairwise_swap_search(best_h, best_n, p5_end, max_sweeps=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine coordinate descent (80s)
    if time.time() < deadline - 30:
        p6_end = min(time.time() + 80, deadline - 20)
        h6, c6 = fine_coord_descent(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final active set polish (40s)
    if time.time() < deadline - 10:
        p7_end = min(time.time() + 40, deadline - 5)
        h7, c7 = phase_active_set_opt(best_h, best_n, p7_end, top_k=3, rounds=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very final fine step (10s)
    if time.time() < deadline - 2:
        p8_end = deadline - 1
        h8, c8 = fine_coord_descent(best_h, best_n, p8_end)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
