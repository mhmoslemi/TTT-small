        t6 = min(time.time() + 80, deadline - 180)
        if t6 > time.time() + 10:
            h6, c6 = sqp_optimize(best_h, n_high, t6, max_iter=200, rho_init=0.05)
            if c6 < best_c:
                best_c, best_h = c6, h6

        # Momentum PG at high res
        t7 = min(time.time() + 60, deadline - 100)
        if t7 > time.time() + 10:
            h7, c7 = momentum_pg(best_h, n_high, t7, alpha=5000, max_iter=1500)
            if c7 < best_c:
                best_c, best_h = c7, h7

        # Local search at high res
        t8 = min(time.time() + 50, deadline - 40)
        if t8 > time.time() + 5:
            h8, c8 = local_search(best_h, n_high, t8, radius=25)
            if c8 < best_c:
                best_c, best_h = c8, h8

    # Phase 6: Final SQP polish
    n_final = len(best_h)
    if time.time() < deadline - 15:
        h_final, c_final = sqp_optimize(best_h, n_final, deadline - 5,
                                         max_iter=150, rho_init=0.02)
        if c_final < best_c:
            best_c, best_h = c_final, h_final

    # Final feasibility
    n_fin = len(best_h)
    best_h = project_feasible(best_h, n_fin)
    best_c = compute_c5(best_h, n_fin)

    return best_h, best_c, n_fin
