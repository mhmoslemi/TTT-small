            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

**Key structural insight**: The overlap at shift k is:
$O_k = \sum_i h[i](1-h[i+k]) dx$ (for k ≥ 0, i ranges appropriately)
$= \sum_i h[i] dx - \sum_i h[i]h[i+k] dx$ (over valid range)

The first term depends on k (it's the integral of h over the valid range), and the second is the auto-correlation.

At the optimum, we want all $O_k$ to be as small as possible (at the maximum). 

**Alternative formulation**: Let $A_k = \sum_i h[i]h[i+k] dx$ (auto-correlation at lag k). Then:
$O_k = S_k - A_k$ where $S_k = \sum_i h[i] dx$ over the valid range.

For a 0/1 function with exactly n/2 ones:
- At k=0: $O_0 = \sum h[i](1-h[i]) dx = 0$
- At k>0: $O_k = $ (# of i where h[i]=1 and h[i+k]=0) * dx = (count of 1s in valid range - count of consecutive 11 at lag k) * dx

For a "well-spread" 0/1 pattern, the consecutive 11 count at each lag k is roughly (n/2)^2 / n = n/4 (by the birthday problem / pigeonhole principle), so $O_k ≈ (n/2 - n/4) dx = n/4 * 2/n = 0.5$.

But we can do better than 0.5 by arranging the 1s to minimize the maximum "excess" at any single shift.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(2000, 500.0), (8000, 2000.0), (20000, 5000.0), (50000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def shift_grad_single(h, n, shift_idx):
    dx_val = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] = (1.0 - h[k:]) * dx_val
            grad[k:] += (-h[:nv]) * dx_val
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:] = (1.0 - h[:nv]) * dx_val
            grad[:nv] += (-h[kk:]) * dx_val
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, max_active=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:max_active]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            no_improve += 1
            if no_improve > 30:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.25, 0.08)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-10:
                break
            if no_improve > 40:
                break
    return best_h, best_c

def balance_step(h, n, active_weights, step_size):
    """Take a step in the direction that decreases the weighted sum of active overlaps."""
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for idx, wt in active_weights:
        g = shift_grad_single(h, n, idx)
        grad += wt * g
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return h.copy()
    direction = -grad / grad_norm
    h_new = h + step_size * direction
    h_new = project_feasible(h_new, n)
    return h_new

def run_local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.005 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad_mag = np.zeros(n)
            for idx in active:
                grad_mag += np.abs(shift_grad_single(h, n, idx))
            grad_mag /= len(active)
            top = np.argsort(-grad_mag)[:min(radius * 2, n)]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
