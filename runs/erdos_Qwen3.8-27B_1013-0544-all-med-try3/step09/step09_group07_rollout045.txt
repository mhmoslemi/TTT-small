            # For this lag, try to reduce overlap by moving mass
            # h[i]*(1-h[i+m]) should be small
            # If h[i] is large and h[i+m] is small, moving mass from i to i+m helps
            # If h[i] is small and h[i+m] is large, moving mass from i+m to i helps
            range_size = n - m
            # Sample pairs
            n_try = min(range_size, 5000)
            indices = rng.choice(range_size, size=min(n_try, range_size), replace=False)
            for i in indices:
                if time.time() > deadline:
                    break
                j = i + m
                if k > 0:
                    # Moving mass from i to j: h[i]=h[i], h[j]=h[j+m]... 
                    # Actually for lag m: pairs are (i, i+m)
                    if h[i] > 0.01 and h[j] < 0.99:
                        # Try moving small amount from i to j
                        d = min(0.01, (h[i] - 0.001) * 0.3, (0.999 - h[j]) * 0.3)
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0, 1)
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            any_improve = True
                            max_inner += 1
                            if max_inner > 2000:
                                break
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > 0.01 and h[i] < 0.99:
                        d = min(0.01, (h[j] - 0.001) * 0.3, (0.999 - h[i]) * 0.3)
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0, 1)
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            any_improve = True
                            max_inner += 1
                            if max_inner > 2000:
                                break
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
                else:
                    # k < 0, lag is -m, pairs are (i+m, i)
                    i2 = i + m
                    j2 = i
                    if h[i2] > 0.01 and h[j2] < 0.99:
                        d = min(0.01, (h[i2] - 0.001) * 0.3, (0.999 - h[j2]) * 0.3)
                        h[i2] -= d
                        h[j2] += d
                        h = np.clip(h, 0, 1)
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            any_improve = True
                            max_inner += 1
                            if max_inner > 2000:
                                break
                        else:
                            h[i2] += d
                            h[j2] -= d
                            h = project_feasible(h, n)
                    elif h[j2] > 0.01 and h[i2] < 0.99:
                        d = min(0.01, (h[j2] - 0.001) * 0.3, (0.999 - h[i2]) * 0.3)
                        h[i2] += d
                        h[j2] -= d
                        h = np.clip(h, 0, 1)
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            any_improve = True
                            max_inner += 1
                            if max_inner > 2000:
                                break
                        else:
                            h[i2] -= d
                            h[j2] += d
                            h = project_feasible(h, n)

        if not any_improve:
            break

    return best_h, best_c

def fine_grained_search(h, n, deadline, rng):
    """Very fine local search with adaptive step sizes and all lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for s_idx, base_s in enumerate(step_sizes):
        if time.time() > deadline:
            break
        improved = False
        iters = 0
        max_iters = 30000 - s_idx * 4000
        
        while iters < max_iters and time.time() < deadline:
            iters += 1
            c_all = compute_all_lags(h, n)
            # Pick a random top lag
            top_k = np.argsort(c_all)[-20:]
            k_star = top_k[rng.randint(len(top_k))]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            
            # Determine direction: reduce overlap h[i]*(1-h[j]) or h[j]*(1-h[i])
            overlap_ij = h[i] * (1 - h[j])
            overlap_ji = h[j] * (1 - h[i])
            
            if overlap_ij > overlap_ji + 1e-8:
                # Move mass from i to j
                d = min(base_s * rng.uniform(0.2, 1.0), h[i] * 0.2, (1.0 - h[j]) * 0.2)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
            elif overlap_ji > overlap_ij + 1e-8:
                # Move mass from j to i
                d = min(base_s * rng.uniform(0.2, 1.0), h[j] * 0.2, (1.0 - h[i]) * 0.2)
                if d < 1e-10:
                    continue
                h[i] += d
                h[j] -= d
            else:
                # Random small move
                d = base_s * 0.1 * rng.uniform(-1, 1)
                if abs(d) < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
            
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            
            if c_test < best_c - 1e-15:
                best_c = c_test
                best_h = h.copy()
                improved = True
                # If it improved, try more at this step size
            else:
                h = best_h.copy()
    
    return best_h, best_c

def gradient_descent_nesterov(h, n, deadline, alpha=10000, lr=0.01, momentum=0.9):
    """Nesterov accelerated gradient descent on smooth max."""
    h = project_feasible(h.copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    
    t = 0
    stagnation = 0
    while time.time() < deadline and stagnation < 2000:
        t += 1
        # Nesterov lookahead
        h_y = h + momentum * v
        h_y = np.clip(h_y, 0, 1)
        grad, _ = compute_gradient(h_y, n, alpha)
        # Projected gradient step with penalty
        diff = h_y.sum() - target_sum
        grad += 0.1 * diff * np.ones(n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        v = h_new - h
        h = h_new
        
        if t % 5 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-14:
                best_c = c_now
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 500:
                    # Reduce learning rate
                    lr *= 0.5
                    stagnation = 0
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h, n_parent)

    # Phase 1: Optimize at parent resolution with penalty minimax
    if time.time() < deadline - 600:
        dl = min(deadline - 550, time.time() + 120)
        h_opt, c_opt = penalty_minimax(best_h, n_parent, dl, alpha_start=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: L-BFGS-B polish at parent resolution
    if time.time() < deadline - 500:
        for alpha in [100000, 1000000, 10000000]:
            if time.time() > deadline - 450:
                break
            h_opt = lbfgsb_optimize(best_h, n_parent, alpha, time.time() + 25, maxiter=2000)
            c_opt = compute_c5(h_opt, n_parent)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: Upscale to higher resolution and optimize
    n_high = 1000
    if time.time() < deadline - 400:
        h_high = expand_to_resolution(best_h, n_parent, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high

        # Optimize at high resolution
        dl_high = min(deadline - 300, time.time() + 150)
        h_opt, c_opt = penalty_minimax(h_high, n_high, dl_high, alpha_start=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        if time.time() < deadline - 200:
            for alpha in [500000, 5000000]:
                if time.time() > deadline - 180:
                    break
                h_opt = lbfgsb_optimize(best_h, n_high, alpha, time.time() + 20, maxiter=1500)
                c_opt = compute_c5(h_opt, n_high)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 4: Nesterov gradient descent on the best solution
    if time.time() < deadline - 150:
        n_curr = len(best_h)
        dl_ng = min(deadline - 120, time.time() + 60)
        h_opt, c_opt = gradient_descent_nesterov(best_h, n_curr, dl_ng, alpha=50000, lr=0.005)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Systematic coordinate descent
    if time.time() < deadline - 100:
        n_curr = len(best_h)
        dl_cd = min(deadline - 60, time.time() + 80)
        h_opt, c_opt = systematic_coord_descent(best_h, n_curr, dl_cd, rng, n_rounds=30)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Fine-grained search
    if time.time() < deadline - 50:
        n_curr = len(best_h)
        dl_fg = min(deadline - 30, time.time() + 50)
        h_opt, c_opt = fine_grained_search(best_h, n_curr, dl_fg, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Try a different starting point at high resolution
    if time.time() < deadline - 80:
        n_try = 1000
        # Create a perturbed version
        h_pert = best_h[:n_parent] + rng.normal(0, 0.05, n_parent)
        h_pert = project_feasible(h_pert, n_parent)
        h_pert_high = expand_to_resolution(h_pert, n_parent, n_try)
        dl_pt = min(deadline - 50, time.time() + 50)
        h_opt, c_opt = penalty_minimax(h_pert_high, n_try, dl_pt, alpha_start=100000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, deadline - 5, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation and cleanup
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
