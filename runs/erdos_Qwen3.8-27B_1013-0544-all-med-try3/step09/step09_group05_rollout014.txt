        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def equalize_obj(x, n, active_idx, alpha, mu):
    """Minimize spread of active shifts plus their mean (to push them down)."""
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mean_v = vals.mean()
    spread = vals.std()
    val = mean_v + 10.0 * spread
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def equalize_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mean_v = vals.mean()
    n_act = len(active_idx)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, mm in enumerate(active_idx):
        k = mm - (n - 1)
        coeff = dx / n_act
        if k >= 0:
            nv = n - k
            grad[0:nv] += coeff * (1.0 - x[k:n])
            grad[k:n] += coeff * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += coeff * (1.0 - x[0:nv])
            grad[0:nv] += coeff * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optim(h, n, deadline, configs=None, bounds=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 20.0), (1000, 200.0), (5000, 2000.0),
                   (20000, 20000.0), (50000, 100000.0),
                   (100000, 500000.0), (200000, 2000000.0), (500000, 10000000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxfun': 50000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_equalize(h, n, deadline, top_m=20, n_rounds=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    for rnd in range(n_rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        # Find shifts within a tight band of the max
        threshold = m - max(1e-6, 0.002 * m)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > top_m:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_m]]
        # Optimize to equalize and reduce active shifts
        for mu in [1.0, 10.0, 100.0]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(equalize_obj, best_h, jac=equalize_grad,
                              args=(n, active, 100, mu), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def nesterov_pg(h, n, deadline, alpha=100000, lr0=0.002, momentum=0.85):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    x = h.copy()
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = x + momentum * v
        grad = smooth_max_grad(y, n, alpha, 0.05)
        v = momentum * v + grad
        x_new = y - lr * v
        x_new = np.clip(x_new, 0.0, 1.0)
        x_new = project_feasible(x_new, n)
        v = x_new - y
        x = x_new
        it += 1
        if it % 500 == 0:
            c = compute_c5(x, n)
            if c < best_c - 1e-16:
                best_c = c
                best_h = x.copy()
            lr *= 0.995
    c = compute_c5(x, n)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def exhaustive_local_search(h, n, deadline, window=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            count = 0
            for i in range(n):
                count += 1
                if count % 500 == 0 and time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move from i to j
                    if best_h[i] - step >= 0.0 and best_h[j] + step <= 1.0:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
                    # Move from j to i
                    if best_h[j] - step >= 0.0 and best_h[i] + step <= 1.0:
                        best_h[j] -= step
                        best_h[i] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += step
                            best_h[i] -= step
    return best_h, best_c

def random_restart_optim(h, n, deadline, bounds, n_tries=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for trial in range(n_tries):
        if time.time() > deadline - 10:
            break
        scale = 0.02 * (1.0 - 0.8 * trial / n_tries) + 0.002
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        cfgs = [(10000, 10000.0), (50000, 100000.0), (200000, 1000000.0)]
        for alpha, mu in cfgs:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(smooth_max_obj, h_pert, jac=smooth_max_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 20000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    if alphas is None:
        alphas = [200000, 500000, 1000000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, 1.0), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-19})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample_downsample(h, n, n_coarse, deadline):
    """Optimize at coarser resolution, then upsample and refine."""
    x_fine = np.linspace(0, 2, n)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_c = np.interp(x_coarse, x_fine, h)
    h_c = project_feasible(h_c, n_coarse)
    bounds_c = [(0.0, 1.0)] * n_coarse
    h_c, c_c = lbfgs_optim(h_c, n_coarse, deadline,
        configs=[(500, 50.0), (2000, 500.0), (10000, 5000.0), (50000, 50000.0)],
        bounds=bounds_c)
    h_f = np.interp(x_fine, x_coarse, h_c)
    h_f = project_feasible(h_f, n)
    return h_f

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_orig = 300

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    bounds = [(0.0, 1.0)] * n_orig

    # Phase 1: Initial L-BFGS with moderate alpha
    t1 = min(t0 + 150, deadline - 600)
    best_h, best_c = lbfgs_optim(best_h, n_orig, t1,
        configs=[(100, 10.0), (500, 100.0), (2000, 1000.0),
                 (10000, 10000.0), (50000, 100000.0)],
        bounds=bounds)

    # Phase 2: Active set equalization
    if time.time() < deadline - 400:
        t2 = min(time.time() + 150, deadline - 400)
        best_h, best_c = active_set_equalize(best_h, n_orig, t2, top_m=15, n_rounds=12)

    # Phase 3: High-alpha L-BFGS
    if time.time() < deadline - 300:
        t3 = min(time.time() + 150, deadline - 300)
        best_h, best_c = lbfgs_optim(best_h, n_orig, t3,
            configs=[(50000, 100000.0), (100000, 500000.0),
                     (200000, 2000000.0), (500000, 10000000.0)],
            bounds=bounds)

    # Phase 4: Nesterov momentum
    if time.time() < deadline - 200:
        t4 = min(time.time() + 80, deadline - 200)
        best_h, best_c = nesterov_pg(best_h, n_orig, t4, alpha=100000, lr0=0.001, momentum=0.85)

    # Phase 5: Another active set round
    if time.time() < deadline - 150:
        t5 = min(time.time() + 60, deadline - 150)
        best_h, best_c = active_set_equalize(best_h, n_orig, t5, top_m=10, n_rounds=8)

    # Phase 6: SLSQP polish
    if time.time() < deadline - 100:
        t6 = min(time.time() + 60, deadline - 100)
        best_h, best_c = slqp_polish(best_h, n_orig, t6,
            alphas=[200000, 500000, 2000000])

    # Phase 7: Random restarts
    if time.time() < deadline - 50:
        t7 = min(time.time() + 40, deadline - 50)
        best_h, best_c = random_restart_optim(best_h, n_orig, t7, bounds, n_tries=10)

    # Phase 8: Exhaustive local search (most important for final squeeze)
    if time.time() < deadline - 15:
        t8 = deadline - 10
        best_h, best_c = exhaustive_local_search(best_h, n_orig, t8, window=25)

    # Final validation
    best_h = project_feasible(best_h, n_orig)
    best_c = compute_c5(best_h, n_orig)

    return best_h, best_c, n_orig
