import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_active(c_vec, top_k=5, tol_frac=0.002):
    max_val = c_vec.max()
    threshold = max_val - max(1e-10, tol_frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def smax_obj(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(active)):
        if w[ii] < 1e-14:
            continue
        idx = active[ii]
        k = idx - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smax_obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2

def smax_grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if w[mm] < 1e-12:

        k = mm - (n - 1)

        if k >= 0:

    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 25:
            break

                              args=(n, alpha, active, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': max(100, maxit // 4), 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit, 'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)

                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                elif c_new <= best_c + 1e-15:

            except Exception:

def slqp_opt(h, n, deadline, alpha=500000, maxiter=1000):
    def slqp_obj(x, n_, alpha_):
        c = compute_c5_vec(x, n_)
        m = c.max()
        sh = alpha_ * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha_)
        else:
            return float(m + np.log(np.sum(np.exp(sh)))) / alpha_
    def sq_fun(x, n_):
        return x.sum() - n_ / 2.0
    def sq_jac(x, n_):
        return np.ones(n_)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def subgradient_refine(h, n, deadline, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    for it in range(n_iter):

        c_vec = compute_c5_vec(best_h, n)

        tol = max(1e-9, 0.002 * c_max)

        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n

        for idx in active:

                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(best_h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:

            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:

            if eta < 1e-5:

    return best_h, best_c

def multi_target_obj(x, n, targets, weights, mu):

    val = sum(weights[i] * c[targets[i]] for i in range(len(targets)))

    return float(val)
