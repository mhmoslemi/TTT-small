            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:n] -= wm * h[:nv]
    return grad

def obj_and_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    w = smooth_max_weights(c, alpha)
    grad = fast_gradient(w, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return float(val), grad

def active_obj_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    val = smooth_max_val(vals, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    w = smooth_max_weights(vals, alpha)
    grad = fast_gradient_from_active(w, active, n, x)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return float(val), grad

def fast_gradient_from_active(w, active, n, x):
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for wi, m in zip(w, active):
        if wi < 1e-15:
            continue
        k = abs(m - n1)
        wm = wi * dx
        if k == 0:
            grad[:n] += wm * (1.0 - 2.0 * x)
        else:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:n] -= wm * x[:nv]
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu, maxit_list=None, bounds_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(500, int(remaining * 4))

        def fun(x):
            v, _ = obj_and_grad(x, n, alpha, mu)
            return v

        def jac(x):
            _, g = obj_and_grad(x, n, alpha, mu)
            return g

        def funjac(x):
            return obj_and_grad(x, n, alpha, mu)

        try:
            res = minimize(funjac, best_h, jac=None, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40, 'factor': 1.0})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def active_set_phase(h, n, deadline, top_k=8, alphas=None, mu=0.003):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]

    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            idx = np.argsort(-c_vec[active])[:top_k]
            active = active[idx]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 3))

            def afun(x):
                return active_obj_grad(x, n, active, alpha, mu)[0]

            def ajac(x):
                return active_obj_grad(x, n, active, alpha, mu)[1]

            def afunjac(x):
                return active_obj_grad(x, n, active, alpha, mu)

            try:
                res = minimize(afunjac, best_h, jac=None, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': maxit * 5,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    return best_h, best_c

def exact_line_search(h, n, direction, deadline):
    best_t = 0.0
    best_c = compute_c5(h, n)
    t = 0.01
    for _ in range(30):
        if time.time() > deadline:
            break
        h_try = h + t * direction
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_t = t
            t *= 1.3
        else:
            t *= 0.5
            if t < 1e-8:
                break
    return h + best_t * direction, best_c

def subgrad_with_line_search(h, n, deadline, n_iter=2000, tol_expand=0.002):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01

    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        dx = 2.0 / n
        grad = np.zeros(n)
        n1 = n - 1
        cnt = 0.0
        for idx in active:
            k = abs(idx - n1)
            if k == 0:
                grad[:n] += (1.0 - 2.0 * h)
            else:
                nv = n - k
                grad[:nv] += (1.0 - h[k:])
                grad[k:n] -= h[:nv]
            cnt += 1
        grad *= dx / cnt

        h_try = h - eta * grad
        h_new, c_new = exact_line_search(h, n, -grad, deadline)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.05)
        else:
            eta *= 0.97

        h = h_new

    return best_h, best_c

def local_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_c = c_vec.max()
    active_set = set(np.where(c_vec >= max_c - 0.002 * max_c)[0].tolist())

    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d_idx, d in enumerate(deltas):
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            c_vec = compute_c5_vec(best_h, n)
                            max_c = c_vec.max()
                            active_set = set(np.where(c_vec >= max_c - 0.002 * max_c)[0].tolist())
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            c_vec = compute_c5_vec(best_h, n)
                            max_c = c_vec.max()
                            active_set = set(np.where(c_vec >= max_c - 0.002 * max_c)[0].tolist())
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def multi_restart_optimize(h0, n, deadline, n_restarts=3):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    rng = np.random.RandomState(np.int32(sum(h0 * n) * 1000))

    for r in range(n_restarts):
        remaining = deadline - time.time()
        if remaining < 60:
            break
        h = best_h.copy()
        noise_scale = 0.02 / (r + 1)
        h = h + rng.randn(n) * noise_scale
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def sq_fun(x):
        return x.sum() - target

    def sq_jac(x):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac}]

    def sfun(x):
        c = compute_c5_vec(x, n)
        return smooth_max_val(c, alpha)

    try:
        res = minimize(sfun, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16,
                               'eps': 1e-8})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS at original resolution with progressive alpha
    p1_deadline = t0 + min(180, budget_s * 0.18)
    alphas_p1 = [50, 200, 1000, 5000, 20000, 100000]
    maxit_p1 = [1500, 1500, 2000, 2500, 3000, 3000]
    h1, c1 = lbfgs_phase(best_h, best_n, p1_deadline, alphas_p1, mu=0.01, maxit_list=maxit_p1)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set at original resolution
    if time.time() < deadline - 400:
        p2_deadline = min(time.time() + 100, deadline - 400)
        alphas_p2 = [10000, 50000, 200000, 1000000]
        h2, c2 = active_set_phase(best_h, best_n, p2_deadline, top_k=6, alphas=alphas_p2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=1000 and optimize
    n_up = 1000
    if time.time() < deadline - 350:
        h_up = upsample(best_h, n_up)
        c_up = compute_c5(h_up, n_up)

        alphas_p3 = [50, 200, 1000, 5000, 20000, 100000, 500000]
        maxit_p3 = [1500, 2000, 2500, 3000, 3000, 3000, 3000]
        p3_deadline = min(time.time() + 250, deadline - 200)
        h3, c3 = lbfgs_phase(h_up, n_up, p3_deadline, alphas_p3, mu=0.005, maxit_list=maxit_p3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_up

    # Phase 4: Active set at best resolution
    if time.time() < deadline - 150:
        p4_deadline = min(time.time() + 150, deadline - 100)
        alphas_p4 = [50000, 200000, 1000000, 5000000, 20000000]
        h4, c4 = active_set_phase(best_h, best_n, p4_deadline, top_k=5, alphas=alphas_p4, mu=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Subgradient with line search
    if time.time() < deadline - 80:
        p5_deadline = min(time.time() + 60, deadline - 50)
        h5, c5 = subgrad_with_line_search(best_h, best_n, p5_deadline, n_iter=3000, tol_expand=0.003)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Multi-restart with small perturbations
    if time.time() < deadline - 60:
        h6, c6 = multi_restart_optimize(best_h, best_n, min(time.time() + 30, deadline - 40), n_restarts=2)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish
    if time.time() < deadline - 30:
        for alpha_s in [200000, 1000000, 5000000]:
            if time.time() > deadline - 20:
                break
            p7_deadline = min(time.time() + 20, deadline - 15)
            h7, c7 = slqp_polish(best_h, best_n, p7_deadline, alpha=alpha_s, maxiter=800)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Local pair search (fine)
    if time.time() < deadline - 5:
        p8_deadline = deadline - 2
        h8, c8 = local_pair_search(best_h, best_n, p8_deadline, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final active set micro-optimization
    if time.time() < deadline - 1:
        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 4:
            active = np.argsort(-c_vec[active])[:4]
        else:
            active = np.array(active)

        if len(active) > 0:
            bounds = [(0.0, 1.0)] * best_n
            alpha_fin = 50000000

            def finfun(x):
                return active_obj_grad(x, best_n, active, alpha_fin, 0.001)[0]

            def finjac(x):
                return active_obj_grad(x, best_n, active, alpha_fin, 0.001)[1]

            def finfunjac(x):
                return active_obj_grad(x, best_n, active, alpha_fin, 0.001)

            try:
                res = minimize(finfunjac, best_h, jac=None, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 200, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': 1000,
                                       'maxls': 30, 'maxcor': 20})
                h_fin = project_feasible(res.x, best_n)
                c_fin = compute_c5(h_fin, best_n)
                if c_fin < best_c:
                    best_c = c_fin
                    best_h = h_fin
            except Exception:
                pass

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    # Verify constraint
    assert abs(best_h.sum() * (2.0 / final_n) - 1.0) < 1e-8, f"Sum constraint violated: {best_h.sum() * 2.0 / final_n}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bound constraint violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
