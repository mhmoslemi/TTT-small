from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_box_sum(v, n):
    """Project v onto {0 <= h <= 1, sum(h) = n/2} using bisection on lambda."""
    target = n / 2.0
    # As lambda -> -inf, clip(v - lambda, 0, 1) -> 1 for all, sum -> n
    # As lambda -> +inf, clip(v - lambda, 0, 1) -> 0 for all, sum -> 0
    lo = min(v.min() - 1.0, -2.0)
    hi = max(v.max() + 1.0, 2.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Add/subtract from non-bound entries
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    vals = c[active]
    mx = vals.max()

    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:

def prox_grad(h, n, eta, alpha=500000, deadline_check=None):
    """One step of proximal gradient on the smooth max."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - n1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    
    h_new = h - eta * grad
    h_new = project_box_sum(h_new, n)
    return h_new

def coord_descent(h, n, deadline, max_rounds=50):
    """Coordinate descent: for each i, try to adjust h[i] optimally."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try small adjustments to h[i]
            # Keep sum = target by adjusting another element
            orig = best_h[i]
            deltas = [orig * 0.01, orig * 0.001, 1.0 - orig, (1.0 - orig) * 0.01]
            for d in deltas:
                for sign in [1, -1]:
                    new_val = orig + sign * abs(d)
                    if new_val < 0 or new_val > 1:
                        continue
                    # Adjust another element to maintain sum
                    # Find a "free" element to adjust
                    delta = -(new_val - orig)
                    # Try to distribute delta to a random element
                    j = (i + 1) % n  # simple: adjust neighbor
                    new_j = best_h[j] + delta
                    if 0 <= new_j <= 1:
                        best_h[i] = new_val
                        best_h[j] = new_j
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = orig
                            best_h[j] = best_h[j] - delta
        if not improved:
            break
    return best_h, best_c

def pair_refine(h, n, deadline, radius=20):
    """Fine pair search with small steps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):  # stride 3 for speed
                if time.time() > deadline:
                    break
                lo = max(1, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        if sign > 0:
                            if best_h[i] + d > 1.0 or best_h[j] - d < 0.0:
                                continue
                            best_h[i] += d
                            best_h[j] -= d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-17:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] -= d
                                best_h[j] += d
                        else:
                            if best_h[i] - d < 0.0 or best_h[j] + d > 1.0:
                                continue
                            best_h[i] -= d
                            best_h[j] += d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-17:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] += d
                                best_h[j] -= d
    return best_h, best_c

    vm = np.zeros(n)
    eta = eta0
    t_count = 1.0
    mu_nest = 0.9
    while time.time() < deadline:
        y = h + mu_nest * vm
        grad = nesterov_step(y, n, alpha, eta, mu_nest)
        h_new = h - eta * grad
        h_new = project_box_sum(h_new, n)
        vm = h_new - h
        h = h_new
        c_new = compute_c5(h, n)
        t_count += 1.0
        mu_nest = (1.0 + 1.0 / np.cbrt(t_count)) / 3.0
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            eta = min(eta * 1.005, 0.01)
        else:
            eta *= 0.97
    return best_h, best_c

def pair_refine(h, n, deadline, radius=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        if time.time() > deadline - 5:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            stride = max(1, n // 200)
            for i in range(0, n, stride):
                if time.time() > deadline - 3:
                    break
                lo = max(1, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, stride):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def perturb_reopt(best_h, best_n, rng, deadline, scale=0.005, n_trials=2):
    best_c = compute_c5(best_h, best_n)
    for _ in range(n_trials):
        if time.time() > deadline - 30:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = project_box_sum(h_p, best_n)
        c_p = compute_c5(h_p, best_n)
        start = h_p if c_p < best_c else best_h.copy()
        end = min(time.time() + 40, deadline - 20)
        h_opt, c_opt = run_active(start, best_n, end, top_k=6,
            alphas=[500000, 2000000, 10000000, 50000000])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def flip_try(h, n, deadline):
    c_orig = compute_c5(h, n)
    h_flip = 1.0 - h
    h_flip = project_box_sum(h_flip, n)
    c_flip = compute_c5(h_flip, n)
    if c_flip < c_orig:
        return h_flip, c_flip
    return h, c_orig

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_box_sum(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: L-BFGS at original resolution
    p1_end = min(t0 + 200, deadline - 500)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.01, maxiters=[1500, 2000, 2000, 2500, 3000, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active-set focused optimization
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 150, deadline - 300)
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=8,
            alphas=[200000, 1000000, 5000000, 20000000, 100000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Nesterov gradient descent
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 80, deadline - 200)
        h3, c3 = grad_descent_loop(best_h, best_n, p3_end, alpha=5000000, eta0=0.001)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try upsampled resolution
    n_high = 1000
    if time.time() < deadline - 200:
        x_fine = np.linspace(0, 2, n_high)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_box_sum(h_fine, n_high)
        p4_end = min(time.time() + 180, deadline - 100)
        h4, c4 = run_lbfgs(h_fine, n_high, p4_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxiters=[1000, 1500, 1500, 2000, 2000, 2000, 2000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high

        if time.time() < deadline - 80:
            p4b_end = min(time.time() + 60, deadline - 50)
            h4b, c4b = run_active(best_h, best_n, p4b_end, top_k=6,
                alphas=[1000000, 5000000, 20000000])
            if c4b < best_c:
                best_c = c4b
                best_h = h4b

    # Phase 5: Flip and try
    if time.time() < deadline - 100:
        best_h, best_c = flip_try(best_h, best_n, deadline)
        p5_end = min(time.time() + 40, deadline - 60)
        h5, c5 = run_active(best_h, best_n, p5_end, top_k=6,
            alphas=[1000000, 5000000, 20000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Perturbation + re-optimize
    if time.time() < deadline - 60:
        best_h, best_c = perturb_reopt(best_h, best_n, rng, deadline,
            scale=0.003, n_trials=2)

    # Phase 7: Final active-set polish
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 30, deadline - 15)
        h7, c7 = run_active(best_h, best_n, p7_end, top_k=4,
            alphas=[5000000, 20000000, 100000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search
    if time.time() < deadline - 5:
        p8_end = deadline - 2
        h8, c8 = pair_refine(best_h, best_n, p8_end, radius=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final verification
    final_n = len(best_h)
    best_h = project_box_sum(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
