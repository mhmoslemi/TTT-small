For a fixed lag m, the contribution to the objective from that lag is:
c[m] = ∑_{i=0}^{n-m-1} h[i+m] * (1-h[i]) * dx

To minimize max_k c[k], we want to minimize the worst lag. For a specific worst lag m, we want to minimize c[m] subject to all other lags not exceeding c[m].

This is a constrained optimization problem. The gradient of c[m] with respect to h[i] is:
- For i in [m, n-1]: ∂c[m]/∂h[i] = (1-h[i-m]) * dx
- For i in [0, n-m-1]: ∂c[m]/∂h[i] = -h[i+m] * dx

So the steepest descent direction for c[m] is:
For i in [0, n-1]:
  grad[i] = (1-h[i-m])*dx (if i >= m) - h[i+m]*dx (if i < n-m)

This can be negative (meaning we should increase h[i]) or positive (meaning we should decrease h[i]).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    final_diff = target_sum - h.sum()
    if abs(final_diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += final_diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    n_active = np.sum(w > 1e-6)
    for k in range(2*n-1):
        if w[k] < 1e-6:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def grad_descent_adaptive(h, n, deadline, alpha_start=50, alpha_end=10000, lr=0.3, momentum=0.92):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        progress = min(1.0, it / 50000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        g = grad_smooth_max(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 3000:
            lr *= 0.8
            v *= 0.5
            patience = 0
            if lr < 0.01:
                break
    return best_h, best_c

def worst_lag_gradient_step(h, n, deadline, max_iter=50000):
    """Directly minimize the worst lag using its gradient."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            # Focus on a random nonzero lag
            lag = 1
        m = abs(lag)
        if m >= n:
            break
        # Gradient of c[lag]
        grad = _lag_gradient(h_curr, n, lag) * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        # Try decreasing step sizes
        improved = False
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = project_feasible(h_curr - lr * grad_normed, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h_curr = h_new
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def intensive_local_search(h, n, deadline, top_k=3, sweeps=200):

    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for sweep in range(sweeps):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved_this_sweep = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            for d in [0.001, 0.0005, 0.0002, 0.0001]:
                if time.time() > deadline:
                    break
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    diff_ij = h_curr[i] - h_curr[j]
                    if abs(diff_ij) < 1e-9:
                        continue
                    if diff_ij > 0:
                        delta = min(d, diff_ij * 0.4, h_curr[i] * 0.9, (1.0 - h_curr[j]) * 0.9)
                    else:
                        delta = -min(d, -diff_ij * 0.4, h_curr[j] * 0.9, (1.0 - h_curr[i]) * 0.9)
                    if abs(delta) < 1e-11:
                        continue
                    h_curr[i] -= delta
                    h_curr[j] += delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_curr.copy()
                        improved_this_sweep = True
                        c_all = compute_all_lags(h_curr, n)
                    else:
                        h_curr[i] += delta
                        h_curr[j] -= delta
        if not improved_this_sweep:
            # Try all lags with tiny steps
            c_all = compute_all_lags(h_curr, n)
            top_idx2 = np.argsort(c_all)[-15:]
            found_any = False
            for k_idx in top_idx2:
                if time.time() > deadline:
                    break
                lag = k_idx - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    continue
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    diff_ij = h_curr[i] - h_curr[j]
                    if abs(diff_ij) < 1e-9:
                        continue
                    if diff_ij > 0:
                        delta = min(0.0005, diff_ij * 0.3)
                    else:
                        delta = -min(0.0005, -diff_ij * 0.3)
                    if abs(delta) < 1e-11:
                        continue
                    h_curr[i] -= delta
                    h_curr[j] += delta
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_curr.copy()
                        found_any = True
                        improved_this_sweep = True
                        c_all = compute_all_lags(h_curr, n)
                        break
                    else:
                        h_curr[i] += delta
                        h_curr[j] -= delta
                if found_any:
                    break
            if not found_any:

    return best_h, best_c

def random_local_search(h, n, deadline, n_iter=2000000, rng=None):
    """Random local search targeting worst lag."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

        k_star = int(np.argmax(c_all))

        if lag == 0:
            lag = rng.randint(1, n - 1)
        m = abs(lag)
        i = rng.randint(0, n - m)
        j = i + m

            continue
        direction = 1 if h_curr[i] > h_curr[j] else -1

            h_curr[j] -= delta
    return best_h, best_c

def slsqp_top_lags(h, n, top_m, alpha, maxiter=2000):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        grad = np.zeros(len(x))
        for ii in range(top_m):
            lag = top_idx[ii] - (n - 1)
            grad += w[ii] * _lag_gradient(x, n, lag)
        return grad * dx
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_jac(x):
        return np.ones(len(x))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_round_and_refine(h, n, deadline):

    return project_feasible(h_dst, n_dst)

            best_h = h_opt
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Verify constraints
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    assert abs(best_h.sum() - n_final / 2.0) < 1e-8
    
    return best_h, best_c, n_final
