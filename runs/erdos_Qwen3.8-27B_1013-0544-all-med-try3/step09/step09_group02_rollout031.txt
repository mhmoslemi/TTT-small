def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

def shift_grad_single(h, n, shift_idx):
    """Gradient of C_k at h, where shift_idx is the index in the full correlation array."""
    dx = 2.0 / n
    grad = np.zeros(n)

    # Minimize t
    # subject to: C_k(h) + grad_k · d ≤ t for k in active
    #             -h[i] ≤ d[i] ≤ 1-h[i]
    #             sum(d) = 0
    #             -delta ≤ d[i] ≤ delta
    
    n_var = n + 1  # d[0..n-1] and t
    c_obj = np.zeros(n_var)
    c_obj[n] = 1.0  # minimize t
    
    c_vec = compute_c5_vec(h, n)
    A_ub_list = []
    b_ub_list = []

    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        row = np.zeros(n_var)
        row[0:n] = g
        row[n] = -1.0

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad
