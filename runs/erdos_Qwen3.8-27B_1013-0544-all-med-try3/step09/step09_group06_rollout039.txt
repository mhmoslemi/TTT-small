import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha, top_m=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        c_sub = c[idx]
        m_sub = c_sub.max()
        shifted = alpha * (c_sub - m_sub)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        grad = np.zeros(n)
        for i, ki in enumerate(idx):
            if w[i] < 1e-10:
                continue
            lag = ki - (n - 1)
            grad += w[i] * _lag_gradient(h, n, lag)
        return grad * dx
    else:
        w = smooth_max_weights(c, alpha)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * _lag_gradient(h, n, lag)
        return grad * dx

def grad_descent_adaptive(h, n, deadline, alpha_start=200, alpha_end=5000,
                          max_iter=50000, base_lr=0.5):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    c_curr = best_c
    lr = base_lr
    alpha = alpha_start
    stagnation = 0
    alpha_increase = 1.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h_curr, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        # Backtracking line search
        step = lr
        improved = False
        for ls in range(20):
            h_try = project_feasible(h_curr - step * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_curr - 1e-15:
                h_curr = h_try
                c_curr = c_try
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try smaller fixed steps
            for trial_step in [0.01, 0.005, 0.001]:
                h_try = project_feasible(h_curr - trial_step * g, n)
                c_try = compute_c5(h_try, n)
                if c_try < c_curr - 1e-15:
                    h_curr = h_try
                    c_curr = c_try
                    improved = True
                    break
        if c_curr < best_c - 1e-14:
            best_c = c_curr
            best_h = h_curr.copy()
            stagnation = 0
        else:
            stagnation += 1
        # Adaptive alpha
        if stagnation > 50:
            alpha = min(alpha * alpha_increase, alpha_end)
            lr = max(0.05, lr * 0.8)
            stagnation = 0
        if it % 200 == 199:
            lr = max(0.05, lr * 0.95)
    return best_h, best_c

def _sum_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _top_m_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h, n, top_m, deadline, alpha=100000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    try:
        res = minimize(_top_m_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or (hasattr(res, 'fun') and res.fun < _top_m_obj(h, n, top_idx, alpha)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def local_search_smart(h, n, deadline, top_m=5):

            if lag == 0:

            else:

                    pairs.append((i, i - m))
            # Sort pairs by potential improvement

                scores = []
                for i in range(n - m):
                    # Current contribution: h[i+m]*(1-h[i])
                    # If we increase h[i] by d and decrease h[i+m] by d:
                    # New contribution: (h[i+m]-d)*(1-h[i]-d) vs h[i+m]*(1-h[i])
                    # Change = -d*(1-h[i]) - d*h[i+m] + d^2
                    # Negative (good) when d*(1-h[i]+h[i+m]) > d^2, usually true for small d
                    # But we want to REDUCE the total c, so we want the change to be negative
                    # That means we want h[i+m] > 0 and h[i] < 1
                    score = h[i + m] * (1 - h[i])
                    scores.append(score)
                scores = np.array(scores)
                order = np.argsort(scores)[::-1][:200]
            else:
                scores = []
                for i in range(m, n):
                    score = h[i] * (1 - h[i - m])
                    scores.append(score)
                scores = np.array(scores)
                order = np.argsort(scores)[::-1][:200]
                
            for idx in order:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i = idx
                    j = i + m
                else:
                    j = idx
                    i = j - m

                    if h[j] > h[i]:
                        # Move from j to i: decrease h[j], increase h[i]
                        d = min(delta, (h[j] - h[i]) * 0.5, h[j] * 0.8)
                        if d < 1e-10:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            c_all = compute_all_lags(h, n)
                            improved_this_lag = True
                            rounds_since_improve = 0
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] > h[j]:
                        # Move from i to j: decrease h[i], increase h[j]
                        d = min(delta, (h[i] - h[j]) * 0.5, h[i] * 0.8)
                        if d < 1e-10:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            c_all = compute_all_lags(h, n)
                            improved_this_lag = True
                            rounds_since_improve = 0
                            break
                        else:
                            h[i] += d
                            h[j] -= d
                    else:
                        # Equal, try small perturbation
                        d = delta * 0.5
                        if d < min(h[i], 1-h[j]) - 1e-10:
                            h[i] -= d
                            h[j] += d
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:

                                c_all = compute_all_lags(h, n)
                                improved_this_lag = True
                                rounds_since_improve = 0
                                break
                            else:
                                h[i] += d
                                h[j] -= d
                if improved_this_lag:
                    break
        if not improved_this_lag:
            rounds_since_improve += 1
    return best_h, best_c
