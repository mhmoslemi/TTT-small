    h_start = np.clip(h_start, 0, 1)
    try:
        res = minimize(obj_and_grad_lbfgs, h_start, method='L-BFGS-B',
                       bounds=bounds, jac=True, args=(n, alpha, penalty_coeff),
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40,
                                'iprint': -1, 'maxfun': maxiter * 3})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h_start.copy(), n)

def fista_optimize(h, n, alpha, deadline, penalty_coeff=1e5, lr_init=0.002, top_k=20):
    """FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) style optimization."""
    h_curr = h.copy()
    h_prev = h.copy()
    t_mom = 1.0
    lr = lr_init
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(20000):
        if time.time() > deadline:
            break
        grad = grad_smooth_max(h_curr, n, alpha, top_k)
        target_sum = n / 2.0
        diff = h_curr.sum() - target_sum
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        
        # Gradient step
        h_step = h_curr - lr * grad
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        h_mom = h_step + ((t_mom - 1.0) / t_new) * (h_step - h_prev)
        # Project
        h_new = project_feasible(h_mom, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
        t_mom = t_new
        lr = lr_init / (1.0 + 0.01 * it)
    
    return best_h, best_c

def targeted_lag_optimization(h, n, deadline, rng, top_lags=5):
    """Directly optimize to reduce the top overlapping lags via small perturbations."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for outer in range(5000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_lags:]
        
        improved = False
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Try to reduce overlap at this lag by swapping mass
            n_pairs = n - m
            if n_pairs <= 0:
                continue
            # Focus on high-overlap pairs
            if lag > 0:
                overlap_vals = h[:n_pairs] * (1 - h[lag:])
            else:
                overlap_vals = h[m:] * (1 - h[:n-m])
            
            top_pairs = np.argsort(overlap_vals)[-20:]
            for p in top_pairs:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i, j = p, p + lag
                else:
                    i, j = p + m, p
                
                # Move mass from h[i] to h[j] (if h[i] > h[j]) or vice versa
                if h[i] > h[j] + 1e-6:
                    d = min(0.005, (h[i] - h[j]) * 0.1)
                    old_i, old_j = h[i], h[j]
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = old_i, old_j
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-6:
                    d = min(0.005, (h[j] - h[i]) * 0.1)
                    old_i, old_j = h[i], h[j]
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = old_i, old_j
                        h = project_feasible(h, n)
        
        if not improved:
            # Try random small perturbations
            for _ in range(10):
                i1 = rng.randint(0, n)
                i2 = rng.randint(0, n)
                if i1 == i2:
                    continue
                d = rng.uniform(-0.003, 0.003)
                old_i, old_j = h[i1], h[i2]
                h[i1] -= d
                h[i2] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[i1], h[i2] = old_i, old_j
                    h = project_feasible(h, n)
            if not improved:
                break
    
    return best_h, best_c

def multi_scale_search(h_start, n_start, deadline, rng):
    """Search at multiple resolutions, keeping the best."""
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n_start)
    
    resolutions = [n_start]
    # Try higher resolutions
    for mult in [1.5, 2.0, 3.0]:
        n_try = int(n_start * mult)
        if n_try > 3000 or n_try < 100:
            continue
        resolutions.append(n_try)
    
    for n_try in resolutions:
        if time.time() > deadline - 30:
            break
        if n_try != n_start:
            # Interpolate
            x_src = np.linspace(0, 2, len(best_h))
            x_dst = np.linspace(0, 2, n_try)
            h_try = np.interp(x_dst, x_src, best_h)
            h_try = project_feasible(h_try, n_try)
        else:
            h_try = best_h.copy()
        
        # Optimize at this resolution
        remaining = deadline - time.time() - 20
        if remaining < 10:
            break
        
        # L-BFGS-B
        t_sub = min(40, remaining * 0.4)
        h_opt = lbfgsb_optimize(h_try, n_try, 100000, time.time() + t_sub, maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            best_n = n_try
        else:
            best_n = len(best_h)
        
        # FISTA
        if time.time() < deadline - 20:
            t_sub2 = min(20, (deadline - 20) - time.time())
            h_fista, c_fista = fista_optimize(h_opt if c_opt < best_c else h_try,
                                               n_try if c_opt < best_c else n_start,
                                               time.time() + t_sub2, alpha=500000)
            if c_fista < best_c:
                best_c = c_fista
                best_h = h_fista
                best_n = n_try
    
    return best_h, best_c

def simulated_annealing_targeted(h, n, deadline, rng, alpha=100000, t_init=0.005, t_min=0.00002, cool=0.9999):
    """SA with targeted moves on high-overlap lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stuck = 0
    
    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-4:]
        k_star = top_idx[rng.randint(len(top_idx))]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue
        
        i = rng.randint(0, n - m)
        j = i + m
        if h[i] <= h[j] + 1e-6:
            T *= cool
            continue
        
        d = rng.uniform(0.001, 0.03)
        d = min(d, (h[i] - h[j]) * 0.3)
        if d < 1e-8:
            T *= cool
            continue
        
        c_old_max = c_all.max()
        old_i, old_j = h[i], h[j]
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        c_new_max = c_new.max()
        
        de = c_new_max - c_old_max
        accept = de < 0 or (T > t_min and rng.random() < np.exp(-max(0, de) / T))
        
        if accept:
            if c_new_max < best_c - 1e-14:
                best_c = float(c_new_max)
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i], h[j] = old_i, old_j
            h = project_feasible(h, n)
            stuck += 1
        
        T *= cool
        if stuck > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            stuck = 0
            T = max(T, 0.001)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    
    # Phase 1: L-BFGS-B with progressive alpha at initial resolution
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    maxiters = [200, 300, 400, 500, 600, 800, 1000]
    
    for idx_a, (alpha, maxit) in enumerate(zip(alphas, maxiters)):
        if time.time() > deadline - 400:
            break
        remaining = deadline - 400 - time.time()
        if remaining < 10:
            break
        t_alloc = min(30 + 5 * idx_a, remaining * 0.5)
        h_opt = lbfgsb_optimize(best_h, best_n, alpha, time.time() + t_alloc, maxiter=maxit)
        c_opt = compute_c5(h_opt, best_n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: FISTA at current best
    if time.time() < deadline - 300:
        t_alloc = min(60, (deadline - 300) - time.time())
        h_fista, c_fista = fista_optimize(best_h, best_n, time.time() + t_alloc,
                                          alpha=500000, lr_init=0.001, top_k=15)
        if c_fista < best_c:
            best_c = c_fista
            best_h = h_fista
    
    # Phase 3: Targeted lag optimization
    if time.time() < deadline - 250:
        t_alloc = min(80, (deadline - 250) - time.time())
        h_targ, c_targ = targeted_lag_optimization(best_h, best_n, time.time() + t_alloc, rng, top_lags=4)
        if c_targ < best_c:
            best_c = c_targ
            best_h = h_targ
    
    # Phase 4: Multi-scale search (try higher resolutions)
    if time.time() < deadline - 150:
        t_alloc = deadline - 150 - time.time()
        h_ms, c_ms = multi_scale_search(best_h, best_n, time.time() + t_alloc, rng)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms
            best_n = len(best_h)
    
    # Phase 5: SA at best resolution
    if time.time() < deadline - 80:
        t_alloc = min(60, (deadline - 80) - time.time())
        h_sa, c_sa = simulated_annealing_targeted(best_h, best_n, time.time() + t_alloc, rng,
                                                   alpha=200000, t_init=0.004, t_min=0.00001, cool=0.9999)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 6: Final L-BFGS-B polish
    if time.time() < deadline - 30:
        t_alloc = min(25, (deadline - 30) - time.time())
        h_opt = lbfgsb_optimize(best_h, best_n, 10000000, time.time() + t_alloc, maxiter=3000)
        c_opt = compute_c5(h_opt, best_n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 7: Final targeted optimization
    if time.time() < deadline - 5:
        t_alloc = min(15, (deadline - 5) - time.time())
        h_targ, c_targ = targeted_lag_optimization(best_h, best_n, time.time() + t_alloc, rng, top_lags=3)
        if c_targ < best_c:
            best_c = c_targ
            best_h = h_targ
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
