    return best_h, best_c

def targeted_local_search(h, n, deadline, rng, max_iters=100000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    no_improve = 0
    step_sizes = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    ss_idx = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-8:]

        improved = False
        base_d = step_sizes[ss_idx]

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            # Try a few random pairs at this lag
            for trial in range(5):
                if time.time() > deadline:
                    break
                if lag > 0:
                    i = rng.randint(0, n - m)
                    j_from = i + m
                    j_to = i
                else:
                    i = rng.randint(0, n - m)
                    j_from = i
                    j_to = i + m

                diff_h = h[j_from] - h[j_to]
                if abs(diff_h) < 0.005:
                    continue
                d = min(base_d, abs(diff_h) * 0.4)
                if d < 1e-8:
                    continue
                sign = 1.0 if diff_h > 0 else -1.0

                h[j_from] -= sign * d
                h[j_to] += sign * d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-13:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                else:
                    h[j_from] += sign * d
                    h[j_to] -= sign * d
                    h = project_feasible(h, n)

        if not improved:
            no_improve += 1
            if no_improve > 500:
                ss_idx = min(ss_idx + 1, len(step_sizes) - 1)
                no_improve = 0
                if ss_idx == len(step_sizes) - 1:
                    break

    return best_h, best_c

def subgradient_descent(h, n, deadline, max_iters=20000, top_k=3):
    """Direct sub-gradient on the max with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = subgrad_max(h, n, top_k=top_k)
        # Project gradient onto the tangent space of sum constraint
        grad_tangent = grad - grad.mean()

        h_new = h - lr * grad_tangent
        h_new = project_feasible(h_new, n)
        f_new = compute_c5(h_new, n)

        if f_new < best_c - 1e-13:
            best_c = f_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.9
                no_improve = 0
                if lr < 0.0001:
                    break

    return best_h, best_c

def interpolate_resolve(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = np.arange(n_src) * (2.0 / n_src) + 1.0 / n_src
    x_dst = np.arange(n_dst) * (2.0 / n_dst) + 1.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binarize_and_relax(h, n, deadline, rng):
    """Threshold to binary with optimal n/2 ones, then relax."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1

    # Slight random perturbation to break symmetry
    h_pert = h_bin + rng.normal(0, 0.05, n)
    h_pert = project_feasible(h_pert, n)

    # Optimize from this starting point
    h_opt = lbfgsb_optimize(h_pert, n, 500000, min(deadline, time.time() + 40), maxiter=2000)
    return h_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    deadline = t0 + budget_s - 5

    n = len(best_h)

    # Phase 1: L-BFGS-B with increasing alpha at parent resolution
    alpha_schedule = [1000, 5000, 50000, 200000, 1000000, 5000000]
    maxiter_schedule = [2000, 2000, 3000, 3000, 3000, 3000]
    for ci, (alpha, maxit) in enumerate(zip(alpha_schedule, maxiter_schedule)):
        if time.time() > deadline - 400:
            break
        remaining = deadline - 400 - time.time()
        if remaining < 10:
            break
        t_end = time.time() + min(remaining * 0.8, 80)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t_end, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Frank-Wolfe refinement
    if time.time() < deadline - 250:
        fw_deadline = min(deadline - 150, time.time() + 100)
        h_opt, c_opt = frank_wolfe_refine(best_h, n, fw_deadline, max_iter=8000, alpha=5000000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Sub-gradient descent on exact max
    if time.time() < deadline - 200:
        sg_deadline = min(deadline - 100, time.time() + 60)
        h_opt, c_opt = subgradient_descent(best_h, n, sg_deadline, max_iters=15000, top_k=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B with very high alpha (near-exact max)
    if time.time() < deadline - 180:
        for alpha in [10000000, 50000000, 200000000]:
            if time.time() > deadline - 100:
                break
            t_end = time.time() + min(40, deadline - 100 - time.time())
            if t_end <= time.time():
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, t_end, maxiter=5000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # Also try sub-gradient from the new point
            if time.time() < deadline - 80:
                h_opt2, c_opt2 = subgradient_descent(best_h, n, time.time() + 20, max_iters=5000, top_k=5)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 5: Targeted local search at parent resolution
    if time.time() < deadline - 80:
        ls_deadline = min(deadline - 30, time.time() + 80)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, rng, max_iters=200000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_try in [600, 800, 1000]:
        if time.time() > deadline - 100:
            break
        remaining = deadline - 60 - time.time()
        if remaining < 30:
            break
        h_interp = interpolate_resolve(best_h, len(best_h), n_try)
        c_interp = compute_c5(h_interp, n_try)

        # Optimize at new resolution
        t_end1 = time.time() + min(30, remaining * 0.35)
        h_opt = lbfgsb_optimize(h_interp, n_try, 200000, t_end1, maxiter=3000)
        c_opt = compute_c5(h_opt, n_try)

        if time.time() < deadline - 40:
            t_end2 = time.time() + min(25, remaining * 0.3)
            h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, t_end2, maxiter=3000)
            c_opt = compute_c5(h_opt, n_try)

        # Frank-Wolfe at new resolution
        if time.time() < deadline - 30:
            h_opt, c_opt = frank_wolfe_refine(h_opt, n_try, min(deadline - 25, time.time() + 20), max_iter=3000, alpha=5000000)
            c_opt = compute_c5(h_opt, n_try)

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Binarize and relax as perturbation
    if time.time() < deadline - 50:
        n_now = len(best_h)
        h_bin_relax = binarize_and_relax(best_h, n_now, deadline, rng)
        c_bin = compute_c5(h_bin_relax, n_now)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin_relax
            # Polish
            if time.time() < deadline - 20:
                h_opt = lbfgsb_optimize(best_h, n_now, 10000000, time.time() + 20, maxiter=2000)
                c_opt = compute_c5(h_opt, n_now)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 8: Final intensive local search
    if time.time() < deadline - 15:
        n_final = len(best_h)
        h_opt, c_opt = targeted_local_search(best_h, n_final, deadline - 3, rng, max_iters=300000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final sub-gradient + FW polish
    if time.time() < deadline - 8:
        n_final = len(best_h)
        h_opt, c_opt = subgradient_descent(best_h, n_final, time.time() + (deadline - 8 - time.time()) * 0.5, max_iters=5000, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 4:
            h_opt, c_opt = frank_wolfe_refine(best_h, n_final, deadline - 2, max_iter=3000, alpha=10000000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
