```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
```

```python
def active_set_optimize(h, n, deadline, epsilon=0.01):
    """Optimize by focusing on active set of shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(20):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Active set: shifts within epsilon of max
        active = np.where(c_vec >= max_val - epsilon)[0]
        
        if len(active) == 0:
            break
        
        # Optimize to equalize the active shifts and reduce them
        # Use SLSQP: minimize t subject to f_k(h) <= t for k in active
        def obj(x_t):
            x = x_t[:-1]
            t_val = x_t[-1]
            c = compute_c5_vec(x, n)
            return t_val
        
        def obj_jac(x_t):
            return np.append(np.zeros(n), 1.0)
        
        cons = []
        for k_idx in active:
            def con(x_t, ki=k_idx):
                x = x_t[:-1]
                c = compute_c5_vec(x, n)
                return c[ki] - x_t[-1]
            def con_jac(x_t, ki=k_idx):
                x = x_t[:-1]
                # gradient of c[ki] w.r.t. x
                n_ = n
                dx_ = 2.0 / n_
                k = ki - (n_ - 1)
                g = np.zeros(n_)
                if k >= 0:
                    nv = n_ - k
                    g[0:nv] = (1.0 - x[k:n_]) * dx_
                    g[k:n_] += (-x[0:nv]) * dx_
                else:
                    kk = -k
                    nv = n_ - kk
                    g[kk:n_] = (1.0 - x[0:nv]) * dx_
                    g[0:nv] += (-x[kk:n_]) * dx_
                jac = np.append(g, -1.0)
                return jac
            cons.append({'type': 'ineq', 'fun': con, 'jac': con_jac})
        
        # Equality constraint: sum h = n/2
        def eq_con(x_t):
            return x_t[:-1].sum() - n / 2.0
        def eq_con_jac(x_t):
            return np.append(np.ones(n), 0.0)
        cons.append({'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac})
        
        x0 = np.append(best_h, max_val)
        bounds_ext = bounds + [(0.0, 1.0)]
```

c_k(δ) = base_k + a_k * δ

The optimal δ is where the "upper envelope" of the lines has its minimum. This can be found by:
1.Computing the values at δ=0 and the slopes
2. The minimum is at the intersection of two lines, or at a boundary

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_shift_grad(h, n, k_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = k_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def lse_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-13:
            continue
        grad += w[k_idx] * corr_shift_grad(x, n, k_idx)
    return grad

def topset_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def topset_grad(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k_idx in zip(w, active):
        if wi < 1e-13:
            continue
        grad += wi * corr_shift_grad(x, n, k_idx)
    return grad

def run_lbfgs(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad, args=(n, alpha),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 20000, 'ftol': 1e-20,
                                  'gtol': 1e-14, 'maxfun': 200000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

            above = np.where(cur_h > 0.5)[0]
            below = np.where(cur_h < 0.5)[0]
            if len(above) > 0 and len(below) > 0:
                i = above[rng.randint(len(above))]
                j = below[rng.randint(len(below))]
            else:
                i, j = rng.randint(n, 2)
                if i == j:
                    continue
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        elif r < 0.85:
            i, j = rng.randint(n, 2)
            if i == j:
                continue
            amt = rng.uniform(0.01, 0.15)
            if cur_h[i] >= amt and cur_h[j] <= 1 - amt:
                cur_h[i] -= amt
                cur_h[j] += amt
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        else:
            i = rng.randint(n)
            cur_h[i] += rng.normal(0, 0.05)
            cur_h[i] = np.clip(cur_h[i], 0, 1)
        cur_h = project_feasible(cur_h, n)
        new_c = compute_c5(cur_h, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            accepted += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h = old_i
    return best_h, best_c

def run_fine_pairs(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        n_pairs = min(2000, int((deadline - time.time()) * 200))
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for direction in [1, -1]:
                for step in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    di = direction * step
                    dj = -direction * step
                    if best_h[i] + di < 0 or best_h[i] + di > 1:
                        continue
                    if best_h[j] + dj < 0 or best_h[j] + dj > 1:
                        continue
                    old_i, old_j = best_h[i], best_h[j]
                    best_h[i] += di
                    best_h[j] += dj
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] = old_i
                        best_h[j] = old_j
        if not improved and sweep > 20:
            break
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def structured_init(n, freq):
    h = np.zeros(n)
    for i in range(n):
        if np.sin(2 * np.pi * freq * i / n) > 0:
            h[i] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = random_binary_init(500, seed=42)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS at original resolution (quick)
    if time.time() < deadline - 600:
        t1 = min(time.time() + 40, deadline - 600)
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            alphas=[500, 2000, 10000, 50000, 200000, 1000000])

    # Phase 2: Upsample to n=800
    n2 = 800
    h2 = upsample(best_h, len(best_h), n2)
    c2 = compute_c5(h2, n2)
    if c2 < best_c:
        best_h, best_c = h2, c2
    else:
        best_h, best_c = h2, c2

    if time.time() < deadline - 450:
        t2 = min(time.time() + 120, deadline - 450)
        h2, c2 = run_lbfgs(h2, n2, t2,
            alphas=[500, 2000, 10000, 50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Top-set focused optimization at n=800
    if time.time() < deadline - 350:
        t3 = min(time.time() + 100, deadline - 350)
        h3, c3 = run_topset_lbfgs(best_h, n2, t3, max_rounds=30, top_k=6)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: SA at n=800
    if time.time() < deadline - 250:
        t4 = min(time.time() + 100, deadline - 250)
        h4, c4 = run_sa(best_h, n2, t4, t_init=0.002, t_final=1e-7, seed=seed)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Momentum subgradient
    if time.time() < deadline - 180:
        t5 = min(time.time() + 60, deadline - 180)
        h5, c5 = run_momentum_subgrad(best_h, n2, t5, step0=0.0005, momentum=0.85)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Fine pair search
    if time.time() < deadline - 120:
        t6 = min(time.time() + 60, deadline - 120)
        h6, c6 = run_fine_pairs(best_h, n2, t6)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Try higher resolution n=1200
    n3 = 1200
    if time.time() < deadline - 100:
        h7 = upsample(best_h, n2, n3)
        c7 = compute_c5(h7, n3)
        if c7 < best_c:
            best_h, best_c = h7, c7
        else:
            h7, c7 = h7, c7
        if time.time() < deadline - 40:
            t7 = min(time.time() + 55, deadline - 40)
            h7, c7 = run_lbfgs(h7, n3, t7,
                alphas=[1000, 5000, 20000, 100000, 500000])
            if c7 < best_c:
                best_h, best_c = h7, c7

    # Phase 8: Second SA round at best resolution
    if time.time() < deadline - 40:
        n_best = len(best_h)
        t8 = min(time.time() + 30, deadline - 40)
        h8, c8 = run_sa(best_h, n_best, t8, t_init=0.001, t_final=1e-8, seed=seed + 100)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Phase 9: Final top-set polish
    if time.time() < deadline - 5:
        n_best = len(best_h)
        h9, c9 = run_topset_lbfgs(best_h, n_best, deadline - 2, max_rounds=10, top_k=4)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
