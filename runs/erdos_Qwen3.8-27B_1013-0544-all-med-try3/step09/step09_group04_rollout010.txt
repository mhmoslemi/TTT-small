import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(sh)] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

def act_grad(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def de_mutation(pop, F):
    """DE/rand/1 mutation"""
    n = len(pop)
    N = len(pop[0])
    mutant = np.zeros(N)
    a, b, c = np.random.choice(n, 3, replace=False)
    mutant = pop[a] + F * (pop[b] - pop[c])
    return mutant

def de_crossover(trial, current, CR):
    """Binary crossover"""
    N = len(current)
    j_rand = np.random.randint(N)
    mask = np.random.random(N) <= CR
    mask[j_rand] = True
    return np.where(mask, trial, current)

def run_de(h_start, n, deadline, pop_size=30, F=0.5, CR=0.7, max_gens=200):
    """Differential Evolution"""
    rng = np.random.default_rng(42)
    target = n / 2.0
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = h_start.copy()
    pop[1] = project_feasible(h_start + 0.1 * rng.standard_normal(n), n)
    pop[2] = project_feasible(np.clip(1 - h_start + 0.05 * rng.standard_normal(n), 0, 1) * target / np.clip(1 - h_start, 0.01, 1).sum() * n / 2, n)
    for i in range(3, pop_size):
        pop[i] = project_feasible(rng.random(n) * 0.5 + 0.25, n)
        # Ensure sum constraint
        pop[i] = project_feasible(pop[i], n)
    
    costs = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    best_idx = np.argmin(costs)
    best_h = pop[best_idx].copy()
    best_c = costs[best_idx]
    
    no_improve = 0
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        if no_improve > 100:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            # Mutation
            mutant = de_mutation(pop, F)
            mutant = np.clip(mutant, 0, 1)
            # Crossover
            trial = de_crossover(mutant, pop[i], CR)
            trial = project_feasible(trial, n)
            # Selection
            cost = compute_c5(trial, n)
            if cost <= costs[i]:
                pop[i] = trial
                costs[i] = cost
                if cost < best_c - 1e-10:
                    best_c = cost
                    best_h = trial.copy()
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
    return best_h, best_c

def run_smooth_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = maxiters[i] if maxiters and i < len(maxiters) else max(200, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 5,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
