        if time.time() > deadline:
            break
        h_raw = 0.5 + params[:p] @ cos_bas + params[p:] @ sin_bas
        h = np.clip(h_raw, 0.0, 1.0)
        diff = h.sum() - n / 2.0
        c = compute_all_lags(h, n)
        c_max = c.max()

        if c_max < best_c - 1e-14:
            best_c = float(c_max)
            best_h = h.copy()
            best_params = params.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        # Gradient
        m = c.max()
        shifted = alpha * (c - m)
        sh_max = shifted.max()
        if sh_max > 500:
            w = np.exp(shifted - sh_max)
        else:
            w = np.exp(shifted)
        w /= w.sum()

        grad_h = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-12:
                continue
            lag = k_idx - (n - 1)
            if lag >= 0:
                grad_h[lag:] += w[k_idx] * h[:n - lag]
                grad_h[:n - lag] -= w[k_idx] * h[lag:]
            else:
                idx = -lag
                grad_h[idx:] += w[k_idx] * h[:n - idx]
                grad_h[:n - idx] -= w[k_idx] * h[idx:]
        grad_h *= 2.0 / n

        grad_params = np.zeros(2 * p)
        grad_params[:p] = cos_bas.T @ grad_h
        grad_params[p:] = sin_bas.T @ grad_h

        # Adaptive LR
        lr_eff = lr * (0.9995 ** it)
        params -= lr_eff * grad_params

        # Keep params small to maintain feasibility
        norm_p = np.linalg.norm(params)
        if norm_p > 0.45:
            params *= 0.45 / norm_p

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: Fourier basis optimization at n=500
    if time.time() < deadline - 400:
        p_fourier = 40
        fd = min(deadline - 380, time.time() + 60)
        h_four, c_four = fourier_optimize(p_fourier, n_init, best_h, fd, n_starts=3)
        if c_four < best_c:
            best_c = c_four
            best_h = h_four

    # Phase 2: L-BFGS-B at n=500 with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 350:
            break
        remaining = deadline - 350 - time.time()
        if remaining < 10:
            break
        h_opt = lbfgsb_smooth_max(best_h, n_init, alpha, time.time() + min(35, remaining * 0.25),
                                  maxiter=3000)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Expand to n=1000 and optimize
    n_high = 1000
    if time.time() < deadline - 250:
        h_high = expand_to_resolution(best_h, len(best_h), n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.005:  # Allow small increase from interpolation
            for alpha in [5000, 50000, 500000]:
                if time.time() > deadline - 150:
                    break
                remaining = deadline - 150 - time.time()
                if remaining < 15:
                    break
                h_opt = lbfgsb_smooth_max(h_high, n_high, alpha,
                                          time.time() + min(25, remaining * 0.3),
                                          maxiter=4000)
                c_opt = compute_c5(h_opt, n_high)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    h_high = h_opt
                else:
                    h_high = h_opt  # Keep the optimized version even if not globally better
    else:
        h_high = best_h
        n_high = len(best_h)

    # Phase 4: Targeted local search
    if time.time() < deadline - 200:
        ls_deadline = min(deadline - 180, time.time() + 120)
        h_opt, c_opt = targeted_local_search(best_h, len(best_h), ls_deadline, rng, top_k=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: SA
    if time.time() < deadline - 150:
        sa_dead = min(deadline - 130, time.time() + 100)
        h_opt, c_opt = sa_optimize(best_h, len(best_h), sa_dead, rng,
                                   alpha_sa=100000, t_init=0.004, t_min=0.00001, cool=0.99992)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: L-BFGS-B polish with high alpha
    if time.time() < deadline - 80:
        for alpha in [2000000, 20000000]:
            if time.time() > deadline - 50:
                break
            h_opt = lbfgsb_smooth_max(best_h, len(best_h), alpha, time.time() + 20, maxiter=5000)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: GRFD basis at high resolution
    if time.time() < deadline - 100:
        p_gd = 30
        gd_dead = min(deadline - 80, time.time() + 60)
        h_gd, c_gd = gradient_descent_basis(best_h, len(best_h), p_gd, gd_dead,
                                            alpha=100000, lr=0.0005, max_iter=15000)
        if c_gd < best_c:
            best_c = c_gd
            best_h = h_gd

    # Phase 8: Final intensive local search
    if time.time() < deadline - 20:
        ls2_deadline = min(deadline - 5, time.time() + 50)
        h_opt, c_opt = targeted_local_search(best_h, len(best_h), ls2_deadline, rng, top_k=6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final tiny SA
    if time.time() < deadline - 10:
        sa2_dead = min(deadline - 3, time.time() + 20)
        h_opt, c_opt = sa_optimize(best_h, len(best_h), sa2_dead, rng,
                                   alpha_sa=500000, t_init=0.001, t_min=0.000005, cool=0.99995)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
