def pair_descent(h, n, deadline, max_radius=40):

    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue

                        best_h[i] -= step

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

                                       'gtol': 1e-12, 'maxfun': 30000,
                                       'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return best_h, best_c

def pair_descent(h, n, deadline, max_radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - step >= 0 and best_h[j] + step <= 1:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
                    if best_h[j] - step >= 0 and best_h[i] + step <= 1:
                        best_h[j] -= step
                        best_h[i] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += step
                            best_h[i] -= step
    return best_h, best_c

def gradient_refine(h, n, deadline, alpha=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    while time.time() < deadline:
        grad = smooth_max_grad(best_h, n, alpha, 0.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        improved = False
        for mult in [lr, lr * 0.5, lr * 0.1, lr * 0.01, lr * 0.001]:
            h_new = best_h - mult * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def multisat_restart(h, n, deadline, n_restarts=3, alpha=50000, mu=10.0, std=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        h_pert = best_h + np.random.randn(n) * std
        h_pert = project_feasible(h_pert, n)
        try:
            res = minimize(smooth_max_obj, h_pert, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': 20000,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    n = 1000
    h = upsample(h_init, n_orig, n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Initial L-BFGS with increasing smoothness at n=1000
    t1 = min(t0 + 120, deadline - 550)
    best_h, best_c = run_lbfgs_phase(best_h, n, t1,
        [ (50, 1.0), (200, 5.0), (1000, 10.0), (5000, 20.0),
          (20000, 50.0), (50000, 100.0) ],
        maxiter=3000)

    # Phase 2: Active set - broader set of shifts
    if time.time() < deadline - 400:
        t2 = min(time.time() + 130, deadline - 400)
        best_h, best_c = run_active_set_phase(best_h, n, t2, top_k=10,
            alphas=[5000, 50000, 200000])

    # Phase 3: High-alpha L-BFGS for sharp max approximation
    if time.time() < deadline - 280:
        t3 = min(time.time() + 100, deadline - 280)
        best_h, best_c = run_lbfgs_phase(best_h, n, t3,
            [ (100000, 50.0), (500000, 100.0), (2000000, 200.0) ],
            maxiter=5000)

    # Phase 4: Active set - narrow focus on top shifts
    if time.time() < deadline - 180:
        t4 = min(time.time() + 80, deadline - 180)
        best_h, best_c = run_active_set_phase(best_h, n, t4, top_k=4,
            alphas=[50000, 200000, 1000000, 5000000])

    # Phase 5: Multi-restart with perturbations
    if time.time() < deadline - 120:
        t5 = min(time.time() + 60, deadline - 120)
        best_h, best_c = multisat_restart(best_h, n, t5, n_restarts=4,
            alpha=200000, mu=50.0, std=0.015)

    # Phase 6: Gradient refinement
    if time.time() < deadline - 60:
        t6 = min(time.time() + 40, deadline - 60)
        best_h, best_c = gradient_refine(best_h, n, t6, alpha=500000)

    # Phase 7: Another active set round after gradient refine
    if time.time() < deadline - 40:
        t7 = min(time.time() + 25, deadline - 38)
        best_h, best_c = run_active_set_phase(best_h, n, t7, top_k=3,
            alphas=[200000, 1000000])

    # Phase 8: Fine pair descent
    if time.time() < deadline - 15:
        t8 = min(time.time() + 20, deadline - 12)
        best_h, best_c = pair_descent(best_h, n, t8, max_radius=25)

    # Phase 9: Final gradient refine
    if time.time() < deadline - 5:
        t9 = min(time.time() + 5, deadline - 3)
        best_h, best_c = gradient_refine(best_h, n, t9, alpha=2000000)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
