def project_feasible(h, n):

                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def linearized_active_step(h, n, active_set, alpha_step=1.0):
    """One step of linearized active-set method."""
    c_vec = compute_c5_vec(h, n)
    c_active = c_vec[active_set]
    t_current = c_active.max()
    
    # Compute gradients of active constraints
    # c_k(h) = sum_i h[i](1-h[i-s_k]) * dx for shift s_k
    # grad c_k / grad h[j] = dx * [(1-h[j-s_k]) - h[j+s_k]] (when valid)
    
    m = len(active_set)
    A = np.zeros((m, n))  # Gradient matrix
    dx = 2.0 / n
    
    for row, k_idx in enumerate(active_set):
        s = k_idx - (n - 1)
        for j in range(n):
            g = 0.0
            # h[j] as first factor: h[j] * (1-h[j-s]), valid when 0 <= j-s < n
            if 0 <= j - s < n:
                g += (1.0 - h[j - s])
            # h[j] as second factor: h[j+s] * (1-h[j]), valid when 0 <= j+s < n
            if 0 <= j + s < n:
                g += (-h[j + s])
            A[row, j] = g * dx
    
    # Solve LP: min t s.t. A*(h-h0) + c_active <= t, 0<=h<=1, sum(h)=n/2
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # This is a LP with n+1 variables and m+n+1 constraints
    
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # Minimize t
    
    # Inequality constraints: A*(h-h0) + c_active - t <= 0
    # i.e., A*h - t <= -c_active + A*h0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = -c_active + A @ h
    
    # Sum constraint: sum(h) = n/2 (as equality)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        # Line search: h = h0 + alpha * (h_new - h0)
        best_alpha = alpha_step
        best_c = compute_c5(h, n)
        for a in [alpha_step, alpha_step*0.5, alpha_step*0.25, alpha_step*0.1]:
            h_try = h + a * (h_new - h)
            h_try = np.clip(h_try, 0, 1)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_alpha = a
        h_result = np.clip(h + best_alpha * (h_new - h), 0, 1)
        h_result = project_feasible(h_result, n)
        return h_result, compute_c5(h_result, n)
    
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)
