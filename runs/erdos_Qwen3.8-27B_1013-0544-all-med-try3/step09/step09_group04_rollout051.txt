    m = c.max()

    w = np.exp(sh - mx)
    w /= w.sum()
    
    # Compute G = ∑ w_k * ∇c_k(h)
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-14:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def mm_project(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: h_j = clip(v_j - lambda, 0, 1), sum = target
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(v - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target - h_new.sum()
    if abs(diff) > 1e-13:
        free = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if free.sum() > 0:
            h_new[free] += diff / free.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def mm_optimize(h, n, deadline, max_iters=50000, mu_init=0.005):
    """MM (minorization) optimization"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = mu_init
    alpha = 50000
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        c_max = c.max()
        
        # Soft-max weights for subgradient
        sh = alpha * (c - c_max)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        
        # Compute combined gradient
        G = np.zeros(n)
        for mm in range(2 * n - 1):
            wi = w[mm]
            if wi < 1e-14:
                continue
            k = mm - (n - 1)
            if k >= 0:
                nv = n - k
                G[0:nv] += wi * (1.0 - h[k:n]) * dx
                G[k:n] += wi * (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                G[kk:n] += wi * (1.0 - h[0:nv]) * dx
                G[0:nv] += wi * (-h[kk:n]) * dx
        
        # MM step: v = h - mu * G, project
        v = h - mu * G
        h_new = mm_project(v, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            mu = min(mu * 0.99, 0.02)  # gradually increase step
        else:
            mu *= 0.8
        
        h = h_new
        
        # Adaptive alpha
        if it % 1000 == 0:
            alpha *= 2
            if alpha > 1000000:
                alpha = 1000000
    
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(100, int(remaining))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 15,
                                    'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
