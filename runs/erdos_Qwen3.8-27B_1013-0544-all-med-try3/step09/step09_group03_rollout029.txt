```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_sweep(h, n, c_vec, lam=0.0):
    """One full sweep of coordinate descent. h is modified in place."""
    center = n - 1
    for i in range(n):
        h_i = h[i]
        
        # Compute slope vector
        s = np.zeros(2 * n - 1)
        s[center - i : center + n - i] += (1.0 - h)
        s[center + i - n + 1 : center + i + 1] += -h[::-1]
        
        # 1D subproblem: min_v max_k (c_vec[k] + s[k]*(v - h_i)) + lam*v
        # = min_v max_k (c_vec[k] - s[k]*h_i + s[k]*v) + lam*v
        a = c_vec - s * h_i
        b = s + lam  # combined slope (including Lagrange term)
        
        # Ternary search on v ∈ [0, 1]
        lo = 0.0
        hi = 1.0
        for _ in range(80):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(a + b * m1)
            f2 = np.max(a + b * m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        v_opt = (lo + hi) / 2.0
        
        # Update
        h[i] = v_opt
        c_vec += s * (v_opt - h_i)
    
    return h, c_vec

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (1000, 500.0), (5000, 5000.0), (20000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_sweeps=100, verbose=False, deadline=None):
    """
    Coordinate descent for minimizing max_k c_k.
    h is modified in place.
    """
    center = n - 1
    h = h.copy()
    c_vec = np.correlate(h, 1 - h, mode='full')  # not multiplied by dx
    
    best_c = np.max(c_vec)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        start_time = time.time()
        
        for i in range(n):
            h_i = h[i]
            
            # Compute slope vector (vectorized)
            s = np.zeros(2 * n - 1)
            s[center - i : center + n - i] += (1.0 - h)
            s[center + i - n + 1 : center + i + 1] += -h[::-1]
            
            # 1D subproblem: min_v max_k (c_vec[k] + s[k]*(v - h_i))
            # Ternary search on v ∈ [0, 1]
            a = c_vec - s * h_i
            b = s
            
            lo = 0.0
            hi = 1.0
            for _ in range(80):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a + b * m1)
                f2 = np.max(a + b * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_opt = (lo + hi) / 2.0
            
            # Update
            delta = v_opt - h_i
            if abs(delta) > 1e-15:
                h[i] = v_opt
                c_vec += s * delta
        
        # Project to feasibility
        h = project_feasible(h, n)
        c_vec = np.correlate(h, 1 - h, mode='full')
        
        current_c = np.max(c_vec)
        if verbose:
            print(f"Sweep {sweep}: C5_unscaled = {current_c:.8f}, time = {time.time()-start_time:.2f}s")
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        else:
            # No improvement, check if we should continue
            h = best_h.copy()  # revert
            c_vec = np.correlate(h, 1 - h, mode='full')
    
    return h, best_c
