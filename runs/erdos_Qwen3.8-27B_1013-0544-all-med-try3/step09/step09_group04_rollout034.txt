    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_fitness(h, n, k=3, alpha=100):
    """Fitness: top-k lags sum with soft-max smoothing"""
    c = compute_c5_vec(h, n)
    sorted_c = np.sort(c)[::-1]
    top_k = sorted_c[:k]
    base = top_k.sum() / k
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        sm = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        sm = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(0.5 * base + 0.5 * sm)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

            best_h = h6
            best_n = n6
        if time.time() < deadline - 30:
            h6b, c6b = gradient_polish_multi(best_h, best_n, deadline - 15,
                                             alphas=[10000, 100000, 500000],
                                             mu=0.005)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Subgradient + pair search fine-tuning
    if time.time() < deadline - 20:
        cur_n = len(best_h)
        p7_end = min(time.time() + 15, deadline - 5)
        h7, c7 = subgrad_refine(best_h, cur_n, p7_end,
                                n_iter=min(5000, int((p7_end - time.time()) * 200)))
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    if time.time() < deadline - 3:
        h8, c8 = pair_search(best_h, len(best_h), deadline - 1, radius=50)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
