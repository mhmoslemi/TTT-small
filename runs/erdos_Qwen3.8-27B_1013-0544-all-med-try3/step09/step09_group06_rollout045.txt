    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lsei_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        w = np.exp(s - s.max())
    else:
        w = np.exp(s)
    return w / w.sum()

    c = np.correlate(h, 1.0 - h, mode='full') * dx

    active = np.where(w > 1e-10)[0]
    for k in active:

        grad += w[k] * _compute_lag_grad(h, n, lag)

    h_curr = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    t = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_threshold(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_b[idx] = 1.0
    return h_b

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def _lsei(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        return float(m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _lsei_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        w = np.exp(s - s.max())
    else:
        w = np.exp(s)
    return w / w.sum()

def _grad_lsei(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = _lsei_w(c, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        lag = k - (n - 1)
        grad += w[k] * _lag_grad(h, n, lag)
    return grad * dx

def nesterov_hg(h, n, deadline, alpha0, alpha1, max_steps):
    h_curr = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    t = 1.0
    stall = 0
    for it in range(max_steps):
        if time.time() > deadline:
            break
        frac = it / max(max_steps - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        g = _grad_lsei(y, n, alpha)
        mt = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = h_curr + (mt - 1.0) * (h_curr - y)
        lr = 0.02 / (alpha * (2.0 / n))
        h_new = project_feasible(y_new - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            h_curr = best_h.copy()
            y = best_h.copy()
            t = 1.0
            stall = 0
            alpha0 = max(alpha0 * 0.7, 100.0)
            alpha = alpha0
        t += 1.0
        y = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def _obj_top(x, n, idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full') * dx
    act = cc[idx]
    return _lsei(act, alpha)

def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return _lsei(c, alpha)

def _con_sum(x, n):
    return float(np.sum(x) - n / 2.0)

def _con_jac(x, n):
    return np.ones_like(x)

def slsqp_solve(h, n, alpha, maxiter, top_m):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_jac, 'args': (n,)}]
    if top_m is not None:
        c = compute_all_lags(h, n)
        top_idx = tuple(np.argsort(c)[-top_m:])
        try:
            res = minimize(_obj_top, h, args=(n, top_idx, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_obj_full, h, args=(n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

def local_search_multi(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    no_improve = 0
    for rnd in range(200):
        if time.time() > deadline:
            break
        top_lags = np.argsort(c_all)[-8:]
        improved = False
        for k_star in top_lags:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for d in [0.05, 0.02, 0.01, 0.005, 0.002]:
                if time.time() > deadline:
                    break
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-8:
                        delta = min(d, (h[i] - h[j]) * 0.3)
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                            c_all = compute_all_lags(h, n)
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-8:
                        delta = min(d, (h[j] - h[i]) * 0.3)
                        if delta < 1e-10:

                        h[i] += delta

def binary_swap_refine(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

        h[j] += delta

def swap_binary_search(h, n, deadline, rng):
    h_bin = binary_threshold(h, n)
    best_c = compute_c5(h_bin, n)
