But let me just go with the simpler O(n^2 * |candidates|) approach with |candidates| = 10-20. It's simpler to implement and likely fast enough.

Let me also think about a different approach: **simulated annealing / stochastic search**.

At each step:
1. Pick a random coordinate i
2. Propose a new value h[i] (e.g., random in [0,1] or a perturbation)
3. Accept if it improves C5 or with some probability (Metropolis)

This is simple but might be slow. However, with the incremental update, each step is O(n), so we can do many steps.

5. Coordinate descent with incremental updates at n=2000

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.correlate(h, 1 - h, mode='full').max() * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Softmax-smoothed objective and gradient
def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

# Active set version
def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    return grad

    best_c = c_vec.max()
    dx = 2.0 / n
    total_n = 2 * n - 1

    no_improve_count = 0

        sweep += 1
        sweep_improved = False

            best_local_c = best_c
            best_new_val = old_val

            # coef[mm] = change in c_vec[mm] per unit change in h[i]

            #              and as h[i-k]*(1-h[i]) if i-k >= 0
            #   if k < 0: let kk = -k, h[i] appears as h[i+kk]*(1-h[i]) if i+kk < n
            #              and as h[i]*(1-h[i+kk]) ... wait let me think more carefully

            #     contribution: (1 - h[i+k]) where k = mm-(n-1), need 0 <= i+k < n
            #   As the second factor: j+k = i, need j >= 0, i.e., i-k >= 0, i.e., i - (mm-(n-1)) >= 0
            #     contribution: -h[j] = -h[i-k] where k = mm-(n-1)
            
            # So coef[mm] = dx * [1_{valid}(1 - h[i+k]) - 1_{valid2} * h[i-k]]
            # where k = mm - (n-1)
            
            coef = np.zeros(total_n)
            k_range = np.arange(-n+1, n)

            valid1 = (k_range >= -i) & (k_range < n - i)
            # For h[i] as second factor: k such that 0 <= i-k < n, i.e., i-n < k <= i
            valid2 = (k_range > i - n) & (k_range <= i)
            
            k1 = k_range + i  # index of h[i+k]
            k2 = i - k_range  # index of h[i-k]
            
            coef[valid1] += dx * (1.0 - best_h[np.clip(k1[valid1], 0, n-1)])
            coef[valid2] += dx * (-best_h[np.clip(k2[valid2], 0, n-1)])
            
            # Special case: k=0 (mm = n-1), h[i] appears as h[i]*(1-h[i])
            # The coefficient should be (1 - 2*h[i]) * dx for the linear part
            # But we also have the -h[i]^2 term which makes it non-linear
            # For the linear approx: coef[n-1] should include (1-h[i]) + (-h[i]) = 1 - 2*h[i]
            # Let's check: valid1 includes k=0 (when -i <= 0 < n-i, i.e., i < n, always true)
            #   contribution: (1 - h[i+0]) = 1 - h[i]
            # valid2 includes k=0 (when i-n < 0 <= i, i.e., i >= 0, always true)
            #   contribution: -h[i-0] = -h[i]
            # Total: (1-h[i]) + (-h[i]) = 1 - 2*h[i]
            # But the actual contribution is h[i]*(1-h[i]) which is h[i] - h[i]^2
            # Linearization at h[i]: h[i] - h[i]^2 ≈ (1-2*h[i]) * (h[i] - h[i]) ... 
            # Actually, d/dh[i] [h[i] - h[i]^2] = 1 - 2*h[i]
            # So the linear coefficient is correct: 1 - 2*h[i]
            # The non-linear part is -h[i]^2 which we'll handle by checking the actual value

            for offset in offsets:
                new_val = old_val + offset
                if new_val < 0 or new_val > 1:
                    continue
                if abs(new_val - old_val) < 1e-10:
                    continue

                new_c = c_vec + delta * coef

                # c_0 changes by (new_val - old_val) - (new_val^2 - old_val^2)

                # = delta * (old_val - new_val)
                # = -delta^2
                # So new_c[n-1] should have -delta^2 added
                new_c[n-1] -= delta * delta
                
                local_max = new_c.max()
                if local_max < best_local_c - 1e-12:

                c_vec += delta * coef
                c_vec[n-1] -= delta * delta
                best_h[i] = best_new_val

            if no_improve_count >= 3:

# Upsampling
def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=None)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=3, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:

                                        'gtol': 1e-18, 'maxfun': maxit * 10,
                                        'maxls': 40, 'maxcor': 40})

                c_new = compute_c5(h_new, n)

                    best_h = h_new

            except Exception:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.correlate(h, 1 - h, mode='full').max() * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):

        try:

                                    'gtol': 1e-18, 'maxfun': maxit * 12,

def compute_coef(h, n, i):
    dx = 2.0 / n
    total_n = 2 * n - 1
    k_range = np.arange(-n + 1, n)
    valid1 = (k_range >= -i) & (k_range < n - i)
    valid2 = (k_range > i - n) & (k_range <= i)
    k1 = k_range + i
    k2 = i - k_range
    coef = np.zeros(total_n)
    idx1 = np.where(valid1)[0]
    idx2 = np.where(valid2)[0]
    coef[idx1] += dx * (1.0 - h[np.clip(k1[idx1], 0, n - 1)])
    coef[idx2] -= dx * h[np.clip(k2[idx2], 0, n - 1)]
    return coef

def coord_descent(h, n, deadline, n_cand=25):
    best_h = h.copy()
    c_vec = compute_c5_vec(h, n)
    best_c = float(c_vec.max())
    total_n = 2 * n - 1
    z_idx = n - 1
    rng = np.random.RandomState(123)

    coarse_deltas = np.concatenate([
        np.array([0.0]),
        np.unique(np.concatenate([
            np.linspace(-0.2, 0.2, 15),
            np.array([-0.5, -0.3, 0.3, -0.8, 0.8])
        ]))
    ])

    fine_deltas = np.concatenate([
        np.array([0.0]),
        np.unique(np.concatenate([
            np.linspace(-0.05, 0.05, 15),
            np.array([-0.1, 0.1, -0.2, 0.2])
        ]))
    ])

    sweep = 0
    no_improve = 0
    use_coarse = True

    while time.time() < deadline:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        deltas = coarse_deltas if use_coarse else fine_deltas

        for i in order:
            old_val = best_h[i]
            if old_val < 1e-12:
                deltas_use = deltas[deltas > 0]
            elif old_val > 1.0 - 1e-12:
                deltas_use = deltas[deltas < 0]
            else:
                deltas_use = deltas

            if len(deltas_use) <= 1:
                continue

            coef = compute_coef(best_h, n, i)
            base = c_vec.copy()

            valid_d = deltas_use[~np.isclose(deltas_use, 0.0, atol=1e-12)]
            if len(valid_d) == 0:
                continue

            new_vals = base[None, :] + valid_d[:, None] * coef[None, :]
            new_vals[:, z_idx] -= valid_d[:, None] ** 2

            local_max = new_vals.max(axis=1)
            best_idx = np.argmin(local_max)

            if local_max[best_idx] < best_c - 5e-10:
                delta = valid_d[best_idx]
                new_val = old_val + delta
                if 0.0 <= new_val <= 1.0:
                    c_vec = base + delta * coef
                    c_vec[z_idx] -= delta * delta
                    best_h[i] = new_val
                    if c_vec.max() < best_c:
                        best_c = float(c_vec.max())
                        sweep_improved = True

        if not sweep_improved:
            no_improve += 1
            if no_improve >= 2:
                if use_coarse:
                    use_coarse = False
                    no_improve = 0
                    continue
                else:
                    break
        else:
            no_improve = 0

    return best_h, best_c

                        best_local = lm
                        best_di, best_dj = a, b

            if abs(best_di) > 1e-12 or abs(best_dj) > 1e-12:
                test_h = best_h.copy()
                test_h[i] = vi + best_di
                test_h[j] = vj + best_dj
                test_h = project_feasible(test_h, n)
                tc = compute_c5(test_h, n)
                if tc < best_c - 1e-12:
                    best_h = test_h
                    best_c = tc
                    improved_any = True
        if not improved_any:
            break
    return best_h, best_c
