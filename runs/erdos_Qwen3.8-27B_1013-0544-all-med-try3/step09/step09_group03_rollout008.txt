def coord_descent(h, n, deadline, max_iters=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    dx = 2.0 / n
    center = n - 1
    top_k = min(10, 2 * n - 1)
    worst_indices = np.argpartition(c_vec, -top_k)[-top_k:]
    worst_c = c_vec[worst_indices].max()
    rng = np.random.RandomState(99)
    it = 0
    deltas = [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002, 0.0002, -0.0002]
    while it < max_iters:
        if time.time() > deadline:
            break
        j = rng.randint(n)
        h_val = best_h[j]
        improved = False
        for delta in deltas:
            nv = h_val + delta
            if nv < -1e-12 or nv > 1.0 + 1e-12:
                continue
            best_h[j] = nv
            # Quick check: update worst lags
            max_c = 0.0
            for idx in worst_indices:
                k = idx - center
                # Recompute c_vec[idx] directly
                if k >= 0:
                    nv_ = n - k
                    if nv_ > 0:
                        cv = np.dot(best_h[:nv_], 1.0 - best_h[k:]) * dx
                    else:
                        cv = 0.0
                else:
                    kk = -k
                    nv_ = n - kk
                    if nv_ > 0:
                        cv = np.dot(best_h[kk:], 1.0 - best_h[:nv_]) * dx
                    else:
                        cv = 0.0
                if cv > max_c:
                    max_c = cv
            if max_c < best_c - 1e-15:
                best_c = max_c
                # Update c_vec for worst indices
                for wi in worst_indices:
                    k = wi - center
                    if k >= 0:
                        nv_ = n - k
                        if nv_ > 0:
                            c_vec[wi] = np.dot(best_h[:nv_], 1.0 - best_h[k:]) * dx
                    else:
                        kk = -k
                        nv_ = n - kk
                        if nv_ > 0:
                            c_vec[wi] = np.dot(best_h[kk:], 1.0 - best_h[:nv_]) * dx
                improved = True
                break
            else:
                best_h[j] = h_val
        if not improved:
            best_h[j] = h_val
        it += 1
        if it % 5000 == 0:
            c_vec = compute_c5_vec(best_h, n)
            actual_c = c_vec.max()
            if actual_c < best_c:
                best_c = actual_c
            worst_indices = np.argpartition(c_vec, -top_k)[-top_k:]
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 25
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_base = 400

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: L-BFGS at base resolution (adaptive alpha chain)
    t1 = min(t0 + 180, deadline - 750)
    if t1 > time.time() + 15:
        best_h, best_c = run_lbfgs_chain(best_h, n_base, t1,
            alphas=(200, 500, 1000, 3000, 8000, 20000),
            mu_scale=0.05, maxiter_scale=1.0)

    # Phase 2: Perturbation search at base
    t2 = min(t0 + 280, deadline - 600)
    if t2 > time.time() + 10:
        best_h, best_c = perturbation_trials(best_h, n_base, t2, n_trials=5)

    # Phase 3: Quick SLSQP polish at base
    t3 = min(t0 + 340, deadline - 550)
    if t3 > time.time() + 10:
        h3, c3 = slqp_polish(best_h, n_base, t3, alpha=2000000, maxiter=3000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upscale to n=1000 and optimize
    n1 = 1000
    t4_end = min(t0 + 600, deadline - 350)
    if time.time() < t4_end - 30:
        h1000 = interpolate_to(best_h, n_base, n1)
        best_h, best_c = run_lbfgs_chain(h1000, n1, t4_end,
            alphas=(500, 2000, 8000, 30000, 80000),
            mu_scale=0.05, maxiter_scale=1.2)

    # Phase 5: Perturbation at n=1000
    t5 = min(t0 + 720, deadline - 250)
    if time.time() < t5 - 15:
        best_h, best_c = perturbation_trials(best_h, n1, t5, n_trials=4)

    # Phase 6: Upscale to n=2000 and optimize
    n2 = 2000
    t6_end = min(t0 + 950, deadline - 100)
    if time.time() < t6_end - 40:
        h2000 = interpolate_to(best_h, n1, n2)
        best_h, best_c = run_lbfgs_chain(h2000, n2, t6_end,
            alphas=(2000, 8000, 30000),
            mu_scale=0.05, maxiter_scale=1.5)

    # Phase 7: Coordinate descent at final resolution
    t7 = deadline - 5
    if time.time() < t7 - 10:
        best_h, best_c = coord_descent(best_h, n2, t7, max_iters=100000)

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
