import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (8000, 5000.0), (20000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def periodic_perturb(h, n, period, amp, rng):

    if period > n:

    t = np.arange(n)
    pert = amp * np.sin(2 * np.pi * t / period + rng.uniform(0, 2 * np.pi))
    h_new = h + pert
    return project_feasible(h_new, n)

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()

    else:

    return float(val)

            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def true_subgrad_step(h, n, top_m=8):
    """Compute a subgradient of the true max at the top shifts."""
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    # Use top top_m shifts with softmax weighting
    top_idx = np.argsort(-c_vec)[:top_m]
    top_vals = c_vec[top_idx]
    # Temperature for weighting
    temp = max(top_vals[0] - top_vals[-1], 1e-10)
    w_vals = np.exp(5.0 * (top_vals - c_max) / temp)
    w_vals /= w_vals.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for ii in range(top_m):
        idx = top_idx[ii]
        k = idx - center
        wm = w_vals[ii] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    # Remove mean to maintain sum constraint direction
    grad -= grad.mean()
    return grad, c_max

def subgradient_method(h, n, deadline, max_iter=100000):
    """Steepest descent on the true (non-smooth) max objective."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, c_cur = true_subgrad_step(best_h, n, top_m=8)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        found = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = best_h - s * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_h = h_new
                best_c = c_new
                found = True
                step = min(s * 1.2, 0.08)
                break
        if not found:
            step *= 0.7
            if step < 1e-9:
                break
    return best_h, best_c

def pair_local_search(h, n, deadline, n_rounds=500, rng=None):
    """Local search using pairwise adjustments (maintains sum constraint)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if rng is None:
        rng = np.random.RandomState(777)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        top5 = np.argsort(-c_vec)[:5]
        
        improved = False
        for trial in range(1000):
            if time.time() > deadline:
                return best_h, best_c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Choose delta to reduce the worst overlap
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.015, -0.015]:
                nv_i = best_h[i] + delta
                nv_j = best_h[j] - delta
                if nv_i < -1e-10 or nv_i > 1.0 + 1e-10:
                    continue
                if nv_j < -1e-10 or nv_j > 1.0 + 1e-10:
                    continue
                best_h[i] = nv_i
                best_h[j] = nv_j
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] -= delta
                    best_h[j] += delta
            if improved:
                break
        if not improved and rnd > 100:
            # Try targeted: focus on worst shift
            c_vec = compute_c5_vec(best_h, n)
            worst_idx = int(np.argmax(c_vec))
            k_w = worst_idx - (n - 1)
            # Find contributing points
            if k_w >= 0:
                nv = n - k_w
                contrib = best_h[0:nv] * (1 - best_h[k_w:n])
            else:
                kk = -k_w
                nv = n - kk
                contrib = best_h[kk:n] * (1 - best_h[0:nv])
            top_pts = np.argsort(-contrib)[:10]
            if k_w >= 0:
                top_pts = list(top_pts) + [p + k_w for p in top_pts if p + k_w < n]
            else:
                top_pts = list(top_pts + kk) + [p for p in top_pts]
            unique_pts = list(set(int(p) for p in top_pts if 0 <= p < n))
            for a in range(len(unique_pts)):
                for b in range(a + 1, len(unique_pts)):
                    if time.time() > deadline:
                        return best_h, best_c
                    ia, ib = unique_pts[a], unique_pts[b]
                    for delta in [0.008, -0.008, 0.004, -0.004, 0.012, -0.012]:
                        nv_a = best_h[ia] + delta
                        nv_b = best_h[ib] - delta
                        if not (0 <= nv_a <= 1 and 0 <= nv_b <= 1):
                            continue
                        best_h[ia] = nv_a
                        best_h[ib] = nv_b
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        best_h[ia] -= delta
                        best_h[ib] += delta
                    if improved:
                        break
                if improved:
                    break
            if not improved and rnd > 200:
                break
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    """Run L-BFGS with a sequence of (alpha, mu) configs."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (8000, 5000.0), (20000, 20000.0), (50000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxfun': 150000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def gen_perturbations(h, n, rng, n_perts=12):
    """Generate diverse perturbation starting points."""
    perts = []
    # Gaussian
    for mag in [0.04, 0.08, 0.12, 0.18]:
        perts.append(project_feasible(h + mag * rng.randn(n), n))
    # Periodic (odd harmonics to break symmetry)
    for freq in [3, 7, 15]:
        t = np.arange(n)
        perts.append(project_feasible(h + 0.08 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 6)), n))
    # Block
    for _ in range(2):
        hc = h.copy()
        bs = rng.randint(0, n - 60)
        bl = rng.randint(30, 120)
        hc[bs:bs + bl] += 0.06 * rng.randn(bl)
        perts.append(project_feasible(hc, n))
    return perts[:n_perts]

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=3000):
    """SLSQP for final polishing."""
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def micro_refine(h, n, deadline):
    """Very fine single-point and pair adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for _ in range(5000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst_idx = int(np.argmax(c_vec))
        k_w = worst_idx - (n - 1)
        dx = 2.0 / n
        
        # Find the most contributing (i) for this shift
        if k_w >= 0:
            nv = n - k_w
            terms = best_h[0:nv] * (1 - best_h[k_w:n])
            i_best = int(np.argmax(terms))
            candidates = [i_best, i_best + k_w]
        else:
            kk = -k_w
            nv = n - kk
            terms = best_h[kk:n] * (1 - best_h[0:nv])
            i_best = int(np.argmax(terms))
            candidates = [i_best + kk, i_best]
        
        improved = False
        for ci in candidates:
            if ci < 0 or ci >= n:
                continue
            # Try adjusting this point with all others (sampled)
            for cj in range(0, n, max(1, n // 50)):
                if cj == ci:
                    continue
                for delta in [0.002, -0.002, 0.001, -0.001, 0.003, -0.003]:
                    nv_ci = best_h[ci] + delta
                    nv_cj = best_h[cj] - delta
                    if nv_ci < -1e-10 or nv_ci > 1.0 + 1e-10:
                        continue
                    if nv_cj < -1e-10 or nv_cj > 1.0 + 1e-10:
                        continue
                    best_h[ci] = nv_ci
                    best_h[cj] = nv_cj
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                        break
                    best_h[ci] -= delta
                    best_h[cj] += delta
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try single-point adjustments (with re-projection)
            for ci in candidates:
                if ci < 0 or ci >= n:
                    continue
                for delta in [0.001, -0.001, 0.0005, -0.0005]:
                    best_h[ci] += delta
                    h_proj = project_feasible(best_h, n)
                    c = compute_c5(h_proj, n)
                    if c < best_c - 1e-16:
                        best_h = h_proj
                        best_c = c
                        improved = True
                        break
                    best_h[ci] -= delta
            if not improved:
                break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    h = project_feasible(h_init, n)
    c = compute_c5(h, n)
    
    # Phase 1: L-BFGS warmup (90s)
    t1 = min(t0 + 90, deadline - 850)
    if t1 > time.time() + 5:
        h, c = lbfgs_optimize(h, n, t1,
            configs=[(100, 50.0), (500, 200.0), (2000, 1000.0),
                     (8000, 5000.0), (20000, 20000.0)])
    
    # Phase 2: True subgradient refinement (150s)
    t2 = min(t0 + 240, deadline - 700)
    if t2 > time.time() + 5:
        h, c = subgradient_method(h, n, t2, max_iter=80000)
    
    # Phase 3: Multi-restart with L-BFGS (200s)
    t3 = min(t0 + 440, deadline - 500)
    if t3 > time.time() + 15:
        perts = gen_perturbations(h, n, rng, n_perts=12)
        for hp in perts:
            if time.time() > t3 - 15:
                break
            # Quick L-BFGS
            h_opt, c_opt = lbfgs_optimize(hp, n, time.time() + 14,
                configs=[(500, 200.0), (2000, 1000.0), (8000, 5000.0)])
            if c_opt < c:
                c = c_opt
                h = h_opt.copy()
            # Quick subgradient
            if time.time() < t3 - 15:
                h_opt2, c_opt2 = subgradient_method(h_opt, n, time.time() + 5, max_iter=5000)
                if c_opt2 < c:
                    c = c_opt2
                    h = h_opt2.copy()
    
    # Phase 4: SLSQP polish at base (60s)
    t4 = min(t0 + 500, deadline - 400)
    if t4 > time.time() + 5:
        h, c = slqp_polish(h, n, t4, alpha=3000000, maxiter=4000)
    
    # Phase 5: Pairwise local search at base (80s)
    t5 = min(t0 + 580, deadline - 320)
    if t5 > time.time() + 5:
        h, c = pair_local_search(h, n, t5, n_rounds=400, rng=rng)
    
    # Phase 6: Upsample to n=1000 and optimize (300s)
    n_up = 1000
    t6_start = max(time.time(), t0 + 590)
    if t6_start < deadline - 250:
        h_up = interpolate_to(h, n, n_up)
        c_up = compute_c5(h_up, n_up)
        t6a = min(t6_start + 160, deadline - 120)
        if t6a > time.time() + 10:
            h_up, c_up = lbfgs_optimize(h_up, n_up, t6a,
                configs=[(200, 50.0), (1000, 200.0), (3000, 1000.0),
                         (10000, 5000.0), (30000, 20000.0), (60000, 40000.0)])
        if c_up < c:
            c = c_up
            h = h_up.copy()
        # Subgradient at high res
        t6b = min(time.time() + 60, deadline - 60)
        if t6b > time.time() + 5 and len(h) == n_up:
            h, c = subgradient_method(h, n_up, t6b, max_iter=30000)
        # Pair search at high res
        t6c = min(time.time() + 50, deadline - 30)
        if t6c > time.time() + 5 and len(h) == n_up:
            h, c = pair_local_search(h, n_up, t6c, n_rounds=300, rng=rng)
    
    # Phase 7: SLSQP final polish (50s)
    if time.time() < deadline - 30:
        h, c = slqp_polish(h, len(h), deadline - 20, alpha=5000000, maxiter=2000)
    
    # Phase 8: Micro refinement
    if time.time() < deadline - 10:
        h, c = micro_refine(h, len(h), deadline - 5)
    
    # Final feasibility
    n_final = len(h)
    h = project_feasible(h, n_final)
    c = compute_c5(h, n_final)
    
    return h, c, n_final
