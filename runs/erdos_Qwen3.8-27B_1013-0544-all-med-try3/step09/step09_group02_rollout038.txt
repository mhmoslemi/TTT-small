from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_c5(h_bin, n):
    """Compute C5 for a binary sequence using the autocorrelation trick."""
    # For binary h: overlap[k] = dx * (n/2 - autocorr_lin[k])
    # where autocorr_lin[k] = sum_{i} h[i]*h[i+k]
    # We can compute this via the full correlate
    dx = 2.0 / n
    # correlate(h, h) gives autocorrelation
    # But we want correlate(h, 1-h) = correlate(h,1) - correlate(h,h)
    # correlate(h, 1) at shift k = sum of h over the overlapping window
    # This is just the running sum... 
    # Actually, let's just use the direct formula
    return float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)

def binary_to_continuous(h_bin, n, margin=0.05):
    """Convert binary to continuous with small margin for optimization."""
    h = h_bin.astype(float).copy()
    # Add small noise to break ties
    h[h == 1] = 1.0 - np.random.uniform(0, margin, size=(h==1).sum())
    h[h == 0] = np.random.uniform(0, margin, size=(h==0).sum())
    return project_feasible(h, n)

def simulated_annealing_binary(h_bin, n, deadline, max_sweeps=10000):
    """Simulated annealing on binary sequences."""
    best_h = h_bin.copy()
    best_c = binary_c5(h_bin, n)
    cur_h = h_bin.copy()
    cur_c = best_c
    
    n_ones = int(n / 2)
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    
    t_start = time.time()
    t_end = deadline
    total_time = t_end - t_start
    
    beta = 10.0  # cooling rate
    sweep = 0
    stuck_count = 0
    
    while time.time() < deadline and sweep < max_sweeps:
        frac = (time.time() - t_start) / total_time if total_time > 0 else 1.0
        temp = (1.0 - frac) ** beta * 0.1  # temperature schedule
        
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        
        n_trials = min(2000, len(ones) * len(zeros))
        improved = False
        
        for _ in range(n_trials):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            
            if i == j:

            # Make swap
            cur_h[i] = 0
            cur_h[j] = 1
            new_c = binary_c5(cur_h, n)
            
            delta = new_c - cur_c
            
            if delta < 0 or (temp > 0 and np.random.random() < np.exp(-delta / max(temp, 1e-10))):
                cur_c = new_c
                # Update lists
                ones = np.where(cur_h == 1)[0]
                zeros = np.where(cur_h == 0)[0]
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
                    stuck_count = 0
                improved = True
            else:
                cur_h[i] = 1
                cur_h[j] = 0
        
        sweep += 1
        if not improved:
            stuck_count += 1
            if stuck_count > 20:
                break
    
    return best_h, best_c

def hill_climbing_binary(h_bin, n, deadline, max_sweeps=50000):

    best_h = h_bin.copy()
    best_c = binary_c5(h_bin, n)
    cur_h = h_bin.copy()
    cur_c = best_c

        ones = np.where(cur_h == 1)[0]

        np.random.shuffle(ones)
        np.random.shuffle(zeros)

                return cur_h, best_c
            for j in zeros:
                # Try swap

                cur_h[j] = 1

                else:

def smoothed_obj(x, n, active_indices, alpha):
    """Smoothed max objective over active shifts only."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_active = c[active_indices]
    m = c_active.max()
    sh = alpha * (c_active - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smoothed_grad(x, n, active_indices, alpha, mu):
    """Gradient of smoothed objective."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_active = c[active_indices]
    m = c_active.max()
    sh = alpha * (c_active - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in active_indices:
        if w[idx - active_indices[0]] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx - active_indices[0]] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    
    # Add constraint gradient

    best_c = compute_c5(h_bin, n)

        frac = (time.time() - t_start) / total_time
        temp = max(0.001, (1 - frac) ** 3 * 0.02)

        zeros = np.where(cur_h == 0)[0]

            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            cur_h[i] = 0
            cur_h[j] = 1
            new_c = compute_c5(cur_h, n)
            cur_h[i] = 1
            cur_h[j] = 0
            delta = new_c - cur_c
            if delta < best_delta:
                best_delta = delta
                best_move = (i, j, new_c)
        
        if best_move is not None:

            # Accept if improvement or with probability exp(-delta/temp)

                cur_h[j] = 1
                cur_c = new_c
                if new_c < best_c:

            if tot < 1e-14:

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def smooth_max_obj(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_act = c[active]
    m = c_act.max()
    sh = alpha * (c_act - m)
    mx = sh.max()
    if mx > 300:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_act = c[active]
    m = c_act.max()
    sh = alpha * (c_act - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(len(active)):
        if w[ii] < 1e-14:
            continue
        idx = active[ii]
        k = idx - (n - 1)
        wm = w[ii] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])
