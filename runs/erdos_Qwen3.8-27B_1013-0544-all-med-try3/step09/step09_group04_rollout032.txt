from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def compute_autocorr(h, n):
    dx = 2.0 / n
    return np.correlate(h, h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def weighted_overlap_obj(x, n, w, mu):
    # Minimize Σ_k w_k * overlap_k(x) + mu * (sum(x) - n/2)^2
    # overlap_k(x) = correlate(x, 1-x, 'full')[k] * dx
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    val = np.dot(w, c)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def weighted_overlap_grad(x, n, w, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_fictitious_play(h_init, n, deadline, T=200, eta=0.5, inner_iters=50):

    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)

    w = np.ones(n_shifts) / n_shifts

                          bounds=bounds,

                                   'gtol': 1e-14, 'maxfun': inner_iters * 5})

        c_new = compute_c5(h_new, n)

    return best_h, best_c

    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution (original smooth max)
    p1_end = t0 + 60
    h1, c1 = run_lbfgs_smooth(best_h, best_n, p1_end,
        alphas=[500, 5000, 50000, 500000],
        mu=0.01, maxiters=[500, 800, 1200, 1500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Fictitious play at original resolution
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 150, deadline - 280)
        h2, c2 = run_fictitious_play(best_h, best_n, p2_end,
                                      T_max=300, eta0=1.5, inner_iters=60)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Autocorrelation reformulation at original resolution
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 80, deadline - 230)
        h3, c3 = run_lbfgs_acorr(best_h, best_n, p3_end,
                                  alphas=[1000, 10000, 100000, 1000000],
                                  mu=0.01, maxiters=[400, 600, 1000, 1200])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and optimize
    n_mid = 1000
    if time.time() < deadline - 350:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c + 0.005:
            p4_end = min(time.time() + 200, deadline - 300)
            h4, c4 = run_lbfgs_smooth(h_mid, n_mid, p4_end,
                alphas=[50, 500, 5000, 50000, 200000, 1000000],
                mu=0.005, maxiters=[500, 800, 1200, 1500, 2000, 2500])
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n_mid

    # Phase 5: Fictitious play at n=1000
    if time.time() < deadline - 250:
        p5_end = min(time.time() + 150, deadline - 200)
        h5, c5 = run_fictitious_play(best_h, best_n, p5_end,
                                      T_max=200, eta0=1.0, inner_iters=40)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active set optimization
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 100, deadline - 120)
        h6, c6 = run_active_opt(best_h, best_n, p6_end, top_k=5,
            alphas=[100000, 500000, 2000000, 10000000, 50000000],
            max_rounds=20)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Single constraint dive (focus on worst shift)
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 60, deadline - 70)
        h7, c7 = run_single_dive(best_h, best_n, p7_end, alpha=20000000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Try n=2000 upsample
    n_high = 2000
    if time.time() < deadline - 180 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.003:
            p8_end = min(time.time() + 120, deadline - 100)
            h8, c8 = run_lbfgs_smooth(h_high, n_high, p8_end,
                alphas=[20, 100, 500, 2000, 10000, 50000],
                mu=0.003, maxiters=[400, 600, 800, 1000, 1200, 1500])
            if c8 < best_c:
                best_c = c8
                best_h = h8
                best_n = n_high

    # Phase 9: Active set at final resolution
    if time.time() < deadline - 80:
        p9_end = min(time.time() + 50, deadline - 50)
        h9, c9 = run_active_opt(best_h, best_n, p9_end, top_k=4,
            alphas=[500000, 2000000, 10000000],
            max_rounds=15)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Gradient-guided local refinement
    if time.time() < deadline - 40:
        p10_end = min(time.time() + 30, deadline - 20)
        h10, c10 = grad_guided_local(best_h, best_n, p10_end, n_steps=30000)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 11: Final single-dive + active set polish
    if time.time() < deadline - 15:
        p11_end = deadline - 5
        h11, c11 = run_single_dive(best_h, best_n, p11_end, alpha=50000000)
        if c11 < best_c:
            best_c = c11
            best_h = h11

    # Absolute final: one more L-BFGS with very high alpha on worst constraint
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 50000000, 0.0005),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 100, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 400})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n

def run(seed=42, budget_s=1100, **kwargs):
    return running(seed=seed, budget_s=budget_s, **kwargs)
