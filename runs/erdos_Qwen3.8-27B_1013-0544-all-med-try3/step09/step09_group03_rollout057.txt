import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_push(h, n):
    """Convert to binary, maintaining sum constraint by adjusting boundary values."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = int(round(target - h_b.sum()))
    if diff == 0:
        return h_b
    if diff > 0:
        # Need more ones
        order = np.argsort(-h)
        for i in order[:diff]:
            if h_b[i] == 0.0:
                h_b[i] = 1.0
    else:
        order = np.argsort(h)
        for i in order[:-diff]:
            if h_b[i] == 1.0:
                h_b[i] = 0.0
    return h_b

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def local_search_binary(h, n, max_iter=5000, deadline=None):
    """Local search on binary pattern using swaps."""
    best_c = compute_c5(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        improved = False
        # Try random swaps
        n_swaps = min(50, len(ones) * len(zeros))
        rng = np.random.RandomState(it + 42)
        for _ in range(n_swaps):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                improved = True
                # Update sets
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            else:
                h[i], h[j] = h[j], h[i]  # revert
        if not improved:
            # Try systematic search over nearby swaps
            found = False
            for i in range(0, min(20, n)):
                if h[i] == 0 and deadline and time.time() > deadline:
                    break
                for j in range(max(0, i-20), min(n, i+20)):
                    if j != i and h[j] == 1:
                        h[i], h[j] = h[j], h[i]
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                            found = True
                            ones = np.where(h == 1.0)[0]
                            zeros = np.where(h == 0.0)[0]
                            i = -1  # restart
                            break
                        else:
                            h[i], h[j] = h[j], h[i]
                if found:
                    break
            if not found:
                break
    return h, best_c

def simulated_annealing_binary(h, n, t0=1.0, t_min=0.01, iterations=50000, deadline=None, seed=42):
    """Simulated annealing on binary pattern."""
    rng = np.random.RandomState(seed)
    current_c = compute_c5(h, n)
    best_c = current_c
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(iterations):
        if deadline and time.time() > deadline:
            break
        T = t0 * (t_min / t0) ** (it / max(iterations, 1))
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = new_c - current_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c

def make_periodic_pattern(n, P, ones_per_period):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    if ones_per_period > 0:
        # Place ones_per_period ones in each period
        # Spread them evenly
        positions = np.round(np.arange(ones_per_period) * P / ones_per_period).astype(int)
        for p in range(n // P):
            for pos in positions:
                idx = p * P + pos
                if idx < n:
                    h[idx] = 1.0
    # Adjust to get exactly n/2 ones
    target = n / 2
    diff = int(round(target - h.sum()))
    if diff > 0:
        order = np.argsort(-h)
        count = 0
        for i in range(n):
            if h[i] == 0 and count < diff:
                h[i] = 1.0
                count += 1
            elif h[i] == 1 and count < diff and i in positions and count < diff:
                # Try to fix by moving
                pass
    elif diff < 0:
        order = np.argsort(h)
        count = 0
        for i in range(n):
            if h[i] == 1 and count < -diff:
                h[i] = 0.0
                count += 1
    return h

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (400, 100.0), (1000, 400.0),
                   (3000, 1000.0), (8000, 5000.0), (20000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = int(n // 2)
    cur = int(h_b.sum())
    diff = target - cur
    if diff > 0:
        order = np.argsort(-h)
        cnt = 0
        for i in order:
            if h_b[i] == 0.0 and cnt < diff:
                h_b[i] = 1.0
                cnt += 1
    elif diff < 0:
        order = np.argsort(h)
        cnt = 0
        for i in order:
            if h_b[i] == 1.0 and cnt < -diff:
                h_b[i] = 0.0
                cnt += 1
    return h_b

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (800, 200.0), (2000, 1000.0),
                   (5000, 4000.0), (15000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 8:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=3000000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def local_search_binary(h, n, max_iter=10000, deadline=None, seed=42):
    best_c = compute_c5(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(seed)
    stall = 0
    kicks = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(20):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                stall = 0
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            else:
                h[i], h[j] = h[j], h[i]
        if improved:
            continue
        stall += 1
        if stall > 30 and kicks < 10:
            kicks += 1
            for _ in range(3 + kicks):
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            stall = 0
        elif stall > 60:
            break
    return h, best_c

def targeted_local_search(h, n, max_iter=5000, deadline=None, seed=42):
    best_c = compute_c5(h, n)
    cvec = compute_c5_vec(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(seed)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        worst_idx = int(np.argmax(cvec))
        k_worst = worst_idx - (n - 1)
        if abs(k_worst) > 0 and abs(k_worst) < n:
            if k_worst > 0:
                ii = np.where((h[0:n-k_worst] == 0) & (h[k_worst:n] == 1))[0]
            else:
                kk = -k_worst
                ii = np.where((h[kk:n] == 0) & (h[0:n-kk] == 1))[0]
            if len(ii) > 0:
                for _ in range(30):
                    idx = ii[rng.randint(len(ii))]
                    if k_worst > 0:
                        pos0 = idx
                        pos1 = idx + k_worst
                    else:
                        pos0 = idx + (-k_worst)
                        pos1 = idx
                    if rng.random() < 0.5 and len(zeros) > 1:
                        j2 = zeros[rng.randint(len(zeros))]
                        if j2 == pos1:
                            continue
                        h[pos0], h[j2] = h[j2], h[pos0]
                    else:
                        i2 = ones[rng.randint(len(ones))]
                        if i2 == pos0:
                            continue
                        h[pos1], h[i2] = h[i2], h[pos1]
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        stall = 0
                        improved = True
                        ones = np.where(h == 1.0)[0]
                        zeros = np.where(h == 0.0)[0]
                        cvec = compute_c5_vec(h, n)
                        break
                    else:
                        if rng.random() < 0.5:
                            h[pos0], h[j2] = h[j2], h[pos0]
                        else:
                            h[pos1], h[i2] = h[i2], h[pos1]
        if not improved:
            for _ in range(10):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    stall = 0
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    cvec = compute_c5_vec(h, n)
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            stall += 1
            if stall > 40:
                break
    return h, best_c

def simulated_annealing_binary(h, n, t0=3.0, t_min=0.001, iterations=100000, deadline=None, seed=42):
    rng = np.random.RandomState(seed)
    current_c = compute_c5(h, n)
    best_c = current_c
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    dx = 2.0 / n
    for it in range(iterations):
        if deadline is not None and time.time() > deadline:
            break
        T = t0 * (t_min / t0) ** (it / max(iterations, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = (new_c - current_c) / dx
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-min(delta / T, 500))):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_and_optimize(h, n, deadline, n_trials=6, seed=42):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    mags = [0.03, 0.06, 0.01, 0.05, 0.02, 0.08]
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        t_sub = min(time.time() + 20, deadline - 10)
        if t_sub > time.time() + 3:
            try:
                h_opt, c_opt = run_lbfgs(h_p, n, t_sub,
                    configs=[(1000, 400.0), (5000, 4000.0), (15000, 15000.0)])
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
        h_bin = binary_push(h_p, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n0 = 500

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: L-BFGS at base resolution (~90s)
    t1 = min(t0 + 90, deadline - 400)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n0, t1,
            configs=[(200, 50.0), (800, 200.0), (2000, 1000.0),
                     (5000, 4000.0), (15000, 15000.0)])

    # Phase 2: SLSQP polish (~30s)
    if time.time() < deadline - 300:
        h_s, c_s = slqp_polish(best_h, n0, min(time.time() + 30, deadline - 280),
                               alpha=3000000, maxiter=2000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 3: Binary local search at base (~50s)
    hb = binary_push(best_h, n0)
    cb = compute_c5(hb, n0)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < deadline - 200:
        h_ls, c_ls = local_search_binary(best_h.copy(), n0, max_iter=15000,
                                          deadline=min(time.time() + 40, deadline - 190),
                                          seed=seed)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()

    # Phase 4: Targeted local search (~30s)
    if time.time() < deadline - 160:
        h_ts, c_ts = targeted_local_search(best_h.copy(), n0, max_iter=8000,
                                            deadline=min(time.time() + 25, deadline - 150),
                                            seed=seed + 100)
        if c_ts < best_c:
            best_c = c_ts
            best_h = h_ts.copy()

    # Phase 5: SA at base (~50s)
    if time.time() < deadline - 130:
        h_sa, c_sa = simulated_annealing_binary(binary_push(best_h, n0), n0,
                                                 t0=4.0, t_min=0.002, iterations=150000,
                                                 deadline=min(time.time() + 45, deadline - 120),
                                                 seed=seed + 200)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        h_ls2, c_ls2 = local_search_binary(best_h.copy(), n0, max_iter=5000,
                                            deadline=min(time.time() + 15, deadline - 100),
                                            seed=seed + 300)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2.copy()

    # Phase 6: Perturbation + optimize at base (~40s)
    if time.time() < deadline - 100:
        best_h, best_c = perturb_and_optimize(best_h, n0,
                                               min(time.time() + 40, deadline - 90),
                                               n_trials=5, seed=seed + 400)

    # Phase 7: Upsample to n=1000, continuous opt (~100s)
    n1 = 1000
    if time.time() < deadline - 180:
        h1 = interpolate_to(best_h, n0, n1)
        t7 = min(time.time() + 80, deadline - 100)
        if t7 > time.time() + 15:
            best_h, best_c = run_lbfgs(h1, n1, t7,
                configs=[(500, 200.0), (2000, 1000.0), (5000, 5000.0), (15000, 20000.0)])
        hb1 = binary_push(best_h, n1)
        cb1 = compute_c5(hb1, n1)
        if cb1 < best_c:
            best_c = cb1
            best_h = hb1

    # Phase 8: Binary local search at n=1000 (~60s)
    if time.time() < deadline - 100:
        hb = binary_push(best_h, n1)
        h_ls, c_ls = local_search_binary(hb, n1, max_iter=12000,
                                          deadline=min(time.time() + 50, deadline - 90),
                                          seed=seed + 500)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()

    # Phase 9: Targeted search + SA at n=1000 (~50s)
    if time.time() < deadline - 60:
        h_ts, c_ts = targeted_local_search(best_h.copy(), n1, max_iter=6000,
                                            deadline=min(time.time() + 20, deadline - 50),
                                            seed=seed + 600)
        if c_ts < best_c:
            best_c = c_ts
            best_h = h_ts.copy()
        if time.time() < deadline - 30:
            h_sa, c_sa = simulated_annealing_binary(binary_push(best_h, n1), n1,
                                                     t0=2.0, t_min=0.002, iterations=80000,
                                                     deadline=min(time.time() + 25, deadline - 25),
                                                     seed=seed + 700)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
            h_ls2, c_ls2 = local_search_binary(best_h.copy(), n1, max_iter=4000,
                                                deadline=min(time.time() + 10, deadline - 20),
                                                seed=seed + 800)
            if c_ls2 < best_c:
                best_c = c_ls2
                best_h = h_ls2.copy()

    # Phase 10: Upsample to n=2000 (~80s)
    n2 = 2000
    if time.time() < deadline - 80:
        h2 = interpolate_to(best_h, n1, n2)
        t10 = min(time.time() + 50, deadline - 30)
        if t10 > time.time() + 10:
            best_h, best_c = run_lbfgs(h2, n2, t10,
                configs=[(1000, 500.0), (5000, 5000.0), (10000, 20000.0)])
        hb2 = binary_push(best_h, n2)
        cb2 = compute_c5(hb2, n2)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2

    # Phase 11: Final binary local search (~30s)
    if time.time() < deadline - 15:
        nf = len(best_h)
        hbf = binary_push(best_h, nf)
        h_fs, c_fs = local_search_binary(hbf, nf, max_iter=8000,
                                          deadline=min(time.time() + 25, deadline - 5),
                                          seed=seed + 900)
        if c_fs < best_c:
            best_c = c_fs
            best_h = h_fs.copy()

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
