import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n * 0.5) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-13:
            continue
        k = mm - (n - 1)
        wd = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wd * (1.0 - x[k:])
            grad[k:] += wd * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wd * (1.0 - x[:nv])
            grad[:nv] += wd * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n * 0.5)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxits=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bnds = [(0.0, 1.0)] * n
    for i, a in enumerate(alphas):
        rem = deadline - time.time()
        if rem < 8:
            break
        mid = maxits[i] if maxits is not None and i < len(maxits) else max(500, int(rem * 4))
        try:
            r = minimize(sm_obj, best_h, jac=sm_grad, args=(n, a, mu),
                        method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': mid, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxfun': mid * 12, 'maxls': 100, 'maxcor': 60})
            hn = project_feasible(r.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-14:
                best_c, best_h = cn, hn
            elif cn <= best_c + 1e-11:
                best_h = hn
        except Exception:
            pass
    return best_h, best_c

def upsample(h, no, nn):
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nn)

def grad_local_search(h, n, deadline, alpha=1e6):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    step = 0.02
    dx = 2.0 / n
    for it in range(200000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        w = np.exp(s - mx)
        w /= w.sum()
        g = np.zeros(n)
        for mm in range(2 * n - 1):
            wm = w[mm]
            if wm < 1e-13:
                continue

            if k >= 0:

                g[kk:] += wd * (1.0 - best_h[:nv])
                g[:nv] += wd * (-best_h[kk:])
        h_new = best_h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new
            step = min(step * 1.02, 0.05)
        else:

def coord_descent(h, n, deadline, alpha=1e5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for epoch in range(200):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        w = np.exp(s - mx)
        w /= w.sum()
        for i in range(n):
            if time.time() > deadline:
                break
            # Compute gradient component for h[i]
            gi = 0.0
            for mm in range(2 * n - 1):
                wm = w[mm]
                if wm < 1e-13:
                    continue
                k = mm - (n - 1)
                if k >= 0:
                    if i < n - k:
                        gi += wm * dx * (1.0 - best_h[i + k])
                    if i >= k:
                        gi += wm * dx * (-best_h[i - k])
                else:

                        gi += wm * dx * (1.0 - best_h[i - kk])

            # Binary search for optimal h[i]

                h_val = max(0.0, best_h[i] - 0.01)

                continue
            best_h[i] = h_val
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                best_h[i] = h_val  # keep (projection may have changed other values)
    return best_h, best_c
