def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)

    w = np.exp(shifted - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)

    threshold = 1e-12
    active = w > threshold
    
    for mm in range(2 * n - 1):
        if not active[mm]:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pair_search(h0, n, deadline, radius=50):

            maxit = max(400, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_set_obj, best_h, jac=active_set_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 80, 'maxcor': 80})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def fine_pair_search(h0, n, deadline, radius=40):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            step = max(1, n // 500)
            for i in range(0, n, step):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, step):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h0, n, deadline, n_iter=5000):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.01)
        else:
            eta *= 0.99
        best_h = h_new
    return best_h, best_c

def perturb_and_optimize(h0, n, rng, scale, deadline):
    h_pert = h0 + scale * rng.randn(n)
    h_pert = project_feasible(h_pert, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h_pert, jac=smooth_max_grad,
                      args=(n, 500000, 0.0), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 800, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h_pert, compute_c5(h_pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Work at n=1000 for better representation
    n_work = 1000
    if n_orig != n_work:
        x_fine = np.linspace(0, 2, n_work, endpoint=False)
        x_coarse = np.linspace(0, 2, n_orig, endpoint=False)
        h_work = np.interp(x_fine, x_coarse, h0)
        h_work = project_feasible(h_work, n_work)
    else:
        h_work = h0.copy()
    
    # Phase 1: Moderate alpha L-BFGS (establish basin)
    p1_end = min(t0 + 180, deadline - 500)
    if time.time() < p1_end - 30:
        h_work, c_work = run_lbfgs(h_work, n_work, p1_end,
            alphas=[50, 200, 1000, 5000, 20000, 50000],
            mu=0.05, maxiters=[1500, 2000, 2500, 3000, 3000, 3000])
        
        # Also try at original resolution
        h_orig_opt, c_orig_opt = run_lbfgs(h0, n_orig, min(p1_end, time.time() + 60),
            alphas=[100, 1000, 10000],
            mu=0.05, maxiters=[2000, 2000, 2000])
        if c_orig_opt < best_c:
            best_c = c_orig_opt
            best_h = h_orig_opt.copy()
            best_n = n_orig
    
    # Phase 2: High alpha L-BFGS at n_work
    p2_end = min(t0 + 500, deadline - 350)
    if time.time() < p2_end - 30:
        h_work, c_work = run_lbfgs(h_work, n_work, p2_end,
            alphas=[100000, 300000, 500000, 1000000, 3000000, 5000000],
            mu=0.005, maxiters=[3000, 3500, 4000, 4000, 5000, 5000])
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n_work
    
    # Phase 3: Active-set refinement at best resolution
    p3_end = min(t0 + 780, deadline - 200)
    if time.time() < p3_end - 30:
        h_work, c_work = run_active_set(best_h, best_n, p3_end,
            top_k=5, alphas=[100000, 500000, 2000000, 10000000,
                            50000000, 200000000, 1000000000])
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
            best_n = best_n
    
    # Phase 4: Subgradient refinement
    p4_end = min(t0 + 880, deadline - 100)
    if time.time() < p4_end - 20:
        h_sub, c_sub = subgrad_refine(best_h, best_n, p4_end, n_iter=4000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub.copy()
    
    # Phase 5: Perturbation restarts
    if time.time() < deadline - 60:
        for scale in [0.005, 0.002, 0.001]:
            if time.time() > deadline - 35:
                break
            h_p, c_p = perturb_and_optimize(best_h, best_n, rng, scale, deadline - 35)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()
    
    # Phase 6: Fine pair search polish
    p6_end = deadline - 5
    if time.time() < p6_end:
        h_fs, c_fs = fine_pair_search(best_h, best_n, p6_end, radius=25)
        if c_fs < best_c:
            best_c = c_fs
            best_h = h_fs.copy()
    
    # Final: ensure feasibility
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
