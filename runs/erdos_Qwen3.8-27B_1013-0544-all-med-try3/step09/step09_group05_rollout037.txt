    """Find optimal h[i] given all other values fixed.
    c_vec: current correlation values
    s: slopes s_k for each shift k
    b: intercepts b_k = c_k - h[i]*s_k
    """
    # f(t) = max_k (b_k + t * s_k)
    # We want to minimize f(t) over t in [0,1]
    
    # Quick check: evaluate at endpoints
    f0 = np.max(b)  # f(0) = max b_k
    f1 = np.max(b + s)  # f(1) = max (b_k + s_k)
    
    # If slopes are all positive, minimum is at t=0
    if s.min() >= 0:
        return 0.0
    # If slopes are all negative, minimum is at t=1
    if s.max() <= 0:
        return 1.0
    
    # Otherwise, find the breakpoint
    # The upper envelope of lines has a unique minimum
    # where the slope changes from negative to positive
    # Use bisection on the derivative
    
    def f_at(t):
        return np.max(b + t * s)
    
    def slope_at(t):
        # The slope of the upper envelope at t is the s_k of the line
        # that achieves the max at t
        vals = b + t * s
        max_val = vals.max()
        # Find all lines achieving the max
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        # The slope is between min and max of active s_k
        return s[active].min(), s[active].max()
    
    # Bisection on the zero of the "subdifferential"
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_lo, s_hi = slope_at(mid)
        if s_lo > 0:
            hi = mid  # All active slopes positive, minimum is to the left
        elif s_hi < 0:
            lo = mid  # All active slopes negative, minimum is to the right
        else:
            # The subdifferential contains 0, so mid is optimal
            lo = hi = mid
            break
    
    return (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        wm = w[mm] * dx
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=1.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(obj_smooth, h, jac=grad_smooth,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-12, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descend(h, n, max_passes=500, time_budget=120.0, recompute_every=20):
    """Exact coordinate descent exploiting piecewise-linear structure of the max."""
    t0 = time.time()
    c_vec = compute_c5_vec(h, n)
    n_shifts = 2 * n - 1
    best_c = c_vec.max()
    m_idx = np.arange(n_shifts)
    
    for pass_i in range(max_passes):
        if time.time() - t0 > time_budget:
            break
        
        any_improve = False
        for i in range(n):
            # Compute slopes s[m] for all shifts m
            # s[m] = (1-h[i-m]) * [0<=i-m<n] - h[i+m] * [0<=i+m<n]
            # where m is the correlation index (0 to 2n-2)
            i_m = i - m_idx
            i_p = i + m_idx
            
            s = np.zeros(n_shifts)
            mask_n = (i_m >= 0) & (i_m < n)
            mask_p = (i_p >= 0) & (i_p < n)
            s[mask_n] = 1.0 - h[i_m[mask_n]]
            s[mask_p] -= h[i_p[mask_p]]
            
            # Minimize f(t) = max(c_vec + t*s) over t in [-h[i], 1-h[i]]
            lo = -h[i]
            hi = 1.0 - h[i]
            
            if hi - lo < 1e-14:
                continue
            
            # Quick checks
            if s.min() > 1e-13:
                best_t = lo
            elif s.max() < -1e-13:
                best_t = hi
            else:
                # Bisection on subdifferential
                a, b = lo, hi
                for _ in range(55):
                    mid = (a + b) * 0.5
                    vals = c_vec + mid * s
                    mx = vals.max()
                    # Find subdifferential at mid
                    diff = mx - vals
                    tol = mx * 1e-11 + 1e-16
                    active = diff <= tol
                    if not active.any():
                        active = (vals == mx)
                    sm = s[active].min()
                    sx = s[active].max()
                    if sm > 1e-13:
                        b = mid
                    elif sx < -1e-13:
                        a = mid
                    else:
                        a = b = mid
                        break
                best_t = (a + b) * 0.5
            
            old_val = h[i]
            new_val = old_val + best_t
            if new_val < 0.0:
                new_val = 0.0
            elif new_val > 1.0:
                new_val = 1.0
            delta = new_val - old_val
            
            if abs(delta) > 1e-15:
                h[i] = new_val
                c_vec += delta * s
                cur_max = c_vec.max()
                if cur_max < best_c:
                    best_c = cur_max
                    any_improve = True
        
        # Periodically recompute to correct accumulated error
        if (pass_i + 1) % recompute_every == 0:
            c_vec = compute_c5_vec(h, n)
            cur_max = c_vec.max()
            if cur_max < best_c:
                best_c = cur_max
        
        if not any_improve and pass_i > 5:
            # Check if we're truly at a local min
            c_check = compute_c5_vec(h, n)
            if c_check.max() >= best_c - 1e-14:
                if pass_i > 20:
                    break
    
    # Final projection
    h = project_feasible(h, n)
    c_vec = compute_c5_vec(h, n)
    return h, c_vec.max()

def multi_start_coord_descent(h, n, num_starts=5, time_budget=60.0):
    """Try random perturbations and coordinate descent from each."""
    t0 = time.time()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for start_i in range(num_starts):

        n_pert = rng.randint(10, 50)

        if c_new < best_c:

            best_h = h_new
