        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        result = np.clip(res.x, 0, 1)
    except Exception:
        result = h.copy()
    return project_feasible(result, n)

def subgradient_method(h, n, deadline, target, max_iters=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = target if target is not None else best_c
    dx = 2.0 / n
    t_max = 0.15
    avg_list = []
    N = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1 - h, mode='full')
        c_scaled = c_all * dx
        F_val = c_scaled.max()
        
        if F_val < best_c - 1e-15:
            best_c = F_val
            best_h = h.copy()
        if F_val < T - 1e-8:
            T = F_val + 0.0001

        # Compute subgradient (weighted by activity)
        m = c_scaled.max()
        active = c_scaled > m - 0.002
        if not active.any():
            active = c_scaled > m - 0.01
        n_active = active.sum()
        if n_active == 0:
            active = c_scaled >= m - 0.005
        
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            lag = k - (n - 1)
            weight = 1.0 / n_active
            if lag == 0:
                grad += weight * dx * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += weight * dx * (1.0 - h[0:n - lag])
                grad[0:n - lag] += weight * dx * (-h[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += weight * dx * (1.0 - h[idx:n])
                grad[idx:n] += weight * dx * (-h[0:n - idx])

        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-20:
            break

        gap = F_val - T
        if gap < 1e-8:
            t = t_max / (1.0 + 0.01 * it)
        else:
            t = min(t_max, gap / (g_norm_sq * 1.5))
            t = max(t, 1e-6)

        h_new = h - t * grad
        h_new = project_feasible(h_new, n)
        
        N += 1
        if len(avg_list) < 50:
            avg_list.append(h_new.copy())
        else:
            avg_list.pop(0)
        avg_h = np.mean(avg_list, axis=0)
        avg_h = project_feasible(avg_h, n)
        avg_c = compute_c5(avg_h, n)
        if avg_c < best_c - 1e-14:
            best_c = avg_c
            best_h = avg_h.copy()

        h = h_new

    return best_h, best_c

def active_lag_coordinate_descent(h, n, deadline, top_k=8, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.04, 0.01, 0.003, 0.001, 0.0003, 0.0001]
    
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    total_iters = 0

    while time.time() < deadline:
        c_all = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False
        total_improves = 0

        for step in step_sizes:
            if time.time() > deadline:
                break
            rounds = 0
            for _ in range(500):
                if time.time() > deadline:
                    break
                if rounds > 50:
                    break
                k_star = top_idx[np.random.randint(top_k)]
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                num_pairs = n - m
                i = np.random.randint(0, num_pairs)
                j = i + m
                if h[i] > h[j] + 1e-8:
                    d = min(step, (h[i] - h[j]) * 0.3)
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = np.correlate(h, 1 - h, mode='full') * (2.0/n)
                    if c_test.max() < best_c - 1e-14:
                        best_c = float(c_test.max())
                        best_h = h.copy()
                        improved = True
                        rounds += 1
                        c_all = c_test
                        top_idx = np.argsort(c_all)[-top_k:]
                    else:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    d = min(step, (h[j] - h[i]) * 0.3)
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_test = np.correlate(h, 1 - h, mode='full') * (2.0/n)
                    if c_test.max() < best_c - 1e-14:
                        best_c = float(c_test.max())
                        best_h = h.copy()
                        improved = True
                        rounds += 1
                        c_all = c_test
                        top_idx = np.argsort(c_all)[-top_k:]
                    else:
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                total_iters += 1

        if not improved:
            break

    return best_h, best_c

def geometric_staircase_init(n, rng):
    """Create structured initial guesses."""
    patterns = []
    # Pattern 1: blocks
    block_sizes = [n//5, n//4, n//3]
    for bs in block_sizes:
        h = np.zeros(n)
        h[:bs] = 1.0
        h = project_feasible(h, n)
        patterns.append(h)
    # Pattern 2: periodic
    for period in [3, 4, 5, 7, 8]:
        h = np.zeros(n)
        h[0::period] = 1.0
        h[1::period] = 0.5
        h = project_feasible(h, n)
        patterns.append(h)
    # Pattern 3: cosine
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3]:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x / 2)
        h = project_feasible(h, n)
        patterns.append(h)
    # Pattern 4: random
    for _ in range(3):
        h = rng.uniform(0, 1, n)
        h = project_feasible(h, n)
        patterns.append(h)
    return patterns

def two_scale_optimize(h_coarse, n_coarse, n_fine, alpha, deadline, maxiter=1000):
    """Optimize at fine resolution starting from coarse solution interpolated."""
    x_c = np.linspace(0, 2, n_coarse + 1)[:-1]
    x_f = np.linspace(0, 2, n_fine + 1)[:-1]
    h_init = np.interp(x_f, x_c, h_coarse)
    h_init = project_feasible(h_init, n_fine)
    h_opt = lbfgsb_optimize(h_init, n_fine, alpha, deadline, maxiter=maxiter)
    return h_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n = len(h_parent)
    except Exception:
        h_parent = None
        n = 500

    best_h = None
    best_c = float('inf')

    if h_parent is not None:
        best_h = project_feasible(h_parent, n)
        best_c = compute_c5(best_h, n)
    else:
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with adaptive alpha (aggressive schedule)
    alpha_schedule = [5000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    maxiter_schedule = [500, 800, 1000, 1200, 1500, 1500, 1500]
    
    for alpha, maxit in zip(alpha_schedule, maxiter_schedule):
        if time.time() > deadline - 350:
            break
        remaining = deadline - 350 - time.time()
        if remaining < 10:
            break
        t_alloc = min(remaining, 45)
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + t_alloc, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient method with Polyak step
    if time.time() < deadline - 300:
        sg_deadline = min(deadline - 250, time.time() + 150)
        h_opt, c_opt = subgradient_method(best_h, n, sg_deadline, target=best_c - 0.001)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: L-BFGS-B polish after subgradient
    if time.time() < deadline - 200:
        for alpha in [50000000, 200000000]:
            if time.time() > deadline - 100:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 30, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Active lag coordinate descent
    if time.time() < deadline - 150:
        cd_deadline = min(deadline - 100, time.time() + 120)
        h_opt, c_opt = active_lag_coordinate_descent(best_h, n, cd_deadline, top_k=10,
                                                     step_sizes=[0.02, 0.005, 0.001, 0.0003, 0.0001, 0.00003])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Try higher resolution (n=1000)
    if time.time() < deadline - 100:
        n_fine = 1000
        x_c = np.linspace(0, 2, n + 1)[:-1]
        x_f = np.linspace(0, 2, n_fine + 1)[:-1]
        h_fine = np.interp(x_f, x_c, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        
        for alpha in [100000, 2000000, 50000000]:
            if time.time() > deadline - 50:
                break
            t_alloc = min(deadline - time.time() - 30, 40)
            if t_alloc < 5:
                break
            h_opt = lbfgsb_optimize(h_fine, n_fine, alpha, time.time() + t_alloc, maxiter=1500)
            c_opt = compute_c5(h_opt, n_fine)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_fine

    # Phase 6: Multi-restart with perturbations
    if time.time() < deadline - 80:
        for restart in range(5):
            if time.time() > deadline - 40:
                break
            h_pert = best_h + rng.normal(0, 0.02, n)
            h_pert = project_feasible(h_pert, n)
            
            # Quick L-BFGS-B
            t_alloc = min(deadline - time.time() - 30, 25)
            if t_alloc < 5:
                break
            h_opt = lbfgsb_optimize(h_pert, n, 5000000, time.time() + t_alloc, maxiter=1000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Subgradient from best, targeting slightly below
    if time.time() < deadline - 40:
        target = best_c - 0.0005
        sg_deadline = min(deadline - 20, time.time() + 50)
        h_opt, c_opt = subgradient_method(best_h, n, sg_deadline, target=target)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        h_opt = lbfgsb_optimize(best_h, n, 500000000, deadline - 5, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final coordinate descent polish
    if time.time() < deadline - 3:
        h_opt, c_opt = active_lag_coordinate_descent(best_h, n, deadline - 1, top_k=5,
                                                     step_sizes=[0.001, 0.0003, 0.0001, 0.00003])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(500, 0.5), 500)
        best_c = compute_c5(best_h, 500)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
