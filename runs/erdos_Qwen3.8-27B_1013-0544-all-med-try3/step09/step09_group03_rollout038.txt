def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            if tot < 1e-14:

**Local search design:**

The key idea is to try swaps (or more generally, paired adjustments) and accept them if they reduce the max overlap.

For a binary solution, a swap means: pick i where h[i]=1 and j where h[j]=0, flip both.

For a non-binary solution, we can do: increase h[i] by δ and decrease h[j] by δ (or use the projections).

**Efficient evaluation of swap impact:**

When we flip h[i] and h[j], the correlation changes only at shifts that involve i or j. Specifically:
- Shift k is affected if i or j is in the "valid range" for that shift.
- For shift +m: affected if i or j is in [0, n-1-m] (as the "second" index) or in [m, n-1] (as the "first" index).
- Essentially, almost all shifts are affected (except maybe |k| very close to n).

So we can't easily check only a few shifts. We need to recompute the full correlation or at least the affected shifts.

But for n ≤ 3000, recomputing the full correlation takes O(n) time (using FFT or direct computation). np.correlate is O(n²) for direct computation but NumPy might use FFT internally... actually, I think np.correlate uses direct computation for 1D, which is O(n²). So for local search at n=3000:
- Each swap evaluation: ~10ms (recompute correlation)
- We can do ~1000 swaps per second
- In 300 seconds: ~300,000 swaps

But wait, we need to be smarter. A random swap is unlikely to help. We should:
1. Identify the worst shift k*
2. Find swaps that specifically reduce r(k*)
3. Check that they don't increase other shifts too much

For shift k*, r(k*) = Σ_{j=0}^{n-1-k*} h[j]*(1-h[j+k*])

To reduce r(k*), we want to find j where h[j] is large and h[j+k*] is small (a "hot spot"), and either:
- Decrease h[j] (flip 1→0 or reduce value)
- Increase h[j+k*] (flip 0→1 or increase value)

But we need to maintain the sum constraint, so we pair with a compensating change.

    2. If not improved in last X iterations, use simulated annealing acceptance
    3. Find the "hot spots" for k*: positions j where h[j] is large and h[j+k*] is small

    5. Evaluate the new max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha, mu):
    """Combined objective and gradient computation."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_val = sh.max()
    w = np.exp(sh - mx_val)
    w /= w.sum()
    
    if mx_val > 500:
        val = m + (mx_val + np.log(np.sum(np.exp(sh - mx_val)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    
    # Gradient
    dx = 2.0 / n
    g = np.zeros(n)
    g += w[n-1] * dx * (1.0 - 2.0 * x)
    for m_ in range(1, n):
        wp = w[n-1+m_] * dx
        wn = w[n-1-m_] * dx
        if wp > 1e-14:
            g[:n-m_] += -wp * x[m_:]
            g[m_:] += wp * (1.0 - x[:n-m_])
        if wn > 1e-14:
            g[:n-m_] += wn * (1.0 - x[m_:])
            g[m_:] += -wn * x[:n-m_]
    g += 2.0 * mu * (x.sum() - n / 2.0)
    
    return val, g

def lbfgs_optimize(h, n, deadline, alphas=(1000, 5000, 20000, 100000, 500000), mu=1000.0):
    """Run L-BFGS-B with increasing alpha for the smooth max."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(
                lambda x: obj_and_grad(x, n, alpha, mu)[0],
                h, 
                jac=lambda x: obj_and_grad(x, n, alpha, mu)[1],
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 50000}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
