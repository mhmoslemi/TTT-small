        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()

        w = np.exp(shifted - mx) if mx > 500 else np.exp(shifted)

        w_top = w_top / w_top.sum()
        w = np.zeros_like(c)

        h = h_new
        
        # Decay learning rate
        if t % 100 == 0:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_grad_indices(n):
    i = np.arange(n)[:, None].astype(np.int32)
    u = np.arange(n)[None, :].astype(np.int32)
    w_idx = (i - u + (n - 1)).astype(np.int32)
    v = np.arange(n)[None, :].astype(np.int32)
    w_idx2 = (v - i + (n - 1)).astype(np.int32)
    mask = (v >= i).astype(np.float64)
    return w_idx, w_idx2, mask

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    return m + logsum / alpha

    mx = shifted.max()
    if mx > 500:

    else:

    val = m + np.log(w_top.sum() * np.exp(mx)) / alpha  # hmm this isn't right

    W_mat = w[w_idx]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_grad_indices(n):
    i = np.arange(n)[:, None].astype(np.int32)
    u = np.arange(n)[None, :].astype(np.int32)
    w_idx = (i - u + (n - 1)).astype(np.int32)
    v = np.arange(n)[None, :].astype(np.int32)
    w_idx2 = (v - i + (n - 1)).astype(np.int32)
    mask = (v >= i).astype(np.float64)
    return w_idx, w_idx2, mask

def smooth_max_val_w(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
        w = np.exp(shifted - mx)
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
        w = np.exp(shifted)
    w = w / w.sum()
    return m + logsum / alpha, w

def tophk_sm(c, alpha, k):
    n = len(c)
    k = min(k, n)
    top_idx = np.argpartition(c, -k)[-k:]
    top_c = c[top_idx]
    m = top_c.max()
    shifted = alpha * (top_c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
        w_top = np.exp(shifted - mx)
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
        w_top = np.exp(shifted)
    w_top = w_top / w_top.sum()
    w = np.zeros(n)
    w[top_idx] = w_top
    return m + logsum / alpha, w

def get_grad(h, n, w, w_idx, w_idx2, mask):
    dx = 2.0 / n
    W_mat = w[w_idx]
    A = W_mat @ (1.0 - h)
    W_mat2 = w[w_idx2] * mask
    B = W_mat2 @ h
    return dx * (A - B)

def adam_run(h, n, deadline, w_idx, w_idx2, mask, alpha, top_k, lr_init, n_iters=None):
    h = h.copy()
    mvp = np.zeros(n)
    vvp = np.zeros(n)
    t = 0
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    iters_no_improve = 0
    patience = 800

    while time.time() < deadline:
        if n_iters is not None and t >= n_iters:
            break
        t += 1
        c = compute_all_lags(h, n)
        if top_k == 1:
            val, w = smooth_max_val_w(c, alpha)
        else:
            val, w = tophk_sm(c, alpha, top_k)
        grad = get_grad(h, n, w, w_idx, w_idx2, mask)
        mvp = b1 * mvp + (1 - b1) * grad
        vvp = b2 * vvp + (1 - b2) * (grad ** 2)
        m_hat = mvp / (1 - b1 ** t)
        v_hat = vvp / (1 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h.copy()
            iters_no_improve = 0
        else:
            iters_no_improve += 1
        if iters_no_improve > patience:
            lr *= 0.5
            iters_no_improve = 0
            if lr < 1e-6:
                break
        lr *= 0.9995
    return best_h, best_c

def lbfgsb_phase(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
            w = np.exp(shifted - mx)
        else:
            logsum = np.log(np.sum(np.exp(shifted)))
            w = np.exp(shifted)
        w = w / w.sum()
        dx = 2.0 / n
        obj_val = m + logsum / alpha
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        W_mat = w[w_idx]
        A = W_mat @ (1.0 - x)
        W_mat2 = w[w_idx2] * mask
        B = W_mat2 @ x
        grad = dx * (A - B)
        grad += 2e6 * diff * np.ones(n)
        return obj_val, grad

    global w_idx, w_idx2, mask
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def coordinate_descent(h, n, deadline, rng, n_passes=50):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.time() > deadline:
                break
            orig = h[i]
            if orig <= 1e-8 or orig >= 1.0 - 1e-8:
                continue
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                new_val = orig + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    orig = h[i]
                else:
                    h[i] = orig
                    h = project_feasible(h, n)
        if not improved:
            # Try smaller steps
            improved2 = False
            for i in perm:
                if time.time() > deadline:
                    break
                orig = h[i]
                if orig <= 1e-8 or orig >= 1.0 - 1e-8:
                    continue
                for delta in [0.002, -0.002, 0.001, -0.001]:
                    new_val = orig + delta
                    if new_val < 0 or new_val > 1:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:

                        best_h = h.copy()
                        improved2 = True
                        orig = h[i]
                    else:

                        h = project_feasible(h, n)
            if not improved2:

    return best_h, best_c

def stochastic_local_search(h, n, deadline, rng, n_iters=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    top_k = np.argsort(c_all)[-10:]
    T = 0.01
    it_no_imp = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break

            continue
        m = abs(lag)
        if m >= n:
            continue
        i = rng.randint(0, n - m)
        j = i + m
        if h[i] > h[j] + 1e-7:
            d = rng.uniform(0.002, min(0.05, (h[i] - h[j]) * 0.3))
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            c_new = compute_all_lags(h, n)
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()

                top_k = np.argsort(c_all)[-10:]
                it_no_imp = 0
            else:
                de = max_new - c_all.max()
                if rng.random() > np.exp(-de / max(T, 1e-10)):
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                it_no_imp += 1
        elif h[j] > h[i] + 1e-7:
            d = rng.uniform(0.002, min(0.05, (h[j] - h[i]) * 0.3))
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            c_new = compute_all_lags(h, n)
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                c_all = c_new
                top_k = np.argsort(c_all)[-10:]

                    h = project_feasible(h, n)

        T *= 0.9999
        if it_no_imp > 5000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-10:]
            T = 0.005
            it_no_imp = 0
