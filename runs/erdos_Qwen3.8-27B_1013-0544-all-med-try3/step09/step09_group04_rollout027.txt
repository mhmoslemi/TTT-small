import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pso_search(n, n_particles, n_iter, deadline, initial_h=None):
    """PSO to find good starting point."""
    rng = np.random.RandomState(42)
    target = n / 2.0
    
    # Initialize particles
    positions = np.zeros((n_particles, n))
    velocities = np.zeros((n_particles, n))
    
    if initial_h is not None:
        positions[0] = initial_h.copy()
        for i in range(1, n_particles):
            # Random binary-ish starts
            h_rand = np.zeros(n)
            n_ones = n // 2
            idx = rng.choice(n, n_ones, replace=False)
            h_rand[idx] = 1.0
            # Add some randomness
            h_rand += rng.uniform(-0.3, 0.3, n)
            positions[i] = project_feasible(h_rand, n)
    else:
        for i in range(n_particles):
            h_rand = np.zeros(n)
            n_ones = n // 2
            idx = rng.choice(n, n_ones, replace=False)
            h_rand[idx] = 1.0
            h_rand += rng.uniform(-0.3, 0.3, n)
            positions[i] = project_feasible(h_rand, n)
    
    pbest = positions.copy()
    pbest_c = np.array([compute_c5(positions[i], n) for i in range(n_particles)])
    gbest_idx = np.argmin(pbest_c)
    gbest = pbest[gbest_idx].copy()
    gbest_c = pbest_c[gbest_idx]
    
    w0, w1 = 0.9, 0.4  # Inertia weight decay
    c1, c2 = 2.0, 2.0  # Cognitive and social coefficients
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        w = w0 - (w0 - w1) * (it / n_iter)
        
        for i in range(n_particles):
            r1 = rng.rand(n)
            r2 = rng.rand(n)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - positions[i]) + c2 * r2 * (gbest - positions[i])
            positions[i] = positions[i] + velocities[i]
            positions[i] = project_feasible(positions[i], n)
            
            c_i = compute_c5(positions[i], n)
            if c_i < pbest_c[i]:
                pbest_c[i] = c_i
                pbest[i] = positions[i].copy()
                if c_i < gbest_c:
                    gbest_c = c_i
                    gbest = positions[i].copy()
    
    return gbest, gbest_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_opt(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(50):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 10:
                break
            maxit = max(100, int(rem2 * 1.5))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, 0.002), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': maxit * 10,
                                        'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0, 1)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0, 1)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    target = n / 2.0
    positions = np.zeros((n_particles, n))
    velocities = np.zeros((n_particles, n))
    
    for i in range(n_particles):
        h_rand = np.zeros(n)
        n_ones = n // 2
        idx = rng.choice(n, n_ones, replace=False)
        h_rand[idx] = 1.0
        h_rand += rng.uniform(-0.5, 0.5, n)
        positions[i] = project_feasible(h_rand, n)
    
    pbest = positions.copy()
    pbest_c = np.array([compute_c5(pbest[i], n) for i in range(n_particles)])
    gbest_idx = np.argmin(pbest_c)
    gbest = pbest[gbest_idx].copy()
    gbest_c = float(pbest_c[gbest_idx])
    
    w0, w1 = 0.9, 0.3
    c1, c2 = 1.8, 2.0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        w = w0 - (w0 - w1) * (it / max(1, n_iter))
        
        for i in range(n_particles):
            r1 = rng.rand(n)
            r2 = rng.rand(n)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - positions[i]) + c2 * r2 * (gbest - positions[i])
            positions[i] = positions[i] + velocities[i]
            positions[i] = project_feasible(positions[i], n)
            
            c_i = compute_c5(positions[i], n)
            if c_i < pbest_c[i]:
                pbest_c[i] = c_i
                pbest[i] = positions[i].copy()
                if c_i < gbest_c:
                    gbest_c = c_i
                    gbest = positions[i].copy()
    
    return gbest, gbest_c

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxit * 15,
                                    'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_opt(h, n, deadline, top_k=5, alphas=None, tol_frac=0.002):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000, 100000000]
    
    for rnd in range(200):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, tol_frac * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(200, int(rem2 * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxfun': maxit * 12,
                                        'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        sweeps = 0
        while improved and time.time() < deadline:
            improved = False
            sweeps += 1
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-10 and best_h[j] - d >= -1e-10:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-10 and best_h[j] + d <= 1.0 + 1e-10:
                        best_h[i] -= d
                        best_h[j] += d
