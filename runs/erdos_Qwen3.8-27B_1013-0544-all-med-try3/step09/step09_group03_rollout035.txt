       # The unconstrained minimizer would be s[i] = 0 if grad[i] > 0, 1 if grad[i] < 0
       # But we need sum(s) = n/2
       # So we set some to 0, some to 1, and possibly one fractional
       
       # Sort by gradient values
       order = np.argsort(grad)
       s = np.zeros(n)
       # Set the lowest gradient entries to 1, highest to 0
       # We need exactly n/2 units of "mass"
       # Fill from lowest gradient: set to 1 until we've placed n/2 units
       
       k = int(n / 2)
       if k == 0:
           return s
       s[order[:k]] = 1.0
       # Check sum
       # If n is even, this gives exactly n/2
       # If n is odd... but n should be even for sum(h) = n/2 to be integer
       
       return s

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def ls_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def ls_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += -wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += -wm * x[kk:]
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=1000.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        remaining = deadline - time.time()
        maxiter = max(500, int(remaining * 8000))
        maxfun = max(2000, int(remaining * 80000))
        try:
            res = minimize(ls_obj, h, jac=ls_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxfun,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def frank_wolfe(h, n, deadline, top_m=3):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    half_n = int(n // 2)
    
    for t in range(500000):
        if time.time() > deadline:
            break
        c_vec = np.correlate(h, 1 - h, mode='full') * dx
        m = c_vec.max()
        if m < best_c - 0.01:
            break
        
        top_idx = np.argsort(c_vec)[-top_m:]
        top_vals = c_vec[top_idx]
        w = top_vals - top_vals.min() + 0.05
        w /= w.sum()
        
        grad = np.zeros(n)
        for j, ti in enumerate(top_idx):
            k = ti - center
            wm = w[j] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[:nv] += wm * (1.0 - h[k:])
                    grad[k:] += -wm * h[:nv]
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:] += wm * (1.0 - h[:nv])
                    grad[:nv] += -wm * h[kk:]
        
        order = np.argsort(grad)
        s = np.zeros(n)
        s[order[:half_n]] = 1.0
        
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = np.clip(h_new, 0.0, 1.0)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def coord_descent(h, n, deadline, n_rounds=500, delta0=0.004):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    center = n - 1
    delta = delta0
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = np.correlate(best_h, 1 - best_h, mode='full') * dx if (dx := 2.0/n) else 0
        c_vec = compute_c5_vec(best_h, n)
        top_idx = np.argsort(c_vec)[-5:]
        
        for ti in top_idx:
            k = ti - center
            if k >= 0:
                affected = list(range(n - k)) + list(range(k, n))
            else:
                kk = -k
                affected = list(range(n - kk)) + list(range(0, n + kk))
            affected = list(set(affected))
            if len(affected) > 80:
                affected = sorted(affected, key=lambda j: abs(j - n // 2))[:80]
        
        improved = False
        for j in affected:
            if time.time() > deadline:
                return best_h, best_c
            for d in [delta, -delta]:
                nv = best_h[j] + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                best_h[j] = nv
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    best_h[j] = nv - d
        if improved:
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
        else:
            delta *= 0.6
            if delta < 1e-7:
                break
    
    return best_h, best_c
