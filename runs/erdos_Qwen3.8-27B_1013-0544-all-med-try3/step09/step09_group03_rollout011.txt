    c = compute_c5_vec(h, n)

            if tot < 1e-14:

            h += (diff / tot) * h

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    active = np.where(w > 1e-8)[0]

                grad[:nl] -= w[pos] * x[la:]
            else:
                grad[:nl] += w[pos] * (1.0 - x[la:])

    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxls': 50,
                               'max_fun': 50000})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
                diff -= 1.0
                if abs(diff) < 1e-12:
                    break
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
                diff += 1.0
