    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_topk(h, n, deadline, kvals, mu=0.005):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for round_i in range(10):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        for k in kvals:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int(remaining * 2))
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, k, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_active_lbfgs(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 25:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active], kind='stable')[:top_k]
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 12:
                break
            maxit = max(400, int(remaining * 3))
            try:
                def act_o(x, _n, _a, _al, _mu):
                    cv = compute_c5_vec(x, _n)
                    vals = cv[_a]
                    mx = vals.max()
                    sh = _al * (vals - mx)
                    smx = sh.max()
                    if smx > 500:
                        v = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / _al
                    else:
                        v = mx + np.log(np.sum(np.exp(sh))) / _al
                    if _mu > 0:
                        v += _mu * (x.sum() - _n / 2.0) ** 2
                    return float(v)
                # Can't define closure, use sm_obj with hack: just use topk approach
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(len(active), 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_sweeps=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            orig = best_h[i]
            # Try small perturbation
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.0001, -0.0001]:
                new_val = orig + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Adjust another point to maintain sum
                # Find a point to adjust
                j = (i + 1) % n
                adj = orig - new_val
                new_j = best_h[j] + adj
                if new_j < 0 or new_j > 1:
                    continue
                best_h[i] = new_val
                best_h[j] = new_j
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = orig
                    best_h[j] = best_h[j] - adj  # revert
            if not improved and time.time() > deadline - 0.5:
                break
        # Re-project after each sweep
        best_h = project_feasible(best_h, n)
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def downsample(h, n_from, n_to):
    if n_to == n_from:
        return h.copy()
    x_fine = np.linspace(0, 2, n_from)
    x_coarse = np.linspace(0, 2, n_to)
    h_new = np.interp(x_coarse, x_fine, h)
    return project_feasible(h_new, n_to)

def upsample(h, n_from, n_to):
    if n_to == n_from:
        return h.copy()
    x_coarse = np.linspace(0, 2, n_from)
    x_fine = np.linspace(0, 2, n_to)
    h_new = np.interp(x_fine, x_coarse, h)
    return project_feasible(h_new, n_to)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    h0 = project_feasible(h0, n0)
    
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: Coarse-to-fine at n=250
    n_coarse = 250
    if time.time() < deadline - 600:
        h_coarse = downsample(h0, n0, n_coarse)
        c_coarse = compute_c5(h_coarse, n_coarse)
        p1_end = min(t0 + 120, deadline - 580)
        h1, c1 = run_lbfgs(h_coarse, n_coarse, p1_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
            mu=0.01, maxiters=[800, 1000, 1200, 1500, 1500, 1500, 1500])
        if c1 < compute_c5(h_coarse, n_coarse):
            h_coarse = h1
            c_coarse = c1
        # Upsample back to n0
        h_up = upsample(h_coarse, n_coarse, n0)
        c_up = compute_c5(h_up, n0)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n0
    
    # Phase 2: L-BFGS at n0 with alpha annealing
    if time.time() < deadline - 500:
        p2_end = min(t0 + 300, deadline - 480)
        h2, c2 = run_lbfgs(best_h, best_n, p2_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.01, maxiters=[1500, 2000, 2500, 2500, 3000, 3000, 3500, 4000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = best_n
    
    # Phase 3: Top-k optimization at n0
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 120, deadline - 330)
        h3, c3 = run_topk(best_h, best_n, p3_end, kvals=[2, 3, 5, 8, 12], mu=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n
    
    # Phase 4: Try n=1000
    n_hi = 1000
    if time.time() < deadline - 250:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        p4_end = min(time.time() + 250, deadline - 200)
        h4, c4 = run_lbfgs(h_hi, n_hi, p4_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.005, maxiters=[1000, 1500, 2000, 2000, 2500, 2500, 3000, 3000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_hi
    
    # Phase 5: Top-k at best resolution
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 100, deadline - 160)
        h5, c5 = run_topk(best_h, best_n, p5_end, kvals=[2, 4, 6, 10], mu=0.003)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = best_n
    
    # Phase 6: Try complement (1-h) as diverse start
    if time.time() < deadline - 120:
        h_comp = project_feasible(1.0 - best_h, best_n)
        c_comp = compute_c5(h_comp, best_n)
        # Brief optimization on complement
        p6_end = min(time.time() + 50, deadline - 100)
        h6, c6 = run_lbfgs(h_comp, best_n, p6_end,
            alphas=[1000, 10000, 100000, 500000],
            mu=0.01, maxiters=[1000, 1500, 1500, 1500])
        # Map back: complement of optimized complement
        h6_back = project_feasible(1.0 - h6, best_n)
        c6_back = compute_c5(h6_back, best_n)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        elif c6_back < best_c:
            best_c = c6_back
            best_h = h6_back
    
    # Phase 7: Subgradient refinement
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n
    
    # Phase 8: Coordinate descent
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 20, deadline - 10)
        h8, c8 = coordinate_descent(best_h, best_n, p8_end, n_sweeps=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n
    
    # Phase 9: Fine pair search
    if time.time() < deadline - 5:
        p9_end = max(deadline - 2, time.time() + 15)
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=40)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = best_n
    
    # Phase 10: Final SLSQP (very short)
    if time.time() < deadline - 2:
        p10_end = deadline - 1
        h10, c10 = run_slqp(best_h, best_n, p10_end, alpha=1000000, maxiter=300)
        if c10 < best_c:
            best_c = c10
            best_h = h10
            best_n = best_n
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
