    # Get top-k indices
    top_idx = np.argsort(c)[-k_top:]
    c_top = c[top_idx]
    m = c_top.max()

    mx = sh.max()

def topk_smooth_grad(h, n, k_top, alpha, mu):

    mx = sh.max()

    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def full_smooth_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def topk_smooth_obj(h, n, k_top, alpha, mu):
    c = compute_c5_vec(h, n)
    top_idx = np.argsort(c)[-k_top:]
    c_top = c[top_idx]
    m = c_top.max()
    sh = alpha * (c_top - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def topk_smooth_grad(h, n, k_top, alpha, mu):
    c = compute_c5_vec(h, n)
    top_idx = np.argsort(c)[-k_top:]
    c_top = c[top_idx]
    m = c_top.max()
    sh = alpha * (c_top - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    for i, idx in enumerate(top_idx):
        if w[i] < 1e-13:
            continue
        k = idx - center
        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def run_topk_lbfgs(h, n, deadline, k_top, alphas, mu=1.0, maxiter=50000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(topk_smooth_obj, h, jac=topk_smooth_grad,
                          args=(n, k_top, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def run_full_lbfgs(h, n, deadline, alphas, mu=1.0, maxiter=30000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(full_smooth_obj, h, jac=full_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 80000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, na=n: float(x.sum() - na / 2.0),
             'jac': lambda x, na=n: np.ones(na)}]
    try:
        res = minimize(full_smooth_obj, h, jac=full_smooth_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x') and res.success is not False:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < compute_c5(h, n):
                return h_new, c_new
    except Exception:
        pass
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def local_coordinate_descent(h, n, deadline, n_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    worst_idx = int(np.argmax(c_vec))
    dx = 2.0 / n
    center = n - 1
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - center
        
        candidates = []
        if k_worst >= 0:
            nv = n - k_worst
            for j in range(nv):
                val = h[j] * (1.0 - h[j + k_worst])
                candidates.append((val, j, j + k_worst, k_worst))
        else:
            kk = -k_worst
            nv = n - kk
            for j in range(nv):
                val = h[j + kk] * (1.0 - h[j])
                candidates.append((val, j + kk, j, kk))
        
        candidates.sort(key=lambda x: -x[0])
        
        improved = False
        for val, i1, i2, kk in candidates[:30]:
            if time.time() > deadline:
                break
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                # Adjust i1 up and i2 down (or vice versa)
                for sign in [1, -1]:
                    n1 = h[i1] + sign * delta
                    n2 = h[i2] - sign * delta
                    if n1 < -1e-10 or n1 > 1 + 1e-10:
                        continue
                    if n2 < -1e-10 or n2 > 1 + 1e-10:
                        continue
                    h[i1] = np.clip(n1, 0, 1)
                    h[i2] = np.clip(n2, 0, 1)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i1] = best_h[i1]
                        h[i2] = best_h[i2]
                if improved:
                    break
            if improved:
                h = best_h.copy()
                break
        if not improved and rnd > 50:
            break
    
    return best_h, best_c

def interpolate_to_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=6):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777 + n)
    mags = [0.03, 0.06, 0.01, 0.08, 0.02, 0.05]
    for t in range(n_trials):
        if time.time() > deadline - 40:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(topk_smooth_obj, h_p, jac=topk_smooth_grad,
                          args=(n, 3, 5000, 1.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 8000, 'ftol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Top-k L-BFGS at base resolution (k_top=5, moderate alpha)
    t1 = min(t0 + 120, deadline - 700)
    if t1 > time.time() + 10:
        best_h, best_c = run_topk_lbfgs(best_h, n_base, t1, k_top=5,
            alphas=[200, 1000, 5000, 20000, 100000], mu=1.0)
    
    # Phase 2: Top-k L-BFGS focusing on single worst shift (k_top=1, high alpha)
    t2 = min(t0 + 250, deadline - 550)
    if t2 > time.time() + 10:
        best_h, best_c = run_topk_lbfgs(best_h, n_base, t2, k_top=1,
            alphas=[50000, 200000, 1000000], mu=0.5)
    
    # Phase 3: SLSQP polish at base
    t3 = min(t0 + 340, deadline - 450)
    if t3 > time.time() + 10:
        h3, c3 = slqp_polish(best_h, n_base, t3, alpha=5000000, maxiter=8000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Local coordinate descent at base
    t4 = min(t0 + 400, deadline - 400)
    if t4 > time.time() + 5:
        best_h, best_c = local_coordinate_descent(best_h, n_base, t4, n_rounds=150)
    
    # Phase 5: Top-k L-BFGS again at base (k_top=3)
    t5 = min(t0 + 480, deadline - 350)
    if t5 > time.time() + 10:
        best_h, best_c = run_topk_lbfgs(best_h, n_base, t5, k_top=3,
            alphas=[5000, 20000, 100000, 500000], mu=1.0)
    
    # Phase 6: Upsample to n=1000
    n1 = 1000
    if time.time() < deadline - 300:
        h1000 = interpolate_to_h(best_h, n_base, n1)
        t6 = min(time.time() + 120, deadline - 200)
        if t6 > time.time() + 10:
            h1000, c1000 = run_topk_lbfgs(h1000, n1, t6, k_top=5,
                alphas=[500, 2000, 10000, 50000], mu=1.0)
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        
        t6b = min(time.time() + 80, deadline - 120)
        if t6b > time.time() + 10:
            best_h, best_c = run_topk_lbfgs(best_h, n1, t6b, k_top=1,
                alphas=[50000, 200000, 1000000], mu=0.5)
    
    # Phase 7: SLSQP at n=1000
    if time.time() < deadline - 100:
        t7 = min(time.time() + 60, deadline - 60)
        if t7 > time.time() + 10:
            h7, c7 = slqp_polish(best_h, len(best_h), t7, alpha=10000000, maxiter=5000)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Perturbation at current resolution
    if time.time() < deadline - 80:
        t8 = min(time.time() + 60, deadline - 30)
        if t8 > time.time() + 10:
            best_h, best_c = perturbation_search(best_h, len(best_h), t8, n_trials=5)
    
    # Phase 9: Local coordinate descent at final resolution
    if time.time() < deadline - 20:
        best_h, best_c = local_coordinate_descent(best_h, len(best_h), deadline - 5, n_rounds=100)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
