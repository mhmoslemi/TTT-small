def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(sh - mx)))
        w = np.exp(sh - mx)
    else:
        logsum = np.log(np.sum(np.exp(sh)))
        w = np.exp(sh)
    w = w / w.sum()
    val = m + logsum / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return float(val), grad

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        val, _ = smooth_obj_and_grad(x, na, al)
        return val
    def jac_fn(x, na, al):
        _, g = smooth_obj_and_grad(x, na, al)
        return g
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

        h_opt, c_opt = run_lbfgs_continuation(h0, n, t_ord_end, alpha_schedule)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n0 = 500
    
    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: L-BFGS continuation at base resolution (~90s)
    t1 = min(t0 + 90, deadline - 800)
    if t1 > time.time() + 10:
        sch1 = [(20, 1.0), (100, 10.0), (500, 50.0), (2000, 200.0),
                (8000, 1000.0), (30000, 5000.0), (100000, 20000.0)]
        best_h, best_c = run_lbfgs_continuation(best_h, n0, t1, sch1)
    
    # Phase 2: SLSQP polish at base (~40s)
    t2 = min(t0 + 130, deadline - 700)
    if t2 > time.time() + 10:
        h2, c2 = slqp_polish(best_h, n0, t2, alpha=500000, maxiter=5000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n1 = 1000 and optimize (~180s)
    n1 = 1000
    t3_end = min(t0 + 310, deadline - 500)
    if t3_end > time.time() + 20:
        h1 = interpolate_to(best_h, n0, n1)
        # Multiple starts: parent-based + perturbed
        rng = np.random.RandomState(seed + 1)
        h1_pert = project_feasible(h1 + 0.03 * (rng.rand(n1) - 0.5), n1)
        starts = [h1, h1_pert]
        sch2 = [(50, 5.0), (200, 20.0), (1000, 100.0), (5000, 500.0),
                (20000, 2000.0), (50000, 5000.0), (100000, 20000.0)]
        best_h, best_c = multi_start_optimize(starts, n1, t3_end, sch2)
        
        # Quick SLSQP at n1
        t3b = min(time.time() + 50, deadline - 400)
        if t3b > time.time() + 10:
            h3b, c3b = slqp_polish(best_h, n1, t3b, alpha=1000000, maxiter=3000)
            if c3b < best_c:
                best_c = c3b
                best_h = h3b
    
    # Phase 4: Upsample to n2 = 2000 and optimize (~300s)
    n2 = 2000
    t4_end = min(t0 + 700, deadline - 300)
    if t4_end > time.time() + 30:
        h2u = interpolate_to(best_h, len(best_h), n2)
        sch3 = [(50, 5.0), (300, 20.0), (1500, 100.0), (6000, 500.0),
                (20000, 2000.0), (50000, 5000.0), (100000, 20000.0)]
        # Just one start at this resolution for time
        h2u, c2u = run_lbfgs_continuation(h2u, n2, t4_end, sch3)
        if c2u < best_c:
            best_c = c2u
            best_h = h2u
    
    # Phase 5: Gradient descent refinement at n2 (~150s)
    t5_end = min(time.time() + 150, deadline - 100)
    if t5_end > time.time() + 20:
        h5, c5 = gradient_descent_refine(best_h, len(best_h), t5_end,
                                          alpha=100000, steps=3000, lr_init=0.005, lr_decay=0.9995)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Coordinate descent on active shifts (~100s)
    t6_end = min(time.time() + 100, deadline - 30)
    if t6_end > time.time() + 15:
        best_h, best_c = coordinate_refine(best_h, len(best_h), t6_end, n_rounds=30)
    
    # Phase 7: Final SLSQP polish (~30s)
    t7_end = min(time.time() + 30, deadline - 5)
    if t7_end > time.time() + 8:
        h7, c7 = slqp_polish(best_h, len(best_h), t7_end, alpha=2000000, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
