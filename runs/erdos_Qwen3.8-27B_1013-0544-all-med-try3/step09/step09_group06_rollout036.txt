import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction for floating point
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argsort(1.0 - h)[-max(1, int(abs(diff) * 2)):]
            h[idx] = np.minimum(1.0, h[idx] + abs(diff) / len(idx))
        else:
            idx = np.argsort(h)[-max(1, int(abs(diff) * 2)):]
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w = w / s
    return w

def grad_overlaps(h, n, weights):
    """Compute gradient of weighted sum of overlaps. weights is over 2n-1 lags."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            # d/dh[i] of sum_{j=0}^{n-m-1} (1-h[j])*h[j+m]
            # = (1-h[i-m]) for i >= m  +  (-h[i+m]) for i < n-m
            if m < n:
                grad[m:] += wk * (1.0 - h[0:n-m])
                grad[0:n-m] += wk * (-h[m:])
        else:
            m = -lag
            # d/dh[i] of sum_{j=0}^{n-m-1} h[j]*(1-h[j+m])
            # = (1-h[i+m]) for i < n-m  +  (-h[i-m]) for i >= m
            if m < n:
                grad[0:n-m] += wk * (1.0 - h[m:])
                grad[m:] += wk * (-h[0:n-m])
    return grad * dx

def projected_gradient_step(h, n, alpha, step_size, momentum, v, deadline):
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    g = grad_overlaps(h, n, w)
    v = momentum * v + step_size * g
    h_new = project_feasible(h - v, n)
    return h_new, v

def projected_grad_optimize(h, n, deadline, alpha_start=50, alpha_end=50000,
                            n_steps=3000, momentum=0.9, lr_start=1.0, lr_end=0.05):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        # Adaptive parameters
        frac = it / max(1, n_steps - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        # Periodic restart
        if it % 500 == 499:
            v *= 0.5
        
        h_new, v = projected_gradient_step(h, n, alpha, lr, momentum, v, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def paired_coord_descent(h, n, deadline, n_passes=100, random_order=False, rng=None):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for pas in range(n_passes):
        if time.time() > deadline:
            break
        improved = 0
        if random_order and rng is not None:
            pairs = list(range(n - 1))
            rng.shuffle(pairs)
        else:
            pairs = list(range(n - 1))
        
        for i in pairs:
            if time.time() > deadline:
                break
            j = i + 1
            s = h[i] + h[j]
            if abs(s - 1.0) < 1e-10:
                # Already at boundary, skip
                continue
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-10:
                continue
            
            # Coarse grid
            n_g = 20
            ts = np.linspace(t_lo, t_hi, n_g)
            orig_i, orig_j = h[i], h[j]
            best_t = orig_i
            best_c_local = best_c
            for t in ts:
                h[i] = t
                h[j] = s - t
                c = compute_c5(h, n)
                if c < best_c_local - 1e-15:
                    best_c_local = c
                    best_t = t
            h[i], h[j] = orig_i, orig_j
            
            if best_t != orig_i:
                # Refine
                lo = max(t_lo, best_t - 0.05)
                hi = min(t_hi, best_t + 0.05)
                ts2 = np.linspace(lo, hi, 25)
                for t in ts2:

                    h[j] = s - t

                h[i], h[j] = orig_i, orig_j
                
                # Apply

                h[j] = s - best_t
                improved += 1
                if best_c_local < best_c:
                    best_c = best_c_local
                    best_h = h.copy()
        
        if improved == 0 and pas > 3:
            break
    
    return best_h, best_c

def active_set_refine(h, n, deadline, top_m=5, n_iters=200):
    """Iteratively balance the top-m active lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_m:]
        c_top = c_all[top_idx]
        c_max = c_top.max()
        c_min = c_top.min()
        
        if c_max - c_min < 1e-8:
            break
        
        # Compute gradient of (c_max - c_min)^2 / 2 approximated by
        # weight on max lags positive, min lags negative
        weights = np.zeros(2 * n - 1)
        for k in top_idx:
            if c_all[k] > c_max - 1e-6:
                weights[k] = 1.0
            elif c_all[k] < c_min + 1e-6:
                weights[k] = -1.0
            else:
                weights[k] = (c_all[k] - (c_max + c_min) / 2) / max(c_max - c_min, 1e-10)
        
        # Normalize
        w_norm = weights / max(weights.sum(), 1e-15)
        g = grad_overlaps(h, n, w_norm)
        
        # Line search
        best_lr = 0.0
        best_c_ls = best_c
        for lr in [0.5, 0.2, 0.1, 0.05, 0.01, 0.005, 0.001]:
            h_trial = project_feasible(h - lr * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c_ls:
                best_c_ls = c_trial
                best_lr = lr
            h_trial = project_feasible(h + lr * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c_ls:
                best_c_ls = c_trial
                best_lr = -lr
        
        if best_lr == 0.0:
            break
        
        h = project_feasible(h + best_lr * g, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_refine(h, n, alpha, top_m, maxiter=2000):
    """SLSQP on the active set with smooth max."""
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        return smooth_max_val(active, alpha)
    
    def obj_grad(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        w = smooth_max_weights(active, alpha)
        # Build full weights vector
        full_w = np.zeros(2 * n - 1)
        for idx, k in enumerate(top_idx):
            full_w[k] = w[idx]
        return grad_overlaps(x, n, full_w)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()

        ls_deadline = min(phase3_end - 40, time.time() + 80)
        h2, c2 = paired_coord_descent(best_h, n, ls_deadline, n_passes=50,
                                       random_order=False)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # 3c: Active set balancing
    if time.time() < phase3_end - 30:
        h3, c3 = active_set_refine(best_h, n, min(phase3_end - 20, time.time() + 60),
                                    top_m=5, n_iters=100)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # 3d: Another round of paired CD
    if time.time() < phase3_end - 20:
        h4, c4 = paired_coord_descent(best_h, n, min(phase3_end - 10, time.time() + 60),
                                       n_passes=30)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # 3e: Swept pair random search
    if time.time() < phase3_end - 10:
        h5, c5_val = swept_pair_search(best_h, n, min(phase3_end - 5, time.time() + 60),
                                        rng, n_trials=30000)
        if c5_val < best_c:
            best_c = c5_val
            best_h = h5
    
    # Phase 4: Final intensive polishing (remaining time)
    n = len(best_h)
    
    # 4a: High-alpha gradient with momentum
    if time.time() < deadline - 30:
        alloc4 = min(60, (deadline - time.time()) * 0.4)
        h6, c6 = projected_grad_optimize(best_h, n, time.time() + alloc4,
                                           alpha_start=5000, alpha_end=200000,
                                           n_steps=2000, momentum=0.95,
                                           lr_start=0.5, lr_end=0.01)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # 4b: Final paired CD with more passes
    if time.time() < deadline - 15:
        h7, c7 = paired_coord_descent(best_h, n, min(deadline - 10, time.time() + 60),
                                       n_passes=80)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # 4c: Active set finalize
    if time.time() < deadline - 10:
        h8, c8 = active_set_refine(best_h, n, min(deadline - 5, time.time() + 30),
                                    top_m=4, n_iters=100)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # 4d: Final micro paired CD
    if time.time() < deadline - 3:
        h9, c9 = paired_coord_descent(best_h, n, deadline - 1, n_passes=50)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
