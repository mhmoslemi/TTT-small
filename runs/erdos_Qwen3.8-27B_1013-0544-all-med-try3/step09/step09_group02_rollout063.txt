def solve_qp_for_binary_base(p, n):
    """Given binary base p, find optimal correction q and resulting C5."""
    from scipy.optimize import linprog
    
    dx = 2.0 / n
    n_var = n + 1  # q[0..n-1] and t
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    # Compute binary C_k values
    p_1mp = 1 - p  # 1-p
    c_bin = np.correlate(p, p_1mp, mode='full') * dx  # 2n-1 values
    
    # Build constraint matrix for each lag
    # C_k(q) ≈ c_bin[k] + dx * sum_i [q[i]*(1-p[i+k]) - p[i]*q[i+k]]
    # We need: c_bin[k] + dx * sum_i [...] ≤ t
    # => dx * sum_i [...] - t ≤ -c_bin[k]
    
    n_lags = 2*n - 1
    A_ub = np.zeros((n_lags, n_var))
    b_ub = np.zeros(n_lags)
    
    for lag_idx in range(n_lags):
        k = lag_idx - (n - 1)  # actual shift
        if k >= 0:
            nv = n - k
            # sum_i q[i]*(1-p[i+k]) for i=0..nv-1
            for i in range(nv):
                A_ub[lag_idx, i] += dx * (1 - p[i+k])
            # sum_i -p[i]*q[i+k] for i=0..nv-1 => sum_j -p[j-k]*q[j] for j=k..n-1
            for j in range(k, n):
                A_ub[lag_idx, j] += -dx * p[j-k]
        else:
            kk = -k
            nv = n - kk
            # sum_i q[i]*(1-p[i-kk]) for i=kk..n-1
            for i in range(kk, n):
                A_ub[lag_idx, i] += dx * (1 - p[i-kk])
            # sum_i -p[i]*q[i-kk] for i=kk..n-1 => sum_j -p[j+kk]*q[j] for j=0..nv-1
            for j in range(nv):
                A_ub[lag_idx, j] += -dx * p[j+kk]
        
        A_ub[lag_idx, n] = -1.0  # -t
        b_ub[lag_idx] = -c_bin[lag_idx]
    
    # Bound constraints: -p[i] ≤ q[i] ≤ 1-p[i]
    # For p[i]=1: -1 ≤ q[i] ≤ 0
    # For p[i]=0: 0 ≤ q[i] ≤ 1
    
    bounds = []
    for i in range(n):
        if p[i] == 1:
            bounds.append((-1.0, 0.0))
        else:
            bounds.append((0.0, 1.0))
    bounds.append((None, None))  # t is free
    
    # Equality: sum(q) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    return res

def lp_correction(h_center, n, rho=0.5):

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp_constraint(h_center, n):
    """Build the linearized constraint matrix for all lags."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c_vals = np.correlate(h_center, 1 - h_center, mode='full') * dx
    A_lag = np.zeros((n_lags, n))
    for lag_idx in range(n_lags):
        k = lag_idx - (n - 1)
        if k >= 0:
            nv = n - k
            A_lag[lag_idx, 0:nv] = dx * (1.0 - h_center[k:n])
            A_lag[lag_idx, k:n] += -dx * h_center[0:nv]
        else:
            kk = -k
            nv = n - kk
            A_lag[lag_idx, kk:n] = dx * (1.0 - h_center[0:nv])
            A_lag[lag_idx, 0:nv] += -dx * h_center[kk:n]
    return A_lag, c_vals

def lp_step(h_center, n, rho, active_only=False, n_active_top=30):
    """Solve LP trust-region step. Returns (delta, t_value) or (None, None)."""
    dx = 2.0 / n
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0

    A_lag, c_vals = build_lp_constraint(h_center, n)
    n_lags = 2 * n - 1

    if active_only:
        max_val = c_vals.max()
        thresh = max_val - max(1e-10, 0.01 * max_val)
        active_mask = c_vals >= thresh
        if active_mask.sum() > n_active_top:
            order = np.argsort(-c_vals)
            active_mask = np.zeros(n_lags, dtype=bool)
            active_mask[order[:n_active_top]] = True
        A_lag = A_lag[active_mask]
        c_vals_sel = c_vals[active_mask]
    else:
        c_vals_sel = c_vals

    n_sel = len(c_vals_sel)
    A_ub = np.hstack([A_lag, -np.ones((n_sel, 1))])
    b_ub = -c_vals_sel

    bounds = []
    for i in range(n):
        lo = max(h_center[i] - 1.0, -rho)
        hi = min(1.0 - h_center[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, None))

    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None

def iterative_lp_opt(h, n, deadline, max_iter=500, rho_init=0.08, active_only=False):
    """Iterative LP trust-region optimization."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        delta, t_lp = lp_step(h, n, rho, active_only=active_only)
        if delta is None:
            rho *= 0.6
            if rho < 1e-12:
                break
            continue

        step_scale = 1.0
        accepted = False
        for _ in range(4):
            h_new = h + step_scale * delta
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                accepted = True
                if rho < 0.06:
                    rho = min(rho * 1.15, 0.1)
                break
            step_scale *= 0.5

        if not accepted:
            no_improve += 1
            rho *= 0.7
            if rho < 1e-12 or no_improve > 40:
                break

    return best_h, best_c

def subgradient_descent(h, n, deadline, eta_init=0.01):
    """Subgradient descent on the max C_k."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta_init
    dx = 2.0 / n

    for it in range(20000):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h, n)
        k_star = int(np.argmax(c_vec))
        k = k_star - (n - 1)

        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += -h[0:nv] * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += -h[kk:n] * dx

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        for scale in [eta, eta * 0.5, eta * 0.25]:
            h_new = h - scale * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                if eta < 0.005:
                    eta = min(eta * 1.05, eta_init)
                break
        else:
            eta *= 0.9
            if eta < 1e-10:
                break

    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mus):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for i, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 5000, 'ftol': 1e-19,
                                  'gtol': 1e-17, 'maxfun': 80000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_lp_init(h, n, deadline):
    """Create binary threshold and solve LP for correction."""
    p = (h > 0.5).astype(float)
    n_ones = int(p.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(p == 1)[0]
        order = np.argsort(-h[ones_idx])
        remove_idx = ones_idx[order[target:]]
        p[remove_idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(p == 0)[0]
        order = np.argsort(-h[zeros_idx])
        add_idx = zeros_idx[order[:target - n_ones]]
        p[add_idx] = 1

    # Iterative LP from binary start
    p_feas = project_feasible(p, n)
    h_out, c_out = iterative_lp_opt(p_feas, n, deadline, max_iter=100, rho_init=0.4)
    return h_out, c_out

def multi_start_lp(h, n, deadline, n_starts=5):

        if c_opt < best_c - 1e-15:
