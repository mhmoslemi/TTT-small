Actually, let me think about this differently. What if I use a completely different optimization strategy?

**Alternating Direction Method (ADMM-like)**:
- Introduce auxiliary variable t: minimize t
- Subject to: c_k(h) ≤ t for all k, 0≤h≤1, sum(h)=n/2
- This is a non-convex problem (c_k is quadratic in h), but we can use alternating:
  - Fix h, update t = max_k c_k(h) (trivial)
  - Fix t, minimize sum_k max(0, c_k(h) - t)^2 (smooth penalty on active constraints)

Or **Proximal Gradient**:
- F(h) = max_k c_k(h) (non-smooth)
- Use proximal gradient: h_{t+1} = argmin F(h_t) + <g, h-h_t> + (1/2L)||h-h_t||^2
- Then project to feasible set

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, j):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = j - (n - 1)
    if k >= 0:
        nv = n - k
        grad[k:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[k:n]) * dx
    else:
        m = -k
        nv = n - m
        grad[m:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[m:n]) * dx
    return grad

def smooth_obj_lse(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_lse(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        g = shift_grad(x, n, mm)
        grad += w[mm] * g
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smooth_obj_power(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-30)
    cp = c ** p
    val = (cp.sum()) ** (1.0 / p)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_power(x, n, p, mu):
    c = np.maximum(compute_c5_vec(x, n), 1e-30)
    cp = c ** p
    sp = cp.sum()
    F = sp ** (1.0 / p)
    weights = c ** (p - 1) / sp
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if weights[mm] < 1e-14:
            continue
        grad += weights[mm] * shift_grad(x, n, mm)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05, top_k=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            top_order = np.argsort(-c_vec[active])
            active = active[top_order[:top_k]]
        if len(active) == 0:
            break
        delta, t_opt = sqp_lp_step(h, n, list(active), rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-10 or no_improve > 30:
                break
    return best_h, best_c

def subgradient_polish(h, n, deadline, n_iter_max=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.01
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        grad = np.zeros(n)
        for j in active:
            grad += shift_grad(h, n, j)
        grad /= max(len(active), 1)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = step_size / gnorm
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            if step_size < 0.05:
                step_size *= 1.2
        else:
            step_size *= 0.7
            if step_size < 1e-6:
                break
    return best_h, best_c

def local_search_aggressive(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            top_pos = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def local_search_2swap(h, n, deadline, n_trials=20000):
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(12345)
    trial = 0
    while trial < n_trials and time.time() < deadline:
        trial += 1
        if trial % 500 == 0:
            pass
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        i3 = rng.integers(0, n)
        i4 = rng.integers(0, n)
        if i1 == i2 or i1 == i3 or i1 == i4 or i2 == i3 or i2 == i4 or i3 == i4:
            continue
        if i1 == i2 or i3 == i4:
            continue
        # Try swapping h[i1],h[i2] with h[i3],h[i4]
        orig = [h[i1].copy(), h[i2].copy(), h[i3].copy(), h[i4].copy()]
        h[i1], h[i2] = orig[2], orig[3]
        h[i3], h[i4] = orig[0], orig[1]
        if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1 and 0 <= h[i3] <= 1 and 0 <= h[i4] <= 1:
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
            else:
                h[i1], h[i2] = orig[0], orig[1]
                h[i3], h[i4] = orig[2], orig[3]
        else:
            h[i1], h[i2] = orig[0], orig[1]
            h[i3], h[i4] = orig[2], orig[3]
    return h, best_c

def fourier_perturb(h, n, scale, rng):
    h_new = h.copy()
    phases = rng.uniform(0, 2 * np.pi, size=6)
    amps = rng.uniform(0, scale, size=6)
    modes = rng.choice(range(2, 12), size=6, replace=False)
    x = np.linspace(0, 2, n)
    for m, a, ph in zip(modes, amps, phases):
        h_new += a * np.sin(np.pi * m * x + ph)
    return project_feasible(h_new, n)

def run_lbfgs_configs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj_lse, h, jac=smooth_grad_lse,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-19,
                                   'gtol': 1e-16, 'maxfun': 80000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_power_mean(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for p, mu in configs:

        try:
            res = minimize(smooth_obj_power, h, jac=smooth_grad_power,

                          options={'maxiter': 5000, 'ftol': 1e-19,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

def slaqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a)
        else:
            return float(m + np.log(np.sum(np.exp(sh)))) / a
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

        best_h, best_c = local_search_aggressive(best_h, n_orig, t5, radius=70)

    # Phase 6: 2-swap local search

    # Phase 7: Another SQP round

    # Phase 8: SLSQP polish
    t8 = min(t0 + 550, deadline - 300)
    if t8 > time.time() + 5:

        if c8 < best_c:
            best_c = c8

    # Phase 9: Fourier perturbation + re-optimize (escape local optima)
    if time.time() < deadline - 250:
        for pert_scale in [0.02, 0.01, 0.005]:
            if time.time() > deadline - 220:
                break
            h_pert = fourier_perturb(best_h, n_orig, pert_scale, rng)
            t_pert = min(time.time() + 30, deadline - 180)

                    configs=[(2000, 500.0), (8000, 5000.0)])
                t_sq = min(time.time() + 20, deadline - 150)
                if t_sq > time.time() + 3:
                    h_pert, c_pert = sqp_optimize(h_pert, n_orig, t_sq, max_iter=100, rho_init=0.03)

                    best_h = h_pert

    # Phase 10: Upsample to higher resolution

    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)

        t_up1 = min(time.time() + 50, deadline - 120)
        if t_up1 > time.time() + 5:
            h_high, c_high = run_lbfgs_configs(h_high, n_high, t_up1,
                configs=[(1000, 200.0), (4000, 2000.0), (12000, 10000.0)])

        t_up2 = min(time.time() + 40, deadline - 60)
        if t_up2 > time.time() + 3:
            h_high, c_high = sqp_optimize(h_high, n_high, t_up2, max_iter=200, rho_init=0.04, top_k=10)

        t_up3 = min(time.time() + 30, deadline - 25)
        if t_up3 > time.time() + 3:
            h_high, c_high = local_search_aggressive(h_high, n_high, t_up3, radius=40)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Phase 11: Final polish
    if time.time() < deadline - 15:
        t_fin = deadline - 5
        h_fin, c_fin = slaqp_polish(best_h, len(best_h), t_fin, alpha=8000000, maxiter=3000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
