        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag_idx):
    """Gradient of f_k w.r.t. h, where lag_idx is the index in the full correlation."""
    dx = 2.0 / n
    grad = np.zeros(n)

        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(h, n, alpha, k_top):
    c = compute_c5_vec(h, n)
    m = c.max()
    idx = np.argsort(-c)[:k_top]
    c_k = c[idx]
    sh = alpha * (c_k - c_k.max())
    sm = sh.max()
    logsum = sm + np.log(np.sum(np.exp(sh - sm)))
    return float(c_k.max() + logsum / alpha)

def smooth_max_grad(h, n, alpha, k_top):
    c = compute_c5_vec(h, n)
    idx = np.argsort(-c)[:k_top]
    c_k = c[idx]
    sh = alpha * (c_k - c_k.max())
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        g = lag_grad(h, n, int(idx[ii]))
        grad += w[ii] * g
    return grad

def sqp_step(h, n, active_idx, rho, c_vec):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    n_act = len(active_idx)
    n_ineq = n_act + 4 * n
    A_ub = np.zeros((n_ineq, n_var))
    b_ub = np.zeros(n_ineq)
    row = 0
    for idx in active_idx:
        g = lag_grad(h, n, int(idx))
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[int(idx)]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.06, max_active=35, slack_frac=0.001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):

            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            active = [int(np.argmax(c_vec))]
        if len(active) > max_active:
            order = np.argsort(-c_vec[active])
            active = [int(active[i]) for i in order[:max_active]]
        delta, t = sqp_step(h, n, active, rho, c_vec)
        if delta is None:
            rho *= 0.5
            if rho < 1e-11:
                break
            no_improve += 5
            continue
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0

                rho = min(rho * 1.3, 0.15)

def window_optimize(h, n, deadline, win_size=12, max_rounds=50):

                h_test = h.copy()
                h_test[s:e] = xw

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    c_vec_init = compute_c5_vec(h, n)
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        if sm > 500:
            return float(m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def eq_fun(x):
        return float(x.sum() - n / 2.0)
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def projected_gradient_descent(h, n, deadline, lr_init=0.01, decay=0.95):
