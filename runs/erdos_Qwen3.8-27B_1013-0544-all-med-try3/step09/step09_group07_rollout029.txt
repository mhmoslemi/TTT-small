```python
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

            if tot < 1e-15:
```

```python
                break
        
        if not improved_any and step < 0.0005:
            break
    
    return best_h, best_c
```

This should be fast because:
- No projection needed (sum is preserved by the move)
- Each evaluation is just one np.correlate call (~50μs for n=500)
- Systematic sweep of all positions

Let me estimate the time:
- Per position: ~50μs (correlate) + comparison = ~60μs
- Per lag: (n-m) * 60μs ≈ 500 * 60μs = 30ms
- Per top_k lags: 5 * 30ms = 150ms
- Per pass: 150ms
- Full sweep (all step sizes): ~10 passes average * 150ms = 1.5s per step size, 9 step sizes = ~13.5s

So the local search takes about 15-20 seconds per resolution. Very manageable!

Now let me think about the gradient optimization. L-BFGS-B with n=500 variables and a smooth max objective. Each function evaluation is:
- compute_all_lags: ~50μs
- smooth_max_grad: O(n²) = 500*999 ≈ 500,000 ops ≈ 0.5ms

So each L-BFGS-B iteration is about 1ms (function + gradient + line search overhead). With 2000 iterations, that's about 2 seconds. Very fast!

I can afford many more iterations. Let me use 5000-10000 iterations per alpha phase.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        wk = w[k] * dx
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            m = abs(lag)
            if m >= n:
                continue
            grad[m:] += wk * (1.0 - h[:n-m])
            grad[:n-m] += wk * (-h[m:])
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0

    def obj_and_grad(x):
        xc = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(xc, n)
        obj_val = smooth_max_value(c, alpha)
        diff = xc.sum() - target_sum
        obj_val += 1e4 * diff * diff
        grad = smooth_max_grad(xc, n, alpha)
        grad += 2.0 * 1e4 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        remaining = deadline - time.time()
        if remaining < 1:
            return project_feasible(h.copy(), n)
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def systematic_local_search(h, n, deadline, top_k=5):
    step_sizes = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for step in step_sizes:
        if time.time() > deadline:
            break
        improved_any = False
        max_passes = 10 if step > 0.005 else (20 if step > 0.001 else 40)
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-top_k:]
            improved = False

            for k_star in top_idx:
                if time.time() > deadline:
                    break
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue

                for i in range(n - m):
                    j = i + m
                    if h[i] > h[j] + 1e-10:
                        d = min(step, (h[i] - h[j]) * 0.25, h[i] * 0.4, (1.0 - h[j]) * 0.4)
                        if d < 1e-13:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
                    elif h[j] > h[i] + 1e-10:
                        d = min(step, (h[j] - h[i]) * 0.25, (1.0 - h[i]) * 0.4, h[j] * 0.4)
                        if d < 1e-13:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d

            if not improved:
                break

        if not improved_any and step < 0.0003:
            break

    return best_h, best_c

def focused_lag_descent(h, n, deadline, top_k=3, max_rounds=500):
    """Gradient descent focused only on the top k lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            # Compute gradient for just this lag
            grad_k = np.zeros(n)
            grad_k[m:] += (1.0 - h[:n-m]) * dx
            grad_k[:n-m] += (-h[m:]) * dx

            # Try moving in -gradient direction for each coordinate
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-10:
                    d = min(0.01, 0.001 * (h[i] - h[j]) / max(1e-10, abs(grad_k[i] - grad_k[j])))
                    d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                    if d < 1e-12:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
                elif h[j] > h[i] + 1e-10:
                    d = min(0.01, 0.001 * (h[j] - h[i]) / max(1e-10, abs(grad_k[i] - grad_k[j])))
                    d = min(d, (1.0 - h[i]) * 0.3, h[j] * 0.3)
                    if d < 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d

        if not improved:
            break

    return best_h, best_c

def simulated_annealing_structured(h, n, deadline, rng, alpha_sa=100000, t_init=0.005, t_min=0.00001, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.7:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.08, 0.003 + T * 5)
            if h[i] > h[j]:
                d = rng.uniform(0.001, max_step)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i]:
                d = -rng.uniform(0.001, max_step)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-3:]
            k_star = top_k[rng.randint(3)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-6:
                d = rng.uniform(0.001, 0.03)
                d = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-6:
                d = -rng.uniform(0.001, 0.03)
                d = max(d, -(h[j] - h[i]) * 0.3, -(1.0 - h[i]) * 0.3, -h[j] * 0.3)
            else:
                T *= cool
                continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        h[i] -= d
        h[j] += d
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = c_new.max()

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 50:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool

        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c
```
