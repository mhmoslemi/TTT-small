            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    n_active = int(np.sum(w > 1e-10))
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w[k] * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - x[0:n-lag])
            grad[0:n-lag] += w[k] * (-x[lag:n])
        else:
            ml = -lag
            grad[0:n-ml] += w[k] * (1.0 - x[ml:n])
            grad[ml:n] += w[k] * (-x[0:n-ml])
    return grad * dx

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def topk_objective(x, n, k_top):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    idx = np.argpartition(c, -k_top)[-k_top:]
    active = c[idx]
    m = active.max()
    alpha = 50.0
    shifted = alpha * (active - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def topk_gradient(x, n, k_top):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    idx = np.argpartition(c, -k_top)[-k_top:]
    active = c[idx]
    m = active.max()
    alpha = 50.0
    shifted = alpha * (active - m)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += w[ii] * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:n] += w[ii] * (1.0 - x[0:n-lag])
            grad[0:n-lag] += w[ii] * (-x[lag:n])
        else:
            ml = -lag
            grad[0:n-ml] += w[ii] * (1.0 - x[ml:n])
            grad[ml:n] += w[ii] * (-x[0:n-ml])
    return grad * dx

def lbfgsb_run(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    def obj(x):
        return lse_objective(x, n, alpha)
    def jac(x):
        return lse_gradient(x, n, alpha)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol,
                                                  'gtol': 1e-12, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_run(h, n, alpha, maxiter=5000, top_m=5):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    if top_m is not None:
        c = compute_all_lags(h, n)
        top_idx = np.argpartition(c, -top_m)[-top_m:]
        def obj(x):
            cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
            active = cc[top_idx]
            m = active.max()
            shifted = alpha * (active - m)
            mx = shifted.max()
            if mx > 500:
                return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
            return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
        def jac(x):
            cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
            active = cc[top_idx]
            m = active.max()
            shifted = alpha * (active - m)
            mx = shifted.max()
            if mx > 500:
                w = np.exp(shifted - mx)
            else:
                w = np.exp(shifted)
            w = w / w.sum()
            g = np.zeros(n)
            for ii, k in enumerate(top_idx):
                if w[ii] < 1e-10:
                    continue
                lag = k - (n - 1)
                if lag == 0:
                    g[:] += w[ii] * (1.0 - 2.0 * x)
                elif lag > 0:
                    g[lag:n] += w[ii] * (1.0 - x[0:n-lag])
                    g[0:n-lag] += w[ii] * (-x[lag:n])
                else:
                    ml = -lag
                    g[0:n-ml] += w[ii] * (1.0 - x[ml:n])
                    g[ml:n] += w[ii] * (-x[0:n-ml])
            return g * (2.0/n)
    else:
        obj = lambda x: lse_objective(x, n, alpha)
        jac = lambda x: lse_gradient(x, n, alpha)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def precompute_jacobian(h, n):
    """Compute J[k, p] = dC[k]/dh[p] * dx for all lags k and positions p."""
    dx = 2.0 / n
    J = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        lag = k - (n - 1)
        if lag == 0:
            J[k, :] = (1.0 - 2.0 * h) * dx
        elif lag > 0:
            J[k, lag:] += (1.0 - h[0:n-lag]) * dx
            J[k, 0:n-lag] += (-h[lag:]) * dx
        else:
            ml = -lag
            J[k, 0:n-ml] += (1.0 - h[ml:]) * dx
            J[k, ml:] += (-h[0:n-ml]) * dx
    return J

def efficient_local_search(h, n, deadline, max_rounds=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        J = precompute_jacobian(h, n)
        k_star = int(np.argmax(c_all))
        current_max = c_all[k_star]
        
        # Find all lags near the max
        threshold = current_max - 0.001
        near_max = c_all >= threshold
        near_max_count = int(np.sum(near_max))
        
        improved = False
        steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
        
        for delta in steps:
            if time.time() > deadline:
                break
            best_move = None
            best_delta_c = 0.0
            best_i, best_j = -1, -1
            
            # Try all pairs (i, j) where we move from j to i or i to j
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    # Moving delta from j to i
                    if h[j] < delta or h[i] > 1.0 - delta:
                        continue
                    # Change in c: delta * (J[:, i] - J[:, j])
                    dc = delta * (J[:, i] - J[:, j])
                    new_c = c_all + dc
                    new_max = float(np.max(new_c))
                    if new_max < current_max - 1e-15:
                        if best_move is None or new_max < best_delta_c:
                            best_move = 'j_to_i'
                            best_delta_c = new_max
                            best_i, best_j = i, j
                    # Moving delta from i to j
                    if h[i] < delta or h[j] > 1.0 - delta:
                        continue
                    dc = delta * (J[:, j] - J[:, i])
                    new_c = c_all + dc
                    new_max = float(np.max(new_c))
                    if new_max < current_max - 1e-15:
                        if best_move is None or new_max < best_delta_c:
                            best_move = 'i_to_j'
                            best_delta_c = new_max
                            best_i, best_j = i, j
            
            if best_move is not None:
                if best_move == 'j_to_i':
                    h[best_j] -= delta
                    h[best_i] += delta
                else:
                    h[best_i] -= delta
                    h[best_j] += delta
                h = np.clip(h, 0, 1)
                c_all = compute_all_lags(h, n)
                new_max = float(np.max(c_all))
                if new_max < best_c - 1e-15:
                    best_c = new_max
                    best_h = h.copy()
                    improved = True
                    c_all = compute_all_lags(h, n)
                    J = precompute_jacobian(h, n)
                    current_max = new_max
                    threshold = current_max - 0.001
                    if improved:
                        break  # Restart with smaller steps
            
            if best_move is None and delta > 0.001:
                # Try random pairs for this delta
                for _ in range(200):
                    if time.time() > deadline:
                        break
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                    if i == j:
                        continue
                    d = delta * np.random.uniform(0.5, 2.0)
                    if h[j] < d or h[i] > 1.0 - d:
                        continue
                    dc = d * (J[:, i] - J[:, j])
                    new_c = c_all + dc
                    if float(np.max(new_c)) < current_max - 1e-15:
                        h[best_j] = h[j] - d
                        h[best_i] = h[i] + d
                        h = np.clip(h, 0, 1)
                        c_all = compute_all_lags(h, n)
                        new_max = float(np.max(c_all))
                        if new_max < best_c - 1e-15:
                            best_c = new_max
                            best_h = h.copy()
                            improved = True
                        break
        
        if not improved:
            break
    
    return project_feasible(best_h, n), best_c

def surgical_local_search(h, n, deadline):
    """Very fine-tuned local search targeting the exact max lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        current_max = c_all[k_star]
        
        # Try very small adjustments
        found = False
        for delta in [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
            best_gain = 1e-15
            best_pair = None
            best_dir = None
            
            # Focus on positions that contribute most to the max lag
            lag = k_star - (n - 1)
            # Get positions involved in the max lag
            if lag > 0:
                positions = list(range(n - lag))
                pos2 = [p + lag for p in positions]
            elif lag < 0:
                ml = -lag
                positions = list(range(ml, n))
                pos2 = [p - ml for p in positions]
            else:
                positions = list(range(n))
                pos2 = list(range(n))
            
            for idx in range(len(positions)):
                if time.time() > deadline:
                    break
                i = positions[idx]
                j = pos2[idx]
                # Move from i to j (reduce h[i], increase h[j])
                d = min(delta, h[i] * 0.99, (1-h[j]) * 0.99)
                if d < 1e-10:
                    continue
                c_new = compute_c5_at_lag(h, n, i, j, d, k_star)
                if c_new < current_max - best_gain:
                    # Full check
                    h[i] -= d
                    h[j] += d
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-15:
                        best_gain = best_c - c_test
                        best_pair = (i, j, d, -1)
                        # Don't break, keep searching
                    else:
                        h[i] += d
                        h[j] -= d
                
                # Move from j to i
                if i != j:
                    d = min(delta, h[j] * 0.99, (1-h[i]) * 0.99)
                    if d < 1e-10:
                        continue
                    h[j] -= d
                    h[i] += d
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-15:
                        best_gain = best_c - c_test
                        best_pair = (i, j, d, 1)
                    else:
                        h[j] += d
                        h[i] -= d
            
            if best_pair is not None:
                i, j, d, direction = best_pair
                if direction == -1:
                    h[i] -= d
                    h[j] += d
                else:
                    h[j] -= d
                    h[i] += d
                h = np.clip(h, 0, 1)
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
                best_h = h.copy()
                found = True
                break
        
        if not found:
            break
    
    return best_h, best_c

def compute_c5_at_lag(h, n, i, j, d, k_star):
    """Quick estimate of c at k_star after moving d from i to j."""
    dx = 2.0 / n
    lag = k_star - (n - 1)
    if lag > 0:
        c = 0.0
        for p in range(n - lag):
            hi = h[p] - (d if p == i else 0.0) + (d if p == j else 0.0)
            hj = h[p + lag] - (d if p + lag == i else 0.0) + (d if p + lag == j else 0.0)
            c += hi * (1.0 - hj)
        return c * dx
    elif lag < 0:
        ml = -lag
        c = 0.0
        for p in range(ml, n):
            hi = h[p] - (d if p == i else 0.0) + (d if p == j else 0.0)
            hj = h[p - ml] - (d if p - ml == i else 0.0) + (d if p - ml == j else 0.0)
            c += hi * (1.0 - hj)
        return c * dx
    else:
        return None  # Lag 0 is always 0 for the max

def momentum_optimize(h, n, deadline, alphas=None, lr=1.5, momentum=0.9):
    if alphas is None:
        alphas = [30, 100, 300, 1000, 3000, 10000]
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    time_per_alpha = (deadline - time.time()) / len(alphas)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        v = np.zeros(n)
        phase_deadline = time.time() + time_per_alpha
        patience = 0
        lr_curr = lr
        
        while time.time() < phase_deadline:
            g = lse_gradient(h_curr, n, alpha)
            v = momentum * v + lr_curr * g
            h_new = project_feasible(h_curr - v, n)
            c_new = compute_c5(h_new, n)
            
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
            
            h_curr = h_new
            
            if patience > 500:
                lr_curr *= 0.5
                v *= 0.3
                patience = 0
                if lr_curr < 0.01:
                    break
    
    return best_h, best_c

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Momentum with increasing alpha (exploration)
    t_check = time.time()
    if deadline - t_check > 150:
        h_new, c_new = momentum_optimize(h, n, t_check + 120,
                                          alphas=[20, 50, 100, 300])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 2: L-BFGS-B refinement
    if time.time() < deadline - 100:
        for alpha in [50, 200, 1000]:
            if time.time() > deadline - 80:
                break
            h_new = lbfgsb_run(h, n, alpha, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            h = h_new
    
    # Phase 3: SLSQP on top lags
    if time.time() < deadline - 50:
        for top_m in [3, 5]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_run(h, n, 5000, maxiter=3000, top_m=top_m)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 4: Full SLSQP
    if time.time() < deadline - 30:
        h_new = slsqp_run(h, n, 20000, maxiter=5000, top_m=None)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Efficient local search
    if time.time() < deadline - 20:
        ls_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = efficient_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 6: Surgical refinement
    if time.time() < deadline - 3:
        h_new, c_new = surgical_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
