        val = m + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

            act = np.where(c_vec >= thr)[0]
            if len(act) > nshifts:

            grad = np.zeros(n)
            for mm in act:
                k = int(mm) - nrm
                if k >= 0:
                    nv = n - k
                    grad[:nv] += (1.0 - h[k:]) * dx_val
                    grad[k:] += (-h[:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:] += (1.0 - h[:nv]) * dx_val
                    grad[:nv] += (-h[kk:]) * dx_val
            if len(act) > 0:
                grad /= len(act)
            sel = np.argsort(-np.abs(grad))[:nidx]
            found = False
            for ai in range(len(sel)):
                if time.time() > deadline:
                    return h, best_c
                i = sel[ai]
                for bj in range(len(sel)):
                    if ai == bj:
                        continue
                    j = sel[bj]
                    for sgn in (1, -1):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                found = True
                            else:
                                h[i] -= sgn * d
                                h[j] += sgn * d
            if not found:
                break
    return h, best_c


def _upsample(h, n0, n1):
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n1)
    return np.interp(x1, x0, h)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _smooth_obj(x, n, alpha, mu, active):
    c_full = compute_c5_vec(x, n)
    c_sel = c_full[active]
    m = c_sel.max()
    sh = alpha * (c_sel - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0.0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def _smooth_grad(x, n, alpha, mu, active):
    c_full = compute_c5_vec(x, n)
    c_sel = c_full[active]
    m = c_sel.max()
    sh = alpha * (c_sel - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nrm = n - 1
    grad = np.zeros(n)
    for wi, mm in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(mm) - nrm
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
    if mu > 0.0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def _get_active(c_vec, top_k, tol):
    if top_k is None or top_k >= len(c_vec):
        return np.arange(len(c_vec))
    mx = c_vec.max()
    thr = mx - max(1e-8, tol * mx)
    idx = np.where(c_vec >= thr)[0]
    if len(idx) > top_k:
        idx = idx[np.argsort(-c_vec[idx])][:top_k]
    return np.sort(idx)


def _lbfgs_step(h, n, deadline, alpha, mu, top_k, tol, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    active = _get_active(c_vec, top_k, tol)
    try:

                       args=(n, alpha, mu, active),
                       method='L-BFGS-B',

                                'gtol': 1e-14, 'maxfun': 50000, 'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                    return h, best_c
                i = sel[ai]
                for bj in range(len(sel)):
                    if ai == bj:
                        continue
                    j = sel[bj]
                    for sgn in (1, -1):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                found = True
                            else:
                                h[i] -= sgn * d
                                h[j] += sgn * d
            if not found:
                break
    return h, best_c


def _upsample(h, n0, n1):
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n1)
    return np.interp(x1, x0, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        rng0 = np.random.RandomState(seed)
        h0 = np.clip(0.5 + rng0.randn(500) * 0.1, 0.0, 1.0)
        n0 = 500

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < dl - 700:
        t1 = min(t0 + 80, dl - 700)
        p1 = [
            (200, 0.01, None, 0.0, 3000),
            (2000, 0.005, 10, 0.001, 3000),
            (10000, 0.0, 5, 0.001, 3000),
            (100000, 0.0, 3, 0.001, 3000),
        ]
        for a, mu, tk, tol, mi in p1:
            if time.time() > t1 - 2:
                break
            hn = _lbfgs_step(best_h, n0, t1, a, mu, tk, tol, mi)
            cn = compute_c5(hn, n0)
            if cn < best_c:
                best_c = cn
                best_h = hn

    # Phase 2: Upsample to n=1000
    NW = 1000
    hw = _upsample(best_h, n0, NW)
    hw = project_feasible(hw, NW)
    cw = compute_c5(hw, NW)
    if cw < best_c:
        best_c = cw
        best_h = hw.copy()
        best_n = NW

    # Phase 3: Main L-BFGS at n=1000 (corrected gradient)
    if time.time() < dl - 350:
        t3 = min(time.time() + 480, dl - 350)
        p3 = [
            (50, 0.01, None, 0.0, 3000),
            (200, 0.01, None, 0.0, 3000),
            (500, 0.005, None, 0.0, 3000),
            (2000, 0.005, 10, 0.001, 4000),
            (5000, 0.005, 10, 0.001, 4000),
            (10000, 0.005, 5, 0.001, 4000),
            (50000, 0.0, 5, 0.001, 5000),
            (100000, 0.0, 5, 0.001, 5000),
            (500000, 0.0, 3, 0.001, 5000),
            (1000000, 0.0, 3, 0.001, 5000),
            (5000000, 0.0, 3, 0.001, 5000),
        ]
        for a, mu, tk, tol, mi in p3:
            if time.time() > t3 - 5:
                break
            hn = _lbfgs_step(hw, NW, t3, a, mu, tk, tol, mi)
            cn = compute_c5(hn, NW)
            if cn < cw - 1e-15:
                cw = cn
                hw = hn
            if cw < best_c:
                best_c = cw
                best_h = hw.copy()
                best_n = NW

    # Phase 4: Second focused round with tighter active set
    if time.time() < dl - 200:
        t4 = min(time.time() + 180, dl - 200)
        p4 = [
            (10000, 0.0, 8, 0.0005, 4000),
            (100000, 0.0, 5, 0.0005, 4000),
            (500000, 0.0, 4, 0.0005, 5000),
            (2000000, 0.0, 3, 0.0005, 5000),
            (5000000, 0.0, 3, 0.0005, 5000),
            (20000000, 0.0, 2, 0.0005, 5000),
        ]
        for a, mu, tk, tol, mi in p4:
            if time.time() > t4 - 3:
                break
            hn = _lbfgs_step(hw, NW, t4, a, mu, tk, tol, mi)
            cn = compute_c5(hn, NW)
            if cn < cw - 1e-15:
                cw = cn
                hw = hn
            if cw < best_c:
                best_c = cw
                best_h = hw.copy()
                best_n = NW

    # Phase 5: Third round – another full pass with best settings
    if time.time() < dl - 130:
        t5 = min(time.time() + 100, dl - 130)
        p5 = [
            (50000, 0.0, 5, 0.001, 5000),
            (500000, 0.0, 3, 0.001, 5000),
            (5000000, 0.0, 3, 0.0005, 5000),
        ]
        for a, mu, tk, tol, mi in p5:
            if time.time() > t5 - 3:
                break
            hn = _lbfgs_step(hw, NW, t5, a, mu, tk, tol, mi)
            cn = compute_c5(hn, NW)
            if cn < cw - 1e-15:
                cw = cn
                hw = hn
            if cw < best_c:
                best_c = cw
                best_h = hw.copy()
                best_n = NW

    # Phase 6: Perturbation restarts
    if time.time() < dl - 80:
        t6 = min(time.time() + 80, dl - 80)
        rng = np.random.RandomState(seed + 1)
        for trial in range(3):
            if time.time() > t6 - 20:
                break
            h_p = hw + rng.randn(NW) * 0.003
            h_p = project_feasible(h_p, NW)
            c_p = compute_c5(h_p, NW)
            if c_p < best_c + 0.003:
                hn = _lbfgs_step(h_p, NW, t6, 5000, 0.0, 5, 0.001, 2000)
                c_n = compute_c5(hn, NW)
                if c_n < c_p:
                    h_p, c_p = hn, c_n
                hn = _lbfgs_step(h_p, NW, t6, 500000, 0.0, 3, 0.001, 2000)
                c_n = compute_c5(hn, NW)
                if c_n < c_p:
                    h_p, c_p = hn, c_n
                if c_p < best_c:
                    best_c = c_p
                    best_h = h_p.copy()
                    best_n = NW
                    hw, cw = h_p.copy(), c_p

    # Phase 7: Gradient-guided local search
    if time.time() < dl - 35:
        t7 = min(time.time() + 60, dl - 35)
        hw, cw = _grad_local(hw, NW, t7, 5, 25,
                             [0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if cw < best_c:
            best_c = cw
            best_h = hw.copy()
            best_n = NW

    # Phase 8: Ultra-fine local search
    if time.time() < dl - 8:
        t8 = min(time.time() + 20, dl - 8)
        hw, cw = _grad_local(hw, NW, t8, 3, 15,
                             [0.0001, 0.00005, 0.00002, 0.00001])
        if cw < best_c:
            best_c = cw
            best_h = hw.copy()
            best_n = NW

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
