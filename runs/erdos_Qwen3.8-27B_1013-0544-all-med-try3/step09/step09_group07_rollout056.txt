        for ii in order[:num_ones - target]:
            h_bin[idx[ii]] = 0
    elif num_ones < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        for ii in order[:target - num_ones]:
            h_bin[idx[ii]] = 1

    # Blend: h_blend = (1-t)*h + t*h_bin for t in [0,1], pick best
    best_res = None
    best_c = compute_c5(h, n)
    for t in [0.3, 0.5, 0.7, 1.0]:
        h_blend = (1 - t) * h + t * h_bin
        h_blend = project_feasible(h_blend, n)
        h_opt = lbfgsb_fast(h_blend, n, 50000, min(deadline, time.time() + 25), maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_res = h_opt
    if best_res is not None:
        return best_res
    return h

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing alpha (fast gradient)
    alpha_schedule = [(5000, 800), (20000, 1000), (100000, 1200), (500000, 1500)]
    for alpha, maxit in alpha_schedule:
        if time.time() > deadline - 500:
            break
        t_alloc = min(40, (deadline - 500 - time.time()))
        if t_alloc < 5:
            break
        h_opt = lbfgsb_fast(best_h, n, alpha, time.time() + t_alloc, maxiter=maxit, top_k=20)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Targeted local search
    if time.time() < deadline - 350:
        ls_deadline = min(deadline - 250, time.time() + 180)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, rng, K=12)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Simulated annealing
    if time.time() < deadline - 250:
        sa_deadline = min(deadline - 180, time.time() + 200)
        h_opt, c_opt = simulated_annealing_fast(best_h, n, sa_deadline, rng,
                                                  alpha_sa=50000, t_init=0.005,
                                                  t_min=0.00005, cool=0.9999)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish
    if time.time() < deadline - 150:
        for alpha in [200000, 2000000, 20000000]:
            if time.time() > deadline - 60:
                break
            h_opt = lbfgsb_fast(best_h, n, alpha, time.time() + 25, maxiter=2000, top_k=15)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Binary snap and relax
    if time.time() < deadline - 120:
        h_opt = binary_snap_and_relax(best_h, n, deadline, rng)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            # Polish the improved binary-relaxed solution
            h_opt2 = lbfgsb_fast(best_h, n, 500000, time.time() + 20, maxiter=1500)
            c_opt2 = compute_c5(h_opt2, n)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2

    # Phase 6: Try higher resolution
    for n_try in [600, 800]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - 80 - time.time()
        if remaining < 50:
            break
        h_interp = expand_resolution(best_h, len(best_h), n_try)
        # Optimize at new resolution
        h_opt = lbfgsb_fast(h_interp, n_try, 20000, time.time() + min(30, remaining * 0.35), maxiter=1200, top_k=25)
        h_opt = lbfgsb_fast(h_opt, n_try, 500000, time.time() + min(20, remaining * 0.25), maxiter=1500, top_k=20)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            # Quick local search at new resolution
            if time.time() < deadline - 40:
                h_opt2, c_opt2 = targeted_local_search(best_h, n_try, min(deadline - 10, time.time() + 40), rng, K=10)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 7: Another SA round
    if time.time() < deadline - 60:
        n_final = len(best_h)
        sa_dl = min(deadline - 20, time.time() + 80)
        h_opt, c_opt = simulated_annealing_fast(best_h, n_final, sa_dl, rng,
                                                  alpha_sa=100000, t_init=0.003,
                                                  t_min=0.00003, cool=0.99995)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 15:
        n_final = len(best_h)
        h_opt = lbfgsb_fast(best_h, n_final, 50000000, deadline - 8, maxiter=3000, top_k=10)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final targeted local search
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt, c_opt = targeted_local_search(best_h, n_final, max(deadline - 1, time.time() + 15), rng, K=15)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
