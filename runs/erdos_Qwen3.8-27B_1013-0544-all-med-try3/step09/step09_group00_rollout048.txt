        pos = int(k * alpha_par) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
        k += 1
    rem = n / 2.0 - h.sum()
    if abs(rem) > 1e-10 and len(seen) < n:
        zeros = [i for i in range(n) if i not in seen]
        if zeros:
            h[zeros[0]] += rem
    return project_feasible(h, n)

def make_random_bin(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

def lbfgs_round(h, n, alpha, mu, maxit, deadline):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(maxit, max(100, int(rem * 3)))
    try:
        res = minimize(obj_soft, h, jac=grad_soft,
                       args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'maxfun': maxit * 8,
                                'ftol': 1e-24, 'gtol': 1e-18, 'maxls': 40,
                                'maxcor': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def active_set_round(h, n, top_k, alpha, mu, maxit, deadline):
    c = compute_c5_vec(h, n)
    mx = c.max()
    thresh = mx - max(1e-9, 0.002 * mx)
    active = np.where(c >= thresh)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c[active])[:top_k])
    if len(active) == 0:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(maxit, max(100, int(rem * 2)))
    try:
        res = minimize(obj_topk, h, jac=grad_topk,
                       args=(n, active, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'maxfun': maxit * 6,
                                'ftol': 1e-24, 'gtol': 1e-18, 'maxls': 30,
                                'maxcor': 40})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def pair_swap_optimize(h, n, deadline, max_rounds=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        mx = c_vec.max()
        tol = max(1e-10, 0.005 * mx)
        active = np.where(c_vec >= mx - tol)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= max(1, len(active))
        pos = np.where((grad > 1e-10) & (best_h > 1e-8))[0]
        neg = np.where((grad < -1e-10) & (best_h < 1.0 - 1e-8))[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        best_gain = 0.0
        best_i, best_j = -1, -1
        for i in pos[:30]:
            for j in neg[:30]:
                g = grad[i] - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i, best_j = i, j
        if best_i < 0 or best_gain < 1e-12:
            break
        amt = min(best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-10:
            break
        amt = min(amt, best_gain * 0.5)
        old_i, old_j = best_h[best_i], best_h[best_j]
        best_h[best_i] = old_i - amt
        best_h[best_j] = old_j + amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
        else:
            best_h[best_i] = old_i
            best_h[best_j] = old_j
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def quantize_push(h, n, deadline, n_frac_allowed=10):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for _ in range(500):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        mx = c_vec.max()
        frac = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        n_frac = int(frac.sum())
        if n_frac <= n_frac_allowed:
            break
        idx = np.where(frac)[0]
        improved = False
        for i in idx:
            time_el = time.time()
            if time_el > deadline:
                break
            for target_val in [0.0, 1.0]:
                old = best_h[i]
                best_h[i] = target_val
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] = old
                    best_h = project_feasible(best_h, n)
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c

def multi_resolution_optimize(seed, budget_s):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    # Determine starting point
    resolutions = [100, 200, 400, 500, 800]
    
    # Try parent first
    best_overall_h = None
    best_overall_c = 999.0
    best_overall_n = 0
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        h0 = project_feasible(h0, n0)
        c0 = compute_c5(h0, n0)
        best_overall_h = h0.copy()
        best_overall_c = c0
        best_overall_n = n0
        if n0 in resolutions:
            resolutions.remove(n0)
        resolutions.append(n0)
        resolutions = sorted(set(resolutions))
    except Exception:
        pass
    
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    sqrt2 = np.sqrt(2.0)
    
    # Quick scan over resolutions
    per_res_time = max(80, int((budget_s - 50) / len(resolutions)))
    
    for n in resolutions:
        if time.time() > deadline - 60:
            break
        res_deadline = min(time.time() + per_res_time, deadline - 50)
        
        # Multiple initializations
        starts = []
        starts.append(make_beatty(n, golden / 2.0))
        starts.append(make_beatty(n, golden / 3.0))
        starts.append(make_beatty(n, sqrt2 / 2.0))
        starts.append(make_random_bin(n, rng))
        starts.append(make_random_bin(n, rng))
        starts.append(np.full(n, 0.5))
        
        # If we have a parent at a different resolution, interpolate
        if best_overall_h is not None and len(best_overall_h) != n:
            h_interp = interpolate_to_resolution(best_overall_h, len(best_overall_h), n)
            starts.append(h_interp)
        
        best_h_res = starts[0]
        best_c_res = compute_c5(starts[0], n)
        
        for s in starts:
            c_s = compute_c5(s, n)
            if c_s < best_c_res:
                best_c_res = c_s
                best_h_res = s.copy()
        
        # Quick L-BFGS warmup
        alphas_warm = [10, 50, 200, 1000]
        for a in alphas_warm:
            if time.time() > res_deadline - 5:
                break
            h_tmp, c_tmp = lbfgs_round(best_h_res, n, a, 0.1, 300, res_deadline)
            if c_tmp < best_c_res:
                best_c_res = c_tmp
                best_h_res = h_tmp
        
        # Track best
        if best_c_res < best_overall_c:
            best_overall_c = best_c_res
            best_overall_h = best_h_res.copy()
            best_overall_n = n
    
    return best_overall_h, best_overall_c, best_overall_n

def focused_optimize(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Smooth max L-BFGS with increasing alpha
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        h_tmp, c_tmp = lbfgs_round(best_h, n, a, 0.01, 2000, deadline)
        if c_tmp < best_c - 1e-15:
            best_c = c_tmp
            best_h = h_tmp
    
    # Phase 2: Active set refinement (multiple rounds)
    for top_k in [3, 5, 8, 12, 20]:
        for _ in range(3):
            if time.time() > deadline - 30:
                break
            h_tmp, c_tmp = active_set_round(best_h, n, top_k,
                                            alpha=1000000, mu=0.001,
                                            maxit=1500, deadline=deadline)
            if c_tmp < best_c - 1e-15:
                best_c = c_tmp
                best_h = h_tmp
    
    # Phase 3: Pair swap optimization
    if time.time() < deadline - 60:
        h_tmp, c_tmp = pair_swap_optimize(best_h, n, deadline - 50, max_rounds=5000)
        if c_tmp < best_c - 1e-15:
            best_c = c_tmp
            best_h = h_tmp
    
    # Phase 4: Another round of active set
    for top_k in [3, 5]:
        for _ in range(2):
            if time.time() > deadline - 30:
                break
            h_tmp, c_tmp = active_set_round(best_h, n, top_k,
                                            alpha=5000000, mu=0.0005,
                                            maxit=2000, deadline=deadline)
            if c_tmp < best_c - 1e-15:
                best_c = c_tmp
                best_h = h_tmp
    
    # Phase 5: Quantization push
    if time.time() < deadline - 20:
        h_tmp, c_tmp = quantize_push(best_h, n, deadline - 10, n_frac_allowed=6)
        if c_tmp < best_c - 1e-15:
            best_c = c_tmp
            best_h = h_tmp
    
    # Phase 6: Final pair swap
    if time.time() < deadline - 15:
        h_tmp, c_tmp = pair_swap_optimize(best_h, n, deadline - 8, max_rounds=2000)
        if c_tmp < best_c - 1e-15:
            best_c = c_tmp
            best_h = h_tmp
    
    return best_h, best_c

def fine_tune_pairs(h, n, deadline, radius=30, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    c_vec = compute_c5_vec(best_h, n)
    mx = c_vec.max()
    tol = max(1e-9, 0.003 * mx)
    active_shifts = np.where(c_vec >= mx - tol)[0]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Focus on variables that affect active shifts
            affected = set()
            for idx in active_shifts:
                k = idx - (n - 1)
                if k >= 0:
                    affected.update(range(k))
                    affected.update(range(k, n))
                else:
                    kk = -k
                    affected.update(range(kk))
                    affected.update(range(n - kk))
            
            candidates = sorted(affected, key=lambda x: (best_h[x] < 0.1) + (2.0 if best_h[x] > 0.9 else 0))
            candidates = candidates[:min(len(candidates), 100)]
            
            for i in candidates:
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = project_feasible(best_h, n)
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
                    # Try moving mass from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        best_h = project_feasible(best_h, n)
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Phase A: Multi-resolution scan (use ~40% of budget)
    scan_deadline = t0 + int(budget_s * 0.35)
    best_h, best_c, best_n = multi_resolution_optimize(seed, scan_deadline - t0)
    
    if best_h is None:
        # Fallback
        n = 500
        best_h = make_random_bin(n, np.random.RandomState(seed))
        best_c = compute_c5(best_h, n)
        best_n = n
    
    # Phase B: Focused optimization at best resolution (use ~45% of budget)
    if time.time() < deadline - 60:
        focus_deadline = min(time.time() + int(budget_s * 0.45), deadline - 50)
        h_f, c_f = focused_optimize(best_h, best_n, focus_deadline)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    # Phase C: Try another resolution with interpolated best (use ~10% of budget)
    if time.time() < deadline - 40:
        for n_alt in [best_n // 2, best_n * 2]:
            if n_alt < 50 or n_alt > 4000 or time.time() > deadline - 30:
                break
            h_alt = interpolate_to_resolution(best_h, best_n, n_alt)
            c_alt = compute_c5(h_alt, n_alt)
            if c_alt < best_c:
                # Optimize at this resolution
                alt_deadline = min(time.time() + 60, deadline - 25)
                h_ai, c_ai = focused_optimize(h_alt, n_alt, alt_deadline)
                if c_ai < best_c:
                    best_c = c_ai
                    best_h = h_ai
                    best_n = len(h_ai)
    
    # Phase D: Fine-tuning (remaining time)
    if time.time() < deadline - 15:
        ft_deadline = deadline - 8
        h_ft, c_ft = fine_tune_pairs(best_h, best_n, ft_deadline,
                                     radius=25,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001,
                                             0.00005, 0.00002, 0.00001])
        if c_ft < best_c:
            best_c = c_ft
            best_h = h_ft
    
    # Final very fine adjustments
    if time.time() < deadline - 5:
        h_fv, c_fv = pair_swap_optimize(best_h, best_n, deadline - 2, max_rounds=3000)
        if c_fv < best_c:
            best_c = c_fv
            best_h = h_fv
    
    # Final projection and validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
