def binary_flips(h, n, deadline, n_rounds=200):

    best_c = compute_c5(h, n)
    best_h = h.copy()

            break
        improved = False

        # Try flipping a 1 to 0 and a 0 to 1
        ones = np.where(h > 0.5)[0]

    return x.sum() - n / 2.0

def slqp_con_jac(x, n):
    return np.ones(n)

def run_slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):
    try:
        res = minimize(obj_f, h, jac=grad_f, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=[(0, 1)] * n,
                      constraints=[{'type': 'eq', 'fun': slqp_con_fun, 'jac': slqp_con_jac, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except:
        return project_feasible(h.copy(), n), compute_c5(h, n)

            h_new = project_feasible(res.x, n)
