    return best_h, best_c

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, int(idx))
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[int(idx)]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.04):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-9, 0.002 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.4, 0.08)
        else:
            consecutive_no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if consecutive_no_improve > 25:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (1000, 500.0), (3000, 5000.0), (8000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, h_old[:1]]), period=2.0)
    h_new = project_feasible(h_new, n_new)
    return h_new

def get_initial():
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        return h_init, n_init
    except Exception:
        return None, 0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h_init, n_init = get_initial()
    if h_init is None:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Quick L-BFGS at initial n (30s)
    t1 = min(t0 + 30, deadline - 900)
    if t1 > time.time() + 3:
        h1, c1 = run_lbfgs(best_h, best_n, t1, configs=[(300, 100.0), (1500, 1000.0)])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=1000
    n_mid = 1000
    if time.time() < deadline - 500:
        h_mid = upsample_h(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)

        # L-BFGS at n=1000 (120s)
        t2 = min(time.time() + 120, deadline - 380)
        if t2 > time.time() + 5:
            h_mid, c_mid = run_lbfgs(h_mid, n_mid, t2,
                configs=[(300, 50.0), (1000, 500.0), (4000, 5000.0), (10000, 20000.0)])

        # SQP at n=1000 (80s)
        t3 = min(time.time() + 80, deadline - 280)
        if t3 > time.time() + 5:
            h_mid, c_mid = sqp_optimize(h_mid, n_mid, t3, max_iter=300, rho_init=0.03)

        # Subgradient at n=1000 (40s)
        t4 = min(time.time() + 40, deadline - 220)
        if t4 > time.time() + 3:
            h_sg, c_sg = subgradient_optimize(h_mid, n_mid, t4, max_iter=3000)
            if c_sg < c_mid:
                h_mid, c_mid = h_sg, c_sg

        # Local search at n=1000 (30s)
        t5 = min(time.time() + 30, deadline - 180)
        if t5 > time.time() + 3:
            h_mid, c_mid = targeted_local_search(h_mid, n_mid, t5, center_radius=25)

        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
            best_n = n_mid

    # Phase 3: Upsample to n=2000
    n_high = 2000
    if time.time() < deadline - 200:
        h_high = upsample_h(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)

        # L-BFGS at n=2000 (180s)
        t6 = min(time.time() + 180, deadline - 100)
        if t6 > time.time() + 5:
            h_high, c_high = run_lbfgs(h_high, n_high, t6,
                configs=[(500, 100.0), (2000, 1000.0), (8000, 10000.0), (20000, 50000.0)])

        # SQP at n=2000 (60s)
        t7 = min(time.time() + 60, deadline - 50)
        if t7 > time.time() + 5:
            h_high, c_high = sqp_optimize(h_high, n_high, t7, max_iter=150, rho_init=0.02)

        # Subgradient at n=2000 (50s)
        t8 = min(time.time() + 50, deadline - 30)
        if t8 > time.time() + 3:
            h_sg, c_sg = subgradient_optimize(h_high, n_high, t8, max_iter=2000)
            if c_sg < c_high:
                h_high, c_high = h_sg, c_sg

        # Local search at n=2000 (30s)
        t9 = min(time.time() + 30, deadline - 10)
        if t9 > time.time() + 3:
            h_high, c_high = targeted_local_search(h_high, n_high, t9, center_radius=20)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high

    # Final: try to polish at current best resolution
    if time.time() < deadline - 10:
        n_final_try = best_n
        h_final = project_feasible(best_h, n_final_try)
        c_final = compute_c5(h_final, n_final_try)
        t_final = deadline - 5
        if t_final > time.time() + 5:
            h_final, c_final = run_lbfgs(h_final, n_final_try, t_final,
                configs=[(5000, 5000.0), (15000, 30000.0)])
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
            best_n = n_final_try

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
