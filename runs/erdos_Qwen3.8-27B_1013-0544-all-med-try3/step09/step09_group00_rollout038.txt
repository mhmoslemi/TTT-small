def swap_effect(h, i, j, n):
    """Compute change in correlation when h[i]:1->0 and h[j]:0->1"""
    # For each lag ell from -(n-1) to (n-1), compute the change
    # C[ell] = sum_{j: 0<=j<n, 0<=j+ell<n} h[j] * (1-h[j+ell])
    # After swap: h'[i]=0, h'[j]=1, rest same
    # Change in C[ell] = sum of changes at positions involving i or j
    
    eff = np.zeros(2*n-1)  # indexed by ell + (n-1)
    
    for ell in range(-(n-1), n):
        # Positions where h[i] appears in C[ell]:
        # 1. As first factor: j=i, need 0<=i<n and 0<=i+ell<n
        # 2. As second factor: j+ell=i, need 0<=i-ell<n and 0<=i<n
        # Similarly for h[j]
        
        delta = 0
        
        # h[i]: 1->0
        # First factor: was 1*(1-h[i+ell]), now 0. Change: -(1-h[i+ell]) if valid
        if 0 <= i + ell < n:
            delta -= (1 - h[i+ell])
        # Second factor: h[j']*(1-h[i]) where j'+ell=i, j'=i-ell
        # Was h[i-ell]*(1-1)=0, now h[i-ell]*(1-0)=h[i-ell]. Change: +h[i-ell] if valid
        if 0 <= i - ell < n:
            delta += h[i-ell]
        
        # h[j]: 0->1
        # First factor: was 0*(1-h[j+ell])=0, now 1*(1-h[j+ell]). Change: +(1-h[j+ell]) if valid
        if 0 <= j + ell < n:
            delta += (1 - h[j+ell])
        # Second factor: h[j']*(1-h[j]) where j'+ell=j, j'=j-ell
        # Was h[j-ell]*(1-0)=h[j-ell], now h[j-ell]*(1-1)=0. Change: -h[j-ell] if valid
        if 0 <= j - ell < n:
            delta -= h[j-ell]
        
        eff[ell + n - 1] = delta
    
    return eff

def make_beatty_binary(n, alpha):

            E_on[j, m4] -= h_curr[jme[m4]]
        return E_off, E_on
    
    E_off, E_on = get_effects(h)
    best_h = h.copy()
    best_c = current_c
    
    tabu_i = np.zeros(n, dtype=int)  # iteration until which i is tabu
    tabu_j = np.zeros(n, dtype=int)
    current_iter = 0
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    rng = np.random.RandomState(777)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        current_iter = it
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find best non-tabu swap
        best_max_val = current_c + 10  # large
        best_i = -1
        best_j = -1
        
        for ii in range(len(ones)):
            i = ones[ii]
            if tabu_i[i] > current_iter:
                continue
            base = c + E_off[i]
            for jj in range(len(zeros)):
                j = zeros[jj]
                if tabu_j[j] > current_iter:
                    continue
                new_max = (base + E_on[j]).max()
                if new_max < best_max_val:
                    best_max_val = new_max
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            # All moves are tabu or no improvement; allow tabu violation for best move
            for ii in range(len(ones)):
                i = ones[ii]
                base = c + E_off[i]
                for jj in range(len(zeros)):
                    j = zeros[jj]
                    new_max = (base + E_on[j]).max()
                    if new_max < best_max_val:
                        best_max_val = new_max
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
        
        # Apply move
        h[best_i] = 0.0
        h[best_j] = 1.0
        tabu_i[best_i] = current_iter + tabu_len
        tabu_j[best_j] = current_iter + tabu_len
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        c = np.correlate(h, 1.0 - h, mode='full')
        current_c = c.max()
        E_off, E_on = get_effects(h)
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
    
    return best_h, best_c

def multi_start_search(n, deadline, seed=42):
    """Try multiple initializations at given resolution"""
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = 1.0
    
    inits = []
    # Beatty sequences
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    inits.append(make_beatty_binary(n, golden / 2.0))
    inits.append(make_beatty_binary(n, np.sqrt(2.0) / 2.0))
    inits.append(make_beatty_binary(n, np.sqrt(3.0) / 2.0))
    inits.append(make_beatty_binary(n, 0.381966))  # 1/golden^2
    inits.append(make_beatty_binary(n, 0.618034))  # 1/golden
    # Random
    for _ in range(4):
        h_r = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_r[idx] = 1.0
        inits.append(h_r)
    
    for init in inits:
        if time.time() > deadline - 60:
            break
        c_init = compute_c5(init, n)
        if c_init < best_c:
            h_g, c_g = greedy_binary_search(init, n, min(time.time() + 40, deadline - 30), max_sweeps=20)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g
            else:
                if c_init < best_c:
                    best_c = c_init
                    best_h = init.copy()
        elif best_h is None:
            best_c = c_init
            best_h = init.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    best_h = None
    best_c = 1.0
    best_n = 0
    
    # Try to use parent
    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        pass
    
    if parent_h is not None:
        # Round parent to binary
        h_bin = make_binary(parent_h, parent_n)
        c_bin = compute_c5(h_bin, parent_n)
        c_cont = compute_c5(parent_h, parent_n)
        
        if c_cont < best_c:
            best_c = c_cont
            best_h = parent_h.copy()
            best_n = parent_n
        
        # Also try the binary version
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
            best_n = parent_n
        
        # Run greedy on the binary parent
        if time.time() < deadline - 100:
            h_g, c_g = greedy_binary_search(h_bin, parent_n, min(time.time() + 120, deadline - 100), max_sweeps=50)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g
                best_n = parent_n
        
        # Tabu search on best binary
        if time.time() < deadline - 80:
            h_t, c_t = tabu_search(best_h, best_n, min(time.time() + 80, deadline - 60), tabu_len=50, max_iter=30000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = best_n
        
        # Kick and search
        if time.time() < deadline - 40:
            h_k, c_k = kick_and_search(best_h, best_n, min(time.time() + 60, deadline - 30), n_kicks=10, kick_size=2)
            if c_k < best_c:
                best_c = c_k
                best_h = h_k
                best_n = best_n
    
    # Try different resolutions with multi-start
    resolutions = []
    if parent_n > 0:
        resolutions = [parent_n, parent_n, parent_n + 50, parent_n + 100, parent_n - 50]
    else:
        resolutions = [400, 500, 600, 700]
    
    # Add some new resolutions
    resolutions.append(600)
    resolutions.append(800)
    
    # Deduplicate and filter
    resolutions = sorted(set([r for r in resolutions if r >= 200]))
    
    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        if n_try == best_n and parent_h is not None:
            continue  # Already tried this resolution with parent
        
        # Multi-start at this resolution
        h_ms, c_ms = multi_start_search(n_try, min(time.time() + 60, deadline - 60), seed=seed + n_try)
        if h_ms is not None and c_ms < best_c:
            best_c = c_ms
            best_h = h_ms
            best_n = n_try
    
    # If we found something at a new resolution, do more optimization
    if time.time() < deadline - 60:
        h_final, c_final = greedy_binary_search(best_h, best_n, min(time.time() + 40, deadline - 50), max_sweeps=30)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Simulated annealing for escape
    if time.time() < deadline - 30:
        h_sa, c_sa = simulated_annealing(best_h, best_n, min(time.time() + 30, deadline - 25), max_iter=20000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Final polish
    if time.time() < deadline - 5:
        h_pol, c_pol = greedy_binary_search(best_h, best_n, min(time.time() + 5, deadline - 2), max_sweeps=10)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
