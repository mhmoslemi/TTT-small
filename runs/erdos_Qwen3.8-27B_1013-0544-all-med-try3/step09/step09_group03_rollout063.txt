import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_t(A, B):
    """Find t in [0,1] minimizing max_k (A[k] + t*B[k])."""
    n_s = len(A)
    if n_s == 0:
        return 0.5
    pos = B > 1e-15
    neg = B < -1e-15
    zero = ~pos & ~neg
    
    if not np.any(pos):
        return 1.0
    if not np.any(neg):
        return 0.0
    
    g0 = np.max(A[pos])
    h0 = np.max(np.where(zero, A, -np.inf)) if np.any(zero) else -np.inf
    hn = np.max(A[neg])
    h0 = max(h0, hn)
    
    if g0 >= h0:
        return 0.0
    
    g1 = np.max(A[pos] + B[pos])
    hn1 = np.max(A[neg] + B[neg])
    h1 = max(hn1, h0) if np.any(zero) else hn1
    
    if g1 <= h1:
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        gm = np.max(A[pos] + mid * B[pos])
        hm_neg = np.max(A[neg] + mid * B[neg])
        hm = max(hm_neg, h0) if np.any(zero) else hm_neg
        if gm > hm:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent_h(h, n, n_sweeps=500, deadline=None, verbose=False):
    """Coordinate descent: optimize each h[i] to minimize max correlation."""
    c = compute_c5_vec(h, n)
    best_c = float(np.max(c))
    target = n / 2.0
    n_shifts = 2 * n - 1
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline - 5:
            break
        for i in range(n):
            # Compute B_k for coordinate i
            B = np.zeros(n_shifts)
            for k in range(-n + 1, n):
                idx = k + n - 1
                # h[i] appears as: (1-h[i+k]) if 0<=i+k<n, minus h[i-k] if 0<=i-k<n
                ik = i + k
                i_k = i - k
                bval = 0.0
                if 0 <= ik < n:
                    bval += (1.0 - h[ik])
                if 0 <= i_k < n:
                    bval -= h[i_k]
                B[idx] = bval
            
            # A_k = c[k] - h[i] * B[k]
            A = c - h[i] * B
            
            # Find optimal t
            t_star = find_optimal_t(A, B)
            
            # Update c and h
            delta = t_star - h[i]
            if abs(delta) > 1e-15:
                c += delta * B
                h[i] = t_star
        
        # Project to maintain sum
        h = project_feasible(h, n)
        c = compute_c5_vec(h, n)
        new_max = float(np.max(c))
        if new_max < best_c:
            best_c = new_max
    
    return h, best_c

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (5000, 3000.0), (10000, 8000.0), (30000, 20000.0),
                   (80000, 50000.0)]

        try:

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_sa(h, n, deadline, n_iters=50000):

    rng = np.random.RandomState(777)
    h_b = np.where(h > 0.5, 1.0, 0.0)
    # Fix sum
    ones = np.sum(h_b == 1.0)
    target = n // 2

        idx_1 = np.where(h_b == 1.0)[0]
        remove = idx_1[rng.choice(len(idx_1), size=ones - target, replace=False)]
        h_b[remove] = 0.0
    elif ones < target:
        idx_0 = np.where(h_b == 0.0)[0]
        add = idx_0[rng.choice(len(idx_0), size=target - ones, replace=False)]
        h_b[add] = 1.0

        if time.time() > deadline - 30:

        idx_0 = np.where(h_b == 0.0)[0]
        if len(idx_1) == 0 or len(idx_0) == 0:
            break
        i = idx_1[rng.randint(len(idx_1))]
        j = idx_0[rng.randint(len(idx_0))]
        h_b[i], h_b[j] = 0.0, 1.0

                best_c = c_cd
                best_h = h_cd
    
    # Phase 8: Final targeted refinement (30s)
    if time.time() < deadline - 10:
        best_h, best_c = targeted_worst_shift(best_h, len(best_h), deadline - 3, n_rounds=100)
    
    # Final: one more coord descent pass
    if time.time() < deadline - 3:
        nf = len(best_h)
        h_final, c_final = coord_descent_h(best_h, nf, n_sweeps=200, deadline=deadline)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
