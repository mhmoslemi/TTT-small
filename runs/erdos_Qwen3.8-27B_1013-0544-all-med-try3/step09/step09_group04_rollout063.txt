    best_c = corr.max()
    dx = 2.0 / n
    one_minus_h = 1.0 - best_h

        improved = True

            improved = False

def point_change_dc(h, one_minus_h, n, i, delta):
    """Compute change in correlation vector (with dx factor) for h[i] += delta."""
    n_corr = 2 * n - 1
    dc = np.zeros(n_corr)
    dx = 2.0 / n
    
    # Term 1: delta * (1-h)[i + t - (n-1)] for valid indices
    lo1 = max(0, (n-1) - i)
    hi1 = min(n_corr-1, (n-1) - i + n - 1)
    if lo1 <= hi1:
        idx = i + np.arange(lo1, hi1+1) - (n-1)
        dc[lo1:hi1+1] += delta * one_minus_h[idx]
    
    # Term 2: -delta * h[i - t + (n-1)] for valid indices
    lo2 = max(0, i)
    hi2 = min(n_corr-1, i + n - 1)
    if lo2 <= hi2:
        idx = i - np.arange(lo2, hi2+1) + (n-1)
        dc[lo2:hi2+1] += -delta * h[idx]
    
    return dc * dx

    n_corr = 2 * n - 1

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    val = m + lse / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for t in range(len(w)):
        if w[t] < 1e-14:
            continue
        k = t - (n - 1)
        wm = w[t] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        lse = smx + np.log(np.sum(np.exp(sh - smx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    val = mx + lse / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def point_dc(h, one_minus_h, n, i, delta):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    dc = np.zeros(n_corr)
    lo1 = max(0, (n - 1) - i)
    hi1 = min(n_corr - 1, (n - 1) - i + n - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + i - (n - 1)
        dc[lo1:hi1 + 1] += delta * one_minus_h[idx]
    lo2 = max(0, i)
    hi2 = min(n_corr - 1, i + n - 1)
    if lo2 <= hi2:
        idx = i - np.arange(lo2, hi2 + 1) + (n - 1)
        dc[lo2:hi2 + 1] += -delta * h[idx]
    return dc * dx

def fast_pair_search(h, n, deadline, radius=80):
    best_h = project_feasible(h.copy(), n)
    corr = compute_c5_vec(best_h, n)
    best_c = corr.max()
    one_minus_h = 1.0 - best_h
    n_corr = 2 * n - 1

    deltas = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    rng = np.random.RandomState(42)

    for d in deltas:
        max_passes = 20
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for jj in range(lo, hi):
                    if i == jj:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[jj] - d >= -1e-12:
                        dc = point_dc(best_h, one_minus_h, n, i, d) + point_dc(best_h, one_minus_h, n, jj, -d)
                        new_max = (corr + dc).max()
                        if new_max < best_c - 1e-14:
                            best_h[i] += d
                            best_h[jj] -= d
                            best_h = np.clip(best_h, 0, 1)
                            one_minus_h = 1.0 - best_h
                            corr += dc
                            best_c = new_max
                            improved = True
                    if best_h[i] - d >= -1e-12 and best_h[jj] + d <= 1.0 + 1e-12:
                        dc = point_dc(best_h, one_minus_h, n, i, -d) + point_dc(best_h, one_minus_h, n, jj, d)
                        new_max = (corr + dc).max()
                        if new_max < best_c - 1e-14:
                            best_h[i] -= d
                            best_h[jj] += d
                            best_h = np.clip(best_h, 0, 1)
                            one_minus_h = 1.0 - best_h
                            corr += dc
                            best_c = new_max
                            improved = True
            if not improved:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.005, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = max(50, maxiters[i])
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 8,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000]
    gen = 0
    while time.time() < deadline:
        gen += 1
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        remaining = deadline - time.time()
        if remaining < 8:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(50, int((deadline - time.time()) * 0.8))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': maxit * 8,
                                        'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def de_lowres(n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    pop = []
    base = np.full(n, 0.5)
    pop.append(base.copy())
    for _ in range(30):
        cand = base.copy()
        mask = rng.rand(n) < 0.5
        cand[mask] = 1.0
        cand[~mask] = 0.0
        pop.append(project_feasible(cand, n))
    for _ in range(10):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        cand = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
        pop.append(project_feasible(cand, n))

    fitness = np.array([compute_c5(p, n) for p in pop])
    best_idx = np.argmin(fitness)
    np_size = len(pop)
    CR = 0.8
    F = 0.7

    for gen in range(2000):
        if time.time() > deadline:
            break
        for i in range(np_size):
            if time.time() > deadline:
                break
            a = pop[best_idx].copy()
            b_idx = rng.choice([j for j in range(np_size) if j != i and j != best_idx])
            c_idx = rng.choice([j for j in range(np_size) if j != i and j != b_idx and j != best_idx])
            v = a + F * (pop[b_idx] - pop[c_idx])
            v = np.clip(v, 0, 1)
            v = project_feasible(v, n)
            fit_v = compute_c5(v, n)
            if fit_v < fitness[i]:
                pop[i] = v
                fitness[i] = fit_v
                if fit_v < fitness[best_idx]:
                    best_idx = i
    return pop[best_idx], fitness[best_idx]

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > 10:
            order = np.argsort(-c_vec[active])
            active = active[order[:10]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.015, 0.012)
        else:
            eta *= 0.96
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS polish at original resolution
    p1_end = min(t0 + 40, deadline - 200)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
                        alphas=[100, 500, 2000, 10000, 50000],
                        mu=0.01, maxiters=[500, 800, 1000, 1200, 1500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Fast pair search at original resolution
    p2_end = min(t0 + 180, deadline - 150)
    h2, c2 = fast_pair_search(best_h, best_n, p2_end, radius=60)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Active-set L-BFGS at original resolution
    p3_end = min(t0 + 230, deadline - 120)
    h3, c3 = run_active_opt(best_h, best_n, p3_end, top_k=5,
                            alphas=[50000, 200000, 1000000, 5000000])
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Upsample to n=1000 and optimize
    n_mid = 1000
    if time.time() < deadline - 250 and best_n < n_mid:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c + 0.002:
            p4a_end = min(time.time() + 80, deadline - 180)
            h4, c4 = run_lbfgs(h_mid, n_mid, p4a_end,
                               alphas=[20, 100, 500, 2000, 10000, 50000],
                               mu=0.004, maxiters=[400, 600, 800, 1000, 1200, 1500])
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n_mid
            p4b_end = min(time.time() + 60, deadline - 120)
            h4b, c4b = run_active_opt(best_h, best_n, p4b_end, top_k=5,
                                      alphas=[100000, 500000, 2000000, 10000000])
            if c4b < best_c:
                best_c = c4b
                best_h = h4b

    # Phase 5: Fast pair search at current best resolution
    p5_end = min(time.time() + 120, deadline - 100)
    h5, c5 = fast_pair_search(best_h, best_n, p5_end, radius=50)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: DE at low resolution for diversity + upsample
    if time.time() < deadline - 120:
        n_de = 60
        p6_end = min(time.time() + 30, deadline - 80)
        h_de, c_de = de_lowres(n_de, p6_end, seed=seed + 1)
        if c_de < best_c + 0.05:
            h_de_up = upsample(h_de, n_de, max(best_n, 500))
            c_de_up = compute_c5(h_de_up, max(best_n, 500))
            p6b_end = min(time.time() + 40, deadline - 40)
            h6, c6 = run_lbfgs(h_de_up, max(best_n, 500), p6b_end,
                               alphas=[100, 1000, 10000, 100000],
                               mu=0.01, maxiters=[300, 500, 700, 800])
            if c6 < best_c:
                best_c = c6
                best_h = h6
                best_n = len(h6)

    # Phase 7: Upsample to higher resolution and optimize
    n_high = 1500
    if time.time() < deadline - 100 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.003:
            p7a_end = min(time.time() + 60, deadline - 50)
            h7, c7 = run_lbfgs(h_high, n_high, p7a_end,
                               alphas=[10, 50, 200, 1000, 5000, 20000],
                               mu=0.003, maxiters=[300, 500, 700, 800, 1000, 1200])
            if c7 < best_c:
                best_c = c7
                best_h = h7
                best_n = n_high
            p7b_end = min(time.time() + 40, deadline - 20)
            h7b, c7b = run_active_opt(best_h, best_n, p7b_end, top_k=4,
                                      alphas=[500000, 2000000, 10000000])
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Phase 8: Final pair search at best resolution
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 50, deadline - 5)
        h8, c8 = fast_pair_search(best_h, best_n, p8_end, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Subgradient refinement
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, n_iter=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final polish: one more active-set run
    if time.time() < deadline - 1:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 10000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 200, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': 500})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
