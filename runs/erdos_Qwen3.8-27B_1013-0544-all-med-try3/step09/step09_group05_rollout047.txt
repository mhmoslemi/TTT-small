                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_wrap_obj(x, n, alpha):
    return lse_obj(x, n, alpha, 0.0)

def slqp_wrap_jac(x, n, alpha):
    return lse_grad(x, n, alpha, 0.0)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, obj_fn, jac_fn, maxiter=3000, ftol=1e-16, gtol=1e-12):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=jac_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def slqp_optimize(h, n, deadline, alpha=100000, maxiter=2000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_wrap_obj, h, jac=slqp_wrap_jac, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def top_shifts_iterate(h, n, deadline, top_k=5, max_rounds=15):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Active shifts: within 0.5% of max
        threshold = max_val - max(1e-8, 0.005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        if len(active) == 0:
            break
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 3:
                break
            obj = lambda x, a=alpha: topequal_obj(x, n, active, a, 0.01)
            jac = lambda x, a=alpha: topequal_grad(x, n, active, a, 0.01)
            try:
                res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-17,
                                        'gtol': 1e-13, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def fine_pair_search(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline:
            break
        changed = True
        while changed:
            changed = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            changed = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            changed = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    alpha = 50000
    base_steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for base_step in base_steps:
        if time.time() > deadline:
            break
        n_iter = 0
        while time.time() < deadline and n_iter < 500:
            n_iter += 1
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) == 0:
                break
            if len(active) > 10:
                active = active[np.argsort(-c_vec[active])[:10]]
            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad_single(h, n, int(idx))
            grad /= len(active)
            # Find best 2-point move along negative gradient
            g_sorted = np.argsort(-np.abs(grad))[:30]
            found = False
            for i_idx in range(len(g_sorted)):
                if time.time() > deadline:
                    return h, best_c
                i = int(g_sorted[i_idx])
                for j_idx in range(len(g_sorted)):
                    if i_idx == j_idx:
                        continue
                    j = int(g_sorted[j_idx])
                    gi, gj = grad[i], grad[j]
                    if gi > gj:
                        # decrease h[i], increase h[j]
                        step = base_step
                        if h[i] - step >= 0 and h[j] + step <= 1:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                found = True
                            else:
                                h[i] += step
                                h[j] -= step
                    else:
                        step = base_step
                        if h[i] + step <= 1 and h[j] - step >= 0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                found = True
                            else:
                                h[i] -= step
                                h[j] += step
            if not found:
                break
    return h, best_c

def multi_resolution_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_fine = np.linspace(0, 2, n, endpoint=True)
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50:
            continue
        remaining = deadline - time.time()
        if remaining < 30:
            break
        t_alloc = min(40.0, remaining * 0.3)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=True)
        h_c = np.interp(x_coarse, x_fine, h, left=0, right=0)
        h_c = project_feasible(h_c, n_coarse)
        # Optimize at coarse level
        obj = lambda x: lse_obj(x, n_coarse, 10000, 0.5)
        jac = lambda x: lse_grad(x, n_coarse, 10000, 0.5)
        bounds_c = [(0.0, 1.0)] * n_coarse
        try:
            res = minimize(obj, h_c, jac=jac, method='L-BFGS-B',
                           bounds=bounds_c,
                           options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
            h_c = project_feasible(res.x, n_coarse)
        except Exception:
            pass
        # Upsample back
        h_up = np.interp(x_fine, x_coarse, h_c, left=0, right=0)
        h_up = project_feasible(h_up, n)
        c_up = compute_c5(h_up, n)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    # Work at higher resolution
    n = 1000
    x_orig = np.linspace(0, 2, n_orig, endpoint=True)
    x_new = np.linspace(0, 2, n, endpoint=True)
    if n_orig != n:
        h_work = np.interp(x_new, x_orig, h_init, left=0, right=0)
    else:
        h_work = h_init.copy()
    h_work = project_feasible(h_work, n)
    best_h = h_work.copy()
    best_c = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n

    # Stage 1: L-BFGS with p-norm (p=4) to smooth the landscape
    if time.time() < deadline - 500:
        t_alloc = min(80.0, deadline - time.time() - 500)
        if t_alloc > 20:
            mu = 0.1
            obj_p4 = lambda x: pnorm_obj(x, n, 4, mu)
            jac_p4 = lambda x: pnorm_grad(x, n, 4, mu)
            try:
                res = minimize(obj_p4, h_work, jac=jac_p4, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h_work = h_new
            except Exception:
                pass

    # Stage 2: L-BFGS with p-norm (p=8)
    if time.time() < deadline - 450:
        t_alloc = min(60.0, deadline - time.time() - 450)
        if t_alloc > 15:
            mu = 0.05
            obj_p8 = lambda x: pnorm_obj(x, n, 8, mu)
            jac_p8 = lambda x: pnorm_grad(x, n, 8, mu)
            try:
                res = minimize(obj_p8, h_work, jac=jac_p8, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h_work = h_new
            except Exception:
                pass

    # Stage 3: L-BFGS with log-sum-exp, increasing alpha
    lse_configs = [
        (500, 0.5), (2000, 0.2), (10000, 0.1), (50000, 0.05),
        (100000, 0.02), (500000, 0.01), (1000000, 0.005)
    ]
    for alpha, mu in lse_configs:
        if time.time() > deadline - 200:
            break
        t_alloc = min(50.0, deadline - time.time() - 200)
        if t_alloc < 10:
            break
        obj_lse = lambda x, a=alpha, m=mu: lse_obj(x, n, a, m)
        jac_lse = lambda x, a=alpha, m=mu: lse_grad(x, n, a, m)
        try:
            res = minimize(obj_lse, h_work, jac=jac_lse, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-17,
                                    'gtol': 1e-13, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h_work = h_new
        except Exception:
            pass

    # Stage 4: Top-shifts equalization
    if time.time() < deadline - 150:
        t4 = min(80.0, deadline - time.time() - 150)
        if t4 > 20:
            h_work, best_c = top_shifts_iterate(h_work, n, time.time() + t4,
                                                 top_k=5, max_rounds=12)
            if best_c >= compute_c5(best_h, n):
                h_work = best_h.copy()
            else:
                best_h = h_work.copy()

    # Stage 5: SLSQP with hard constraint
    if time.time() < deadline - 100:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 90:
                break
            t5 = min(30.0, deadline - time.time() - 90)
            if t5 < 10:
                break
            h_work, best_c = slqp_optimize(h_work, n, time.time() + t5,
                                           alpha=alpha, maxiter=1500)
            if best_c >= compute_c5(best_h, n):
                h_work = best_h.copy()
            else:
                best_h = h_work.copy()

    # Stage 6: Another round of top-shifts (tighter)
    if time.time() < deadline - 80:
        t6 = min(50.0, deadline - time.time() - 80)
        if t6 > 15:
            h_tmp, c_tmp = top_shifts_iterate(best_h, n, time.time() + t6,
                                               top_k=3, max_rounds=10)
            if c_tmp < best_c:
                best_c = c_tmp
                best_h = h_tmp

    # Stage 7: Multi-resolution refinement
    if time.time() < deadline - 60:
        t7 = min(40.0, deadline - time.time() - 60)
        if t7 > 20:
            h_tmp, c_tmp = multi_resolution_refine(best_h, n, time.time() + t7)
            if c_tmp < best_c:
                best_c = c_tmp
                best_h = h_tmp

    # Stage 8: High-alpha L-BFGS polish
    if time.time() < deadline - 40:
        for alpha in [5000000, 10000000, 20000000]:
            if time.time() > deadline - 35:
                break
            obj_pol = lambda x, a=alpha: lse_obj(x, n, a, 0.001)
            jac_pol = lambda x, a=alpha: lse_grad(x, n, a, 0.001)
            try:
                res = minimize(obj_pol, best_h, jac=jac_pol, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxfun': 20000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Stage 9: Gradient-guided local search
    if time.time() < deadline - 20:
        t9 = min(15.0, deadline - time.time() - 10)
        if t9 > 5:
            best_h, best_c = gradient_local_search(best_h, n, time.time() + t9)

    # Stage 10: Final fine pair search
    if time.time() < deadline - 3:
        t10 = max(2.0, deadline - time.time() - 2)
        best_h, best_c = fine_pair_search(best_h, n, time.time() + t10, radius=30)

    # Final SLSQP if time allows
    if time.time() < deadline - 1:
        h_final, c_final = slqp_optimize(best_h, n, deadline - 0.5, alpha=500000, maxiter=500)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    n_final = len(best_h)
    return best_h, best_c, n_final
