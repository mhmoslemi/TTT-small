def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            if tot < 1e-14:

            if tot < 1e-14:

            h += (diff / tot) * h

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1

        k = idx - center

        if k >= 0:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (10000, 5000.0), (20000, 10000.0),
                   (50000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def threshold_binary(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    target = n / 2.0
    diff = int(round(target - h_bin.sum()))
    if diff > 0:
        frac = h.copy()
        frac[h_bin == 1.0] = -1  # Exclude current 1s
        idx_zero = np.where(h_bin == 0.0)[0]
        order = idx_zero[np.argsort(-h[idx_zero])]
        for t in range(min(diff, len(order))):
            h_bin[order[t]] = 1.0
    elif diff < 0:
        idx_one = np.where(h_bin == 1.0)[0]
        order = idx_one[np.argsort(h[idx_one])]
        for t in range(min(-diff, len(order))):
            h_bin[order[t]] = 0.0
    return h_bin

                best_h = h_b.copy()
                ones = np.where(h_b == 1.0)[0]
                zeros = np.where(h_b == 0.0)[0]
                improved = True
                stall = 0
            else:
                h_b[i] = 1.0
                h_b[j] = 0.0
        
        # Phase B: Targeted swaps (aim at worst shift)
        if not improved:
            c_vec = compute_c5_vec(h_b, n)
            worst_idx = int(np.argmax(c_vec))
            k_w = worst_idx - (n - 1)
            
            # Try to reduce the worst shift
            n_targeted = 100
            for _ in range(n_targeted):
                if time.time() > deadline:
                    return best_h, best_c
                a = rng.randint(len(ones))
                b = rng.randint(len(zeros))
                i, j = ones[a], zeros[b]
                
                # Check if this swap helps the worst shift
                if k_w >= 0:
                    helps_i = (i + k_w < n) and (h_b[i + k_w] == 0.0)  # removing a contributor
                    hurts_j = (j + k_w < n) and (h_b[j + k_w] == 0.0)  # adding a contributor
                    if not helps_i and hurts_j:
                        continue  # Skip clearly bad moves
                else:
                    kw = -k_w
                    helps_i = (i - kw >= 0) and (h_b[i - kw] == 0.0)
                    hurts_j = (j - kw >= 0) and (h_b[j - kw] == 0.0)
                    if not helps_i and hurts_j:
                        continue

                h_b[j] = 1.0
                c = compute_c5(h_b, n)

                    best_c = c
                    best_h = h_b.copy()
                    ones = np.where(h_b == 1.0)[0]

                    h_b[j] = 0.0

                break
    
    return best_h, best_c

def upsample_binary(h, n_old, n_new):
    """Upsample a (possibly binary) signal."""
    if n_old == n_new:
        return h.copy()
    # Linear interpolation
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
