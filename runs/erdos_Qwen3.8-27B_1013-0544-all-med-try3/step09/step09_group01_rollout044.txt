            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(500, int(remaining * 4)) if maxit_list is None else maxit_list[i]
        
        try:
            res = minimize(obj_func, best_h, jac=grad_func,
                          args=(n, alpha, mu, None), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_phase(h, n, deadline, top_k=10, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000]
    
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            idx_sorted = np.argsort(-c_vec[active])
            active = active[idx_sorted[:top_k]]
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(obj_func, best_h, jac=grad_func,
                              args=(n, alpha, 0.001, active), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_phase(h, n, deadline, n_iter=10000, eta_init=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta_init
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.97
        
        h = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=50, steps=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-9 and best_h[j] - d >= -1e-9:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h[i] = min(1.0, max(0.0, best_h[i]))
                        best_h[j] = min(1.0, max(0.0, best_h[j]))
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-9 and best_h[j] + d <= 1.0 + 1e-9:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h[i] = min(1.0, max(0.0, best_h[i]))
                        best_h[j] = min(1.0, max(0.0, best_h[j]))
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def make_binary(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        h_bin[sorted_idx[n_ones]] = remaining
    return project_feasible(h_bin, n)

def slqp_phase(h, n, deadline, alpha=100000):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    maxit = max(500, int((deadline - time.time()) * 2))
    try:
        res = minimize(obj_func, h, args=(n, alpha, 0.0, None),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    
    # Phase 1: L-BFGS at original resolution (n_orig)
    p1_end = min(t0 + 250, deadline - 500)
    h1, c1 = lbfgs_phase(h0, n_orig, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01)
    best_h, best_c, best_n = h1, c1, n_orig
    
    # Phase 2: Active set at n_orig
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 120, deadline - 400)
        h2, c2 = active_set_phase(best_h, best_n, p2_end, top_k=8,
            alphas=[20000, 100000, 500000, 2000000])
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, best_n
    
    # Phase 3: Upsample to n=1000
    n_fine = 1000
    if time.time() < deadline - 350:
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p3_end = min(time.time() + 250, deadline - 250)
        h3, c3 = lbfgs_phase(h_fine, n_fine, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.005)
        
        if c3 < best_c:
            best_h, best_c, best_n = h3, c3, n_fine
    
    # Phase 4: Active set at best resolution
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 200)
        h4, c4 = active_set_phase(best_h, best_n, p4_end, top_k=6,
            alphas=[50000, 200000, 1000000, 5000000])
        if c4 < best_c:
            best_h, best_c, best_n = h4, c4, best_n
    
    # Phase 5: Subgradient refinement
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 60, deadline - 100)
        h5, c5 = subgrad_phase(best_h, best_n, p5_end, n_iter=5000, eta_init=0.008)
        if c5 < best_c:
            best_h, best_c, best_n = h5, c5, best_n
    
    # Phase 6: Binary restart
    if time.time() < deadline - 200:
        h_bin = make_binary(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        h_candidate = h_bin.copy()
        c_candidate = c_bin
        
        p6_end = min(time.time() + 100, deadline - 150)
        h6, c6 = lbfgs_phase(h_candidate, best_n, p6_end,
            alphas=[100, 1000, 10000, 100000, 500000],
            mu=0.01)
        if c6 < best_c:
            best_h, best_c, best_n = h6, c6, best_n
    
    # Phase 7: Pair search
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = pair_search(best_h, best_n, p7_end, radius=40,
            steps=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c7 < best_c:
            best_h, best_c, best_n = h7, c7, best_n
    
    # Phase 8: Final subgradient + active set
    if time.time() < deadline - 15:
        p8_end = min(time.time() + 10, deadline - 10)
        h8, c8 = subgrad_phase(best_h, best_n, p8_end, n_iter=2000, eta_init=0.003)
        if c8 < best_c:
            best_h, best_c, best_n
