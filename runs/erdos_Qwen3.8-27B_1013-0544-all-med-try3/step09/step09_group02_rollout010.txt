            break

        # Use L-BFGS-B

            val += mu * (x.sum() - n/2.0)**2

def toeplitz_apply(w_shifts, h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def shift_grad_single(x, n, mm):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = mm - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] = (1.0 - x[k:]) * dx
        grad[k:] += (-x[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] = (1.0 - x[:nv]) * dx
        grad[:nv] += (-x[kk:]) * dx
    return grad

def irls_polish(h, n, deadline):
    """IRLS with projected gradient, using Toeplitz FFT for speed."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    num_shifts = 2 * n - 1
    mu = 50.0
    dx = 2.0 / n
    
    # Schedule: increasing focus on worst shifts
    betas = [2, 5, 10, 20, 50, 100, 200, 500]
    
    for beta in betas:
        if time.time() > deadline:
            break
        w = np.ones(num_shifts) / num_shifts
        lr = 1.0 / (beta * 10 + 1.0)
        
        no_improve = 0
        for it in range(2000):
            if time.time() > deadline:
                break
            
            # Compute gradient using Toeplitz
            t = w[n-1 : 2*n-1].copy()
            Ah = toeplitz_apply(t, h, n)
            
            grad = -2.0 * dx * Ah + 2.0 * mu * (h.sum() - n / 2.0)
            
            # Adaptive step size
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            h = h_new
            
            # Update weights periodically
            if it % 20 == 19:
                c_vec = compute_c5_vec(h, n)
                # Shift to prevent overflow
                c_shifted = c_vec - c_vec.min()
                w = w * np.exp(beta * c_shifted)
                w /= w.sum()
            
            # Decay learning rate
            lr *= 0.995
            
            if no_improve > 300:
                break
    
    return best_h, best_c

def coordinate_descent(h, n, deadline):
    """Targeted coordinate descent on worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            if len(active) == 0:
                break
            
            # Compute gradient for active shifts
            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad_single(best_h, n, idx)
            grad /= len(active)
            
            # Find best pair (i,j): increase h[i], decrease h[j]
            # Want grad[i] very negative (increasing h[i] helps) and grad[j] very positive
            # Net effect: -grad[i]*step + grad[j]*step should be negative → grad[i] < grad[j]
            # Also need: best_h[i] + step <= 1, best_h[j] - step >= 0
            
            neg_candidates = np.argsort(grad)[:(n//10 + 10)]
            pos_candidates = np.argsort(-grad)[:(n//10 + 10)]
            
            best_cand = None
            best_cand_c = best_c
            
            for i_val in neg_candidates:
                i = int(i_val)
                if best_h[i] + step > 1.0 + 1e-10:
                    continue
                for j_val in pos_candidates:
                    j = int(j_val)
                    if i == j or best_h[j] - step < -1e-10:
                        continue
                    best_h[i] += step
                    best_h[j] -= step
                    c = compute_c5(best_h, n)
                    if c < best_cand_c:
                        best_cand_c = c
                        best_cand = (int(i), int(j))
                    else:
                        best_h[i] -= step
                        best_h[j] += step
                if best_cand is not None:
                    break
            
            if best_cand is not None:
                improved = True
                best_c = best_cand_c
                # Recompute from scratch to avoid accumulated errors
                # Actually h is already modified, so just continue
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: L-BFGS warmup (40s)
    t1 = min(t0 + 40, deadline - 800)
    if t1 > time.time() + 5:
        bounds = [(0.0, 1.0)] * n_orig
        for alpha, mu in [(100, 50.0), (500, 200.0), (2000, 1000.0)]:
            if time.time() > t1:
                break
            try:
                res = minimize(smooth_obj, best_h, jac=smooth_grad,
                              args=(n_orig, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: IRLS optimization (250s)
    t2 = min(t0 + 290, deadline - 500)
    if t2 > time.time() + 5:
        best_h, best_c = irls_polish(best_h, n_orig, t2)
    
    # Phase 3: High-α L-BFGS (100s)
    t3 = min(t0 + 390, deadline - 350)
    if t3 > time.time() + 5:
        bounds = [(0.0, 1.0)] * n_orig
        for alpha, mu in [(5000, 2000.0), (20000, 5000.0), (100000, 10000.0)]:
            if time.time() > t3:
                break
            try:
                res = minimize(smooth_obj, best_h, jac=smooth_grad,
                              args=(n_orig, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
