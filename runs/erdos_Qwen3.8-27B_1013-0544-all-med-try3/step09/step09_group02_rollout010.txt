def project_feasible(h, n):

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k):
    """Gradient of overlap at shift k with respect to h.
    k is the shift index in [0, 2n-2], actual shift = k - (n-1)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    lag = k - (n - 1)
    if lag >= 0:
        # overlap = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag]) * dx
        nv = n - lag
        grad[0:nv] += (1.0 - h[lag:n]) * dx
        grad[lag:n] += (-h[0:nv]) * dx
    else:
        ll = -lag
        # overlap = sum_{i=ll}^{n-1} h[i] * (1-h[i-ll]) * dx
        nv = n - ll
        grad[ll:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[ll:n]) * dx
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgradient_polish(h, n, deadline, n_iter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active[:min(20, len(active))]:
            grad += np.abs(grad_shift(h, n, int(idx)))
        grad /= min(20, len(active))
        grad -= grad.mean()
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.9
            if step < 1e-7:
                break
    return best_h, best_c

def perturb_and_reopt(h, n, deadline, seed=123):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for attempt in range(5):
        if time.time() > deadline - 20:
            break
        noise = rng.randn(n) * 0.02
        noise -= noise.mean()
        h_pert = project_feasible(h + noise, n)
        c_pert = compute_c5(h_pert, n)
        t_sub = min(time.time() + 15, deadline - 5)
        h_opt, c_opt = sqp_optimize(h_pert, n, t_sub, max_iter=100, rho_init=0.04, max_active=20)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup at original n
    t1 = min(t0 + 40, deadline - 700)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(200, 50.0), (2000, 500.0), (20000, 5000.0), (100000, 50000.0)])

    # Phase 2: SQP optimization
    t2 = min(t0 + 150, deadline - 550)
    if t2 > time.time() + 5:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=600, rho_init=0.07, max_active=30)

    # Phase 3: Local search at original n
    t3 = min(t0 + 200, deadline - 500)
    if t3 > time.time() + 3:
        best_h, best_c = local_search(best_h, n_orig, t3, radius=50)

    # Phase 4: Subgradient polish
    t4 = min(t0 + 230, deadline - 470)
    if t4 > time.time() + 3:
        best_h, best_c = subgradient_polish(best_h, n_orig, t4)

    # Phase 5: SLSQP
    t5 = min(t0 + 270, deadline - 430)
    if t5 > time.time() + 5:
        h5, c5 = slqp_polish(best_h, n_orig, t5, alpha=5000000, maxiter=4000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Upsample to n=1000
    n_hi = 1000
    if time.time() < deadline - 350:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)

        t6 = min(time.time() + 80, deadline - 250)
        if t6 > time.time() + 5:
            h_hi, c_hi = run_lbfgs(h_hi, n_hi, t6,
                configs=[(500, 100.0), (5000, 1000.0), (50000, 10000.0), (200000, 50000.0)])

        t7 = min(time.time() + 100, deadline - 150)
        if t7 > time.time() + 5:
            h_hi, c_hi = sqp_optimize(h_hi, n_hi, t7, max_iter=500, rho_init=0.06, max_active=35)

        t8 = min(time.time() + 40, deadline - 80)
        if t8 > time.time() + 3:
            h_hi, c_hi = local_search(h_hi, n_hi, t8, radius=40)

        t9 = min(time.time() + 25, deadline - 40)
        if t9 > time.time() + 3:
            h_hi, c_hi = subgradient_polish(h_hi, n_hi, t9)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi

        # Phase 7: Perturbation at n=1000
        if time.time() < deadline - 50:
            t10 = min(time.time() + 40, deadline - 10)
            h_p, c_p = perturb_and_reopt(best_h, len(best_h), t10, seed=seed+1)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 8: Try n=2000 if time allows
    if time.time() < deadline - 80:
        n_xl = 2000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_xl)
        h_xl = np.interp(x_new, x_old, best_h)
        h_xl = project_feasible(h_xl, n_xl)
        c_xl = compute_c5(h_xl, n_xl)

        t11 = min(time.time() + 40, deadline - 40)
        if t11 > time.time() + 5:
            h_xl, c_xl = run_lbfgs(h_xl, n_xl, t11,
                configs=[(2000, 500.0), (20000, 5000.0)])

        t12 = min(time.time() + 30, deadline - 10)
        if t12 > time.time() + 3:
            h_xl, c_xl = sqp_optimize(h_xl, n_xl, t12, max_iter=200, rho_init=0.04, max_active=25)

        if c_xl < best_c:
            best_c = c_xl
            best_h = h_xl

    # Final: SLSQP polish and feasibility
    if time.time() < deadline - 5:
        h_fin, c_fin = slqp_polish(best_h, len(best_h), deadline - 2, alpha=10000000, maxiter=5000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
