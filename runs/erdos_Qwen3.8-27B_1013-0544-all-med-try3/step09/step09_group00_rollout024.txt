                nv = n - kk

        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Try alternative starts at n0
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_h = make_beatty(n0, golden / 2.0)
    c_alt = compute_c5(alt_h, n0)
    if c_alt < best_c:
        best_c = c_alt
        best_h = alt_h
    alt_h2 = make_beatty(n0, np.sqrt(2.0) / 2.0)
    c_alt2 = compute_c5(alt_h2, n0)
    if c_alt2 < best_c:
        best_c = c_alt2
        best_h = alt_h2
    alt_h3 = make_beatty(n0, 1.0 / np.sqrt(3.0))
    c_alt3 = compute_c5(alt_h3, n0)
    if c_alt3 < best_c:
        best_c = c_alt3
        best_h = alt_h3
    
    # Phase 1: L-BFGS at n0
    p1_end = t0 + 200
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 500000, 2000000],
        mu=0.01, maxit_list=[800, 1000, 1200, 1500, 2000, 2500, 3000, 3000, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: SLP refinement
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 120, deadline - 300)
        h2, c2 = slp_refine(best_h, best_n, p2_end, max_rounds=80, trust_radius=0.1)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Multi-active L-BFGS
    if time.time() < deadline - 280:
        p3_end = min(time.time() + 100, deadline - 250)
        h3, c3 = multi_active(best_h, best_n, p3_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Coordinate descent
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 170)
        h4, c4 = coord_descent(best_h, best_n, p4_end,
            step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=80000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Fine pair search
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = fine_pair(best_h, best_n, p5_end, radius=40,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Perturbation and search
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 60, deadline - 60)
        h6, c6 = perturb_and_search(best_h, best_n, p6_end, rng, n_perturbs=4)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Try higher resolution
    if time.time() < deadline - 60 and best_n < 800:
        n_high = min(800, best_n * 2)
        # Upsample by holding values
        h_high = np.zeros(n_high)
        ratio = n_high / best_n
        idx_map = (np.arange(n_high) * best_n / n_high).astype(int)
        idx_map = np.clip(idx_map, 0, best_n - 1)
        h_high[:] = best_h[idx_map]
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        
        if time.time() < deadline - 40:
            p7_end = min(time.time() + 50, deadline - 30)
            h7, c7 = phase_lbfgs(h_high, n_high, p7_end,
                alphas=[100, 1000, 10000, 100000, 500000],
                mu=0.01, maxit_list=[500, 800, 1000, 1500, 2000])
            if c7 < c_high:
                h_high, c_high = h7, c7
            
            if time.time() < deadline - 20:
                h7b, c7b = coord_descent(h_high, n_high, deadline - 10,
                    step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                    max_steps=30000)
                if c7b < c_high:
                    h_high, c_high = h7b, c7b
            
            if c_high < best_c:
                best_c = c_high
                best_h = h_high
                best_n = n_high
    
    # Phase 8: Final SLP + fine adjustments
    if time.time() < deadline - 15:
        h8, c8 = slp_refine(best_h, best_n, deadline - 5, max_rounds=40, trust_radius=0.05)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
