**Alternative approach: Direct min-max with bisection**

For a given C, check feasibility:
- Find h such that 0 ≤ h ≤ 1, sum(h) = n/2, and for all k: corr_k * dx ≤ C
- This is: find h such that for all k: sum_{valid j} h[j](1-h[j+k]) ≤ C/dx

This is a non-convex feasibility problem (the constraints are non-convex: h[j] - h[j]h[j+k] is non-convex in h).

But we can use SCA (Successive Convex Approximation):
- Linearize each constraint around the current h:
  h[j](1-h[j+k]) ≈ h0[j](1-h0[j+k]) + (1-h0[j+k])(h[j]-h0[j]) - h0[j](h[j+k]-h0[j+k])
  = h0[j] - h0[j]h0[j+k] + (1-h0[j+k])δ_j - h0[j]δ_{j+k}
  
  where δ_i = h[i] - h0[i]

- This gives a linear constraint in h (or δ)
- Solve the LP: minimize 0 subject to linearized constraints, 0≤h≤1, sum(h)=n/2
- If feasible, update h0 and repeat
- If infeasible, the LP gives a direction to reduce infeasibility

This is a powerful approach! Let me implement it.

```python
from scipy.optimize import linprog

def sca_feasibility(h0, n, C_target, max_iters=100):
    dx = 2.0 / n
    C_limit = C_target / dx
    h = h0.copy()
    for it in range(max_iters):
        # Compute gradient (linearization) of each constraint
        # Constraint k: sum_j h[j](1-h[j+k]) ≤ C_limit
        # Linearized: sum_j [h0[j] - h0[j]h0[j+k] + (1-h0[j+k])(h[j]-h0[j]) - h0[j](h[j+k]-h0[j+k])] ≤ C_limit
        
        # This simplifies to:
        # c_k(h0) + sum_j [(1-h0[j+k]) - h0[j+k]] δ_j ... 
        # Actually let me just compute:
        # c_k_linear = c_k(h0) + sum_i grad_k[i] * (h[i] - h0[i])
        # where grad_k = d c_k / d h evaluated at h0
        
        # So: c_k(h0) + grad_k · (h - h0) ≤ C_limit
        # grad_k · h ≤ C_limit - c_k(h0) + grad_k · h0
        
        # For all k: A @ h ≤ b
        # 0 ≤ h ≤ 1
        # sum(h) = n/2
        
        # But building the A matrix for 2n-1 constraints × n variables is O(n²) memory
        # For n=1000, that's 1999 × 1000 = ~2M entries, which is fine
        
        A_rows = []
        b_vals = []
        c_vec = compute_c5_vec(h, n)
        for idx in range(len(c_vec)):
            p = idx - (n - 1)
            # Compute grad of this specific correlation at h
            row = np.zeros(n)
            if p >= 0 and p < n:
                nv = n - p
                row[:nv] += (1.0 - h[p:p+nv])
                row[p:] += (-h[:nv])
            elif p < 0 and -p < n:
                q = -p
                nv = n - q
                row[q:] += (1.0 - h[:nv])
                row[:nv] += (-h[q:q+nv])
            
            c_k = c_vec[idx]
            # Linearized: c_k + row·(h_h_new - h) ≤ C_limit
            # row·h_new ≤ C_limit - c_k + row·h
            rhs = C_limit - c_k + np.dot(row, h)
            A_rows.append(row)
            b_vals.append(rhs)
        
        A = np.array(A_rows)
        b = np.array(b_vals)
        
        # Add sum constraint: sum(h) = n/2
        # For linprog, we need equality constraint: A_eq @ h = b_eq
        
        A_eq = np.ones((1, n))
        b_eq = np.array([n/2.0])
        
        bounds = [(0, 1)] * n
        c_obj = np.zeros(n)  # minimize 0
        
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if res.status == 0:
            return res.x, True
        else:
            # Infeasible - need to increase C_target
            # Use the dual to find the most violated constraint
            # For now, just return infeasible
            return h, False
    return h, False

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slexp_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def slexp_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def obj_lbfgs(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val = slexp_val(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def grad_lbfgs(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    w = slexp_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        p = idx - (n - 1)
        wm = w[idx] * dx
        if p >= 0 and p < n:
            nv = n - p
            grad[:nv] += wm * (1.0 - h[p:])
            grad[p:] += wm * (-h[:nv])
        elif p < 0 and -p < n:
            q = -p
            nv = n - q
            grad[q:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[q:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def obj_active(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    val = slexp_val(vals, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def grad_active(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    wv = np.exp(sh - smx)
    wv /= wv.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(wv, active):
        if wi < 1e-14:
            continue
        p = idx - (n - 1)
        wm = wi * dx
        if p >= 0 and p < n:
            nv = n - p
            grad[:nv] += wm * (1.0 - h[p:])
            grad[p:] += wm * (-h[:nv])
        elif p < 0 and -p < n:
            q = -p
            nv = n - q
            grad[q:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[q:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def interpolate(h_old, n_old, n_new):
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h_old)

def run_lbfgs(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time()
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(obj_lbfgs, best_h, jac=grad_lbfgs,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_sca(h, n, deadline, max_iters=30, top_k=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline - 20:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        C_current = c_vec.max()
        C_target = C_current  # Try to maintain current level
        
        # Select top-k active constraints
        order = np.argsort(-c_vec)
        active = order[:top_k]
        
        # Build LP
        A_rows = []
        b_vals = []
        for idx in active:
            p = idx - (n - 1)
            row = np.zeros(n)
            if p >= 0 and p < n:
                nv = n - p
                row[:nv] += (1.0 - best_h[p:])
                row[p:] += (-best_h[:nv])
            elif p < 0 and -p < n:
                q = -p
                nv = n - q
                row[q:] += (1.0 - best_h[:nv])
                row[:nv] += (-best_h[q:])
            
            c_k = c_vec[idx]
            rhs = C_target - c_k + np.dot(row, best_h)
            A_rows.append(row)
            b_vals.append(rhs)
        
        A = np.array(A_rows)
        b = np.array(b_vals)
        A_eq = np.ones((1, n))
        b_eq = np.array([n / 2.0])
        c_obj = np.zeros(n)
        bounds = [(0, 1)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status == 0:
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                # Always update to new point for better linearization
                if not np.allclose(res.x, best_h, atol=1e-10):
                    best_h = h_new
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Compute pseudo-gradient
        grad = np.zeros(n)
        for idx in active:
            p = idx - (n - 1)
            if p >= 0 and p < n:
                nv = n - p
                grad[:nv] += dx_val * (1.0 - best_h[p:])
                grad[p:] += dx_val * (-best_h[:nv])
            elif p < 0 and -p < n:
                q = -p
                nv = n - q
                grad[q:] += dx_val * (1.0 - best_h[:nv])
                grad[:nv] += dx_val * (-best_h[q:])
        grad /= len(active)
        
        # Find best pair: decrease where grad > 0, increase where grad < 0
        pos = np.where((grad > 1e-10) & (best_h > 1e-10))[0]
        neg = np.where((grad < -1e-10) & (best_h < 1e-10 + 1.0))[0]
        
        if len(pos) == 0 or len(neg) == 0:
            break
        
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos[:30]:
            for j in neg[:30]:
                g = grad[i] - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        # Try multiple step sizes
        improved = False
        for step in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]:
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
        
        if not improved:
            break
    
    return best_h, best_c

def active_set_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    configs = [
        (3, 50000, 200000),
        (5, 200000, 1000000),
        (8, 1000000, 5000000),
        (15, 5000000, 20000000),
        (30, 20000000, 100000000),
    ]
    
    for top_k, alpha_lo, alpha_hi in configs:
        if time.time() > deadline - 20:
            break
        alphas = [alpha_lo, int(alpha_lo * 3), int(alpha_hi)]
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                continue
            
            maxit = max(200, 1000)
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, 0.001),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def make_beatty(n, a):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * a) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)
