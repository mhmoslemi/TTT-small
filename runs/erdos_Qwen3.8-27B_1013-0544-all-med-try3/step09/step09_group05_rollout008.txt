def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, top_idx):
    """Smooth max of selected correlations."""
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)

    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_gradient_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj_top(x, n, alpha, top_idx):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_top(x, n, alpha, top_idx):

    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)

    dx = 2.0 / n
    grad = np.zeros(n)

        k = idx - (n - 1)

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

def lp_active_step(h, n, top_k=8, deadline=None):
    """LP-based active set step: linearize top-k constraints, solve LP, line search."""
    c = compute_c5_vec(h, n)
    max_val = c.max()
    # Find active shifts (within 0.5% of max or top k)
    thresh = max_val - max(1e-8, 0.005 * max_val)
    active = np.where(c >= thresh)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c[active])[:top_k]]
    
    dx = 2.0 / n
    n_act = len(active)
    
    # Build LP: min t s.t. g_k . h - t <= g_k . h - f_k(h) for active k
    #            0 <= h <= 1, sum(h) = n/2
    # Variables: [h(0),...,h(n-1), t]  (n+1 vars)
    # c_obj: minimize t -> [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    
    for ii, idx in enumerate(active):
        g = get_gradient_single(h, n, idx)
        f_val = c[idx]
        A_ub[ii, 0:n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = g @ h - f_val
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            h_lp = project_feasible(h_lp, n)
            # Line search
            best_h = h.copy()
            best_c = compute_c5(h, n)
            alphas = [1.0, 0.75, 0.5, 0.33, 0.25, 0.125, 0.0625]

                h_trial = h + a * (h_lp - h)

                    best_c = c_trial
                    best_h = h_trial
            return best_h, best_c

        pass
    return h.copy(), compute_c5(h, n)

def pair_swap_refine(h, n, deadline, radius=50):

    best_c = compute_c5(h, n)

    for d in deltas:

                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

def gradient_line_search(h, n, deadline, top_k=5):

            grad += get_gradient_single(h, n, idx)
        grad /= len(active)

        neg_mask = grad < -1e-14
        max_step = 1e6
        if pos_mask.any():
            max_step = min(max_step, ((bounds_hi - h)[pos_mask] / grad[pos_mask]).min())
        if neg_mask.any():
            max_step = min(max_step, (h[neg_mask] / (-grad[neg_mask])).min())
        max_step = max(max_step, 1e-10)
        
        # Try several step sizes
        best_alpha = 0
        best_trial_c = best_c
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            alpha = frac * max_step
            h_trial = h - alpha * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_trial_c:
                best_trial_c = c_trial
                best_alpha = alpha

            break
        h = h - best_alpha * grad
        h = project_feasible(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
