def compute_correlation_fft(h, n):

    F = np.fft.fft(1.0 - h, N)
    r = np.real(np.fft.ifft(H * np.conj(F)))
    c = np.concatenate([r[N-n+1:], r[:n]])
    return c

def project_feasible(h, n):
