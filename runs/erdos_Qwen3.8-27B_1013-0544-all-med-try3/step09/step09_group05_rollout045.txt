def sum_con_jac(x, n):
    return np.ones(n)


def slqp_run(x, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, x, args=(n, alpha, 0.0), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return x.copy(), compute_c5(x, n)


def get_active_shifts(c_vec, frac=0.02, max_k=30):
    mx = c_vec.max()
    thresh = mx - max(1e-8, frac * mx)
    active = np.where(c_vec >= thresh)[0]
    if len(active) > max_k:
        active = active[np.argsort(-c_vec[active])[:max_k]]
    return active


def local_pair_search(h, n, deadline, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            if time.time() > deadline:
                break
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                for j in range(0, n, 2):
                    if abs(i - j) < 3 or abs(i - j) > n - 3:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c


def gradient_local_search(h, n, deadline, base_steps=None):
    best_c = compute_c5(h, n)
    if base_steps is None:
        base_steps = [0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in base_steps:
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        mx = c_vec.max()
        thresh = mx - max(1e-7, 0.005 * mx)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        
        grad = np.zeros(n)
        for idx in active:
            grad += grad_single_shift(h, n, idx)
        grad /= len(active)
        
        g_abs = np.abs(grad)
        top_idxs = np.argsort(-g_abs)[:80]
        
        found = True
        while found and time.time() < deadline:
            found = False
            for ii in range(len(top_idxs)):
                if time.time() > deadline:
                    return h, best_c
                i = top_idxs[ii]
                for jj in range(len(top_idxs)):
                    if ii == jj:
                        continue
                    j = top_idxs[jj]
                    gi, gj = grad[i], grad[j]
                    
                    if gi > gj:
                        # decrease h[i], increase h[j]
                        if h[i] - step >= 0 and h[j] + step <= 1:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                found = True
                            else:
                                h[i] += step
                                h[j] -= step
                    else:
                        # increase h[i], decrease h[j]
                        if h[i] + step <= 1 and h[j] - step >= 0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                found = True
                            else:
                                h[i] -= step
                                h[j] += step
    return h, best_c


def lbfgs_phase(h, n, deadline, alpha_list, mu_list, maxiter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha, mu in zip(alpha_list, mu_list):
        if time.time() > deadline - 10:
            break
        current = best_h.copy()
        t_remaining = deadline - time.time()
        if t_remaining < 15:
            break
        try:
            res = minimize(soft_max_obj, current, jac=soft_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': int(maxiter * 2),
                                    'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def active_set_lbfgs(h, n, deadline, alpha, max_k=20, iterations=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for _ in range(iterations):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_shifts(c_vec, frac=0.01, max_k=max_k)
        if len(active) < 2:
            break
        try:
            res = minimize(top_k_obj, best_h, jac=top_k_grad,
                           args=(n, active, alpha, 10.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-18,
                                    'gtol': 1e-13, 'maxfun': 10000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Upsample to n=1000
    N = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, N)
    h = np.interp(x_new, x_old, best_h)
    h = project_feasible(h, N)
    c = compute_c5(h, N)
    if c < best_c:
        best_c = c
        best_h = h
    else:
        best_h = h
    
    # Phase 1: Global L-BFGS (low to medium alpha)
    t1 = min(t0 + 180, deadline - 700)
    h, c = lbfgs_phase(h, N, t1,
                       alpha_list=[50, 200, 1000],
                       mu_list=[5.0, 50.0, 500.0],
                       maxiter=3000)
    if c < best_c:
        best_c = c
        best_h = h
    
    # Phase 2: Higher alpha L-BFGS
    t2 = min(t0 + 380, deadline - 500)
    h, c = lbfgs_phase(h, N, t2,
                       alpha_list=[3000, 10000, 50000],
                       mu_list=[2000.0, 10000.0, 50000.0],
                       maxiter=4000)
    if c < best_c:
        best_c = c
        best_h = h
    
    # Phase 3: Active-set focused optimization
    t3 = min(t0 + 530, deadline - 350)
    h, c = active_set_lbfgs(h, N, t3, alpha=20000, max_k=15, iterations=4)
    if c < best_c:
        best_c = c
        best_h = h
    
    # Phase 4: SLSQP with exact constraint
    t4 = min(t0 + 640, deadline - 280)
    for alpha in [50000, 200000]:
        if time.time() > t4 - 30:
            break
        h_new, c_new = slqp_run(best_h, N, alpha, maxiter=2000, deadline=t4)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Random restarts (2 perturbations)
    t5 = min(t0 + 800, deadline - 200)
    for restart in range(2):
        if time.time() > t5 - 60:
            break
        h_pert = best_h.copy()
        noise = rng.normal(0, 0.02, N)
        h_pert = np.clip(h_pert + noise, 0.0, 1.0)
        h_pert = project_feasible(h_pert, N)
        c_pert = compute_c5(h_pert, N)
        
        if c_pert < best_c + 0.001:
            t_sub = min(time.time() + 55, t5 - 10)
            h_pert, c_pert = lbfgs_phase(h_pert, N, t_sub,
                                         alpha_list=[5000, 50000],
                                         mu_list=[5000.0, 50000.0],
                                         maxiter=2000)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert
    
    # Phase 6: Gradient-guided local search
    t6 = min(t0 + 950, deadline - 100)
    h, c = gradient_local_search(best_h, N, t6,
                                  base_steps=[0.003, 0.001, 0.0005, 0.0002, 0.0001])
    if c < best_c:
        best_c = c
        best_h = h
    
    # Phase 7: Pair search (coarse to fine)
    t7 = min(t0 + 1050, deadline - 30)
    h, c = local_pair_search(best_h, N, t7,
                              steps=[0.003, 0.001, 0.0003, 0.0001])
    if c < best_c:
        best_c = c
        best_h = h
    
    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 10:
        h_final, c_final = slqp_run(best_h, N, 100000, maxiter=1500, deadline=deadline - 2)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Also evaluate at original resolution for comparison
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
