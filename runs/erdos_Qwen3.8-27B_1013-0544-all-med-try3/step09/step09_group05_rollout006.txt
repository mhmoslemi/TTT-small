                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, obj_fn, grad_fn, bounds, maxiter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 10, 'maxls': 40})
        if time.time() < deadline:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def gradient_local_search(h, n, deadline, max_iters=5000):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    base_step = 0.005
    steps_scale = [1.0, 0.5, 0.25, 0.1, 0.05]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Compute averaged gradient over active shifts
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Find top indices by gradient magnitude
        top_idx = np.argsort(-np.abs(grad))[:30]
        
        improved = False
        for sm in steps_scale:
            if time.time() > deadline:
                break
            step = base_step * sm
            for i_pos in range(len(top_idx)):
                if time.time() > deadline:
                    return h, best_c
                i = top_idx[i_pos]
                for j_pos in range(i_pos + 1, len(top_idx)):
                    j = top_idx[j_pos]
                    gi, gj = grad[i], grad[j]
                    # Move mass from higher gradient to lower (negative gradient direction)
                    if gi > gj:
                        di, dj = -step, step
                    else:
                        di, dj = step, -step
                    if h[i] + di >= 0 and h[j] + dj <= 1:
                        h[i] += di
                        h[j] += dj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= di
                            h[j] -= dj
        if not improved:
            base_step *= 0.6
            if base_step < 1e-7:
                break
    return h, best_c

def perturbation_restart(h, n, deadline, rng, num_restarts=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for r in range(num_restarts):
        if time.time() > deadline - 30:
            break
        # Random perturbation preserving feasibility
        perm = rng.permutation(n)
        h_pert = h.copy()
        perturb_amount = 0.02 / np.sqrt(n)
        delta = rng.standard_normal(n) * perturb_amount
        delta -= delta.sum()  # preserve sum
        h_pert += delta
        h_pert = project_feasible(h_pert, n)
        
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS on perturbed point
        t_sub = min(40, deadline - time.time() - 5)
        if t_sub < 10:
            break
        if c_pert < best_c + 0.005:
            # Try to optimize from this point
            active, _ = get_active_shifts(h_pert, n, top_k=8, rel_tol=0.01)
            alpha = 50000
            obj = lambda x: smooth_max_obj(x, n, alpha, active, 100.0)
            gr = lambda x: smooth_max_grad(x, n, alpha, active, 100.0)
            h_opt, c_opt = run_lbfgs(h_pert, n, time.time() + t_sub, obj, gr, bounds, maxiter=500)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        
        # Also try from current best with different active set
        active2, _ = get_active_shifts(best_h, n, top_k=5, rel_tol=0.002)
        alpha2 = 200000
        obj2 = lambda x: smooth_max_obj(x, n, alpha2, active2, 500.0)
        gr2 = lambda x: smooth_max_grad(x, n, alpha2, active2, 500.0)
        h_opt2, c_opt2 = run_lbfgs(best_h, n, time.time() + min(30, deadline - time.time() - 5),
                                    obj2, gr2, bounds, maxiter=400)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2
    
    return best_h, best_c

def active_set_optimize(h, n, deadline):
    """Iteratively refine the active set and optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(15):
        if time.time() > deadline - 10:
            break
        # Get current active shifts
        active, c_vec = get_active_shifts(best_h, n, top_k=8, rel_tol=0.005)
        
        # Try different alpha values
        for alpha in [50000, 200000, 800000, 2000000]:
            if time.time() > deadline - 10:
                break
            mu = 100.0 + alpha * 0.01
            obj = lambda x, a=alpha, act=active: smooth_max_obj(x, n, a, act, mu)
            gr = lambda x, a=alpha, act=active: smooth_max_grad(x, n, a, act, mu)
            h_new, c_new = run_lbfgs(best_h, n, time.time() + min(30, deadline - time.time() - 10),
                                      obj, gr, bounds, maxiter=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Phase 1: Full smooth max L-BFGS (moderate alpha)
    if time.time() < deadline - 400:
        t1_end = min(t0 + 150, deadline - 400)
        for alpha, mu in [(200, 50.0), (1000, 200.0), (5000, 1000.0)]:
            if time.time() > t1_end - 10:
                break
            obj = lambda x, a=alpha: full_smooth_obj(x, n, a, mu)
            gr = lambda x, a=alpha: full_smooth_grad(x, n, a, mu)
            h, best_c = run_lbfgs(h, n, min(time.time() + 60, t1_end), obj, gr, bounds, maxiter=3000)
    
    # Phase 2: Active set optimization (progressively tighter)
    if time.time() < deadline - 250:
        h, best_c = active_set_optimize(h, n, deadline - 150)
    
    # Phase 3: Very high alpha on top 5 shifts
    if time.time() < deadline - 150:
        active, _ = get_active_shifts(h, n, top_k=5, rel_tol=0.003)
        for alpha in [1000000, 5000000, 10000000]:
            if time.time() > deadline - 130:
                break
            mu = 500.0
            obj = lambda x, a=alpha, act=active: smooth_max_obj(x, n, a, act, mu)
            gr = lambda x, a=alpha, act=active: smooth_max_grad(x, n, a, act, mu)
            h, best_c = run_lbfgs(h, n, min(time.time() + 40, deadline - 110), obj, gr, bounds, maxiter=3000)
    
    # Phase 4: Gradient local search
    if time.time() < deadline - 80:
        h, best_c = gradient_local_search(h, n, deadline - 60, max_iters=8000)
    
    # Phase 5: Another active set round from potentially improved point
    if time.time() < deadline - 60:
        h, best_c = active_set_optimize(h, n, deadline - 30)
    
    # Phase 6: Perturbation restarts
    if time.time() < deadline - 20:
        h, best_c = perturbation_restart(h, n, deadline - 5, rng, num_restarts=8)
    
    # Final: one more local search
    if time.time() < deadline - 3:
        h, best_c = gradient_local_search(h, n, deadline - 2, max_iters=3000)
    
    # Ensure feasibility
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    return h, best_c, n
