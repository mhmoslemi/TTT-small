import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target_sum / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag]) * dx
            grad[0:n - lag] += w[k] * (-h[lag:n]) * dx
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n]) * dx
            grad[idx:n] += w[k] * (-h[0:n - idx]) * dx
    return grad

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:n] += (1.0 - h[0:n - lag]) * dx
        grad[0:n - lag] += (-h[lag:n]) * dx
    else:
        m = -lag
        grad[0:n - m] += (1.0 - h[m:n]) * dx
        grad[m:n] += (-h[0:n - m]) * dx
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0
    n_local = n
    alpha_local = alpha
    penalty_local = penalty_coeff
    target_local = target_sum

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full')
        obj_val = smooth_max_value(c, alpha_local)
        diff = x.sum() - target_local
        obj_val += penalty_local * diff * diff
        grad = smooth_max_grad(x, n_local, alpha_local)
        grad += 2.0 * penalty_local * diff * np.ones(n_local)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):

    x_dst = np.linspace(0, 2, n_dst)

def gradient_descent_polish(h, n, deadline, lr_init=0.05, maxiter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 500000
    lr = lr_init

    for it in range(maxiter):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(h, n, alpha)
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-12:
            break
        step = lr * grad / gnorm
        for trial_lr in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            h_new = h - trial_lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                break
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
        if it % 100 == 0:
            desired_alpha = min(1e10, 500000 * (1 + it // 500))
            if desired_alpha > alpha:
                alpha = desired_alpha

    return best_h, best_c
