from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2} using waterfilling."""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)

def gradient_flow(h, n, deadline, n_steps=5000, eta0=0.005):
    """Projected gradient descent on the smooth max objective."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    alpha = 5000000
    eta = eta0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        # Active set: all constraints within tol of max
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.02, 0.02)
        else:
            eta *= 0.95
    return best_h, best_c

def coordinate_descent(h, n, deadline, radius=30):
    """Local coordinate descent with momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def lbfgs_optimize(h, n, deadline, alphas, mu, maxit_list):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(200, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 20,
                                    'maxls': 40, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(25):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(100, int(rem2 * 1.5))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 20,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def gradient_flow(h, n, deadline, n_steps=20000, eta0=0.005):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = eta0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx_val
                grad[k:] += (-best_h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx_val
                grad[:nv] += (-best_h[kk:]) * dx_val
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.01, 0.02)
        else:
            eta *= 0.96
    return best_h, best_c

def single_constraint_dive(h, n, deadline, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = int(np.argmax(c_vec))
        active = np.array([worst])
        rem = deadline - time.time()
        if rem < 5:
            break
        maxit = max(200, int(rem * 3))
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(n, active, alpha, 0.0005),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 20,
                                    'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def make_diverse_starts(n, seed, count=4):
    rng = np.random.RandomState(seed)
    starts = []
    hb = (rng.rand(n) < 0.5).astype(float)
    starts.append(project_feasible(hb, n))
    xs = np.linspace(0, 2, n)
    hs = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * xs), n)
    starts.append(hs)
    hf = project_feasible(0.5 + 0.2 * np.sin(3 * np.pi * xs) + 0.1 * np.sin(7 * np.pi * xs), n)
    starts.append(hf)
    hr = project_feasible(0.5 + 0.25 * rng.randn(n), n)
    starts.append(hr)
    while len(starts) < count:
        hq = project_feasible(0.5 + 0.2 * rng.randn(n), n)
        starts.append(hq)
    return starts[:count]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except (NameError, Exception):
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution
    p1_dead = t0 + 55
    h1, c1 = lbfgs_optimize(best_h, best_n, p1_dead,
        alphas=[50, 500, 5000, 50000, 500000],
        mu=0.01, maxit_list=[1200, 1800, 2200, 2800, 3500])
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Upsample to n=1500 and optimize with annealed alpha
    n2 = 1500
    h2_init = upsample(best_h, best_n, n2)
    c2_init = compute_c5(h2_init, n2)
    if c2_init < best_c + 0.01:
        p2_dead = min(t0 + 260, deadline - 380)
        h2, c2 = lbfgs_optimize(h2_init, n2, p2_dead,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.003, maxit_list=[1000, 1200, 1500, 1800, 2000, 2200, 2500, 3000, 3500])
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2, n2

    # Phase 3: Active set with aggressive tracking
    if time.time() < deadline - 280:
        p3_dead = min(time.time() + 140, deadline - 230)
        h3, c3 = active_set_optimize(best_h, best_n, p3_dead, top_k=10,
            alphas=[50000, 200000, 1000000, 5000000])
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Gradient flow for fine-tuning
    if time.time() < deadline - 200:
        p4_dead = min(time.time() + 80, deadline - 160)
        h4, c4 = gradient_flow(best_h, best_n, p4_dead, n_steps=20000, eta0=0.004)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Multi-start from diverse initializations
    if time.time() < deadline - 180 and best_n >= 1000:
        diverse = make_diverse_starts(best_n, seed, count=3)
        for d_h in diverse:
            if time.time() > deadline - 100:
                break
            d_dead = min(time.time() + 55, deadline - 80)
            hd, cd = lbfgs_optimize(d_h, best_n, d_dead,
                alphas=[500, 5000, 50000, 500000, 2000000],
                mu=0.005, maxit_list=[600, 800, 1000, 1200, 1500])
            if cd < best_c:
                best_c, best_h = cd, hd

    # Phase 6: Active set final polish
    if time.time() < deadline - 100:
        p6_dead = min(time.time() + 70, deadline - 50)
        h6, c6 = active_set_optimize(best_h, best_n, p6_dead, top_k=6,
            alphas=[200000, 1000000, 5000000, 20000000])
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Ultra-focused single constraint dive
    if time.time() < deadline - 40:
        p7_dead = deadline - 5
        h7, c7 = single_constraint_dive(best_h, best_n, p7_dead,
            alphas=[5000000, 20000000, 100000000])
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
