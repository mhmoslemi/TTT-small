import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            scale = diff / tot
            h += scale * room
            h = np.clip(h, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            scale = diff / tot
            h += scale * h
            h = np.clip(h, 0, 1)
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

                                    'maxls': 50, 'maxcor': 80})

        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.03, 0.02)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 1):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = best_h.copy()
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = best_h.copy()
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def single_coordinate_run(h, n, deadline, n_coords=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n_coords is None:
        n_coords = n
    idx_list = list(range(n_coords))
    d_sigmas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in d_sigmas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in idx_list:
                if time.time() > deadline:
                    break
                for sign in [1.0, -1.0]:
                    new_val = best_h[i] + sign * d
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    old_val = best_h[i]
                    best_h[i] = new_val
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = best_h.copy()
                        improved = True
                    else:
                        best_h[i] = old_val
                        best_h = project_feasible(best_h, n)
                        break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 50
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 500000],
        mu=0.01, maxiters=[500, 800, 1200, 1500, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set at original resolution
    p2_end = t0 + 100
    h2, c2 = run_active_opt(best_h, best_n, p2_end, top_k=5,
        alphas=[100000, 1000000, 10000000, 100000000])
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Upsample to n=800 and optimize
    n_hi = 800
    if time.time() < deadline - 350:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c + 0.003:
            p3_end = min(time.time() + 150, deadline - 300)
            h3, c3 = run_lbfgs(h_hi, n_hi, p3_end,
                alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                mu=0.005, maxiters=[400, 600, 800, 1200, 1500, 2000, 2500, 3000])
            if c3 < best_c:
                best_c = c3
                best_h = h3
                best_n = n_hi

    # Phase 4: Active set at higher resolution
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 170)
        h4, c4 = run_active_opt(best_h, best_n, p4_end, top_k=4,
            alphas=[200000, 1000000, 5000000, 50000000, 100000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Single worst constraint repeated dive
    if time.time() < deadline - 130:
        p5_end = min(time.time() + 50, deadline - 100)
        bounds = [(0.0, 1.0)] * best_n
        for dive in range(10):
            if time.time() > p5_end:
                break
            c_vec = compute_c5_vec(best_h, best_n)
            worst = np.argmax(c_vec)
            active = np.array([worst])
            for a in [20000000, 100000000]:
                if time.time() > p5_end:
                    break
                try:
                    res = minimize(act_obj, best_h, jac=act_grad,
                                   args=(best_n, active, a, 0.0005),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 400, 'ftol': 1e-25,
                                            'gtol': 1e-20, 'maxfun': 1500,
                                            'maxls': 40, 'maxcor': 30})
                    h_new = project_feasible(res.x, best_n)
                    c_new = compute_c5(h_new, best_n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass

    # Phase 6: Stochastic local search (escape local optima)
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 50, deadline - 50)
        h6, c6 = stochastic_local_search(best_h, best_n, p6_end, rng,
                                          n_trials=6, perturb_k=6,
                                          perturb_mag=0.02,
                                          alphas=(50000, 500000, 5000000))
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Projected gradient refinement
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 25, deadline - 20)
        h7, c7 = projected_grad_refine(best_h, best_n, p7_end, max_steps=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search
    if time.time() < deadline - 15:
        p8_end = min(time.time() + 15, deadline - 5)
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=25)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Single coordinate on worst region
    if time.time() < deadline - 3:
        p9_end = deadline - 1
        c_vec = compute_c5_vec(best_h, best_n)
        # Focus on coordinates where h is "active" (not 0 or 1)
        active_coords = np.where((best_h > 0.01) & (best_h < 0.99))[0]
        if len(active_coords) > 0:
            h9, c9 = single_coordinate_run(best_h, best_n, p9_end,
                                           n_coords=len(active_coords))
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Final: single worst constraint push
    if time.time() < deadline - 0.5:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 500000000, 0.0005),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 200, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': 500,
                                    'maxls': 30, 'maxcor': 20})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
