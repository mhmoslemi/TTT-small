import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    omx = 1.0 - x
    for mm in range(len(w)):
        wi = w[mm]
        if wi < 1e-14:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * omx[k:]
            grad[k:] -= wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * omx[:nv]
            grad[:nv] -= wm * x[kk:]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, alpha, mu):

        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)

    dx = 2.0 / n
    grad = np.zeros(n)

def upsample(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old + 1)
    x_new = np.linspace(0, 2, n_new + 1)
    h_ext = np.concatenate([h, [h[0]]])
    h_interp = np.interp(x_new, x_old, h_ext)
    return project_feasible(h_interp[:n_new], n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_list=None, topk_mode=False, topk_size=8):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 30:
            break

            c_vec = compute_c5_vec(best_h, n)
            mx = c_vec.max()
            thresh = mx - max(1e-9, 0.001 * mx)
            active = np.where(c_vec >= thresh)[0]
            if len(active) > topk_size:

            g = topk_grad

            maxit = maxit_list[i]

                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})

                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-12:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

            ones = no
            zeros = nz

            best_h[oi] = 1.0
            best_h[oz] = 0.0
    final = project_feasible(best_h, n)

        best_c = final_c
