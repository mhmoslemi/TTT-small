import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def smooth_max_topk_grad(c, alpha, x, n, active):
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def smooth_max_topk_val(c, alpha, active):
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_topk_val(c, alpha, active)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_topk_grad(c, alpha, x, n, active)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)

def gradient_step_search(h, n, deadline, step_sizes=None, max_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    dx_val = 2.0 / n
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:20]:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active[:20])
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            if len(pos_idx) > 30:
                pos_idx = pos_idx[np.argsort(-grad[pos_idx])[:30]]
            if len(neg_idx) > 30:
                neg_idx = neg_idx[np.argsort(grad[neg_idx])[:30]]
            best_gain = 0
            best_i = -1
            best_j = -1
            for ii in pos_idx:
                gi = grad[ii]
                for jj in neg_idx:
                    g = gi - grad[jj]
                    if g > best_gain:
                        best_gain = g
                        best_i = ii
                        best_j = jj
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=60, deltas=None, max_evals=200000):

def make_beatty(n, frac):
    h = np.zeros(n)
    positions = set()
    n_ones = n // 2
    for k in range(1, 4 * n):
        pos = int(k * frac) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            for idx in zeros:
                add = min(remaining, 1.0)
                h[idx] += add
                remaining -= add
                if abs(remaining) < 1e-10:
                    break
    return project_feasible(h, n)

def interpolate_h(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old

    h_new = np.interp(x_new, x_old, np.concatenate([h_old[-1:], h_old, h_old[:1]]))
    return project_feasible(h_new, n_new)

            best_h = h_a

    # Phase 4: Gradient-guided search
    t_phase4_end = min(t0 + 800, deadline - 150)
    if time.time() < t_phase4_end:
        h_g, c_g = gradient_step_search(best_h, best_n, t_phase4_end,
                                         step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                         max_steps=80000)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    # Phase 5: Perturb and re-optimize (escape local optima)
    t_phase5_end = min(t0 + 950, deadline - 80)
    if time.time() < t_phase5_end:
        h_p, c_p = perturb_reoptimize(best_h, best_n, t_phase5_end, rng,
                                       n_perturb=15, n_restarts=8)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    # Phase 6: Fine local search
    t_phase6_end = min(t0 + 1050, deadline - 30)
    if time.time() < t_phase6_end:
        h_f, c_f = fine_local_search(best_h, best_n, t_phase6_end,
                                       radius=50,
                                       deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                       max_evals=100000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 5:
        h_s, c_s = slqp_polish(best_h, best_n, deadline - 2,
                                alphas=[1000000, 5000000], maxiter=300)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 8: Very fine final adjustments
    if time.time() < deadline - 1:
        h_v, c_v = fine_local_search(best_h, best_n, deadline - 1,
                                       radius=30,
                                       deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                       max_evals=30000)
        if c_v < best_c:
            best_c = c_v
            best_h = h_v

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
