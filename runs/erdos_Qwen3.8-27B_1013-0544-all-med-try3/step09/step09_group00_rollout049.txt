But actually, for a continuous variable (not just binary), the "swap" is different. We can:
1. Transfer δ from h[i] to h[j]: h[i] -= δ, h[j] += δ
2. This maintains the sum constraint
3. We need h[i] ≥ 0 and h[j] ≤ 1

For binary-like optimization, δ is either 0 or 1 (flip) or h[i]-h[j] (swap).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorize: for each lag mm, k = mm - (n-1)
    k_vals = np.arange(len(c)) - (n - 1)  # shape (2n-1,)
    # Positive k: grad[0:n-k] += w*dx*(1-x[k:n]) and grad[k:n] += w*dx*(-x[0:n-k])
    # Negative k: grad[-k:n] += w*dx*(1-x[0:n+k]) and grad[0:n+k] += w*dx*(-x[-k:n])
    
    pos = k_vals >= 0
    neg = k_vals < 0
    
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = k_vals[mm]
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad
```

Actually, the gradient loop is O(n²) because for each of the 2n-1 lags, we update O(n) elements. For n=500, that's about 10^6 operations. For n=1000, about 4*10^6. This is fine for L-BFGS but might be slow if done thousands of times.

```python
def make_grad_vectors(h, n):
    """Precompute gradient vectors for all positions."""
    k_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    grads = np.zeros((n, 2*n-1))
    for i in range(n):
        valid1 = (i + k_vals >= 0) & (i + k_vals < n)
        grads[i, valid1] += (1.0 - h[i + k_vals[valid1]])
        valid2 = (i - k_vals >= 0) & (i - k_vals < n)
        grads[i, valid2] -= h[i - k_vals[valid2]]
    return grads
```

Wait, this is O(n² * n) = O(n³) total, which for n=500 is 1.25*10^8. That's too slow to recompute at each SA step.

```python
def single_grad(h, n, i):
    k_vals = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    valid1 = (i + k_vals >= 0) & (i + k_vals < n)
    g[valid1] += (1.0 - h[i + k_vals[valid1]])
    valid2 = (i - k_vals >= 0) & (i - k_vals < n)
    g[valid2] -= h[i - k_vals[valid2]]
    return g
```

This is O(n) per call. For a pair (i,j), we compute two vectors and take the difference. Total O(n) per SA step.

But in Python, even O(n) with numpy takes some time due to array creation. For n=500, maybe 50μs per step. With 1M steps, that's 50s. Doable.

Actually, let me think about this differently. For the SA approach, I don't need the full c_vec update. I just need to know if the max increases.

If I change h[i] by +δ and h[j] by -δ:
- c_vec changes by δ*(g_i - g_j) where g_i, g_j are the single-point gradients
- The new c_vec = c_vec + δ*(g_i - g_j)
- The max of the new c_vec

To avoid computing the full c_vec each time, I can:
1. Track c_vec explicitly
2. Update it incrementally: c_vec += δ*(g_i - g_j)
3. The max is just c_vec.max() - O(n) scan

For n=500, c_vec has 999 elements. Scanning for max is trivial.

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14 * max(1, n):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        g += 2*mu*(x.sum() - n/2.0)
    return g

def run_lbfgs(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)]*n
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        maxit = max(500, int((deadline - time.time()) * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit*10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except:
            pass
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 5*n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n/2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def interleave_parent_to_n(h_parent, n_parent, n_target):
    """Interpolate parent to target resolution."""
    x_parent = np.linspace(0, 2, n_parent, endpoint=False)
    x_target = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_target, x_parent, h_parent)
    return project_feasible(h_interp, n_target)
