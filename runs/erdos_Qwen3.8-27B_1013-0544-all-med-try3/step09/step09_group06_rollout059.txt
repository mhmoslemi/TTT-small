from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def adam_optimize(h, n, deadline, alpha=500, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-based gradient descent with smooth max."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.time() < deadline:
        t += 1
        # Adaptive alpha: increase over time to focus more on the max
        alpha_t = alpha * (1 + 0.1 * t / 100)
        
        # Compute gradient
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        shifted = alpha_t * (c - c.max())
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        n_active = np.sum(w > 1e-8)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            lag = k - (n - 1)
            m_off = abs(lag)
            if m_off >= n:
                continue
            if lag == 0:
                grad[:] += w[k] * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[m_off:n] += w[k] * (1.0 - h[0:n-m_off])
                grad[0:n-m_off] += w[k] * (-h[m_off:n])

                grad[m_off:n] += w[k] * (-h[0:n-m_off])
        grad *= dx

        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        h_new = project_feasible(h_new, n)

        if c_new < best_c:

            best_h = h_new.copy()
        h = h_new

    h = h.copy()
    best_c = compute_c5(h, n)

        c_all = compute_all_lags(h, n)

            # For each pair, determine direction (should h[i] decrease and h[i+m] increase, or vice versa)
            diffs = h[pairs] - h[pairs + m_off]

                if time.time() > deadline:

                # Positive diff: h[i] > h[i+m], try to reduce h[i] and increase h[i+m]
                pos_mask = diffs > delta * 0.5
                neg_mask = diffs < -delta * 0.5
                
                for mask in [pos_mask, neg_mask]:
                    if not np.any(mask):

                    idxs = np.where(mask)[0]
                    rng_idx = np.random.choice(idxs, size=min(20, len(idxs)), replace=False)
                    for idx in rng_idx:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def adam_optimize(h_init, n, deadline, alpha_start=200, alpha_end=2000, lr=0.05, max_iter=5000):

    t = 0

        # Adaptive alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** (t / total_iters)

        shifted = alpha * (c - c_max)

        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        grad = np.zeros(n)

                continue

                grad[m_off:n] += w[k] * (-h[0:n-m_off])

def lp_feasibility_step(h, n, T, deadline):
    """Try to find h' close to h that satisfies all C(k) ≤ T."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    
    # Build A matrix for linearized constraints
    # C(k; h') ≈ c[k] + grad_c(k) @ (h' - h)
    # Want: c[k] + grad_c(k) @ h' ≤ T + grad_c(k) @ h
    # i.e.: grad_c(k) @ h' ≤ T + grad_c(k) @ h - c[k]
    
    m_lags = 2 * n - 1
    A = np.zeros((m_lags, n))
    b = np.zeros(m_lags)
    
    for k in range(m_lags):
        lag = k - (n - 1)
        m_off = abs(lag)
        if m_off >= n:
            b[k] = np.inf  # no constraint
            continue
        if lag == 0:
            g = (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            g[m_off:n] = (1.0 - h[0:n-m_off])
            g[0:n-m_off] = -h[m_off:n]
        else:
            g = np.zeros(n)
            g[0:n-m_off] = (1.0 - h[m_off:n])
            g[m_off:n] = -h[0:n-m_off]
        
        A[k, :] = g * dx
        b[k] = T + np.dot(g * dx, h) - c[k]
    
    # Also add: sum(h) = n/2 (equality)
    # And: 0 ≤ h ≤ 1 (bounds)
    
    # Minimize 0 (feasibility problem)
    c_obj = np.zeros(n)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n
    
    # Filter out infinite b entries
    valid = np.isfinite(b)
    A_valid = A[valid]
    b_valid = b[valid]
    
    if A_valid.shape[0] == 0:
        return h
    
    try:
        res = linprog(c_obj,
                     A_ub=A_valid, b_ub=b_valid,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds,
                     method='highs')
        if res.success:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h

def compute_swap_delta(h, n, i, j, delta, c_all):
    """Compute change in all lags when h[i] -= delta and h[j] += delta."""
    dx = 2.0 / n
    delta_c = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m_off = abs(lag)
        if m_off >= n:
            continue
        # ∂C(lag)/∂h[i]
        gi = 0.0
        if lag == 0:
            gi = (1 - 2 * h[i])
        elif lag > 0:
            if i < n - lag:
                gi += (1 - h[i + lag])
            if i >= lag:
                gi += (-h[i - lag])
        else:  # lag < 0
            m = -lag
            if i < n - m:
                gi += (1 - h[i + m])
            if i >= m:
                gi += (-h[i - m])

        if lag == 0:
            gj = (1 - 2 * h[j])

            if j < n - m:
                gj += (1 - h[j + m])
            if j >= m:
                gj += (-h[j - m])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max_vec(h, n, alpha):
    """Vectorized gradient of smooth max."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-7:
            continue
        lag = k - (n - 1)
        m_off = abs(lag)
        if m_off >= n:
            continue
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[m_off:] += w[k] * (1.0 - h[:n-m_off])
            grad[:n-m_off] += w[k] * (-h[m_off:])
        else:
            grad[:n-m_off] += w[k] * (1.0 - h[m_off:])
            grad[m_off:] += w[k] * (-h[:n-m_off])
    return grad * dx, c

def adam_optimize(h_init, n, deadline, alpha_start=100, alpha_end=5000, lr=0.05):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    t = 0
    
    while time.time() < deadline:
        t += 1
        alpha = alpha_start * (alpha_end / alpha_start) ** (t / 10000.0)
        alpha = min(alpha, 50000)
        
        grad, c = grad_smooth_max_vec(h, n, alpha)
        
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = c.max()  # approx (should be close to compute_c5)
        c_new = compute_c5(h_new, n)  # exact
        h = h_new
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        
        if t % 2000 == 0:
            lr *= 0.85
    
    return best_h, best_c

# Constraint functions for SLSQP (top-level, no closures)
def _c5_sum_eq_fun(x, n):
    return np.sum(x) - n / 2.0

def _c5_sum_eq_jac(x, n):
    return np.ones(len(x))

def _c5_slsqp_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    shifted = np.clip(shifted, -500, 500)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _c5_slsqp_grad(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        if w[j] < 1e-8:
            continue
        lag = k - (n - 1)
        m_off = abs(lag)
        if m_off >= n:
            continue
        if lag == 0:
            grad += w[j] * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[m_off:] += w[j] * (1.0 - x[:n-m_off])
            grad[:n-m_off] += w[j] * (-x[m_off:])
        else:
            grad[:n-m_off] += w[j] * (1.0 - x[m_off:])
            grad[m_off:] += w[j] * (-x[:n-m_off])
    return grad * dx

def slsqp_polish(h, n, top_m=5, alpha=100000, maxiter=2000):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c5_sum_eq_fun, 'jac': _c5_sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_c5_slsqp_obj, h, jac=_c5_slsqp_grad,
                      args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    best_c = float(c_all.max())

        lag_star = k_star - (n - 1)
        m_star = abs(lag_star)
        if m_star == 0 or m_star >= n:
            break
        
        improved = False
        for delta in [0.04, 0.02, 0.01, 0.005, 0.002]:
            if time.time() > deadline:

            for i in range(n - m_star):
                if time.time() > deadline:
                    break
                j = i + m_star
                diff = h[i] - h[j]
                if diff < 1e-12:
                    continue
                step = min(delta, abs(diff) * 0.5)
                if step < 1e-12:
                    continue
                direction = 1.0 if diff > 0 else -1.0
                # h[i] -= direction*step, h[j] += direction*step
                # Compute delta_c
                delta_c = np.zeros(n_lags)
                for k in range(n_lags):
                    lag = k - (n - 1)
                    if lag > 0:
                        gi = 0.0
                        if i < n - lag: gi += (1.0 - h[i + lag])
                        if i >= lag: gi += (-h[i - lag])
                        gj = 0.0
                        if j < n - lag: gj += (1.0 - h[j + lag])
                        if j >= lag: gj += (-h[j - lag])
                    elif lag < 0:
                        m2 = -lag
                        gi = 0.0
                        if i < n - m2: gi += (1.0 - h[i + m2])
                        if i >= m2: gi += (-h[i - m2])
                        gj = 0.0
                        if j < n - m2: gj += (1.0 - h[j + m2])
                        if j >= m2: gj += (-h[j - m2])
                    else:
                        gi = 1.0 - 2.0 * h[i]
                        gj = 1.0 - 2.0 * h[j]
                    delta_c[k] = dx * direction * step * (gi - gj)
                
                new_max = (c_all + delta_c).max()
                if new_max < best_c - 1e-15:
                    h[i] -= direction * step
                    h[j] += direction * step
                    c_all = c_all + delta_c
                    best_c = new_max
                    improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)
