                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)

        if not improved:
            # Small perturbation
            h = best_h + rng.normal(0, 0.002, n)
            h = project_feasible(h, n)
            ch = compute_c5(h, n)
            if ch < best_c:
                best_c = ch
                best_h = h.copy()
            if time.time() > deadline - 30:
                break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = None
        n = 500

    if h_init is not None:
        best_h = project_feasible(h_init, n)
    else:
        best_h = project_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick L-BFGS polish with moderate alpha
    remaining = deadline - time.time()
    if remaining > 80:
        t_end = min(deadline - 70, time.time() + 50)
        h_opt = lbfgsb_optimize(best_h, n, 200000, t_end, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Higher alpha L-BFGS
    remaining = deadline - time.time()
    if remaining > 100:
        t_end = min(deadline - 90, time.time() + 50)
        h_opt = lbfgsb_optimize(best_h, n, 2000000, t_end, maxiter=2500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Gradient-guided local search
    remaining = deadline - time.time()
    if remaining > 300:
        ls_deadline = min(deadline - 280, time.time() + 220)
        h_opt, c_opt = gradient_guided_search(best_h, n, ls_deadline, rng, topk=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS polish after local search
    remaining = deadline - time.time()
    if remaining > 100:
        t_end = min(deadline - 90, time.time() + 55)
        h_opt = lbfgsb_optimize(best_h, n, 10000000, t_end, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Balancing search
    remaining = deadline - time.time()
    if remaining > 200:
        bal_deadline = min(deadline - 180, time.time() + 150)
        h_opt, c_opt = balancing_search(best_h, n, bal_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: More local search
    remaining = deadline - time.time()
    if remaining > 150:
        ls_deadline = min(deadline - 130, time.time() + 120)
        h_opt, c_opt = gradient_guided_search(best_h, n, ls_deadline, rng, topk=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Try n=600
    remaining = deadline - time.time()
    if remaining > 180 and n < 600:
        n600 = 600
        h600 = interpolate_resolution(best_h, n, n600)
        t_end = min(deadline - 140, time.time() + 60)
        h600 = lbfgsb_optimize(h600, n600, 500000, t_end, maxiter=2000)
        c600 = compute_c5(h600, n600)
        if c600 < best_c:
            best_c = c600
            best_h = h600
            n = n600
            # Quick local search at n600
            remaining = deadline - time.time()
            if remaining > 80:
                ls_end = min(deadline - 60, time.time() + 60)
                h_opt, c_opt = gradient_guided_search(best_h, n, ls_end, rng, topk=4)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 8: Final L-BFGS polish
    remaining = deadline - time.time()
    if remaining > 40:
        n_final = len(best_h)
        t_end = min(deadline - 15, time.time() + 30)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, t_end, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final micro local search
    remaining = deadline - time.time()
    if remaining > 15:
        n_final = len(best_h)
        ls_end = min(deadline - 2, time.time() + 12)
        h_opt, c_opt = gradient_guided_search(best_h, n_final, ls_end, rng, topk=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
