import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        elif diff < -1e-14:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_w(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_w(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_opt(h, n, alpha, deadline, maxiter=3000, pen=1e8):
    target_sum = n / 2.0
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        ov = smooth_max_val(c, alpha)
        d = x.sum() - target_sum
        ov += pen * d * d
        g = smooth_max_grad(x, n, alpha) + 2.0 * pen * d
        return ov, g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def gradient_momentum(h, n, alpha, deadline, lr=0.002, momentum=0.9, max_iters=50000):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        c_all = compute_all_lags(h, n)
        g = smooth_max_grad(h, n, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        if it % 2000 == 0:
            lr *= 0.9
    return best_h, best_c

def targeted_local_search(h, n, deadline, rng, top_k=5, min_step=5e-6):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_scale = 1.0
    stall = 0
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False
        for scale in [1.0, 0.6, 0.3, 0.15, 0.07, 0.03]:
            if time.time() > deadline:
                break
            base = 0.015 * scale

                    break
                ki = top_idx[rng.randint(top_k)]
                lag = ki - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                i = rng.randint(0, n - m)
                j = i + m
                if h[i] > h[j] + 1e-8:
                    d = min(base, (h[i] - h[j]) * 0.3)
                    if d < min_step:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_t = compute_c5(h, n)
                    if c_t < best_c - 1e-15:
                        best_c = c_t
                        best_h = h.copy()
                        improved = True
                        c_all = compute_all_lags(h, n)
                        top_idx = np.argsort(c_all)[-top_k:]
                    else:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    d = min(base, (h[j] - h[i]) * 0.3)
                    if d < min_step:
                        continue
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_t = compute_c5(h, n)
                    if c_t < best_c - 1e-15:
                        best_c = c_t
                        best_h = h.copy()

                    else:

                        h = project_feasible(h, n)
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    return best_h, best_c

def coarse_to_fine(h_src, n_src, n_coarse, deadline, alpha_start=500):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_c = np.interp(x_coarse, x_src, h_src)
    h_c = project_feasible(h_c, n_coarse)
    alphas = [alpha_start, alpha_start * 10, alpha_start * 100, alpha_start * 500]
    for a in alphas:
        if time.time() > deadline:
            break
        h_c = lbfgsb_opt(h_c, n_coarse, a, time.time() + 30, maxiter=800)
    x_fine = np.linspace(0, 2, n_src, endpoint=False)
    h_f = np.interp(x_fine, x_coarse, h_c)
    return project_feasible(h_f, n_src)

def simulated_anneal(h, n, deadline, rng, alpha_sa=100000, t_init=0.005, t_floor=2e-5, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stall = 0
    while time.time() < deadline and T > t_floor:
        r = rng.random()
        if r < 0.6:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.1, 0.005 + T * 6)
            if h[i] > h[j]:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.5, (1.0 - h[j]) * 0.5)
            elif h[j] > h[i]:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.5, -(1.0 - h[i]) * 0.5)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-4:]
            k_star = top_k[rng.randint(min(4, len(top_k)))]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-7:
                d = rng.uniform(0.002, 0.05)
                d = min(d, (h[i] - h[j]) * 0.4)
            elif h[j] > h[i] + 1e-7:
                d = -rng.uniform(0.002, 0.05)
                d = max(d, -(h[j] - h[i]) * 0.4)
            else:
                T *= cool
                continue
        if abs(d) < 1e-10:
            T *= cool
            continue
        c_old = compute_all_lags(h, n)
        e_old = smooth_max_val(c_old, alpha_sa)
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_val(c_new, alpha_sa)
        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_floor:
            if -de / T < 25:
                accept = rng.random() < np.exp(-de / T)
        if accept:
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            stall += 1
        T *= cool
        if stall > 50000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            stall = 0
            T = max(T, 0.003)
    return best_h, best_c

def interpolate_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
