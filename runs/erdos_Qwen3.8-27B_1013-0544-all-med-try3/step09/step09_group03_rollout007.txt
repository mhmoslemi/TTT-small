```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk

    return grad

        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 300000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new.copy()
        except Exception:
            pass
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgrad_one_shift(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n])
            grad[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv])
            grad[0:nv] += (-h[kk:n])
    return grad * dx

def subgradient_method(h, n, deadline, lr_init=0.02, n_iter=20000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    c_vec = compute_c5_vec(h, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 10 == 0:
            c_vec = compute_c5_vec(h, n)
        k_worst = int(np.argmax(c_vec))
        k = k_worst - (n - 1)
        grad = subgrad_one_shift(h, n, k)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.05, 0.08)
        else:

    return best_h, best_c

def pair_swap_descent(h, n, deadline, n_passes=200, n_trials_per_pass=300):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(42)

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (15000, 8000.0), (50000, 30000.0),
                   (100000, 60000.0), (200000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 200000, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

                mags=[0.004, 0.008, 0.012, 0.003, 0.015, 0.006, 0.01, 0.005,
                      0.02, 0.007])
            # Quick SLSQP after perturbation
            t6b = min(time.time() + 25, deadline - 30)
            if t6b > time.time() + 5:
                h6b, c6b = slqp_polish(best_h, n1, t6b, alpha=8000000, maxiter=4000)
                if c6b < best_c:
                    best_c = c6b
                    best_h = h6b

    # Phase 6: Directed refinement (30s)
    if time.time() < deadline - 50:
        t7 = min(time.time() + 25, deadline - 20)
        if t7 > time.time() + 5:
            best_h, best_c = directed_refine(best_h, n1, t7, n_rounds=500)

    # Phase 7: Upsample to n=2000 (200s)
    n2 = 2000
    if time.time() < deadline - 150:
        h2000 = interpolate_to(best_h, n1, n2)
        c2000 = compute_c5(h2000, n2)

        t8 = min(time.time() + 100, deadline - 60)
        if t8 > time.time() + 10:
            h2000, c2000 = run_lbfgs(h2000, n2, t8,
                configs=[(2000, 500.0), (8000, 3000.0), (30000, 15000.0),
                         (80000, 50000.0), (200000, 100000.0)])

        if c2000 < best_c:
            best_c = c2000
            best_h = h2000

            # SLSQP at n=2000
            t9 = min(time.time() + 50, deadline - 25)
            if t9 > time.time() + 8:
                h9, c9 = slqp_polish(best_h, n2, t9, alpha=10000000, maxiter=3000)
                if c9 < best_c:
                    best_c = c9
                    best_h = h9

            # Subgradient at n=2000
            t10 = min(time.time() + 30, deadline - 15)
            if t10 > time.time() + 5:
                h10, c10 = subgradient_method(best_h, n2, t10, lr_init=0.01, n_iter=20000)
                if c10 < best_c:
                    best_c = c10
                    best_h = h10

    # Phase 8: Final directed refinement (20s)
    if time.time() < deadline - 8:
        t11 = min(time.time() + 18, deadline - 5)
        if t11 > time.time() + 3:
            best_h, best_c = directed_refine(best_h, len(best_h), t11, n_rounds=300)

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
