```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_smooth_obj(x, n, alpha, k_top, mu):
    """Smooth max using only top-k shifts."""
    c = compute_c5_vec(x, n)
    # Get top k shifts
    if k_top < len(c):
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        c_top = c[top_idx]
    else:
        c_top = c
    m = c_top.max()
    sh = alpha * (c_top - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_smooth_grad(x, n, alpha, k_top, mu):
    """Gradient of active smooth max."""
    c = compute_c5_vec(x, n)
    if k_top < len(c):
        top_idx = np.argpartition(c, -k_top)[-k_top:]
    else:
        top_idx = np.arange(len(c))
    c_top = c[top_idx]
    m = c_top.max()
    sh = alpha * (c_top - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

        except Exception:

    """Coordinate descent: try flipping individual values."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n

    center = n - 1

def binary_push(h, n):
    """Push towards binary while maintaining sum."""
    h_b = np.zeros(n)
    sorted_idx = np.argsort(-h)
    # Set top n/2 to 1, rest to 0
    h_b[sorted_idx[:n//2]] = 1.0
    # But we need sum = n/2 exactly, so this works if n is even
    if n % 2 == 0:
        return h_b
    else:
        # Need one fractional value
        h_b[sorted_idx[:n//2]] = 1.0
        h_b[sorted_idx[n//2]] = h[sorted_idx[n//2]] - h[sorted_idx[n//2 + 1]]
        # Actually let's just use project_feasible
        return project_feasible(h_b, n)

            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)

        return np.ones(na)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (10000, 3000.0), (50000, 20000.0), (200000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 500000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000, k_top=10):
    def obj_fn(x, na, al, kt):
        c = compute_c5_vec(x, na)
        if kt < len(c):
            top_idx = np.argpartition(c, -kt)[-kt:]
            c_top = c[top_idx]
        else:
            c_top = c
        m = c_top.max()
        sh = al * (c_top - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha, k_top), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def coordinate_descent(h, n, deadline, max_rounds=200, delta_range=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if delta_range is None:
        delta_range = [0.5, -0.5, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        progress = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            old_val = h[i]
            for delta in delta_range:
                new_val = old_val + delta
                if new_val < -0.01 or new_val > 1.01:
                    continue
                new_val = max(0.0, min(1.0, new_val))
                h[i] = new_val
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    progress = True
                    break
                else:
                    h[i] = old_val
        if not progress and rnd > 15:
            break
    return best_h, best_c

def binary_push(h, n):
    h_b = np.zeros(n)
    sorted_idx = np.argsort(-h)
    half = n // 2
    h_b[sorted_idx[:half]] = 1.0
    if n % 2 == 1:
        h_b[sorted_idx[half]] = 2.0 * h[sorted_idx[half]] - 1.0
        h_b = np.clip(h_b, 0, 1)
    return project_feasible(h_b, n)

    return project_feasible(h_new, n_new)

            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Phase 9: Coordinate descent at higher res
    if time.time() < deadline - 100:
        t10 = min(time.time() + 50, deadline - 50)
        if t10 > time.time() + 3:
            best_h, best_c = coordinate_descent(best_h, len(best_h), t10, max_rounds=80,
                delta_range=[0.01, -0.01, 0.005, -0.005, 0.02, -0.02])

    # Phase 10: Upsample to n=2000 for final push
    n2 = 2000
    if time.time() < deadline - 80:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        t11 = min(time.time() + 50, deadline - 30)
        if t11 > time.time() + 5:
            h2000, c2000 = run_lbfgs(h2000, n2, t11,
                configs=[(2000, 500.0), (10000, 3000.0), (50000, 20000.0), (200000, 100000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000

        if time.time() < deadline - 15:
            h_final, c_final = coordinate_descent(best_h, len(best_h), deadline - 5,
                max_rounds=40, delta_range=[0.005, -0.005, 0.01, -0.01])
            if c_final < best_c:
                best_c = c_final
                best_h = h_final

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
