import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    if span < 1e-14:
        return 0.0
    n_pts = 150
    deltas = np.linspace(delta_min, delta_max, n_pts)
    best_val = 1e18
    best_t = 0

                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)

    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: L-BFGS at original resolution with alpha annealing
    p1_end = t0 + 250
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[30, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.005,
        maxit_per=[1500, 2000, 2000, 3000, 3000, 4000, 5000, 5000, 5000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set refinement at original resolution
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 120, deadline - 400)
        h2, c2 = run_active_set(best_h, best_n, p2_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to 2x resolution
    n_hi = n0 * 2
    if time.time() < deadline - 380:
        h_hi = upsample(best_h, best_n, n_hi)
        p3_end = min(time.time() + 280, deadline - 280)
        h3, c3 = run_lbfgs(h_hi, n_hi, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
            mu=0.003,
            maxit_per=[1000, 1500, 2000, 2500, 3000, 4000, 5000, 5000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_hi
    
    # Phase 4: Active set at higher resolution
    if time.time() < deadline - 300:
        p4_end = min(time.time() + 100, deadline - 250)
        h4, c4 = run_active_set(best_h, best_n, p4_end, top_k=8,
            alphas=[200000, 1000000, 5000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = len(h4)
    
    # Phase 5: 1D minimax pair search (key improvement)
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 180, deadline - 150)
        h5, c5 = pair_minimax_search(best_h, best_n, p5_end, top_k=35, max_sweeps=500)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    if time.time() < deadline - 100:

            deltas=[0.003, 0.001, 0.0003, 0.0001])
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = len(h6)
    
    # Phase 7: Another round of 1D minimax
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 45, deadline - 30)
        h7, c7 = pair_minimax_search(best_h, best_n, p7_end, top_k=40, max_sweeps=300)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = len(h7)
    
    # Phase 8: SLSQP polish
    if time.time() < deadline - 20:
        for alp in [500000, 2000000, 10000000]:
            if time.time() > deadline - 8:
                break
            h8, c8 = slqp_polish(best_h, best_n, deadline - 5, alpha=alp, maxiter=500)
            if c8 < best_c:
                best_c = c8
                best_h = h8
                best_n = len(h8)
    
    # Phase 9: Final 1D minimax polish
    if time.time() < deadline - 3:
        h9, c9 = pair_minimax_search(best_h, best_n, deadline - 1, top_k=50, max_sweeps=200)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = len(h9)

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
