def project_feasible(h, n):

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def block_swap(h, n, rng, n_blocks=8):
    """Randomly swap and reverse blocks of h"""
    h_new = h.copy()
    block_size = n // n_blocks
    for _ in range(rng.integers(2, 5)):
        i = rng.integers(0, n_blocks - 1)
        j = rng.integers(0, n_blocks - 1)
        if i == j:
            continue
        si, sj = i * block_size, j * block_size
        h_new[si:si+block_size], h_new[sj:sj+block_size] = h_new[sj:sj+block_size].copy(), h_new[si:si+block_size].copy()
    if rng.random() < 0.3:
        # Reverse a block
        i = rng.integers(0, n_blocks)
        si = i * block_size
        h_new[si:si+block_size] = h_new[si:si+block_size][::-1]
    return project_feasible(h_new, n)

def subgrad_step(h, n, active, eta):
    """One subgradient step on the active constraints"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= len(active)
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new

def project_feasible(h, n):
    target = n / 2.0

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def block_swap_perturb(h, n, rng, n_blocks=10):
    h_new = h.copy()
    block_size = n // n_blocks

    for _ in range(rng.integers(2, 4)):

        j = rng.integers(0, n_blocks - 1)

        si = i * block_size
        sj = j * block_size
        tmp = h_new[si:si + block_size].copy()
        h_new[si:si + block_size] = h_new[sj:sj + block_size]

    """Focus intensely on reducing the single worst shift."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for it in range(100):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst_idx = np.argmax(c_vec)
        active = np.array([worst_idx])

        # Try multiple alphas for the single worst constraint
        for a in [alpha, alpha * 10, alpha * 100]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, a, 0.0005), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': 2000,
                                        'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

        # Also try top-3 worst
        c_vec2 = compute_c5_vec(best_h, n)
        top3 = np.argsort(-c_vec2)[:3]
        for a in [alpha * 5]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, top3, a, 0.0005), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': 3000,
                                        'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    return best_h, best_c


def multi_start_refine(h, n, deadline, rng, n_starts=15):
    """Multiple structured perturbations followed by quick local refinement."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        r = rng.random()
        if r < 0.3:
            candidate = block_swap_perturb(best_h, n, rng, n_blocks=rng.integers(6, 20))
        elif r < 0.6:
            scale = 0.02 + 0.03 * rng.random()
            candidate = best_h + rng.standard_normal(n) * scale
            candidate = project_feasible(candidate, n)
        elif r < 0.8:
            # Sinusoidal perturbation
            freq = rng.integers(2, 15)
            phase = rng.random() * 2 * np.pi
            amp = 0.03 + 0.05 * rng.random()
            x = np.linspace(0, 2, n)
            candidate = best_h + amp * np.sin(2 * np.pi * freq * x / 2 + phase)
            candidate = project_feasible(candidate, n)
        else:
            # Invert a random interval
            candidate = best_h.copy()
            start = rng.integers(0, n - n // 10)
            length = rng.integers(n // 20, n // 4)
            candidate[start:start + length] = 1.0 - candidate[start:start + length]
            candidate = project_feasible(candidate, n)

        # Quick local refinement
        time_left = deadline - time.time()
        if time_left < 15:
            break
        alloc = min(20, max(5, int(time_left / 3)))
        h_ref, c_ref = run_lbfgs(candidate, n, time.time() + alloc,
                                  alphas=[10000, 100000, 1000000],
                                  mu=0.01, maxiters=[300, 500, 500])
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = t0 + 40
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
                       alphas=[500, 5000, 50000, 500000],
                       mu=0.01, maxiters=[1000, 1500, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: DE global search at n=500
    p2_end = t0 + 220
    if time.time() < p2_end:
        h2, c2 = run_de(best_h, best_n, p2_end, pop_size=35, seed_val=seed)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Multi-start refinement at n=500
    p3_end = t0 + 340
    if time.time() < p3_end:
        h3, c3 = multi_start_refine(best_h, best_n, p3_end, rng, n_starts=12)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and refine
    n_mid = 1000
    p4_start = time.time()
    if p4_start < deadline - 350:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        p4_end = min(p4_start + 200, deadline - 300)
        h4, c4 = run_lbfgs(h_mid, n_mid, p4_end,
                           alphas=[10, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                           mu=0.005, maxiters=[400, 600, 800, 1000, 1500, 2000, 2500, 3000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_mid

    # Phase 5: Active set at current best resolution
    p5_end = min(time.time() + 120, deadline - 200)
    if time.time() < p5_end:
        h5, c5 = run_active_opt(best_h, best_n, p5_end, top_k=5,
                                alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: DE at n=1000 (shorter, for escaping local minima at higher res)
    if best_n >= 800 and time.time() < deadline - 180:
        p6_end = min(time.time() + 80, deadline - 150)
        h6, c6 = run_de(best_h, best_n, p6_end, pop_size=20, seed_val=seed + 1)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try n=2000
    n_high = 2000
    if time.time() < deadline - 120 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        p7_end = min(time.time() + 80, deadline - 60)
        h7, c7 = run_lbfgs(h_high, n_high, p7_end,
                           alphas=[5, 50, 200, 1000, 5000, 20000],
                           mu=0.002, maxiters=[300, 500, 700, 1000, 1200, 1500])
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_high

    # Phase 8: Targeted shift reduction
    p8_end = min(time.time() + 60, deadline - 30)
    if time.time() < p8_end:
        h8, c8 = targeted_shift_reduction(best_h, best_n, p8_end, alpha=50000000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Subgradient refinement
    p9_end = min(time.time() + 40, deadline - 10)
    if time.time() < p9_end:
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, max_iter=15000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final surgical active set push
    if time.time() < deadline - 3:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            for a in [100000000, 500000000]:
                if time.time() > deadline - 2:
                    break
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(best_n, active, a, 0.0005),
                               method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * best_n,
                               options={'maxiter': 300, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': 1000,
                                        'maxls': 40, 'maxcor': 30})
                h_final = project_feasible(res.x, best_n)
                c_final = compute_c5(h_final, best_n)
                if c_final < best_c:
                    best_c = c_final
                    best_h = h_final
        except Exception:
            pass

    # Final verification and cleanup
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
