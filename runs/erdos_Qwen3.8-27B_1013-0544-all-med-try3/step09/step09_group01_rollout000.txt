    m = c.max()

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_shift_optimize(h, n, deadline, top_k=6):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50000, 200000, 1000000, 5000000, 20000000]
    
    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-9, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit*5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def binary_local_search(h, n, deadline, max_iters=50000):
    """Local search on binary/near-binary landscape."""
    # Convert to binary
    h_bin = np.zeros(n)
    n_ones = int(n / 2)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    h_bin = project_feasible(h_bin, n)
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    
    # Tabu list
    tabu = set()
    tabu_tenure = 50
    
    it = 0
    no_improve = 0
    while it < max_iters and time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - 1e-10)[0]
        
        found_move = False
        # Try all single-bit moves
        for i in range(n):
            if (i) in tabu:
                continue
            if best_h[i] > 0.5:
                # Try to set to 0
                new_h = best_h.copy()
                new_h[i] = 0.0
                new_h = project_feasible(new_h, n)
                c = compute_c5(new_h, n)
                if c < best_c - 1e-13:
                    best_h = new_h
                    best_c = c
                    tabu.add(i)
                    found_move = True
                    no_improve = 0
                    it += 1
                    break
                # Check if this move is "promising" for the active set
            else:
                # Try to set to 1
                new_h = best_h.copy()
                new_h[i] = 1.0
                new_h = project_feasible(new_h, n)
                c = compute_c5(new_h, n)
                if c < best_c - 1e-13:
                    best_h = new_h
                    best_c = c
                    tabu.add(i)
                    found_move = True
                    no_improve = 0
                    it += 1
                    break
        
        if not found_move:
            no_improve += 1
            # Try pair swaps
            for i in range(0, n, 3):  # Sparse to save time
                if best_h[i] <= 0.5:
                    continue
                for j in range(0, n, 3):
                    if i == j or best_h[j] >= 0.5:
                        continue
                    new_h = best_h.copy()
                    new_h[i] = 0.0
                    new_h[j] = 1.0
                    c = compute_c5(new_h, n)
                    if c < best_c - 1e-13:
                        best_h = new_h
                        best_c = c
                        tabu.add(i)
                        tabu.add(j)
                        found_move = True
                        no_improve = 0
                        it += 1
                        break
                if time.time() > deadline:
                    break
                if found_move:
                    break
        
        if not found_move:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Manage tabu list
        if len(tabu) > n // 3:
            tabu = set(list(tabu)[:n // 6])
        it += 1
    
    return best_h, best_c

def subgrad_with_momentum(h, n, deadline, max_iters=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    momentum = 0.5
    v = np.zeros(n)
    tol = 0.002
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        active = np.where(c_vec >= c_max - tol * c_max)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        v = momentum * v - eta * grad
        h_new = best_h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.002, 0.03)
        else:
            v = 0.5 * v
            h_new = best_h - 0.5 * eta * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
            eta *= 0.995
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Sample pairs to save time
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    elif best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if
