import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_grad_at_k(h, n, k):
    """Gradient of c_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    absk = abs(k)
    if absk >= n:
        return grad
    nv = n - absk
    if k >= 0:
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        grad += w[idx] * c5_grad_at_k(x, n, idx - (n - 1))
    return grad

def active_obj(x, n, active_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def active_grad(x, n, active_idx, alpha):

    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):

            continue

        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n

    # Phase 1: Quick L-BFGS with increasing alpha at native resolution
    t_phase1 = t0 + 120
    if time.time() < t_phase1:
        for alpha in [100, 500, 2000, 10000, 50000]:
            if time.time() > t_phase1 - 10:
                break
            h1, c1 = run_lbfgs_alpha(best_h, best_n, t_phase1 - 5, alpha, 0.005, 1500)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    # Phase 2: Active-set focused optimization
    t_phase2 = t0 + 280
    if time.time() < t_phase2 - 20:
        h2, c2 = active_set_optimize(best_h, best_n, t_phase2 - 10,
                                     top_k=5, alphas=[5000, 20000, 100000], max_rounds=8)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient refinement
    t_phase3 = t0 + 400
    if time.time() < t_phase3 - 10:
        h3, c3 = subgradient_refine(best_h, best_n, t_phase3 - 5, n_iter=5000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Fine pair search
    t_phase4 = t0 + 550
    if time.time() < t_phase4 - 10:
        h4, c4 = fine_pair_search(best_h, best_n, t_phase4 - 5,
                                  radius=60,
                                  deltas=[0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try upsample to n=1000
    n_up = 1000
    t_phase5 = t0 + 750
    if time.time() < t_phase5 - 100:
        h_up, c_up = try_upsample(best_h, best_n, n_up, t_phase5)
        if time.time() < t_phase5 - 30:
            for alpha in [200, 1000, 5000, 20000]:
                if time.time() > t_phase5 - 20:
                    break
                h_up2, c_up2 = run_lbfgs_alpha(h_up, n_up, t_phase5 - 15, alpha, 0.003, 1000)
                if c_up2 < c_up:
                    h_up = h_up2
                    c_up = c_up2
            # Downsample back
            if c_up < best_c:
                x_new = np.linspace(0, 2, n_up)
                x_old = np.linspace(0, 2, best_n)
                h_ds = np.interp(x_old, x_new, h_up)
                h_ds = project_feasible(h_ds, best_n)
                c_ds = compute_c5(h_ds, best_n)
                if c_ds < best_c:
                    best_c = c_ds
                    best_h = h_ds

    # Phase 6: Multi-restart with perturbations
    t_phase6 = t0 + 900
    if time.time() < t_phase6 - 30:
        for s in range(4):
            if time.time() > t_phase6 - 25:
                break
            h_p = perturb_start(best_h, best_n, seed + 100 + s * 7, scale=0.005)
            c_p = compute_c5(h_p, best_n)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()
            # Quick refine
            t_sub = min(time.time() + 30, t_phase6 - 20)
            h_pr, c_pr = subgradient_refine(best_h, best_n, t_sub, n_iter=2000)
            if c_pr < best_c:
                best_c = c_pr
                best_h = h_pr

    # Phase 7: Final active-set + pair search
    t_phase7 = deadline
    if time.time() < t_phase7 - 30:
        h7, c7 = active_set_optimize(best_h, best_n, t_phase7 - 20,
                                     top_k=4, alphas=[50000, 200000, 1000000], max_rounds=4)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        if time.time() < t_phase7 - 8:
            h7b, c7b = fine_pair_search(best_h, best_n, t_phase7 - 3,
                                        radius=40,
                                        deltas=[0.001, 0.0003, 0.0001, 0.00003])
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
