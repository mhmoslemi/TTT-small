import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_max_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_max_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, maxiter_per):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]

            maxit = max(500, int(remaining * 4))
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

def steepest_max_descent(h, n, deadline, n_iter=2000):
    """Direct steepest descent on the max using active gradients."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        
        # Active set: shifts within 1% of max
        tol = max(1e-10, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx_step = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_step
                grad[k:n] += (-best_h[0:nv]) * dx_step
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_step
                grad[0:nv] += (-best_h[kk:n]) * dx_step
        grad /= len(active)
        
        # Line search
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.1]:
            h_trial = project_feasible(best_h - trial_eta * grad, n)

                best_c = c_trial
                best_h = h_trial
                eta = min(eta * 1.01, 0.01)
                break

def upsample(h, n_new):
    n_old = len(h)

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def project_feasible(h, n):

            tot = h.sum()
            if tot < 1e-14:

    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

        val = m + np.log(np.sum(np.exp(shifted))) / alpha

        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def active_max_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def active_max_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def eq_fun(x, n):
    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(len(x))


def run_lbfgs(h, n, deadline, alphas, mu, maxiter_per):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def run_active_refinement(h, n, deadline, max_rounds=25):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        sorted_idx = np.argsort(-c_vec)
        top_ks = [2, 4, 6, 10, 15]
        for k_top in top_ks:
            if time.time() > deadline - 8:
                break
            active = sorted_idx[:k_top]
            alphas = [10000, 50000, 200000, 1000000, 5000000]
            for alpha in alphas:
                if time.time() > deadline - 6:
                    break
                maxit = max(200, int((deadline - time.time()) * 2))
                try:
                    res = minimize(active_max_obj, best_h, jac=active_max_grad,
                                  args=(n, active, alpha, 0.002), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-22,
                                           'gtol': 1e-18, 'maxfun': maxit * 12,
                                           'maxls': 50, 'maxcor': 40})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    return best_h, best_c


def steepest_max_descent(h, n, deadline, n_iter=5000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.004
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx_step = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_step
                grad[k:n] += (-best_h[0:nv]) * dx_step
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_step
                grad[0:nv] += (-best_h[kk:n]) * dx_step
        grad /= len(active)
        found = False
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1]:
            h_trial = project_feasible(best_h - trial_eta * grad, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-16:
                best_c = c_trial
                best_h = h_trial
                eta = min(eta * 1.005, 0.015)
                found = True
                break
        if not found:
            eta *= 0.97
    return best_h, best_c
