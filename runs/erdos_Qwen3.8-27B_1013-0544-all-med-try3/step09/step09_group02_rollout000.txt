from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def all_shift_grads(h, n):
    """Compute gradients for all shifts. Returns (2n-1, n) matrix."""
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grads[idx, 0:nv] = (1.0 - h[k:n]) * dx
                grads[idx, k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grads[idx, kk:n] = (1.0 - h[0:nv]) * dx
                grads[idx, 0:nv] += (-h[kk:n]) * dx
    return grads

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only compute for shifts with significant weight
    thresh = 1e-8 / w.sum()
    active = np.where(w > thresh)[0]
    for idx in active:
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def sqp_step(h, n, active_idx, c_vec, grads_matrix, rho):
    """Solve SQP LP subproblem using precomputed gradients."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        A_ub[row, 0:n] = grads_matrix[idx]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, rho)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

        h_new, c_new = sqp_refined(h_new, n_new, t2, max_iter=150, rho_init=0.02, n_active=20)
    return h_new, c_new, n_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SQP warmup (80s)
    t1 = min(t0 + 80, deadline - 800)
    if t1 > time.time() + 5:
        best_h, best_c = sqp_refined(best_h, n, t1, max_iter=300, rho_init=0.04, n_active=30)

    # Phase 2: Fine SQP with small trust region (120s)
    t2 = min(t0 + 200, deadline - 650)
    if t2 > time.time() + 5:
        best_h, best_c = sqp_refined(best_h, n, t2, max_iter=400, rho_init=0.015, n_active=35)

    # Phase 3: Fine coordinate search (80s)
    t3 = min(t0 + 280, deadline - 550)
    if t3 > time.time() + 5:
        best_h, best_c = fine_coord_search(best_h, n, t3, n_top=6, n_steps_per_round=300)

    # Phase 4: Another tight SQP (80s)
    t4 = min(t0 + 360, deadline - 450)
    if t4 > time.time() + 5:
        best_h, best_c = sqp_refined(best_h, n, t4, max_iter=300, rho_init=0.01, n_active=40)

    # Phase 5: Perturbation restarts (120s)
    t5 = min(t0 + 480, deadline - 350)
    if t5 > time.time() + 10:
        best_h, best_c = perturbation_restart(best_h, n, t5, n_restarts=5,
                                              sigma_list=[0.02, 0.015, 0.01, 0.005, 0.003])

    # Phase 6: Fine coordinate search again (60s)
    t6 = min(t0 + 540, deadline - 280)
    if t6 > time.time() + 5:
        best_h, best_c = fine_coord_search(best_h, n, t6, n_top=5, n_steps_per_round=400)

    # Phase 7: Tight SQP (60s)
    t7 = min(t0 + 600, deadline - 220)
    if t7 > time.time() + 5:
        best_h, best_c = sqp_refined(best_h, n, t7, max_iter=250, rho_init=0.008, n_active=40)

    # Phase 8: Try upsample to n=1000 (180s)
    n_high = 1000
    t8 = min(t0 + 780, deadline - 100)
    if time.time() < t8 - 50:
        h_high, c_high, _ = upsample_and_optimize(best_h, n, n_high, t8)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = n_high

    # Phase 9: Try n=400 (sometimes coarser grids give better discrete values)
    t9 = min(t0 + 850, deadline - 60)
    if time.time() < t9 - 40:
        n_mid = 400
        x_old = np.linspace(0, 2, len(best_h))
        x_mid = np.linspace(0, 2, n_mid)
        h_mid = np.interp(x_mid, x_old, best_h)
        h_mid = project_feasible(h_mid, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        t_mid = min(time.time() + 50, deadline - 30)
        if t_mid > time.time() + 10:
            h_mid, c_mid = sqp_refined(h_mid, n_mid, t_mid, max_iter=200, rho_init=0.01, n_active=25)
        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
            n = n_mid

    # Phase 10: SLSQP final polish
    if time.time() < deadline - 15:
        h_final, c_final = slqp_polish(best_h, len(best_h), deadline - 3, alpha=2000000, maxiter=2000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final: one more ultra-fine SQP
    if time.time() < deadline - 3:
        n_fin = len(best_h)
        best_h, best_c = sqp_refined(best_h, n_fin, deadline - 1, max_iter=100, rho_init=0.005, n_active=30)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
