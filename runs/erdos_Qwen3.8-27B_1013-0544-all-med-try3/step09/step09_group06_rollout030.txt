import numpy as np
import time


def fft_convolve(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.empty(0, dtype=float)
    out_len = la + lb - 1
    nfft = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, nfft)
    fb = np.fft.rfft(b, nfft)
    out = np.fft.irfft(fa * fb, nfft)
    return out[:out_len]


def all_lags(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    f = 1.0 - h
    conv = fft_convolve(h, f[::-1])
    return conv[::-1] * dx


def fft_max_c(h):
    return float(np.max(all_lags(h)))


def smooth_max_weights(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    z = alpha * (c - m)
    zmax = float(np.max(z))
    if zmax > 50.0:
        e = np.exp(z - zmax)
        logsum = zmax + float(np.log(np.sum(e)))
    else:
        e = np.exp(z)
        logsum = float(np.log(np.sum(e)))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return m, w
    w = e / s
    return m + logsum / alpha, w


def top_k_weights(c, k):
    c = np.asarray(c, dtype=float)
    L = len(c)
    k = int(min(max(1, k), L))
    if k == L:
        return np.ones(L, dtype=float) / float(L)
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(L, dtype=float)
    w[idx] = 1.0 / float(k)
    return w


def weighted_gradient(h, w):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    f = 1.0 - h

    conv1 = fft_convolve(f[::-1], w)
    tf = conv1[n - 1:2 * n - 1][::-1]

    conv2 = fft_convolve(h[::-1], w[::-1])
    th = conv2[n - 1:2 * n - 1][::-1]

    return dx * (tf - th)


def enforce_sum_with_bounds(y, target):
    y = np.asarray(y, dtype=float).copy()
    diff = target - float(np.sum(y))
    if abs(diff) <= 1e-14:
        return y

    interior = (y > 1e-12) & (y < 1.0 - 1e-12)
    cnt = int(np.sum(interior))
    if cnt > 0:
        y[interior] += diff / float(cnt)
        y = np.clip(y, 0.0, 1.0)
        diff = target - float(np.sum(y))
        if abs(diff) > 1e-14:
            interior = (y > 1e-12) & (y < 1.0 - 1e-12)
            if np.sum(interior) > 0:
                j = int(np.where(interior)[0][0])
                y[j] += diff
                y = np.clip(y, 0.0, 1.0)
    else:
        if diff > 0:
            j = int(np.argmax(1.0 - y))
            y[j] = min(1.0, y[j] + diff)
        else:
            j = int(np.argmin(y))
            y[j] = max(0.0, y[j] + diff)
    return y


def project_box_sum_fast(x, n):
    target = 0.5 * n
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    s = float(np.sum(y))
    diff = target - s
    for _ in range(25):
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - y
            tot = float(np.sum(room))
            if tot < 1e-15:
                break
            y += (diff / tot) * room
        else:
            tot = float(np.sum(y))
            if tot < 1e-15:
                break
            y += (diff / tot) * y
        y = np.clip(y, 0.0, 1.0)
        diff = target - float(np.sum(y))
    return enforce_sum_with_bounds(y, target)


def project_box_sum_exact(x, n):
    target = 0.5 * n
    x = np.asarray(x, dtype=float)

    y0 = np.clip(x, 0.0, 1.0)
    s0 = float(np.sum(y0))
    if abs(s0 - target) <= 1e-12:
        return enforce_sum_with_bounds(y0, target)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - hi, 0.0, 1.0)
    return enforce_sum_with_bounds(y, target)


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    x_old = (np.arange(n) + 0.5) * dx
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def make_simple_start(n, code):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if code == 0:
        h = 1.0 - np.abs(x - 1.0)
    elif code == 1:
        h = np.abs(x - 1.0)
    elif code == 2:
        h = 0.5 + 0.5 * np.cos(np.pi * x)
    elif code == 3:
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
    elif code == 4:
        h = 0.5 + 0.5 * np.sin(2.0 * np.pi * x)
    else:
        h = 0.5 + 0.35 * np.cos(2.0 * np.pi * x) * np.sin(3.0 * np.pi * x)
    return project_box_sum_fast(h, n)


def perturbed_h(base, n, rng, scale):
    base = np.asarray(base, dtype=float)
    noise = rng.normal(0.0, scale, n)
    return project_box_sum_fast(base + noise, n)


def adam_phase(start_h, n, mode, param, deadline, rng, best_h, best_c, base_lr=0.04):
    h = project_box_sum_fast(start_h, n).copy()
    m1 = np.zeros(n, dtype=float)
    m2 = np.zeros(n, dtype=float)
    b1t = 1.0
    b2t = 1.0
    t = 0
    lr = float(base_lr)
    no_best = 0
    accept_fail = 0

    c = all_lags(h)
    cval = float(np.max(c))
    if cval < best_c - 1e-12:
        best_c = cval
        best_h = h.copy()

    while time.time() < deadline:
        if cval < best_c - 1e-12:
            best_c = cval
            best_h = h.copy()
            no_best = 0
        else:
            no_best += 1

        if no_best > 1400:
            h = perturbed_h(best_h, n, rng, 0.010 + 0.002 * (no_best % 6))
            m1.fill(0.0)
            m2.fill(0.0)
            b1t = 1.0
            b2t = 1.0
            t = 0
            lr = max(0.012, 0.65 * base_lr)
            no_best = 0
            accept_fail = 0
            c = all_lags(h)
            cval = float(np.max(c))
            continue

        if mode == "topk":
            w = top_k_weights(c, int(param))
        else:
            _val, w = smooth_max_weights(c, float(param))

        g = weighted_gradient(h, w)
        gnorm = float(np.max(np.abs(g)))
        if gnorm < 1e-11:
            break
        g = g / gnorm

        t += 1
        b1t *= 0.9
        b2t *= 0.999
        m1 = 0.9 * m1 + 0.1 * g
        m2 = 0.999 * m2 + 0.001 * (g * g)

        mh = m1 / (1.0 - b1t)
        vh = m2 / (1.0 - b2t)
        step = -lr * mh / (np.sqrt(vh) + 1e-8)
        step -= float(np.mean(step))
        maxstep = float(np.max(np.abs(step)))
        if maxstep > 0.06:
            step *= 0.06 / maxstep

        new_h = project_box_sum_fast(h + step, n)
        new_c_arr = all_lags(new_h)
        new_c = float(np.max(new_c_arr))

        if new_c <= cval + 1e-10:
            h = new_h
            c = new_c_arr
            cval = new_c
            accept_fail = 0
            if new_c < cval - 1e-12:
                lr = min(0.075, lr * 1.02)
            else:
                lr = max(0.5 * base_lr, lr * 0.985)
        else:
            accept_fail += 1
            half_step = 0.5 * step
            half_h = project_box_sum_fast(h + half_step, n)
            half_arr = all_lags(half_h)
            half_c = float(np.max(half_arr))
            if half_c <= cval + 1e-10:
                h = half_h
                c = half_arr
                cval = half_c
                accept_fail = max(0, accept_fail - 2)
                lr *= 0.9
            else:
                lr = max(0.008, 0.6 * lr)
                step *= 0.25
            if accept_fail > 45:
                lr *= 0.6
                accept_fail = 0

        if t % 1200 == 0:
            h = project_box_sum_exact(h, n)
            c = all_lags(h)
            cval = float(np.max(c))

    if cval < best_c - 1e-12:
        best_c = cval
        best_h = h.copy()

    best_h = project_box_sum_exact(best_h, n)
    best_c = fft_max_c(best_h)
    return best_h, best_c


def local_pair_search(h_start, n, deadline, rng):
    h = project_box_sum_exact(h_start, n).copy()
    best_h = h.copy()
    best_c = fft_max_c(h)
    no_round = 0

    while time.time() < deadline and no_round < 7:
        if time.time() > deadline - 2:
            break

        c = all_lags(h)
        cval = float(np.max(c))
        if cval < best_c - 1e-12:
            best_c = cval
            best_h = h.copy()

        top = min(12, len(c))
        top_idx = np.argpartition(c, -top)[-top:]
        order = np.argsort(c[top_idx])[::-1]
        if len(order) > 1:
            order = order[rng.permutation(len(order))]

        improved = False
        for pos in order:
            if improved:
                break

            idx = int(top_idx[pos])
            d = idx - (n - 1)
            if d == 0:
                continue
            m = abs(d)

            if d > 0:
                a = np.arange(0, n - m, dtype=int)
                b = a + m
            else:
                a = np.arange(m, n, dtype=int)
                b = a - m

            diffs = h[a] - h[b]
            cap = np.minimum(h[a], 1.0 - h[b])
            mask = (diffs > 1e-5) & (cap > 1e-5)
            if not np.any(mask):
                continue

            valid = np.nonzero(mask)[0]
            vals = diffs[valid] * cap[valid]
            if vals.size > 70:
                part = np.argpartition(vals, -70)[-70:]
                part = part[np.argsort(vals[part])[::-1]]
            else:
                part = np.arange(vals.size)

            for pi in part:
                if improved:
                    break
                vi = int(valid[pi])
                ai = int(a[vi])
                bi = int(b[vi])
                capv = float(cap[vi])

                for frac in (0.05, 0.15, 0.35, 0.70):
                    delta = frac * capv
                    if delta < 1e-8:
                        continue
                    old_a = h[ai]
                    old_b = h[bi]
                    h[ai] = old_a - delta
                    h[bi] = old_b + delta

                    nc = fft_max_c(h)
                    if nc < best_c - 1e-12:
                        best_c = nc
                        best_h = h.copy()
                        improved = True
                        no_round = 0
                        break
                    else:
                        h[ai] = old_a
                        h[bi] = old_b

        if not improved:
            no_round += 1
            scale = 0.004 + 0.002 * no_round
            h = perturbed_h(h, n, rng, scale)
            if time.time() > deadline - 8:
                break

    best_h = project_box_sum_exact(best_h, n)
    best_c = fft_max_c(best_h)
    return best_h, best_c


def optimize_h(h_start, n, deadline, rng):
    h = project_box_sum_exact(h_start, n)
    best_c = fft_max_c(h)
    best_h = h.copy()

    now = time.time()
    if deadline - now < 4:
        return best_h, best_c

    total = max(2.0, deadline - now)
    specs = [
        ("topk", 4, 0.10),
        ("smooth", 500, 0.10),
        ("topk", 8, 0.10),
        ("smooth", 2000, 0.10),
        ("topk", 5, 0.10),
        ("smooth", 8000, 0.12),
        ("smooth", 30000, 0.12),
        ("topk", 3, 0.12),
        ("smooth", 120000, 0.14),
    ]

    base_lr = 0.05 if n <= 700 else 0.035
    start = now
    for mode, param, frac in specs:
        phase_deadline = start + float(frac) * total
        if phase_deadline > deadline:
            phase_deadline = deadline
        if phase_deadline - time.time() > 1.5:
            best_h, best_c = adam_phase(
                best_h, n, mode, param, phase_deadline, rng, best_h, best_c,

            )
            if time.time() > deadline - 1:
                break

    best_h = project_box_sum_exact(best_h, n)
    best_c = fft_max_c(best_h)
    return best_h, best_c

        parent_alloc = min(360.0, 0.38 * remaining)
    else:
        parent_alloc = min(560.0, 0.58 * remaining)
    if parent_alloc > 12.0:
        best_h, best_c = optimize_h(best_h, n_best, time.time() + parent_alloc, rng)

    for n_try in candidate_list(n_best):
        remaining = deadline - time.time()
        if remaining < 210.0:
            break
        alloc = min(230.0, remaining - 110.0)
        if alloc < 70.0:
            break

        h0 = resample_h(best_h, n_try)
        h0 = project_box_sum_fast(h0, n_try)
        h1, c1 = optimize_h(h0, n_try, time.time() + alloc, rng)
        if c1 < best_c - 1e-10:
            best_h = h1
            best_c = c1
            n_best = n_try
            break

    remaining = deadline - time.time()
    if remaining > 110.0:
        hr = perturbed_h(best_h, n_best, rng, 0.020)
        alloc = min(130.0, remaining - 70.0)
        if alloc > 50.0:
            h1, c1 = optimize_h(hr, n_best, time.time() + alloc, rng)
            if c1 < best_c - 1e-10:
                best_h = h1
                best_c = c1

    remaining = deadline - time.time()
    if remaining > 25.0:
        local_deadline = min(deadline - 3.0, time.time() + max(20.0, min(90.0, remaining * 0.6)))
        best_h, _ = local_pair_search(best_h, n_best, local_deadline, rng)

    remaining = deadline - time.time()
    if remaining > 25.0:
        local_deadline = min(deadline - 1.0, time.time() + min(60.0, remaining * 0.7))
        best_h, _ = local_pair_search(best_h, n_best, local_deadline, rng)

    best_h = project_box_sum_exact(best_h, n_best)
    c5 = final_c5(best_h, n_best)
    return best_h, c5, int(n_best)
