    viol = np.maximum(c - C, 0)
    active = viol > 0
    # d(d f_k)/d h for active k
    grad = np.zeros(n)
    for k in range(len(c)):
        if not active[k]:
            continue
        weight = 2 * rho * viol[k]
        # gradient of f_k w.r.t. h
        k_idx = k - (n-1)
        if k_idx >= 0:

            grad[0:nv] += weight * (1 - h[k_idx:n])
            grad[k_idx:n] += weight * (-h[0:nv])

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
