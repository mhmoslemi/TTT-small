import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

_RUN_DEADLINE = 0.0


def minimize_callback(x, f, g, k):
    if time.time() > _RUN_DEADLINE:

def sample_coords(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    return (np.arange(n) + 0.5) * (2.0 / n)


def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def c5_val(h, n):
    return float(np.max(c5_vec(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = h.reshape(-1)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(200):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h
            room[~mask] = 0.0
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            ratio = diff / tot
            if ratio > 1.0:
                ratio = 1.0
            h = np.clip(h + ratio * room, 0.0, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            val = h.copy()
            val[~mask] = 0.0
            tot = float(val.sum())
            if tot <= 1e-15:
                break
            ratio = -diff / tot
            if ratio > 1.0:
                ratio = 1.0
            h = np.clip(h - ratio * val, 0.0, 1.0)

    for _ in range(100):
        s = float(h.sum())
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            if add <= 1e-18:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            if sub <= 1e-18:
                break
            h[idx] -= sub

    return h


def resample(h, n):
    n = int(n)
    old = int(len(h))
    if old == n:
        return h.copy()
    if old == 0:
        return np.zeros(n, dtype=float)
    return np.interp(sample_coords(n), sample_coords(old), h)


def resample_nearest(h, n):
    n = int(n)
    old = int(len(h))
    if old == n:
        return h.copy()
    if old == 0:
        return np.zeros(n, dtype=float)
    idx = np.round((np.arange(n) + 0.5) * (float(old) / float(n)) - 0.5).astype(int)
    idx = np.clip(idx, 0, old - 1)
    return h[idx].astype(float)


def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    k = int(shift_idx) - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad


def select_active(c_vec, K):
    maxval = float(np.max(c_vec))
    tol = max(1e-11, 1e-5 * maxval)
    idx = np.where(c_vec >= maxval - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c_vec))])
    if idx.size > K:
        order = np.argsort(-c_vec[idx])
        idx = idx[order[:K]]
    return idx, maxval


def trust_lp(h, n, active, c_vec, rho, maxval):
    nvar = n + 1
    c_obj = np.zeros(nvar, dtype=float)
    c_obj[n] = 1.0

    A_ub = np.zeros((len(active), nvar), dtype=float)
    b_ub = np.empty(len(active), dtype=float)
    for r, idx in enumerate(active):
        idx = int(idx)
        A_ub[r, 0:n] = shift_grad_single(h, n, idx)
        A_ub[r, n] = -1.0
        b_ub[r] = -float(c_vec[idx])

    A_eq = np.zeros((1, nvar), dtype=float)
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1, dtype=float)

    t_lo = max(0.0, maxval - 1.0)
    t_hi = 10.0
    bounds = [(-rho, rho)] * n
    bounds.append((t_lo, t_hi))

    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
        if res.success:
            return np.asarray(res.x[0:n], dtype=float)
    except Exception:
        pass
    return None


def fallback_direction(h, n, active, rho):
    g = np.zeros(n, dtype=float)
    for idx in active:
        g += shift_grad_single(h, n, int(idx))
    g /= max(1, len(active))
    g -= g.mean()
    d = -g
    dm = float(np.max(np.abs(d)))
    if dm < 1e-15:
        return None
    return rho * (d / dm)


def line_search_step(h, n, d, ref_c):
    lams = (1.0, 0.6, 0.3, 0.15, 0.075, 0.0375, 0.01875, 0.009375)
    best_h = None
    best_c = ref_c
    for lam in lams:
        h_try = project_feasible(h + lam * d, n)
        c_try = c5_val(h_try, n)
        if c_try < best_c - 1e-16:
            best_c = c_try
            best_h = h_try
    return best_h, best_c


def trust_optimize(h, n, deadline, max_iter=500, rho_init=0.02, topK=10, max_stall=12):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_val(h, n)
    cur_h = h.copy()
    rho = max(float(rho_init), 1e-6)
    stall = 0
    it = 0

    while it < max_iter and time.time() < deadline:
        c_vec = c5_vec(cur_h, n)
        active, maxval = select_active(c_vec, topK)
        if active.size == 0:
            break
        if maxval < best_c:
            best_c = maxval
            best_h = cur_h.copy()

        delta = trust_lp(cur_h, n, active, c_vec, rho, maxval)
        if delta is None or not np.all(np.isfinite(delta)):
            delta = fallback_direction(cur_h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            it += 1
            continue

        dmax = float(np.max(np.abs(delta)))
        if dmax < 1e-13:
            break

        new_h, new_c = line_search_step(cur_h, n, delta, best_c)
        if new_h is not None:
            best_h = new_h.copy()
            best_c = new_c
            cur_h = new_h.copy()
            stall = 0
            if rho < rho_init:
                rho = min(rho * 1.25, rho_init)
        else:
            stall += 1
            rho *= 0.5
            if rho < 1e-9 or stall > max_stall:
                break
        it += 1

    return best_h, best_c


def smooth_obj(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    s = float(np.sum(np.exp(sh)))
    val = m + math.log(max(s, 1e-300)) / alpha
    if mu:
        diff = float(np.sum(x)) - 0.5 * float(n)
        val += 0.5 * mu * diff * diff
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = c5_vec(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    w = np.exp(sh)
    ws = float(w.sum())
    if ws <= 0.0:
        w[:] = 0.0
        w[int(np.argmax(c))] = 1.0
        ws = 1.0
    w /= ws

    grad = np.zeros(n, dtype=float)
    idx = np.where(w > 1e-12)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size > 40:
        idx = idx[np.argsort(-w[idx])[:40]]
    wt = w[idx]
    wt_sum = float(wt.sum())
    if wt_sum > 0.0:
        wt /= wt_sum
    for j in range(idx.size):
        grad += float(wt[j]) * shift_grad_single(x, n, int(idx[j]))

    if mu:
        diff = float(np.sum(x)) - 0.5 * float(n)
        grad += mu * diff
    return grad


def lbfgs_local(h, n, deadline, alphas=None, mu=5.0, maxiter=None):
    global _RUN_DEADLINE
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = (200.0, 800.0, 3000.0)
    if maxiter is None:
        maxiter = int(max(60, min(320, 110000.0 / max(int(n), 1))))

    for alpha in alphas:
        if time.time() > deadline - 2.0:
            break
        _RUN_DEADLINE = max(_RUN_DEADLINE, deadline - 1.0)
        options = {
            "maxiter": maxiter,
            "ftol": 1e-18,
            "gtol": 1e-13,
            "maxfun": max(200, 5 * maxiter),
            "maxls": 20,
        }
        try:
            res = minimize(
                smooth_obj,
                h,
                jac=smooth_grad,
                args=(n, float(alpha), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options=options,
                callback=minimize_callback,
            )
            h_new = project_feasible(res.x, n)
            c_new = c5_val(h_new, n)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c


def add_zero_sum_noise(h, n, rng, amp):
    v = rng.normal(0.0, amp, n)
    v -= v.mean()
    return project_feasible(h + v, n)


def gen_random_feasible(n, rng):
    return project_feasible(rng.random(n), n)


def gen_binary_feasible(n, rng):
    x = np.zeros(n, dtype=float)
    m = n // 2
    if m > 0:
        idx = rng.choice(n, size=m, replace=False)
        x[idx] = 1.0
    if n % 2 == 1:
        x[0] = 0.5
    return project_feasible(x, n)


def gen_blocks(n, rng):
    x = np.empty(n, dtype=float)
    pos = 0
    val = int(rng.integers(0, 2))
    while pos < n:
        low = 2
        high = min(8, n - pos) + 1
        if high <= low:
            high = low + 1
        ln = int(rng.integers(low, high))
        ln = min(ln, n - pos)
        x[pos:pos + ln] = val
        pos += ln
        val = 1 - val
    return project_feasible(x, n)


def gen_period(n, period, duty, phase):
    if period <= 0.0:
        period = 0.5
    x = sample_coords(n)
    t = ((x - phase) / period) % 1.0
    h = (t < duty).astype(float)
    return project_feasible(h, n)


def gen_sine(n, freq, amp, phase):
    x = sample_coords(n)
    h = 0.5 + amp * np.sin(2.0 * math.pi * freq * x + phase)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def gen_thue(n):
    x = np.zeros(n, dtype=float)
    for i in range(n):
        if (i.bit_count() & 1):
            x[i] = 1.0
    return project_feasible(x, n)


def gen_lowdisp(n, alpha):
    x = np.arange(n, dtype=float) * float(alpha)
    frac = x - np.floor(x)
    h = (frac < 0.5).astype(float)
    return project_feasible(h, n)


def make_fine_from_coarse(h_c, n_target):
    h_i = project_feasible(resample(h_c, n_target), n_target)
    c_i = c5_val(h_i, n_target)

    h_n = project_feasible(resample_nearest(h_c, n_target), n_target)
    c_n = c5_val(h_n, n_target)

    h_t = project_feasible((h_i > 0.5).astype(float), n_target)
    c_t = c5_val(h_t, n_target)

    h_nt = project_feasible((h_n > 0.5).astype(float), n_target)
    c_nt = c5_val(h_nt, n_target)

    best_c = c_i
    best_h = h_i
    if c_n < best_c:
        best_c = c_n
        best_h = h_n
    if c_t < best_c:
        best_c = c_t
        best_h = h_t
    if c_nt < best_c:
        best_c = c_nt
        best_h = h_nt
    return best_h


def refine_fine(h, n, deadline, max_trust_iter=400, use_lbfgs=True):

def coarse_search(h_base, n_base, deadline, rng):
    top = []
    counter = 0

    if n_base >= 350:
        coarse_ns = (64, 128, 256)
    elif n_base >= 180:
        coarse_ns = (64, 128)
    else:
        coarse_ns = (32, 64)

    for nc in coarse_ns:
        if nc >= n_base:
            continue

        parent_c = project_feasible(resample(h_base, nc), nc)
        cands = []

        cands.append(parent_c)
        cands.append(project_feasible((parent_c > 0.5).astype(float), nc))
        cands.append(gen_thue(nc))
        cands.append(gen_lowdisp(nc, math.sqrt(2.0) - 1.0))
        cands.append(gen_lowdisp(nc, 1.0 / (1.0 + math.sqrt(5.0))))

        for p in (0.25, 0.3, 0.333, 0.4, 0.5, 0.6, 0.75):
            cands.append(gen_period(nc, p, 0.5, 0.0))
            cands.append(gen_period(nc, p, 0.5, 0.5 * p))

        for _ in range(6):
            cands.append(gen_random_feasible(nc, rng))
        for _ in range(6):
            cands.append(gen_binary_feasible(nc, rng))
        for _ in range(6):
            cands.append(gen_blocks(nc, rng))

        for f in (2.0, 3.0, 4.0, 5.0):
            cands.append(gen_sine(nc, f, 0.95, 0.0))
            cands.append(gen_sine(nc, f, 0.95, 0.5 * math.pi))

        per = 0.55 if nc <= 64 else (1.1 if nc <= 128 else 1.9)
        remaining = max(0.0, deadline - time.time())
        if remaining < 25.0:
            break
        max_cands = max(6, int(remaining / max(0.25, per)))
        if len(cands) > max_cands:
            step = float(len(cands)) / float(max_cands)
            cands = [cands[int(i * step)] for i in range(max_cands)]

        for h_cand in cands:
            if time.time() > deadline - 60.0:
                break
            h_cand = project_feasible(h_cand, nc)
            cand_dead = min(deadline - 60.0, time.time() + per)
            h_loc, _ = trust_optimize(
                h_cand,
                nc,
                cand_dead,
                max_iter=350,
                rho_init=0.06,
                topK=6,
                max_stall=8,
            )

            hf_i = project_feasible(resample(h_loc, n_base), n_base)
            cf_i = c5_val(hf_i, n_base)

            hf_n = project_feasible(resample_nearest(h_loc, n_base), n_base)
            cf_n = c5_val(hf_n, n_base)

            hf_t = project_feasible((hf_i > 0.5).astype(float), n_base)
            cf_t = c5_val(hf_t, n_base)

            score = min(cf_i, cf_n, cf_t)
            top.append((score, nc, counter, h_loc.copy(), cf_i, cf_n, cf_t))
            counter += 1
            top.sort()
            if len(top) > 12:
                top = top[:12]

    return top

                mu=5.0,
                maxiter=90,
            )
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
                best_n = len(best_h)

    # Multi-resolution global exploration.
    coarse_end = min(
        time.time() + max(120.0, 0.23 * budget),
        deadline - max(320.0, 0.35 * budget),
    )
    coarse_top = []
    if coarse_end > time.time() + 30.0:
        coarse_top = coarse_search(best_h, n_base, coarse_end, rng)

    # Refine the most promising coarse-to-fine candidates.
    refine_end = min(
        time.time() + max(180.0, 0.29 * budget),
        deadline - max(260.0, 0.28 * budget),
    )
    if refine_end > time.time() + 60.0 and coarse_top:
        top_count = 5
        for item in coarse_top[:top_count]:
            if time.time() > refine_end - 50.0:
                break
            score, nc, counter, h_c, _, _, _ = item
            if score > best_c + 0.02:
                continue
            h_f = make_fine_from_coarse(h_c, n_base)
            rem = refine_end - time.time()
            left = max(1, top_count - (len(coarse_top[:top_count]) - 0))
            per_cand = max(25.0, rem / float(left))
            cand_dead = min(refine_end - 45.0, time.time() + per_cand)
            h2, c2 = refine_fine(h_f, n_base, cand_dead, max_trust_iter=350, use_lbfgs=True)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = len(best_h)

    # Higher-resolution refinement if useful.
    high_end = min(
        time.time() + max(80.0, 0.12 * budget),
        deadline - max(140.0, 0.14 * budget),
    )
    n_high = int(n_base) + 300
    if n_high < 800:
        n_high = 800
    if n_high > 1000:
        n_high = 1000
    if n_high % 2 != 0:
        n_high += 1
    if n_high <= best_n:
        n_high = best_n

    if n_high > best_n and high_end > time.time() + 45.0:
        h_hi = make_fine_from_coarse(best_h, n_high)
        h_hi, c_hi = refine_fine(h_hi, n_high, high_end, max_trust_iter=320, use_lbfgs=(n_high <= 820))
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            best_n = len(best_h)

    # Final polishing.
    final_end = deadline - 5.0

    if time.time() < final_end - 80.0:
        h_snap = project_feasible((best_h > 0.5).astype(float), best_n)
        c_snap = c5_val(h_snap, best_n)
        if c_snap < best_c + 0.004:
            t_snap = min(final_end - 55.0, time.time() + 40.0)
            h_snap, c_snap = trust_optimize(
                h_snap,
                best_n,
                t_snap,
                max_iter=350,
                rho_init=0.015,
                topK=12 if best_n <= 700 else 8,
                max_stall=12,
            )
            if c_snap < best_c:
                best_c = c_snap
                best_h = h_snap.copy()
                best_n = len(best_h)

    if time.time() < final_end - 35.0:
        tfin = min(final_end - 25.0, time.time() + max(35.0, 0.14 * (final_end - time.time())))
        best_h, best_c = trust_optimize(
            best_h,
            best_n,
            tfin,
            max_iter=700,
            rho_init=0.012,
            topK=14 if best_n <= 700 else 8,
            max_stall=16,
        )

    if best_n <= 820 and time.time() < final_end - 12.0:
        tfin = min(final_end - 8.0, time.time() + max(12.0, 0.10 * (final_end - time.time())))
        best_h, best_c = lbfgs_local(
            best_h,
            best_n,
            tfin,
            alphas=(3000.0, 10000.0, 30000.0),
            mu=5.0,
            maxiter=100,
        )

    if time.time() < final_end - 3.0:
        best_h, best_c = trust_optimize(
            best_h,
            best_n,
            max(0.0, final_end - 3.0),
            max_iter=600,
            rho_init=0.008,
            topK=12 if best_n <= 700 else 8,
            max_stall=18,
        )

    best_h = project_feasible(best_h, best_n)
    best_c = c5_val(best_h, best_n)
    return best_h, float(best_c), int(best_n)
