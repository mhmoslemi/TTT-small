```python
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_W_idx(n):
    """Build index matrix for fast gradient. W_idx[j,i] = i-j+n-1"""
    j = np.arange(n)
    i = np.arange(n)
    W_idx = (i[None, :] - j[:, None] + n - 1).astype(np.int64)
    return W_idx

def make_obj_and_grad(n, alpha, mu, W_idx):
    """Create objective and gradient functions."""
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        exp_c = np.exp(alpha * (c - m))
        val = m + np.log(exp_c.sum()) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def grad(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        exp_c = np.exp(alpha * (c - m))
        W = exp_c / exp_c.sum()
        dx_val = 2.0 / n
        W_x = W[W_idx]

        S2 = W_x.T @ x
        g = dx_val * (N1 - S1 - S2)
        g += 2.0 * mu * (x.sum() - n / 2.0)

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

    best_c = compute_c5(h, n)
    dx_val = 2.0 / n

        max_val = c_vec.max()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_W_idx(n):
    j = np.arange(n)
    i = np.arange(n)
    W_idx = (i[None, :] - j[:, None] + n - 1).astype(np.int64)
    return W_idx

def soft_obj(x, n, alpha, mu, W_idx):
    c = compute_c5_vec(x, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    val = m + np.log(exp_c.sum()) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_grad(x, n, alpha, mu, W_idx):
    c = compute_c5_vec(x, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    W = exp_c / exp_c.sum()
    dx_val = 2.0 / n
    W_x = W[W_idx]
    N1 = W_x.sum(axis=1)
    S1 = W_x @ x
    S2 = W_x.T @ x
    g = dx_val * (N1 - S1 - S2)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_round(h, n, W_idx, alpha, mu, deadline, max_iter=20000):
    bounds = [(0.0, 1.0)] * n
    args = (n, alpha, mu, W_idx)
    best_c = compute_c5(h, n)
    try:
        res = minimize(soft_obj, h, args=args, jac=soft_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxfun': 500000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def active_refine(h, n, deadline, top_k=10, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = np.where(c_vec >= thresh)[0]
        
        if len(active) > top_k:
            top_idx = np.argsort(-c_vec[active])[:top_k]
            active = active[top_idx]
        
        grad = np.zeros(n)
        for mm in active:
            k = int(mm) - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx_val * (1.0 - h[k:n])
                grad[k:n] += -dx_val * h[0:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx_val * (1.0 - h[0:nv])
                grad[0:nv] += -dx_val * h[kk:n]
        grad /= len(active)
        
        improved = False
        for lr in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_trial = h - lr * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    W_idx = build_W_idx(n)
    
    # Phase 1: L-BFGS warmup with moderate alpha
    t1 = min(t0 + 150, deadline - 400)
    for alpha, mu in [(200, 50.0), (1000, 100.0), (5000, 100.0)]:
        if time.time() > t1:
            break
        h, best_c = lbfgs_round(h, n, W_idx, alpha, mu, t1, max_iter=5000)
    
    # Phase 2: Active set refinement
    t2 = min(t0 + 300, deadline - 300)
    if time.time() < t2:
        h, best_c = active_refine(h, n, t2, top_k=15, max_iter=5000)
    
    # Phase 3: L-BFGS with higher alpha
    t3 = min(t0 + 500, deadline - 200)
    for alpha, mu in [(10000, 200.0), (50000, 500.0), (100000, 500.0)]:
        if time.time() > t3:
            break
        h, best_c = lbfgs_round(h, n, W_idx, alpha, mu, t3, max_iter=10000)
    
    # Phase 4: Active set refinement
    t4 = min(t0 + 650, deadline - 100)
    if time.time() < t4:
        h, best_c = active_refine(h, n, t4, top_k=8, max_iter=10000)
    
    # Phase 5: Final L-BFGS
    t5 = min(t0 + 800, deadline - 50)
    if time.time() < t5:
        for alpha, mu in [(200000, 500.0), (500000, 1000.0)]:
            if time.time() > t5:
                break
            h, best_c = lbfgs_round(h, n, W_idx, alpha, mu, t5, max_iter=5000)
    
    # Phase 6: Final active set
    if time.time() < deadline:
        h, best_c = active_refine(h, n, deadline, top_k=5, max_iter=5000)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    return h, best_c, n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_W_idx(n):
    j = np.arange(n)
    i = np.arange(n)
    W_idx = (i[None, :] - j[:, None] + n - 1).astype(np.int64)
    return W_idx

def soft_obj(x, n, alpha, mu, W_idx):
    c = compute_c5_vec(x, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    val = m + np.log(exp_c.sum()) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_grad(x, n, alpha, mu, W_idx):
    c = compute_c5_vec(x, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    W = exp_c / exp_c.sum()
    dx_val = 2.0 / n
    W_x = W[W_idx]
    N1 = W_x.sum(axis=1)
    S1 = W_x @ x
    S2 = W_x.T @ x
    g = dx_val * (N1 - S1 - S2)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_round(h, n, W_idx, alpha, mu, deadline, max_iter=10000):
    bounds = [(0.0, 1.0)] * n
    args = (n, alpha, mu, W_idx)
    best_c = compute_c5(h, n)
    try:
        res = minimize(soft_obj, h, args=args, jac=soft_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxfun': 500000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def shift_grad_single(h, n, shift_idx, dx_val):
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = dx_val * (1.0 - h[k:n])
        if nv > 0:
            grad[k:n] += -dx_val * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = dx_val * (1.0 - h[0:nv])
        if nv > 0:
            grad[0:nv] += -dx_val * h[kk:n]
    return grad

def active_refine(h, n, deadline, top_k=10, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = np.where(c_vec >= thresh)[0]

        if len(active) > top_k:
            top_idx = np.argsort(-c_vec[active])[:top_k]
            active = active[top_idx]

        if len(active) == 0:
            break

        grad = np.zeros(n)
        for ii in range(len(active)):
            grad += shift_grad_single(h, n, int(active[ii]), dx_val)
        grad /= len(active)

        improved = False
        for lr in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_trial = h - lr * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial.copy()
                h = h_trial
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break

    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30):
    """Efficient local search: try moving mass between pairs of nearby indices."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001]

    for step in steps:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]

            # Compute gradient magnitude to find where to focus
            dx_val = 2.0 / n
            grad_mag = np.zeros(n)
            for ii in range(min(len(active), 20)):
                grad_mag += np.abs(shift_grad_single(h, n, int(active[ii]), dx_val))

            top_pos = np.argsort(-grad_mag)[:min(n, radius * 4)]

            for i_val in top_pos:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try: increase i, decrease j
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try: decrease i, increase j
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] += step
                            h[j] -= step
                    h = best_h  # keep searching from best

    return best_h, best_c
```
