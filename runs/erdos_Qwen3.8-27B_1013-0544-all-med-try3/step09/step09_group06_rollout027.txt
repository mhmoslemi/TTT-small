def top_gradient_counts(h, n, idxs):

   for idx in idxs:
      m = int(idx) - (n-1)
      if m == 0:
         g += 1.0 - 2.0*h

         g[m:n] += 1.0 - h[0:n-m]

import numpy as np
import time


def project_h(y, n):
    y = np.asarray(y, dtype=float)
    target = 0.5 * float(n)

    x = np.clip(y, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - target) < 1e-10:
        return x

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid

    t = 0.5 * (lo + hi)
    x = np.clip(y + t, 0.0, 1.0)

    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            add = min(diff, max(0.0, 1.0 - x[idx]))
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            sub = min(-diff, x[idx])
            x[idx] -= sub
    return x


def correlate_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def top_lag_indices(c, k):
    k = int(k)
    m = len(c)
    if k <= 0:
        return np.array([], dtype=int)
    if k >= m:
        return np.arange(m)
    if k == 1:
        return np.array([int(np.argmax(c))], dtype=int)
    return np.argpartition(c, -k)[-k:]


def top_gradient_counts(h, n, idxs):
    g = np.zeros(n, dtype=float)
    n1 = n - 1
    cnt = 0
    for idx in idxs:
        m = int(idx) - n1
        if m == 0:
            g += 1.0 - 2.0 * h
        elif m > 0:
            g[m:n] += 1.0 - h[0:n - m]
            g[0:n - m] -= h[m:n]
        else:
            am = -m
            g[0:n - am] += 1.0 - h[am:n]
            g[am:n] -= h[0:n - am]
        cnt += 1
    if cnt > 1:
        g /= float(cnt)
    return g


def apply_single_move(h, c, p, delta):
    if abs(delta) < 1e-16:
        return
    n = len(h)
    c[p:p + n] += delta * (1.0 - h[n - 1::-1])
    c[n - 1 - p:2 * n - 1 - p] += -delta * h
    c[n - 1] -= delta * delta
    h[p] += delta


def rotation_search_from_c(h, c, n, deadline, rng):
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c_full = c.copy()
    best_c_val = float(np.max(c) * dx)

    if time.time() >= deadline:
        return best_h, best_c_val, best_c_full

    if n <= 600:
        shifts = np.arange(n)
    else:
        m = min(n, 240)
        if m > 1:
            chosen = rng.choice(n, size=m - 1, replace=False)
            shifts = np.concatenate((np.array([0], dtype=int), chosen.astype(int)))
        else:
            shifts = np.array([0], dtype=int)

    evaluate_comp = n <= 550

    for s in shifts:
        if time.time() > deadline:
            break
        s = int(s)
        hs = np.roll(h, s)
        cs = correlate_counts(hs)
        val = float(np.max(cs) * dx)
        if val < best_c_val - 1e-14:
            best_c_val = val
            best_h = hs.copy()
            best_c_full = cs

        if evaluate_comp and time.time() < deadline:
            hc = 1.0 - hs
            cc = correlate_counts(hc)
            valc = float(np.max(cc) * dx)
            if valc < best_c_val - 1e-14:
                best_c_val = valc
                best_h = hc.copy()
                best_c_full = cc

    return best_h, best_c_val, best_c_full


def gradient_descent_from_c(h, c, n, deadline, rng):
    h = h.copy()
    c = c.copy()
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = float(np.max(c) * dx)

    v = np.zeros(n, dtype=float)
    stuck = 0

    phases = (
        (32, 0.018, 0.88),
        (16, 0.022, 0.86),
        (8, 0.026, 0.84),
        (4, 0.030, 0.82),
        (1, 0.025, 0.80),
    )
    fracs = (0.30, 0.25, 0.20, 0.15, 0.10)

    t_start = time.time()
    total = deadline - t_start
    if total <= 1.0:
        c_final = correlate_counts(best_h)
        val_final = float(np.max(c_final) * dx)
        if val_final < best_c - 1e-14:
            best_c = val_final
        return best_h, best_c, c_final

    acc = 0.0
    for pi in range(len(phases)):
        top_k, step, mom = phases[pi]
        phase_deadline = t_start + total * (acc + fracs[pi])
        acc += fracs[pi]

        while time.time() < phase_deadline - 0.5:
            current_c = float(np.max(c) * dx)
            if current_c < best_c - 1e-14:
                best_c = current_c
                best_h = h.copy()

            idxs = top_lag_indices(c, top_k)
            if idxs.size == 0:
                break

            g = top_gradient_counts(h, n, idxs)
            gn = float(np.linalg.norm(g))

            if gn < 1e-13:
                stuck += 5
                continue

            v = mom * v - step * (g / gn)
            vn = float(np.linalg.norm(v))
            if vn > 0.4:
                v *= 0.4 / vn

            base_top = float(np.mean(c[idxs]) * dx)
            trial_h = None
            trial_cfull = None
            trial_c = current_c
            trial_top = base_top

            for alpha in (1.0, 0.6, 0.35, 0.15, 0.05):

                c_trial = correlate_counts(trial)
                val = float(np.max(c_trial) * dx)
                tmean = float(np.mean(c_trial[idxs]) * dx)

                if val < best_c - 1e-14:
                    best_c = val
                    best_h = trial.copy()

                if val < trial_c - 1e-14 or (val < trial_c + 2e-12 and tmean < trial_top - 1e-14):
                    trial_c = val
                    trial_top = tmean
                    trial_h = trial
                    trial_cfull = c_trial

            if trial_h is not None:
                h = trial_h
                c = trial_cfull
                stuck = 0
            else:
                stuck += 1
                v *= 0.5
                if stuck > 50:
                    step *= 0.7
                    v *= 0.25
                    stuck = 0
                    if step < 1e-4:
                        break
                    h = project_h(h + rng.normal(0.0, 0.012, n), n)
                    c = correlate_counts(h)

    c_final = correlate_counts(best_h)
    val_final = float(np.max(c_final) * dx)
    if val_final < best_c - 1e-14:
        best_c = val_final
    return best_h, best_c, c_final

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == 0:
        return np.full(n_new, 0.5)
    if n_old == n_new:
        return np.asarray(h_old, dtype=float).copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new)


def random_binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    ones = n // 2
    idx = rng.permutation(n)
    if ones > 0:
        h[idx[:ones]] = 1.0
    if n % 2 == 1 and n > 0:
        h[idx[ones]] = 0.5
    return h
