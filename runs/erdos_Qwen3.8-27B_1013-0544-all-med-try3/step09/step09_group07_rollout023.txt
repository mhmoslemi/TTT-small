import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂f/∂h[j] = (1-h[j-lag])*1[j-lag>=0] - h[j+lag]*1[j+lag<n]
            # j-lag >= 0 means j >= lag
            # j+lag < n means j < n-lag
            if lag < n:
                # j in [lag, n-1]: contributes (1-h[j-lag])
                grad[lag:n] += wk * (1.0 - h[0:n-lag])
                # j in [0, n-1-lag]: contributes -h[j+lag]
                grad[0:n-lag] += wk * (-h[lag:n])
        else:
            m_abs = -lag
            if m_abs < n:
                # j in [m_abs, n-1]: contributes (1-h[j-m_abs])
                grad[m_abs:n] += wk * (1.0 - h[0:n-m_abs])
                # j in [0, n-1-m_abs]: contributes -h[j+m_abs]
                grad[0:n-m_abs] += wk * (-h[m_abs:n])
    return grad * dx

        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff
        grad = smooth_max_grad(x, n, alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        elif -n < lag < 0:
            m_abs = -lag
            grad[m_abs:n] += wk * (1.0 - h[0:n-m_abs])
            grad[0:n-m_abs] += wk * (-h[m_abs:n])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_val(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * 1e8 * diff * np.ones(n)
        return obj_val, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 80,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def targeted_local_search(h, n, deadline, rng, top_k=5, max_rounds=200000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    
    rounds = 0
    no_improve = 0
    
    while time.time() < deadline and rounds < max_rounds:
        rounds += 1
        top_indices = np.argsort(c_all)[-top_k:]
        current_max = c_all.max()
        
        if current_max < best_c - 1e-15:
            best_c = float(current_max)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30000:
            h = best_h + rng.normal(0, 0.003, n)
            h = project_feasible(h, n)
            c_all = compute_all_lags(h, n)
            no_improve = 0
            continue
        
        # For each top lag, find the most beneficial move
        best_move = None
        best_new_max = current_max + 1e-15
        
        for k_idx in top_indices:
            lag = k_idx - (n - 1)
            if lag == 0:
                # For lag 0: f_0 = sum h[i](1-h[i]), gradient is 1-2h
                # Most beneficial: decrease h where h>0.5, increase where h<0.5
                # Move from high h to low h
                i = int(np.argmax(h))
                j = int(np.argmin(h))
                if h[i] <= h[j] + 1e-8:
                    continue
                d = min(0.01, (h[i] - h[j]) * 0.2)
                if d < 1e-9:
                    continue
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_test = compute_all_lags(h, n)
                new_max = c_test.max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_move = (i, j, d, c_test.copy())
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
            else:
                m = abs(lag)
                if m >= n:
                    continue
                # For lag l>0: f_l = sum_{i=0}^{n-l-1} h[i]*(1-h[i+l])
                # Move from i to i+l (decrease h[i], increase h[i+l])
                # Benefit: reduces the i-th term
                # Or move from i+l to i (decrease h[i+l], increase h[i])
                # Benefit: reduces via (1-h[i+l+lag]) in the (i+l)-th term... 
                # Actually for positive lag: term i is h[i]*(1-h[i+lag])
                # Decreasing h[i] reduces term i by delta*(1-h[i+lag])
                # Increasing h[i+lag] reduces term (i-lag) by h[i-lag]*delta (if valid)
                
                if lag > 0:
                    # Try moving from i to i+lag
                    benefits = np.zeros(n - lag)
                    for i in range(n - lag):
                        # Moving d from h[i] to h[i+lag]:
                        # Reduces term i by d*(1-h[i+lag])
                        # Reduces term (i-lag) if i>=lag: by d... 
                        # Actually just compute the "score" as h[i]*(1-h[i+lag])
                        # Higher score = more reduction if we decrease h[i]
                        # But we also increase h[i+lag] which affects other terms
                        benefits[i] = h[i] * (1.0 - h[i + lag])
                    
                    # Try top few candidates
                    top_candidates = np.argsort(benefits)[-min(5, n-lag):]
                    for i in top_candidates:
                        d = min(0.005, h[i] * 0.15, (1.0 - h[i+lag]) *  0.15)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[i + lag] += d
                        h = project_feasible(h, n)
                        c_test = compute_all_lags(h, n)
                        new_max = c_test.max()
                        if new_max < best_new_max:
                            best_new_max = new_max
                            best_move = (i, i + lag, d, c_test.copy())
                        h[i] += d
                        h[i + lag] -= d
                        h = project_feasible(h, n)
                else:
                    m_abs = -lag
                    if m_abs >= n:
                        continue
                    # For negative lag: f = sum h[i]*(1-h[i-m_abs]) for valid i
                    # Similar logic
                    benefits = np.zeros(n - m_abs)
                    for i in range(n - m_abs):
                        benefits[i] = h[i] * (1.0 - h[i + m_abs])
                    
                    top_candidates = np.argsort(benefits)[-min(5, n-m_abs):]
                    for i in top_candidates:
                        d = min(0.005, h[i] * 0.15, (1.0 - h[i+m_abs]) * 0.15)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[i + m_abs] += d
                        h = project_feasible(h, n)
                        c_test = compute_all_lags(h, n)
                        new_max = c_test.max()
                        if new_max < best_new_max:
                            best_new_max = new_max
                            best_move = (i, i + m_abs, d, c_test.copy())
                        h[i] += d
                        h[i + m_abs] -= d
                        h = project_feasible(h, n)
        
        if best_move is not None and best_new_max < current_max - 1e-15:
            i, j, d, c_test = best_move
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            c_all = c_test
        else:
            # Try random small perturbation
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i != j and h[i] > h[j] + 1e-8:
                d = min(0.003, (h[i]-h[j])*0.1)
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_test = compute_all_lags(h, n)
                if c_test.max() < current_max + 1e-14:
                    c_all = c_test
    
    return best_h, best_c

def gradient_descent_local(h, n, deadline, rng, alpha=200000, base_lr=0.02):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = base_lr
    no_improve = 0
    
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        current_max = c.max()
        
        if current_max < best_c - 1e-15:
            best_c = float(current_max)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 8000:
            h = best_h + rng.normal(0, 0.004, n)
            h = project_feasible(h, n)
            no_improve = 0
            lr = base_lr
            continue
        
        grad = smooth_max_grad(h, n, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-13:
            break
        
        step_size = lr / (grad_norm + 1e-10)
        h_new = project_feasible(h - step_size * grad, n)
        
        c_new = compute_all_lags(h_new, n)
        if c_new.max() < current_max + 1e-15:
            h = h_new
            lr *= 0.9995
        else:
            lr *= 0.85
            if lr < 1e-7:
                h = best_h + rng.normal(0, 0.002, n)
                h = project_feasible(h, n)
                lr = base_lr * 0.5
    
    return best_h, best_c

    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
