```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorized gradient computation
    for idx in range(len(w)):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-13:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    total = 0.0

            total += dx * np.sum(x[:nv] * (1.0 - x[k:]))
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sobj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sgrad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = n - 1
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        k = idx - nn
        wm = w[idx] * dx
        if k > 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        elif k < 0:
            t = -k
            nv = n - t
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[t:])
                grad[t:] += wm * (-x[:nv])
        else:
            grad += wm * (1.0 - 2.0 * x)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def aobj(x, n, sidx, mu):
    dx = 2.0 / n
    total = 0.0
    nn = n - 1
    for idx in sidx:
        k = idx - nn
        if k == 0:
            total += dx * np.dot(x, 1.0 - x)
        elif k > 0:
            total += dx * np.dot(x[k:], 1.0 - x[:n-k])
        else:
            t = -k
            total += dx * np.dot(x[:n-t], 1.0 - x[t:])
    total += mu * (x.sum() - n / 2.0) ** 2
    return float(total)

def agrad(x, n, sidx, mu):
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = n - 1
    for idx in sidx:
        k = idx - nn
        if k == 0:
            grad += 2.0 * dx * (0.5 - x)
        elif k > 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += dx * (1.0 - x[k:])
                grad[k:] += dx * (-x[:nv])
        else:
            t = -k
            nv = n - t
            if nv > 0:
                grad[:nv] += dx * (1.0 - x[t:])
                grad[t:] += dx * (-x[:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, obj, obj_args, jac, jac_args, bounds, maxiter, deadline):
    try:
        res = minimize(obj, h, jac=jac, args=obj_args,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'maxfun': maxiter * 4,
                               'ftol': 1e-17, 'gtol': 1e-15})
        return res.x
    except Exception:
        return h.copy()

def run_slsqp(h, n, obj, obj_args, jac, jac_args, bounds, maxiter, deadline):
    try:
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)

        res = minimize(obj, h, jac=jac, args=obj_args,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h.copy()
