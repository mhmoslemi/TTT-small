            maxit = max(100, int(rem2 * 1.2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, mu), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-24,
                                        'gtol': 1e-20, 'maxfun': maxit * 8,
                                        'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c


def run_full_loop(h, n, deadline, alphas=None, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 1.5))
        try:
            res = minimize(full_smooth_obj, best_h, jac=full_smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-24,
                                    'gtol': 1e-20, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def random_perturb(h, n, rng, sigma=0.02):
    noise = rng.normal(0, sigma, size=n)
    h_new = h + noise
    return project_feasible(h_new, n)


def coordinate_descent(h, n, deadline, active, n_sweeps=200, step_scales=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_scales is None:
        step_scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        # Recompute active set
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.005 * max_val)
        act = np.where(c_vec >= threshold)[0]
        if len(act) > 5:
            order = np.argsort(-c_vec[act])
            act = act[order[:5]]
        for scale in step_scales:
            improved = False
            for m in act:
                k = m - (n - 1)
                if k >= 0:
                    nv = n - k
                    # Try moving mass in the overlap region
                    for i in range(0, nv, 3):
                        if best_h[i] > 0.01 and best_h[i + k] < 0.98:
                            # Increase h[i], decrease h[i+k]
                            best_h[i] += scale
                            best_h[i + k] -= scale
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i] -= scale
                                best_h[i + k] += scale
                        if best_h[i] < 0.98 and best_h[i + k] > 0.01:
                            # Decrease h[i], increase h[i+k]
                            best_h[i] -= scale
                            best_h[i + k] += scale
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i] += scale
                                best_h[i + k] -= scale
                else:
                    kk = -k
                    nv = n - kk
                    for i in range(0, nv, 3):
                        if best_h[i + kk] > 0.01 and best_h[i] < 0.98:
                            best_h[i + kk] += scale
                            best_h[i] -= scale
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i + kk] -= scale
                                best_h[i] += scale
            if not improved:
                break
        if not improved:
            break
    return project_feasible(best_h, n), best_c


def subgradient_refine(h, n, deadline, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.005, 0.005)
        else:
            eta *= 0.995
        h = h_new
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Load initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick full L-BFGS at original resolution
    p1_end = min(t0 + 70, deadline - 800)
    h1, c1 = run_full_loop(best_h, best_n, p1_end,
                           alphas=[50, 200, 1000, 5000, 20000, 100000],
                           mu=0.01, maxiters=[800, 1000, 1200, 1500, 2000, 2500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Multi-restart exploration at original resolution
    n_restarts = 5
    t_restart_start = time.time()
    t_restart_end = min(t0 + 140, deadline - 700)
    if t_restart_end > t_restart_start + 30:
        per_restart = (t_restart_end - t_restart_start) / n_restarts
        for r in range(n_restarts):
            if time.time() > t_restart_end - 10:
                break
            perturbation = random_perturb(best_h, best_n, rng, sigma=0.015 * (r + 1) / 3)
            # Try a different structural perturbation
            if r % 2 == 0:
                # Shift-based perturbation
                shift = rng.randint(-5, 6)
                perturbation = np.roll(best_h, shift)
                perturbation = project_feasible(perturbation, best_n)
            else:
                # Smooth bump perturbation
                center = rng.randint(n_best_n if 'n_best_n' in dir() else best_n // 3,
                                     (best_n // 3) * 2)
                width = best_n // 8
                bump = np.exp(-0.5 * ((np.arange(best_n) - center) / width) ** 2)
                perturbation = best_h + 0.03 * (bump - bump.mean())
                perturbation = project_feasible(perturbation, best_n)
            r_end = min(time.time() + per_restart - 5, deadline - 700)
            hr, cr = run_full_loop(perturbation, best_n, r_end,
                                   alphas=[100, 1000, 10000, 50000],
                                   mu=0.01, maxiters=[500, 800, 1000, 1500])
            if cr < best_c:
                best_c = cr
                best_h = hr

    # Phase 3: Active set refinement at original resolution
    if time.time() < deadline - 600:
        p3_end = min(time.time() + 80, deadline - 500)
        h3, c3 = run_active_loop(best_h, best_n, p3_end, top_k=6,
                                 alphas=[100000, 500000, 2000000, 10000000, 50000000],
                                 mu=0.002)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and optimize
    n_mid = 1000
    if time.time() < deadline - 400:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c + 0.002:
            p4_end = min(time.time() + 180, deadline - 300)
            h4, c4 = run_full_loop(h_mid, n_mid, p4_end,
                                   alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                                   mu=0.005, maxiters=[600, 800, 1000, 1200, 1500, 2000, 2500, 3000])
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n_mid

    # Phase 5: Active set at n=1000
    if time.time() < deadline - 250:
        p5_end = min(time.time() + 120, deadline - 180)
        h5, c5 = run_active_loop(best_h, best_n, p5_end, top_k=5,
                                 alphas=[200000, 1000000, 5000000, 20000000, 100000000],
                                 mu=0.001)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Upsample to n=1500
    n_high = 1500
    if time.time() < deadline - 250 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.003:
            p6_end = min(time.time() + 130, deadline - 130)
            h6, c6 = run_full_loop(h_high, n_high, p6_end,
                                   alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                                   mu=0.003, maxiters=[500, 700, 900, 1100, 1400, 1800, 2200, 2500])
            if c6 < best_c:
                best_c = c6
                best_h = h6
                best_n = n_high

    # Phase 7: Active set at highest resolution
    if time.time() < deadline - 120:
        p7_end = min(time.time() + 80, deadline - 60)
        h7, c7 = run_active_loop(best_h, best_n, p7_end, top_k=4,
                                 alphas=[500000, 2000000, 10000000, 50000000, 200000000],
                                 mu=0.001)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Coordinate descent polish
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 30, deadline - 15)
        h8, c8 = coordinate_descent(best_h, best_n, p8_end, active=None,
                                    n_sweeps=50, step_scales=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Subgradient refinement
    if time.time() < deadline - 20:
        p9_end = min(time.time() + 15, deadline - 8)
        h9, c9 = subgradient_refine(best_h, best_n, p9_end, n_iter=10000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final tight active set push
    if time.time() < deadline - 5:
        p10_end = deadline - 2
        h10, c10 = run_active_loop(best_h, best_n, p10_end, top_k=3,
                                   alphas=[10000000, 50000000, 200000000],
                                   mu=0.0005)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
