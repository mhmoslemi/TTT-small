import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_all_grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        k = idx - (n-1)
        if k >= 0:
            nv = n - k
            G[idx, :nv] += (1.0 - h[k:k+nv]) * dx
            if k > 0:
                G[idx, k:n] += -h[:nv] * dx
        else:
            kk = -k
            nv = n - kk
            G[idx, kk:n] += (1.0 - h[:nv]) * dx
            G[idx, :nv] += -h[kk:n] * dx
    return G

def local_search_gradient(h, n, deadline, max_iter=100000):
    """Local search using gradient-based swap evaluation."""
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    G = compute_all_grads(h, n)
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_step_idx = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if time.time() > deadline - 0.5:
            break
            
        improved = False
        max_c = c_vec.max()
        
        # Find positions most affecting the max correlation
        active = np.where(c_vec >= max_c - 0.005 * max_c)[0]
        # For active shifts, find positions with large gradient
        active_grads = G[active, :]  # shape (n_active, n)
        abs_grad = np.abs(active_grads).mean(axis=0)
        
        # Try swapping from high-grad positions to low-grad positions
        # Sort by gradient magnitude
        order = np.argsort(-abs_grad)
        
        for i in order[:50]:
            if time.time() > deadline:
                break
            for j in order[50:150]:  # swap to positions with lower grad
                if i == j:
                    continue
                for eps in [0.001, 0.0005, 0.0002]:

                    new_c_vec = c_vec + dc
                    new_max = new_c_vec.max()

                        h[j] += eps

                        c_vec = new_c_vec

            if improved and it % 10 == 0:

                G = compute_all_grads(h, n)

                break

            if best_step_idx < len(steps) - 1:
                best_step_idx += 1
            else:
                break
        
        if it % 100 == 0:
            G = compute_all_grads(h, n)
            c_vec = compute_c5_vec(h, n)
    
    return h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_all_grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            G[idx, :nv] += (1.0 - h[k:k + nv]) * dx
            if k > 0:
                G[idx, k:n] += -h[:nv] * dx
        else:
            kk = -k
            nv = n - kk
            G[idx, kk:n] += (1.0 - h[:nv]) * dx
            G[idx, :nv] += -h[kk:n] * dx
    return G

def shift_grad_single(h, n, idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] = (1.0 - h[k:k + nv]) * dx
        grad[k:k + nv] += -h[:nv] * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:kk + nv] = (1.0 - h[:nv]) * dx
        grad[:nv] += -h[kk:kk + nv] * dx
    return grad

def sqp_lp_solve(h, n, active_idx, c_vec):

    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    n_act = len(active_idx)

    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        grad = shift_grad_single(h, n, idx)
        A_ub[row, :n] = grad
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:

        pass
    return None

def smooth_obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()

    grad = np.zeros(n)
    active_mask = w > 1e-14

    return grad
