def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(c) / len(c)
    else:
        w = w / s
    return w

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < n:
                upper = n - lag
                grad[0:upper] += w[k] * (1.0 - h[lag:lag + upper])
                grad[lag:n] += w[k] * (-h[0:upper])

                upper = n - m
                grad[0:upper] += w[k] * (-h[m:m + upper])
                grad[m:n] += w[k] * (1.0 - h[0:upper])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 60,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def equalization_search(h, n, deadline, rng):
    """Try to equalize the top lags by targeted moves."""
    h = h.copy()

            break
        c_all = compute_all_lags(h, n)
        top_K = 5
        top_idx = np.argsort(c_all)[-top_K:]
        top_vals = c_all[top_idx]
        top_mean = top_vals.mean()
        top_max = top_vals.max()
        
        if top_max - top_mean < 1e-6:
            break
        
        # Focus on the lag with the highest value
        k_star = top_idx[np.argmax(top_vals)]
        lag = k_star - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        if m >= n:
            continue
        
        # Find the pair that gives the most reduction for this specific lag
        if lag > 0:
            # c_lag = dx * sum h[i]*(1-h[i+lag])
            # To reduce: decrease h[i] or increase h[i+lag] where h[i] is high and h[i+lag] is low
            pairs = h[0:n-m] - h[lag:n]
        else:
            m = -lag
            pairs = h[m:n] - h[0:n-m]
            # For negative lag, c = dx * sum h[i+m]*(1-h[i])
            # To reduce: decrease h[i+m] or increase h[i]
            # So we want h[i+m] high and h[i] low
            pairs = h[m:n] - h[0:n-m]
        
        # Pick a random candidate from the top pairs
        n_candidates = min(50, len(pairs))
        if n_candidates < 2:
            continue
        top_pairs = np.argsort(pairs)[-n_candidates:]
        i_idx = top_pairs[rng.randint(n_candidates)]
        
        if lag > 0:
            i = i_idx
            j = i + lag
        else:
            i = i_idx
            j = i + m
        
        if h[j] > h[i]:
            i, j = j, i
        
        d = min(0.01, (h[i] - h[j]) * 0.2)
        if d < 1e-10:
            continue
        
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)

        for s in range(n_starts):
            if time.time() > phase6_deadline:
                break
            h_opt, c_opt = diverse_start_optimize(best_h, n, phase6_deadline, rng, seed_offset=s)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                # Polish the new best
                h_opt2 = lbfgsb_optimize(best_h, n, 50000000, min(deadline - 80, time.time() + 20), maxiter=1000)
                c_opt2 = compute_c5(h_opt2, n)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2
    
    # Phase 7: Final direction-based local search
    if time.time() < deadline - 30:
        phase7_deadline = min(deadline - 15, time.time() + 60)
        h_opt, c_opt = direction_local_search(best_h, n, phase7_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Very fine local search with tiny steps
    if time.time() < deadline - 10:
        h = best_h.copy()
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-6:]
        for _ in range(10000):
            if time.time() > deadline - 5:
                break
            k_star = top_idx[rng.randint(len(top_idx))]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-7:
                d = min(0.0003, (h[i] - h[j]) * 0.15)
                if d < 1e-11:
                    continue
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
            elif h[j] > h[i] + 1e-7:
                d = min(0.0003, (h[j] - h[i]) * 0.15)
                if d < 1e-11:
                    continue
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
    
    # Final L-BFGS-B polish if time remains
    if time.time() < deadline - 5:
        h_opt = lbfgsb_optimize(best_h, n, 500000000, deadline - 3, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
