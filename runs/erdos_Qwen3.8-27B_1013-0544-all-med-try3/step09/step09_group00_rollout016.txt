def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue

def upsample(h, n_old, n_new):
    h_new = np.zeros(n_new)
    scale = n_new / n_old
    for i in range(n_new):
        src = i / scale
        i0 = int(src)
        i1 = min(i0 + 1, n_old - 1)
        frac = src - i0
        h_new[i] = (1 - frac) * h[i0] + frac * h[i1]
    return project_feasible(h_new, n_new)

        per_start = max(10, (quick_end - time.time()) / max(1, len(starts) - i))
        h_opt, c_opt = lbfgs_phase(s, n0, time.time() + per_start,
                                    alphas=[50, 500, 5000, 50000], mu=0.01)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            best_n = n0

    # Phase 2: Intensive L-BFGS
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 200, deadline - 350)
        h2, c2 = lbfgs_phase(best_h, best_n, p2_end,
                             alphas=[100, 1000, 10000, 50000, 200000, 1000000],
                             mu=0.005)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Active set optimization
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 120, deadline - 200)
        h3, c3 = active_set_phase(best_h, best_n, p3_end, top_k=6)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient local search
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 150)
        h4, c4 = grad_local_search(best_h, best_n, p4_end, use_anneal=True)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try n=1000
    n1k = 1000
    if time.time() < deadline - 350:
        h_up = upsample(best_h, best_n, n1k)
        h_up = binarize(h_up, n1k, rng)
        h1k, c1k = lbfgs_phase(h_up, n1k, time.time() + 80,
                                alphas=[50, 500, 5000], mu=0.01)
        if time.time() < deadline - 250:
            h1k, c1k = active_set_phase(h1k, n1k, time.time() + 60, top_k=4)
        if time.time() < deadline - 180:
            h1k, c1k = grad_local_search(h1k, n1k, time.time() + 40, use_anneal=True)
        if c1k < best_c - 1e-6:
            best_c = c1k
            best_h = h1k
            best_n = n1k

    # Phase 6: Binary hill climb at current best resolution
    if time.time() < deadline - 120:
        h_bin = binarize(best_h, best_n, rng)
        p6_end = min(time.time() + 50, deadline - 80)
        h6, c6 = binary_hill_climb(h_bin, best_n, p6_end, rng)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # Phase 7: Fine pair search
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=30)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Another round of gradient search
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 20, deadline - 15)
        h8, c8 = grad_local_search(best_h, best_n, p8_end, use_anneal=True)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final very fine adjustments
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection
    fn = len(best_h)
    best_h = project_feasible(best_h, fn)
    best_c = compute_c5(best_h, fn)

    return best_h, best_c, fn
