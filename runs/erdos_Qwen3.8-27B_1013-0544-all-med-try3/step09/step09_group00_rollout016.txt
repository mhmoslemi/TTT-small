        i = np.random.randint(n)
        j = np.random.randint(n)

        # new_max for each j: max(base + Ddx[j])

        new_maxes = candidates.max(axis=1)  # shape (len(zeros),)

            best_j = zeros[best_idx]
    
    if best_i < 0:

    dc = (Ddx[best_j] - Ddx[best_i])  # change in c
    return best_i, best_j, dc, best_gain
