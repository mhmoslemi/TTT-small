import time
import math
import numpy as np
from scipy.optimize import minimize

_LAG_ORIENT = {}
LB_STATE = {'n': 0, 'alpha': 1.0, 'penalty': 1e6, 'target': 0.0, 'top': 128}


def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0

    for _ in range(70):
        lam = 0.5 * (lo + hi)
        hs = np.clip(h - lam, 0.0, 1.0)
        s = float(hs.sum())
        if s > target:
            lo = lam
        else:
            hi = lam

    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(5):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        interior_mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior_mask))
        if cnt > 0:
            h[interior_mask] += diff / float(cnt)
        else:
            if diff > 0:
                zeros = np.where(h < 0.5)[0]
                if zeros.size:
                    h[zeros[0]] = 1.0
            else:
                ones = np.where(h > 0.5)[0]
                if ones.size:
                    h[ones[0]] = 0.0
        h = np.clip(h, 0.0, 1.0)

    return h


def _lag_orientation(n):
    if n in _LAG_ORIENT:
        return _LAG_ORIENT[n]

    if n <= 1:
        rev = False
        _LAG_ORIENT[n] = rev
        return rev

    h = np.linspace(0.1, 0.9, n)
    q = 1.0 - h
    c = np.correlate(h, q, mode="full")

    pos = float(np.dot(h[:-1], q[1:]))
    neg = float(np.dot(h[1:], q[:-1]))

    score_rev = abs(c[n - 2] - pos) + abs(c[n] - neg)
    score_norm = abs(c[n - 2] - neg) + abs(c[n] - pos)
    rev = score_rev < score_norm

    _LAG_ORIENT[n] = rev
    return rev


def compute_lags(h, n):
    q = 1.0 - h
    c = np.correlate(h, q, mode="full")
    if n > 1 and _lag_orientation(n):
        c = c[::-1]
    return c * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(compute_lags(h, n)))


def smooth_max_value(f, alpha):
    m = float(np.max(f))
    alpha = float(alpha)
    if alpha <= 0.0:
        return m
    sh = alpha * (f - m)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return m + math.log(s) / alpha


def smooth_max_weights(f, alpha):
    m = float(np.max(f))
    alpha = float(alpha)
    if alpha <= 0.0:
        return np.full_like(f, 1.0 / float(f.size))
    sh = alpha * (f - m)
    w = np.exp(sh)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.full_like(f, 1.0 / float(f.size))
    else:
        w = w / s
    return w


def gradient_from_weights(h, n, w, top):
    g = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    nz = np.where(w > 1e-12)[0]
    if nz.size > top:
        nz = nz[np.argsort(w[nz])[-top:]]

    for idx in nz:
        wi = float(w[idx])
        lag = int(idx - (n - 1))
        if lag == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            g[m:n] += wi * (1.0 - h[0:n - m])
            g[0:n - m] -= wi * h[m:n]
        else:
            m = -lag
            g[0:n - m] += wi * (1.0 - h[m:n])
            g[m:n] -= wi * h[0:n - m]

    return g * dx


def lb_obj_fun(x):
    state = LB_STATE
    n = int(state["n"])
    alpha = float(state["alpha"])
    penalty = float(state["penalty"])
    target = float(state["target"])
    top = int(state["top"])

    h_raw = np.asarray(x, dtype=float)
    h = np.clip(h_raw, 0.0, 1.0)

    f = compute_lags(h, n)
    val = smooth_max_value(f, alpha)

    diff = float(h.sum()) - target
    val += penalty * diff * diff

    out = np.zeros(n, dtype=float)
    below = h_raw < 0.0
    above = h_raw > 1.0
    if np.any(below):
        out[below] = -h_raw[below]
    if np.any(above):
        out[above] = h_raw[above] - 1.0
    val += 1e4 * float(np.sum(out * out))

    w = smooth_max_weights(f, alpha)
    grad = gradient_from_weights(h, n, w, top)
    grad += 2.0 * penalty * diff
    grad += 2.0e4 * out

    return float(val), grad


def lbfgsb_optimize(h, n, alpha, maxiter, penalty=1e6):
    h0 = project_feasible(h, n).copy()
    LB_STATE["n"] = int(n)
    LB_STATE["alpha"] = float(alpha)
    LB_STATE["penalty"] = float(penalty)
    LB_STATE["target"] = 0.5 * int(n)
    W = 2 * int(n) - 1
    if alpha < 1000.0:
        top = min(W, 400)
    else:
        top = min(W, 160)
    LB_STATE["top"] = int(top)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lb_obj_fun,
            h0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-18,
                "gtol": 1e-8,
                "maxls": 40,
                "iprint": -1,
            },
        )
        hout = np.clip(res.x, 0.0, 1.0)
    except Exception:
        hout = h0

    return project_feasible(hout, n)


def make_ramp(n, decreasing=True):

def expand_to_resolution(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    x_src = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / float(n_src))
    x_dst = (np.arange(n_dst, dtype=float) + 0.5) * (2.0 / float(n_dst))
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def add_smooth_noise(base, amp, rng):
    n = int(len(base))
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    ph = rng.uniform(0.0, 2.0 * math.pi, 4)

        + 0.3 * np.cos(3.0 * np.pi * x + ph[2])
    )
    return project_feasible(base + p, n)


def make_unit_delta(omh, hrev, n, p, q):
    W = 2 * n - 1
    udf = np.zeros(W, dtype=float)

    s = n - 1 - p
    e = 2 * n - 1 - p
    udf[s:e] -= omh

    s = p
    e = n + p
    udf[s:e] += hrev

    s = n - 1 - q
    e = 2 * n - 1 - q
    udf[s:e] += omh

    s = q
    e = n + q
    udf[s:e] -= hrev

    cross = []
    m = q - p
    if m != 0:
        im = n - 1 + m
        im2 = n - 1 - m
        if 0 <= im < W:
            cross.append(im)
        if 0 <= im2 < W:
            cross.append(im2)

    return udf, cross

                    continue
                scalar = dx * d
                quad = dx * d * d
                cand = f + scalar * udf
                for idx in cross:
                    cand[idx] += quad
                cm = float(np.max(cand))
                if cm < best_cand - 1e-13:
                    best_cand = cm
                    best_d = d

            if best_d > 0.0:
                d = best_d
                scalar = dx * d
                quad = dx * d * d

                h[p] -= d
                h[q] += d
                omh[p] += d
                omh[q] -= d
                hrev[n - 1 - p] -= d
                hrev[n - 1 - q] += d

                f += scalar * udf
                for idx in cross:
                    f[idx] += quad

                C = best_cand
                accepted = True
                if C < best_c - 1e-14:
                    best_c = C
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                if moves % 50 == 0:
                    top_idx = np.argsort(f)[-top_k:]
                break

        if not accepted:
            no_improve += 1
            if no_improve > perturb_every:
                if rng.random() < 0.55:
                    h = project_feasible(best_h + rng.normal(0.0, 0.006, n), n)
                else:
                    h = project_feasible(h + rng.normal(0.0, 0.010, n), n)
                f = compute_lags(h, n)
                C = float(np.max(f))
                omh = 1.0 - h
                hrev = h[::-1].copy()
                top_idx = np.argsort(f)[-top_k:]
                if C < best_c:
                    best_c = C
                    best_h = h.copy()
                no_improve = 0

        moves += 1

    h_fin = project_feasible(best_h, n)
    c_fin = compute_c5(h_fin, n)
    if c_fin < best_c:
        best_c = c_fin
    return h_fin, float(best_c)


def anneal_local(h, n, deadline, rng, t_init=0.004, t_min=1e-5, cool=0.99995):
    n = int(n)
    h = project_feasible(h, n).copy()
    f = compute_lags(h, n)
    C = float(np.max(f))

    best_h = h.copy()
    best_c = C

    dx = 2.0 / float(n)
    omh = 1.0 - h
    hrev = h[::-1].copy()

    top_k = min(2 * n - 1, 25)
    top_idx = np.argsort(f)[-top_k:]

    T = t_init
    no_best = 0
    moves = 0

    while time.time() < deadline and T > t_min:
        if moves % 1000 == 0:
            f = compute_lags(h, n)
            C = float(np.max(f))
            omh = 1.0 - h
            hrev = h[::-1].copy()
            top_idx = np.argsort(f)[-top_k:]
            if C < best_c:
                best_c = C
                best_h = h.copy()
                no_best = 0
            if no_best > 20000 or C > best_c + 0.08:
                h = project_feasible(best_h + rng.normal(0.0, 0.005, n), n)
                f = compute_lags(h, n)
                C = float(np.max(f))
                omh = 1.0 - h
                hrev = h[::-1].copy()
                top_idx = np.argsort(f)[-top_k:]
                no_best = 0

        dirs = None
        if rng.random() < 0.75 and top_idx.size > 0:
            k = int(top_idx[rng.randint(top_idx.size)])
            m = k - (n - 1)
            if m != 0 and abs(m) < n:
                L = n - abs(m)
                if L > 1:
                    a = int(rng.randint(0, L))
                    if m > 0:
                        p = a
                        q = a + m
                    else:
                        q = a
                        p = a - m
                    if rng.random() < 0.50:
                        dirs = ((p, q),)
                    else:
                        dirs = ((q, p),)

        if dirs is None:
            p = int(rng.randint(0, n))
            q = int(rng.randint(0, n))
            if p == q:
                T *= cool
                moves += 1
                continue
            if rng.random() < 0.50:
                if h[p] >= h[q]:
                    dirs = ((p, q),)
                else:
                    dirs = ((q, p),)
            else:
                dirs = ((p, q),)

        accepted_any = False
        p, q = dirs[0]
        maxd = min(float(h[p]), float(omh[q]))
        if maxd <= 1e-13:
            T *= cool
            moves += 1
            continue

        base = min(0.5 * maxd, 0.003 + 3.0 * T)
        base = max(1e-5, base)
        d = maxd * 0.999 * max(0.1, min(1.0, rng.uniform(0.3, 1.0)))
        d = min(d, base)
        if d <= 0.0:
            T *= cool
            moves += 1
            continue

        udf, cross = make_unit_delta(omh, hrev, n, p, q)
        scalar = dx * d
        quad = dx * d * d
        cand = f + scalar * udf
        for idx in cross:
            cand[idx] += quad
        cm = float(np.max(cand))

        de = cm - C
        accept = de < -1e-14
        if not accept and de > 0.0:
            ratio = de / T
            if ratio < 40.0 and rng.random() < math.exp(-ratio):
                accept = True

        if accept:
            h[p] -= d
            h[q] += d
            omh[p] += d
            omh[q] -= d
            hrev[n - 1 - p] -= d
            hrev[n - 1 - q] += d

            f += scalar * udf
            for idx in cross:
                f[idx] += quad

            C = cm
            accepted_any = True
            if C < best_c - 1e-14:
                best_c = C
                best_h = h.copy()
                no_best = 0
            else:
                no_best += 1

        T *= cool
        moves += 1

    h_fin = project_feasible(best_h, n)
    c_fin = compute_c5(h_fin, n)
    if c_fin < best_c:
        best_c = c_fin
    return h_fin, float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if budget_s >= 900.0:
        n_main = 1200
    elif budget_s >= 450.0:
        n_main = 800
    else:
        n_main = 400
    n_main -= n_main % 2
    if n_main < 4:
        n_main = 4

    best_h = None
    best_c = float("inf")

    if parent is not None and len(parent) > 0:
        hp = project_feasible(parent, len(parent))
        cp = compute_c5(hp, len(parent))
        best_h = hp
        best_c = cp

    work_h = None
    work_c = float("inf")

    if parent is not None and len(parent) > 0:
        hw = expand_to_resolution(parent, len(parent), n_main)
        cw = compute_c5(hw, n_main)
        work_h = hw
        work_c = cw
    else:
        work_h = make_ramp(n_main, True)
        work_c = compute_c5(work_h, n_main)

    for dec in (True, False):
        hr = make_ramp(n_main, dec)
        cr = compute_c5(hr, n_main)
        if cr < work_c:
            work_h = hr
            work_c = cr

    for amp in (0.02, 0.05, 0.10):
        hpert = add_smooth_noise(work_h, amp, rng)
        cpert = compute_c5(hpert, n_main)
        if cpert < work_c:
            work_h = hpert
            work_c = cpert

    if work_c < best_c:
        best_h = work_h.copy()
        best_c = work_c

    if time.time() < deadline - 450.0:
        for alpha, mx in ((300.0, 160), (3000.0, 220), (30000.0, 280)):
            if time.time() > deadline - 400.0:
                break
            hopt = lbfgsb_optimize(work_h, n_main, alpha, mx, penalty=1e6)
            copt = compute_c5(hopt, n_main)
            if copt < work_c - 1e-14:
                work_c = copt
                work_h = hopt
                if copt < best_c:
                    best_c = copt
                    best_h = work_h.copy()

    if time.time() < deadline - 260.0:
        dl = min(deadline - 220.0, time.time() + 300.0)
        hloc, cloc = local_search(work_h, n_main, dl, rng)
        if cloc < work_c - 1e-14:
            work_c = cloc
            work_h = hloc
        if cloc < best_c:
            best_c = cloc
            best_h = hloc.copy()

    if time.time() < deadline - 140.0:
        dl = min(deadline - 90.0, time.time() + 150.0)
        han, can = anneal_local(work_h, n_main, dl, rng)
        if can < work_c - 1e-14:
            work_c = can
            work_h = han
        if can < best_c:
            best_c = can
            best_h = han.copy()

    if time.time() < deadline - 35.0:
        dl = min(deadline - 12.0, time.time() + 150.0)
        hloc, cloc = local_search(work_h, n_main, dl, rng)
        if cloc < work_c - 1e-14:
            work_c = cloc
            work_h = hloc
        if cloc < best_c:
            best_c = cloc
            best_h = hloc.copy()

    if time.time() < deadline - 160.0:
        n_high = len(best_h) + 500
        n_high -= n_high % 2
        n_high = max(1200, n_high)
        n_high = min(1600, n_high)
        if n_high >= 8:
            hh = expand_to_resolution(best_h, len(best_h), n_high)
            ch = compute_c5(hh, n_high)

            if time.time() < deadline - 130.0:
                hh = lbfgsb_optimize(hh, n_high, 30000.0, 90, penalty=1e6)
                ch = compute_c5(hh, n_high)

            dh = min(deadline - 25.0, time.time() + 130.0)
            hh, ch = local_search(hh, n_high, dh, rng)

            if ch < best_c:
                best_c = ch
                best_h = hh.copy()

    if best_h is None:
        best_h = make_ramp(n_main, True)
        best_c = compute_c5(best_h, len(best_h))

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, float(best_c), int(len(best_h))
