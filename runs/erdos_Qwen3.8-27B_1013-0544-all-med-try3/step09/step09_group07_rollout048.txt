import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sse_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sse_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def sse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = sse_weights(c, alpha)
    grad = np.zeros(n)
    nk = 2 * n - 1
    for k in range(nk):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n-idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n-idx])
    return grad * dx

def lbfgs_obj(x, n, alpha, penalty):
    target_sum = n / 2.0
    xc = np.clip(x, 0.0, 1.0)
    c = compute_all_lags(xc, n)
    obj = sse_value(c, alpha)
    diff = xc.sum() - target_sum
    obj += penalty * diff * diff
    g = sse_gradient(xc, n, alpha)
    g += 2.0 * penalty * diff
    return obj, g

def lbfgsb_optimize(h0, n, alpha, t_end, maxiter=3000, penalty=1e6):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h0, method='L-BFGS-B', jac=True,
                       bounds=bounds, args=(n, alpha, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
