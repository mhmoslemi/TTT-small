            if tot < 1e-15:

Since Δc[m] = δ * [f(m-i) - f(m-j)] and |f| ≤ 1, the maximum possible increase in any c[m] is δ. So if δ is very small, the change is small.

The key lags to check are:
- The current maximum lag(s)
- Any lag where f(m-i) - f(m-j) is maximized (potential new maximum)

For the local search, I can:
1. Compute c[m] for all m
2. For the worst lag, compute Δc for each candidate pair
3. Check if the swap reduces the worst lag without increasing any other lag above it
4. Execute the best swap

This is much faster than recomputing the full correlate each time.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_lse(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_lse_grad_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def subgrad_phase(h, n, deadline, top_k=3, alpha=3000, lr0=0.4, decay=0.99995, momentum=0.92):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr0
    imp_count = 0
    stall = 0
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_k:]
        ct = c[top_idx]
        w = smooth_lse_grad_weights(ct, alpha)
        g = np.zeros(n)
        for ii in range(len(top_idx)):
            if w[ii] < 1e-10:
                continue
            lag = top_idx[ii] - (n - 1)
            g += w[ii] * lag_gradient(h, n, lag)
        g *= (2.0 / n)
        v = momentum * v + lr * g
        h = project_feasible(h - v, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            imp_count += 1
            stall = 0
        else:
            stall += 1
        lr *= decay
        if stall > 3000:
            lr *= 0.8
            v *= 0.3
            stall = 0
        if imp_count > 50000 and lr < 0.01:
            break
    return best_h, best_c

def fast_pair_delta(h, n, i, j, delta):
    """Analytically compute change in all lag values for swap h[i]+=delta, h[j]-=delta."""
    # f(x) = 1 - 2*h[x] for valid x, else 0
    def f(x):
        if 0 <= x < n:
            return 1.0 - 2.0 * h[x]
        return 0.0
    fi = f
    fj = f
    # For each m in [0, 2n-2]: delta_c[m] = delta * (fi(m-i) - fj(m-j))
    # We only need to check if max increases
    # fi(m-i) is nonzero when 0<=m-i<n, i.e., i<=m<i+n
    # fj(m-j) is nonzero when 0<=m-j<n, i.e., j<=m<j+n
    # The range where either is nonzero: [min(i,j), max(i,j)+n-1]
    m_lo = max(0, min(i, j))
    m_hi = min(2*n-2, max(i, j) + n - 1)
    # For m in [m_lo, m_hi], compute delta_c
    # fi(m-i): valid when 0<=m-i<n -> i<=m<i+n -> m-i in [0,n-1]
    # fj(m-j): valid when 0<=m-j<n -> j<=m<j+n -> m-j in [0,n-1]
    # We need max of c[m] + delta*(fi(m/i) - fi(m/j)) over all m
    # Instead of computing for all m, just check the affected range
    # and the current max lag
    return fi, fj, m_lo, m_hi

def local_search_fast(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        # Compute the "pressure" at each pair for the worst lag
        # For lag>0: c[lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # We want to reduce this. Decrease h[i+lag] or increase h[i].
        # Gradient for lag>0: grad[i+lag] += (1-h[i]), grad[i] += -h[i+lag]
        # So positions where h[i+lag] > h[i] are "bad" (contribute more to overlap)
        
        n_improved = 0
        deltas = [0.005, 0.002, 0.001, 0.0005]
        
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    # Bad if h[j] is high and h[i] is low (h[j]*(1-h[i]) is large)
                    # Move mass from j to i
                    if h[j] > h[i] + 0.01:
                        delta = min(d, (h[j] - h[i]) * 0.15, h[j] * 0.08, (1 - h[i]) * 0.08)
                    elif h[i] > h[j] + 0.01:
                        delta = -min(d, (h[i] - h[j]) * 0.15, h[i] * 0.08, (1 - h[j]) * 0.08)
                    else:
                        continue
                    if abs(delta) < 1e-8:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        n_improved += 1
                        if n_improved > 200:
                            break
                    else:
                        h[i], h[j] = old_i, old_j
                        h = project_feasible(h, n)
            if n_improved > 200:
                break
        
        if n_improved == 0:
            # Try across all top lags
            c = compute_all_lags(h, n)
            top_lags = np.argsort(c)[-8:]
            any_imp = False
            for kl in top_lags:
                if time.time() > deadline:
                    break
                lag2 = kl - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in range(0, n - m2):
                    if time.time() > deadline:
                        break
                    j2 = i + m2
                    if lag2 > 0:
                        if h[j2] > h[i] + 0.05:
                            delta = min(0.002, (h[j2]-h[i])*0.1, h[j2]*0.05, (1-h[i])*0.05)
                        elif h[i] > h[j2] + 0.05:
                            delta = -min(0.002, (h[i]-h[j2])*0.1, h[i]*0.05, (1-h[j2])*0.05)
                        else:
                            continue
                    else:
                        if h[i] > h[j2] + 0.05:
                            delta = min(0.002, (h[i]-h[j2])*0.1, h[i]*0.05, (1-h[j2])*0.05)
                        elif h[j2] > h[i] + 0.05:
                            delta = -min(0.002, (h[j2]-h[i])*0.1, h[j2]*0.05, (1-h[i])*0.05)
                        else:
                            continue
                    if abs(delta) < 1e-8:
                        continue
                    old_i, old_j = h[i], h[j2]
                    h[i] += delta
                    h[j2] -= delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        n_improved += 1
                        any_imp = True
                    else:
                        h[i], h[j2] = old_i, old_j
                        h = project_feasible(h, n)
            if not any_imp:
                break
    
    return best_h, best_c

def annealed_local_search(h, n, deadline, rng, max_iter=50000):
    """Simulated annealing on pair swaps for the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = compute_c5(h, n)
    
    T0 = 0.002
    Tf = 0.0001
    n_iter = 0
    while time.time() < deadline and n_iter < max_iter:
        n_iter += 1
        T = T0 * (Tf / T0) ** (n_iter / max_iter)
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        i = rng.randint(0, n - m)
        j = i + m
        if lag > 0:
            if h[j] > h[i]:
                delta = rng.uniform(0.001, 0.005)
            elif h[i] > h[j]:
                delta = -rng.uniform(0.001, 0.005)
            else:
                delta = rng.uniform(-0.002, 0.002)
        else:
            if h[i] > h[j]:
                delta = rng.uniform(0.001, 0.005)
            elif h[j] > h[i]:
                delta = -rng.uniform(0.001, 0.005)
            else:
                delta = rng.uniform(-0.002, 0.002)
        
        delta = max(-0.01, min(0.01, delta))
        if abs(delta) < 1e-8:
            continue
        delta = min(delta, h[i], 1 - h[j])
        delta = max(delta, -h[j], h[i] - 1)
        if abs(delta) < 1e-8:
            continue
        
        old_i, old_j = h[i], h[j]
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        delta_c = c_new - c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
            c = c_new
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
            h = project_feasible(h, n)
    
    return best_h, best_c

def slsqp_objective(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ct = c[top_idx]
    return smooth_lse(ct, alpha)

def slsqp_grad(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ct = c[top_idx]
    w = smooth_lse_grad_weights(ct, alpha)
    g = np.zeros(n)
    for ii in range(len(top_idx)):
        if w[ii] < 1e-10:
            continue
        lag = top_idx[ii] - (n - 1)
        g += w[ii] * lag_gradient(x, n, lag)
    return g * dx

def slsqp_polish(h, n, alpha, maxiter, top_k):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    bounds = [(0.0, 1.0)] * n
    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(slsqp_objective, h, args=(n, alpha, top_idx),
                      jac=lambda x: slsqp_grad(x, n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def resolve_nudge(h, n, deadline, rng):
    """Try to improve by nudging individual coordinates."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Focus on coordinates where h is "fractional" (not near 0 or 1)
    frac_mask = (h > 0.05) & (h < 0.95)
    frac_idx = np.where(frac_mask)[0]
    
    for _ in range(200):
        if time.time() > deadline:
            break
        if len(frac_idx) == 0:
            break
        i = frac_idx[rng.randint(0, len(frac_idx))]
        c = compute_all_lags(h, n)
        # Gradient at position i
        g_i = 0
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            if lag == 0:
                g_i += (1 - 2 * h[i])
            elif abs(lag) < n:
                m = abs(lag)
                if lag > 0:
                    if i >= m:
                        g_i += (1 - h[i - m])
                    if i < n - m:
                        g_i += (-h[i + m])
                else:
                    if i < n - m:
                        g_i += (1 - h[i + m])
                    if i >= m:
                        g_i += (-h[i - m])
        # Try to move h[i] in direction -sign(g_i)
        direction = -1 if g_i > 0 else 1
        for step in [0.01, 0.005, 0.002, 0.001]:
            delta = direction * step
            new_val = h[i] + delta
            if new_val < 0 or new_val > 1:
                continue
            old_i = h[i]
            h[i] = new_val
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                break
            else:
                h[i] = old_i
                h = project_feasible(h, n)
        # Update frac mask occasionally
        if _ % 20 == 0:
            frac_mask = (h > 0.03) & (h < 0.97)
            frac_idx = np.where(frac_mask)[0]
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

            best_c = c_an
            best_h = h_an

    if time.time() < p3_end:
        for top_k in [3, 5]:
            if time.time() > p3_end - 5:

            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_k:]
            h_s = slsqp_polish(best_h, n, 500000, maxiter=3000, top_k=top_k)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 4: Multi-resolution attempt (15% budget)
    p4_end = t0 + 0.95 * (budget_s - 5)
    if time.time() < p4_end:
        x_best = np.linspace(0, 2, n)
        for n_try in [n - 10, n + 10, n - 20, n + 20]:
            if time.time() > p4_end - 10:
                break
            if n_try < 100:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
            h_start = project_feasible(h_start, n_try)
            alloc = min(30, (p4_end - time.time()) * 0.3)
            if alloc < 10:

            # Quick subgradient

                                      top_k=3, alpha=5000, lr0=0.2, decay=0.99997, momentum=0.93)
            # Quick local search
            h_r, c_r = local_search_fast(h_r, n_try, time.time() + alloc * 0.25, max_rounds=50)
            if c_r < best_c:
                best_c = c_r
                # Interpolate back to original n
                x_r = np.linspace(0, 2, n_try)
                best_h = np.interp(x_best, x_r, h_r)
                best_h = project_feasible(best_h, n)
    
    # Phase 5: Final fine-tuning (5% budget)
    if time.time() < deadline:
        # Coordinate nudge
        h_n, c_n = resolve_nudge(best_h, n, deadline - 2, rng)
        if c_n < best_c:
            best_c = c_n
            best_h = h_n
        
        # Tiny local search
        h_f, c_f = local_search_fast(best_h, n, deadline - 0.5, max_rounds=50)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    # Final validation
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1 + 1e-15)
    assert abs(best_h.sum() - n / 2.0) < 1e-8
    best_h = np.clip(best_h, 0, 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
