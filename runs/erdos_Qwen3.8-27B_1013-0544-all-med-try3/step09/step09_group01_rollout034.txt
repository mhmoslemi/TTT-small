                            improved = True
                        else:

                            h[j] += d

                        h[i] -= d
                        h[j] += d

def triple_search(h, n, deadline, radius=30):

    best_c = compute_c5(h, n)

        k = np.random.randint(n)

        if j < lo or j >= hi or k < lo or k >= hi:
            continue
        orig = (h[i], h[j], h[k])
        h[i] += d
        h[j] -= d / 2
        h[k] -= d / 2
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1 and 0 <= h[k] <= 1:
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
                continue
        h[i], h[j], h[k] = orig
    return best_h, best_c

def binary_start(n, rng):

def block_start(n, block_len=7):
    h = np.zeros(n)
    pattern_pos = [0, 1, 3]
    n_blocks = n // block_len
    remainder = n % block_len
    for b in range(n_blocks):
        base = b * block_len
        for p in pattern_pos:
            h[base + p] = 1.0
    for r in range(remainder):
        h[n_blocks * block_len + r] = 1.0 if r < 3 else 0.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return project_feasible(h, n)
