```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
```

**Approach: Projected subgradient with momentum**

Since we want to minimize f(h) = max_k O(k) (where O(k) are the overlap values), we can use:
1. Compute the current worst lag k*
2. Compute the subgradient: ∇O(k*)/∂h
3. Take a step in the negative subgradient direction
4. Project onto the feasible set

The subgradient of O(k) = Σ_i h[i](1-h[i+k]) * dx with respect to h is:
For lag m (where m > 0):
O(m) = Σ_{i=0}^{n-1-m} h[i](1-h[i+m]) * dx

∂O(m)/∂h[i] = (1-h[i+m])*dx for i < n-m (from the first factor)
            + (-h[i-m])*dx for i ≥ m (from the second factor, where h[i] appears as h[i-m+m])
            So for i in [m, n-1-m]: ∂O(m)/∂h[i] = (1-h[i+m])*dx - h[i-m]*dx
For i < m: ∂O(m)/∂h[i] = (1-h[i+m])*dx
For i > n-1-m: ∂O(m)/∂h[i] = -h[i-m]*dx

This is O(n) to compute for a single lag, so it's efficient.

With momentum (Nesterov acceleration):
v_{t+1} = β * v_t - lr * g_t
h_{t+1} = project(h_t + v_{t+1})

where g_t is the subgradient at the worst lag.

```python
def subgradient_optimize(h, n, deadline, lr=0.01, momentum=0.9, top_k=5):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    t = 0
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        
        # Weighted subgradient from top worst lags
        weights = c_all[top_idx] - c_all[top_idx[0]] + 1  # emphasize worst
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            m = k - (n - 1)
            if m == 0:
                g += w * (1 - 2*h) * dx
            elif m > 0:
                g[:n-m] += w * (1 - h[m:]) * dx
                g[m:] += w * (-h[:-m]) * dx
            else:
                im = -m
                g[:n-im] += w * (1 - h[im:]) * dx
                g[im:] += w * (-h[:n-im]) * dx
        
        # Nesterov step
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        t += 1
        lr = 0.01 / (1 + 0.01 * t)  # adaptive learning rate
    
    return best_h, best_c
```

```python
import numpy as np
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()


def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx


def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    penalty_coeff = 1e6

    def obj_and_grad(x):
        xc = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(xc, n)
        obj_val = smooth_max_value(c, alpha)
        diff = xc.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(xc, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def binary_anneal(h, n, deadline, rng, t_init=0.3, t_min=0.01, cool=0.999):

        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c_new = compute_c5(h_bin, n)
        c_old = best_c_bin if np.array_equal(h_bin, best_bin) else compute_c5(best_bin, n)

        if c_new < c_old:
            best_bin = h_bin.copy()
            best_c_bin = c_new
        else:
            de = c_new - c_old
            if de / T < 50 and rng.random() < np.exp(-de / T):
                pass  # accept
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]

        T *= cool

    # Relax the best binary solution
    h_relaxed = best_bin.astype(float)
    return h_relaxed


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B at parent resolution with alpha sweep
    n = n_init
    alpha_schedule = [2000, 10000, 50000, 200000, 1000000]
    for alpha in alpha_schedule:
        remaining = deadline - time.time() - 150
        if remaining < 10:
            break
        t_limit = time.time() + min(25, remaining * 0.15)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t_limit, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Aggressive targeted local search at parent resolution
    if time.time() < deadline - 100:
        ls_deadline = min(deadline - 80, time.time() + 120)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, rng,
                                             top_k=10, step_base=0.004)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-lag local search (more systematic)
    if time.time() < deadline - 80:
        ml_deadline = min(deadline - 60, time.time() + 80)
        h_opt, c_opt = multi_lag_local_search(best_h, n, ml_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish after local search
    if time.time() < deadline - 60:
        for alpha in [500000, 5000000]:
            if time.time() > deadline - 40:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 20, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Try alternative resolutions
    alt_resolutions = []
    for dr in [-40, -20, 20, 40, 60, 80]:
        nr = n_init + dr
        if 200 <= nr <= 1200:
            alt_resolutions.append(nr)

    for nr in sorted(alt_resolutions, key=lambda x: abs(x - n_init)):
        remaining = deadline - time.time() - 30
        if remaining < 20:
            break
        if remaining < 40:
            # Quick pass
            if time.time() > deadline - 10:
                break
            h_try = interpolate_resolution(best_h, len(best_h), nr)
            c_try = compute_c5(h_try, nr)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
            continue

        h_try = interpolate_resolution(best_h, len(best_h), nr)
        c_try = compute_c5(h_try, nr)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

        # Optimize at this resolution
        for alpha in [10000, 100000]:
            if time.time() > deadline - 25:
                break
            h_opt = lbfgsb_optimize(h_try, nr, alpha, time.time() + min(15, remaining * 0.25),
                                    maxiter=1500)
            c_opt = compute_c5(h_opt, nr)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h_try = h_opt

        # Quick local search
        if time.time() < deadline - 15:
            h_opt, c_opt = targeted_local_search(h_try, nr, deadline - 10, rng,
                                                 top_k=6, step_base=0.003)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Binary annealing as diversification
    if time.time() < deadline - 50:
        h_bin = binary_anneal(best_h, len(best_h), time.time() + 30, rng)
        # Relax the binary solution
        h_relaxed = lbfgsb_optimize(h_bin, len(best_h), 100000, time.time() + 15, maxiter=2000)
        c_relaxed = compute_c5(h_relaxed, len(best_h))
        if c_relaxed < best_c:
            best_c = c_relaxed
            best_h = h_relaxed

    # Phase 7: Final intensive local search
    if time.time() < deadline - 8:
        n_final = len(best_h)
        h_opt, c_opt = targeted_local_search(best_h, n_final, deadline - 3, rng,
                                             top_k=6, step_base=0.002)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 10000000, deadline - 1, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
