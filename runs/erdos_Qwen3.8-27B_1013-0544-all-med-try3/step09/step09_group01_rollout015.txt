from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    logsum = mx + np.log(np.sum(np.exp(sh - mx)))
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nc = len(c)  # 2n-1
    center = n - 1
    grad = np.zeros(n)
    for mm in range(nc):
        if w[mm] < 1e-15:
            continue
        k = mm - center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def tops_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    logsum = smx + np.log(np.sum(np.exp(sh - smx)))
    val = mx + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def tops_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:

        k = idx - center
        wm = wi * dx
        if k >= 0:

            if nv > 0:

            nv = n - kk

def binary_local_search(h_bin, n, deadline, max_rounds=50):
    """Local search on binary sequence via swaps."""
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try random swaps
        n_swaps = min(len(ones) * 3, 5000)
        for _ in range(n_swaps):
            if time.time() > deadline:
                break
            i = min(ones) if ones else 0
            j = min(zeros) if zeros else 0
            # Random selection
            i = list(ones)[np.random.randint(len(ones))] if ones else 0
            j = list(zeros)[np.random.randint(len(zeros))] if zeros else 0

            h[j] = 1
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
                improved = True
            else:
                h[i] = 1
                h[j] = 0
        if not improved:

            T = 0.001

def compute_c5_vec(h, n):

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0

                if c6b < best_c:
                    best_c, best_h = c6b, h6b
    
    # === Phase 7: Try different n values ===
    for n_try in [600, 800]:
        if time.time() > deadline - 100:
            break
        x_c = np.linspace(0, 2, best_n)
        x_t = np.linspace(0, 2, n_try)
        h_t = np.interp(x_t, x_c, best_h)
        h_t = project_feasible(h_t, n_try)
        c_t = compute_c5(h_t, n_try)
        if c_t < best_c + 0.01:
            p7_dl = min(time.time() + 60, deadline - 80)
            h7, c7 = run_lbfgs(h_t, n_try, p7_dl,
                alphas=[500, 5000, 50000, 200000],
                mu=0.005, maxit_per=[2000, 3000, 3000, 3000])
            if c7 < best_c:
                best_c, best_h, best_n = c7, h7, n_try
    
    # === Phase 8: Final top-k polish ===
    if time.time() < deadline - 30:
        p8_dl = deadline - 5
        h8, c8 = run_topk_rounds(best_h, best_n, p8_dl, top_k=4,
            alphas=[500000, 2000000, 10000000], max_rounds=5)
        if c8 < best_c:
            best_c, best_h = c8, h8
    
    # === Phase 9: Nesterov final ===
    if time.time() < deadline - 5:
        p9_dl = deadline - 1
        h9, c9 = nesterov_max(sm_obj, sm_grad, best_h, best_n,
            [(0.0, 1.0)] * best_n, p9_dl, step0=0.003, max_iter=5000)
        if c9 < best_c:
            best_c, best_h = c9, h9
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
