                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.0002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) < 1:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 5e-4 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        h = h_new
    
    return best_h, best_c

def pair_search_fast(h, n, deadline, radius=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            n_active = 0
            for i in range(n):
                if time.time() > deadline:
                    break
                if h[i] < 1e-6 or h[i] > 1 - 1e-6:
                    continue
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[j] < 1e-6 or h[j] > 1 - 1e-6:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha, maxiter):
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    
    def sq_jac(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2)
    return project_feasible(h_new, n_new)

def random_feasible(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n)

def binary_start(n, rng, period_ratio=None):
    h = np.zeros(n)
    if period_ratio is not None:
        period = max(2, int(n * period_ratio))
        ones = max(1, period // 2)
        for p in range(ones):
            h[p::period] = 1.0
    else:
        n_ones = n // 2
        idx = rng.choice(n, size=n_ones, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        best_c = c1a
        best_h = h1a
        best_n = n_start
    
    # Try complement start with remaining time
    remaining = deadline - time.time()
    if remaining > 50:
        h1b, c1b = lbfgs_optimize(h0_complement, n_start, time.time() + min(80, remaining * 0.4),
                                   [100, 1000, 10000, 100000, 500000],
                                   0.005, [1500, 2000, 2500, 2500, 3000])
        if c1b < best_c:
            best_c = c1b
            best_h = h1b
            best_n = n_start
    
    # Phase 2: Upsample to 2*n_start (if n_start <= 1000)
    n2 = min(2 * n_start, 2000)
    if n2 > best_n and time.time() < deadline - 300:
        h2_init = upsample(best_h, best_n, n2)
        p2_end = time.time() + min(250, (deadline - time.time()) * 0.5)
        alphas2 = [10, 50, 200, 1000, 5000, 20000, 100000]
        maxit2 = [1000, 1500, 2000, 2500, 2500, 3000, 3000]
        h2, c2 = lbfgs_optimize(h2_init, n2, p2_end, alphas2, 0.003, maxit2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2
    
    # Phase 3: Active set optimization at best_n
    if time.time() < deadline - 200:
        p3_end = time.time() + min(150, (deadline - time.time()) * 0.5)
        alphas3 = [5000, 20000, 100000, 500000, 2000000, 5000000]
        h3, c3 = active_set_optimize(best_h, best_n, p3_end, top_k=5, alphas=alphas3)

    if best_n < n4 and time.time() < deadline - 200:
        h4_init = upsample(best_h, best_n, n4)
        p4_end = time.time() + min(180, (deadline - time.time()) * 0.55)
        alphas4 = [10, 50, 200, 1000, 5000, 20000, 80000]

        h7, c7 = pair_search_fast(best_h, best_n, p7_end, radius=15)

            best_c = c7
            best_h = h7
            best_n = best_n
    
    # Phase 8: SLSQP polish
    if time.time() < deadline - 20:
        for alpha_sl in [200000, 1000000]:
            if time.time() > deadline - 12:
                break
            p8_end = time.time() + min(15, deadline - time.time() - 3)
            h8, c8 = slqp_polish(best_h, best_n, p8_end, alpha_sl, maxiter=800)
            if c8 < best_c:
                best_c = c8
                best_h = h8
                best_n = best_n
    
    # Phase 9: Final subgradient + pair
    if time.time() < deadline - 8:
        h9, c9 = subgrad_refine(best_h, best_n, deadline - 5, n_iter=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = best_n
    
    # Final validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
