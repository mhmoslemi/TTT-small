import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    val = smooth_max_val(c, alpha)

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-16:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2

def topK_grad(x, n, active, alpha, mu):

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_sturmian_start(n, slope):
    h = np.zeros(n)
    x = 0.0
    n_ones = n // 2
    count = 0
    for i in range(n):
        x += slope
        if x >= 1.0:
            x -= 1.0
            h[i] = 1.0
            count += 1
    remaining = n_ones - h.sum()
    if remaining > 0:
        zeros = np.where(h == 0)[0]
        for j in range(int(remaining)):
            if j < len(zeros):
                h[zeros[j]] = 1.0
    elif remaining < 0:
        ones = np.where(h == 1)[0]
        for j in range(int(-remaining)):
            if j < len(ones):
                h[ones[j]] = 0.0
    return project_feasible(h, n)

def exact_local_search(h, n, deadline, max_rounds=50000):

            break
        improved = True
        while improved and time.time() < deadline and rounds_done < max_rounds:
            improved = False
            rounds_done += 1
            c_vec = compute_c5_vec(best_h, n)

            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx
                    grad[k:n] += (-best_h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                    grad[0:nv] += (-best_h[kk:n]) * dx
            grad /= max(1, len(active))
            pos_idx = np.where((grad > 1e-10) & (best_h > step))[0]

            best_gain = 0

    cur_h, cur_c = phase_lbfgs(cur_h, n_orig, p1_end,
        alphas=[10, 50, 200, 1000, 5000, 50000, 200000, 1000000],
        mu=0.01, maxit_list=[500, 800, 1200, 1500, 1500, 1500, 1500, 1500])
    if cur_c < best_c:
        best_c = cur_c
        best_h = cur_h

    # === Phase 2: Active set optimization (120-320s) ===
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 180, deadline - 350)
        cur_h, cur_c = phase_active_set(cur_h, n_orig, p2_end,
            top_k=4, alphas=[200000, 1000000, 5000000, 20000000, 100000000, 500000000],
            rounds=25)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h

    # === Phase 3: Exact local search (320-480s) ===
    if time.time() < deadline - 280:
        p3_end = min(time.time() + 140, deadline - 250)
        cur_h, cur_c = exact_local_search(cur_h, n_orig, p3_end, max_rounds=30000)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h

    # === Phase 4: Active set again with different params (480-580s) ===
    if time.time() < deadline - 220:
        p4_end = min(time.time() + 80, deadline - 200)
        cur_h, cur_c = phase_active_set(cur_h, n_orig, p4_end,
            top_k=3, alphas=[100000000, 500000000, 2000000000],
            rounds=10)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h

    # === Phase 5: Swap local search (580-680s) ===
    if time.time() < deadline - 160:
        p5_end = min(time.time() + 80, deadline - 140)
        cur_h, cur_c = swap_local_search(cur_h, n_orig, p5_end, max_rounds=15000)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h

    # === Phase 6: Try at n=1000 (680-850s) ===
    n_high = 1000
    if time.time() < deadline - 180:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        p6_end = min(time.time() + 100, deadline - 160)
        h_high, c_high = phase_lbfgs(h_high, n_high, p6_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.01, maxit_list=[500, 800, 1000, 1000, 1000])
        if time.time() < deadline - 80:
            p6b_end = min(time.time() + 60, deadline - 60)
            h_high, c_high = phase_active_set(h_high, n_high, p6b_end,
                top_k=3, alphas=[5000000, 50000000, 200000000],
                rounds=10)
        if time.time() < deadline - 40:
            p6c_end = min(time.time() + 30, deadline - 30)
            h_high, c_high = exact_local_search(h_high, n_high, p6c_end, max_rounds=10000)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high

    # === Phase 7: Fine pair search at best resolution (850-980s) ===
    if time.time() < deadline - 80:
        p7_end = min(time.time() + 100, deadline - 60)
        best_h, best_c = fine_pair_search(best_h, best_n, p7_end,
            radius=35, deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if best_n > 600:
            n_final = n_final

    # === Phase 8: SLSQP polish (980-1060s) ===
    if time.time() < deadline - 50:
        for alpha in [5000000, 50000000, 200000000]:
            if time.time() > deadline - 10:
                break
            h_slq, c_slq = run_slqp(best_h, best_n, deadline - 5, alpha=alpha, maxiter=500)
            if c_slq < best_c:
                best_c = c_slq
                best_h = h_slq

    # === Phase 9: Final fine adjustments (1060s - deadline) ===
    if time.time() < deadline - 3:
        best_h, best_c = fine_pair_search(best_h, best_n, deadline - 2,
            radius=20, deltas=[0.00005, 0.00002, 0.00001, 0.000005])

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
