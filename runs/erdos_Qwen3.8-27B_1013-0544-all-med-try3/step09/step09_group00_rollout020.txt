            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad_full(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.005, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active_set(h, n, deadline, top_k=8, alphas=None, rounds=30):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Use a wider threshold to capture near-active constraints
        threshold = max_val - max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 12:
                break
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, 0.0005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in step_sizes:
        stall = 0
        while time.time() < deadline and steps_done < max_steps and stall < 5000:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            
            # Find best positive and negative gradient positions
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = -1
            best_i = -1
            best_j = -1
            # Check top candidates by gradient magnitude
            pos_sorted = pos_idx[np.argsort(-grad[pos_idx])[:30]]
            neg_sorted = neg_idx[np.argsort(grad[neg_idx])[:30]]
            for i in pos_sorted:
                for j in neg_sorted:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-14:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            h_proj = project_feasible(best_h, n)
            c_new = compute_c5(h_proj, n)
            steps_done += 1
            if c_new < best_c:
                best_c = c_new
                best_h = h_proj
                stall = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stall += 1
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        stall = 0
        while improved and time.time() < deadline and stall < 200:
            improved = False
            stall = 0
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        h_proj = project_feasible(best_h, n)
                        c = compute_c5(h_proj, n)
                        if c < best_c:
                            best_c = c
                            best_h = h_proj
                            improved = True
                            stall = 0
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            stall += 1
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        h_proj = project_feasible(best_h, n)
                        c = compute_c5(h_proj, n)
                        if c < best_c:
                            best_c = c
                            best_h = h_proj
                            improved = True
                            stall = 0
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            stall += 1
                    if stall > 300:
                        break
    return best_h, best_c

def far_pair_search(h, n, deadline, deltas=None):
    """Try pair moves between distant positions (outside local radius)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001]
    rng = np.random.RandomState(12345)
    for d in deltas:
        stall = 0
        max_trials = 50000
        for trial in range(max_trials):
            if time.time() > deadline or stall > 5000:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try moving from i to j
            if best_h[i] >= d and best_h[j] <= 1.0 - d:
                best_h[i] -= d
                best_h[j] += d
                h_proj = project_feasible(best_h, n)
                c = compute_c5(h_proj, n)
                if c < best_c:
                    best_c = c
                    best_h = h_proj
                    stall = 0
                else:
                    best_h[i] += d
                    best_h[j] -= d
                    stall += 1
            # Try moving from j to i
            if best_h[j] >= d and best_h[i] <= 1.0 - d:
                best_h[j] -= d
                best_h[i] += d
                h_proj = project_feasible(best_h, n)
                c = compute_c5(h_proj, n)
                if c < best_c:
                    best_c = c
                    best_h = h_proj
                    stall = 0
                else:
                    best_h[j] += d
                    best_h[i] -= d
                    stall += 1
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    """Nearest-neighbor upsample."""
    if n_old == n_new:
        return h.copy()
    indices = (np.arange(n_new) * n_old) // n_new
    h_new = h[indices]
    # Fix the sum constraint
    h_new = project_feasible(h_new, n_new)
    return h_new

def perturbation_restart(h, n, deadline, rng, num_restarts=5, noise_scale=0.02):
    """Perturb the solution and re-optimize to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(num_restarts):
        if time.time() > deadline - 30:
            break
        # Add random perturbation
        perturbed = best_h + rng.randn(n) * noise_scale
        perturbed = project_feasible(perturbed, n)
        # Quick L-BFGS from perturbed point
        rem = deadline - time.time() - 30
        if rem < 20:
            break
        try:
            res = minimize(obj_full, perturbed, jac=grad_full,
                          args=(n, 50000, 0.001), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max(100, int(rem * 2)),
                                   'ftol': 1e-20, 'gtol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def multi_active_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (4, 50000, 4),
        (6, 200000, 4),
        (8, 1000000, 3),
        (12, 5000000, 3),
        (20, 20000000, 2),
    ]
    for top_k, alpha, rounds in configs:
        if time.time() > deadline - 15:
            break
        for rnd in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            maxit = max(150, int((deadline - time.time()) * 1.2))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, 0.0003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    # Get parent solution
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Generate alternative starts at n_parent
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty_start(n_parent, golden / 2.0),
        make_beatty_start(n_parent, np.sqrt(2.0) / 2.0),
        make_beatty_start(n_parent, 1.0 / np.e),
    ]
    for hs in alt_starts:
        c = compute_c5(hs, n_parent)
        if c < best_c:
            best_c = c
            best_h = hs.copy()

    # ===== Phase 1: L-BFGS warmup at parent resolution =====
    p1_end = t0 + 120
    h1, c1 = phase_lbfgs(best_h, n_parent, p1_end,
        alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.005, maxit_list=[500, 800, 1000, 1200, 1500, 1800, 2000, 2500])
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = n_parent

    # ===== Phase 2: Active set at parent resolution =====
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 150, deadline - 300)
        h2, c2 = multi_active_optimize(best_h, n_parent, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_parent

    # ===== Phase 3: Gradient-guided search at parent resolution =====
    if time.time() < deadline - 280:
        p3_end = min(time.time() + 120, deadline - 250)
        h3, c3 = grad_guided_search(best_h, n_parent, p3_end,
            step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=150000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_parent

    # ===== Phase 4: Fine pair search at parent resolution =====
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 180)
        h4, c4 = fine_pair_search(best_h, n_parent, p4_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_parent

    # ===== Phase 5: Try higher resolution n=1000 =====
    n_hi = 1000
    h_hi = upsample(best_h, best_n, n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = phase_lbfgs(h_hi, n_hi, p5_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.003, maxit_list=[300, 500, 800, 1000, 1200])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_hi

    # ===== Phase 6: Active set at best resolution =====
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 50, deadline - 60)
        h6, c6 = multi_active_optimize(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # ===== Phase 7: Far pair search =====
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 35, deadline - 40)
        h7, c7 = far_pair_search(best_h, best_n, p7_end,
            deltas=[0.0005, 0.0002, 0.0001])
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n

    # ===== Phase 8: Perturbation restart =====
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 25, deadline - 20)
        h8, c8 = perturbation_restart(best_h, best_n, p8_end, rng,
            num_restarts=3, noise_scale=0.01)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n

    # ===== Phase 9: Final fine local search =====
    if time.time() < deadline - 5:
        p9_end = min(time.time() + 20, deadline - 3)
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001])
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = best_n

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c, final_n
