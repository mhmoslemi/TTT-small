        val = smooth_max_val(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    
    def grad(x, n, alpha, mu):
        c = compute_c5_vec(x, n)
        w = smooth_max_grad(c, alpha)
        dx_v = 2.0 / n
        g = np.zeros(n)
        for wi, idx in zip(w, range(len(c))):
            if wi < 1e-14:
                continue
            g += wi * grad_c_k(x, n, idx)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(200, int(remaining / len(alphas) * 3))
        try:
            res = minimize(obj, best_h, jac=grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-15:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c

def active_set_phase(h, n, deadline, configs):
    """Iterate: identify active set, optimize with L-BFGS on active set."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for top_k, alpha, rounds in configs:
        for rnd in range(rounds):
            remaining = deadline - time.time()
            if remaining < 20:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            
            maxit = max(200, int(remaining / max(1, top_k)))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, 0.001),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                break
    return best_h, best_c

def fast_grad_search(h, n, deadline, step_sizes):
    """Gradient-guided pair search using active lags only."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for step in step_sizes:
        steps = 0
        while time.time() < deadline:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            # Compute average gradient of active lags
            avg_grad = np.zeros(n)
            for idx in active:
                avg_grad += grad_c_k(best_h, n, idx)
            avg_grad /= len(active)
            
            # Find best pair: decrease h[i] and increase h[j]
            # where avg_grad[i] > 0 and avg_grad[j] < 0
            pos_mask = (avg_grad > 1e-8) & (best_h > 1e-8)
            neg_mask = (avg_grad < -1e-8) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            # Only check top 50 candidates each way
            for i in pos_idx[:50]:
                gi = avg_grad[i]
                for j in neg_idx[:50]:
                    g = gi - avg_grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0 or best_gain < 1e-10:
                break
            
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            
            # Only check active lags for speed
            c_act = compute_c5_active(best_h, n, active)
            new_max = c_act.max()
            
            if new_max < best_c - 1e-16:
                # Verify with full C5 occasionally
                if steps % 100 == 0:
                    c_full = compute_c5(best_h, n)
                    if c_full < best_c - 1e-16:
                        best_c = c_full
                        steps += 1
                    else:
                        best_h[best_i] += amt
                        best_h[best_j] -= amt
                        best_h = project_feasible(best_h, n)
                else:
                    steps += 1
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                steps += 1
            
            if steps > 500000:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = np.clip(best_h, 0, 1)
                        best_h = project_feasible(best_h, n)
                        c_act = compute_c5_active(best_h, n, active)
                        if c_act.max() < best_c - 1e-14:
                            c_full = compute_c5(best_h, n)
                            if c_full < best_c - 1e-14:
                                best_c = c_full
                                improved = True
                                continue
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = np.clip(best_h, 0, 1)
                        best_h = project_feasible(best_h, n)
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = np.clip(best_h, 0, 1)
                        best_h = project_feasible(best_h, n)
                        c_act = compute_c5_active(best_h, n, active)
                        if c_act.max() < best_c - 1e-14:
                            c_full = compute_c5(best_h, n)
                            if c_full < best_c - 1e-14:
                                best_c = c_full
                                improved = True
                                continue
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = np.clip(best_h, 0, 1)
                        best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = max(0, min(1, remaining))
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    # Actually, let's not use periodic interpolation, just linear
    h_interp = np.interp(x_new, x_old + 2.0, np.concatenate([h_old, h_old[:1]]))
    h_interp = np.clip(h_interp, 0, 1)
    return project_feasible(h_interp, n_new)

def slqp_polish(h, n, deadline, alpha, maxiter):
    def obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        return smooth_max_val(c, alpha)
    
    def jac(x, n, alpha):
        c = compute_c5_vec(x, n)
        w = smooth_max_grad(c, alpha)
        g = np.zeros(n)
        for wi, idx in zip(w, range(len(c))):
            if wi < 1e-14:
                continue
            g += wi * grad_c_k(x, n, idx)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    
    try:
        rem = max(10, int(deadline - time.time()))
        maxit = min(maxiter, max(100, rem * 5))
        res = minimize(obj, h, jac=jac, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
