from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, frac=0.005, max_k=20):
    c = compute_c5_vec(h, n)
    threshold = c.max() - frac * c.max()
    active = np.where(c >= threshold)[0]
    if len(active) > max_k:
        order = np.argsort(-c[active])
        active = active[order[:max_k]]
    return active, c.max()

def smooth_obj(x, n, active, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n/2.0
    val += mu * s * s
    return float(val)

def smooth_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    offset = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:

        k = idx - offset
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n/2.0)

    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_list:

            break

        active, _ = get_active_shifts(h, n, frac=0.01, max_k=15)
        
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, active, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new
        except Exception:
            pass

def compute_c5(h, n):

def compute_c5_vec(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, frac, max_k):
    c = compute_c5_vec(h, n)
    threshold = c.max() - frac * c.max()
    active = np.where(c >= threshold)[0]
    if len(active) > max_k:
        order = np.argsort(-c[active])
        active = active[order[:max_k]]
    return active, float(c.max())

def smooth_obj(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smooth_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    offset = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-16:
            continue
        k = idx - offset
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Load initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Initial optimization at parent resolution (120s)
    t1 = min(t0 + 120, deadline - 600)
    if time.time() < t1:
        h, best_c = iterative_active_optimize(h, n, t1, max_rounds=10)
        best_h = h.copy()
    
    # Phase 2: Multi-perturbation to escape local optima (100s)
    t2 = min(t0 + 220, deadline - 400)
    if time.time() < t2 and best_h is not None:
        h2, c2 = multi_perturbation(best_h, n, t2, n_pert=4)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: More aggressive active-set optimization (250s)
    t3 = min(t0 + 470, deadline - 250)
    if time.time() < t3:
        h3, c3 = iterative_active_optimize(best_h, n, t3, max_rounds=25)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try higher resolution (150s)
    t4 = min(t0 + 620, deadline - 150)
    if time.time() < t4:
        best_h, best_c, n = try_higher_res(best_h, n, best_c, t4)
    
    # Phase 5: Local pair search for fine tuning (200s)
    t5 = min(t0 + 820, deadline - 80)
    if time.time() < t5:
        h5, c5 = pair_search_fine(best_h, len(best_h), t5, radius=20,
                                   max_steps=[0.003, 0.001, 0.0005, 0.0002])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Final polish (80s)
    t6 = deadline - 5
    if time.time() < t6:
        best_h, best_c = final_polish(best_h, len(best_h), t6)
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
