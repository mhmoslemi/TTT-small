    if not active.any():

    w[active] = 1.0 / active.sum()

            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

Then the change for swap (a, b) is: dx * (cost[b][k] - cost[a][k]) for each k.

The maximum c_k after the swap is: max_k (c_k + dx*(cost[b][k] - cost[a][k]))

If I precompute cost[p][k] for all p and k, that's O(n²) memory and O(n²) time. For n=500, that's 250K entries, which is fine.

Then for each swap (a, b), the new C5 is max_k (c_k + dx*(cost[b][k] - cost[a][k])). Finding the best swap is O(n² * n) = O(n³) in the worst case, which for n=500 is 125M operations. That might be too slow in Python.

Better approach: for each position b (0→1), find the position a (1→0) that maximizes the reduction in the worst c_k. This is still O(n²) per b, giving O(n³) total.

        # c_k = dx * sum_{valid i} h[i]*(1-h[i+k])
        # If we flip a from 1->0: c_k decreases by dx*(1-h[a+k]) if a+k valid
        # If we flip b from 0->1: c_k increases by dx*(1-h[b+k]) if b+k valid
        
        # We want to minimize c_k after swap:
        # New c_k = c_k - dx*(1-h[a+k]) + dx*(1-h[b+k])
        # = c_k + dx*(h[a+k] - h[b+k])  (for the worst shift)
        # To minimize: maximize h[b+k] - h[a+k], i.e., want h[b+k]=1 and h[a+k]=0
        
        # But we also need to check other shifts don't increase above c_max
        
        dx = 2.0 / n
        best_reduction = 0
        best_a = -1
        best_b = -1
        
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        # For the worst shift k_worst:
        if k_worst >= 0:
            # c_k = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # Flip a (1->0): reduction = dx*(1-h[a+k]) if a <= n-1-k
            # Flip b (0->1): increase = dx*(1-h[b+k]) if b <= n-1-k
            for a in ones:
                if a > n - 1 - k_worst:
                    continue
                h_ak = best_h[a + k_worst]
                red_a = (1 - h_ak)  # reduction from removing a
                for b in zeros:
                    if b > n - 1 - k_worst:
                        continue
                    h_bk = best_h[b + k_worst]
                    inc_b = (1 - h_bk)  # increase from adding b
                    net = red_a - inc_b
                    if net > best_reduction:
                        best_reduction = net
                        best_a = a
                        best_b = b
        # ... similar for k_worst < 0
        
        if best_a >= 0 and best_reduction > 0:
            # Apply swap and check if overall C5 improved
            best_h[best_a] = 0
            best_h[best_b] = 1
            new_c = compute_c5(best_h, n)
            if new_c < best_c:
                best_c = new_c
            else:
                # Revert
                best_h[best_a] = 1
                best_h[best_b] = 0
        else:
            break  # No improvement possible for this worst shift
    
    return best_h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, n, alpha, mu):
    """Smooth objective (log-sum-exp) and its gradient."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    s = exp_sh.sum()
    w = exp_sh / s
    val = m + (mx + np.log(s)) / alpha
    
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    # Only iterate over active weights
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
    
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    
    return float(val), grad

def lbfgs_run(h, n, deadline, alphas, mu=5000.0, maxiter_per_alpha=10000):
    """L-BFGS-B with increasing smoothness."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(obj_grad, best_h, jac=True,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxiter_per_alpha * 10, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_run(h, n, deadline, alpha=200000, maxiter=5000):
    """SLSQP with equality constraint."""
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    
    def obj_g(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        exp_sh = np.exp(sh - mx)
        s = exp_sh.sum()
        w = exp_sh / s
        dx = 2.0 / n
        grad = np.zeros(n)
        center = n - 1
        active = np.where(w > 1e-12)[0]
        for idx in active:
            k = idx - center
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[:nv] += wm * (1.0 - x[k:])
                    grad[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:] += wm * (1.0 - x[:nv])
                    grad[:nv] += wm * (-x[kk:])
        return float(m + (mx + np.log(s)) / alpha), grad
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
