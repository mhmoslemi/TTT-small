import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        v = m + np.log(np.sum(np.exp(s))) / alpha
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - x[k:])
                g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    else:

    f += mu * (x.sum() - n / 2.0) ** 2

                g[:nv] += wm * (-x[kk:])

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

            tot = room.sum()
            if tot < 1e-13:

            if tot < 1e-13:
                break

    diff = target - h.sum()
    if abs(diff) > 1e-13:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = interior.sum()

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue

        if k >= 0:

            nv = n - kk

                g[kk:] += wm * (1.0 - x[:nv])
