import numpy as np
import time
import math
from scipy.optimize import minimize

try:
    from scipy.fft import next_fast_len as _next_fast_len_impl
except Exception:
    _next_fast_len_impl = None

_CACHE = {}
_STATE = {"deadline": 0.0}


def _nfl(m):
    m = int(m)
    if m < 1:
        return 1
    if _next_fast_len_impl is not None:
        try:
            v = int(_next_fast_len_impl(m))
            if v >= m:
                return v
        except Exception:
            pass
    p = 1
    while p < m:
        p <<= 1
    return p


def get_aux(n):
    if n in _CACHE:
        return _CACHE[n]
    P = _nfl(3 * n - 2)
    off = n - 1
    M = 2 * n - 1
    if n > 1:
        r = np.arange(1, n, dtype=int)
        cut = n - r
        idx_neg = off - r
    else:
        r = np.empty(0, dtype=int)
        cut = np.empty(0, dtype=int)
        idx_neg = np.empty(0, dtype=int)
    i = np.arange(n, dtype=int)
    p_start = off - i
    p_end1 = M - i
    aux = (P, off, M, r, cut, idx_neg, i, p_start, p_end1)
    _CACHE[n] = aux
    return aux


def full_values_and_total(x, n):
    P, off, M, r, cut, idx_neg, i, p_start, p_end1 = get_aux(n)
    xf = np.asarray(x, dtype=float)
    Fh = np.fft.rfft(xf, P)
    auto = np.fft.irfft(Fh * np.conjugate(Fh), P)[:n]
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(xf, out=pref[1:])
    total = pref[n]

    vals = np.empty(M, dtype=float)
    vals[off] = total - auto[0]
    if n > 1:
        br = auto[r]
        pc = pref[cut]
        vals[off + 1:] = pc - br
        vals[idx_neg] = (total - pc) - br
    return vals, total


def smooth_max_value(c, alpha):
    m = float(c.max())
    if not np.isfinite(m):
        return 1e6
    sh = alpha * (c - m)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return m
    val = m + (sm + math.log(s)) / alpha
    if not np.isfinite(val):
        return m
    return float(val)


def full_obj(x, n, alpha, mu):
    vals, total = full_values_and_total(x, n)
    c = (2.0 / float(n)) * vals
    if not np.all(np.isfinite(c)):
        return 1e6
    val = smooth_max_value(c, alpha)
    if mu > 0.0:
        diff = total - 0.5 * float(n)
        val += mu * diff * diff
    if not np.isfinite(val):
        return 1e6
    return float(val)


def full_grad(x, n, alpha, mu):
    P, off, M, r, cut, idx_neg, i, p_start, p_end1 = get_aux(n)
    xf = np.asarray(x, dtype=float)
    vals, total = full_values_and_total(xf, n)
    dx = 2.0 / float(n)
    c = dx * vals

    m = float(c.max())
    sh = alpha * (c - m)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        w = np.zeros(M, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
        if not np.all(np.isfinite(w)):
            w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
            ws = float(w.sum())
            if ws > 0.0:
                w /= ws

    wpref = np.empty(M + 1, dtype=float)
    wpref[0] = 0.0
    np.cumsum(w, out=wpref[1:])
    A = wpref[p_end1] - wpref[p_start]

    Fh = np.fft.rfft(xf, P)
    Fw = np.fft.rfft(w, P)
    convB = np.fft.irfft(Fh * Fw, P)
    B = convB[off:off + n]

    Fwr = np.fft.rfft(w[::-1], P)
    convC = np.fft.irfft(Fh * Fwr, P)
    C = convC[off:off + n]

    grad = dx * (A - B - C)
    if mu > 0.0:
        diff = total - 0.5 * float(n)
        grad += (2.0 * mu * diff)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def compute_c5_np(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    v = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(v) * dx)


def project_l2(h, n):
    target = 0.5 * float(n)
    x = np.asarray(h, dtype=float).copy()
    if x.size < n:
        pad = np.zeros(n - x.size, dtype=float)
        x = np.concatenate([x, pad])
    elif x.size > n:
        x = x[:n]
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    if n == 0:
        return x

    s = float(np.sum(x))
    if abs(s - target) > 1e-14:
        lo = float(np.min(x)) - 1.0
        hi = float(np.max(x)) + 1.0
        for _ in range(80):
            lam = 0.5 * (lo + hi)
            y = np.clip(x - lam, 0.0, 1.0)
            sy = float(np.sum(y))
            if sy > target:
                lo = lam
            else:
                hi = lam
        x = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(80):
        s = float(np.sum(x))
        diff = target - s
        if diff == 0.0 or abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - x
            max_room = float(room.max())
            if max_room <= 1e-15:
                break
            if diff <= max_room + 1e-14:
                j = int(np.argmax(room))
                x[j] = min(1.0, x[j] + diff)
            else:
                idx = np.where(room > 1e-15)[0]
                rs = float(room[idx].sum())
                if rs <= 0.0:
                    break
                x[idx] += diff * room[idx] / rs
                x = np.clip(x, 0.0, 1.0)
        else:
            mass = x
            min_mass = float(mass.min())
            if (-diff) <= min_mass + 1e-14:
                j = int(np.argmin(mass))
                x[j] = max(0.0, x[j] + diff)
            else:
                idx = np.where(mass > 1e-15)[0]
                ms = float(mass[idx].sum())
                if ms <= 0.0:
                    break
                x[idx] += diff * mass[idx] / ms
                x = np.clip(x, 0.0, 1.0)

    for _ in range(30):
        s = float(np.sum(x))
        diff = target - s
        if diff == 0.0:
            break
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            j = int(np.argmax(1.0 - x))
            if 1.0 - x[j] <= 1e-15:
                break
            x[j] = min(1.0, x[j] + diff)
        else:
            j = int(np.argmin(x))
            if x[j] < 1e-15:
                break
            x[j] = max(0.0, x[j] + diff)

    return x


def interpolate_h(h, n_from, n_to):
    if n_to == n_from:
        return np.asarray(h, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_to, endpoint=False)
    u = np.interp(x1, x0, np.asarray(h, dtype=float), left=h[0], right=h[-1])
    return project_l2(u, n_to)


def _timeout_callback(xk):
    try:
        if time.time() > _STATE["deadline"]:
            return True
    except Exception:
        pass
    return False


def _lbf_maxiter(n, seconds):
    if n <= 800:
        rate = 900
    elif n <= 1600:
        rate = 520
    elif n <= 3000:
        rate = 300
    else:
        rate = 140
    v = int(max(500, min(50000, float(seconds) * float(rate))))
    return v


def _active_lag_value(x, L, n):
    if L >= 0:
        nv = n - int(L)
        if nv <= 0:
            return 0.0
        return float(np.dot(x[:nv], 1.0 - x[int(L):n]))
    r = -int(L)
    nv = n - r
    if nv <= 0:
        return 0.0
    return float(np.dot(x[r:n], 1.0 - x[:nv]))


def active_obj(x, n, active, alpha, mu):
    active = np.asarray(active, dtype=int)
    m = active.size
    if m == 0:
        d = float(np.sum(x) - 0.5 * float(n))
        return float(mu * d * d) if mu > 0 else 0.0

    vals = np.empty(m, dtype=float)
    dx = 2.0 / float(n)
    for j in range(m):
        vals[j] = _active_lag_value(x, int(active[j]), n)
    c = dx * vals
    val = smooth_max_value(c, alpha)
    if mu > 0.0:
        diff = float(np.sum(x) - 0.5 * float(n))
        val += mu * diff * diff
    if not np.isfinite(val):
        return 1e6
    return float(val)


def active_grad(x, n, active, alpha, mu):
    active = np.asarray(active, dtype=int)
    m = active.size
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)

    if m > 0:
        vals = np.empty(m, dtype=float)
        for j in range(m):
            vals[j] = _active_lag_value(x, int(active[j]), n)
        c = dx * vals

        cm = float(c.max())
        sh = alpha * (c - cm)
        sm = float(sh.max())
        e = np.exp(sh - sm)
        s = float(e.sum())
        if (not np.isfinite(s)) or s <= 0.0:
            w = np.zeros(m, dtype=float)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
            if not np.all(np.isfinite(w)):
                w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
                ws = float(w.sum())
                if ws > 0.0:
                    w /= ws

        for j in range(m):
            L = int(active[j])
            wm = dx * float(w[j])
            if wm == 0.0:
                continue
            if L >= 0:
                nv = n - L
                if nv > 0:
                    grad[:nv] += wm * (1.0 - x[L:n])
                    grad[L:n] += -wm * x[:nv]
            else:
                rr = -L
                nv = n - rr
                if nv > 0:
                    grad[rr:n] += wm * (1.0 - x[:nv])
                    grad[:nv] += -wm * x[rr:n]

    if mu > 0.0:
        diff = float(np.sum(x) - 0.5 * float(n))
        grad += (2.0 * mu * diff)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def select_active(h, n, top_k, tol_frac):
    vals, _ = full_values_and_total(h, n)
    c = (2.0 / float(n)) * vals
    mx = float(c.max())
    tol = max(1e-10, tol_frac * mx)
    idx = np.where(c >= mx - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    if top_k > 0 and idx.size > top_k:
        order = np.argsort(-c[idx])[:top_k]
        idx = idx[order]
    off = n - 1
    lags = (idx - off).astype(int)
    return lags.tolist()


def active_polish(h, n, deadline, polish_seconds, top_k=24, alphas=None, mu=0.01):

    best = h.copy()
    bc = compute_c5_np(best, n)
    if polish_seconds <= 0 or time.time() >= deadline - 8:
        return best, bc

    if alphas is None:
        alphas = [20000, 100000, 400000]
    bounds = [(0.0, 1.0)] * n

    if n <= 800:
        rate = 500
    elif n <= 1600:
        rate = 300
    elif n <= 3000:
        rate = 180
    else:
        rate = 90

    per = max(5.0, float(polish_seconds) / (float(len(alphas)) * 2.0))

        for _ in range(2):
            rem = deadline - time.time()
            if rem < 12:
                break
            lags = select_active(best, n, top_k, 0.002)
            if not lags:
                break
            stage_sec = min(per, rem - 6.0)
            if stage_sec < 6.0:
                break
            _STATE["deadline"] = time.time() + stage_sec
            maxit = int(max(300, min(20000, stage_sec * float(rate))))
            try:
                res = minimize(
                    active_obj, best, jac=active_grad,
                    args=(n, lags, alpha, mu),
                    method="L-BFGS-B",
                    bounds=bounds,
                    callback=_timeout_callback,
                    options={
                        "maxiter": maxit,

            elif cc <= bc:
