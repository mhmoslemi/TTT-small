                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_tmp = compute_c5(h, n)
                if c_tmp < best_c - 1e-14:
                    best_c = c_tmp
                    best_h = h.copy()
                    improved = True
                    break
            if not improved and stuck > 5:
                # Larger random perturbation
                scale = 0.01 * (1 + stuck * 0.5)
                hp = h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)
                c_hp = compute_c5(hp, n)
                if c_hp < best_c:
                    h = hp
                    best_c = c_hp
                    best_h = hp.copy()
                    stuck = 0
                else:
                    h = hp
                    if stuck > 20:
                        break
        else:
            stuck = 0
        trial += 1
    return best_h, best_c

def slsqq_polish(h, n, alpha, maxiter=2000, top_m=None):
    dx = 2.0 / n
    if top_m is not None:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
    else:
        top_idx = None
    
    def obj(x):
        dx2 = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx2
        if top_idx is not None:
            active = cc[top_idx]
        else:
            active = cc
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Gradient with increasing alpha
    t_phase1 = deadline * 0.35
    p1_dl = time.time() + t_phase1
    for alpha in [50, 200, 1000, 5000]:
        if time.time() > p1_dl:
            break
        phase_dl = min(p1_dl, time.time() + t_phase1 * 0.25)
        h_new, c_new = gradient_phase(h, n, alpha, phase_dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 2: SLSQP polish
    if time.time() < deadline - 20:
        for tm in [3, 5]:
            if time.time() > deadline - 10:
                break
            h_new = slsqq_polish(h, n, 100000, maxiter=1500, top_m=tm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 3: Fast local search
    if time.time() < deadline - 5:
        ls_dl = deadline - 2
        h_new, c_new = fast_local_search(best_h, n, ls_dl, top_k=5, n_trials=300000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Start from initial construction at its native resolution
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        alloc = min(300, (deadline - t0) * 0.30)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc = min(250, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_resolution(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try higher resolutions
    n_candidates = sorted(set([600, 700, 800]) - {n_init})
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.30)
        if alloc < 50:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multi-start at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.015 * (1.0 + 0.4 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(60, (deadline - time.time() - 25) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive local search
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        if time.time() < deadline - 12:
            h_new = slsqq_polish(best_h, n, 200000, maxiter=2000, top_m=5)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = fast_local_search(best_h, n, deadline - 1, top_k=5, n_trials=500000)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
