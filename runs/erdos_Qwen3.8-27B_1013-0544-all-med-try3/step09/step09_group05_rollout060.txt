def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_lp_h(g, n):
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            n_over = n - k
            A_ub[m, :n_over] = g[k:k + n_over]
        else:
            kk = -k
            n_over = n - kk
            A_ub[m, kk:kk + n_over] = g[:n_over]
        A_ub[m, n] = -1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    b_ub = np.zeros(2 * n - 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.clip(res.x[:n], 0.0, 1.0)
    return None

def alternating_lp(h, n, max_iters, deadline, pert_interval=200, pert_size=0.005):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Step A: optimize h given g = 1 - h
        g = 1.0 - h
        h_new = solve_lp_h(g, n)
        if h_new is not None:
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            h = h_new
        else:
            no_improve += 1
        # Step B: optimize g given h
        g_new = solve_lp_g(h, n)
        if g_new is not None:
            g_new = project_feasible(g_new, n)
            h_try = project_feasible(1.0 - g_new, n)
            c_try = compute_c5(h_try, n)

                best_c = c_try
                best_h = h_try.copy()
                no_improve = 0
                h = h_try
            else:

        else:

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    else:

    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
