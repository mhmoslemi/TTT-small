def smooth_grad_active(x, n, top_k, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()

        idx_arr = np.argpartition(-c, top_k)[:top_k]
    else:
        idx_arr = np.arange(len(c))
    c_a = c[idx_arr]

    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()

        w = np.zeros_like(w)
        w[np.argmax(sh)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    center = n - 1
    for i in range(len(idx_arr)):

        wm = w[i] * dx

                grad[0:nv] += wm * (-x[kk:n])

    return grad

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def slqp_with_constraints(h, n, deadline, top_k=20, c_target=None):
    """SLSQP with explicit upper-bound constraints on top shifts."""
    if c_target is None:
        c_target = compute_c5(h, n) + 0.01
    # Get current top-k shifts
    c_vec = compute_c5_vec(h, n)
    top_idx = np.argpartition(-c_vec, top_k)[:top_k]
    center = n - 1
    
    def obj_fn(x):
        return smooth_obj_full(x, n, 50000, 0.0)
    
    def obj_grad(x):
        return smooth_grad_full(x, n, 50000, 0.0)
    
    constraints = []
    for idx in top_idx:
        k = idx - center
        def make_con(idx, k):
            def con_func(x):
                if k >= 0:
                    nv = n - k
                    return float(np.dot(x[0:nv], 1 - x[k:n]) * (2.0/n)) - c_target
                else:
                    kk = -k
                    nv = n - kk
                    return float(np.dot(x[kk:n], 1 - x[0:nv]) * (2.0/n)) - c_target
            def con_jac(x):
                g = np.zeros(n)
                dx = 2.0 / n
                if k >= 0:

                    g[0:nv] += (1.0 - x[k:n]) * dx
                    g[k:n] += (-x[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += (1.0 - x[0:nv]) * dx
                    g[0:nv] += (-x[kk:n]) * dx
                return g
            return {'type': 'ineq', 'fun': con_func, 'jac': con_jac}
        constraints.append(make_con(idx, k))
    
    # Add sum constraint
    constraints.append({'type': 'eq', 'fun': lambda x: float(x.sum() - n/2), 
                        'jac': lambda x: np.ones(n)})
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):

    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def targeted_refine_fast(h, n, deadline):
    """Fast targeted refinement: only modify points affecting worst shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(30):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        dx = 2.0 / n
        
        # Find affected indices
        if k_worst >= 0:
            idx_a = np.arange(0, n - k_worst)
            idx_b = np.arange(k_worst, n)
        else:
            idx_a = np.arange(-k_worst, n)
            idx_b = np.arange(0, n + k_worst)
        
        # Try small adjustments on a subset
        n_affected = min(len(idx_a), 200)
        step = max(1, n_affected // 200)
        
        found_improve = False
        candidates = list(idx_a[::step]) + list(idx_b[::step])
        
        for i in candidates:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002]:
                nv = h[i] + delta
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h[i] = nv
                    found_improve = True
                    break
                else:
                    h[i] = best_h[i]
            if found_improve:
                h = best_h.copy()
                break
        if not found_improve:
            break
    
    return best_h, best_c
