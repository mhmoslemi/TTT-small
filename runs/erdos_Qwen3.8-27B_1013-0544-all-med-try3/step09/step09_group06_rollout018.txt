import time

            if tot < 1e-14:
                break
            h += (diff / tot) * h

    c = compute_all_lags(h, n)
    idx = np.argsort(c)[-top_m:]
    return float(c[idx].max())

def smooth_top_max(c, alpha, top_m=10):
    # Only consider top_m lags
    top_idx = np.argsort(c)[-top_m:]
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def grad_top_lag(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def adam_optimize(h, n, deadline, alpha=5000, top_m=5, lr=0.5):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_params = np.zeros(n)
    v_params = np.zeros(n)
    t = 0

        else:

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] -= h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n-m]
    return grad

def adam_phase(h, n, deadline, top_m=3, lr=0.3, binarize_rate=0.0):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    patience = 0
    
    while time.time() < deadline:
        t += 1
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += lag_gradient(h_curr, n, lag)
        g /= len(top_idx)
        dx_val = 2.0 / n
        g *= dx_val
        
        if binarize_rate > 0:
            g += binarize_rate * 2.0 * h_curr * (1.0 - h_curr) * (1.0 - 2.0 * h_curr)
        
        m_buf = b1 * m_buf + (1.0 - b1) * g
        v_buf = b2 * v_buf + (1.0 - b2) * g * g
        m_hat = m_buf / (1.0 - b1 ** t)
        v_hat = v_buf / (1.0 - b2 ** t)
        
        h_new = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > 200:
            lr *= 0.85
            patience = 0
            if lr < 0.003:
                break
    
    return best_h, best_c

def guided_ls(h, n, deadline, rng):
    """Gradient-guided local search on the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(5000):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        # Try top 2 lags
        top2 = np.argsort(c_all)[-2:]
        improved = False
        
        for k_idx in top2:
            if time.time() > deadline:
                break
            lag = int(k_idx) - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            
            g = lag_gradient(h, n, lag)
            # scores[i] = g[i] - g[i+m]: positive means moving from i to i+m helps
            scores = g[0:n-m] - g[m:n]
            
            # Find top few pairs
            top_pairs = np.argsort(scores)[::-1][:20]
            
            for pi in top_pairs:
                if time.time() > deadline:
                    break
                i = int(pi)
                j = i + m
                if g[i] <= g[j]:
                    continue
                
                max_move = min(h[i], 1.0 - h[j])
                if max_move < 1e-10:
                    continue
                
                # Try decreasing step sizes
                success = False
                for frac in [0.3, 0.1, 0.05, 0.02, 0.01]:
                    delta = frac * max_move
                    if delta < 1e-12:

                    h[i] -= delta
                    h[j] += delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True

                    else:
                        # Revert
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h, n)
                
                if success:

        if not improved:
            break
    
    return best_h, best_c

    hp = h + rng.normal(0, scale, n)

def _slsqp_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1.0 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    for wi, idx in zip(w, top_idx):
        lag = int(idx) - (n - 1)
        g += wi * lag_gradient(x, n, lag)
    g *= dx / alpha  # chain rule for log-sum-exp / alpha
    return g
```

Hmm, actually the gradient of the smooth max is more complex. The smooth max is:
f(x) = (1/α) * log(Σ exp(α * c_k))

∂f/∂x = (1/α) * (Σ α * exp(α * c_k) * ∂c_k/∂x) / (Σ exp(α * c_k))
       = Σ w_k * ∂c_k/∂x
where w_k = exp(α * c_k) / Σ exp(α * c_j)

And ∂c_k/∂x = dx * (gradient of the correlation at lag k, as computed by lag_gradient)

So: ∂f/∂x_i = Σ_k w_k * dx * (∂c_k/∂x_i)

In SLSQP, I need to provide the objective and its gradient. But actually, SLSQP can also work without providing the gradient (it uses finite differences). For n=500, finite differences would require 500 function evaluations per gradient computation, which is too expensive.

With 500 variables, SLSQP typically needs 200-500 iterations. So 500 * O(n²) = 500 * 250,000 = 125M operations. At ~1M ops/ms in NumPy, that's ~125 seconds. Too slow for a single SLSQP call.

Actually, I just realized there's another approach I haven't considered: **use the problem structure more directly**.

The objective is:
max_k c_k where c_k = dx * Σ_i h[i](1-h[i+k])

For a given set of "active" lags (where c_k is close to the max), we can write:
min_h max_{k ∈ K} dx * h^T A_k h + b_k^T h

where A_k and b_k depend on the specific lag k. This is a min-max quadratic program.

We can solve this using a **bisection on t**:
For a given t, check if there exists h with:
  dx * Σ_i h[i](1-h[i+k]) ≤ t for all k ∈ K
  0 ≤ h_i ≤ 1
  Σ h_i = n/2

This feasibility problem is a quadratic constraint satisfaction problem. For |K| = 3-5, it's a QCQP with 3-5 quadratic constraints. This can be solved with cvxpy (if we relax to SDP) or with a specialized solver.

But this is getting too complex for the time budget. Let me stick with the Adam + local search approach.

Let me write the final code now. I'll keep it clean and focused.

One more optimization: instead of computing the full `np.correlate(h, 1-h, mode='full')` for every evaluation, I can compute just the correlations at the relevant lags. This saves factor of ~2 (only n lags instead of 2n-1). But np.correlate is very fast in C, so this micro-optimization probably isn't worth it.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n - m])
        grad[0:n - m] -= h[m:n]
    else:
        grad[0:n - m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n - m]
    return grad
