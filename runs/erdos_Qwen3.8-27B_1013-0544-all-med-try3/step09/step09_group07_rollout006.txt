import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13 and abs(h.max() - 1.0) > 1e-13 and abs(h.min()) > 1e-13:
        h += diff / n
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        out = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        out = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(out)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    nh = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        wk = wk * dx
        lag = k - nh
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += wk * (1.0 - h[lag:])
            grad[lag:] += -wk * h[:n-lag]
        else:
            p = -lag
            grad[:n-p] += -wk * h[p:]
            grad[p:] += wk * (1.0 - h[:n-p])
    return grad

def lbfgsb_run(h, n, alpha, deadline, maxiter=3000):
    def f_and_g(x):
        x = np.clip(x, 0, 1)
        return smooth_max_val(compute_all_lags(x, n), alpha), smooth_max_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_and_g, h, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return project_feasible(h.copy(), n)

def adam_optimize(h, n, alpha, deadline, lr=0.01, maxiter=10000):
    h = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    while t < maxiter and time.time() < deadline:
        grad = smooth_max_grad(h, n, alpha)
        # Add gradient for sum constraint (soft)
        diff = h.sum() - n/2.0
        grad += 10.0 * diff
        
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if t % 500 == 0:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c
