        w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        w /= s

        if wk < 1e-10:
            continue

        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif lag > 0:

            grad[0:n-lag] += wk * (-h[lag:n])
        else:

            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] += wk * (-h[m:n])

            else:
                lr *= 1.05
            lr = max(0.01, min(lr, 5.0))
        h_curr = h_new
        it += 1
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def grad_objective(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] += wk * (-h[m:n])
    return grad * dx

def gradient_phase(h, n, deadline, alpha0=30, alpha1=5000):
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    lr = 0.3
    momentum = 0.92
    it = 0
    while time.time() < deadline:
        progress = min(1.0, it / 400.0)
        alpha = alpha0 * (alpha1 / alpha0) ** progress
        g = grad_objective(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if it % 100 == 99:
            if c_new > best_c + 0.005:
                lr *= 0.7
            else:
                lr = min(lr * 1.05, 2.0)
            lr = max(0.005, lr)
        h_curr = h_new
        it += 1
    return best_h, best_c

def local_search_pairs(h, n, deadline, top_m=5):

    best_h = h.copy()
    best_c = compute_c5(h, n)

def slsqp_polish(h, n, deadline, top_m=5):
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    alpha = 30000

    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max_val(active, alpha)

    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        w = smooth_max_weights(active, alpha)
        grad = np.zeros(n)
        for ii, k in enumerate(top_idx):
            if w[ii] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad[:] += w[ii] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[ii] * (1.0 - x[0:n-lag])
                grad[0:n-lag] += w[ii] * (-x[lag:n])
            else:
                m = -lag
                grad[m:n] += w[ii] * (1.0 - x[0:n-m])
                grad[0:n-m] += w[ii] * (-x[m:n])
        return grad * dx

    target_sum = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def try_resolution(h_best, n_best, n_new, deadline):
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_best)
    h_new = project_feasible(h_new, n_new)
    c_new = compute_c5(h_new, n_new)
    
    # Quick gradient phase
    phase1_end = time.time() + (deadline - time.time()) * 0.3
    h_g, c_g = gradient_phase(h_new, n_new, phase1_end, alpha0=50, alpha1=3000)
    if c_g < c_new:
        c_new = c_g
        h_new = h_g
    
    # Local search
    h_ls, c_ls = local_search_pairs(h_new, n_new, deadline, top_m=5)
    if c_ls < c_new:
        c_new = c_ls
        h_new = h_ls
    
    return h_new, c_new
