        v = h_new - h
        h = h_new
        lr *= 0.999

    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_momentum(h, n, deadline, alpha=20000, lr_init=0.005, momentum=0.92, topk=10):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    target_sum = n / 2.0
    lr = lr_init
    stuck_count = 0

    while time.time() < deadline:
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        grad = compute_gradient(y, n, alpha, topk=topk)
        diff = y.sum() - target_sum
        grad += 0.005 * diff * np.ones(n)

        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        v = h_new - h
        h = h_new
        lr *= 0.9995

        if stuck_count > 500:
            h = best_h + np.random.RandomState(stuck_count).normal(0, 0.01, n)
            h = project_feasible(h, n)
            v = np.zeros(n)
            lr = lr_init
            stuck_count = 0

    return best_h, best_c

def coordinate_descent_maxlag(h, n, deadline, max_sweeps=5000, top_lags=3):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    dx = 2.0 / n
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        c_raw = np.correlate(h, 1 - h, mode='full')
        top_idx = np.argsort(c_raw)[-top_lags:]

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            order = rng.permutation(n)
            for j in order:
                if lag > 0:
                    s = 0.0
                    if j + lag < n:
                        s += (1.0 - h[j + lag])
                    if j >= lag:
                        s -= h[j - lag]
                else:
                    idx = -lag
                    s = 0.0
                    if j >= idx:
                        s += (1.0 - h[j - idx])
                    if j + idx < n:
                        s -= h[j + idx]

                if s == 0:
                    continue
                step = 0.01
                old_val = h[j]
                new_val = np.clip(old_val - step * s, 0.0, 1.0)
                if abs(new_val - old_val) < 1e-10:
                    continue
                h[j] = new_val
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    no_improve = 0
                else:
                    h[j] = old_val

        if no_improve > 0 and sweep % 100 == 0:
            no_improve += 1
        if no_improve > 5:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            no_improve = 0

    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_relax_optimize(h, n, alpha, deadline, maxiter=2000):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return lbfgsb_optimize(h_bin, n, alpha, deadline, maxiter=maxiter)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: Progressive L-BFGS-B with topk focus
    phases = [
        (50, 300, None),
        (200, 300, None),
        (1000, 400, None),
        (5000, 400, 20),
        (20000, 500, 10),
        (100000, 600, 5),
        (500000, 800, 3),
        (2000000, 1000, 2),
        (10000000, 1200, 1),
    ]
    for alpha, maxit, topk in phases:
        if time.time() > deadline - 300:
            break
        remaining = deadline - 300 - time.time()
        if remaining < 8:
            break
        tlim = time.time() + min(remaining, 60)
        h_opt = lbfgsb_optimize(best_h, n, alpha, tlim, maxiter=maxit, topk=topk)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient with momentum
    if time.time() < deadline - 250:
        tlim = min(deadline - 150, time.time() + 200)
        h_opt, c_opt = subgradient_momentum(best_h, n, tlim, alpha=100000, lr_init=0.003, momentum=0.93, topk=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: L-BFGS-B polish
    if time.time() < deadline - 150:
        for alpha, topk in [(5000000, 2), (20000000, 1), (100000000, 1)]:
            if time.time() > deadline - 60:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 35, maxiter=2500, topk=topk)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Coordinate descent on max lag
    if time.time() < deadline - 120:
        tlim = min(deadline - 60, time.time() + 120)
        h_opt, c_opt = coordinate_descent_maxlag(best_h, n, tlim, max_sweeps=3000, top_lags=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Random perturbation + re-optimize
    if time.time() < deadline - 150:
        for noise in [0.03, 0.08, 0.15]:
            if time.time() > deadline - 80:
                break
            h_pert = best_h + rng.normal(0, noise, n)
            h_pert = project_feasible(h_pert, n)
            h_opt = lbfgsb_optimize(h_pert, n, 5000, time.time() + 25, maxiter=1000)
            h_opt = lbfgsb_optimize(h_opt, n, 500000, time.time() + 30, maxiter=2000, topk=5)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Binary relaxation
    if time.time() < deadline - 100:
        h_bin = binary_relax_optimize(best_h, n, 200000, time.time() + 50, maxiter=2000)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
            h_opt = lbfgsb_optimize(best_h, n, 5000000, time.time() + 25, maxiter=2000, topk=2)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Try higher resolution
    for n_try in [n + 250, n * 2]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - 40 - time.time()
        if remaining < 50:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 2000, time.time() + min(25, remaining * 0.25), maxiter=1500)
        h_opt = lbfgsb_optimize(h_opt, n_try, 100000, time.time() + min(20, remaining * 0.2), maxiter=2500, topk=10)
        h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, time.time() + min(15, remaining * 0.15), maxiter=2500, topk=3)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final polish
    if time.time() < deadline - 15:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, deadline - 3, maxiter=3000, topk=2)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt, c_opt = coordinate_descent_maxlag(best_h, n_final, deadline - 1, max_sweeps=2000, top_lags=2)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
