def smart_local_search(h, n, deadline, top_k=40, max_rounds=500):

    best_h = h.copy()
    best_c = compute_c5(h, n)

        if time.time() > deadline:
            break

        c_max = c_vec.max()
        threshold = c_max - max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) < 5:
            active = np.argsort(-c_vec)[:10]
        
        # Compute gradient
        g = max_subgradient(best_h, n, active)
        
        # Find top coordinates by |gradient|
        abs_g = np.abs(g)
        top_coords = np.argsort(-abs_g)[:top_k]
        
        improved = False
        # Try small transfers between top coordinates
        for ii in range(len(top_coords)):
            if time.time() > deadline:
                return best_h, best_c
            i = top_coords[ii]
            for jj in range(ii+1, len(top_coords)):
                j = top_coords[jj]
                
                # Determine direction: transfer from high-grad to low-grad
                if g[i] > g[j]:
                    # Decrease h[i], increase h[j]
                    direction = (i, j, -1, 1)
                else:
                    direction = (i, j, 1, -1)
                
                i, j, di, dj = direction
                
                # Line search: try a few step sizes
                for step in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
                    if best_h[i] + di*step < 0 or best_h[i] + di*step > 1:
                        continue
                    if best_h[j] + dj*step < 0 or best_h[j] + dj*step > 1:
                        continue
                    
                    best_h[i] += di * step
                    best_h[j] += dj * step
                    c_trial = compute_c5(best_h, n)
                    
                    if c_trial < best_c - 1e-14:
                        best_c = c_trial
                        improved = True
                    else:
                        best_h[i] -= di * step
                        best_h[j] -= dj * step
                    break  # Only try one step size per pair per direction
        
        if not improved:
            break
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, config):
    """L-BFGS with log-sum-exp smoothing."""
    alpha, mu = config
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        
        g = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                g[:nv] += wm * (1.0 - x[k:])
                g[k:] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[kk:])
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                               'maxfun': 20000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] += dx * (1.0 - h[k:])
        grad[k:] += -dx * h[:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += dx * (1.0 - h[:nv])
        grad[:nv] += -dx * h[kk:]
    return grad

def max_subgradient(h, n, active_shifts):
    g = np.zeros(n)
    for s in active_shifts:
        g += shift_gradient(h, n, s)
    g /= float(len(active_shifts))
    return g

def lbf_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbf_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def do_lbfgs(h, n, deadline, alpha, mu, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h, n)
    try:
        res = minimize(lbf_obj, h, jac=lbf_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16,

        c_new = compute_c5(h_new, n)
        return h_new, c_new

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
