def smooth_grad_from_w(h, w, n):
    # w length 2n-1 nonnegative
    dx=2.0/n
    if n<=1 return zeros
    w0=w[n-1]
    if np.all(w==0) return zeros
    # to improve, if sum weights not 1? no need. But if weights scaled, gradient scaled. We normalize outside.
    q = w[0:n-1][::-1]
    p = w[n:2*n]
    B = p + q
    # const
    if n>1:
       # head_const = suffix sum q length n
       head = np.empty(n); 
       if n-1>0:

             C_tail = np.convolve(D,h,'full')[:n]
          else zeros
       grad = w0*(1-2*h)+ const - C_head - C_tail
    else ...
    return grad * dx

def _fix_sum(y,target,n):

    diff = target - float(np.sum(y))

        diff = target - float(np.sum(y))
        if abs(diff) <= 1e-15:
            break

            room = 1.0 - y
            tot = float(room.sum())
            if tot <= 1e-15:

            else:
                y += diff * room / tot

def grad_full_weights(h,w,n):
    dx=2.0/n
    if n <= 1:
      return np.zeros(n)
    w0=w[n-1]
    if float(np.sum(w)) <= 0:
       return np.zeros(n)
    q=w[0:n-1][::-1]
    p=w[n:2*n]
    B=p+q
    head=np.empty(n)
    # head suffix q
    # q length n-1. compute suffix from right.
    if n-1 > 0:
       # use cumulative on reversed
       head_tail = np.cumsum(q[::-1])[::-1]
       head[:n-1] = head_tail
       head[n-1] = 0.0
    else:
       head.fill(0.0)
    pp = np.empty(n)
    pp[0]=0.0
    if n>1:
       pp[1:] = np.cumsum(p)
    const = head + pp
    if float(np.max(np.abs(B))) > 0:
       C_head = np.convolve(B[::-1], h, mode='full')[n-1:2*n-1]
       D=np.zeros(n)
       D[1:] = B
       C_tail = np.convolve(D, h, mode='full')[:n]
    else:
       C_head=np.zeros(n); C_tail=np.zeros(n)
    grad = w0*(1.0-2.0*h) + const - C_head - C_tail
    return grad * dx

def _fix_sum(y, target, n):
    y = np.asarray(y, dtype=float)
    diff = target - float(np.sum(y))
    if abs(diff) <= 1e-14:
        return y
    for _ in range(3):
        diff = target - float(np.sum(y))
        if abs(diff) <= 1e-15:
            break
        if diff > 0.0:

                y = y + diff * room / tot
        else:
            tot = float(y.sum())
            if tot <= 1e-15:
                y = y - diff / max(1, n)
            else:
                y = y + diff * y / tot
        y = np.clip(y, 0.0, 1.0)

    diff = target - float(np.sum(y))
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - y))
            add = min(float(diff), float(1.0 - y[idx]))
            y[idx] += add
        else:
            idx = int(np.argmax(y))
            sub = min(float(-diff), float(y[idx]))
            y[idx] -= sub
    y = np.clip(y, 0.0, 1.0)
    return y


def _project_exact(y, n):
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * float(n)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(y - tau, 0.0, 1.0)


def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(y.sum())
    if abs(s - target) <= 1e-13:
        return _fix_sum(y, target, n)

    diff = target - s
    if diff > 0.0:
        room = 1.0 - y
        tot = float(room.sum())
        if tot > 0.0:
            y += diff * room / tot
    else:
        if s > 0.0:
            y += diff * y / s

    y = np.clip(y, 0.0, 1.0)
    if abs(float(y.sum()) - target) > 1e-8:
        y = _project_exact(y, n)
    return _fix_sum(y, target, n)


def _logsumexp_value(vals, alpha):
    alpha = float(max(1.0, alpha))
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    es = float(np.sum(e))
    if es <= 0.0:
        return m
    return m + (mx + float(np.log(es))) / alpha


def full_weights(c, alpha):
    alpha = float(max(1.0, alpha))
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    es = float(np.sum(e))
    if es <= 0.0:
        return m, np.full_like(c, 1.0 / max(1, c.size))
    w = e / es
    val = m + (mx + float(np.log(es))) / alpha
    return val, w


def select_top_weights(c, alpha, top_m):
    alpha = float(max(1.0, alpha))
    idx = np.argsort(c)[-top_m:]
    vals = c[idx]
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    es = float(np.sum(e))
    if es <= 0.0:
        wt = np.full(top_m, 1.0 / max(1, top_m))
    else:
        wt = e / es
    val = m + (mx + float(np.log(es))) / alpha
    return idx, wt, val


def grad_top_indices(h, idx, wt, n):
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    for k, weight in zip(idx, wt):
        if weight <= 1e-16:
            continue
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += weight * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            if m < n:
                grad[m:n] += weight * (1.0 - h[0:n - m])
                grad[0:n - m] -= weight * h[m:n]
        else:
            m = -lag
            if m < n:
                grad[0:n - m] += weight * (1.0 - h[m:n])
                grad[m:n] -= weight * h[0:n - m]
    return grad * dx


def grad_full_weights(h, w, n):
    dx = 2.0 / float(n)
    if n <= 1:
        return w[0] * (1.0 - 2.0 * h) * dx

    w0 = w[n - 1]
    if float(np.sum(w)) <= 0.0:
        return np.zeros(n, dtype=float)

    q = w[0:n - 1][::-1]
    p = w[n:2 * n]
    B = p + q

    head = np.empty(n, dtype=float)
    if n - 1 > 0:
        head[:n - 1] = np.cumsum(q[::-1])[::-1]
        head[n - 1] = 0.0
    else:
        head.fill(0.0)

    pref = np.empty(n, dtype=float)
    pref[0] = 0.0
    if n > 1:
        pref[1:] = np.cumsum(p)

    const = head + pref

    if float(np.max(np.abs(B))) > 0.0:
        c_head = np.convolve(B[::-1], h, mode="full")[n - 1:2 * n - 1]
        D = np.zeros(n, dtype=float)
        D[1:] = B
        c_tail = np.convolve(D, h, mode="full")[:n]
    else:
        c_head = np.zeros(n, dtype=float)
        c_tail = np.zeros(n, dtype=float)

    grad = w0 * (1.0 - 2.0 * h) + const - c_head - c_tail
    return grad * dx


def shape_direction(h, g):
    g = np.asarray(g, dtype=float).copy()
    g = g - float(np.mean(g))

    g[(h <= 1e-12) & (g > 0.0)] = 0.0
    g[(h >= 1.0 - 1e-12) & (g < 0.0)] = 0.0

    g = g - float(np.mean(g))

    g[(h <= 1e-12) & (g > 0.0)] = 0.0
    g[(h >= 1.0 - 1e-12) & (g < 0.0)] = 0.0

    s = float(np.max(np.abs(g))) if g.size else 0.0
    if s < 1e-14:
        return None
    return g / s


def steepest_polish(h, n, deadline, alpha=200.0, top_m=3, stall_limit=120, step_init=0.2):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)
    c_cur = all_lags(h, n)
    cur_c = float(np.max(c_cur))

    step = float(step_init)
    stall = 0

    while time.time() < deadline:
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        if top_m is None:
            val, w = full_weights(c_cur, alpha)
            grad = grad_full_weights(h, w, n)
        elif top_m == 1:
            ii = int(np.argmax(c_cur))
            idx = np.array([ii], dtype=int)
            wt = np.array([1.0], dtype=float)
            val = float(c_cur[ii])
            grad = grad_top_indices(h, idx, wt, n)
        else:
            idx, wt, val = select_top_weights(c_cur, alpha, top_m)
            if top_m <= 80:
                grad = grad_top_indices(h, idx, wt, n)
            else:
                w = np.zeros(len(c_cur), dtype=float)
                w[idx] = wt
                grad = grad_full_weights(h, w, n)

        g = shape_direction(h, grad)
        if g is None:
            break

        trial_muls = (1.0, 0.5, 0.25, 2.0, 0.1)
        best_trial_c = cur_c
        best_trial_h = h
        best_mult = 1.0

        for mult in trial_muls:
            if time.time() > deadline:
                break
            s = step * mult
            if s <= 1e-9:
                continue
            cand = project_feasible(h - s * g, n)
            c_new = all_lags(cand, n)
            cc = float(np.max(c_new))
            if cc < best_trial_c - 1e-16:
                best_trial_c = cc
                best_trial_h = cand
                best_mult = mult

        if best_trial_c < best_c - 1e-16:
            best_c = best_trial_c
            best_h = best_trial_h
            h = best_trial_h
            c_cur = all_lags(h, n)
            cur_c = float(np.max(c_cur))
            step = float(max(1e-5, step * best_mult))
            stall = 0
        else:
            stall += 1
            step = float(max(1e-5, step * 0.7))
            if stall % 25 == 0:
                step = float(max(step, 0.05))
            if stall > stall_limit or step < 1e-7:
                break

    return best_h, best_c


def lag_grad_raw(h, n, lag):
    g = np.zeros(n, dtype=float)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = lag
        if m < n:
            g[m:n] = 1.0 - h[0:n - m]
            g[0:n - m] = -h[m:n]
    else:
        m = -lag
        if m < n:
            g[0:n - m] = 1.0 - h[m:n]
            g[m:n] = -h[0:n - m]
    return g


def pair_local_search(h, n, deadline, rng, max_rounds=80, pairs_per_round=12):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)
    stall = 0

    for _ in range(max_rounds):
        if time.time() > deadline:
            break

        c_cur = all_lags(h, n)
        cur_c = float(np.max(c_cur))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        k_idx = np.argsort(c_cur)[-3:]
        improved = False

        for k in k_idx:
            if time.time() > deadline:
                break
            lag = int(k) - (n - 1)
            g = lag_grad_raw(h, n, lag) * (2.0 / n)

            pos = np.where(g > 1e-14)[0]
            neg = np.where(g < -1e-14)[0]
            if pos.size == 0 or neg.size == 0:
                continue

            for _ in range(pairs_per_round):
                if time.time() > deadline:
                    break
                i = int(pos[rng.randint(pos.size)])
                j = int(neg[rng.randint(neg.size)])

                d = 0.05 * (0.25 + 0.75 * rng.random())
                d = min(d, 0.8 * h[i], 0.8 * (1.0 - h[j]))
                if d < 1e-7:
                    continue

                oi = h[i]
                oj = h[j]
                h[i] = oi - d
                h[j] = oj + d

                cc = eval_c5(h, n)
                if cc < best_c - 1e-15:
                    best_c = cc
                    best_h = h.copy()
                    improved = True
                    c_cur = all_lags(h, n)
                    break
                else:
                    h[i] = oi
                    h[j] = oj

            if improved:
                break

        if improved:
            h = best_h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break

    return best_h, best_c


def _sum_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def _sum_eq_jac(x, n):
    return np.ones(n, dtype=float)


def _slsqp_active_obj(x, n, idx, alpha):
    c = all_lags(x, n)
    return _logsumexp_value(c[idx], alpha)


def _slsqp_active_grad(x, n, idx, alpha):
    c = all_lags(x, n)
    idx, wt, _ = select_top_weights(c, alpha, len(idx))
    return grad_top_indices(x, idx, wt, n)


def _slsqp_full_obj(x, n, alpha):
    c = all_lags(x, n)
    return _logsumexp_value(c, alpha)


def _slsqp_full_grad(x, n, alpha):
    c = all_lags(x, n)
    _, w = full_weights(c, alpha)
    return grad_full_weights(x, w, n)


def slsqp_refine(h, n, top_m, alpha, maxiter, deadline):
    h = project_feasible(h, n)
    if time.time() > deadline - 2:
        return h

    c = all_lags(h, n)

    if top_m is None:
        fun = _slsqp_full_obj
        grad = _slsqp_full_grad
        args = (n, alpha)
    else:
        idx = np.argsort(c)[-top_m:]
        fun = _slsqp_active_obj
        grad = _slsqp_active_grad
        args = (n, idx, alpha)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_eq, "jac": _sum_eq_jac, "args": (n,)}]
    opts = {"maxiter": int(maxiter), "ftol": 1e-12, "maxfun": int(max(200, 20 * maxiter))}

    try:
        res = minimize(
            fun,
            h,
            args=args,
            jac=grad,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def _lbbf_active_obj(x, n, idx, alpha, penalty):
    c = all_lags(x, n)
    val = _logsumexp_value(c[idx], alpha)
    s = float(np.sum(x) - 0.5 * float(n))
    return val + penalty * s * s


def _lbbf_active_grad(x, n, idx, alpha, penalty):
    c = all_lags(x, n)
    _, wt, _ = select_top_weights(c, alpha, len(idx))
    grad = grad_top_indices(x, idx, wt, n)
    s = float(np.sum(x) - 0.5 * float(n))
    grad = grad + 2.0 * penalty * s * np.ones(n, dtype=float)
    return grad


def lbbf_refine(h, n, deadline, maxiter=500, top_m=5, alpha=300.0, penalty=1e4):
    h = project_feasible(h, n)
    if time.time() > deadline - 3:
        return h

    c = all_lags(h, n)
    idx = np.argsort(c)[-top_m:]
    args = (n, idx, alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    opts = {"maxiter": int(maxiter), "ftol": 1e-12, "maxfun": int(max(200, 20 * maxiter))}

    try:
        res = minimize(
            _lbbf_active_obj,
            h,
            args=args,
            jac=_lbbf_active_grad,
            method="L-BFGS-B",
            bounds=bounds,
            options=opts,
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def polish_solvers(h, n, deadline):
    h = project_feasible(h, n)
    base_c = eval_c5(h, n)

    if time.time() > deadline - 18 or n > 900:
        return h

    if time.time() < deadline - 12:
        maxit = 700 if n <= 600 else 300
        h2 = lbbf_refine(h, n, min(deadline - 4, time.time() + 25), maxiter)
        c2 = eval_c5(h2, n)
        if c2 <= base_c + 1e-12:
            h = h2
            base_c = c2

    if n <= 550 and time.time() < deadline - 16:
        maxit = 1100 if n <= 300 else 700
        h2 = slsqp_refine(h, n, 5, 400.0, maxit, deadline)
        c2 = eval_c5(h2, n)
        if c2 <= base_c + 1e-12:
            h = h2
            base_c = c2

        if n <= 350 and time.time() < deadline - 12:
            h2 = slsqp_refine(h, n, 3, 800.0, maxit, deadline)
            c2 = eval_c5(h2, n)
            if c2 <= base_c + 1e-12:
                h = h2
                base_c = c2

    return h


def optimize_resolution(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)

    if time.time() >= deadline - 3:
        return best_h, best_c

    total = deadline - time.time()
    if total > 12.0:
        dl = min(deadline - 2, time.time() + total * 0.38)
        try:
            h2, c2 = steepest_polish(h, n, dl, 1.0, 1, 140, 0.22)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    total = deadline - time.time()
    if total > 10.0:
        dl = min(deadline - 2, time.time() + total * 0.30)
        try:
            h2, c2 = steepest_polish(h_use, n, dl, 900.0, 3, 130, 0.16)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    total = deadline - time.time()
    if total > 8.0:
        dl = min(deadline - 2, time.time() + total * 0.22)
        try:
            h2, c2 = steepest_polish(h_use, n, dl, 160.0, 10, 120, 0.13)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    if time.time() < deadline - 22:
        try:
            h2 = polish_solvers(h_use, n, deadline)
            c2 = eval_c5(h2, n)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    total = deadline - time.time()
    if total > 8.0:
        dl = min(deadline - 2, time.time() + total * 0.22)
        try:
            h2, c2 = steepest_polish(h_use, n, dl, 40.0, 50, 120, 0.11)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    if time.time() < deadline - 3:
        try:
            hdl = min(deadline - 1, time.time() + max(3.0, (deadline - time.time()) * 0.35))
            h2, c2 = pair_local_search(h_use, n, hdl, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    h_use = best_h
    if time.time() < deadline - 6:
        try:
            hp = h_use + rng.normal(0.0, 0.025, n)
            hp = project_feasible(hp, n)
            h2, c2 = steepest_polish(hp, n, deadline - 1, 700.0, 3, 90, 0.09)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    return best_h, best_c


def make_parentlike(n):
    h = np.zeros(n, dtype=float)
    phi = 0.6180339887498948
    fr = np.mod(np.arange(n, dtype=float) * phi, 1.0)
    h[fr > 0.5] = 1.0
    return project_feasible(h, n)


def interp_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    if n_src == 1:
        out = np.full(n_dst, float(h_src[0]), dtype=float)
        return project_feasible(out, n_dst)
    xs = np.linspace(0.0, 2.0, n_src, endpoint=True)
    xt = np.linspace(0.0, 2.0, n_dst, endpoint=True)
    out = np.interp(xt, xs, h_src, left=float(h_src[0]), right=float(h_src[-1]))
    return project_feasible(out, n_dst)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(max(10.0, budget_s))
    deadline = t0 + budget - 4.0
    rng = np.random.RandomState(seed if seed is not None else 42)

    try:
        h_init = np.asarray(initial_h_values, dtype=float).reshape(-1).astype(float, copy=True)
        n_init = int(len(h_init))
        if n_init < 2:
            raise ValueError("bad parent length")
    except Exception:
        n_init = 500
        h_init = make_parentlike(n_init)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = eval_c5(h_init, n_init)
    tried = set([n_init])

    parent_alloc = min(620.0, max(60.0, (deadline - t0) * 0.55))
    if time.time() < deadline - 10:
        try:
            h2, c2 = optimize_resolution(h_init, n_init, t0 + parent_alloc, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    attempts = [650, 800, 1000, 400, 700, 550, 300, 1200, 900, 250, 450]
    for n_try in attempts:
        if time.time() > deadline - 35:
            break
        if n_try in tried:
            continue
        if not (24 <= n_try <= 1600):
            continue

        rem = deadline - time.time() - 12.0
        if rem < 35.0:
            break

        base_alloc = 100.0
        if n_try < 500:
            base_alloc = 90.0
        elif n_try < 800:
            base_alloc = 95.0
        elif n_try < 1100:
            base_alloc = 75.0
        else:
            base_alloc = 65.0

        alloc = min(base_alloc, rem * 0.45, 120.0)
        if alloc < 22.0:
            break

        tried_add = False
        try:
            if rng.random() < 0.22:
                h_start = make_parentlike(n_try)
            else:
                h_start = interp_h(best_h, len(best_h), n_try)
                if rng.random() < 0.7:
                    h_start = h_start + rng.normal(0.0, 0.045, n_try)
                    h_start = project_feasible(h_start, n_try)
            tried_add = True
            h2, c2 = optimize_resolution(h_start, n_try, time.time() + alloc, rng)
            tried.add(n_try)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            if tried_add:
                tried.add(n_try)

    n_best = int(len(best_h))
    restarts = 0
    while time.time() < deadline - 45:
        rem = deadline - time.time() - 18.0
        if rem < 30.0:
            break
        if restarts == 0 and n_best != n_init and n_best <= 1200:
            hp = make_parentlike(n_best)
        else:
            scale = 0.018 + 0.008 * (restarts % 3)
            hp = best_h + rng.normal(0.0, scale, n_best)
            hp = project_feasible(hp, n_best)

        alloc = min(38.0, rem * 0.45, 55.0)
        if alloc < 18.0:
            break

        try:
            h2, c2 = optimize_resolution(hp, n_best, time.time() + alloc, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                restarts = 0
            else:
                restarts += 1
                if restarts >= 3:
                    break
        except Exception:
            break

    if time.time() < deadline - 8:
        try:
            h2, c2 = optimize_resolution(best_h, len(best_h), deadline - 1, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        except Exception:
            pass

    best_h = project_feasible(best_h, int(len(best_h)))
    best_c = eval_c5(best_h, int(len(best_h)))

    return best_h, float(best_c), int(len(best_h))
