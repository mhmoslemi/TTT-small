    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

    return np.correlate(h, 1.0 - h, mode='full') * dx

    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    if s > 500:
        return m + (s + np.log(np.sum(np.exp(sh - s)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    w = np.exp(sh - s)
    w /= w.sum()
    return w

def grad_all(x, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def grad_active(x, n, active, w_a):
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w_a, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_smooth_jac(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = grad_all(x, n, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return obj_smooth(x, n, alpha, mu), g

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    s = sh.max()
    if s > 500:
        val = mx + (s + np.log(np.sum(np.exp(sh - s)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_active_jac(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    s = sh.max()
    w = np.exp(sh - s)
    w /= w.sum()
    g = grad_active(x, n, active, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return obj_active(x, n, active, alpha, mu), g

def run_lbfgs(h, n, deadline, alphas, mu=0.01, iters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = 2000 if iters is None else iters[i] if i < len(iters) else 2000
        maxit = min(maxit, max(500, int(remaining * 4)))
        try:
            res = minimize(obj_smooth, best_h, jac=obj_smooth_jac,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxit*8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except:
            pass
    return best_h, best_c

def run_active_lbfgs(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        mx = c_vec.max()
        thr = mx - max(1e-8, 0.0002 * mx)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])[:top_k]
        
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(500, int((deadline - time.time()) * 3))
            try:
                res = minimize(obj_active, best_h, jac=obj_active_jac,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxfun': maxit*8,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except:
                pass
    return best_h, best_c

def shift_alternating(h, n, deadline, max_iters=50000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    lr = 0.01
    n_active = 8
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        mx = c_vec.max()
        thr = mx - max(1e-9, 0.0001 * mx)
        active = np.where(c_vec >= thr)[0]
        if len(active) > n_active:
            active = np.argsort(-c_vec[active])[:n_active]
        
        # Compute gradient from active shifts (equal weight)
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - best_h[k:n]) * dx
                g[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - best_h[0:nv]) * dx
                g[0:nv] += (-best_h[kk:n]) * dx
        g /= len(active)
        
        # Projected step
        h_new = best_h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.95
        
        if lr < 1e-6:
            lr = 0.001  # Reset occasionally
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_steps=200000):
    """Targeted coordinate descent: only modify entries with large gradient."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    n_top = 20
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        mx = c_vec.max()
        thr = mx - max(1e-9, 0.0001 * mx)
        active = np.where(c_vec >= thr)[0]
        if len(active) > 6:
            active = np.argsort(-c_vec[active])[:6]
        
        # Gradient from active
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - best_h[k:n]) * dx
                g[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - best_h[0:nv]) * dx
                g[0:nv] += (-best_h[kk:n]) * dx
        g /= len(active)
        
        # Find top entries to modify
        # Increase where g < 0 (decreases objective), decrease where g > 0
        inc_mask = g < -1e-6
        dec_mask = g > 1e-6
        
        # Pick the most impactful entries
        inc_idx = np.where(inc_mask)[0]
        dec_idx = np.where(dec_mask)[0]
        if len(inc_idx) > 0 and len(dec_idx) > 0:
            inc_idx = inc_idx[np.argsort(g[inc_idx])][:n_top]  # Most negative first
            dec_idx = dec_idx[np.argsort(-g[dec_idx])][:n_top]  # Most positive first
        else:
            # All same sign, can't do pair update
            continue
        
        # Try moving mass from dec_idx to inc_idx
        step_size = 0.002
        improved = False
        for di in dec_idx[:n_top//2]:
            if time.time() > deadline:
                break
            for ii in inc_idx[:n_top//2]:
                if best_h[di] - step_size < 0 or best_h[ii] + step_size > 1:
                    continue
                best_h[di] -= step_size
                best_h[ii] += step_size
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improved = True
                else:
                    best_h[di] += step_size
                    best_h[ii] -= step_size
        if not improved:
            step_size *= 0.5
            if step_size < 1e-6:
                step_size = 0.005  # Reset
    return best_h, best_c

def multi_restart(h, n, deadline, n_restarts=3, perturb_scale=0.01):
    """Try small random perturbations and re-optimize."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(12345)
    
    per_restart = (deadline - time.time()) / max(n_restarts, 1)
    
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        # Perturb
        h_pert = best_h + rng.randn(n) * perturb_scale
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick active-set optimization
        end_r = time.time() + per_restart - 5
        h_new, c_new = run_active_lbfgs(h_pert, n, end_r, top_k=4,
                                         alphas=[50000, 200000, 1000000])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Also try the perturbation itself
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except:
        h0 = np.full(200, 0.5)
        n_orig = 200
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS warmup (100s)
    p1_end = t0 + 100

                       iters=[3000, 3000, 3000])

                                    alphas=[10000, 50000, 200000, 1000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Shift-alternating projected gradient (150s)
    if time.time() < deadline - 150:
        p3_end = min(time.time() + 150, deadline - 150)
