def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 20 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run_active_round(h, n, deadline, top_k, alpha, mu=0.001):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-7, 0.0005 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 5:
        return h, compute_c5(h, n)
    maxit = max(100, int(rem * 3))
    maxit = min(maxit, 3000)
    try:
        res = minimize(act_obj, h, jac=act_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 8,
                               'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def fine_pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
        if time.time() > deadline:
            break
    return best_h, best_c

def directed_pair_search(h, n, deadline, radius, delta):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    changes = 0
    while time.time() < deadline - 5 and changes < 50000:
        improved = False
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        active_set = set(int(a) for a in active)
        for idx in active[:min(len(active), 10)]:
            if time.time() > deadline - 10:
                break
            k = int(idx) - (n - 1)
            wm = 1.0 / len(active) * (2.0 / n)
            if k >= 0:
                pairs = [(i, i + k, -1) for i in range(n - k)]
            else:
                kk = -k
                pairs = [(i + kk, i, 1) for i in range(n - kk)]
            for (a, b, direction) in pairs[:200]:
                if direction == -1:
                    if best_h[a] >= delta and best_h[b] <= 1.0 - delta:
                        best_h[a] -= delta
                        best_h[b] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            changes += 1
                        else:
                            best_h[a] += delta
                            best_h[b] -= delta
                else:
                    if best_h[b] >= delta and best_h[a] <= 1.0 - delta:
                        best_h[b] -= delta
                        best_h[a] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            changes += 1
                        else:
                            best_h[b] += delta
                            best_h[a] -= delta
            if not improved and time.time() > deadline - 30:
                break
        if not improved:
            delta /= 2
            if delta < 1e-8:
                break
    return best_h, best_c

def perturb_and_reopt(h, n, deadline, rng, n_trials=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        h_pert = best_h + rng.uniform(-0.01, 0.01, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.005:
            maxit = max(200, int((deadline - time.time() - 10) / (n_trials - t)))
            try:
                res = minimize(full_obj, h_pert, jac=full_grad,
                              args=(n, 200000, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': min(maxit, 1000), 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': 2000,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate alternative starts and pick best
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n_orig, golden / 2.0),
        make_beatty(n_orig, np.sqrt(2.0) / 2.0),
        make_beatty(n_orig, 0.4312),
        make_random_binary(n_orig, rng),
        make_random_binary(n_orig, rng),
    ]
    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: Quick L-BFGS warmup (short, just to smooth)
    p1_end = min(t0 + 60, deadline - 800)
    if p1_end > t0 + 10:
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [1000, 20000, 200000]:
            if time.time() > p1_end:
                break
            rem = p1_end - time.time() - 5
            if rem < 10:
                break
            maxit = max(200, int(rem * 3))
            try:
                res = minimize(full_obj, best_h, jac=full_grad,
                              args=(best_n, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': min(maxit, 1500), 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': min(maxit * 8, 8000),
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Aggressive multi-round active set optimization
    p2_end = min(time.time() + 250, deadline - 450)
    if p2_end > time.time() + 30:
        cur_n = best_n
        configs = [
            (4, 50000, 0.002),
            (5, 100000, 0.001),
            (6, 200000, 0.001),
            (8, 500000, 0.001),
            (10, 1000000, 0.0005),
            (12, 2000000, 0.0005),
            (16, 5000000, 0.0003),
            (20, 10000000, 0.0003),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 20:
                break
            for _ in range(2):
                if time.time() > p2_end - 15:
                    break
                h2, c2 = run_active_round(best_h, cur_n, p2_end - 10, top_k, alpha, mu)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2

    # Phase 3: Try higher resolution
    n_up = best_n * 2
    p3_start = time.time()
    p3_end = min(p3_start + 200, deadline - 300)
    if p3_end > p3_start + 60 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.001:
            bounds_up = [(0.0, 1.0)] * n_up
            cur_n_up = n_up
            for alpha in [5000, 50000, 500000, 2000000]:
                if time.time() > p3_end - 20:
                    break
                rem = p3_end - time.time() - 10
                if rem < 10:
                    break
                maxit = max(300, int(rem * 2))
                maxit = min(maxit, 2000)
                try:
                    res = minimize(full_obj, h_up, jac=full_grad,
                                  args=(cur_n_up, alpha, 0.005), method='L-BFGS-B',
                                  bounds=bounds_up,
                                  options={'maxiter': maxit, 'ftol': 1e-22,
                                           'gtol': 1e-16, 'maxfun': maxit * 8,
                                           'maxls': 40, 'maxcor': 40})
                    h_new = project_feasible(res.x, cur_n_up)
                    c_new = compute_c5(h_new, cur_n_up)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                        best_n = cur_n_up
                        h_up = h_new
                except Exception:
                    pass
            # Active set at higher res
            for top_k, alpha in [(6, 500000), (10, 5000000), (16, 20000000)]:
                if time.time() > p3_end - 15:
                    break
                if best_n == n_up:
                    h3, c3 = run_active_round(best_h, best_n, p3_end - 10, top_k, alpha, 0.0005)
                    if c3 < best_c:
                        best_c = c3
                        best_h = h3

    # Phase 4: Fine directed pair search
    p4_end = min(time.time() + 150, deadline - 200)
    if p4_end > time.time() + 30:
        cur_n = len(best_h)
        h4, c4 = directed_pair_search(best_h, cur_n, p4_end, radius=100, delta=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: More active-set rounds at fine level
    p5_end = min(time.time() + 120, deadline - 120)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        for top_k, alpha, mu in [(4, 2000000, 0.0005), (6, 5000000, 0.0003),
                                  (8, 10000000, 0.0003), (12, 20000000, 0.0002)]:
            if time.time() > p5_end - 15:
                break
            h5, c5 = run_active_round(best_h, cur_n, p5_end - 10, top_k, alpha, mu)
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Very fine pair search
    p6_end = min(time.time() + 100, deadline - 60)
    if p6_end > time.time() + 20:
        cur_n = len(best_h)
        h6, c6 = fine_pair_search(best_h, cur_n, p6_end, radius=40,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation + reoptimization
    p7_end = min(time.time() + 60, deadline - 40)
    if p7_end > time.time() + 20:
        cur_n = len(best_h)
        h7, c7 = perturb_and_reopt(best_h, cur_n, p7_end, rng, n_trials=6)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final ultra-fine search
    p8_end = deadline - 2
    if p8_end > time.time() + 10:
        cur_n = len(best_h)
        h8, c8 = fine_pair_search(best_h, cur_n, p8_end, radius=25,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
