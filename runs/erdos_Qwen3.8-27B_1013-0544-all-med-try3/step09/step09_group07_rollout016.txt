def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):

    c = compute_all_lags(h, n)

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0e8 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 100,
                                'gtol': 1e-12, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)

def binary_perturb(h, n, rng, n_flips=30):

    target = n // 2
    if num_ones > target:

        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    # Flip a few random bits
    for _ in range(n_flips):
        i = rng.randint(0, n)
        h_bin[i] = 1.0 - h_bin[i]
    # Rebalance
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        rng.shuffle(one_idx)

        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None and n_init > 0:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: L-BFGS-B at initial resolution with increasing smoothness
    alpha_schedule = [2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    iters_schedule = [400, 500, 600, 800, 1000, 1200, 1500, 2000]
    for alpha, maxit in zip(alpha_schedule, iters_schedule):
        if time.time() > deadline - 420:
            break
        budget_left = deadline - 420 - time.time()
        if budget_left < 8:
            break
        h_opt = lbfgsb_pass(best_h, n, alpha, time.time() + min(45, budget_left * 0.6), maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Active set gradient descent
    if time.time() < deadline - 300:
        ad = min(deadline - 250, time.time() + 120)
        h_opt, c_opt = active_set_descent(best_h, n, ad, initial_lr=0.01, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: More L-BFGS-B at very high alpha
    if time.time() < deadline - 200:
        for alpha in [50000000, 200000000, 1000000000]:
            if time.time() > deadline - 80:
                break
            h_opt = lbfgsb_pass(best_h, n, alpha, time.time() + 35, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Simulated annealing
    if time.time() < deadline - 180:
        sa_dl = min(deadline - 120, time.time() + 100)
        h_opt, c_opt = simulated_annealing(best_h, n, sa_dl, rng,
                                           alpha_sa=100000, t_init=0.005,
                                           t_min=0.00002, cool=0.99988)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: L-BFGS-B polish after SA
    if time.time() < deadline - 120:
        h_opt = lbfgsb_pass(best_h, n, 500000000, time.time() + 30, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Systematic local search
    if time.time() < deadline - 80:
        ls_dl = min(deadline - 30, time.time() + 100)
        h_opt, c_opt = systematic_local_search(best_h, n, ls_dl, top_lags=6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Try higher resolution n=1000
    if time.time() < deadline - 150:
        n_hi = 1000
        remaining = deadline - 80 - time.time()
        if remaining > 80:
            h_interp = upsample(best_h, len(best_h), n_hi)
            h_opt = lbfgsb_pass(h_interp, n_hi, 100000, time.time() + min(30, remaining * 0.25), maxiter=1000)
            c_opt = compute_c5(h_opt, n_hi)
            h_opt = lbfgsb_pass(h_opt, n_hi, 5000000, time.time() + min(25, remaining * 0.2), maxiter=1500)
            c_opt = compute_c5(h_opt, n_hi)
            h_opt = lbfgsb_pass(h_opt, n_hi, 100000000, time.time() + min(20, remaining * 0.15), maxiter=1500)
            c_opt = compute_c5(h_opt, n_hi)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 8: Binary perturbation + re-optimize
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        for trial in range(3):
            if time.time() > deadline - 40:
                break
            h_pert = binary_perturb(best_h, n_cur, rng, n_flips=20 + trial * 10)
            h_opt = lbfgsb_pass(h_pert, n_cur, 10000000, time.time() + 20, maxiter=1000)
            c_opt = compute_c5(h_opt, n_cur)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 9: Final local search
    if time.time() < deadline - 20:
        n_final = len(best_h)
        ls_dl2 = min(deadline - 5, time.time() + 60)
        h_opt, c_opt = systematic_local_search(best_h, n_final, ls_dl2, top_lags=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 10: Final L-BFGS-B
    if time.time() < deadline - 4:
        n_final = len(best_h)
        h_opt = lbfgsb_pass(best_h, n_final, 500000000, deadline - 2, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
