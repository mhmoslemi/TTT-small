    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def _lag_gradient(h, n, lag):

        grad[0:n-m] += (1.0 - h[m:n])

    a_coeffs = np.zeros(max_harmonics)
    b_coeffs = np.zeros(max_harmonics)
    for k in range(1, max_harmonics + 1):
        a_coeffs[k-1] = np.mean(h * np.cos(np.pi * k * t)) * 2
        b_coeffs[k-1] = np.mean(h * np.sin(np.pi * k * t)) * 2

    def reconstruct(a, b):
        h_rec = np.full(n, 0.5)
        for k in range(1, max_harmonics + 1):
            h_rec += a[k-1] * np.cos(np.pi * k * t)
            h_rec += b[k-1] * np.sin(np.pi * k * t)
        return h_rec

    params = np.concatenate([a_coeffs, b_coeffs])
    best_params = params.copy()
    best_c = compute_c5(h, n)

    def obj_ft(p):
        a = p[:max_harmonics]
        b = p[max_harmonics:]
        h_rec = np.full(n, 0.5)
        for k in range(1, max_harmonics + 1):
            h_rec += a[k-1] * np.cos(np.pi * k * t)
            h_rec += b[k-1] * np.sin(np.pi * k * t)
        h_rec = project_feasible(h_rec, n)
        return compute_c5(h_rec, n)

    # Simple coordinate descent on Fourier coefficients
    for coord in range(2 * max_harmonics):
        if time.time() > deadline:
            break
        step_sizes = [0.01, 0.005, 0.001]
        for ss in step_sizes:
            for direction in [1, -1]:
                p_new = params.copy()
                p_new[coord] += direction * ss
                c_new = obj_ft(p_new)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_params = p_new.copy()
                    params = p_new.copy()
                else:
                    params[coord] = best_params[coord]

    a_best = best_params[:max_harmonics]
    b_best = best_params[max_harmonics:]
    h_best = np.full(n, 0.5)
    for k in range(1, max_harmonics + 1):
        h_best += a_best[k-1] * np.cos(np.pi * k * t)
        h_best += b_best[k-1] * np.sin(np.pi * k * t)
    h_best = project_feasible(h_best, n)
    return h_best, compute_c5(h_best, n)

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Gradient descent with moderate alpha
    if time.time() < deadline - 20:
        pd1 = min(time.time() + (deadline - time.time()) * 0.4, deadline - 30)
        h_new, c_new = gradient_descent_v2(h, n, pd1, alpha_init=100, alpha_final=10000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new

    # Phase 2: SLSQP with top-lag focus
    if time.time() < deadline - 20:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(h, n, alpha=50000, top_m=top_m, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: Targeted local search
    if time.time() < deadline - 5:
        ls_dl = min(deadline - 2, time.time() + (deadline - time.time()) * 0.4)
        h_new, c_new = targeted_local_search(best_h, n, ls_dl, max_rounds=150)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Phase A: Optimize at initial resolution
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        phase_a_end = t0 + min(280, budget_s * 0.28)
        h_opt, c_opt = optimize_pipeline(h_init, n_init, phase_a_end, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        phase_a_end = t0 + min(200, budget_s * 0.2)
        h_opt, c_opt = optimize_pipeline(h_start, n_init, phase_a_end, rng)
        best_h, best_c = h_opt, c_opt

    # Phase B: Try higher resolutions
    resolutions_to_try = [600, 700, 800]
    for n_try in resolutions_to_try:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 40
        if remaining < 70:
            break

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)

        alloc = min(110, remaining * 0.3)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_pipeline(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase C: Fourier pre-optimization to find new basin
    if best_h is not None and time.time() < deadline - 120:
        n = len(best_h)
        ft_end = min(time.time() + 80, deadline - 80)
        h_ft, c_ft = fourier_pre_optimize(best_h, n, ft_end, max_harmonics=20)
        if c_ft < best_c:
            best_c = c_ft
            best_h = h_ft.copy()
            # Re-optimize the Fourier-found solution in full space
            if time.time() < deadline - 40:
                h_opt, c_opt = optimize_pipeline(h_ft, n, deadline - 30, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase D: Multistart at best resolution
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 75))

        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.025, n)
            elif r % 4 == 1:
                freq = rng.uniform(2, 8)
                phase = rng.uniform(0, 2 * np.pi)
                hp = best_h + 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            elif r % 4 == 2:
                hp = best_h.copy()
                idx = rng.choice(n, size=10, replace=False)
                for ii in idx:
                    hp[ii] += rng.uniform(-0.08, 0.08)
            else:
                # Binarize then perturb
                h_bin = np.zeros(n)
                top_idx = np.argsort(best_h)[::-1][:n // 2]
                h_bin[top_idx] = 1.0
                hp = h_bin + rng.normal(0, 0.1, n)

            hp = project_feasible(hp, n)
            alloc = min(65, (deadline - time.time() - 30) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_pipeline(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase E: Final intensive polish
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)

        # SLSQP final
        if time.time() < deadline - 15:
            h_new = slsqp_optimize(best_h, n, alpha=200000, top_m=5, maxiter=3000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        # Very fine local search
        if time.time() < deadline - 3:
            best_h, best_c = targeted_local_search(best_h, n, deadline - 1, max_rounds=300)

        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
