from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    m = c.max()

    mx = sh.max()

def coordinate_descent(h, n, deadline, max_rounds=100):

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alpha, maxiter=5000):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 2, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def gradient_pair_descent(h, n, deadline, max_rounds=200):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-7, 0.003 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        
        grad = np.zeros(n)
        for idx in active[:20]:
            grad += np.abs(shift_grad_single(h, n, idx))
        if len(active) > 0:
            grad /= min(len(active), 20)
        
        order = np.argsort(-np.abs(grad))
        processed = 0
        for i in order[:min(150, n)]:
            if time.time() > deadline:
                return h, best_c
            i = int(i)
            best_improve = 0
            best_j = -1
            best_dir = 0
            best_step = 0
            for step in steps:
                for direction in [1, -1]:
                    new_val = h[i] + direction * step
                    if new_val < -1e-12 or new_val > 1 + 1e-12:
                        continue
                    for j_idx in range(min(30, len(order))):
                        j = int(order[j_idx])
                        if j == i:
                            continue
                        j_val = h[j] - direction * step
                        if j_val < -1e-12 or j_val > 1 + 1e-12:
                            continue
                        h[i] = new_val
                        h[j] = j_val
                        c_try = compute_c5(h, n)
                        h[i] = new_val - direction * step
                        h[j] = j_val + direction * step
                        improve = best_c - c_try
                        if improve > best_improve:
                            best_improve = improve
                            best_j = j
                            best_dir = direction
                            best_step = step
            if best_j >= 0 and best_improve > 1e-14:
                h[i] += best_dir * best_step
                h[best_j] -= best_dir * best_step
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
                processed += 1
        if processed == 0:
            break
    return h, best_c

                best_h = h.copy()
    
    # Phase 5: Another SQP round
    if time.time() < deadline - 150:
        sqp_end2 = min(time.time() + 60, deadline - 100)
        if time.time() < sqp_end2 - 5:
            h, c = sqp_optimize(h, n, sqp_end2, max_iter=150, rho_init=0.04)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 6: Gradient pair descent
    if time.time() < deadline - 80:
        cd_end = min(time.time() + 80, deadline - 50)
        h, c = gradient_pair_descent(h, n, cd_end, max_rounds=100)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 7: Random restarts
    if time.time() < deadline - 60:
        for trial in range(8):
            if time.time() > deadline - 20:
                break
            sigma = 0.005 * (trial + 1)
            h_pert = h + np.random.randn(n) * sigma
            h_pert = project_feasible(h_pert, n)
            h_pert, c_pert = lbfgs_optimize(h_pert, n, time.time() + 15, alpha=10000, maxiter=1500)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()
                h = h_pert
    
    # Phase 8: Try higher resolution n=2000
    n_high = 2000
    if time.time() < deadline - 120:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_high)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_high)
        c_hi = compute_c5(h_hi, n_high)
        
        if time.time() < deadline - 80:
            h_hi, c_hi = lbfgs_optimize(h_hi, n_high, min(time.time() + 50, deadline - 60),
                                         alpha=3000, maxiter=2000)
        
        if time.time() < deadline - 40:
            h_hi, c_hi = sqp_optimize(h_hi, n_high, min(time.time() + 30, deadline - 20),
                                       max_iter=100, rho_init=0.05)
        
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            n = n_high
    
    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 10:
        h_fin, c_fin = slqp_polish(best_h, len(best_h), deadline - 3, alpha=2000000, maxiter=1500)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
            n = len(best_h)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
