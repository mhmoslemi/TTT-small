                    nv = n - kk
                    g[kk:n] += (1.0 - cur_h[0:nv])
                    g[0:nv] += -cur_h[kk:n]
                grad_focus += np.abs(g)
            if len(active) > 0:
                grad_focus /= len(active)
            top_idx = np.argsort(-grad_focus)[:min(200, n)]
            found = False
            for i in top_idx:
                if found or time.time() > deadline:
                    break
                for j in top_idx:
                    if i == j:
                        continue
                    if cur_h[i] - step >= -1e-12 and cur_h[j] + step <= 1.0 + 1e-12:
                        cur_h[i] -= step
                        cur_h[j] += step
                        c = compute_c5(cur_h, n)
                        if c < cur_c - 1e-15:
                            cur_c = c
                            if c < best_c:
                                best_c = c
                                best_h = cur_h.copy()
                            improved = True
                            found = True
                            break
                        else:
                            cur_h[i] += step
                            cur_h[j] -= step
                    elif cur_h[i] + step <= 1.0 + 1e-12 and cur_h[j] - step >= -1e-12:
                        cur_h[i] += step
                        cur_h[j] -= step
                        c = compute_c5(cur_h, n)
                        if c < cur_c - 1e-15:
                            cur_c = c
                            if c < best_c:
                                best_c = c
                                best_h = cur_h.copy()
                            improved = True
                            found = True
                            break
                        else:
                            cur_h[i] -= step
                            cur_h[j] += step
    return best_h, best_c

def randomized_local_search(h, n, deadline, n_trials=300000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_c = best_c
    trials = 0
    for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline or trials >= n_trials:
            break
        while trials < n_trials and time.time() < deadline:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            trials += 1
            if cur_h[i] - step >= -1e-12 and cur_h[j] + step <= 1.0 + 1e-12:
                cur_h[i] -= step
                cur_h[j] += step
                c = compute_c5(cur_h, n)
                if c < cur_c - 1e-15:
                    cur_c = c
                    if c < best_c:
                        best_c = c
                        best_h = cur_h.copy()
                else:
                    cur_h[i] += step
                    cur_h[j] -= step
            elif cur_h[i] + step <= 1.0 + 1e-12 and cur_h[j] - step >= -1e-12:
                cur_h[i] += step
                cur_h[j] -= step
                c = compute_c5(cur_h, n)
                if c < cur_c - 1e-15:
                    cur_c = c
                    if c < best_c:
                        best_c = c
                        best_h = cur_h.copy()
                else:
                    cur_h[i] -= step
                    cur_h[j] += step
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new)
    return h_new

def active_set_qp(h, n, deadline, max_iter=200):
    """SQP-style: solve LP subproblem on active constraints."""
    from scipy.optimize import linprog
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = 0.03
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.002 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        grads = []
        c_vals = []
        for idx in active:
            k = idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - h[k:n])
                g[k:n] += -h[0:nv]
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - h[0:nv])
                g[0:nv] += -h[kk:n]
            g *= (2.0 / n)
            grads.append(g)
            c_vals.append(c_vec[idx])
        n_act = len(active)
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in range(n_act):
            A_ub[row, 0:n] = grads[j]
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[j]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(None, None)] * n + [(0.0, 2.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                rho *= 0.5
                if rho < 1e-10:
                    break
                continue
            delta = res.x[0:n]
            if np.max(np.abs(delta)) < 1e-12:
                break
            h_new = project_feasible(h + delta, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                if rho < 0.05:
                    rho = min(rho * 1.2, 0.08)
            else:
                rho *= 0.5
                if rho < 1e-11:
                    break
        except Exception:
            rho *= 0.5
            if rho < 1e-10:
                break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h.copy(), jac=smooth_grad,
                     args=(n, alpha), method='L-BFGS-B',
                     bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-22,
                              'gtol': 1e-18, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if time.time() < deadline:
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick L-BFGS at original n (40s)
    t1 = min(t0 + 40, deadline - 1000)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs_phase(best_h, n, t1,
            alphas=(500, 2000, 10000, 50000))

    # Phase 2: SQP at original n (50s)
    t2 = min(t0 + 90, deadline - 950)
    if t2 > time.time() + 5:
        best_h, best_c = active_set_qp(best_h, n, t2, max_iter=300)

    # Phase 3: Local search at original n (30s)
    t3 = min(t0 + 120, deadline - 900)
    if t3 > time.time() + 3:
        best_h, best_c = greedy_local_search(best_h, n, t3, max_iter=200000)

    # Phase 4: Upsample to n=1000 and optimize (350s)
    n2 = 1000
    t4_start = time.time()
    if t4_start < deadline - 250:
        h_hi = upsample(best_h, n, n2)
        c_hi = compute_c5(h_hi, n2)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
        cur_h = best_h.copy()

        # L-BFGS at n=1000 (120s)
        t4 = min(t4_start + 120, deadline - 130)
        if t4 > time.time() + 5:
            cur_h, c_hi = run_lbfgs_phase(cur_h, n2, t4,
                alphas=(500, 2000, 10000, 50000, 200000))
            if c_hi < best_c:
                best_c = c_hi
                best_h = cur_h.copy()

        # SQP at n=1000 (80s)
        t5 = min(t4_start + 200, deadline - 100)
        if t5 > time.time() + 5:
            cur_h, c_hi = active_set_qp(cur_h, n2, t5, max_iter=400)
            if c_hi < best_c:
                best_c = c_hi
                best_h = cur_h.copy()

        # Greedy local search at n=1000 (80s)
        t6 = min(t4_start + 280, deadline - 70)
        if t6 > time.time() + 3:
            cur_h, c_hi = greedy_local_search(cur_h, n2, t6, max_iter=400000)
            if c_hi < best_c:
                best_c = c_hi
                best_h = cur_h.copy()

        # Final L-BFGS at n=1000 (50s)
        t7 = min(t4_start + 330, deadline - 30)
        if t7 > time.time() + 5:
            cur_h, c_hi = run_lbfgs_phase(cur_h, n2, t7,
                alphas=(100000, 500000, 2000000))
            if c_hi < best_c:
                best_c = c_hi
                best_h = cur_h.copy()

    # Phase 5: Try n=1500 if time allows (200s)
    n3 = 1500
    t8_start = time.time()
    if t8_start < deadline - 80:
        h_hi3 = upsample(best_h, len(best_h), n3)
        c_hi3 = compute_c5(h_hi3, n3)
        cur_h3 = best_h.copy() if c_hi3 <= best_c else h_hi3

        t8 = min(t8_start + 90, deadline - 60)
        if t8 > time.time() + 5:
            cur_h3, c_hi3 = run_lbfgs_phase(cur_h3, n3, t8,
                alphas=(2000, 10000, 100000, 500000))
            if c_hi3 < best_c:
                best_c = c_hi3
                best_h = cur_h3.copy()

        t9 = min(t8_start + 150, deadline - 30)
        if t9 > time.time() + 3:
            cur_h3, c_hi3 = greedy_local_search(cur_h3, n3, t9, max_iter=200000)
            if c_hi3 < best_c:
                best_c = c_hi3
                best_h = cur_h3.copy()

    # Phase 6: Try n=2000 if time allows
    n4 = 2000
    t10_start = time.time()
    if t10_start < deadline - 60:
        h_hi4 = upsample(best_h, len(best_h), n4)
        c_hi4 = compute_c5(h_hi4, n4)
        cur_h4 = best_h.copy() if c_hi4 <= best_c else h_hi4

        t10 = min(t10_start + 50, deadline - 20)
        if t10 > time.time() + 5:
            cur_h4, c_hi4 = run_lbfgs_phase(cur_h4, n4, t10,
                alphas=(5000, 50000, 200000))
            if c_hi4 < best_c:
                best_c = c_hi4
                best_h = cur_h4.copy()

    # Final: SLSQP-style polish on best solution
    t_final = deadline - 5
    if time.time() < t_final - 10:
        nf = len(best_h)
        h_fin, c_fin = slqp_polish(best_h, nf, t_final, alpha=1000000, maxiter=2000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
