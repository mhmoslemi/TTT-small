```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c * dx
    m = c_scaled.max()
    shifted = alpha * (c_scaled - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n-idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n-idx])
    return grad * dx

def smooth_max_obj(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
        obj_val = smooth_max_obj(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * 1e6 * diff
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50,
                                'iprint': -1, 'gtol': 1e-8})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_optimize(h, n, deadline, rng, alpha_init=1000, eta0=0.01):
    """Projected subgradient with away steps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    eta = eta0
    it = 0
    
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        k_max = np.argmax(c_all)
        lag = k_max - (n - 1)
        
        # Compute gradient of the max lag
        grad = np.zeros(n)
        if lag == 0:
            grad = 1.0 - 2.0 * h
        elif lag > 0:
            grad[lag:n] += (1.0 - h[0:n-lag])
            grad[0:n-lag] += (-h[lag:n])
        else:
            idx = -lag
            grad[0:n-idx] += (1.0 - h[idx:n])
            grad[idx:n] += (-h[0:n-idx])
        
        h_new = h - eta * grad * (2.0/n)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        it += 1
        eta *= 0.9995
        
        # Periodic restart
        if it % 100 == 0:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            eta = max(eta, 0.003)
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def smooth_max_grad(h, n, alpha):

    c_full = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c_full * dx
    m = c_scaled.max()
    shifted = alpha * (c_scaled - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            # Corrected: linear part on [0, n-1-lag], bilinear h[j-lag] on [lag, n-1]
            grad[0:n-lag] += wk * (1.0 - h[lag:n])
            grad[lag:n] += wk * (-h[0:n-lag])

            grad[m2:n] += wk * (1.0 - h[0:n-m2])
            grad[0:n-m2] += wk * (-h[m2:n])

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    penalty = 1e5

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        obj_val = smooth_max_obj(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty * diff
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40,
                                'iprint': -1, 'gtol': 1e-7})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:

def interpolate_resolution(h_src, n_src, n_dst):

    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_hill_climb(h_init, n, deadline, rng, max_rounds=200):
    """Optimize binary pattern via hill climbing."""

    num_ones = int(h_bin.sum())

        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h_init[one_idx])

            h_bin[one_idx[idx]] = 0

        round_count += 1
        c_all = compute_all_lags(h_bin, n)
        top_k = np.argsort(c_all)[-10:]
        for _ in range(200):
            if time.time() > deadline:
                break
            k_idx = top_k[rng.randint(len(top_k))]
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h_bin[i] == h_bin[j]:
                continue
            # Try flip: set h[i] and h[j] to swap values
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c_test = compute_c5(h_bin, n)
            if c_test < best_c - 1e-12:
                best_c = c_test
                improved = True
                c_all = compute_all_lags(h_bin, n)
                top_k = np.argsort(c_all)[-10:]
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, best_c

def targeted_local_search(h, n, deadline, rng, max_rounds=50000):
    """Targeted local search focusing on worst lags with correct directional moves."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds_since_improve = 0
    base_step = 0.015

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-8:]
        step = base_step * (1.0 + 0.5 * rng.random())
        k_idx = top_idx[rng.randint(len(top_idx))]
        lag = k_idx - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        if m >= n:
            continue
        i = rng.randint(0, n - m)
        j = i + m
        if lag > 0:
            # To reduce c_lag: want h[i] and h[j] to be aligned
            # Gradient for h[i] (i < lag or i >= lag matters, but the direction is:
            # decrease h[i] if h[j] is high, increase h[i] if h[j] is low)
            # Effectively: move mass from the "1" to the "0" at this lag
            if h[i] > h[j]:
                d = min(step, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
            else:
                d = min(step, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1.0 - h[i]) * 0.5)
                if d < 1e-10:
                    continue
                h[i] += d
                h[j] -= d
        else:
            # lag < 0, m = -lag: the pair is (i, j=i+m) but the correlation
            # involves h[i+m]*(1-h[i]) effectively (by symmetry)
            if h[j] > h[i]:
                d = min(step, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1.0 - h[i]) * 0.5)
                if d < 1e-10:
                    continue
                h[i] += d
                h[j] -= d
            else:
                d = min(step, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d

        h = project_feasible(h, n)
        c_test = compute_c5(h, n)
        if c_test < best_c - 1e-13:
            best_c = c_test
            best_h = h.copy()
            rounds_since_improve = 0
            base_step = min(0.03, base_step * 1.02)
        else:
            h = best_h.copy()
            rounds_since_improve += 1
            if rounds_since_improve % 100 == 0:
                base_step = max(0.001, base_step * 0.85)
            if rounds_since_improve > 5000:
                base_step = 0.02
                rounds_since_improve = 0
                h = best_h + rng.normal(0, 0.008, n)
                h = project_feasible(h, n)

    return best_h, best_c

def gradient_descent_optimize(h, n, deadline, alpha=50000, lr0=0.005, decay=0.9997):
    """Projected gradient descent with the smooth max objective."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    no_improve = 0

    while time.time() < deadline:
        obj = smooth_max_obj(compute_all_lags(h, n), alpha)
        grad = smooth_max_grad(h, n, alpha)
        # Project gradient to sum-zero subspace
        grad -= grad.mean()
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr = lr * 0.5
                no_improve = 0
                if lr < 1e-6:
                    h = best_h + np.random.randn(n) * 0.005
                    h = project_feasible(h, n)
                    lr = lr0 * 0.3
        lr *= decay

    return best_h, best_c

def simulated_annealing_fast(h, n, deadline, rng, t_init=0.005, t_min=0.00002, cool=0.9999):
    """Fast simulated annealing with pair moves."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stuck = 0

    while time.time() < deadline and T > t_min:
        # Pick move
        r = rng.random()
        if r < 0.5:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-5:]
            k_star = top_k[rng.randint(5)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m

        max_step = min(0.08, 0.003 + T * 5)
        if h[i] > h[j] + 1e-6:
            d = rng.uniform(0.002, max_step)
            d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
        elif h[j] > h[i] + 1e-6:
            d = -rng.uniform(0.002, max_step)
            d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
        else:
            T *= cool
            continue

        if abs(d) < 1e-9:
            T *= cool
            continue

        h_old_c = compute_c5(h, n)
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        h_new_c = compute_c5(h, n)

        de = h_new_c - h_old_c
        accept = False
        if de < -1e-14:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if h_new_c < best_c - 1e-14:
                best_c = float(h_new_c)
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            stuck += 1

        T *= cool
        if stuck > 30000:
            h = best_h + rng.normal(0, 0.01, n)
            h = project_feasible(h, n)
            stuck = 0
            T = max(T, 0.003)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
    best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: L-BFGS-B at parent resolution with increasing alpha (60s)
    alpha_schedule = [500, 5000, 50000, 500000, 2000000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 600:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 12, maxiter=800)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Gradient descent with smooth max (40s)
    if time.time() < deadline - 500:
        gd_deadline = min(deadline - 450, time.time() + 40)
        h_opt, c_opt = gradient_descent_optimize(best_h, n, gd_deadline, alpha=200000, lr0=0.003)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Binary hill climb + relax (60s)
    if time.time() < deadline - 400:
        bin_deadline = min(deadline - 350, time.time() + 25)
        h_bin, c_bin = binary_hill_climb(best_h, n, bin_deadline, rng, max_rounds=100)
        # Relax from binary
        if time.time() < deadline - 300:
            h_relaxed = lbfgsb_optimize(h_bin, n, 200000, time.time() + 30, maxiter=1500)
            c_relaxed = compute_c5(h_relaxed, n)
            if c_relaxed < best_c:
                best_c = c_relaxed
                best_h = h_relaxed

    # Phase 4: Try higher resolutions (300s)
    resolutions_to_try = [750, 1000, 1500]
    for n_try in resolutions_to_try:
        if time.time() > deadline - 200:
            break
        remaining = deadline - 180 - time.time()
        if remaining < 50:
            break
        h_interp = interpolate_resolution(best_h, len(best_h), n_try)
        c_interp = compute_c5(h_interp, n_try)
        # Optimize at new resolution
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, time.time() + min(25, remaining * 0.3), maxiter=1000)
        c_opt = compute_c5(h_opt, n_try)
        h_opt = lbfgsb_optimize(h_opt, n_try, 500000, time.time() + min(20, remaining * 0.25), maxiter=1500)
        c_opt = compute_c5(h_opt, n_try)
        h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, time.time() + min(15, remaining * 0.15), maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try

    # Phase 5: Targeted local search (150s)
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        ls_deadline = min(deadline - 30, time.time() + 120)
        h_opt, c_opt = targeted_local_search(best_h, n_cur, ls_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: SA for escape (80s)
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        sa_deadline = min(deadline - 20, time.time() + 70)
        h_opt, c_opt = simulated_annealing_fast(best_h, n_cur, sa_deadline, rng,
                                                 t_init=0.004, t_min=0.00002, cool=0.99992)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final L-BFGS-B polish (30s)
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_cur, 10000000, deadline - 5, maxiter=3000)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final local search (20s)
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        h_opt, c_opt = targeted_local_search(best_h, n_cur, deadline - 1, rng, max_rounds=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
