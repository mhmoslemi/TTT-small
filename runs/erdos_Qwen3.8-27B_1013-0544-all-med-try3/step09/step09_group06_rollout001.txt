            for _ in range(n_tries):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n - m)
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 0.01:
                    continue
                # Swap to reduce overlap at this lag
                step = min(abs(diff) * 0.3, 0.15, 1.0 - max(h[i], h[j]), min(h[i], h[j]) * 0.8)
                if step < 1e-7:
                    continue
                if diff > 0:
                    h[i] -= step
                    h[j] += step
                else:
                    h[i] += step
                    h[j] -= step
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    if diff > 0:
                        h[i] += step
                        h[j] -= step
                    else:
                        h[i] -= step
                        h[j] += step
                    h = project_feasible(h, n)
        
        if not improved:
            # Try smaller steps
            for trial_step in [0.02, 0.01, 0.005, 0.002]:
                if time.time() > deadline:
                    break
                c_all = compute_all_lags(h, n)
                k_star = int(np.argmax(c_all))
                lag = k_star - (n - 1)
                if lag == 0:
                    break
                m = abs(lag)
                found = False
                for _ in range(100):
                    if time.time() > deadline:
                        break
                    i = rng.randint(0, n - m)
                    j = i + m
                    diff = h[i] - h[j]
                    if abs(diff) < trial_step * 0.5:
                        continue
                    step = trial_step * 0.5
                    if diff > 0:
                        h[i] -= step
                        h[j] += step
                    else:
                        h[i] += step
                        h[j] -= step
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        best_h = h.copy()
                        found = True
                        break
                    else:
                        if diff > 0:
                            h[i] += step
                            h[j] -= step
                        else:
                            h[i] -= step
                            h[j] += step
                        h = project_feasible(h, n)
                if found:
                    improved = True
                    break
            
            if not improved:
                # Escalate: random perturbation
                if rnd > 200 and rnd % 500 == 0:
                    scale = 0.03 + 0.02 * (rnd // 5000)
                    pert = rng.normal(0, scale, n)
                    h_pert = project_feasible(best_h + pert, n)
                    c_pert = compute_c5(h_pert, n)
                    if c_pert < best_c:
                        h = h_pert
                        best_c = c_pert
                        best_h = h_pert.copy()
                    else:
                        h = best_h.copy()
                
                # Try binary projection
                if rnd > 100 and rnd % 500 == 250:
                    h_bin = (h > 0.5).astype(float)
                    # Fix count
                    n_ones = int(h_bin.sum())
                    target = n // 2
                    if n_ones > target:
                        ones = np.where(h_bin == 1)[0]
                        remove = ones[rng.choice(n_ones - target, target - n_ones + n_ones - target, replace=False)[:max(0, n_ones - target)]]
                        h_bin[remove] = 0
                    elif n_ones < target:
                        zeros = np.where(h_bin == 0)[0]
                        add = zeros[rng.choice(n - n_ones, target - n_ones, replace=False)]
                        h_bin[add] = 1
                    h_bin = project_feasible(h_bin, n)
                    c_bin = compute_c5(h_bin, n)
                    if c_bin < best_c:
                        best_c = c_bin
                        best_h = h_bin.copy()
                        h = h_bin.copy()
                
                if rnd > 2000:
                    break
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        if top_m is not None:
            idx = np.argsort(c)[-top_m:]
            c = c[idx]
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def optimize_one_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase A: Smooth max gradient
    t_phase = min(deadline - time.time(), (deadline - time.time()) * 0.35)
    deadline_a = time.time() + t_phase
    h_a, c_a = smooth_max_opt(h, n, deadline_a, alpha=30000, max_iter=15000)
    if c_a < best_c:
        best_c = c_a
        best_h = h_a
    h = h_a
    
    # Phase B: Subgradient
    if time.time() < deadline - 15:
        deadline_b = min(deadline - 10, time.time() + (deadline - time.time()) * 0.2)
        h_b, c_b = subgradient_step(h, n, deadline_b, lr_init=0.008, decay=0.9997, max_iter=30000)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b
        h = h_b
    
    # Phase C: SLSQP polish
    if time.time() < deadline - 10:
        h_c = slsqp_polish(h, n, alpha=100000, maxiter=1500, top_m=4)
        c_c = compute_c5(h_c, n)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c
        
        h_c2 = slsqp_polish(h_c, n, alpha=500000, maxiter=1000, top_m=3)
        c_c2 = compute_c5(h_c2, n)
        if c_c2 < best_c:
            best_c = c_c2
            best_h = h_c2
    
    # Phase D: Intensive local search
    if time.time() < deadline - 3:
        ls_deadline = min(deadline - 1, time.time() + (deadline - time.time()) * 0.5)
        h_d, c_d = intensive_local_search(best_h, n, ls_deadline, rng)
        if c_d < best_c:
            best_c = c_d
            best_h = h_d
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Start from initial construction
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_one_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = golden_ratio_binary(n_init)
        alloc = min(200, (deadline - t0) * 0.2)
        h_opt, c_opt = optimize_one_resolution(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try alternative starting points at same resolution
    n_best = len(best_h)
    alt_starts = [
        golden_ratio_binary(n_best),
        blue_noise_binary(n_best),
    ]
    
    for alt in alt_starts:
        if time.time() > deadline - 100:
            break
        # Interpolate best to help
        x_grid = np.linspace(0, 2, n_best)
        alt_interp = np.interp(x_grid, x_grid, alt)
        # Mix with best
        alt_mix = 0.4 * alt_interp + 0.6 * best_h
        alt_mix = project_feasible(alt_mix, n_best)
        
        alloc = min(80, (deadline - time.time()) * 0.15)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_one_resolution(alt_mix, n_best, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Try different resolutions
    n_candidates = []
    n_best = len(best_h)
    for nc in [350, 400, 450, 550, 600, 650]:
        if nc != n_best and nc not in n_candidates:
            n_candidates.append(nc)
    
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 20
        if remaining < 50:
            break
        alloc = min(100, remaining * 0.2)
        if alloc < 35:
            break
        
        # Interpolate best solution
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        h_opt, c_opt = optimize_one_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multistart perturbations at best resolution
    n_best = len(best_h)
    n_restarts = max(1, int((deadline - time.time() - 30) / 50))
    for r in range(min(n_restarts, 8)):
        if time.time() > deadline - 35:
            break
        scale = 0.015 * (1.0 + 0.5 * r)
        hp = best_h + rng.normal(0, scale, n_best)
        hp = project_feasible(hp, n_best)
        alloc = min(45, (deadline - time.time() - 25) * 0.25)
        if alloc < 20:
            break
        h_opt, c_opt = optimize_one_resolution(hp, n_best, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final intensive local search with remaining time
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ls_deadline = deadline - 1
        h_final, c_final = intensive_local_search(best_h, n, ls_deadline, rng, max_rounds=200000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        
        # Final SLSQP
        if time.time() < deadline - 8:
            h_s = slsqp_polish(best_h, n, alpha=200000, maxiter=1000, top_m=3)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    if best_h is None:
        best_h = project_feasible(golden_ratio_binary(n_init), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
