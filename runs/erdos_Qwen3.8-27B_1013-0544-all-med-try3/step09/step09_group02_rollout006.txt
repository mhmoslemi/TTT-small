import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12 * max(1, n):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    """Gradient of overlap at shift_idx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def sqp_lp(h, n, active_idx, rho):
    """One SQP step via LP."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    
    total_ineq = len(active_idx) + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    
    for idx in active_idx:
        g = shift_grad(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.ones((1, n_var))
    A_eq[0, n] = 0
    b_eq = np.array([0.0])  # sum(delta) = 0
    
    bounds = [(None, None)] * n + [(0.0, 2.0)]  # Wait, t can be negative?
    # Actually t should be able to be negative (we want to minimize t)
    # But t >= 0 makes sense since C5 >= 0
    # Let's allow t to be in [-1, 2]
    bounds = [(None, None)] * n + [(-1.0, 2.0)]
    
    try:

                      bounds=bounds, method='highs')
        if res.success:

    except Exception:
        pass
    return None, None

def subgradient_refine(h, n, deadline, n_steps=5000, step_alpha=0.001):
    """Iterative subgradient descent focused on active shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

            break

        active = sorted_idx[:n_active]

        for idx in active:
            grad += shift_grad(h, n, int(idx))
        grad /= n_active
        
        # Adaptive step size
        step_size = step_alpha / (1 + step * 0.01)
        
        h_new = project_feasible(h - step_size * grad, n)
        c_new = compute_c5(h_new, n)

def coordinate_descent(h, n, deadline, center_radius=60):
    """Fine coordinate descent: try moving mass between pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-6, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute sensitivity: which indices affect the active shifts most
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad(h, n, int(idx)))
            
            top_pos = np.argsort(-grad)[:min(center_radius * 2, n)]

                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)

                    if i == j:
                        continue

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h, n_from, n_to):
    """Upsample h from n_from points to n_to points."""

    x_new = np.linspace(0, 2, n_to)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_to)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * max(1.0, n):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad


def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)] * n + [(-1.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None


def sqp_optimize(h, n, deadline, max_iter=5000, rho_init=0.005, top_k=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    total_sqp_calls = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        sorted_idx = np.argsort(-c_vec)
        n_act = min(top_k, 2 * n - 1)
        active = [int(sorted_idx[i]) for i in range(n_act)]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        total_sqp_calls += 1
        if delta is None:
            rho *= 0.5
            no_improve += 1
            if rho < 1e-10 or no_improve > 50:
                break
            continue
        max_step = np.max(np.abs(delta))
        if max_step < 1e-14:
            rho *= 0.5
            no_improve += 1
            if no_improve > 30:
                break
            continue
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.01:
                rho = min(rho * 1.15, 0.01)
        else:
            no_improve += 1
            rho *= 0.4
            if no_improve > 40 or rho < 1e-10:
                break
    return best_h, best_c


def subgradient_refine(h, n, deadline, n_steps=50000, lr_init=0.002):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        sorted_idx = np.argsort(-c_vec)
        n_act = min(3, len(sorted_idx))
        active = [int(sorted_idx[i]) for i in range(n_act)]
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad(h, n, idx)
        grad /= n_act
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_h, best_c


def coordinate_descent(h, n, deadline, center_radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-7, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad(h, n, int(idx)))
            n_top = min(center_radius * 2, n)
            top_pos = np.argsort(-grad)[:n_top]
            for i_val in top_pos:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c


def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 200.0), (3000, 2000.0), (10000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def upsample(h, n_from, n_to):

    x_new = np.linspace(0.0, 2.0, n_to)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_to)
