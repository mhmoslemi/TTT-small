def nesterov_pair_swap(h, n, t_end, alpha=50000, K=10, eta0=0.01, momentum=0.85):
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_c = compute_c5(x, n)
    
    while time.time() < t_end:
        c = compute_all_lags(x, n)
        w = smooth_max_weights(c, alpha)
        g = grad_smooth_max(x, n, alpha)
        # Project to sum=0
        g -= g.mean()
        
        # Nesterov momentum
        v = momentum * v + g
        
        # Pair swap: transfer mass from high-v to low-v
        # Actually: g tells us direction of improvement
        # We want to decrease the objective, so move against gradient
        # grad[i] > 0 means increasing h[i] increases obj -> decrease h[i]
        # grad[i] < 0 means increasing h[i] decreases obj -> increase h[i]
        
        # Sort by gradient (most positive first = should decrease)
        order_dec = np.argsort(-v)  # most positive v first
        order_inc = np.argsort(v)   # most negative v first
        
        d = eta0 / (1 + np.linalg.norm(v))
        # Clamp to feasible
        for _ in range(K):
            i = order_dec[rng.randint(K)]  # decrease this
            j = order_inc[rng.randint(K)]  # increase this
            if i == j:
                continue
            step = min(d, x[i] * 0.5, (1-x[j]) * 0.5)
            if step < 1e-10:
                continue
            x[i] -= step
            x[j] += step
            # Check improvement
            c_new = compute_all_lags(x, n)
            if c_new.max() < best_c:
                best_c = c_new.max()
                best_x = x.copy()
            else:
                x[i] += step
                x[j] -= step
    
    return best_x, best_c

OK, let me also think about a completely different approach: **coordinate descent with exact line search**.

For each coordinate h[j], given all other coordinates fixed, the objective is:
f(h[j]) = max_k [c_k(h[j])]

where c_k(h[j]) = c_k(0) + h[j] * (∂c_k/∂h[j] at h[j]=0) for the parts that depend on h[j].

Actually, c_k is linear in each coordinate h[j] (since it's a sum of products h[i]*(1-h[j]) which is linear in h[j] for fixed i). So for a fixed k:
c_k = A_k + B_k * h[j]

where A_k and B_k are constants (given other h values).

The objective max_k (A_k + B_k * h[j]) is a piecewise linear function of h[j]. The minimum of this over h[j] ∈ [0,1] (while maintaining the sum constraint) can be found by checking the breakpoints.

But the sum constraint couples all coordinates, so this isn't quite right. In coordinate descent, we fix the sum: if we change h[j], we must change another coordinate to maintain the sum. So it's really "pair coordinate descent."

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx - np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, t_end, maxiter=5000, pen=1e6):
    target = n / 2.0
    def fg(x):
        x = np.clip(x, 0, 1)
        c = compute_all_lags(x, n)
        obj = smooth_max_value(c, alpha)
        diff = x.sum() - target
        obj += pen * diff * diff
        g = grad_smooth_max(x, n, alpha)
        g += 2.0 * pen * diff
        return obj, g
    try:
        res = minimize(fg, h, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def nesterov_pair_optimize(h, n, t_end, rng, alpha=100000, K=15, eta0=0.015, momentum=0.85):
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_c = compute_c5(x, n)
    momentum_cur = momentum
    eta_cur = eta0
    no_improve = 0

    while time.time() < t_end:
        c = compute_all_lags(x, n)
        cur_max = c.max()
        w = smooth_max_weights(c, alpha)
        g = grad_smooth_max(x, n, alpha)
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        v = momentum_cur * v + g

        # Adaptive step
        eta = eta_cur / (1.0 + g_norm * 10)
        eta = np.clip(eta, 0.0005, 0.05)

        # Find good pair: decrease where v is high, increase where v is low
        order_dec = np.argsort(-v)
        order_inc = np.argsort(v)

        improved = False
        for trial in range(K):
            i = order_dec[rng.randint(min(K, n))]
            j = order_inc[rng.randint(min(K, n))]
            if i == j:
                continue
            if v[i] <= 0 and v[j] >= 0:
                continue
            if x[i] - x[j] <= 0 and v[i] - v[j] <= 0:
                step_dir = 1.0
            else:
                step_dir = 1.0

            step = min(eta, x[i] * 0.45, (1.0 - x[j]) * 0.45)
            if step < 1e-10:
                continue

            x[i] -= step
            x[j] += step
            c_new = compute_all_lags(x, n)
            if c_new.max() < best_c - 1e-15:
                best_c = float(c_new.max())
                best_x = x.copy()
                improved = True
                no_improve = 0
                break
            else:
                x[i] += step
                x[j] -= step

        if not improved:
            no_improve += 1
            if no_improve > 500:
                x = best_x + rng.normal(0, 0.003, n)
                x = project_feasible(x, n)
                v = np.zeros(n)
                no_improve = 0
            momentum_cur = max(0.5, momentum_cur * 0.999)
        else:
            momentum_cur = min(0.92, momentum_cur + 0.001)

    return best_x, best_c
