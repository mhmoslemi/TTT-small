import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    """Gradient of C_k (at correlation index shift_idx) with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        # C_k = dx * sum_{i=0}^{nv-1} h[i] * (1 - h[i+k])
        # dC_k/dh[i] = dx * (1 - h[i+k]) for i < nv
        # dC_k/dh[i] = dx * (-h[i-k]) for i >= k (i-k < nv)
        grad[0:nv] += (1.0 - h[k:n]) * dx
        if nv > 0 and k > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        # C_k = dx * sum_{i=kk}^{n-1} h[i] * (1 - h[i-kk])
        # dC_k/dh[i] = dx * (1 - h[i-kk]) for i >= kk
        # dC_k/dh[i] = dx * (-h[i+kk]) for i < nv (i+kk >= kk)
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        if nv > 0 and kk > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        grad += w[mm] * shift_grad_single(h, n, mm)
    return grad

def lbfgs_smooth(h, n, alpha, maxiter, deadline):

    bounds = [(0.0, 1.0)] * n
    
    def obj(x):

        return smooth_max_grad(x, n, alpha)

                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})

Specifically, at each iteration:
1. Linearize C_k(h) around the current h: C_k(h) ≈ C_k(h₀) + ∇C_k(h₀)ᵀ(h - h₀)
2. Solve the resulting LP:
   min t
   s.t. C_k(h₀) + ∇C_k(h₀)ᵀ(h - h₀) ≤ t for all k (active)
        0 ≤ h ≤ 1, Σh = n/2
3. Update h₀ = h*

This is the **linearization method** (also known as the cutting plane method or Kelley's method). It's simple, robust, and works well for problems of this type.

The LP has:
- Variables: h (n vars) + t (1 var) = n+1 variables
- Constraints: |active| (linearization) + 2n (box) + 1 (equality) + 1 (bound on t) ≈ n + |active| + 3

For n=500 and |active|=10, that's about 513 constraints. This is a small LP that HiGHS can solve in milliseconds.

def linearization_step(h, n, active_indices, t_bound=1.0):
    """One step of the linearization (cutting plane) method.
    Returns a new h and the predicted t."""
    from scipy.optimize import linprog
    
    dx = 2.0 / n
    c_vec = compute_c5_vec(h, n)
    A = len(active_indices)
    
    # LP: min t
    # s.t. C_k(h) ≈ c_k + grad_k . (h - h₀) ≤ t  for k in active
    #      -t ≤ 0  (t is unbounded below, but we have a lower bound from the problem)
    #      0 ≤ h[i] ≤ 1
    #      sum(h) = n/2
    #
    # Variables: [h(0), h(1), ..., h(n-1), t]
    # Objective: [0, 0, ..., 0, 1]
    
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: grad_k . h - t <= c_k - grad_k . h₀
    # i.e., grad_k . h - t <= c_k - grad_k . h₀
    # Rearranged: grad_k . h - t + grad_k . h₀ <= c_k
    # But this isn't quite right. Let me redo.
    #
    # Linearization: C_k(h) ≈ C_k(h₀) + ∇C_k(h₀)ᵀ (h - h₀)
    # = C_k(h₀) + ∇C_k(h₀)ᵀ h - ∇C_k(h₀)ᵀ h₀
    # We want: C_k(h₀) + ∇C_k(h₀)ᵀ h - ∇C_k(h₀)ᵀ h₀ ≤ t
    # i.e., ∇C_k(h₀)ᵀ h - t ≤ ∇C_k(h₀)ᵀ h₀ - C_k(h₀)
    
    A_ub = np.zeros((A + 2*n, n_var))
    b_ub = np.zeros(A + 2*n)
    
    for j, idx in enumerate(active_indices):
        g = shift_grad_single(h, n, idx)
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
        b_ub[j] = g @ h - c_vec[idx]
    
    # Box constraints: h[i] ≤ 1 → h[i] ≤ 1
    for i in range(n):
        A_ub[A + i, i] = 1.0
        b_ub[A + i] = 1.0
        A_ub[A + n + i, i] = -1.0
        b_ub[A + n + i] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                 bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    return None, None

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if k > 0 and nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if kk > 0 and nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        grad += w[mm] * shift_grad_single(h, n, mm)
    return grad

def lbfgs_optimize(h, n, deadline, alpha=5000, maxiter=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, n, alpha)

    def grad(x):
        return smooth_max_grad(x, n, alpha)

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': 50000})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

    for k in range(1, M + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * k * x)
        h += b * np.sin(np.pi * k * x)
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    h = h / h.sum() * (n / 2.0)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def fourier_optimize(h, n, M, deadline, maxiter=2000):
    """Optimize h in a low-dimensional Fourier subspace."""
    x = np.linspace(0, 2, n)
    # Build Fourier design matrix
    cols = [np.ones(n)]
    for k in range(1, M + 1):
        cols.append(np.cos(np.pi * k * x))
        cols.append(np.sin(np.pi * k * x))
    Phi = np.column_stack(cols)
    n_params = Phi.shape[1]

    # Initial coefficients: least squares fit
    coeffs, _, _, _ = np.linalg.lstsq(Phi, h, rcond=None)

    def fourier_to_h(c):
        h_new = Phi @ c
        h_new = np.clip(h_new, 0, 1)
        return project_feasible(h_new, n)

    def obj_fourier(c):
        h_tmp = fourier_to_h(c)
        return smooth_max_obj(h_tmp, n, 2000)

    c0 = coeffs.copy()
    bounds_f = [(-1.0, 1.0)] * n_params
    try:
        res = minimize(obj_fourier, c0, method='Nelder-Mead',
                      options={'maxiter': maxiter * 50, 'xatol': 1e-8, 'fatol': 1e-10,
                              'adaptive': True})
        h_new = fourier_to_h(res.x)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def subgradient_momentum(h, n, deadline, alpha_init=0.01, beta=0.9, max_iter=2000):
    """Subgradient method with momentum for the smooth max objective."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    total_mom = n / 2.0  # target sum

    for it in range(max_iter):
        if time.time() > deadline - 3:
            break

        g = smooth_max_grad(h, n, 2000)
        # Add constraint violation gradient
        sum_viol = h.sum() - n / 2.0
        g += 2.0 * sum_viol * np.ones(n)

        v = beta * v - alpha * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Keep step size
            if alpha < alpha_init:
                alpha = min(alpha * 1.05, alpha_init)
        else:
            h = h_new if c_new < compute_c5(h, n) else h  # keep old if worse
            alpha *= 0.95

        if alpha < 1e-10:
            break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    # Load parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n = 400

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Quick L-BFGS warmup on parent (40s)
    t1 = min(t0 + 40, deadline - 700)
    if time.time() < t1 - 5:
        h1, c1 = lbfgs_optimize(h, n, t1, alpha=3000, maxiter=5000)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Linearization method (80s)
    t2 = min(t0 + 120, deadline - 600)
    if time.time() < t2 - 5:
        h2, c2 = linearization_optimize(best_h, n, t2, max_iter=300, rho_init=0.06)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient with momentum (40s)
    t3 = min(t0 + 160, deadline - 550)
    if time.time() < t3 - 5:
        h3, c3 = subgradient_momentum(best_h, n, t3, alpha_init=0.005, max_iter=3000)
    else:
        h3, c3 = best_h, best_c
    h3 = project_feasible(h3, n)
    c3 = compute_c5(h3, n)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: L-BFGS again (30s)
    t4 = min(t0 + 190, deadline - 500)
    if time.time() < t4 - 5:
        h4, c4 = lbfgs_optimize(best_h, n, t4, alpha=8000, maxiter=3000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Linearization again (60s)
    t5 = min(t0 + 250, deadline - 420)
    if time.time() < t5 - 5:
        h5, c5 = linearization_optimize(best_h, n, t5, max_iter=400, rho_init=0.04)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Pairwise search - fine (100s)
    t6 = min(t0 + 350, deadline - 320)
    if time.time() < t6 - 5:
        h6, c6 = pairwise_search(best_h, n, t6, n_top=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try Fourier initialization from scratch (40s)
    t7 = min(t0 + 390, deadline - 280)
    if time.time() < t7 - 5:
        for fs in [seed, seed + 100, seed + 200]:
            if time.time() > t7 - 5:
                break
            h_f = fourier_init(n, 12, fs)
            c_f = compute_c5(h_f, n)
            if c_f < best_c + 0.02:  # Only if reasonably close
                h_f2, c_f2 = lbfgs_optimize(h_f, n, t7 - 5, alpha=5000, maxiter=2000)
                if c_f2 < best_c:
                    best_c = c_f2
                    best_h = h_f2

    # Phase 8: Upsample to higher resolution (200s)
    n_high = max(1000, 2 * n)
    t8_start = max(time.time(), t0 + 400)
    t8 = min(t8_start + 100, deadline - 200)
    if time.time() < t8 - 30:
        h_hi = upsample(best_h, len(best_h), n_high)
        c_hi = compute_c5(h_hi, n_high)

        # Optimize at higher resolution
        t8a = min(time.time() + 40, t8 - 30)
        if time.time() < t8a - 5:
            h_hi, c_hi = lbfgs_optimize(h_hi, n_high, t8a, alpha=5000, maxiter=3000)

        t8b = min(time.time() + 40, t8 - 10)
        if time.time() < t8b - 5:
            h_hi, c_hi = linearization_optimize(h_hi, n_high, t8b, max_iter=200, rho_init=0.05)

        t8c = min(time.time() + 25, deadline - 100)
        if time.time() < t8c - 5:
            h_hi, c_hi = pairwise_search(h_hi, n_high, t8c, n_top=100)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            n = n_high

    # Phase 9: Another round of linearization on best (60s)
    t9 = min(t0 + 620, deadline - 80)
    if time.time() < t9 - 5:
        n_cur = len(best_h)
        h9, c9 = linearization_optimize(best_h, n_cur, t9, max_iter=400, rho_init=0.03)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Very fine pairwise search (80s)
    t10 = min(t0 + 700, deadline - 40)
    if time.time() < t10 - 5:
        n_cur = len(best_h)
        h10, c10 = pairwise_search(best_h, n_cur, t10, n_top=150)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 11: SLSQP final polish (remaining)
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h11, c11 = slqp_polish(best_h, n_cur, deadline - 5, alpha=500000)
        if c11 < best_c:
            best_c = c11
            best_h = h11

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
