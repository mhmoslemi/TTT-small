So Δc_k = dx * [δ*(1-h[i+k]) - δ*h[i-k]] = δ*dx*(1-h[i+k]-h[i-k])

(with appropriate bounds: h[i+k] only exists if i+k < n, h[i-k] only if i-k ≥ 0)

This is O(1) per k! So the total change for all k is O(n) per flip, and a full sweep is O(n²).

For n=500, a full sweep is 250k operations. Very fast!

So the algorithm for binary local search:
1. Threshold h to binary
2. Repeat until no improvement:
   a. For each position i, compute the "gain" of flipping: new_max_c - old_max_c
   b. Flip the position with the best (most negative) gain
   c. Update c_vec incrementally
3. The resulting binary pattern is the local optimum

Then we can use this as initialization for continuous optimization.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smoothed_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smoothed_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 30:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:30]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0
            rho = min(rho * 1.4, 0.15)
        else:
            consecutive_no_improve += 1
            rho *= 0.4
            if rho < 1e-11:
                break
            if consecutive_no_improve > 30:
                break
    return best_h, best_c

def subgrad_optimize(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_c_vec = compute_c5_vec(h, n)
    step = 0.02
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_cur = c_vec.max()
        thresh = c_cur - max(1e-10, 0.002 * c_cur)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= len(active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        trial_h = project_feasible(h - step * grad / grad_norm, n)
        trial_c = compute_c5(trial_h, n)
        if trial_c < best_c - 1e-15:
            best_c = trial_c
            best_h = trial_h.copy()
            h = trial_h
            stall = 0
            step = min(step * 1.1, 0.05)
        else:
            stall += 1
            step *= 0.7
            if step < 1e-7:
                break
            if stall > 100:
                step = 0.01
                stall = 0
                h = best_h.copy()
                if time.time() > deadline:
                    break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (1000, 500.0), (3000, 2000.0), (8000, 5000.0), (20000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smoothed_obj, h, jac=smoothed_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 40000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c_vec = compute_c5_vec(h_bin, n)
    best_c = c_vec.max()
    best_h_bin = h_bin.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if h_bin[i] > 0.5:
                delta = dx
            else:
                delta = dx
            flip_val = 1.0 - 2.0 * h_bin[i]
            new_c_max = best_c
            flip_improved = False
            for k_idx in range(n, 2 * n - 1):
                k = k_idx - (n - 1)
                dc = 0.0
                if k >= 0:
                    j = i
                    if j < n - k:
                        dc += flip_val * (1.0 - h_bin[j + k])
                    j2 = i - k
                    if 0 <= j2 < n - k:
                        dc -= flip_val * h_bin[j2]
                else:
                    kk = -k
                    j = i
                    if j >= kk:
                        dc += flip_val * (1.0 - h_bin[j - kk])
                    j2 = i + kk
                    if j2 < n:
                        dc -= flip_val * h_bin[j2]
                dc *= dx
                old_val = c_vec[k_idx]
                new_val = old_val + dc
                if new_val > new_c_max:
                    new_c_max = new_val
                    if new_c_max > best_c + 1e-12:
                        break
            if new_c_max < best_c - 1e-13:
                h_bin[i] = 1.0 - h_bin[i]
                for k_idx in range(2 * n - 1):
                    k = k_idx - (n - 1)
                    dc = 0.0
                    if k >= 0:
                        j = i
                        if j < n - k:
                            dc += flip_val * (1.0 - h_bin[j + k])
                        j2 = i - k
                        if 0 <= j2 < n - k:
                            dc -= flip_val * h_bin[j2]
                    else:
                        kk = -k
                        j = i
                        if j >= kk:
                            dc += flip_val * (1.0 - h_bin[j - kk])
                        j2 = i + kk
                        if j2 < n:
                            dc -= flip_val * h_bin[j2]
                    c_vec[k_idx] += dc * dx
                best_c = c_vec.max()
                best_h_bin = h_bin.copy()
                improved = True
        if not improved:
            break
    return best_h_bin, best_c

def fine_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    active = np.where(c_vec >= max_val - 0.001 * max_val)[0]
    grad_abs = np.zeros(n)
    for idx in active:
        grad_abs += np.abs(shift_grad_single(h, n, idx))
    if len(active) > 0:
        grad_abs /= len(active)
    n_top = min(100, n)
    top_positions = np.argsort(-grad_abs)[:n_top]
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i_val in top_positions:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                if h[i] < 1e-6 or h[i] > 1 - 1e-6:
                    continue
                for direction in [step, -step]:
                    h_trial = h.copy()
                    delta = direction
                    j_candidates = np.argsort(-((1 - h) if delta > 0 else h)[np.arange(n) != i])
                    for j_val in j_candidates[:50]:
                        j = int(j_val)
                        if j == i:
                            continue
                        if delta > 0 and h[j] < delta + 1e-12:
                            continue
                        if delta < 0 and h[j] > -delta - 1e-12:
                            continue
                        h_trial = h.copy()
                        h_trial[i] += delta
                        h_trial[j] -= delta
                        h_trial = project_feasible(h_trial, n)
                        c_trial = compute_c5(h_trial, n)
                        if c_trial < best_c - 1e-15:
                            best_c = c_trial
                            best_h = h_trial.copy()
                            h = h_trial
                            improved_any = True
                            break
    return best_h, best_c

def bfgs_polish(h, n, deadline, alpha):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smoothed_obj, h, jac=smoothed_grad,
                      args=(n, alpha, 0.0), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 2000, 'ftol': 1e-20,
                               'gtol': 1e-18, 'maxfun': 20000, 'maxls': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh)))) / alpha
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
