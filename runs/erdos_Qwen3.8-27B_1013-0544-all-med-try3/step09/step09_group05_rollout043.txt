```python
def periodic_solve(p, n, deadline):
    """Solve the periodic subproblem and expand to length n."""
    # p variables, p shifts
    # min max_r (2/p) * sum_i h[i]*(1-h[(i+r)%p])
    # s.t. 0<=h<=1, sum(h)=p/2
    
    def period_obj(hp, alpha):
        # Compute all p correlation values
        vals = []
        for r in range(p):
            s = np.sum(hp * (1 - np.roll(hp, r)))
            vals.append((2.0/p) * s)
        vals = np.array(vals)
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    
    def period_grad(hp, alpha):
        vals = np.zeros(p)
        for r in range(p):
            vals[r] = (2.0/p) * np.sum(hp * (1 - np.roll(hp, r)))
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(p)
        for r in range(p):
            if w[r] < 1e-14:
                continue
            # d/dr of sum_i h[i]*(1-h[(i+r)%p])
            # = sum_i [(1-h[(i+r)%p]) - h[i]*delta(i+r, j) for j]
            # For each i, contributes (1-h[(i+r)%p]) to grad[i]
            # and -h[i] to grad[(i+r)%p]
            rolled = np.roll(hp, r)
            grad[:] += w[r] * (2.0/p) * (1 - rolled)
            grad[:] -= w[r] * (2.0/p) * np.roll(hp, -r)
        # Constraint gradient
        grad += 0.01 * (hp.sum() - p/2.0)  # small penalty
        return grad
    
    # Initial: uniform
    hp = np.full(p, 0.5)
    bounds = [(0.0, 1.0)] * p
    
    best_hp = hp.copy()
    best_val = np.inf
    
    for alpha in [10, 100, 1000, 10000, 100000]:
        res = minimize(period_obj, hp, jac=period_grad, args=(alpha,),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-16})
        hp_new = np.clip(res.x, 0, 1)
        # Project to sum constraint
        hp_new = project_feasible(hp_new, p)
        
        # Evaluate
        vals = np.array([(2.0/p) * np.sum(hp_new * (1 - np.roll(hp_new, r))) for r in range(p)])
        val = vals.max()
        if val < best_val:
            best_val = val
            best_hp = hp_new.copy()
        hp = hp_new
    
    # Expand to length n
    m = n // p
    h_full = np.tile(best_hp, m)
    if len(h_full) < n:
        h_full = np.concatenate([h_full, best_hp[:n - len(h_full)]])
    h_full = h_full[:n]
    h_full = project_feasible(h_full, n)
    c = compute_c5(h_full, n)
    
    return h_full, c
```
