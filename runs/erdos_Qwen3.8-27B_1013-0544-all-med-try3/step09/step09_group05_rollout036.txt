For the L-BFGS with log-sum-exp:

- But large alpha makes the problem harder to optimize (sharper minima).

For the local search:

- Use the gradient to prioritize which pairs to try.
- Very small δ (1e-5 to 1e-4) for fine tuning.

Let me also add a technique that I think could be effective: **correlation target**. 

The idea: after computing the correlations, identify the max value C_max. Then, for a few iterations:
1. Pick the shift k* that achieves the max
2. Compute the gradient of correlation_{k*}
3. Take a small step in the negative gradient direction
4. Project to feasible set
5. Check if the max decreased (or if other correlations increased too much)

This is essentially a subgradient method for the minimax problem. It's simpler than L-BFGS but can be more effective for the actual (non-smooth) objective.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        sign = 1.0 if diff > 0 else -1.0
        amt = abs(diff) / n
        h += sign * amt
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_gradient(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_objective(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_gradient(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_shifts(c_vec, n, tol_frac=0.005, max_count=20):
    max_val = c_vec.max()
    threshold = max_val - tol_frac * max_val
    active = np.where(c_vec >= threshold)[0]
    if len(active) > max_count:
        order = np.argsort(-c_vec[active])
        active = active[order[:max_count]]
    return active

def subgradient_step(h, n, step_size):
    """Take a step in the negative subgradient direction of the max correlation."""
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    max_indices = np.where(np.abs(c_vec - max_val) < 1e-10)[0]
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in max_indices:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[kk:n])
    grad /= len(max_indices)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def correlation_aware_pairs(h, n, c_vec, top_k_indices, radius=None):
    """Generate pairs that are likely to reduce the top correlations."""
    if radius is None:
        radius = max(10, n // 50)
    pairs = []
    for idx in top_k_indices:
        k = idx - (n - 1)
        abs_k = abs(k)
        for i in range(0, n - abs_k, max(1, (n - abs_k) // 50)):
            if k >= 0:
                j = i + k
            else:
                j = i - k
            if 0 <= i < n and 0 <= j < n and i != j:
                pairs.append((i, j))
    # Add some local pairs
    for _ in range(min(200, n)):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i != j:
            pairs.append((i, j))
    return pairs

def local_pair_search(h, n, deadline, base_step=0.005, max_iters=50000):
    """Aggressive local 2-point search focusing on top correlations."""
    best_c = compute_c5(h, n)
    n_iter = 0
    step = base_step
    patience = 0
    
    while time.time() < deadline and n_iter < max_iters:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - 0.01 * max_val
        active = np.where(c_vec >= threshold)[0]
        
        # Generate candidate pairs
        n_try = min(200, max(50, n // 10))
        improvements = 0
        
        for _ in range(n_try):
            if time.time() > deadline:
                break
            n_iter += 1
            if len(active) > 0:
                idx = active[np.random.randint(len(active))]
                k = idx - (n - 1)
                abs_k = abs(k)
                if abs_k == 0:
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                elif abs_k >= n:
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                else:
                    i = np.random.randint(0, n - abs_k)
                    j = i + abs_k
                    if np.random.random() < 0.5:
                        i, j = j, i
            else:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            if i == j:
                continue
            
            # Try both directions
            for direction in [1, -1]:
                h[i] += direction * step
                h[j] -= direction * step
                h[i] = np.clip(h[i], 0, 1)
                h[j] = np.clip(h[j], 0, 1)
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improvements += 1
                else:
                    h[i] -= direction * step
                    h[j] += direction * step
                    h = project_feasible(h, n)
        
        if improvements == 0:
            patience += 1
            if patience > 10:
                step *= 0.7
                patience = 0
                if step < 1e-7:
                    break
        else:
            patience = 0
    
    return h, best_c

def lbfgs_optimize(h, n, deadline, alpha, mu, maxiter=5000, label=""):
    """Run L-BFGS with given parameters, return improved solution if found."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    obj = lambda x: lse_objective(x, n, alpha, mu)
    jac = lambda x: lse_gradient(x, n, alpha, mu)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': maxiter * 5, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    except Exception:
        pass
    return h, best_c

def topk_lbfgs(h, n, deadline, top_k, alphas, mu=1.0):
    """Iteratively optimize focusing on top-k correlations."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        for _ in range(3):
            if time.time() > deadline - 5:
                break
            c_vec = compute_c5_vec(h, n)
            active = get_active_shifts(c_vec, n, tol_frac=0.01, max_count=top_k)
            if len(active) == 0:
                break
            obj = lambda x: topk_objective(x, n, active, alpha, mu)
            jac = lambda x: topk_gradient(x, n, active, alpha, mu)
            try:
                res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': 15000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = h_new
                else:
                    break
            except Exception:
                break
    return h, best_c

def upsample(h_old, n_old, n_new):
    """Upsample from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    """Downsample from n_old to n_new points."""

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
