def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n, tol=1e-14):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < tol:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient_fast(h, n, lag):
    """Compute gradient of c(k) with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:

        # d/dh[i] = (1-h[i+k]) for i < n-k
        #           -h[i-k] for i >= k
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += (-h[:-m])
    else:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])

def equalize_top_lags(h, n, deadline, top_m=5, lr=0.01):

    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]

        cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()

        return float(m + np.log(np.sum(np.exp(shifted)))) / alpha
    
    def obj_grad(x):
        dx_ = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx_
        active = cc[top_idx]
        m = active.max()

                continue
            lag = k_idx - (n - 1)
            g += w[idx] * lag_gradient_fast(x, n, lag)
        return g * dx_
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-m] += 1.0 - h[m:]
        g[m:] += -h[:n-m]
    else:
        m = -lag
        g[:n-m] += 1.0 - h[m:]
        g[m:] += -h[:n-m]
    return g

def smooth_max_and_grad(h, n, lags, alpha):
    """Compute smooth max over given lag indices and its gradient."""
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    vals = c_all[lags]
    m = vals.max()
    s = alpha * (vals - m)
    sm = s.max()
    if sm > 500:
        obj = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    else:
        obj = m + np.log(np.sum(np.exp(s)))) / alpha
    
    w = np.exp(s - sm) if sm > 500 else np.exp(s)
    w /= w.sum()
    
    g = np.zeros(n)
    for j, li in enumerate(lags):
        if w[j] > 1e-12:
            lag = li - (n - 1)
            g += w[j] * lag_grad(h, n, lag)
    g *= dx
    return obj, g

def project_grad_descent(h, n, deadline, lags=None, alpha0=500, alpha1=50000,
                         lr0=0.5, mom=0.9, top_m=10):
    """Projected gradient descent with momentum and adaptive alpha."""
    if lags is None:
        c_all = compute_all_lags(h, n)
        lags = np.argsort(c_all)[-top_m:]
        lags.sort()
    
    h_c = h.copy()
    best_h = h_c.copy()
    best_c = compute_c5(best_h, n)
    v = np.zeros(n)
    it = 0
    lr = lr0
    
    while time.time() < deadline:
        # Adaptive alpha
        prog = min(1.0, it / 5000.0)
        alpha = alpha0 * (alpha1 / alpha0) ** prog
        
        _, g = smooth_max_and_grad(h_c, n, lags, alpha)
        v = mom * v + lr * g
        h_new = project_feasible(h_c - v, n)
        c_new = compute_c5(h_new, n)
        it += 1
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h_c = h_new
        
        # Update lags every 200 iters
        if it % 200 == 0:
            c_all = compute_all_lags(h_c, n)
            lags = np.argsort(c_all)[-top_m:]
            lags.sort()
        
        # Decay learning rate
        if it % 1000 == 0:
            lr *= 0.85
            v *= 0.5
    
    return best_h, best_c

def local_search(h, n, deadline, top_m=12, max_rounds=100):
    """Iterative local search focusing on reducing max correlation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_m:]
        improved = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            
            n_pairs = n - m
            step = max(1, n_pairs // 30)
            for i in range(0, n_pairs, step):
                if time.time() > deadline:
                    break
                j = i + m
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                
                # Determine direction: reduce h[i] if h[i] > h[j] (to reduce c at this lag)
                # c(lag) ∝ h[i](1-h[j]) for the pair (i,j) at this lag
                # To reduce c(lag), we want to decrease h[i](1-h[j])
                # If h[i] > h[j]: decrease h[i], increase h[j]
                # If h[i] < h[j]: increase h[i], decrease h[j]
                
                d = h[i] - h[j]
                for delta in [0.02, 0.01, 0.005, 0.001]:
                    if d > 0:
                        delta_use = min(delta, d * 0.3)
                        if delta_use < 1e-8:
                            continue
                        h[i] -= delta_use
                        h[j] += delta_use
                    else:
                        delta_use = min(delta, (-d) * 0.3)
                        if delta_use < 1e-8:
                            continue
                        h[i] += delta_use
                        h[j] -= delta_use
                    
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        if d > 0:
                            h[i] += delta_use
                            h[j] -= delta_use
                        else:
                            h[i] -= delta_use
                            h[j] += delta_use
                        break
        
        if not improved:
            break
    
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter, top_m):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    top_idx = list(top_idx)
    
    def obj(x):
        dx_ = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx_
        vals = cc[top_idx]
        m = vals.max()
        s = alpha * (vals - m)
        sm = s.max()
        if sm > 500:
            return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
        return float(m + np.log(np.sum(np.exp(s)))) / alpha
    
    def obj_jac(x):
        dx_ = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx_
        vals = cc[top_idx]
        m = vals.max()
        s = alpha * (vals - m)
        sm = s.max()
        w = np.exp(s - sm) if sm > 500 else np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for j, ti in enumerate(top_idx):
            if w[j] < 1e-12:
                continue
            lag = ti - (n - 1)
            g += w[j] * lag_grad(x, n, lag)
        return g * dx_
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g

    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] += -h[:n-m]
    return g
