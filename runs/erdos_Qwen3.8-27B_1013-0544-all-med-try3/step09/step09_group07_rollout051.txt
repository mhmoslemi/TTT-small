import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    return w / w.sum()

def compute_gradient(h, n, alpha):
    """Compute gradient of smooth max of all lags w.r.t. h"""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    offset = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - offset
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0 and lag < n:
            # c[k] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i]) * dx
            # d/dh[j] = dx * (1-h[j-lag]) for j=lag..n-1 (h[j] is the "h[i+lag]" term)
            #          + dx * (-h[j+lag]) for j=0..n-1-lag (h[j] is the "1-h[i]" term)
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] += wk * (-h[lag:n])
        elif lag < 0 and lag > -n:
            idx = -lag
            # c[k] = sum_{i=0}^{n-1-idx} h[i] * (1-h[i+idx]) * dx
            # d/dh[j] = dx * (1-h[j+idx]) for j=0..n-1-idx
            #          + dx * (-h[j-idx]) for j=idx..n-1
            grad[0:n-idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n-idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e10):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = compute_gradient(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1, 'maxfun': maxiter * 3})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def gradient_descent_local(h, n, deadline, lr=0.001, max_iter=50000):
    """Simple projected gradient descent on the smooth max"""
    alpha = 100000
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Adaptive alpha
        c_all = compute_all_lags(h, n)
        top_vals = np.sort(c_all)[-10:]
        gap = top_vals[0] - top_vals[-1]
        alpha = max(1000, 500.0 / max(gap, 1e-10))
        alpha = min(alpha, 5000000)

        grad = compute_gradient(h, n, alpha)
        # Add sum constraint gradient
        diff = h.sum() - target_sum
        grad += 2.0 * (diff / n) * n  # penalty gradient

        # Take step
        step = lr / (np.max(np.abs(grad)) + 1e-10)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.01)
        else:
            lr *= 0.9
            if lr < 1e-7:
                break

    return best_h, best_c

            elif h[j] > h[i]:

def binary_to_continuous(h, n, deadline, rng):
    """Start from binary, then relax"""
    # Threshold
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1

    # Relax using L-BFGS-B
    h_relaxed = lbfgsb_optimize(h_bin.astype(float), n, 50000, time.time() + 30, maxiter=2000)
    return h_relaxed
