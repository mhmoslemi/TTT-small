                    best_h = h_ref
                    pop[0] = best_h.copy()
                    fitness[0] = c_ref

        # Inject diversity if stuck
        if no_improve > pop_size * 8:
            idx = rng.randint(pop_size)
            pop[idx] = rng.random(n)
            pop[idx] = project_feasible(pop[idx], n)
            fitness[idx] = compute_c5(pop[idx], n)
            # Also try perturbation of best
            idx2 = rng.randint(pop_size)
            pop[idx2] = best_h + 0.05 * (2 * rng.random(n) - 1)
            pop[idx2] = project_feasible(pop[idx2], n)
            fitness[idx2] = compute_c5(pop[idx2], n)
            no_improve = 0

    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Random subset of positions to check
            n_checks = min(200, n)
            indices = np.random.choice(n, n_checks, replace=False)
            for i in indices:
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                candidates = np.random.choice(hi - lo, min(10, hi - lo - 1), replace=False) + lo
                for j in candidates:
                    if j == i:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_step(h, n, deadline, n_iter=20000, eta_init=0.003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta_init
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 10:
            order = np.argsort(-c_vec[active])
            active = active[order[:10]]
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = np.clip(h - eta * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.005, 0.01)
        else:
            eta *= 0.98
            if eta < 1e-6:
                break
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Load initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: DE at n=300 for global exploration (100s)
    n_de = 300
    p1_deadline = min(t0 + 100, deadline - 500)
    if p1_deadline > t0 + 20:
        h_de, c_de = de_search(n_de, pop_size=60, n_gen_max=500,
                               deadline=p1_deadline, rng=rng)
        if c_de < best_c:
            # Upsample DE result to n_orig for fair comparison
            h_de_up = upsample(h_de, n_de, n_orig)
            c_de_up = compute_c5(h_de_up, n_orig)
            if c_de_up < best_c:
                best_c = c_de_up
                best_h = h_de_up
            best_h_de = h_de
            best_c_de = c_de
        else:
            best_h_de = h_de
            best_c_de = c_de

    # Phase 2: L-BFGS polish at n=500 (60s)
    p2_deadline = min(t0 + 160, deadline - 400)
    if time.time() < p2_deadline:
        h2, c2 = lbfgs_opt(best_h, best_n, p2_deadline,
                           alphas=[500, 5000, 50000, 200000, 1000000],
                           mu=0.005, maxiters_list=[500, 800, 1000, 1200, 1500])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=1000 and optimize (300s)
    n_mid = 1000
    p3_deadline = min(t0 + 500, deadline - 300)
    if time.time() < p3_deadline - 50:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        # Also try upsampled DE solution
        if 'best_h_de' in dir() or True:
            try:
                h_mid_de = upsample(best_h_de, n_de, n_mid)
                c_mid_de = compute_c5(h_mid_de, n_mid)
                if c_mid_de < c_mid:
                    h_mid = h_mid_de
                    c_mid = c_mid_de
            except Exception:
                pass
        if c_mid < best_c + 0.002:
            if c_mid < best_c:
                best_c = c_mid
                best_h = h_mid
                best_n = n_mid
            p3a_deadline = min(time.time() + 150, deadline - 350)
            h3a, c3a = lbfgs_opt(h_mid, n_mid, p3a_deadline,
                                  alphas=[200, 2000, 10000, 50000, 200000, 1000000, 5000000],
                                  mu=0.003, maxiters_list=[400, 600, 800, 1000, 1200, 1500, 2000])
            if c3a < best_c:
                best_c = c3a
                best_h = h3a
                best_n = n_mid

    # Phase 4: Active set optimization at n=1000 (200s)
    n_cur = len(best_h)
    p4_deadline = min(t0 + 700, deadline - 250)
    if time.time() < p4_deadline - 20:
        h4, c4 = active_opt_rounds(best_h, n_cur, p4_deadline,
                                   top_k=5,
                                   alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try n=2000 (200s)
    n_high = 2000
    p5_deadline = min(t0 + 900, deadline - 150)
    if time.time() < p5_deadline - 80 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.003:
            if c_high < best_c:
                best_c = c_high
                best_h = h_high
                best_n = n_high
            p5a = min(time.time() + 100, deadline - 160)
            h5, c5 = lbfgs_opt(h_high, n_high, p5a,
                                alphas=[100, 1000, 5000, 20000, 100000, 500000],
                                mu=0.002, maxiters_list=[300, 500, 700, 800, 1000, 1200])
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_high

    # Phase 6: Active set at final resolution (80s)
    n_cur = len(best_h)
    p6_deadline = min(t0 + 980, deadline - 60)
    if time.time() < p6_deadline - 10:
        h6, c6 = active_opt_rounds(best_h, n_cur, p6_deadline,
                                   top_k=4,
                                   alphas=[500000, 2000000, 10000000, 50000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Subgradient refinement (50s)
    n_cur = len(best_h)
    p7_deadline = min(t0 + 1030, deadline - 30)
    if time.time() < p7_deadline - 5:
        h7, c7 = subgrad_step(best_h, n_cur, p7_deadline, n_iter=20000, eta_init=0.005)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Pair search (remaining time)
    n_cur = len(best_h)
    p8_deadline = deadline - 2
    if time.time() < p8_deadline - 10:
        h8, c8 = pair_search(best_h, n_cur, p8_deadline, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
