import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def topk_obj(x, n, K, mu):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, min(K, len(c)-1))[:K]
    val = c[idx].mean()
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def topk_grad(x, n, K, mu):
    c = compute_c5_vec(x, n)
    idx = np.argpartition(-c, min(K, len(c)-1))[:K]
    grad = np.zeros(n)
    for j in idx:
        grad += shift_grad(x, n, j)
    grad /= K
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for mm in range(2*n-1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n-1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def opt_lse(h, n, deadline, alphas, maxiter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, 1e-4), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def opt_topk(h, n, deadline, K=5, maxiter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(20):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > K:
            active = np.argsort(-c_vec[active])[:K]
        K_use = max(1, len(active))
        try:
            res = minimize(topk_obj, best_h, jac=topk_grad,
                          args=(n, K_use, 1e-3), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
            else:
                break
        except Exception:
            break
    return best_h, best_c

def binary_local_search(h, n, deadline, n_iter=50000):
    # Convert to binary
    h_bin = (h >= 0.5).astype(float)
    # Adjust to have exactly n//2 ones
    n_ones = int(h_bin.sum())
    target_n = n // 2
    if n_ones > target_n:
        # Remove some ones
        order = np.argsort(h_bin * (-h))  # prefer removing ones where h is lowest
        for idx in order:
            if n_ones == target_n:
                break
            if h_bin[idx] == 1:
                h_bin[idx] = 0
                n_ones -= 1
    elif n_ones < target_n:
        order = np.argsort(-h_bin * (1 - h_bin))  # prefer adding zeros where h is highest
        # Actually, prefer positions where h is closest to 0.5 from above
        order = np.argsort(-(h * (1 - h_bin)))  # prefer positions where h is low
        # Simpler: just pick random positions
        for idx in np.random.permutation(n):
            if n_ones == target_n:
                break
            if h_bin[idx] == 0:
                h_bin[idx] = 1
                n_ones += 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    rng = np.random.RandomState(42)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick random pair to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0
        h_bin[j] = 1
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_bin.copy()
            # Update sets
            mask_i = (ones == i)
            ones = np.delete(ones, np.where(mask_i)[0][0])
            mask_j = (zeros == j)
            zeros = np.delete(zeros, np.where(mask_j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return best_h, best_c

def projected_subgradient(h, n, deadline, K=5, lr0=0.01, lr_decay=0.999):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    for it in range(100000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > K:
            active = np.argsort(-c_vec[active])[:K]
        
        grad = np.zeros(n)
        for j in active:
            grad += shift_grad(best_h, n, j)
        grad /= len(active)
        
        # Projected gradient step
        new_h = best_h - lr * grad
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = new_h
            # Adaptive lr: keep or slightly increase
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= lr_decay
            if lr < 1e-8:

    return best_h, best_c

    best_h = h.copy()

    dx = 2.0 / n

                return best_h, best_c
            orig = best_h[i]
            # Try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                new_val = orig + delta
                if new_val < 0 or new_val > 1:

                # Adjust another variable to maintain sum constraint
                # Find the variable with the most "room"
                j = (i + 1) % n
                adj = -delta
                new_j = best_h[j] + adj
                if new_j < 0 or new_j > 1:

                    improved = True
                    orig = new_val

                else:
                    best_h[i] = orig
                    best_h[j] = best_h[j] + delta  # undo
            best_h[i] = orig

            break
    
    return best_h, best_c
