import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_single(h, n, m_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = m_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-15:
            continue
        grad += w[mm] * compute_grad_single(h, n, mm)
    return grad

def momentum_descent(h, n, deadline, alpha_init=500, top_k=5, momentum_init=0.9):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.04
    momentum = momentum_init
    v = np.zeros(n)
    alpha = alpha_init
    
    no_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        c_vec = compute_c5_vec(h, n)
        current_max = c_vec.max()
        
        if current_max < best_c - 1e-15:
            best_c = current_max
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.15)
        else:
            no_improve += 1
            if no_improve > 500:
                break
            if no_improve > 100:
                lr *= 0.95
        
        # Focus on top-k worst shifts
        top_indices = np.argpartition(c_vec, -top_k)[-top_k:]
        top_vals = c_vec[top_indices]
        
        # Softmax weighting over top-k
        sh_w = alpha * 0.1 * (top_vals - top_vals.max())
        w_k = np.exp(sh_w)
        w_k /= w_k.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, wv in zip(top_indices, w_k):
            grad += wv * compute_grad_single(h, n, int(idx))

        grad += bin_reg

        h_new = h + v

            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)

                c_new = current_max

                continue
        else:

        # Slowly increase alpha (sharper max approximation)

            alpha = min(alpha * 1.2, 5000)
    
    return max(best_c, compute_c5(h, n)), best_h if compute_c5(best_h, n) <= compute_c5(h, n) else h

    """Use linearized subproblem on top-k constraints for direction finding."""
    from scipy.optimize import linprog
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    it = 0
    
    while time.time() < deadline and it < max_iter:
        it += 1
        c_vec = compute_c5_vec(h, n)
        current_max = c_vec.max()
        
        if current_max < best_c - 1e-14:
            best_c = current_max
            best_h = h.copy()
        
        # Find top-k active constraints
        top_indices = np.argpartition(c_vec, -top_k)[-top_k:]
        
        # Build LP: minimize t subject to gradient_i · delta + c_i <= t, ||delta||_inf <= eps
        eps = 0.02
        grads = [compute_grad_single(h, n, int(idx)) for idx in top_indices]
        c_vals = [c_vec[idx] for idx in top_indices]
        
        n_var = 2 * n + 1  # delta+ and delta- and t
        c_obj = np.zeros(n_var)
        c_obj[n:2*n] = 1.0  # minimize sum of |delta|
        
        # Constraints: grad_i · delta + c_i - t <= 0 for each i
        # delta = delta+ - delta-
        A_ub = np.zeros((top_k + 2*n, n_var))
        b_ub = np.zeros(top_k + 2*n)
        
        for j in range(top_k):
            A_ub[j, 0:n] = grads[j]
            A_ub[j, n:2*n] = -grads[j]
            A_ub[j, 2*n] = -1.0
            b_ub[j] = -c_vals[j]
        
        # delta+ >= 0, delta+ <= 1-h
        for i in range(n):
            A_ub[top_k + i, i] = -1.0
            b_ub[top_k + i] = -0.0
            A_ub[top_k + n + i, i] = 1.0
            b_ub[top_k + n + i] = 1.0 - h[i]
        
        # delta- >= -1, delta- <= -h  (i.e., h+delta->=0)
        for i in range(n):
            A_ub[top_k + i, n+i] = -1.0
            b_ub[top_k + i] = 0.0
            A_ub[top_k + n + i, n+i] = 1.0
            b_ub[top_k + n + i] = h[i]
        
        A_eq = np.zeros((1, n_var))
        b_eq = np.zeros(1)
        
        bounds = [(0, 1-h[i]) for i in range(n)] + [(0, h[i]) for i in range(n)] + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-4})
            if res.success:
                delta = res.x[0:n] - res.x[n:2*n]
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < current_max:
                    h = h_new
                else:
                    # Try smaller step
                    h_new = project_feasible(h + 0.3 * delta, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < current_max:
                        h = h_new
                    else:
                        eps *= 0.5
                        if eps < 0.001:
                            break
            else:
                eps *= 0.5
                if eps < 0.001:
                    break
        except Exception:
            eps *= 0.5
            if eps < 0.001:
                break
    
    return best_c, best_h

def fine_local_search(h, n, deadline, radius=30):
    """Very fine local search focusing on worst shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            current_max = c_vec.max()
            
            # Find worst shift and its gradient
            worst_idx = int(np.argmax(c_vec))
            grad = compute_grad_single(h, n, worst_idx)
            
            # Find best pair to swap
            # We want to increase h where grad < 0 and decrease where grad > 0
            neg_mask = grad < -1e-10
            pos_mask = grad > 1e-10
            
            pos_indices = np.where(pos_mask)[0]
            neg_indices = np.where(neg_mask)[0]
            
            if len(pos_indices) == 0 or len(neg_indices) == 0:
                break
            
            # Focus on indices near the worst shift region
            k_shift = worst_idx - (n - 1)
            center = max(0, min(n - 1, (n - 1) // 2))
            
            # Sort by gradient magnitude
            pos_sorted = pos_indices[np.argsort(-grad[pos_indices])][:radius * 2]
            neg_sorted = neg_indices[np.argsort(grad[neg_indices])][:radius * 2]
            
            best_delta = None
            best_c_trial = current_max
            
            for i in pos_sorted[:radius]:
                if h[i] < step:
                    continue
                for j in neg_sorted[:radius]:
                    if h[j] > 1.0 - step:
                        continue
                    h[i] -= step
                    h[j] += step
                    c_trial = compute_c5(h, n)
                    if c_trial < best_c_trial:
                        best_c_trial = c_trial
                        best_delta = (int(i), int(j), step)
                    h[i] += step
                    h[j] -= step
            
            if best_delta is not None:
                i, j, s = best_delta
                h[i] -= s
                h[j] += s
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
                    break
    
    return best_c, best_h

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha
    
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    
    def sq_jac_fn(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_thrash(h, n, threshold=0.02):
    """Push values near 0 or 1 to extremes, then fix sum."""
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    # Adjust remaining values to fix sum
    target = n / 2.0
    diff = target - h2.sum()
    free_mask = (h2 >= threshold) & (h2 <= 1.0 - threshold)
    if free_mask.sum() > 0 and abs(diff) > 1e-10:
        n_free = free_mask.sum()
        adj = diff / n_free
        h2[free_mask] = np.clip(h2[free_mask] + adj, threshold, 1.0 - threshold)
        # Second pass
        h2 = project_feasible(h2, n)
    else:
        h2 = project_feasible(h2, n)
    return h2

def runner_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (800, 200.0), (2000, 500.0), (5000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def jac(x):
            return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Upsample to n=1000 for higher resolution
    n_target = 1000
    if n_orig != n_target:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_target)
        h = np.interp(x_new, x_old, h_init)
    else:
        h = h_init.copy()
    n = n_target
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS warmup (40s)
    t1 = min(t0 + 40, deadline - 800)
    if t1 > time.time() + 3:
        h, best_c = runner_lbfgs(h, n, t1,
            configs=[(200, 30.0), (800, 100.0), (2000, 500.0)])
        best_h = h.copy()
    
    # Phase 2: Momentum gradient descent (250s)
    t2 = min(t0 + 290, deadline - 550)
    if t2 > time.time() + 5:
        h, c2 = momentum_descent(h, n, t2, alpha_init=300, top_k=5, momentum_init=0.85)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    # Phase 3: Focused LP descent (150s)
    t3 = min(t0 + 440, deadline - 400)
    if t3 > time.time() + 5:
        h, c3 = focused_lp_descent(h, n, t3, top_k=4, max_iter=500)
        if c3 < best_c:
            best_c = c3
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    # Phase 4: Another momentum pass (120s)
    t4 = min(t0 + 560, deadline - 300)
    if t4 > time.time() + 5:
        h, c4 = momentum_descent(h, n, t4, alpha_init=1000, top_k=3, momentum_init=0.7)
        if c4 < best_c:
            best_c = c4
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    # Phase 5: Binary thrash + re-optimize (60s)
    if time.time() < deadline - 200:
        h_th = binary_thrash(h, n, threshold=0.015)
        t5 = min(time.time() + 40, deadline - 160)
        if t5 > time.time() + 3:
            h_th, c_th = momentum_descent(h_th, n, t5, alpha_init=500, top_k=5, momentum_init=0.8)
            if c_th < best_c:
                best_c = c_th
                best_h = h_th.copy()
                h = h_th.copy()
    
    # Phase 6: Fine local search (100s)
    t6 = min(t0 + 720, deadline - 100)
    if t6 > time.time() + 5:
        h, c6 = fine_local_search(h, n, t6, radius=25)
        if c6 < best_c:
            best_c = c6
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    # Phase 7: SLSQP polish (60s)
    t7 = min(t0 + 830, deadline - 40)
    if t7 > time.time() + 5:
        h7, c7 = slqp_polish(h, n, t7, alpha=3000000, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    
    # Phase 8: Final fine local search (remaining time)
    if time.time() < deadline - 10:
        h, c8 = fine_local_search(h, n, deadline - 5, radius=15)
        if c8 < best_c:
            best_c = c8
            best_h = h.copy()
    
    # Final check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
