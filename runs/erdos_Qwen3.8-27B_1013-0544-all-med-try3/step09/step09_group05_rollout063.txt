```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx, dx):
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_lp(h, n, active_shifts, delta):
    """LP step to minimize linearized max over active shifts."""
    dx = 2.0 / n
    c_all = compute_c5_vec(h, n)
    m = len(active_shifts)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    
    for k_idx in active_shifts:
        g = shift_gradient(h, n, k_idx, dx)
        c_k = c_all[k_idx]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(c_k - g @ h)
    
    lower = np.concatenate([np.maximum(0.0, h - delta), [0.0]])
    upper = np.concatenate([np.minimum(1.0, h + delta), [np.inf]])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lower, upper)),
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), float('inf')

def get_active_shifts(h, n, top_k=20, threshold_frac=0.01):
    c = compute_c5_vec(h, n)
    max_c = c.max()
    threshold = max_c - threshold_frac * max_c
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c[active])][:top_k]
    return active, max_c

    best_c = compute_c5(h, n)
    delta = delta_init

        else:

            if delta < 1e-8:

    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [(200, 20.0), (1000, 200.0), (5000, 2000.0),
                  (20000, 10000.0), (100000, 50000.0)]

            break
        try:

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def top_k_smooth_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

def top_k_smooth_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def coordinate_descent(h, n, deadline):
    """Cycle through variables, adjusting each to minimize the top shift."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        max_idx = c_vec.argmax()
        max_val = c_vec[max_idx]
        active = np.where(c_vec >= max_val - 0.001 * max_val)[0]
        if len(active) > 10:
            active = active[np.argsort(-c_vec[active])][:10]
        
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try adjusting h[i] up or down
            for direction in [1, -1]:
                for delta in [0.001, 0.0005, 0.0002]:
                    trial = best_h.copy()
                    trial[i] += direction * delta
                    if 0 <= trial[i] <= 1:
                        trial = project_feasible(trial, n)
                        c_trial = compute_c5(trial, n)
                        if c_trial < best_c - 1e-14:
                            best_h = trial
                            best_c = c_trial
                            improved = True
                            break
                else:
                    continue
                break
        
        if not improved:
            break
    
    return best_h, best_c

def perturb_reoptimize(h, n, deadline, n_perturbs=5):
    """Try random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(42)
    
    per_perturb_time = max(10, (deadline - time.time()) / n_perturbs)
    
    for p in range(n_perturbs):
        if time.time() > deadline - 10:
            break
        
        t_sub = min(time.time() + per_perturb_time, deadline - 5)
        
        # Different types of perturbation
        h_pert = best_h.copy()
        if p % 3 == 0:
            # Smooth perturbation
            h_pert += 0.05 * rng.standard_normal(n)
        elif p % 3 == 1:
            # Block perturbation
            start = rng.integers(0, n // 2)
            length = rng.integers(n // 8, n // 4)
            end = min(n, start + length)
            h_pert[start:end] += 0.1 * rng.choice([-1, 1])
        else:
            # Sparse perturbation
            idx = rng.choice(n, size=n // 10, replace=False)
            h_pert[idx] += 0.05 * rng.standard_normal(len(idx))
        
        h_pert = project_feasible(h_pert, n)
        
        # Quick L-BFGS
        h_opt, c_opt = run_lbfgs(h_pert, n, t_sub,
            alphas=[(1000, 100.0), (5000, 1000.0), (20000, 10000.0)])
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def equalize_top_shifts(h, n, deadline, n_inner=50):
    """Explicitly try to equalize the top shifts using 2-variable adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for _ in range(n_inner):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Find the two closest-to-max shifts
        sorted_idx = np.argsort(-c_vec)
        worst = sorted_idx[0]  # the max
        second = sorted_idx[1]
        
        # The direction that reduces the worst shift
        g_worst = shift_gradient(best_h, n, worst, dx)
        
        # Find the variable that most reduces the worst shift
        # We want to move h in direction -g_worst, but simultaneously
        # not increase other active shifts too much
        active = sorted_idx[:10]
        g_active = np.array([shift_gradient(best_h, n, k, dx) for k in active])
        
        # Find best 2-point move
        best_improve = 1e-14
        best_move = None
        
        # Sort variables by their effect on the worst shift
        effect = np.abs(g_worst)
        top_vars = np.argsort(-effect)[:30]
        
        for i_idx in range(len(top_vars)):
            if time.time() > deadline:
                break
            i = top_vars[i_idx]
            for j_idx in range(len(top_vars)):
                if i_idx == j_idx:
                    continue
                j = top_vars[j_idx]
                
                # Try 4 directions: (i up, j down), (i down, j up)
                for di, dj in [(-1, 1), (1, -1)]:
                    for step in [0.005, 0.002, 0.001, 0.0005]:
                        trial = best_h.copy()
                        trial[i] += di * step
                        trial[j] += dj * step
                        if (trial[i] < 0 or trial[i] > 1 or 
                            trial[j] < 0 or trial[j] > 1):
                            continue
                        trial = project_feasible(trial, n)
                        c_trial = compute_c5(trial, n)
                        if c_trial < best_c - 1e-14:
                            improve = best_c - c_trial
                            if improve > best_improve:
                                best_improve = improve
                                best_move = (i, j, di * step, dj * step)
        
        if best_move is not None:
            i, j, di, dj = best_move
            best_h[i] += di
            best_h[j] += dj
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
        else:
            break
    
    return best_h, best_c
```

```python
def steepest_descent_step(h, n, active_shifts, eta):
    """Compute steepest descent direction for max function."""
    dx = 2.0 / n
    m = len(active_shifts)
    
    # Compute gradients for active shifts
    grads = np.array([shift_gradient(h, n, k, dx) for k in active_shifts])
    # grads shape: (m, n)
    
    # Find weights w that minimize ||sum w_k * g_k||^2 subject to sum w = 1, w >= 0
    # This is: min_w (G^T w)^T (G^T w) = w^T G G^T w
    # subject to 1^T w = 1, w >= 0
    G = grads  # (m, n)
    M = G @ G.T  # (m, m)
    
    # Solve QP: min w^T M w s.t. 1^T w = 1, w >= 0
    # Using scipy: minimize 0.5 * w^T M w (since gradient is M w)
    from scipy.optimize import minimize as sp_minimize
    w0 = np.ones(m) / m
    res = sp_minimize(lambda w: 0.5 * w @ M @ w, w0,
                      constraints={'type': 'eq', 'fun': lambda w: w.sum() - 1},
                      bounds=[(0, 1)] * m, method='SLSQP')
    
    if res.success:
        w = res.x
    else:
        w = np.ones(m) / m
    
    # Steepest direction
    g = (w[:, None] * grads).sum(axis=0)  # shape (n,)
    
    # Take step
    h_new = h - eta * g
    h_new = project_feasible(h_new, n)
    
    return h_new
```

    m = len(active)
    if m == 0:
        return h, compute_c5(h, n)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []

        for idx in active:

        # Normalize

                        if best_h[i] + di < -1e-15 or best_h[i] + di > 1 + 1e-15:
                            continue
                        if best_h[j] + dj < -1e-15 or best_h[j] + dj > 1 + 1e-15:
                            continue
                        best_h[i] += di
                        best_h[j] += dj
                        best_h = project_feasible(best_h, n)
                        c_try = compute_c5(best_h, n)
                        if c_try < best_c - 1e-14:
                            best_c = c_try
                            improved = True
                        else:
                            best_h[i] -= di
                            best_h[j] -= dj
                            best_h = project_feasible(best_h, n)
            if not improved:
                break
    
    return best_h, best_c

def perturb_and_optimize(h, n, deadline, n_trials=4):
    """Try random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(seed)
    t_per = max(15, (deadline - time.time()) / (n_trials + 1))
    
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        t_sub = min(time.time() + t_per, deadline - 10)
        
        h_p = best_h.copy()
        if t == 0:
            h_p += 0.03 * rng.standard_normal(n)
        elif t == 1:
            start = rng.integers(0, max(1, n - n // 4))
            h_p[start:start + n // 4] += 0.08 * rng.choice([-1, 1])
        elif t == 2:
            idx = rng.choice(n, size=max(1, n // 5), replace=False)
            h_p[idx] += 0.03 * rng.standard_normal(len(idx))
        else:
            # Shape-preserving perturbation
            h_p = best_h + 0.02 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        
        h_p = project_feasible(h_p, n)
        h_opt, c_opt = run_lbfgs(h_p, n, t_sub,
            configs=[(500, 50.0), (5000, 500.0), (20000, 10000.0)])
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(250, 0.5)
        n0 = 250
    
    h = project_feasible(h0, n0)
    best_h = h.copy()
    best_c = compute_c5(h, n0)
    
    # Phase 1: L-BFGS warm-up at original resolution (80s)
    t1 = min(t0 + 80, deadline - 700)
    h, c = run_lbfgs(h, n0, t1,
        configs=[(100, 10.0), (500, 100.0), (2000, 1000.0), (5000, 10000.0)])
    if c < best_c:
        best_c, best_h = c, h.copy()
