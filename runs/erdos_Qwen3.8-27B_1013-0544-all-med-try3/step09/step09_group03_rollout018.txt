```python
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def lse_smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    w /= s
    return w

    val = lse_smooth_max(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2

            ones_set.add(i)
        else:
            # Revert
            b[i] = 0.0
            b[j] = 1.0
```

Now let me also think about the directed binary search:

After computing the worst shift k*, I can try to directly reduce R(k*) by:
- Finding positions i where b[i]=1 and b[i+k*]=0 (these contribute to R(k*))
- Swapping b[i]=1 with b[j]=0 where b[j+k*]=1 (this removes the contribution from i)

This is a more targeted search. Let me implement it:

```python
def directed_binary_search(b, n, deadline, n_rounds=100):
    """Targeted binary search focusing on worst shifts."""
    best_c = compute_c5(b, n)
    best_b = b.copy()
    c_vec = compute_c5_vec(b, n)
    center = n - 1
    dx = 2.0 / n
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        
        # Find the worst shift
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - center
        
        # Try to reduce R(k_worst)
        found = False
        if k_worst >= 0:
            # R(k) = sum b[i] * (1-b[i+k]) for i in [0, n-1-k]
            # Positions where b[i]=1 and b[i+k]=0 contribute
            nk = n - k_worst
            contrib_1 = np.where(b[:nk] == 1)[0]
            for i in contrib_1:
                if b[i+k_worst] == 0:
                    # Try to find j where b[j]=0 and b[j+k_worst]=1
                    # to swap i and j
                    zeros = np.where(b == 0)[0]
                    for j in zeros:
                        if j == i or j >= nk:
                            continue
                        if b[j+k_worst] == 1:
                            # This swap would help: removes contribution from i, doesn't add from j
                            b[i] = 0.0
                            b[j] = 1.0
                            c_new = compute_c5(b, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_b = b.copy()
                                c_vec = compute_c5_vec(b, n)
                                found = True
                                break
                            else:
                                b[i] = 1.0
                                b[j] = 0.0
            if found:
                continue
        
        # If directed search didn't help, try random neighborhood
        # ... (fallback to random swaps)
        
        # Periodically re-evaluate
        if rnd % 5 == 0:
            c_vec = compute_c5_vec(b, n)
    
    return best_b, best_c
```
