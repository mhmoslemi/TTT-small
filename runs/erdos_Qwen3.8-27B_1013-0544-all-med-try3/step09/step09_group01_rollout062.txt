```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha, mu):
    """Combined objective + gradient for log-sum-exp smoothed max."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - mx)
    w /= w.sum()
    
    # Fast gradient
    dx = 2.0 / n
    A = np.cumsum(w)[:n]
    conv_part = np.convolve(w, x, 'full')[:n]
    corr_part = np.correlate(x, w, 'full')[:n]
    grad = dx * (A - conv_part - corr_part)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return float(val), grad

def active_obj_grad(x, n, active, alpha, mu):
    """Objective + gradient using only active lags."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    w_a = np.exp(sh - smx)
    w_a /= w_a.sum()
    
    # Build full w vector
    w = np.zeros(2*n-1)
    w[active] = w_a
    
    dx = 2.0 / n
    A = np.cumsum(w)[:n]
    conv_part = np.convolve(w, x, 'full')[:n]
    corr_part = np.correlate(x, w, 'full')[:n]
    grad = dx * (A - conv_part - corr_part)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return float(val), grad

def lbfgs_step(h, n, deadline, alphas, mu=0.01, maxiters_list=None, use_active=False, top_k=8):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters_list and i < len(maxiters_list):
            maxit = maxiters_list[i]
        else:
            maxit = max(500, int(remaining * 4))
        
        if use_active:
            c_vec = compute_c5_vec(best_h, n)
            threshold = c_vec.max() - max(1e-6, 0.001 * c_vec.max())
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = active[np.argsort(-c_vec[active])[:top_k]]
            
            def obj(x):
                return active_obj_grad(x, n, active, alpha, mu)[0]
            def jac(x):
                return active_obj_grad(x, n, active, alpha, mu)[1]
        else:
            def obj(x):
                return smooth_obj_grad(x, n, alpha, mu)[0]
            def jac(x):
                return smooth_obj_grad(x, n, alpha, mu)[1]
        
        try:
            res = minimize(obj, best_h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit*10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c
```

```python
def smooth_only(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh-mx))))/alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh)))/alpha)

def smooth_jac_only(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0/n
    A = np.cumsum(w)[:n]
    conv = np.convolve(w, x, 'full')[:n]
    corr = np.correlate(x, w, 'full')[:n]
    grad = dx * (A - conv - corr)
    if mu > 0:
        grad += 2*mu*(x.sum() - n/2.0)
    return grad
```

This works. The objective is O(n²) and the gradient is O(n²) (for the conv/correlate), both fast in C.
