    c = compute_c5_vec(h, n)
    # Get top-k shifts
    idx = np.argsort(c)[-top_k:]
    weights = np.exp((c[idx] - c[idx].max()) / 0.01)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for w, m in zip(weights, idx):
        k = int(m) - center
        if k >= 0:
            # dJ/dh[j] = (1-h[j-k]) * [j>=k] - h[j+k] * [j+k<n]
            j_range = range(k, n)
            grad[j_range] += w * (1 - h[j_range - k])
            if n - k > 0:
                j_range2 = range(0, n - k)
                grad[j_range2] += w * (-h[j_range2 + k])
        else:

            grad[j_range] += w * (1 - h[j_range + kk])
            j_range2 = range(kk, n)
            grad[j_range2] += w * (-h[j_range2 - kk])
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)

import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu, thresh=12):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    active_mask = sh > -thresh
    w = np.exp(sh * active_mask - sh[active_mask].max()) if active_mask.any() else np.zeros_like(sh)
    if active_mask.any():
        amax_active = sh[active_mask].max()
        w = np.zeros_like(sh)
        w[active_mask] = np.exp(sh[active_mask] - amax_active)
        w[~active_mask] = 0
        wsum = w.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        w /= wsum
    else:
        return np.zeros(n)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    active_idx = np.where(active_mask)[0]
    for idx in active_idx:
        k = int(idx) - center
        wm = w[idx] * dx
        if wm < 1e-15:
            continue
        if k >= 0:
            grad[k:n] += wm * (1.0 - x[0:n-k])
            grad[0:n-k] += wm * (-x[k:n])
        else:
            kk = -k
            grad[0:n-kk] += wm * (1.0 - x[kk:n])
            grad[kk:n] += wm * (-x[0:n-kk])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 1000.0),
                   (5000, 5000.0), (15000, 20000.0), (50000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 30:
            break
        t_start = time.time()
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 40})
            elapsed = time.time() - t_start
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
            if time.time() > deadline - 60:
                break
        except Exception:
            pass
    return h, best_c

def subgradient_polish(h, n, deadline, top_k=15):

    best_c = compute_c5(h, n)
    lr = 0.005
    n_no_improve = 0

        else:
            n_no_improve += 1
            if n_no_improve % 200 == 0:
