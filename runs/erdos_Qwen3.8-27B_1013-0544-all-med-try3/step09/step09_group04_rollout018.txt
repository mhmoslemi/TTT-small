import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += 0.5 * mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += mu * (x.sum() - n / 2.0)
    return grad

            if len(active) > 8:

                active = active[order[:8]]
            if len(active) == 0:

def pair_descent(h, n, deadline, deltas=None, window=20):
    """Efficient pair coordinate descent"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-15 and best_h[j] - d >= -1e-15:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-15 and best_h[j] + d <= 1.0 + 1e-15:

One important thing: the previous code's pair_search used radius=40, which means for each i, it checks j in [i-40, i+40]. That's 81 values. For n=500, that's 500*81 = 40,500 pairs per direction per delta. With 2 directions and 7 deltas, that's 567,000 evaluations. Each evaluation is O(n) = 500 (for np.correlate). Total: 283M operations. At numpy speed, this is maybe 5-10 seconds. But the "improved" loop could run many times.

I'll use a smaller window (10-15) and fewer deltas to keep it manageable.

Let me finalize the code now. I want to make sure it's correct and efficient.

One more thought: should I try to use cvxpy for the QP subproblem? The active-set problem is:
min max_{m ∈ A} f_m(h) s.t. sum(h) = n/2, 0 ≤ h ≤ 1

This can be written as:
min t s.t. f_m(h) ≤ t for m ∈ A, sum(h) = n/2, 0 ≤ h ≤ 1

Since each f_m is quadratic (bilinear), this is a QCP (quadratically constrained quadratic program). CVXOPT or SCIP could handle it, but the bilinear terms make it non-convex. So CVXOPT wouldn't help directly.

Actually, wait. f_m(h) = dx * sum_i h[i+k]*(1-h[i]) = dx * sum_i h[i+k] - dx * sum_i h[i+k]*h[i]

The second term is bilinear (product of two different variables). This makes the problem non-convex. So standard QP solvers won't work.

However, if we fix all variables except one, the problem becomes a one-dimensional quadratic, which is easy. This is the basis of coordinate descent.

Let me also think about a "steepest descent" approach on the active set:
- Compute the gradient of the max (or smooth max) at the current point
- Take a projected gradient step
- Repeat

This is essentially what L-BFGS does, but with a line search. For a max of quadratics, the landscape is piecewise quadratic, so a damped Newton method might be better. But implementing that is complex.

I'll stick with L-BFGS but make it more robust.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += 0.5 * mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += mu * (x.sum() - n / 2.0)
    return grad

def get_active_set(c_vec, top_k=6, rel_tol=0.001):
    max_val = c_vec.max()
    threshold = max_val - max(1e-12, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    if len(active) == 0:
        active = np.array([int(np.argmax(c_vec))])
    return active

def run_lbfgs_active(h, n, deadline, alphas, mu=0.01, top_k=6, rel_tol=0.001,
                     maxit_per=2000, restart_every=3):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    count = 0
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 8:
            break
        # Re-identify active set every restart_every alphas
        if count % restart_every == 0:
            c_vec = compute_c5_vec(best_h, n)
            active = get_active_set(c_vec, top_k=top_k, rel_tol=rel_tol)
        maxit = min(maxit_per, max(200, int(rem * 2)))
        try:
            res = minimize(smax_obj, best_h, jac=smax_grad,
                           args=(n, active, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': maxit * 12,
                                    'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
        count += 1
    return best_h, best_c

def binary_push_refine(h, n, deadline, threshold=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for rnd in range(30):
        rem = deadline - time.time()
        if rem < 15:
            break
        # Push near-boundary values
        h_push = best_h.copy()
        h_push[h_push < threshold] = 0.0
        h_push[h_push > 1.0 - threshold] = 1.0
        h_push = project_feasible(h_push, n)
        c_push = compute_c5(h_push, n)
        if c_push < best_c:
            best_c = c_push
            best_h = h_push
        
        # Quick active-set optimization from pushed solution
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_set(c_vec, top_k=5, rel_tol=0.0005)
        try:
            res = minimize(smax_obj, best_h, jac=smax_grad,
                           args=(n, active, 500000, 0.003), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': 3000,
                                    'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
        # Decrease threshold for stronger rounding
        threshold *= 0.85
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, window=12):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        passes = 0
        improved = True
        while improved and time.time() < deadline and passes < 50:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try increasing i, decreasing j
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try decreasing i, increasing j
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def single_worst_dive(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for it in range(30):
