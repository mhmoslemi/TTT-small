```python
def compute_delta_c(h, a, b, d, n):
    k = np.arange(-(n-1), n)  # all lags
    delta = np.zeros(2*n-1)
    # h[a-k]
    idx1 = a - k
    mask1 = (idx1 >= 0) & (idx1 < n)
    delta[mask1] += h[idx1[mask1]]
    # -h[b-k]
    idx2 = b - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta[mask2] -= h[idx2[mask2]]
    # +h[a+k]
    idx3 = a + k
    mask3 = (idx3 >= 0) & (idx3 < n)
    delta[mask3] += h[idx3[mask3]]
    # -h[b+k]
    idx4 = b + k
    mask4 = (idx4 >= 0) & (idx4 < n)
    delta[mask4] -= h[idx4[mask4]]
    return d * delta
```

```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[m:n] += (1.0 - h[0:n-m])
        g[0:n-m] += -h[m:n]
    else:
        g[0:n-m] += (1.0 - h[m:n])
        g[m:n] += -h[0:n-m]
    return g

def subgradient_opt(h, n, deadline, top_k=1, max_iter=100000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    it = 0
    while it < max_iter and time.time() < deadline:
        c_all = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c_all)[-top_k:]
        g = np.zeros(n)
        for k2 in top_idx:
            lag = k2 - (n - 1)
            g += _lag_grad(h, n, lag)
        g = g / top_k
        g = g - g.sum() / n
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        progress = it / max_iter
        step = 0.3 * (1.0 - 0.8 * progress) / (1.0 + 0.01 * it)
        if it > max_iter * 0.7:
            step *= 0.3
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h_new
                tot = room.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * room
            else:
                tot = h_new.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * h_new
            h_new = np.clip(h_new, 0.0, 1.0)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            g += w[k] * _lag_grad(x, n, lag)
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, max_rounds=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    stuck = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        c_max_val = c_all[k_star]
        if lag == 0:
            break
        m = abs(lag)
        n_pairs = n - m
        if lag > 0:
            hi_idx = m + np.arange(n_pairs)
            lo_idx = np.arange(n_pairs)
        else:
            lo_idx = m + np.arange(n_pairs)
            hi_idx = np.arange(n_pairs)
        hi_vals = h[hi_idx]
        lo_vals = h[lo_idx]
        move_from_hi = hi_vals > lo_vals + 1e-10
        from_mask = np.where(move_from_hi)[0]
        if len(from_mask) == 0:
            from_mask = np.where(~move_from_hi)[0]
        delta_fractions = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        improved = False
        for frac in delta_fractions:
            if time.time() > deadline:
                break
            for idx in from_mask:
                if time.time() > deadline:
                    break
                a = lo_idx[idx]
                b = hi_idx[idx]
                if frac >= 0:
                    d = frac * (h[b] - h[a])
                else:
                    d = frac * (h[a] - h[b])
                if d < 1e-10:
                    continue
                if d >= h[b] * 0.99 or d >= (1.0 - h[a]) * 0.99:
                    continue
                h[b] -= d
                h[a] += d
                c_new_arr = np.correlate(h, 1.0 - h, mode='full')
                c_new = float(np.max(c_new_arr) * dx)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    if improved and (rnd % 200 == 199):
                        break
                else:
                    h[a] -= d
                    h[b] += d
            if improved:
                break
        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    for rnd in range(20000):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argsort(c_all)[-3:]
        for k2 in top3:
            if time.time() > deadline:
                break
            lag = k2 - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            n_pairs = n - m
            best_pair_d = None
            best_pair_c = best_c
            for trial_frac in [0.001, 0.0005, 0.0002, 0.0001]:
                for i in range(0, n_pairs, max(1, n_pairs // 200)):
                    a = i
                    b = i + m
                    if abs(h[a] - h[b]) < 1e-8:
                        continue
                    if h[b] > h[a]:
                        d = trial_frac * (h[b] - h[a])
                    else:
                        d = -trial_frac * (h[a] - h[b])
                    if abs(d) < 1e-11:
                        continue
                    h[a] += d
                    h[b] -= d
                    c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                    if c_new < best_pair_c:
                        best_pair_c = c_new
                        best_pair_d = (a, b, d)
                    else:
                        h[b] += d
                        h[a] -= d
            if best_pair_d is not None:
                a, b, d = best_pair_d
                h[a] += d
                h[b] -= d
                if best_pair_c < best_c:
                    best_c = best_pair_c
                    best_h = h.copy()
    return best_h, best_c

def multi_start_explore(h_best, n, deadline, rng, num_starts=3):
    best_h = h_best.copy()
    best_c = compute_c5(h_best, n)
    for s in range(num_starts):
        if time.time() > deadline:
            break
        scale = 0.01 * (1.0 + 0.5 * s)
        hp = h_best + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        c0 = compute_c5(hp, n)
        if c0 >= best_c:
            continue
        sub_deadline = min(deadline, time.time() + 30)
        hp, c_sub = subgradient_opt(hp, n, sub_deadline, top_k=1, max_iter=5000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = hp
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    best_c = compute_c5(best_h, n)
    phase1_deadline = t0 + min(120, (deadline - t0) * 0.12)
    h_opt, c_opt = subgradient_opt(best_h, n, phase1_deadline, top_k=2, max_iter=40000)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    if time.time() < deadline - 200:
        h_slq = slsqp_polish(best_h, n, alpha=500000, maxiter=2000)
        c_slq = compute_c5(h_slq, n)
        if c_slq < best_c:
            best_c = c_slq
            best_h = h_slq
    phase3_deadline = min(deadline - 60, time.time() + (deadline - time.time()) * 0.35)
    h_opt, c_opt = targeted_local_search(best_h, n, phase3_deadline, max_rounds=5000)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    if time.time() < deadline - 80:
        h_opt, c_opt = subgradient_opt(best_h, n, min(deadline - 60, time.time() + 80), top_k=3, max_iter=60000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if time.time() < deadline - 40:
        h_slq = slsqp_polish(best_h, n, alpha=2000000, maxiter=1500)
        c_slq = compute_c5(h_slq, n)
        if c_slq < best_c:
            best_c = c_slq
            best_h = h_slq
    if time.time() < deadline - 20:
        h_opt, c_opt = targeted_local_search(best_h, n, min(deadline - 5, time.time() + (deadline - time.time()) * 0.5), max_rounds=10000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if time.time() < deadline - 10:
        h_opt, c_opt = fine_pair_search(best_h, n, deadline - 3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if time.time() < deadline - 5:
        h_opt, c_opt = targeted_local_search(best_h, n, deadline - 1, max_rounds=2000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
