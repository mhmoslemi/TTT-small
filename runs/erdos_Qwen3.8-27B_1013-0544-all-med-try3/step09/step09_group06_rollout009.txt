def targeted_lag_reduction(h, n, m, dx, c_all, max_moves=200):
    """Try to reduce c[m] by transferring mass from i+m to i"""
    h = h.copy()
    improved = 0
    for _ in range(max_moves):
        # Compute contribution of each pair for lag m
        contrib = h[m:n] * (1 - h[0:n-m])  # contribution to c[m] from each pair
        # Moving δ from position i+m to i reduces contrib[i] by approximately
        # δ*(1-h[i]) + δ*h[i+m] = δ*(1 - h[i] + h[i+m])
        # But we also need to check effect on other lags

        if contrib[max_idx] < 1e-8:
            break
        i = max_idx
        a = i + m  # source (high h)
        b = i      # destination (low h)
        # Maximum transfer
        max_delta = min(h[a], (1-h[b]) * 0.5, 0.05)

                improved += 1
                break

                h[a] += delta
                h[b] -= delta
        else:
            break

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_lse(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_lse_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_lse_weights(c, alpha)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            grad += w[k_idx] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k_idx] * (1.0 - h[0:n-lag])
            grad[0:n-lag] += w[k_idx] * (-h[lag:n])
        else:
            grad[0:n-m] += w[k_idx] * (1.0 - h[m:n])
            grad[m:n] += w[k_idx] * (-h[0:n-m])
    return grad * dx

def fast_lag_update(c_all, h, n, a, b, delta):
    dx = 2.0 / n
    new_c = c_all.copy()
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        dc = 0.0
        if 0 <= a - lag < n:
            dc += -delta * (1.0 - h[a - lag])
        if 0 <= a + lag < n:
            dc += h[a + lag] * delta
        if 0 <= b - lag < n:
            dc += delta * (1.0 - h[b - lag])
        if 0 <= b + lag < n:
            dc += -h[b + lag] * delta
        new_c[k_idx] += dx * dc
    return new_c

def grad_momentum(h, n, alpha, deadline, lr=1.5, momentum=0.9, max_iter=15000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h_curr, n, alpha)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 800:
            lr *= 0.6
            v *= 0.4
            patience = 0
        if it % 3000 == 2999:
            lr = max(0.05, lr * 0.8)
    return best_h, best_c

def slsqp_active(h, n, alpha, maxiter, top_m, ftol=1e-15):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    top_sorted = sorted(top_idx)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        active = cc[top_sorted]
        return smooth_lse(active, alpha)
    
    def cons_fun(x):
        return float(np.sum(x) - n / 2.0)
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search_fast(h, n, deadline, max_rounds=200):
    h = h.copy()
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    best_c = float(c_all.max())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-3:]
        improved = False
        
        for k_idx in top_k:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            n_pairs = n - m
            
            if lag > 0:
                contrib = h[m:n] * (1.0 - h[0:n-m])
                order = np.argsort(contrib)[::-1]
            else:
                contrib = h[0:n-m] * (1.0 - h[m:n])
                order = np.argsort(contrib)[::-1]
            
            for trial_i in order[:30]:
                if time.time() > deadline:
                    break
                if lag > 0:
                    b = trial_i
                    a = trial_i + m
                else:
                    a = trial_i + m
                    b = trial_i
                
                if h[a] <= h[b] + 1e-10:
                    continue
                
                max_delta = min(h[a] * 0.4, (1.0 - h[b]) * 0.4, 0.08)
                for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    delta = max_delta * frac
                    if delta < 1e-10:
                        break
                    new_c = fast_lag_update(c_all, h, n, a, b, delta)
                    new_max = float(new_c.max())
                    if new_max < best_c - 1e-13:
                        h[a] -= delta
                        h[b] += delta
                        h = project_feasible(h, n)
                        c_all = compute_all_lags(h, n)
                        best_c = float(c_all.max())
                        improved = True
                        break
                    elif new_max < best_c + 0.5e-13:
                        h[a] -= delta
                        h[b] += delta
                        h = project_feasible(h, n)
                        c_all = compute_all_lags(h, n)
                        best_c = float(c_all.max())
                        improved = True
                        break
        
        if not improved:
            break
    
    return h, best_c

def fine_local_search(h, n, deadline):
    h = h.copy()
    c_all = compute_all_lags(h, n)
    best_c = float(c_all.max())
    dx = 2.0 / n
    
    for rnd in range(5000):
        if time.time() > deadline:
            break
        k_idx = int(np.argmax(c_all))
        lag = k_idx - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        n_pairs = n - m
        
        if lag > 0:
            contrib = h[m:n] * (1.0 - h[0:n-m])
        else:
            contrib = h[0:n-m] * (1.0 - h[m:n])
        
        best_improve = 1e-13
        best_move = None
        top_candidates = np.argsort(contrib)[::-1][:50]
        
        for trial_i in top_candidates:
            if lag > 0:
                b = trial_i
                a = trial_i + m
            else:
                a = trial_i + m
                b = trial_i
            
            if h[a] <= h[b] + 1e-10:
                continue
            
            max_delta = min(h[a] * 0.3, (1.0 - h[b]) * 0.3, 0.02)
            for frac in [1.0, 0.5, 0.25]:
                delta = max_delta * frac
                if delta < 1e-10:
                    break
                new_c = fast_lag_update(c_all, h, n, a, b, delta)
                new_max = float(new_c.max())
                if new_max < best_c - best_improve:
                    best_improve = best_c - new_max
                    best_move = (a, b, delta)
        
        if best_move is not None:
            a, b, delta = best_move
            h[a] -= delta
            h[b] += delta
            h = project_feasible(h, n)
            c_all = compute_all_lags(h, n)
            best_c = float(c_all.max())
        else:
            # Try smaller deltas on all pairs
            found = False
            for trial_i in range(n_pairs):
                if time.time() > deadline:
                    break
                if lag > 0:
                    b = trial_i
                    a = trial_i + m
                else:
                    a = trial_i + m
                    b = trial_i
                
                if h[a] <= h[b] + 1e-10:
                    continue
                
                delta = min(0.005, h[a] * 0.1, (1.0 - h[b]) * 0.1)
                if delta < 1e-10:
                    continue
                new_c = fast_lag_update(c_all, h, n, a, b, delta)
                new_max = float(new_c.max())
                if new_max < best_c - 1e-13:
                    h[a] -= delta
                    h[b] += delta
                    h = project_feasible(h, n)
                    c_all = compute_all_lags(h, n)
                    best_c = float(c_all.max())
                    found = True
                    break
            if not found:
                break
    
    return h, best_c

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    elapsed_at_start = time.time()
    
    # Phase 1: Gradient momentum
    t1 = min(deadline - 60, time.time() + 80)
    for alpha in [30, 80, 200, 500]:
        if time.time() > t1:
            break
        h_new, c_new = grad_momentum(h, n, alpha, min(time.time() + 35, t1), lr=1.2, momentum=0.9)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    # Phase 2: SLSQP on active lags
    if time.time() < deadline - 80:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 50:
                break
            h_new = slsqp_active(h, n, 100000, maxiter=2000, top_m=top_m)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    
    # Phase 3: Fast local search
    if time.time() < deadline - 30:
        ls_deadline = min(deadline - 10, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = local_search_fast(h, n, ls_deadline, max_rounds=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 4: Fine local search
    if time.time() < deadline - 5:
        h_new, c_new = fine_local_search(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Final SLSQP polish
    if time.time() < deadline - 3:
        h_new = slsqp_active(best_h, n, 200000, maxiter=1000, top_m=3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except (NameError, Exception):
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Main optimization at initial resolution
    if h_init is not None:
        alloc = deadline - 200
        h_opt, c_opt = optimize_full(h_init, n_init, alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc = deadline - 200
        h_opt, c_opt = optimize_full(h_start, n_init, alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try higher resolutions
    for n_try in [600, 800, 700]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 20
        if remaining < 60:
            break
        alloc = min(remaining * 0.35, 150)
        if alloc < 40:
            break
        
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(1, int((deadline - time.time() - 30) / 50))
        for r in range(min(n_restarts, 5)):
            if time.time() > deadline - 40:
                break
            scale = 0.015 * (1.0 + 0.2 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_full(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Fine local search
        if time.time() < deadline - 2:
            h_new, c_new = fine_local_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # One more SLSQP
        if time.time() < deadline - 1.5:
            h_new = slsqp_active(best_h, n, 500000, maxiter=500, top_m=3)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
