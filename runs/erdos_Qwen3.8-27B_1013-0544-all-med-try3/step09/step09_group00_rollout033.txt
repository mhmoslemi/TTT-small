from scipy.signal import correlate
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = correlate(h, 1 - h, mode='full') * dx
    return c.max()

def compute_c5_all(h, n):
    dx = 2.0 / n
    return correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    # Bisection-style projection
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    smx = shifted.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(c, alpha, h, n):
    """Gradient of smooth max of correlation w.r.t. h"""
    m = c.max()
    shifted = alpha * (c - m)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c)):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_beatty(n, alpha):
    """Create a binary Beatty sequence of length n with n/2 ones."""
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            h[pos] = 1.0
            if len(positions) >= n_ones:
                break
    # If we didn't get exactly n/2 ones (shouldn't happen), fix it
    current_ones = int(h.sum())
    if current_ones < n_ones:
        zeros = np.where(h == 0)[0]
        h[zeros[:n_ones - current_ones]] = 1.0
    elif current_ones > n_ones:
        ones = np.where(h == 1)[0]
        h[ones[current_ones - n_ones:]] = 0.0
    return h

def binary_local_search(h, n, deadline, max_iter=50000):
    """Binary local search with single-pair flips."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    
    iter_count = 0
    while time.time() < deadline and iter_count < max_iter:
        improved = False
        # Try all pairs
        best_gain = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:

                    if gain > best_gain:

            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            improved = True
        if not improved:

        iter_count += 1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_beatty(n, alpha):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    seen = set()
    for k in range(1, 10 * n + 1):
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions[:n_ones]:
        h[pos] = 1.0
    cur = h.sum()
    if cur < n_ones:
        zeros = np.where(h == 0)[0]
        h[zeros[:int(n_ones - cur)]] = 1.0
    elif cur > n_ones:
        ones = np.where(h == 1)[0]
        h[ones[int(n_ones - cur):]] = 0.0
    return h

def interpolate_to_n(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_interp, n_dst)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 350:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c)):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [10, 50, 200, 1000, 5000, 50000, 200000, 1000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 15:
            break
        maxit = max(200, int(rem * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 5,
                                   'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, top_k=8, alphas=None, rounds=20):
    if alphas is None:
        alphas = [100000, 500000, 2000000, 10000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for rnd in range(rounds):
        rem = deadline - time.time()
        if rem < 20:
            break
        c_vec = compute_c5_all(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            idx_sorted = np.argsort(-c_vec[active])[:top_k]
            active = active[idx_sorted]
        if len(active) == 0:
            break
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 10:
                break
            c_act = c_vec[active]
            mx = c_act.max()
            sh = alpha * (c_act - mx)
            smx = sh.max()
            w = np.exp(sh - smx)
            w /= w.sum()
            dx_val = 2.0 / n
            obj = float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
            
            def act_obj(x, n=n, active=active, alpha=alpha):
                cv = compute_c5_all(x, n)
                va = cv[active]
                m = va.max()
                s = alpha * (va - m)
                sm = s.max()
                if sm > 350:
                    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
                return m + np.log(np.sum(np.exp(s))) / alpha
            
            def act_grad(x, n=n, active=active, alpha=alpha):
                cv = compute_c5_all(x, n)
                va = cv[active]
                m = va.max()
                s = alpha * (va - m)
                sm = s.max()
                ww = np.exp(s - sm)
                ww /= ww.sum()
                gr = np.zeros(n)
                for wi, idx in zip(ww, active):
                    if wi < 1e-14:
                        continue
                    k = idx - (n - 1)
                    wm2 = wi * dx_val
                    if k >= 0:
                        nv = n - k
                        gr[0:nv] += wm2 * (1.0 - x[k:n])
                        gr[k:n] += wm2 * (-x[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        gr[kk:n] += wm2 * (1.0 - x[0:nv])
                        gr[0:nv] += wm2 * (-x[kk:n])
                return gr
            
            maxit = max(100, int(rem2 * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 4,
                                       'maxls': 25, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def binary_local_search(h, n, deadline, rng, max_rounds=200, samples_per_round=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() >= deadline:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        for _ in range(samples_per_round):
            if time.time() >= deadline:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                found = True
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
        if not found:
            break
    return best_h, best_c

def binary_best_search(h, n, deadline, rng, max_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 0
        best_i, best_j = -1, -1
        # Sample pairs
        n_samples = min(2000, len(ones) * len(zeros))
        for _ in range(n_samples):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c = compute_c5(best_h, n)
            best_h[i] = 1.0
            best_h[j] = 0.0
            if c < best_c - 1e-15:
                gain = best_c - c
                if gain > best_gain:
                    best_gain = grad
                    best_i, best_j = i, j
        if best_i >= 0:
            best_h[best_i] = 0.0
            best_h[best_j] = 1.0
            best_c = compute_c5(best_h, n)
        else:
            break
    return best_h, best_c

def two_swap_search(h, n, deadline, rng, max_rounds=100, samples=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() >= deadline:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        found = False
        for _ in range(samples):
            if time.time() >= deadline:
                break
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            best_h[i1] = 0.0
            best_h[i2] = 0.0
            best_h[j1] = 1.0
            best_h[j2] = 1.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                found = True
            else:
                best_h[i1] = 1.0
                best_h[i2] = 1.0
                best_h[j1] = 0.0
                best_h[j2] = 0.0
        if not found:
            break
    return best_h, best_c

def grad_guided_binary(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        c_vec = compute_c5_all(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= max(1, len(active))
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Find best pair: reduce gradient at a 1 position, increase at a 0 position
        best_score = -np.inf
        best_i, best_j = -1, -1
        # Score for flipping i (1->0): -grad[i] (we want grad[i] > 0 to reduce objective)
        # Score for flipping j (0->1): -grad[j] (we want grad[j] < 0 to reduce objective)
        # Combined score: -(grad[i] + grad[j]) ... actually we want to minimize
        # The change in objective ≈ grad[i]*(-1) + grad[j]*(+1) = -grad[i] + grad[j]
        # We want this to be negative, i.e., grad[i] > grad[j]
        g_ones = grad[ones]
        g_zeros = grad[zeros]
        # We want max grad[i] - grad[j] to be as large as possible
        # Just take the max of g_ones and min of g_zeros
        best_i_local = np.argmax(g_ones)
        best_j_local = np.argmin(g_zeros)
        if g_ones[best_i_local] - g_zeros[best_j_local] > 0.01:
            best_i = ones[best_i_local]
            best_j = zeros[best_j_local]
        else:
            # Try all combinations of top candidates
            top_i = np.argsort(-g_ones)[:10]
            top_j = np.argsort(g_zeros)[:10]
            for ii in top_i:
                for jj in top_j:
                    score = g_ones[ii] - g_zeros[jj]
                    if score > best_score:
                        best_score = score
                        best_i = ones[ii]
                        best_j = zeros[jj]
        if best_i < 0 or best_score <= 0:
            break
        best_h[best_i] = 0.0
        best_h[best_j] = 1.0
        c_new = compute_c5(best_h, n)
        if c_new >= best_c:
            best_h[best_i] = 1.0
            best_h[best_j] = 0.0
            break
        best_c = c_new
    return best_h, best_c

def perturbation_search(h, n, deadline, rng, n_perturb=10):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for _ in range(n_perturb):
        if time.time() >= deadline:
            break
        h_pert = best_h.copy()
        n_flips = rng.randint(3, 10)
        ones = np.where(h_pert > 0.5)[0]
        zeros = np.where(h_pert < 0.5)[0]
        for _ in range(n_flips):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            new_ones = np.where(h_pert > 0.5)[0]
            new_zeros = np.where(h_pert < 0.5)[0]
            ones = new_ones
            zeros = new_zeros
        c_pert = compute_c5(h_pert, n)
        # Optimize from perturbed point
        sub_deadline = min(time.time() + 30, deadline - 5)
        h_opt, c_opt = run_lbfgs(h_pert, n, sub_deadline,
                                  alphas=[1000, 10000, 100000], mu=0.01)
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def resolve_and_optimize(h_src, n_src, n_dst, deadline, rng, seed=0):
    if n_dst == n_src:
        h_start = h_src.copy()
    else:
        h_start = interpolate_to_n(h_src, n_src, n_dst)
    # Also try Beatty starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [h_start]
    starts.append(make_beatty(n_dst, golden / 2.0))
    starts.append(make_beatty(n_dst, np.sqrt(2.0) / 2.0))
    starts.append(make_random_binary(n_dst, rng))
    
    # Pick best start
    best_start = None
    best_start_c = np.inf
    for s in starts:
        c = compute_c5(s, n_dst)
        if c < best_start_c:
            best_start_c = c
            best_start = s
    
    # Phase A: L-BFGS
    rem_a = deadline - time.time()
    if rem_a > 30:
        h_a, c_a = run_lbfgs(best_start, n_dst, time.time() + min(rem_a * 0.4, rem_a - 10),
                             alphas=[10, 100, 1000, 10000, 100000, 1000000], mu=0.01)
    else:
        h_a, c_a = best_start, best_start_c
    
    # Phase B: Active set
    rem_b = deadline - time.time()
    if rem_b > 20:
        h_b, c_b = active_set_lbfgs(h_a, n_dst, time.time() + min(rem_b * 0.3, rem_b - 5),
                                     top_k=8, alphas=[50000, 200000, 1000000], rounds=10)
        if c_b < c_a:
            h_a, c_a = h_b, c_b
    
    # Phase C: Binary local search
    rem_c = deadline - time.time()
    if rem_c > 15:
        h_c, c_c = binary_local_search(h_a, n_dst, time.time() + min(rem_c * 0.4, rem_c - 3),
                                         rng, max_rounds=100, samples_per_round=300)
        if c_c < c_a:
            h_a, c_a = h_c, c_c
    
    # Phase D: Grad-guided binary
    rem_d = deadline - time.time()
    if rem_d > 10:
        h_d, c_d = grad_guided_binary(h_a, n_dst, time.time() + min(rem_d * 0.5, rem_d - 2),
                                        max_iter=20000)
        if c_d < c_a:
            h_a, c_a = h_d, c_d
    
    # Phase E: 2-swap
    rem_e = deadline - time.time()
    if rem_e > 8:
        h_e, c_e = two_swap_search(h_a, n_dst, time.time() + min(rem_e * 0.5, rem_e - 2),
                                     rng, max_rounds=50, samples=150)
        if c_e < c_a:
            h_a, c_e = h_e, c_e
    
    return h_a, c_a, n_dst

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = make_beatty(500, (1.0 + np.sqrt(5.0)) / 4.0)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Try alternative starts at original resolution
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n0, golden / 2.0),
        make_beatty(n0, np.sqrt(2.0) / 2.0),
        make_beatty(n0, 0.414),
        make_random_binary(n0, rng),
        make_random_binary(n0, rng),
    ]
    for s in alt_starts:
        c = compute_c5(s, n0)
        if c < best_c:
            best_c = c
            best_h = s.copy()
    
    # Multi-resolution optimization
    resolutions = [n0, 400, 600, 800]
    # Remove duplicates and sort
    resolutions = sorted(set([int(r) for r in resolutions if r % 2 == 0 and r >= 200]))
    
    # Allocate time: more to original resolution, less to others
    time_alloc = {}
    time_left = deadline - time.time()
    # Give 40% to original, split rest among others
    time_alloc[n0] = time_left * 0.40
    others = [r for r in resolutions if r != n0]
    if others:
        per_other = (time_left * 0.60) / len(others)
        for r in others:
            time_alloc[r] = per_other
    else:
        time_alloc[n0] = time_left
    
    for n_try in resolutions:
        if time.time() >= deadline:
            break
        alloc = time_alloc.get(n_try, 60)
        sub_deadline = min(time.time() + alloc, deadline - 20)
        if sub_deadline <= time.time() + 10:
            continue
        
        if n_try == n0:
            h_try = best_h
        else:
            h_try = interpolate_to_n(h0, n0, n_try)
        
        h_opt, c_opt, _ = resolve_and_optimize(h_try, n_try if n_try != n0 else n0,
                                                 n_try, sub_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            best_n = n_try
    
    # Final intensive optimization on the best solution
    rem = deadline - time.time()
    if rem > 30:
        # L-BFGS polish
        sub_dl = min(time.time() + rem * 0.25, deadline - 25)
        h_pol, c_pol = run_lbfgs(best_h, best_n, sub_dl,
                                  alphas=[10000, 100000, 1000000, 5000000], mu=0.005)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
        
        # Active set
        rem2 = deadline - time.time()
        if rem2 > 20:
            sub_dl2 = min(time.time() + rem2 * 0.25, deadline - 20)
            h_act, c_act = active_set_lbfgs(best_h, best_n, sub_dl2,
                                             top_k=10, alphas=[200000, 1000000, 5000000], rounds=15)
            if c_act < best_c:
                best_c = c_act
                best_h = h_act
        
        # Binary search
        rem3 = deadline - time.time()
        if rem3 > 15:
            sub_dl3 = min(time.time() + rem3 * 0.3, deadline - 15)
            h_bin, c_bin = binary_local_search(best_h, best_n, sub_dl3, rng,
                                                 max_rounds=200, samples_per_round=500)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin
            
            # Grad-guided
            rem4 = deadline - time.time()
            if rem4 > 10:
                sub_dl4 = min(time.time() + rem4 * 0.4, deadline - 10)
                h_gg, c_gg = grad_guided_binary(best_h, best_n, sub_dl4, max_iter=50000)
                if c_gg < best_c:
                    best_c = c_gg
                    best_h = h_gg
        
        # 2-swap
        rem5 = deadline - time.time()
        if rem5 > 8:
            sub_dl5 = min(time.time() + rem5 * 0.5, deadline - 5)
            h_2s, c_2s = two_swap_search(best_h, best_n, sub_dl5, rng,
                                           max_rounds=80, samples=200)
            if c_2s < best_c:
                best_c = c_2s
                best_h = h_2s
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
