import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:k+nv]) * dx
            grad[k:k+nv] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:kk+nv] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:kk+nv]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    return grad

def sqp_step(h, n, active_idx, rho):
    """Solve LP: min t s.t. f_k(h) + grad_k·δ ≤ t, |δ|≤rho, sum(δ)=0, 0≤h+δ≤1"""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    
    # Constraints: n_act (linearized) + 4n (box + trust region)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    
    for j in range(n_act):
        idx = active_idx[j]
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    # h[i] + δ[i] ≥ 0  =>  -δ[i] ≤ h[i]
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    # h[i] + δ[i] ≤ 1  =>  δ[i] ≤ 1 - h[i]
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    # δ[i] ≥ -rho  =>  -δ[i] ≤ rho
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    # δ[i] ≤ rho
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=5000, rho_init=0.03, max_active=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Include all shifts within 0.5% of max, up to max_active
        thresh = max_val - max(1e-10, 0.005 * max_val)
        active_set = np.where(c_vec >= thresh)[0]
        
        if len(active_set) > max_active:
            # Sort by value and take top max_active
            vals = c_vec[active_set]
            order = np.argsort(-vals)
            active_set = active_set[order[:max_active]]
        
        delta, t_opt = sqp_step(h, n, active_set, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        
        if np.max(np.abs(delta)) < 1e-13:
            break
        
        # Backtracking line search
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        
        # Try full step first, then halve
        for frac in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_feasible(h + frac * delta, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                no_improve = 0
                # Expand trust region gradually
                if rho < 0.08:
                    rho = min(rho * 1.2, 0.08)
                break
        else:
            no_improve += 1
            rho *= 0.7
            if rho < 1e-10:
                break
            if no_improve > 50:
                # Try perturbing
                if no_improve > 100:
                    break
    
    return best_h, best_c

def block_local_search(h, n, deadline, block_sizes=(2, 3, 4, 5), max_iter=5000):
    best_c = compute_c5(h, n)
    steps = [0.003, 0.002, 0.001, 0.0005, 0.0002]
    
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for bs in block_sizes:
                if time.time() > deadline:
                    return best_h, best_c
                for start in range(0, n - bs, max(1, (n - bs) // 200)):
                    if time.time() > deadline + 1:
                        break
                    # Move 'step' from block [start, start+bs) to a random other block
                    # Actually, let's do: add step to h[start:start+bs] and subtract from h[end:end+bs]
                    end = (start + n // 2) % n
                    if abs(end - start) < bs:
                        continue
                    
                    # Check bounds
                    block1 = h[start:start+bs]
                    block2 = h[end:end+bs]
                    
                    # Distribute step among block elements
                    if block1.max() + step <= 1.0 + 1e-12 and block2.min() - step >= -1e-12:
                        old1 = h[start:start+bs].copy()
                        old2 = h[end:end+bs].copy()
                        h[start:start+bs] += step
                        h[end:end+bs] -= step
                        h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[start:start+bs] = old1
                            h[end:end+bs] = old2
    
    return h, best_c

def lbfgs_optimize(h, n, deadline, alpha_list=None):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:

            h_new = project_feasible(res.x, n)

            if c_new < best_c:

                h = h_new
        except Exception:

    return h, best_c

LP formulation:

- Objective: minimize t
- Constraints:
  - For each active constraint k: c_k(h) + g_k·δ - t ≤ 0, i.e., g_k·δ - t ≤ -c_k(h)
  - For each i: δ[i] ≤ h[i] (ensures h[i]+δ[i] ≥ 0)
  - For each i: -δ[i] ≤ 1-h[i] (ensures h[i]+δ[i] ≤ 1)
  - For each i: δ[i] ≤ rho (trust region)
  - For each i: -δ[i] ≤ rho (trust region)
  - Equality: sum(δ) = 0 (maintain integral constraint)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:k+nv]) * dx
            grad[k:k+nv] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:kk+nv] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:kk+nv]) * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    return grad

def sqp_lp(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=20000, rho_init=0.02, max_active=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    n_iter = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        n_iter += 1
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.003 * max_val)
        active_mask = c_vec >= thresh
        if active_mask.sum() > max_active:
            idxs = np.where(active_mask)[0]
            vals = c_vec[idxs]
            order = np.argsort(-vals)
            active_idx = idxs[order[:max_active]].tolist()
        else:
            active_idx = np.where(active_mask)[0].tolist()
        if not active_idx:
            break
        delta, t_opt = sqp_lp(h, n, active_idx, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            no_improve += 1
            if no_improve > 30:
                break
            continue
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.15, 0.06)
        else:
            step_halves = [0.5, 0.25, 0.125]
            found = False
            for frac in step_halves:
                h_try = project_feasible(h + frac * delta, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
                rho *= 0.7
                if rho < 1e-10:
                    break
                if no_improve > 60:
                    break
    return best_h, best_c

def focused_local_search(h, n, deadline, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.004, 0.002, 0.001, 0.0004, 0.00015]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        it_count = 0
        while improved and time.time() < deadline:
            improved = False
            it_count += 1
            if it_count > 2000:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.008 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad_mag = np.zeros(n)
            for idx in active[:12]:
                grad_mag += np.abs(shift_grad_single(h, n, int(idx)))
            top = np.argsort(-grad_mag)[:40]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                candidates = [int((i + n // 2) % n), int((i + n // 3) % n),
                              int((i + 2 * n // 3) % n), int((i + n // 4) % n),
                              int((i + 3 * n // 4) % n)]
                for j in candidates:
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def lbfgs_optimize(h, n, deadline, alpha_list=None):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [500, 2000, 8000, 30000]
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 10000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_projected_grad(h, n, deadline, alpha_init=200, n_steps=3000):
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    alpha = alpha_init
    lr_init = 0.002
    for step_i in range(n_steps):
        if time.time() > deadline:
            break
        alpha *= 1.002
        grad = smooth_grad(h_curr, n, alpha)
        step_t = lr_init / (1.0 + 0.01 * step_i)
        h_new = h_curr - step_t * grad
        t = (step_i + 1.0) / (step_i + 1.0 + 3.0)
        h_curr, h_prev = h_new, h_new + t * (h_new - h_prev)
        h_curr = project_feasible(h_curr, n)
        c = compute_c5(h_curr, n)
        if c < best_c - 1e-15:
            best_c = c
            h = h_curr.copy()
    return h, best_c

def multi_start_optimize(h_base, n, deadline, n_starts=3):
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n)
    remaining = deadline - time.time()
    if remaining < 30:
        return best_h, best_c
    per_start = remaining / n_starts
    for s in range(n_starts):
        t_end = time.time() + per_start
        if t_end > deadline:
            t_end = deadline
        if t_end < time.time() + 5:
            break
        if s == 0:
            h_start = best_h.copy()
        else:
            h_start = best_h.copy()
            noise_scale = 0.02 / s
            noise = np.random.randn(n) * noise_scale
            h_start = project_feasible(h_start + noise, n)
        h_s, c_s = sqp_optimize(h_start, n, t_end, max_iter=5000, rho_init=0.03, max_active=40)
        if c_s < best_c:

            best_h = h_s
    return best_h, best_c
