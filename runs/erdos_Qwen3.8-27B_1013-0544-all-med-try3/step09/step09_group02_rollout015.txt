                if best_decrease < step * 0.5:
                    continue
                candidate_pairs = []
                if k >= 0:
                    for di in range(5):
                        ii = int(i_pos) + di
                        if ii < n - k:
                            candidate_pairs.append((ii, int(j_pos)))
                    for dj in range(5):
                        jj = int(j_pos) + dj
                        if jj < n:
                            candidate_pairs.append((int(i_pos), jj))
                else:
                    kk = -k
                    for di in range(5):
                        ii = int(i_pos) + di
                        if ii < n:
                            candidate_pairs.append((ii, int(j_pos)))
                    for dj in range(5):
                        jj = int(j_pos) + dj
                        if jj < n - kk:
                            candidate_pairs.append((int(i_pos), jj))
                for (ip, jp) in candidate_pairs[:n_candidates]:
                    if ip == jp:
                        continue
                    if h[ip] + step <= 1.0 + 1e-11 and h[jp] - step >= -1e-11:
                        h[ip] += step
                        h[jp] -= step
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-15:
                            best_c = cn
                            improved = True
                        else:
                            h[ip] -= step
                            h[jp] += step
                    if h[ip] - step >= -1e-11 and h[jp] + step <= 1.0 + 1e-11:
                        h[ip] -= step
                        h[jp] += step
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-15:
                            best_c = cn
                            improved = True
                        else:
                            h[ip] += step
                            h[jp] -= step
    return best_h, best_c

def binary_local_search(h, n, deadline):
    h_bin = np.round(h)
    ones = np.sum(h_bin == 1)
    need = n // 2
    if ones > need:
        h1 = h_bin == 1
        h0 = h_bin == 0
        excess = ones - need
        h1_sorted = np.sort(h[h1])
        thresh_val = h1_sorted[excess - 1]
        to_flip = np.where(h1 & (h <= thresh_val))[0][:excess]
        h_bin[to_flip] = 0
    elif ones < need:
        h1 = h_bin == 1
        h0 = h_bin == 0
        deficit = need - ones
        h0_sorted = np.sort(h[h0])[::-1]
        thresh_val = h0_sorted[deficit - 1]
        to_flip = np.where(h0 & (h >= thresh_val))[0][:deficit]
        h_bin[to_flip] = 1
    best_h = h_bin.astype(float)
    best_c = compute_c5(best_h, n)
    for _ in range(5000):
        if time.time() > deadline - 1:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        top_k = np.argsort(-c_vec)[:5]
        improved = False
        ones_idx = np.where(best_h == 1)[0]
        zeros_idx = np.where(best_h == 0)[0]
        for ki in top_k:
            k = int(ki) - (n - 1)
            if k >= 0:
                valid_ones = [i for i in ones_idx if i < n - k]
                valid_zeros = [j for j in zeros_idx if j >= k]
            else:
                kk = -k
                valid_ones = [i for i in ones_idx if i >= kk]
                valid_zeros = [j for j in zeros_idx if j < n - kk]
            if not valid_ones or not valid_zeros:
                continue
            np.random.shuffle(valid_ones)
            np.random.shuffle(valid_zeros)
            for i in valid_ones[:20]:
                for j in valid_zeros[:20]:
                    if i >= n - max(1, abs(k)) + abs(k) * (1 if k >= 0 else 0):
                        continue
                    best_h[i] = 0
                    best_h[j] = 1
                    cn = compute_c5(best_h, n)
                    if cn < best_c - 1e-15:
                        best_c = cn
                        improved = True
                        break
                    else:
                        best_h[i] = 1
                        best_h[j] = 0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=3000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, nn=n: float(x.sum() - nn / 2.0),
             'jac': lambda x, nn=n: np.ones(nn), 'args': (n,)}]
    try:
        res = minimize(lambda x: smooth_obj(x, n, alpha, 0.0), h,
                      jac=lambda x: smooth_grad(x, n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def thue_morse_h(n):
    h = np.zeros(n)
    for i in range(n):
        t = i
        pc = 0
        while t:
            pc ^= (t & 1)
            t >>= 1
        h[i] = float(pc)
    if n % 2 != 0:
        h[-1] = 0.0
    return h

def optimize_at_n(h_start, n, deadline, label=""):
    h = project_feasible(h_start, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    elapsed_start = time.time()
    # Phase A: L-BFGS
    tA = min(elapsed_start + 60, deadline - (deadline - tA_budget(deadline, elapsed_start)))
    if time.time() < deadline - 30:
        tA = min(deadline - 30, time.time() + 50)
        h, c = lbfgs_phase(h, n, tA, alpha_list=[100, 500, 2000], mu_list=[50, 200, 1000])
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase B: Projected subgradient
    if time.time() < deadline - 30:
        tB = min(deadline - 60, time.time() + 120)
        h, c = proj_subgrad(h, n, tB, alpha=800)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase C: L-BFGS with higher alpha
    if time.time() < deadline - 30:
        tC = min(deadline - 90, time.time() + 100)
        h, c = lbfgs_phase(h, n, tC, alpha_list=[3000, 8000, 15000], mu_list=[1000, 3000, 8000])
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase D: SQP
    if time.time() < deadline - 30:
        tD = min(deadline - 60, time.time() + 100)
        h, c = sqp_optimize(h, n, tD, max_iter=400, rho_init=0.04)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase E: Fast local search
    if time.time() < deadline - 20:
        tE = min(deadline - 30, time.time() + 80)
        h, c = fast_local_search(h, n, tE, n_top_shifts=6, n_candidates=30)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase F: SQP again
    if time.time() < deadline - 20:
        tF = min(deadline - 40, time.time() + 60)
        h, c = sqp_optimize(h, n, tF, max_iter=200, rho_init=0.02)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase G: Final SLSQP
    if time.time() < deadline - 10:
        h2, c2 = slqp_polish(best_h, n, deadline - 3, alpha=5000000, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    if label:
        pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    best_overall_h = None
    best_overall_c = float('inf')
    best_overall_n = 0

    # Get parent initialization
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 0

    # Strategy 1: Optimize parent at n=500
    if h_parent is not None and n_parent == 500:
        t1_end = min(t0 + 200, deadline - 600)
        if t1_end > time.time() + 20:
            h1, c1 = optimize_at_n(h_parent, 500, t1_end, "n500_parent")
            if c1 < best_overall_c:
                best_overall_c = c1
                best_overall_h = h1.copy()
                best_overall_n = 500

    # Strategy 2: Optimize at n=1000 from parent interpolation
    if h_parent is not None:
        n2 = 1000
        x_old = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n2)
        h2_init = np.interp(x_new, x_old, h_parent)
        h2_init = project_feasible(h2_init, n2)
        t2_end = min(deadline - 200, time.time() + 400)
        if t2_end > time.time() + 30:
            h2, c2 = optimize_at_n(h2_init, n2, t2_end, "n1000_parent")
            if c2 < best_overall_c:
                best_overall_c = c2
                best_overall_h = h2.copy()
                best_overall_n = n2
    else:
        # No parent: create from 0.5
        n2 = 500
        h2_init = np.full(n2, 0.5)
        t2_end = min(deadline - 200, time.time() + 300)
        if t2_end > time.time() + 30:
            h2, c2 = optimize_at_n(h2_init, n2, t2_end, "n500_default")
            if c2 < best_overall_c:
                best_overall_c = c2
                best_overall_h = h2.copy()
                best_overall_n = n2

    # Strategy 3: Thue-Morse at n=512 (power of 2, balanced)
    if time.time() < deadline - 100:
        n3 = 512
        h3_init = thue_morse_h(n3)
        h3_init = project_feasible(h3_init, n3)
        t3_end = min(deadline - 80, time.time() + 150)
        if t3_end > time.time() + 20:
            h3, c3 = optimize_at_n(h3_init, n3, t3_end, "tm512")
            if c3 < best_overall_c:
                best_overall_c = c3
                best_overall_h = h3.copy()
                best_overall_n = n3

    # Strategy 4: Thue-Morse at n=1024
    if time.time() < deadline - 100:
        n4 = 1024
        h4_init = thue_morse_h(n4)
        h4_init = project_feasible(h4_init, n4)
        t4_end = min(deadline - 50, time.time() + 150)
        if t4_end > time.time() + 20:
            h4, c4 = optimize_at_n(h4_init, n4, t4_end, "tm1024")
            if c4 < best_overall_c:
                best_overall_c = c4
                best_overall_h = h4.copy()
                best_overall_n = n4

    # Final: if we still have time, do one more polish on the best
    if best_overall_h is not None and time.time() < deadline - 15:
        nf = len(best_overall_h)
        h_fin, c_fin = slqp_polish(best_overall_h, nf, deadline - 3,
                                    alpha=8000000, maxiter=5000)
        if c_fin < best_overall_c:
            best_overall_c = c_fin
            best_overall_h = h_fin

    # Final feasibility
    if best_overall_h is None:
        best_overall_h = np.full(500, 0.5)
        best_overall_n = 500
    nf = len(best_overall_h)
    best_overall_h = project_feasible(best_overall_h, nf)
    best_overall_c = compute_c5(best_overall_h, nf)
    return best_overall_h, best_overall_c, nf
