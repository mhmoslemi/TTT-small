def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def optimize_at_resolution(h, n, deadline, bounds):
    """Full optimization pipeline at a given resolution."""
    # Phase A: Multi-alpha L-BFGS
    h = multi_alpha_lbfgs(h, n, deadline, bounds)
    c = compute_c5(h, n)

    # Phase B: Subgradient refinement
    if time.time() < deadline - 20:
        h, c = subgrad_refine(h, n, deadline - 15, n_iter=8000)

    # Phase C: Active set with very high alpha
    if time.time() < deadline - 10:
        active, _ = get_active(h, n, top_k=4, tol_frac=0.0003)
        h = do_lbfgs(h, n, 100000000, 0.002, active,
                     min(2000, max(100, int((deadline - time.time()) / 10))), bounds)
        c = compute_c5(h, n)

    # Phase D: Randomized local moves
    if time.time() < deadline - 5:
        h, c = randomized_local_moves(h, n, deadline - 2, n_moves=10000)

    # Phase E: Gradient-guided moves
    if time.time() < deadline - 2:
        h, c = gradient_guided_moves(h, n, deadline - 1, n_moves=3000)

    return h, c


def make_periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(min(ones_per_period, period)):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 1e-12:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < -1e-12:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get parent solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h0)
    except Exception:
        h0 = np.full(400, 0.5)
        n_parent = 400

    h0 = project_feasible(h0, n_parent)
    c0 = compute_c5(h0, n_parent)

    best_h = h0.copy()
    best_c = c0
    best_n = n_parent

    # Phase 1: Quick multi-resolution scan (30s each, try ~6 resolutions)
    res_list = sorted(set([n_parent, 250, 300, 400, 500, 600, 750, 800, 1000, 1200]),
                      key=lambda x: abs(x - n_parent))
    # Limit to at most 6 resolutions
    res_list = res_list[:6]

    candidates = []
    for n in res_list:
        if time.time() > deadline - 400:
            break
        x_old = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h0)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        candidates.append((n, h.copy(), c))

    # Also try a periodic start at the parent resolution
    for period in [4, 5, 6, 7, 8]:
        if period > n_parent // 3:
            break
        h_per = make_periodic_start(n_parent, period, period // 2)
        c_per = compute_c5(h_per, n_parent)
        candidates.append((n_parent, h_per.copy(), c_per))

    candidates.sort(key=lambda x: x[2])

    # Phase 2: Main optimization at top 2 candidates (with time budget each)
    time_per_trial = (deadline - time.time() - 100) / min(2, len(candidates))
    for trial in range(min(2, len(candidates))):
        if time.time() > deadline - 80:
            break
        n, h_init, c_init = candidates[trial]
        bounds = [(0.0, 1.0)] * n
        trial_deadline = min(time.time() + time_per_trial, deadline - 60)
        h_opt, c_opt = optimize_at_resolution(h_init, n, trial_deadline, bounds)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

    # Phase 3: Try additional resolution if time allows
    if time.time() < deadline - 100:
        # Try the next best candidate
        for trial in range(2, min(4, len(candidates))):
            if time.time() > deadline - 40:
                break
            n, h_init, c_init = candidates[trial]
            bounds = [(0.0, 1.0)] * n
            trial_deadline = min(time.time() + 50, deadline - 30)
            h_opt, c_opt = optimize_at_resolution(h_init, n, trial_deadline, bounds)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Phase 4: Final refinement at best resolution
    if time.time() < deadline - 15:
        n_best = best_n
        bounds = [(0.0, 1.0)] * n_best
        # One more subgradient pass
        h_ref, c_ref = subgrad_refine(best_h, n_best, deadline - 5, n_iter=5000)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # Phase 5: Final very tight local moves
    if time.time() < deadline - 3:
        h_fin, c_fin = randomized_local_moves(best_h, best_n, deadline - 1, n_moves=5000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
