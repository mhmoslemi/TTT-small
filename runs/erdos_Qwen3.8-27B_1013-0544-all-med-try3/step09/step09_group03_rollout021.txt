```python
    ks = np.arange(-(n-1), n)  # length 2n-1
    
    # For m (0->1): delta_m[k] = (1-h[m+k]) - h[m-k] (with bounds)
    delta_m = np.zeros(2*n-1)
    # m+k valid: 0 <= m+k < n => -m <= k < n-m
    mask_m1 = (ks >= -m) & (ks < n - m)
    # m-k valid: 0 <= m-k < n => m-n < k <= m-1... => m-n+1 <= k <= m
    # Actually: 0 <= m-k < n => m-n < k <= m => m-n+1 <= k <= m (integer)
    # In our range: max(-(n-1), m-n+1) <= k <= min(n-1, m)
    mask_m2 = (ks >= m - n + 1) & (ks <= m)
    
    # For l (1->0): delta_l[k] = -(1-h[l+k]) + h[l-k]
    delta_l = np.zeros(2*n-1)
    mask_l1 = (ks >= -l) & (ks < n - l)
    mask_l2 = (ks >= l - n + 1) & (ks <= l)
    
    # Compute contributions
    # delta_m[k] = (1-h[m+k]) for valid m+k
    valid_m1 = m + ks[mask_m1]
    delta_m[mask_m1] = (1.0 - h[valid_m1])
    
    # delta_m[k] -= h[m-k] for valid m-k
    valid_m2 = m - ks[mask_m2]
    delta_m[mask_m2] -= h[valid_m2]
    
    # delta_l[k] = -(1-h[l+k]) for valid l+k
    valid_l1 = l + ks[mask_l1]
    delta_l[mask_l1] = -(1.0 - h[valid_l1])
    
    # delta_l[k] += h[l-k] for valid l-k
    valid_l2 = l - ks[mask_l2]
    delta_l[mask_l2] += h[valid_l2]
    
    return (delta_m + delta_l) * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))


def continuous_project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def binary_project(h, n):
    idx = np.argsort(-h)[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def compute_flip_effect(h, m, l, n):
    dx = 2.0 / n
    center = n - 1
    ks = np.arange(-(n - 1), n)
    delta_m = np.zeros(2 * n - 1)
    mask_m1 = (ks >= -m) & (ks < n - m)
    mask_m2 = (ks >= m - n + 1) & (ks <= m)
    valid_m1 = m + ks[mask_m1]
    delta_m[mask_m1] = (1.0 - h[valid_m1])
    valid_m2 = m - ks[mask_m2]
    delta_m[mask_m2] -= h[valid_m2]
    delta_l = np.zeros(2 * n - 1)
    mask_l1 = (ks >= -l) & (ks < n - l)
    mask_l2 = (ks >= l - n + 1) & (ks <= l)
    valid_l1 = l + ks[mask_l1]
    delta_l[mask_l1] = -(1.0 - h[valid_l1])
    valid_l2 = l - ks[mask_l2]
    delta_l[mask_l2] += h[valid_l2]
    return (delta_m + delta_l) * dx


def simulated_annealing_binary(h, n, deadline, T0=0.004, T_end=5e-7, tau=None, top_k=10, seed=42):
    c_vec = compute_c5_vec(h, n)
    best_c = float(np.max(c_vec))
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    t_start = time.time()
    if tau is None:
        tau = 200.0
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = max(T0 * np.exp(-elapsed / tau), T_end)
        if T < T_end:
            break
        active_indices = np.argpartition(c_vec, -top_k)[-top_k:]
        current_max = c_vec[active_indices[0]]
        m = zeros_list[rng.randint(len(zeros_list))]
        l = ones_list[rng.randint(len(ones_list))]
        if m == l:
            it += 1
            continue
        delta = compute_flip_effect(h, m, l, n)
        new_active = c_vec[active_indices] + delta[active_indices]
        new_max = float(np.max(new_active))
        delta_c = new_max - current_max
        if delta_c < 0 or (delta_c / T < 700 and rng.rand() < np.exp(-delta_c / T)):
            h[m] = 1.0
            h[l] = 0.0
            zeros_list.remove(m)
            zeros_list.append(l)
            ones_list.remove(l)
            ones_list.append(m)
            c_vec += delta
            true_max = float(np.max(c_vec))
            if true_max < best_c - 1e-15:
                best_c = true_max
                best_h = h.copy()
        it += 1
        if it % 100000 == 0:
            pass
    return best_h, best_c


def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)


def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 200.0 * (x.sum() - n / 2.0)
    return grad


def run_lbfgs(h, n, deadline, configs=None):
    h = continuous_project(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (20000, 10000.0)]
    for alpha in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = continuous_project(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c


def coordinate_descent_binary(h, n, deadline, top_k=5, max_rounds=5000, seed=123):
    c_vec = compute_c5_vec(h, n)
    best_c = float(np.max(c_vec))
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        active_indices = np.argpartition(c_vec, -top_k)[-top_k:]
        current_max = float(c_vec[active_indices[0]])
        best_gain = 0.0
        best_m = -1
        best_l = -1
        n_ones = len(ones_set)
        n_zeros = len(zeros_set)
        sample_ones = rng.choice(list(ones_set), size=min(200, n_ones), replace=False)
        sample_zeros = rng.choice(list(zeros_set), size=min(200, n_zeros), replace=False)
        for m in sample_zeros:
            for l in sample_ones:
                if m == l:
                    continue
                delta = compute_flip_effect(h, m, l, n)
                new_vals = c_vec[active_indices] + delta[active_indices]
                new_max = float(np.max(new_vals))
                gain = current_max - new_max
                if gain > best_gain + 1e-15:
                    best_gain = gain
                    best_m = m
                    best_l = l
        if best_m >= 0:
            h[best_m] = 1.0
            h[best_l] = 0.0
            zeros_set.discard(best_m)
            zeros_set.add(best_l)
            ones_set.discard(best_l)
            ones_set.add(best_m)
            delta = compute_flip_effect(h, best_m, best_l, n)
            # Recompute properly
            c_vec = compute_c5_vec(h, n)
            new_c = float(np.max(c_vec))
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c


def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return continuous_project(h_new, n_new)
```
