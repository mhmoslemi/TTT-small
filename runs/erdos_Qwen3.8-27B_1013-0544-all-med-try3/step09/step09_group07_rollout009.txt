    def obj_and_grad(x):

        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty * diff * diff

                rounds += 1
                continue
            if h[i] > h[j]:
                h[i] -= d
                h[j] += d
            else:
                h[i] += d
                h[j] -= d
            h = project_feasible(h, n)
            c_new = compute_all_lags(h, n)
            if c_new.max() >= max_val + 1e-12:
                if h[i] < 1.0:
                    pass
                # revert
                h2 = best_h.copy()
                # Actually just revert the last change
                if lag > 0:
                    h[i] += d
                    h[j] -= d
                else:
                    h[i] -= d
                    h[j] += d
                h = project_feasible(h, n)
            else:
                c_all = c_new
        rounds += 1

    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, t_init=0.005, t_min=0.00001, cool=0.9999, alpha_sa=500000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-5:]
        k_star = top_k[rng.randint(len(top_k))]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue

        i = rng.randint(0, n - m)
        j = i + m
        if h[i] > h[j] + 1e-8:
            d = rng.uniform(0.002, 0.02)
            d = min(d, (h[i] - h[j]) * 0.3)
        elif h[j] > h[i] + 1e-8:
            d = -rng.uniform(0.002, 0.02)
            d = max(d, -(h[j] - h[i]) * 0.3)
        else:
            T *= cool
            continue

        if abs(d) < 1e-9:
            T *= cool
            continue

        e_old = smooth_max_value(c_all, alpha_sa)
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)

        de = e_new - e_old
        accept = False
        if de < -1e-15:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool
        if it_since_improve > 30000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Polish at parent resolution with corrected gradient
    if time.time() < deadline - 800:
        for alpha in [500, 5000, 50000, 500000]:
            if time.time() > deadline - 780:
                break
            h_opt = lbfgsb_optimize(best_h, n0, alpha, time.time() + 15, maxiter=1500)
            c_opt = compute_c5(h_opt, n0)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 2: Upsample to n=1000 and optimize
    n2 = 1000
    if time.time() < deadline - 500:
        h2 = upsample(best_h, len(best_h), n2)
        c2 = compute_c5(h2, n2)
        for alpha in [100, 1000, 5000, 50000, 200000, 1000000, 5000000]:
            if time.time() > deadline - 480:
                break
            t_sub = min(40, (deadline - 480) - time.time())
            if t_sub < 5:
                break
            h_opt = lbfgsb_optimize(h2, n2, alpha, time.time() + t_sub, maxiter=3000)
            c_opt = compute_c5(h_opt, n2)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h2 = h_opt

    # Phase 3: Gradient local search at best resolution
    n_curr = len(best_h)
    if time.time() < deadline - 350:
        t_ls = min(120, (deadline - 350) - time.time())
        if t_ls > 10:
            h_opt, c_opt = gradient_local_search(best_h, n_curr, time.time() + t_ls, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Upsample to n=2048 and optimize
    n3 = 2048
    if time.time() < deadline - 250:
        h3 = upsample(best_h, len(best_h), n3)
        c3 = compute_c5(h3, n3)
        for alpha in [500, 5000, 50000, 500000, 5000000, 50000000]:
            if time.time() > deadline - 230:
                break
            t_sub = min(30, (deadline - 230) - time.time())
            if t_sub < 5:
                break
            h_opt = lbfgsb_optimize(h3, n3, alpha, time.time() + t_sub, maxiter=2000)
            c_opt = compute_c5(h_opt, n3)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h3 = h_opt

    # Phase 5: SA at best resolution
    n_curr = len(best_h)
    if time.time() < deadline - 100:
        t_sa = min(120, (deadline - 100) - time.time())
        if t_sa > 10:
            h_opt, c_opt = simulated_annealing(best_h, n_curr, time.time() + t_sa, rng,
                                                t_init=0.003, t_min=0.00001,
                                                cool=0.99995, alpha_sa=1000000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: L-BFGS-B polish at final alpha
    n_curr = len(best_h)
    if time.time() < deadline - 50:
        for alpha in [50000000, 200000000]:
            if time.time() > deadline - 25:
                break
            h_opt = lbfgsb_optimize(best_h, n_curr, alpha, time.time() + 25, maxiter=3000)
            c_opt = compute_c5(h_opt, n_curr)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Final local search
    n_curr = len(best_h)
    if time.time() < deadline - 5:
        h_opt, c_opt = gradient_local_search(best_h, n_curr, deadline - 2, rng, max_rounds=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
