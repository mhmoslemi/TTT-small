```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # For step functions, use nearest-neighbor upsampling
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad
```

```python
def binary_local_search(h, n, deadline):
    h = h.copy()
    # Round to binary
    idx = np.argsort(-h)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    h = project_feasible(h_bin, n)
    
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Active lags: top 20 or within 2% of max
        thresh = max_val - max(1e-8, 0.02 * max_val)
        active = np.where(c_vec >= thresh)[0]
        active = active[:30]  # Limit to top 30

        # Try swaps
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                if time.time() > deadline:
                    break
                # Try swap
                h_bin[i] = 0
                h_bin[j] = 1
                h_check = h_bin.copy()
                c_new = compute_c5(h_check, n)

                    improved = True
                    break
                else:

                    h_bin[j] = 0
            if improved:
                break
        if not improved:
            break
```

    This is: count of positions j where h[j]=1 and h[j+kk]=0 (i.e., "1 with 0 to the right")
  
  Now, when we swap h[i]: 1→0 and h[j]: 0→1:
  
  For lag k ≥ 0: c[n-1+k] = sum_{m=k}^{n-1} h[m] * (1-h[m-k])
  - h[i] appears as h[m] when m=i (if i ≥ k): old contribution h[i]*(1-h[i-k]) = 1*(1-h[i-k]), new: 0
    Change: -(1-h[i-k]) if i ≥ k
  - h[i] appears as h[m-k] when m-k=i, i.e., m=i+k (if i+k ≤ n-1): old contribution h[i+k]*(1-h[i]) = h[i+k]*0 = 0, new: h[i+k]*1
    Change: +h[i+k] if i+k ≤ n-1
  - h[j] appears as h[m] when m=j (if j ≥ k): old: 0, new: 1*(1-h[j-k])
    Change: +(1-h[j-k]) if j ≥ k
  - h[j] appears as h[m-k] when m-k=j, i.e., m=j+k (if j+k ≤ n-1): old: h[j+k]*1 = h[j+k], new: h[j+k]*0 = 0
    Change: -h[j+k] if j+k ≤ n-1
  
  So for k ≥ 0:
  Δc[n-1+k] = -(1-h[i-k])*[i≥k] + h[i+k]*[i+k≤n-1] + (1-h[j-k])*[j≥k] - h[j+k]*[j+k≤n-1]
  
  For lag -kk (kk ≥ 0): c[n-1-kk] = sum_{j'=0}^{n-1-kk} h[j'] * (1-h[j'+kk])
  - h[i] appears as h[j'] when j'=i (if i ≤ n-1-kk): old: 1*(1-h[i+kk]), new: 0
    Change: -(1-h[i+kk]) if i ≤ n-1-kk
  - h[i] appears as h[j'+kk] when j'+kk=i, i.e., j'=i-kk (if i-kk ≥ 0): old: h[i-kk]*0 = 0, new: h[i-kk]*1
    Change: +h[i-kk] if i-kk ≥ 0
  - h[j] appears as h[j'] when j'=j (if j ≤ n-1-kk): old: 0, new: (1-h[j+kk])
    Change: +(1-h[j+kk]) if j ≤ n-1-kk
  - h[j] appears as h[j'+kk] when j'+kk=j, i.e., j'=j-kk (if j-kk ≥ 0): old: h[j-kk]*1 = h[j-kk], new: 0
    Change: -h[j-kk] if j-kk ≥ 0
  
  So for lag -kk:
  Δc[n-1-kk] = -(1-h[i+kk])*[i≤n-1-kk] + h[i-kk]*[i-kk≥0] + (1-h[j+kk])*[j≤n-1-kk] - h[j-kk]*[j-kk≥0]
  
  Notice the symmetry: Δc[n-1+k] (for k≥0) is the same formula as Δc[n-1-kk] (for kk=k) with i and j swapped! That makes sense due to the symmetry of the correlation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def make_binary(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)[:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def jac_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_act(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def jac_act(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxits=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxits is not None and i < len(maxits):
            maxit = maxits[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(obj_full, best_h, jac=jac_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_subgrad(h, n, deadline, n_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    t = 0
    h_avg = h.copy().astype(np.float64)
    for t in range(1, n_iter + 1):
        if t % 500 == 0 and time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.005 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        step = 2.0 / (max_val * (1.0 + t * 0.01))
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        h_avg += (h_new - h_avg) / t
        best_h = h_new
    c_final = compute_c5(project_feasible(h_avg, n), n)
    return best_h, min(best_c, compute_c5(best_h, n)), project_feasible(h_avg, n), c_final

def binary_swap_delta(h_bin, n, i, j, active_indices):
    nmid = n - 1
    deltas = np.zeros(len(active_indices))
    for ai, idx in enumerate(active_indices):
        k = idx - nmid
        d = 0.0
        if k >= 0:
            if i >= k and i < n:
                d -= (1.0 - h_bin[i - k])
            if i + k < n:
                d += h_bin[i + k]
            if j >= k and j < n:
                d += (1.0 - h_bin[j - k])
            if j + k < n:
                d -= h_bin[j + k]
        else:
            kk = -k
            if i < n - kk:
                d -= (1.0 - h_bin[i + kk])
            if i - kk >= 0:
                d += h_bin[i - kk]
            if j < n - kk:
                d += (1.0 - h_bin[j + kk])
            if j - kk >= 0:
                d -= h_bin[j - kk]
        deltas[ai] = d
    return deltas

def phase_binary_greedy(h, n, deadline):
    h_bin = make_binary(h, n)
    best_c = compute_c5(h_bin, n)
    ones = sorted(np.where(h_bin == 1)[0].tolist())
    zeros = sorted(np.where(h_bin == 0)[0].tolist())
    stalled = 0
    while time.time() < deadline and stalled < 5:
        c_vec = compute_c5_vec(h_bin, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.01 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 25:
            active = np.array(np.argsort(-c_vec[active])[:25])
        improved = False
        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        for ii in range(len(ones_arr)):
            if time.time() > deadline:
                break
            i = ones_arr[ii]
            for jj in range(len(zeros_arr)):
                j = zeros_arr[jj]
                deltas = binary_swap_delta(h_bin, n, i, j, active)
                new_active_vals = c_vec[active] + deltas
                new_max = new_active_vals.max()
                if new_max < max_val - 1e-12:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    ones.remove(i)
                    ones.append(j)
                    ones.sort()
                    zeros.remove(j)
                    zeros.append(i)
                    zeros.sort()
                    c_vec[active] = new_active_vals
                    full_c = compute_c5_vec(h_bin, n)
                    new_full_max = full_c.max()
                    if new_full_max < best_c - 1e-12:
                        best_c = new_full_max
                    c_vec = full_c
                    max_val = c_vec.max()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stalled += 1
        else:
            stalled = 0
    return h_bin, best_c

def phase_binary_sa(h, n, deadline, rng, T_init=0.05, T_final=0.001, n_iter=100000):

def phase_grad_descent(h, n, deadline, max_steps=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    count = 0
    for step in steps:
        while time.time() < deadline and count < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx
                    grad[k:n] += (-best_h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                    grad[0:nv] += (-best_h[kk:n]) * dx
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] = old_i - amt
            best_h[best_j] = old_j + amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            count += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] = old_i
                best_h[best_j] = old_j
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def phase_fine_pairs(h, n, deadline, radius=50, deltas=None):

                lo = max(0, i - radius)

                        continue
                    if i == j:
                        continue
                    if best_h[i] - d >= 0.0:
                        old = best_h[i].copy()
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old[i]
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        old = best_h[i].copy()
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old[i]
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 2 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def multi_resolution_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n
    
    # Try at current resolution
    if time.time() < deadline - 60:
        end1 = min(time.time() + 100, deadline - 50)
        h1, c1 = phase_lbfgs(best_h, best_n, end1,
            alphas=[50, 500, 5000, 50000, 200000],
            mu=0.01, maxits=[500, 800, 1000, 1500, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    if time.time() < deadline - 40:
        end2 = min(time.time() + 120, deadline - 35)
        h2, c2 = phase_active_opt(best_h, best_n, end2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    # Try higher resolution
    n2 = n * 2
    if time.time() < deadline - 200:
        h_up = upsample_h(best_h, n, n2)
        end3 = min(time.time() + 150, deadline - 150)
        h3, c3 = phase_lbfgs(h_up, n2, end3,
            alphas=[50, 500, 5000, 50000, 200000],
            mu=0.01, maxits=[500, 800, 1000, 1500, 2000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n2
    if time.time() < deadline - 100 and best_n == n2:
        end4 = min(time.time() + 100, deadline - 50)
        h4, c4 = phase_active_opt(best_h, best_n, end4)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    return best_h, best_c, best_n

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(10, int(deadline - time.time()))
        maxit = min(maxiter, max(100, rem * 5))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
