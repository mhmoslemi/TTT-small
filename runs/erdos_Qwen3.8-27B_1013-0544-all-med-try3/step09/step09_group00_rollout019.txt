            h[p] = 1.0
            cnt += 1
            if cnt >= target:
                break
    if cnt < target:
        for i in range(n):
            if h[i] < 0.5 and cnt < target:
                h[i] = 1.0
                cnt += 1
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(bin(i).count('1') % 2)
    # Adjust to have exactly n/2 ones
    n_ones = int(sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n_ones]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[:n_ones - target]] = 0.0
    return project_feasible(h, n)

def upsample(h):
    n = len(h)
    H = np.zeros(2*n)
    H[0::2] = h
    H[1::2] = h
    return H

def get_active(c, n, top_k, tol_frac=0.001):
    mx = c.max()
    thr = mx - max(1e-10, tol_frac * mx)
    active = np.where(c >= thr)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c[active])[:top_k])
    return active
