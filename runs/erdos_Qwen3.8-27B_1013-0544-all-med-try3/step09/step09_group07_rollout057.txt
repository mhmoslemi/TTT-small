def project_feasible_exact(y, n):
    """Project y onto {x : 0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    
    # y - tau, clipped to [0,1]
    # f(tau) = sum(clip(y - tau, 0, 1)) - target
    # f is monotonically decreasing in tau
    # f(tau) = 0 at the solution
    
    def f(tau):
        return np.sum(np.clip(y - tau, 0.0, 1.0)) - target
    
    # Bisection
    lo = y.min() - 1.0  # f(lo) > 0 definitely
    hi = y.max() + 1.0  # f(hi) < 0 definitely
    
    # Actually: f(lo) = sum(clip(y-lo, 0, 1)) - target = sum(1) - target = n - n/2 = n/2 > 0
    # f(hi) = sum(clip(y-hi, 0, 1)) - target = sum(0) - target = -n/2 < 0
    # Good, f(lo) > 0 and f(hi) < 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2
    return np.clip(y - tau, 0.0, 1.0)

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e6):
    """Optimize using L-BFGS-B with smooth max + penalty for sum constraint."""
    target_sum = n / 2.0
    x0 = np.clip(h.copy(), 0.0, 1.0)

    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        dx = 2.0 / n
        c = np.correlate(x_c, 1.0 - x_c, mode='full')
        O = c * dx
        aO = alpha * O
        aO_max = aO.max()
        exp_vals = np.exp(aO - aO_max)
        w = exp_vals / exp_vals.sum()
        obj_val = float(aO_max / alpha + np.log(exp_vals.sum()) / alpha)

                idx = -lag

                i = rng.randint(0, n - m)
                j = i + m
                if h[i] > h[j] + 1e-7:
                    d = rng.uniform(0.002, 0.05)
                    d = min(d, (h[i] - h[j]) * 0.3)
                elif h[j] > h[i] + 1e-7:
                    d = -rng.uniform(0.002, 0.05)
                    d = max(d, -(h[j] - h[i]) * 0.3)
                else:
                    T *= cool
                    continue
            else:
                # Swap-like move: transfer mass between two positions
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j:
                    continue
                if h[i] > h[j] + 1e-6:
                    d = rng.uniform(0.001, 0.06)
                    d = min(d, (h[i] - h[j]) * 0.5)
                elif h[j] > h[i] + 1e-6:
                    d = -rng.uniform(0.001, 0.06)
                    d = max(d, -(h[j] - h[i]) * 0.5)
                else:
                    T *= cool
                    continue

            if abs(d) < 1e-10:
                T *= cool
                continue

            c_old = compute_all_lags(h, n)
            e_old = smooth_max_value(c_old, alpha_sa)

            h[i] -= d
            h[j] += d
            h = project_feasible_exact(h, n)
            c_new = compute_all_lags(h, n)
            e_new = smooth_max_value(c_new, alpha_sa)

            de = e_new - e_old
            accept = False
            if de < -1e-15:
                accept = True
            elif T > t_min:
                ratio = -de / T
                if ratio < 50:
                    accept = rng.random() < np.exp(-ratio)

            if accept:
                max_new = float(np.max(c_new))
                if max_new < best_c - 1e-15:
                    best_c = max_new
                    best_h = h.copy()
                    it_since_improve = 0
                else:
                    it_since_improve += 1
            else:
                h[i] += d
                h[j] -= d
                h = project_feasible_exact(h, n)
                it_since_improve += 1

            T *= cool

        if it_since_improve > 12000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible_exact(h, n)
            it_since_improve = 0
            T = max(T, 0.003)

    return best_h, best_c


def greedy_lag_reduction(h, n, deadline, rng, max_rounds=50000):
    """Directly target the worst lag with greedy coordinate moves."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top5 = np.argsort(c_all)[-5:]
        improved = False

        for k_star in top5[rng.permutation(5)]:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            # Try moves that reduce this specific lag
            if lag > 0:
                # c = sum h[i+lag]*(1-h[i]) for i in [0, n-1-lag]
                # To reduce: decrease h[i+lag] or increase h[i] where h[i+lag]*h[i] is small
                n_valid = n - lag
                # Pick a random valid position and try to transfer mass
                for attempt in range(20):
                    i = rng.randint(0, n_valid)
                    j = i + lag
                    if h[j] > h[i] + 1e-6:
                        d = rng.uniform(0.001, 0.03)
                        d = min(d, (h[j] - h[i]) * 0.4)
                        if d < 1e-9:
                            continue
                        h[j] -= d
                        h[i] += d
                        h = project_feasible_exact(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[j] += d
                            h[i] -= d
                            h = project_feasible_exact(h, n)
                    elif h[i] > h[j] + 1e-6:
                        d = rng.uniform(0.001, 0.03)
                        d = min(d, (h[i] - h[j]) * 0.4)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible_exact(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible_exact(h, n)
            else:
                m2 = -lag
                n_valid = n - m2
                for attempt in range(20):
                    i = rng.randint(0, n_valid)
                    j = i + m2
                    if h[i] > h[j] + 1e-6:
                        d = rng.uniform(0.001, 0.03)
                        d = min(d, (h[i] - h[j]) * 0.4)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible_exact(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible_exact(h, n)
                    elif h[j] > h[i] + 1e-6:
                        d = rng.uniform(0.001, 0.03)
                        d = min(d, (h[j] - h[i]) * 0.4)
                        if d < 1e-9:
                            continue
                        h[j] -= d
                        h[i] += d
                        h = project_feasible_exact(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[j] += d
                            h[i] -= d
                            h = project_feasible_exact(h, n)

        if not improved and rnd > 100:
            # Perturb and try again
            idx = rng.randint(0, n, size=10)
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible_exact(h, n)

    return best_h, best_c
