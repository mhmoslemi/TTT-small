def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

        if time.time() > deadline:
            break
        for _ in range(500):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-7, 0.004 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_gradient(h, n, idx))
            top_pos = np.argsort(-grad)[:min(30, n)]
            i = int(top_pos[np.random.randint(min(10, len(top_pos)))])
            lo = max(0, i - bsize * 3)
            hi = min(n - bsize, lo + bsize * 6)
            if hi <= lo:
                hi = min(n - bsize, lo + bsize + 1)
            j = int(np.random.randint(lo, hi))
            if j + bsize >= n or i + bsize >= n:
                continue
            si = slice(i, i + bsize)
            sj = slice(j, j + bsize)
            if np.min(h[si]) - step >= -1e-12 and np.max(h[sj]) + step <= 1.0 + 1e-12:
                h[si] -= step
                h[sj] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    h[si] += step
                    h[sj] -= step
            if np.min(h[sj]) - step >= -1e-12 and np.max(h[si]) + step <= 1.0 + 1e-12:
                h[sj] -= step
                h[si] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    h[sj] += step
                    h[si] -= step
    return h, best_c

def perturbation_search(h, n, deadline, n_pert=10):
    """Try small random perturbations to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(n_pert):
        if time.time() > deadline:
            break
        h_pert = h.copy()
        idx = np.random.randint(0, n, size=min(20, n))
        h_pert[idx] += np.random.randn(len(idx)) * 0.02
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        # Quick L-BFGS from perturbed starting point
        t_sub = min(time.time() + 8, deadline)
        h_pert, c_pert = lbfgs_phase(h_pert, n, t_sub, [3000, 8000], mu=500.0)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
    return best_h, best_c

def upsample_fn(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_refine(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x):
        return smooth_obj(x, n, alpha)
    def eq_fun(x):
        return float(x.sum() - n / 2.0)
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_cur = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_cur = 500

    best_h = project_feasible(h_init, n_cur)
    best_c = compute_c5(best_h, n_cur)

    # Phase 1: L-BFGS polish at initial resolution (25s)
    t1 = min(t0 + 25, deadline - 900)
    if time.time() < t1:
        best_h, best_c = lbfgs_phase(best_h, n_cur, t1, [800, 3000, 8000, 20000], mu=200.0)

    # Phase 2: Upsample to n=1000 and optimize (80s)
    n_mid = 1000
    if n_cur < n_mid:
        h_mid = upsample_fn(best_h, n_cur, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        t2 = min(time.time() + 40, deadline - 600)
        if time.time() < t2:
            h_mid, c_mid = lbfgs_phase(h_mid, n_mid, t2, [1000, 4000, 10000, 30000], mu=200.0)
        t2b = min(time.time() + 40, deadline - 500)
        if time.time() < t2b:
            h_mid, c_mid = sqp_optimize(h_mid, n_mid, t2b, max_iter=200, rho_init=0.04)
        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
            n_cur = n_mid

    # Phase 3: Heavy SQP at current resolution (100s)
    t3 = min(time.time() + 100, deadline - 400)
    if time.time() < t3:
        best_h, best_c = sqp_optimize(best_h, n_cur, t3, max_iter=400, rho_init=0.035)

    # Phase 4: Fine local search (80s)
    t4 = min(time.time() + 80, deadline - 300)
    if time.time() < t4:
        best_h, best_c = fine_local_search(best_h, n_cur, t4, radius=60)

    # Phase 5: Block search for coordinated moves (50s)
    t5 = min(time.time() + 50, deadline - 250)
    if time.time() < t5:
        best_h, best_c = block_search(best_h, n_cur, t5, bsize=4)

    # Phase 6: Another SQP round after local improvements (70s)
    t6 = min(time.time() + 70, deadline - 200)
    if time.time() < t6:
        best_h, best_c = sqp_optimize(best_h, n_cur, t6, max_iter=250, rho_init=0.02)

    # Phase 7: Ultra-fine local search (60s)
    t7 = min(time.time() + 60, deadline - 130)
    if time.time() < t7:
        best_h, best_c = fine_local_search(best_h, n_cur, t7, radius=40)

    # Phase 8: Perturbation to escape local minima (30s)
    t8 = min(time.time() + 30, deadline - 100)
    if time.time() < t8:
        best_h, best_c = perturbation_search(best_h, n_cur, t8, n_pert=5)

    # Phase 9: Upsample to n=2000 for high-resolution refinement
    n_high = 2000
    if time.time() < deadline - 180:
        h_high = upsample_fn(best_h, n_cur, n_high)
        c_high = compute_c5(h_high, n_high)
        t9 = min(time.time() + 60, deadline - 120)
        if time.time() < t9:
            h_high, c_high = lbfgs_phase(h_high, n_high, t9, [2000, 6000, 15000], mu=300.0)
        t10 = min(time.time() + 50, deadline - 60)
        if time.time() < t10 and c_high < best_c + 0.001:
            h_high, c_high = sqp_optimize(h_high, n_high, t10, max_iter=150, rho_init=0.03)
        t11 = min(time.time() + 40, deadline - 10)
        if time.time() < t11:
            h_high, c_high = fine_local_search(h_high, n_high, t11, radius=30)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_cur = n_high

    # Final: SLSQP polish if time remains
    if time.time() < deadline - 5:
        h_fin, c_fin = slqp_refine(best_h, len(best_h), deadline - 1, alpha=3000000, maxiter=500)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
