        b[row] = rho
        row += 1
    for i in range(n):
        A[row, i] = -1.0
        b[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, max_active=15):
    best_h = h.copy()
    best_c = compute_c5(h)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [int(active[i]) for i in top_order[:max_active]]
        delta, t_opt = sqp_solve_lp(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-10:
                break
            if no_improve > 30:
                break
    return best_h, best_c

def grad_descent_polish(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        g = np.zeros(n)
        for idx in active:
            g += shift_grad(h, n, int(idx))
        g /= len(active)
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g /= g_norm
        for _ in range(10):
            h_try = project_feasible(h + step * (-g))
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                step = min(step * 1.1, 0.05)
                break
            step *= 0.6
        if step < 1e-6:
            break
    return best_h, best_c

def pair_local_search(h, n, deadline, radius=15, max_evals=200000):
    best_c = compute_c5(h)
    evals = 0
    steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c_vec = compute_c5_vec(h)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad(h, n, int(idx)))
            if len(active) > 0:
                grad /= len(active)
            top_frac = np.argsort(-grad)[:min(n, 200)]
            np.random.shuffle(top_frac)
            for i in top_frac:
                if time.time() > deadline or evals > max_evals:
                    return h, best_c
                i = int(i)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    evals += 1
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                            break
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                            break
                        else:
                            h[i] += step
                            h[j] -= step
                    if improved_any:
                        break
    return h, best_c

def slqp_final(h, n, deadline, alpha=5000000):
    def obj_fn(x):
        c = compute_c5_vec(x)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    def eq_fun(x):
        return np.array([x.sum() - n / 2.0])

    def eq_jac(x):
        return np.ones((1, n))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = project_feasible(res.x)
        c_new = compute_c5(h_new)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(500, 0.5)
    n = len(h_init)

    best_h = project_feasible(h_init)
    best_c = compute_c5(best_h)
    best_n = n

    # Phase 1: L-BFGS at original n (40s)
    t1 = min(t0 + 40, deadline - 900)
    if time.time() < t1:
        best_h, best_c = run_lbfgs(best_h, n, t1,
            alphas=[500, 2000, 10000, 50000],
            mus=[10.0, 50.0, 500.0, 2000.0])

    # Phase 2: SQP at n (80s)
    t2 = min(t0 + 120, deadline - 750)
    if time.time() < t2:
        best_h, best_c = sqp_optimize(best_h, n, t2, max_iter=500,
                                       rho_init=0.05, max_active=15)

    # Phase 3: Gradient polish at n (30s)
    t3 = min(t0 + 150, deadline - 700)
    if time.time() < t3:
        best_h, best_c = grad_descent_polish(best_h, n, t3, max_iter=5000)

    # Phase 4: Local search at n (30s)
    t4 = min(t0 + 180, deadline - 650)
    if time.time() < t4:
        best_h, best_c = pair_local_search(best_h, n, t4, radius=10, max_evals=100000)

    # Phase 5: Upsample to n*4 and optimize (main high-res phase)
    n_high = n * 4
    if n_high > 4000:
        n_high = 4000
    if time.time() < deadline - 500:
        h_high = upsample(best_h, n_high)
        c_high = compute_c5(h_high)

        # L-BFGS at high res (120s)
        t5 = min(time.time() + 120, deadline - 350)
        if time.time() < t5 and c_high < best_c + 0.005:
            h_high, c_high = run_lbfgs(h_high, n_high, t5,
                alphas=[1000, 5000, 20000, 100000],
                mus=[50.0, 200.0, 1000.0, 5000.0])

        # SQP at high res (120s)
        t6 = min(time.time() + 120, deadline - 200)
        if time.time() < t6:
            h_high, c_high = sqp_optimize(h_high, n_high, t6, max_iter=400,
                                           rho_init=0.04, max_active=12)

        # Gradient polish (40s)
        t7 = min(time.time() + 40, deadline - 150)
        if time.time() < t7:
            h_high, c_high = grad_descent_polish(h_high, n_high, t7, max_iter=5000)

        # Local search (50s)
        t8 = min(time.time() + 50, deadline - 80)
        if time.time() < t8:
            h_high, c_high = pair_local_search(h_high, n_high, t8,
                                                radius=8, max_evals=150000)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high

    # Phase 6: Try even higher resolution if time allows
    n_very_high = min(best_n * 2, 8000)
    if n_very_high > best_n and time.time() < deadline - 200:
        h_vh = upsample(best_h, n_very_high)
        c_vh = compute_c5(h_vh)
        t9 = min(time.time() + 80, deadline - 100)
        if time.time() < t9 and c_vh < best_c + 0.01:
            h_vh, c_vh = run_lbfgs(h_vh, n_very_high, t9,
                alphas=[5000, 30000, 100000],
                mus=[200.0, 1000.0, 5000.0])
            t10 = min(time.time() + 50, deadline - 50)
            if time.time() < t10:
                h_vh, c_vh = sqp_optimize(h_vh, n_very_high, t10,
                                           max_iter=200, rho_init=0.03, max_active=10)
            if c_vh < best_c:
                best_c = c_vh
                best_h = h_vh
                best_n = n_very_high

    # Phase 7: Final polish on best solution
    if time.time() < deadline - 5:
        h_fin, c_fin = slqp_final(best_h, best_n, deadline - 2, alpha=5000000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility enforcement
    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
