import numpy as np
from scipy.optimize import minimize

def make_even(n):

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(12):

        diff = target - s
        if abs(diff) < 1e-13:

        else:
            mask = h > 1e-12
        cnt = int(mask.sum())

            break

def full_c_array_float(x, n):
    dx = 2.0 / n
    x = np.asarray(x, dtype=float)
    return np.correlate(x, 1.0 - x, mode="full") * dx


def compute_c5_val(x, n):
    return float(np.max(full_c_array_float(x, n)))


def interpolate_h(x, n):
    x = np.asarray(x, dtype=float)
    if x.size == n:
        return project_feasible(x, n)
    old = x.size
    if old == 0:
        return project_feasible(np.full(n, 0.5), n)
    if old == 1:
        return project_feasible(np.full(n, float(x[0])), n)
    xo = np.linspace(0.0, 2.0, old)
    xn = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(xn, xo, x), n)


def binary_from_h(x, n):
    x = np.asarray(x, dtype=float)
    if x.size == n:
        xb = x
    else:
        xb = interpolate_h(x, n)
    xb = np.clip(xb, 0.0, 1.0)
    target = n // 2
    order = np.argsort(-xb, kind="stable")
    h = np.zeros(n, dtype=np.uint8)
    h[order[:target]] = 1
    return h


def random_binary(n, rng):
    h = np.zeros(n, dtype=np.uint8)
    cnt = n // 2
    if cnt > 0:
        idx = rng.choice(n, size=cnt, replace=False)
        h[idx] = 1
    return h


def thue_morse_binary(n, rng):
    m = 1
    while m < n:
        m *= 2
    seq = np.array([1], dtype=np.uint8)
    while seq.size < m:
        seq = np.concatenate((seq, 1 - seq))
    h = seq[:n].copy()
    target = n // 2
    cnt = int(h.sum())
    if cnt < target:
        z = np.flatnonzero(h == 0)
        rng.shuffle(z)
        h[z[:target - cnt]] = 1
    elif cnt > target:
        o = np.flatnonzero(h == 1)
        rng.shuffle(o)
        h[o[:cnt - target]] = 0
    return h


def build_bin_CD(h, n):
    h32 = h.astype(np.int32)
    z32 = (1 - h).astype(np.int32)
    full = np.correlate(h32, z32, mode="full")
    C = np.zeros(n, dtype=np.int32)
    D = np.zeros(n, dtype=np.int32)
    C[1:] = full[n:2 * n - 1]
    D[1:] = full[n - 2::-1]
    return C, D


def build_bin_sets(h):
    zeros = np.flatnonzero(h == 0).astype(np.int32)
    ones = np.flatnonzero(h == 1).astype(np.int32)
    return zeros, ones


def bin_flip_up(p, n, h, z, C, D):
    left = n - p - 1
    if left > 0:
        sl = slice(1, left + 1)
        C[sl] += z[p + 1:p + 1 + left]
        D[sl] -= h[p + 1:p + 1 + left]
    right = p
    if right > 0:
        sl = slice(1, right + 1)
        C[sl] -= h[p - right:p]
        D[sl] += z[p - right:p]
    h[p] = 1
    z[p] = 0


def bin_flip_down(p, n, h, z, C, D):
    left = n - p - 1
    if left > 0:

        C[sl] -= z[p + 1:p + 1 + left]

    right = p

        sl = slice(1, right + 1)
        C[sl] += h[p - right:p]
        D[sl] -= z[p - right:p]
    h[p] = 0
    z[p] = 1

            if M < best_M:
                best_M = M
                best_c = float(M) * 2.0 / n
                best_h = h.copy()
                no_best = 0
            else:
                no_best += 1
            if M > best_M + 20:
                h = best_h.copy()
                z = (1 - h).astype(np.uint8)
                C, D = build_bin_CD(h, n)
                M = best_M
                zeros, ones = build_bin_sets(h)
                no_best = 0
        else:
            bin_flip_up(q, n, h, z, C, D)
            bin_flip_down(p, n, h, z, C, D)
            no_best += 1

        trial += 1
        if no_best > max_no_best:
            h = best_h.copy()
            z = (1 - h).astype(np.uint8)
            C, D = build_bin_CD(h, n)
            pert = int(rng.randint(3, 9))
            for _ in range(pert):
                z2 = np.flatnonzero(h == 0)
                o2 = np.flatnonzero(h == 1)
                if z2.size == 0 or o2.size == 0:
                    break
                p2 = int(z2[int(rng.randint(z2.size))])
                q2 = int(o2[int(rng.randint(o2.size))])
                bin_flip_up(p2, n, h, z, C, D)
                bin_flip_down(q2, n, h, z, C, D)
            M = int(max(int(C.max()), int(D.max())))
            zeros, ones = build_bin_sets(h)
            no_best = 0

    return best_h.astype(float), best_c


def all_lags_smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    x = np.asarray(x, dtype=float)
    full = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(np.max(full))
    shift = alpha * (full - m)
    lse = float(np.log(np.sum(np.exp(shift))))
    val = m + lse / alpha
    s = float(np.sum(x)) - n / 2.0
    if mu != 0.0:
        val += mu * s * s
    return float(val)


def all_lags_smooth_grad(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / n
    full = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(np.max(full))
    shift = alpha * (full - m)
    w = np.exp(shift)
    w /= w.sum()
    grad = np.zeros(n, dtype=float)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = float(w[idx])
        if wi < 1e-13:
            continue
        wm = wi * dx
        if idx == center:
            grad += wm * (1.0 - 2.0 * x)
        elif idx > center:
            k = idx - center
            L = n - k
            grad[:L] += wm * (1.0 - x[k:])
            grad[k:] += -wm * x[:L]
        else:
            k = center - idx
            L = n - k
            grad[:L] += -wm * x[k:]
            grad[k:] += wm * (1.0 - x[:L])
    if mu != 0.0:
        s = float(np.sum(x)) - n / 2.0
        grad += 2.0 * mu * s
    return grad


def lbfgs_all_lags(x0, n, deadline):
    x = project_feasible(x0, n)
    best_x = x.copy()
    best_c = compute_c5_val(x, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(80, 1.0), (300, 2.0), (1000, 5.0), (3000, 10.0), (10000, 20.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(
                all_lags_smooth_obj,
                x,
                jac=all_lags_smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 70, "ftol": 1e-16, "gtol": 1e-12, "maxfun": 300},
            )
            x_new = project_feasible(res.x, n)
            c_new = compute_c5_val(x_new, n)
            if c_new < best_c:
                best_c = c_new
                best_x = x_new.copy()
            if c_new <= best_c + 1e-10:
                x = x_new
            else:
                x = best_x.copy()
        except Exception:
            pass
    return best_x, best_c


def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_jacobian(x, n):
    return np.ones(np.asarray(x).size, dtype=float)


def active_values(x, n, active_idx):
    dx = 2.0 / n
    x = np.asarray(x, dtype=float)
    vals = []
    center = n - 1
    for idx in active_idx:
        idx = int(idx)
        if idx == center:
            vals.append(dx * float(np.dot(x, 1.0 - x)))
        elif idx > center:
            k = idx - center
            L = n - k
            vals.append(dx * float(np.dot(x[:L], 1.0 - x[k:])))
        else:
            k = center - idx
            L = n - k
            vals.append(dx * float(np.dot(x[k:], 1.0 - x[:L])))
    return np.asarray(vals, dtype=float)


def active_smooth_obj(x, n, alpha, mu, active_idx):
    vals = active_values(x, n, active_idx)
    if vals.size == 0:
        val = 0.0
    else:
        m = float(np.max(vals))
        shift = alpha * (vals - m)
        lse = float(np.log(np.sum(np.exp(shift))))
        val = m + lse / alpha
    s = float(np.sum(x)) - n / 2.0
    if mu != 0.0:
        val += mu * s * s
    return float(val)


def active_smooth_grad(x, n, alpha, mu, active_idx):
    x = np.asarray(x, dtype=float)
    vals = active_values(x, n, active_idx)
    grad = np.zeros(n, dtype=float)
    if vals.size > 0:
        m = float(np.max(vals))
        shift = alpha * (vals - m)
        ex = np.exp(shift)
        w = ex / ex.sum()
        dx = 2.0 / n
        center = n - 1
        for idx, wi in zip(active_idx, w):
            idx = int(idx)
            wi = float(wi)
            if wi <= 1e-15:
                continue
            scale = wi * dx
            if idx == center:
                grad += scale * (1.0 - 2.0 * x)
            elif idx > center:
                k = idx - center
                L = n - k
                grad[:L] += scale * (1.0 - x[k:])
                grad[k:] += -scale * x[:L]
            else:
                k = center - idx
                L = n - k
                grad[:L] += -scale * x[k:]
                grad[k:] += scale * (1.0 - x[:L])
    if mu != 0.0:
        s = float(np.sum(x)) - n / 2.0
        grad += 2.0 * mu * s
    return grad


def select_active_indices(full_c, n, size=35, max_total=80, close_tol=1e-5):
    full_c = np.asarray(full_c, dtype=float)
    length = full_c.size
    size = int(max(1, min(size, length)))
    if length <= size:
        arr = np.arange(length)
    else:
        m = float(np.max(full_c))
        top = np.argpartition(-full_c, size - 1)
        close = np.flatnonzero(full_c >= m - close_tol)
        arr = np.unique(np.concatenate((top, close)))
        if arr.size > max_total:
            order = np.argsort(-full_c[arr], kind="stable")
            arr = arr[order[:max_total]]
    return [int(i) for i in arr]


def active_continuous_refine(x0, n, deadline, best_c):
    x = project_feasible(x0, n)
    c0 = compute_c5_val(x, n)
    best_x = x.copy()
    best_c_run = c0
    bounds = [(0.0, 1.0)] * n

    alphas = [500, 2000, 8000, 30000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        full = full_c_array_float(x, n)
        cur = float(np.max(full))
        active = select_active_indices(full, n, size=30, max_total=70, close_tol=1e-5)
        if not active:
            active = [n - 1]
        mu = 1.0
        try:
            res = minimize(
                active_smooth_obj,
                x,
                jac=active_smooth_grad,
                args=(n, alpha, mu, active),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 70, "ftol": 1e-15, "gtol": 1e-11, "maxfun": 300},
            )
            x_new = project_feasible(res.x, n)
            c_new = compute_c5_val(x_new, n)
            if c_new < best_c_run:
                best_c_run = c_new
                best_x = x_new.copy()
            if c_new <= cur + 1e-10:
                x = x_new
            else:
                x = best_x.copy()
        except Exception:
            pass

    if n <= 500 and time.time() < deadline - 25:
        for alpha in [5000, 20000]:
            if time.time() > deadline - 20:
                break
            full = full_c_array_float(x, n)
            cur = float(np.max(full))
            active = select_active_indices(full, n, size=40, max_total=70, close_tol=1e-5)
            if not active:
                active = [n - 1]
            cons = [
                {
                    "type": "eq",
                    "fun": sum_constraint,
                    "jac": sum_jacobian,
                    "args": (n,),
                }
            ]
            try:
                res = minimize(
                    active_smooth_obj,
                    x,
                    jac=active_smooth_grad,
                    args=(n, alpha, 0.0, active),
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 70, "ftol": 1e-12, "disp": 0},
                )
                x_new = project_feasible(res.x, n)
                c_new = compute_c5_val(x_new, n)
                if c_new < best_c_run:
                    best_c_run = c_new
                    best_x = x_new.copy()
                if c_new <= cur + 1e-9:
                    x = x_new
                else:
                    x = best_x.copy()
            except Exception:
                pass

    return best_x, best_c_run


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    work_end = t0 + max(30, int(budget_s) - 25)

    init_arr = None
    try:
        tmp = np.asarray(initial_h_values, dtype=float)
        if tmp.size > 0:
            init_arr = tmp
    except Exception:
        init_arr = None

    if init_arr is None:
        n_p = 500
        parent = np.full(500, 0.5)
    else:
        n_p = int(init_arr.size)
        parent = init_arr.astype(float)

    if n_p < 200:
        n0 = 500
    elif n_p > 1200:
        n0 = 1000
    else:
        n0 = make_even(n_p)

    x0 = interpolate_h(parent, n0) if parent.size != n0 else project_feasible(parent.astype(float), n0)
    best_h = project_feasible(x0, n0)
    best_c = compute_c5_val(best_h, n0)

    d1 = min(t0 + 110, work_end - 550)
    d2a = min(t0 + 330, work_end - 350)
    d2b = min(t0 + 430, work_end - 250)
    d2c = min(t0 + 490, work_end - 150)
    d3 = min(t0 + 660, work_end - 100)
    d4 = min(t0 + 920, work_end - 20)
    d5 = work_end

    if d1 - time.time() > 40:
        if n0 <= 600:
            h1, c1 = lbfgs_all_lags(best_h, n0, d1)
        else:
            h1, c1 = active_continuous_refine(best_h, n0, d1, best_c)
        if c1 < best_c:
            best_c = c1
            best_h = project_feasible(h1, len(h1))

    def _use_candidate(h_candidate, n_candidate, c_candidate):
        nonlocal_best = (best_h, best_c) if True else None
        return nonlocal_best

    if d2a - time.time() > 70:
        h_bin = binary_from_h(best_h, n0)
        hb, cb = binary_local_search(h_bin, n0, d2a, seed)
        if cb < best_c:
            hb_f = project_feasible(hb.astype(float), n0)
            cb_actual = compute_c5_val(hb_f, n0)
            if cb_actual < best_c:
                best_c = cb_actual
                best_h = hb_f

    if d2b - time.time() > 60:
        h_th = thue_morse_binary(n0, np.random.RandomState(seed + 1))
        hb, cb = binary_local_search(h_th, n0, d2b, seed + 1)
        if cb < best_c:
            hb_f = project_feasible(hb.astype(float), n0)
            cb_actual = compute_c5_val(hb_f, n0)
            if cb_actual < best_c:
                best_c = cb_actual
                best_h = hb_f

    if d2c - time.time() > 50:
        h_r = random_binary(n0, np.random.RandomState(seed + 2))
        hb, cb = binary_local_search(h_r, n0, d2c, seed + 2)
        if cb < best_c:
            hb_f = project_feasible(hb.astype(float), n0)
            cb_actual = compute_c5_val(hb_f, n0)
            if cb_actual < best_c:
                best_c = cb_actual
                best_h = hb_f

    if d3 - time.time() > 70 and n0 <= 1200:
        rem = d3 - time.time()
        d3a = time.time() + 0.62 * rem
        h3, c3 = active_continuous_refine(best_h, n0, d3a, best_c)
        if c3 < best_c:
            best_c = c3
            best_h = project_feasible(h3, len(h3))

        if d3 - time.time() > 35:
            rngp = np.random.RandomState(seed + 3)
            x_p = best_h + 0.045 * (rngp.rand(len(best_h)) - 0.5)
            x_p = project_feasible(x_p, n0)
            h3b, c3b = active_continuous_refine(x_p, n0, d3, best_c)
            if c3b < best_c:
                best_c = c3b
                best_h = project_feasible(h3b, len(h3b))

    n1 = make_even(min(2 * n0, 1600))
    if n1 != n0 and d4 - time.time() > 70:
        x1 = interpolate_h(best_h, n1)
        h_bin1 = binary_from_h(x1, n1)
        hb, cb = binary_local_search(h_bin1, n1, d4, seed + 100)
        if cb < best_c:
            hb_f = project_feasible(hb.astype(float), n1)
            cb_actual = compute_c5_val(hb_f, n1)
            if cb_actual < best_c:
                best_c = cb_actual
                best_h = hb_f

    if d5 - time.time() > 45 and len(best_h) <= 1200:
        h5, c5 = active_continuous_refine(best_h, len(best_h), d5, best_c)
        if c5 < best_c:
            best_c = c5
            best_h = project_feasible(h5, len(h5))

    n_final = int(len(best_h))
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5_val(best_h, n_final)
    return np.asarray(best_h, dtype=float), float(best_c), int(n_final)
