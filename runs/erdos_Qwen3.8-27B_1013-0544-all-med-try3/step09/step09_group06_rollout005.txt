    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (target_sum / tot)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    if s_max > 500:
        return m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def soft_max_weights_full(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    if s_max > 500:
        w = np.exp(s - s_max)
    else:
        w = np.exp(s)
    w = w / w.sum()
    return w

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += (-h[m:n])
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += (-h[0:n-m])
    return grad

def grad_smooth_max(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(c)[-top_k:]
        c_act = c[idx]
        m = c_act.max()
        s = alpha * (c_act - m)
        s_max = s.max()
        if s_max > 500:
            w = np.exp(s - s_max)
        else:
            w = np.exp(s)
        w = w / w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(idx):
            if w[i] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[i] * _lag_gradient(h, n, lag)
        return grad * dx
    else:
        w = soft_max_weights_full(c, alpha)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * _lag_gradient(h, n, lag)
        return grad * dx

def momentum_optimize(h, n, deadline, alpha_start=100.0, alpha_end=50000.0,
                      lr_start=0.4, momentum=0.92, top_k=None, patience_limit=800):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    patience = 0

    while time.time() < deadline:
        g = grad_smooth_max(h_curr, n, min(alpha_start, alpha_end) + 
                            (max(alpha_start, alpha_end) - min(alpha_start, alpha_end)) *
                            min(1.0, patience / 3000.0) if patience > 0 else alpha_start,
                            top_k=top_k)
        # Adaptive alpha based on progress
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, patience / 5000.0)
        g = grad_smooth_max(h_curr, n, alpha, top_k=top_k)
        
        v = momentum * v + lr_start * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > patience_limit:
            lr_start *= 0.6
            v *= 0.3
            patience = 0
        if lr_start < 1e-10:
            break
    
    return best_h, best_c

def vectorized_local_search(h, n, deadline, top_m=5, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_m:]
        improved = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            if lag > 0:
                diff = h[:n-m].copy() - h[m:].copy()
            else:
                diff = h[m:].copy() - h[:n-m].copy()
            
            for d in [0.02, 0.01, 0.005, 0.002, 0.001]:
                if time.time() > deadline:
                    break
                if lag > 0:
                    mask = diff > d
                else:
                    mask = (-diff) > d
                n_transfer = int(mask.sum())
                if n_transfer == 0:
                    continue
                idx = np.where(mask)[0]
                h_old = h.copy()
                if lag > 0:
                    h[idx] -= d
                    h[idx + m] += d
                else:
                    h[idx + m] -= d
                    h[idx] += d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                    c_all = compute_all_lags(h, n)
                else:
                    h = h_old
        
        if not improved:
            # Try very small steps on top lags
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-top_m:]
            for k_idx in top_idx:
                lag = k_idx - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                for d in [0.0005, 0.0002]:
                    if time.time() > deadline:
                        break
                    if lag > 0:
                        diff = h[:n-m].copy() - h[m:].copy()
                        mask = diff > d
                    else:
                        diff = h[m:].copy() - h[:n-m].copy()
                        mask = (-diff) > d
                    if mask.sum() == 0:
                        continue
                    idx = np.where(mask)[0]
                    h_old = h.copy()
                    if lag > 0:
                        h[idx] -= d
                        h[idx + m] += d
                    else:
                        h[idx + m] -= d
                        h[idx] += d
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h = h_old
            if not improved:
                break
    
    return h, best_c

def pairwise_local_search(h, n, deadline, top_m=3):
    """Slower but more targeted pairwise search"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_m:]
        improved = False
        
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            for d in [0.005, 0.002, 0.001]:
                if time.time() > deadline:
                    break
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if lag > 0:
                        if h[i] > h[j]:
                            delta = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.5, (1 - h[j]) * 0.5)
                            if delta < 1e-10:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                    else:
                        if h[j] > h[i]:
                            delta = min(d, (h[j] - h[i]) * 0.3, h[j] * 0.5, (1 - h[i]) * 0.5)
                            if delta < 1e-10:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[j] += delta
                                h[i] -= delta
        if not improved:
            break
    
    return h, best_c

def slsqp_active(h, n, alpha, top_m, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        s = alpha * (active - m)
        s_max = s.max()
        if s_max > 500:
            return m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def de_global_search(h_init, n, deadline, pop_size=12, F=0.5, CR=0.7, seed_offset=100):
    rng_local = np.random.RandomState(seed_offset)
    time_budget = deadline - time.time()
    if time_budget < 30:
        return h_init.copy(), compute_c5(h_init, n)
    
    # Initialize population around h_init
    pop = np.zeros((pop_size, n))
    pop[0] = project_feasible(h_init.copy(), n)
    for i in range(1, pop_size):
        scale = 0.08 * (1 + 0.3 * i / pop_size)
        pert = rng_local.normal(0, scale, n)
        pop[i] = project_feasible(h_init + pert, n)
    
    fitness = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]
    
    gen = 0
    while time.time() < deadline - 10:
        gen += 1
        # Adaptive mutation
        F_adapt = F + 0.2 * (1 - min(1.0, gen / 500.0))
        
        for i in range(pop_size):
            if time.time() > deadline - 5:
                break
            others = [j for j in range(pop_size) if j != i]
            a, b, c_idx = rng_local.choice(others, 3, replace=False)
            mutant = pop[a] + F_adapt * (pop[b] - pop[c_idx])
            
            # Binary crossover
            trial = pop[i].copy()
            r = rng_local.random(n)
            trial[r < CR] = mutant[r < CR]
            trial = project_feasible(trial, n)
            
            fit = compute_c5(trial, n)
            if fit < fitness[i]:
                pop[i] = trial
                fitness[i] = fit
                if fit < best_c:
                    best_c = fit
                    best_h = trial.copy()
    
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Momentum with increasing sharpness
    t1 = min(deadline - 80, time.time() + (deadline - time.time()) * 0.35)
    h1, c1 = momentum_optimize(h, n, t1, alpha_start=200.0, alpha_end=20000.0,
                                lr_start=0.3, momentum=0.92, top_k=8, patience_limit=500)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    h = h1
    
    # Phase 2: Higher alpha momentum
    t2 = min(deadline - 40, time.time() + (deadline - time.time()) * 0.3)
    h2, c2 = momentum_optimize(h, n, t2, alpha_start=20000.0, alpha_end=200000.0,
                                lr_start=0.1, momentum=0.94, top_k=4, patience_limit=400)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    h = h2
    
    # Phase 3: SLSQP
    if time.time() < deadline - 25:
        for tm in [3, 5]:
            if time.time() > deadline - 15:
                break
            h3 = slsqp_active(h, n, 100000, tm, maxiter=2000, ftol=1e-16)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c = c3
                best_h = h3
                h = h3
    
    # Phase 4: Vectorized local search
    if time.time() < deadline - 8:
        t4 = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        h4, c4 = vectorized_local_search(h, n, t4, top_m=5, max_rounds=500)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Pairwise local search (finer)
    if time.time() < deadline - 3:
        h5, c5 = pairwise_local_search(best_h, n, deadline - 0.5, top_m=3)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        # Main optimization at parent resolution
        alloc_main = min(deadline - t0 - 200, 400)
        h_opt, c_opt = optimize_pipeline(h_init, n_init, t0 + alloc_main, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(500, 0.5)
        h_opt, c_opt = optimize_pipeline(h_start, 500, t0 + 300, rng)
        best_h, best_c = h_opt, c_opt
    
    # DE global search to escape local minima
    if time.time() < deadline - 120:
        de_time = min(120, deadline - time.time() - 100)
        if de_time > 40:
            h_de, c_de = de_global_search(best_h, len(best_h), time.time() + de_time,
                                          pop_size=10, F=0.5, CR=0.7, seed_offset=seed + 200)
            if c_de < best_c:
                # Polish DE result
                if time.time() < deadline - 60:
                    h_pol, c_pol = optimize_pipeline(h_de, len(best_h),
                                                     min(deadline - 50, time.time() + 60), rng)
                    if c_pol < best_c:
                        best_c = c_pol
                        best_h = h_pol
            # Even if DE didn't beat, try a second round with different params
            elif time.time() < deadline - 80:
                de_time2 = min(60, deadline - time.time() - 70)
                if de_time2 > 30:
                    h_de2, c_de2 = de_global_search(best_h, len(best_h), time.time() + de_time2,
                                                    pop_size=8, F=0.8, CR=0.9, seed_offset=seed + 500)
                    if c_de2 < best_c:
                        if time.time() < deadline - 40:
                            h_pol, c_pol = optimize_pipeline(h_de2, len(best_h),
                                                             min(deadline - 30, time.time() + 40), rng)
                            if c_pol < best_c:
                                best_c = c_pol
                                best_h = h_pol
                        else:
                            best_c = c_de2
                            best_h = h_de2
    
    # Try alternative resolutions
    n_best = len(best_h)
    for n_try in [n_best + 50, n_best - 50, 600]:
        if n_try < 200 or n_try > 800:
            continue
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 20
        if remaining < 50:
            break
        alloc = min(70, remaining * 0.25)
        if alloc < 35:
            break
        
        x_best = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        h_opt, c_opt = optimize_pipeline(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_best = n_try
    
    # Multistart perturbations at best resolution
    if time.time() < deadline - 40:
        n_restarts = max(1, int((deadline - time.time() - 30) / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(40, (deadline - time.time() - 15) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_pipeline(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polishing
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        # Final vectorized local search
        if time.time() < deadline - 1:
            best_h, best_c = vectorized_local_search(best_h, n, deadline - 0.5, top_m=3, max_rounds=200)
        
        # Final SLSQP
        if time.time() < deadline - 0.5:
            h_fin = slsqp_active(best_h, n, 200000, 3, maxiter=1500, ftol=1e-16)
            c_fin = compute_c5(h_fin, n)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
