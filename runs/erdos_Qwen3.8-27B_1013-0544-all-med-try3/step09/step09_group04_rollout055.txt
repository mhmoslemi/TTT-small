Let me think about the actual mathematics more carefully:

C₅ = max_k ∫₀² h(x)(1-h(x+k)) dx

This is the maximum over all shifts k of the "anti-overlap" - where h is 1 but h shifted is 0.

For a step function with n points and dx = 2/n:
- h has sum = n/2 (integral = 1)
- We want to minimize the maximum correlation of h with (1-h) at all shifts

Key insight: If h is roughly 0.5 everywhere, then h(x)(1-h(x+k)) ≈ 0.25 for all k, giving integral ≈ 0.25*2 = 0.5. But we can do much better.

If h is 0 or 1 (binary), then h(x)(1-h(x+k)) is 1 only where h(x)=1 and h(x+k)=0. The integral becomes the measure of {x: h(x)=1, h(x+k)=0}.

For the binary case with n/2 ones and n/2 zeros, we want to arrange them so that for every shift k, the number of (1,0) pairs is minimized.

This is related to the theory of "difference sets" and "optimal binary sequences."

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n as 0/1"""
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Convert to [0,1] with approximately n/2 ones
    # The Thue-Morse sequence has approximately n/2 ones
    h = tm.astype(float)
    return project_feasible(h, n)

def periodic_pattern(n, period, ones_in_period):
    """Create a periodic pattern"""
    h = np.zeros(n)
    pattern = np.zeros(period)
    # Place ones_in_period ones in the pattern
    pattern[:ones_in_period] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return project_feasible(h, n)

def random_binary(n, seed):
    """Random binary-like initialization"""
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    # Make it more binary
    h = (h < 0.5).astype(float)
    return project_feasible(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if not np.isfinite(mx):
        return 1e10
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if not np.isfinite(mx):
        return np.zeros(n)
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if not np.isfinite(smx):
        return 1e10
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if not np.isfinite(smx):
        return np.zeros(n)
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w /= s
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, alphas=None, mu=0.01, maxiters=None):
    """Run L-BFGS-B with increasing alpha"""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 1000, 10000, 100000, 500000]
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = min(3000, max(200, int(remaining * 1.5)))
        try:
            res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def de_search(n, deadline, pop_size=30, seed=42):
    """Differential evolution at low resolution for global exploration"""
    rng = np.random.RandomState(seed)
    target = n / 2.0
    
    # Initialize population
    pop = []
    # Include some structured inits
    pop.append(project_feasible(thue_morse(n), n))
    pop.append(project_feasible(random_binary(n, seed), n))
    pop.append(project_feasible(np.full(n, 0.5) + 0.1 * rng.randn(n), n))
    
    for i in range(pop_size - 3):
        h = rng.random(n)
        h = (h < 0.5).astype(float)
        h = project_feasible(h, n)
        pop.append(h)
    
    # Evaluate
    fitness = [compute_c5(h, n) for h in pop]
    
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]
    
    F = 0.8  # mutation factor
    CR = 0.8  # crossover rate
    
    gen = 0
    while time.time() < deadline:
        gen += 1
        if time.time() > deadline:
            break
        
        # Adaptive parameters
        progress = min(1.0, gen / 200.0)
        F_curr = F * (1.0 - 0.3 * progress)
        CR_curr = CR + 0.1 * progress
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            # Select three distinct individuals
            candidates = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(candidates, 3, replace=False)
            
            # Mutation
            v = pop[a] + F_curr * (pop[b] - pop[c])
            
            # Crossover
            trial = pop[i].copy()
            j0 = rng.randint(n)
            for j in range(n):
                if rng.random() < CR_curr or j == j0:
                    trial[j] = v[j]
            
            trial = project_feasible(trial, n)
            f_trial = compute_c5(trial, n)
            
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c:
                    best_c = f_trial
                    best_h = trial.copy()
    
    return best_h, best_c

def pair_perturb(h, n, deadline, radius=30):
    """Local pair perturbation search"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Try random pairs within radius
            for _ in range(200):
                if time.time() > deadline:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                lo = i - radius
                hi = i + radius + 1
                if j < lo or j >= hi:
                    continue
                
                if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                
                if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
    
    return best_h, best_c

def sin_perturbation(h, n, deadline):
    """Perturb with sinusoidal patterns and optimize"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = np.linspace(0, 2, n)
    
    for freq in [1, 2, 3, 5, 7, 10, 15, 20, 30, 50]:

    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    best_n = n_init

    # Phase 1: Quick refine at initial resolution (25s)
    p1_end = t0 + 25
    h1, c1 = lbfgs_refine(best_h, best_n, p1_end,
        alphas=[200, 2000, 20000, 100000], mu=0.01, maxiters=[400, 600, 800, 1000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=2000
    n_work = 2000
    if time.time() < deadline - 200:
        h_work = upsample(best_h, best_n, n_work)
        c_work = compute_c5(h_work, n_work)
        if c_work <= best_c + 0.002:
            best_h = h_work
            best_n = n_work
            if c_work < best_c:
                best_c = c_work

    # Phase 3: L-BFGS at working resolution (200s)
    p3_end = t0 + 230
    h3, c3 = lbfgs_refine(best_h, best_n, p3_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.005, maxiters=[400, 500, 700, 800, 1000, 1200, 1500, 2000])
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Single lag dive (120s)
    p4_end = t0 + 350
    if time.time() < p4_end - 10:
        h4, c4 = single_lag_dive(best_h, best_n, p4_end)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active set optimization (150s)
    p5_end = t0 + 500
    if time.time() < p5_end - 10:
        h5, c5 = active_set_opt(best_h, best_n, p5_end, top_k=3)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Perturbation exploration (120s)
    p6_end = t0 + 620
    if time.time() < p6_end - 10:
        h6, c6 = perturb_and_refine(best_h, best_n, p6_end, seed=seed + 7)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Binary push + refine (60s)
    p7_end = t0 + 680
    if time.time() < p7_end - 10:
        h_bin = binary_push(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        if c_bin < best_c + 0.01:
            h7, c7 = lbfgs_refine(h_bin, best_n, p7_end,
                alphas=[10000, 100000, 1000000], mu=0.01, maxiters=[500, 800, 1000])
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: DE exploration at lower res then upsample (100s)
    p8_end = t0 + 780
    if time.time() < p8_end - 20:
        n_de = 200
        h_de, c_de = de_explore(n_de, p8_end, pop_size=15, seed=seed + 200)
        if c_de < best_c + 0.01:
            h_de_up = upsample(h_de, n_de, best_n)
            c_de_up = compute_c5(h_de_up, best_n)
            if c_de_up < best_c:
                h8, c8 = lbfgs_refine(h_de_up, best_n, p8_end,
                    alphas=[10000, 100000], mu=0.005, maxiters=[500, 800])
                if c8 < best_c:
                    best_c = c8
                    best_h = h8

    # Phase 9: Final active set push (80s)
    p9_end = t0 + 860
    if time.time() < p9_end - 10:
        h9, c9 = active_set_opt(best_h, best_n, p9_end, top_k=2)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Single lag dive again (60s)
    p10_end = t0 + 920
    if time.time() < p10_end - 5:
        h10, c10 = single_lag_dive(best_h, best_n, p10_end)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 11: Pair micro-optimization (remaining time)
    p11_end = deadline - 1
    if time.time() < p11_end:
        h11, c11 = pair_micro(best_h, best_n, p11_end, radius=20)
        if c11 < best_c:
            best_c = c11
            best_h = h11

    # Final: one more very short active set
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = int(np.argmax(c_vec))
        try:
            res = minimize(multi_lag_obj, best_h, jac=multi_lag_grad,
                           args=(best_n, np.array([worst]), 50000000, 0.0001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 300, 'ftol': 1e-20,
                                    'gtol': 1e-14, 'maxfun': 1000})
            h_f = project_feasible(res.x, best_n)
            c_f = compute_c5(h_f, best_n)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        except Exception:
            pass

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
