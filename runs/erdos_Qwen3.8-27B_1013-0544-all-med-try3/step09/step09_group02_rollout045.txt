def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_k(h, n, k_full_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = k_full_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def projected_subgrad_step(h, n, active_idx, step_size):
    """Take a projected subgradient step using the average gradient of active constraints."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_idx:
        grad += grad_c_k(h, n, idx)
    grad /= len(active_idx)
    
    # Polyak-like step: scale by current max value
    c_vec = compute_c5_vec(h, n)
    cur_max = c_vec.max()
    
    # Line search
    best_h = h.copy()
    best_c = cur_max
    for t_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
        t = step_size * t_mult
        h_trial = h - t * grad
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial
    return best_h, best_c

def kkt_refine(h, n, deadline, max_iter=2000):
    """Refine using KKT conditions: at optimum, the weighted sum of active gradients
    should align with the normal cone of the box constraint."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 1e-5 * max_val)
        active = np.where(c_vec >= thresh)[0]
        
        if len(active) == 0:
            active = np.argsort(-c_vec)[:5]
        if len(active) > 30:
            active = active[np.argsort(-c_vec[active])[:30]]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            grad += grad_c_k(h, n, int(idx))
        grad /= len(active)
        
        # Check KKT: for free variables (0<h[i]<1), grad should be ~0
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if not np.any(free):
            break
        
        free_grad_mag = np.abs(grad[free]).max() if np.any(free) else 0
        
        # Take a small step in negative gradient direction for free variables only
        if free_grad_mag > 1e-14:
            step = min(0.01, 0.001 / free_grad_mag)
            h_trial = h.copy()
            h_trial[free] -= step * grad[free]
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial
                h = h_trial
            else:
                # Try smaller step
                for s in [step * 0.3, step * 0.1, step * 0.03]:
                    h_trial = h.copy()
                    h_trial[free] -= s * grad[free]
                    h_trial = project_feasible(h_trial, n)
                    c_trial = compute_c5(h_trial, n)
                    if c_trial < best_c - 1e-15:
                        best_c = c_trial
                        best_h = h_trial
                        h = h_trial
                        break
                else:
                    if free_grad_mag < 1e-10:
                        break
        else:
            break
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, n_passes=200):
    """Very fine pair search with gradient guidance."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > 20:
            active = active[np.argsort(-c_vec[active])[:20]]
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in active:
            grad += grad_c_k(h, n, int(idx))
        if len(active) > 0:
            grad /= len(active)
        
        # For each position with room to move, try transferring to best partner
        sorted_by_grad_neg = np.argsort(grad)[:30]
        sorted_by_grad_pos = np.argsort(-grad)[:30]
        
        for i in sorted_by_grad_neg:
            if time.time() > deadline:
                break
            if h[i] >= 0.999:
                continue
            # Try transferring mass to positions with high positive gradient
            for j in sorted_by_grad_pos:
                if time.time() > deadline:
                    break
                if i == j or h[j] <= 0.001:
                    continue
                steps = [0.001, 0.003, 0.01, 0.03]
                for step in steps:
                    if h[i] + step > 1.0 or h[j] - step < 0.0:
                        continue
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        h[i] -= step
                        h[j] += step
        
        if not improved:
            # Try finer steps
            for i in sorted_by_grad_neg[:15]:
                if time.time() > deadline:
                    break
                if h[i] >= 0.999:
                    continue
                for j in sorted_by_grad_pos[:15]:
                    if time.time() > deadline:
                        break
                    if i == j or h[j] <= 0.001:
                        continue
                    for step in [0.0003, 0.0001, 0.00003]:
                        if h[i] + step > 1.0 or h[j] - step < 0.0:
                            continue
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] -= step
                            h[j] += step
            if not improved:
                break
    
    return best_h, best_c

def binary_push(h, n, deadline):
    """Try to push h values toward 0 or 1 (near-binary solution)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    threshold = 0.1
    
    for it in range(5000):
        if time.time() > deadline:
            break
        # Find the softest position (furthest from 0 and 1)
        softness = np.minimum(h, 1.0 - h)
        mask = (softness > 0.02) & (softness < 0.98)
        if not np.any(mask):
            break
        
        idxs = np.where(mask)[0]
        # Pick the one that's closest to being pushed
        target = 0 if h[idxs[0]] < 0.5 else 1
        # Try pushing each soft position toward 0 or 1
        i = idxs[np.argmax(softness[mask])]
        push_dir = 1.0 if h[i] < 0.5 else -1.0
        
        for step in [0.01, 0.005, 0.002, 0.001]:
            h_trial = h.copy()
            h_trial[i] += push_dir * step
            if h_trial[i] < 0 or h_trial[i] > 1:
                continue
            # Compensate at another position
            if push_dir > 0:
                # Need to decrease somewhere else
                candidates = np.where(h > step + 0.05)[0]
                candidates = candidates[candidates != i]
            else:
                candidates = np.where(h < 1.0 - step - 0.05)[0]
                candidates = candidates[candidates != i]
            
            if len(candidates) == 0:
                continue
            # Pick candidate that minimizes impact on C5
            best_j = candidates[0]
            best_c_trial = 1.0
            for j in candidates[:10]:
                h_trial2 = h_trial.copy()
                h_trial2[j] -= push_dir * step
                c_t = compute_c5(h_trial2, n)
                if c_t < best_c_trial:
                    best_c_trial = c_t
                    best_j = j
                    best_h_trial = h_trial2.copy()
            
            if best_c_trial < best_c - 1e-15:
                h = best_h_trial
                best_c = best_c_trial
                best_h = best_h_trial.copy()
                break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS with increasing smoothness (80s)
    bounds = [(0.0, 1.0)] * n
    lbfgs_configs = [
        (1000, 50.0),
        (5000, 500.0),
        (20000, 5000.0),
        (100000, 50000.0),
        (500000, 200000.0),
    ]
    for alpha, mu in lbfgs_configs:
        remaining = deadline - 180 - time.time()
        if remaining < 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-22,
                                  'gtol': 1e-18, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    # Phase 2: KKT refinement (60s)
    if deadline - time.time() > 30:
        t2 = min(time.time() + 60, deadline - 100)
        h, best_c = kkt_refine(h, n, t2, max_iter=5000)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Phase 3: Another L-BFGS round with very high alpha (40s)
    if deadline - time.time() > 25:
        for alpha, mu in [(1000000, 500000.0), (5000000, 2000000.0)]:
            if time.time() > deadline - 80:
                break
            try:
                res = minimize(smooth_obj, h, jac=smooth_grad,
                             args=(n, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 2000, 'ftol': 1e-22,
                                      'gtol': 1e-18, 'maxfun': 20000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    
    # Phase 4: Fine pair search (80s)
    if deadline - time.time() > 30:
        t4 = min(time.time() + 80, deadline - 50)
        h, c4 = fine_pair_search(h, n, t4, n_passes=300)
        if c4 < best_c:
            best_c = c4
            best_h = h.copy()
    
    # Phase 5: KKT refinement again (40s)
    if deadline - time.time() > 20:
        t5 = min(time.time() + 40, deadline - 30)
        h, c5 = kkt_refine(h, n, t5, max_iter=10000)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
    
    # Phase 6: Extra fine pair search (30s)
    if deadline - time.time() > 15:
        h, c6 = fine_pair_search(h, n, deadline - 5, n_passes=200)
        if c6 < best_c:
            best_c = c6
            best_h = h.copy()
    
    # Phase 7: Binary push attempt (if time)
    if deadline - time.time() > 20:
        h7, c7 = binary_push(h, n, deadline - 5)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
