```python

import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = int(n / 2.0)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + math.log(float(np.sum(np.exp(sh - mx))))) / alpha
    else:
        val = m + math.log(float(np.sum(np.exp(sh)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
```

```python
import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = int(n / 2.0)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def local_search(h, n, deadline, seed=42, verbose=False):
    """Local search on binary vectors using random swaps."""
    h = to_binary(h, n)
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve_count = 0
    max_no_improve = 5000
    
    while time.time() < deadline:
        if no_improve_count > max_no_improve:
            break
        
        # Choose a random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            no_improve_count = 0
            # Update sets
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]  # Undo
            no_improve_count += 1
    
    return h, best_c

def simulated_annealing(h, n, deadline, seed=42):
    """Simulated annealing on binary vectors."""
    h = to_binary(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    current_c = best_c
    
    T0 = 0.01  # Initial temperature
    T_end = 1e-6  # Final temperature
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_iter = 1000000  # Will be limited by deadline
    total_time = deadline - time.time()
    # Estimate iterations possible
    # Each iteration: 1 swap + 1 correlation = ~20 μs
    # So we can do ~50,000 iterations per second
    
    start = time.time()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute temperature (geometric cooling)
        frac = min(1.0, (time.time() - start) / max(1, total_time))
        T = T0 * (T_end / T0) ** frac
        
        # Choose random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        delta = c_new - current_c
        
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]  # Undo
    
    return best_h, best_c
```
