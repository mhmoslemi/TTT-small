import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - nm1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for wi, idx in zip(w, active):
        k = idx - nm1
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_optimize(h, n, deadline, top_k=10, alphas=None, rounds=20, mu=0.001):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(rounds):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(100, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturb_and_reoptimize(h, n, deadline, rng, n_perturbs=200, perturb_scale=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i in range(n_perturbs):
        if time.time() > deadline - 30:
            break
        # Random perturbation
        scale = perturb_scale * (0.5 ** (i // 50))  # decrease scale over time
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Brief active set optimization from perturbed point
        c_vec = compute_c5_vec(h_pert, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 8:
            active = np.array(np.argsort(-c_vec[active])[:8])
        
        if len(active) > 0:
            maxit = max(50, int((deadline - time.time()) * 0.3))
            try:
                res = minimize(active_obj, h_pert, jac=active_grad,
                              args=(n, active, 100000, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': min(maxit, 200), 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxfun': min(maxit, 200)*3})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def fine_coordinate_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            # Compute gradient
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)

def project_feasible(h, n):

            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
