import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_k_val(h, idx, n):
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        return np.sum(h[k:n] * (1 - h[0:nv])) * (2.0 / n)
    else:
        kk = -k
        nv = n - kk
        return np.sum(h[0:nv] * (1 - h[kk:n])) * (2.0 / n)

def act_k_grad_slot(h, idx, n):
    k = idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n]) * dx
        g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv]) * dx
        g[0:nv] += (-h[kk:n]) * dx
    return g

def active_slsqp_solve(h_init, n, active_indices, deadline, mu=0.001):
    # Variables: x = [t, h_0, ..., h_{n-1}]
    # Minimize x[0] subject to x[0] - f_k(h) >= 0 for k in active_indices
    # and sum(h) = n/2, 0 <= h <= 1
    x0 = np.zeros(n + 1)
    x0[0] = compute_c5(h_init, n) + 0.01
    x0[1:] = h_init
    
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * n

    def objective(x):
        return x[0]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    constraints = []
    for idx in active_indices:
        def make_ineq(i):
            def fun(x):
                h = x[1:]
                return x[0] - act_k_val(h, i, n)
            def jac(x):
                h = x[1:]
                g = np.zeros(n + 1)
                g[0] = 1.0
                gk = act_k_grad_slot(h, i, n)
                g[1:] -= gk
                return g
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        constraints.append(make_ineq(idx))

    def eq_fun(x):
        return x[1:].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    if mu > 0:
        constraints.append({'type': 'ineq', 'fun': lambda x: 1.0 - mu * (x[1:] - 0.5)**2.sum(), 'jac': None})

    maxit = max(500, int((deadline - time.time()) * 2))
    try:
        res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        h_new = project_feasible(np.clip(res.x[1:], 0, 1), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h_init.copy(), n), compute_c5(h_init, n)

            h_new = project_feasible(res.x, n)

                best_h = h_new

            pass

def coordinate_descent(h, n, deadline, n_sweeps=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    for sweep in range(n_sweeps):

            break
        improved_any = True
        while improved_any and time.time() < deadline:

            for i in range(n):

                # 1D optimization over h[i] in [0,1]

                    h[i] = z
                    # Maintain sum constraint approximately by adjusting
                    c = compute_c5(h, n)
                    h[i] = z  # restore

def pair_search(h, n, deadline, radius=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        new_i = best_h[i] + sign * d
                        new_j = best_h[j] - sign * d
                        if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
                            best_h[i] = new_i
                            best_h[j] = new_j
                            best_h = project_feasible(best_h, n)
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] = new_i - sign * d
                                best_h[j] = new_j + sign * d
                                best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)
