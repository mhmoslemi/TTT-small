    sh = alpha * (c - m)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, h, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_topk(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_topk(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(500, int(remaining * 3))
        try:
            res = minimize(obj_smooth, best_h, jac=grad_smooth,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active(h, n, deadline, top_k=8, alphas=None, max_rounds=30):

    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100000, 500000, 2000000, 10000000, 50000000]
    no_improve = 0

        c_vec = compute_c5_vec(best_h, n)

        if len(active) > top_k:

                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)

        if not improved_this_round:

def grad_step_search(h, n, deadline, max_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in base_steps:
        stalled = 0
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            if len(active) > 0:
                grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            np.random.shuffle(pos_idx)
            np.random.shuffle(neg_idx)
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stalled = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stalled += 1
                if stalled > 200:
                    break
    return best_h, best_c

def fine_pairs(h, n, deadline, radius=40, deltas=None):

def upsample(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    h_fine = np.zeros(n_fine)
    if n_fine == n_coarse:
        return h_coarse.copy()
    for i in range(n_fine):
        src = i * n_coarse / n_fine
        i0 = int(src)
        i1 = min(i0 + 1, n_coarse - 1)
        frac = src - i0
        h_fine[i] = (1 - frac) * h_coarse[i0] + frac * h_coarse[i1]
    return project_feasible(h_fine, n_fine)

def make_binary_start(n, rng, alpha_param=None):
    h = np.zeros(n)
    n_ones = n // 2
    if alpha_param is not None:
        positions = set()
        for k in range(1, 4 * n):
            pos = int(k * alpha_param) % n
            positions.add(pos)
            if len(positions) >= n_ones:
                break
        for pos in positions:
            h[pos] = 1.0
        remaining = n / 2.0 - h.sum()
        if abs(remaining) > 1e-10:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] += remaining
    else:
        idx = rng.choice(n, size=n_ones, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)

def perturb(h, n, rng, strength=0.05):
    h_new = h.copy()
    mask = (h_new > 0.05) & (h_new < 0.95)
    h_new[mask] += rng.uniform(-strength, strength, size=mask.sum())
    return project_feasible(h_new, n)

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    def sq_fun(x):
        return x.sum() - n / 2.0
    def sq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac}]
    try:
        rem = max(5, int(deadline - time.time()))
        maxit_use = min(maxiter, max(100, rem * 5))
        res = minimize(obj_smooth, h, jac=grad_smooth,

                      bounds=bounds, constraints=cons,

        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

        else:
            best_h_f = upsample(best_h, n_fine)
            best_c_f = compute_c5(best_h_f, n_fine)

        p2_end = min(time.time() + 350, deadline - 250)
        h2, c2 = phase_lbfgs(best_h_f, n_fine, p2_end,
            alphas=[1000, 10000, 50000, 200000, 1000000, 5000000])
        if c2 < best_c_f:
            best_c_f = c2
            best_h_f = h2

        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_fine

    # Phase 3: Active-set at best resolution
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 180, deadline - 150)
        h3, c3 = phase_active(best_h, best_n, p3_end, top_k=10,
            alphas=[200000, 1000000, 5000000, 20000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient step search
    if time.time() < deadline - 120:
        p4_end = min(time.time() + 80, deadline - 80)
        h4, c4 = grad_step_search(best_h, best_n, p4_end, max_steps=30000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try multiple random starts at n=1000
    if time.time() < deadline - 150 and best_n == n_fine:
        for trial in range(3):
            if time.time() > deadline - 60:
                break
            h_rand = make_binary_start(n_fine, rng, alpha_param=None)
            c_rand = compute_c5(h_rand, n_fine)
            p5_end = min(time.time() + 40, deadline - 50)
            h5, c5 = phase_lbfgs(h_rand, n_fine, p5_end,
                alphas=[5000, 50000, 500000])
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Fine-tune
    if time.time() < deadline - 50:
        p6_end = min(time.time() + 40, deadline - 20)
        h6, c6 = fine_pairs(best_h, best_n, p6_end, radius=30,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Active-set again
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 25, deadline - 10)
        h7, c7 = phase_active(best_h, best_n, p7_end, top_k=6,
            alphas=[1000000, 5000000, 20000000], max_rounds=10)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: SLSQP polish
    if time.time() < deadline - 10:
        for alpha in [5000000, 20000000]:
            if time.time() > deadline - 5:
                break
            h8, c8 = slqp_polish(best_h, best_n, deadline - 2, alpha=alpha, maxiter=300)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Final very fine adjustments
    if time.time() < deadline - 3:
        h9, c9 = fine_pairs(best_h, best_n, deadline - 1, radius=20,
            deltas=[0.0001, 0.00005, 0.00002])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
