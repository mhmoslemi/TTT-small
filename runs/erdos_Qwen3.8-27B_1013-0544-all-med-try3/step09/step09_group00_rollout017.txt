from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n):
    c = compute_c5_vec(h, n)
    m = c.max()
    # Focus on active set
    tol = max(1e-8, 0.005 * m)
    active = c >= (m - tol)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if not active[mm]:
            continue
        k = mm - (n - 1)
        wm = dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad / active.sum()

def make_beatty(n, alpha):
    h = np.zeros(n)
    for k in range(n):
        h[k] = 1.0 if (k * alpha) % 1.0 < 0.5 else 0.0
    # Fix sum
    while h.sum() > n / 2.0:
        h[np.argmax(h)] = 0.0
    while h.sum() < n / 2.0:
        h[np.argmin(h)] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for k in range(n):
        # Thue-Morse: parity of number of 1-bits
        t = bin(k).count('1') % 2
        h[k] = 1.0 if t == 0 else 0.0
    while h.sum() > n / 2.0:
        idx = np.where(h == 1.0)[0]
        if len(idx) == 0: break
        h[idx[0]] = 0.0
    while h.sum() < n / 2.0:
        idx = np.where(h == 0.0)[0]
        if len(idx) == 0: break
        h[idx[0]] = 1.0
    return h

def binary_sa(h, n, deadline, rng, iters=200000):
    """Binary simulated annealing with swap moves"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Make binary
    h_cur = (h > 0.5).astype(float)
    # Fix to exactly n/2 ones
    n_ones = int(np.round(h_cur.sum()))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h_cur == 1.0)[0]
        rng.shuffle(ones)
        for i in range(n_ones - target_ones):
            h_cur[ones[i]] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_cur == 0.0)[0]
        rng.shuffle(zeros)
        for i in range(target_ones - n_ones):
            h_cur[zeros[i]] = 1.0
    
    c_cur = compute_c5(h_cur, n)
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()
    
    ones_list = np.where(h_cur == 1.0)[0].tolist()
    zeros_list = np.where(h_cur == 0.0)[0].tolist()
    
    # Simulated annealing schedule
    T_start = 0.5
    T_end = 0.01
    n_steps = iters
    cooldown = (np.log(T_end) - np.log(T_start)) / n_steps
    
    c_tol = best_c - 1e-12  # We need improvement of at least this
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T_start * np.exp(cooldown * step)
        
        i = rng.randint(0, len(ones_list))
        j = rng.randint(0, len(zeros_list))
        
        pi = ones_list[i]
        pj = zeros_list[j]
        
        # Swap
        h_cur[pi] = 0.0
        h_cur[pj] = 1.0
        c_new = compute_c5(h_cur, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_cur.copy()
        elif c_new <= c_cur + T:
            c_cur = c_new
            # Accept move
            ones_list[i] = pj
            zeros_list[j] = pi
        else:
            # Reject
            h_cur[pi] = 1.0
            h_cur[pj] = 0.0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return val

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        g += 2.0*mu*(x.sum() - n/2.0)
    return g

def active_obj(x, n, act, alpha, mu):
    c = compute_c5_vec(x, n)
    v = c[act]
    m = v.max()
    val = m + np.log(np.sum(np.exp(alpha*(v-m))))/alpha
    if mu > 0:
        val += mu*(x.sum()-n/2.0)**2
    return float(val)

def active_grad(x, n, act, alpha, mu):
    c = compute_c5_vec(x, n)
    v = c[act]
    m = v.max()
    w = np.exp(alpha*(v-m))
    w /= w.sum()
    dx = 2.0/n
    grad = np.zeros(n)
    for wi, idx in zip(w, act):
        k = idx - (n-1)
        wm = wi*dx
        if k >= 0:
            nv = n-k
            grad[0:nv] += wm*(1.0-x[k:n])
            grad[k:n] += wm*(-x[0:nv])
        else:
            kk = -k
            nv = n-kk
            grad[kk:n] += wm*(1.0-x[0:nv])
            grad[0:nv] += wm*(-x[kk:n])
    if mu > 0:
        grad += 2.0*mu*(x.sum()-n/2.0)
    return grad

def make_beatty(n, alpha):
    h = np.zeros(n)
    for k in range(n):
        h[k] = 1.0 if (k*alpha) % 1.0 < 0.5 else 0.0
    # Fix sum
    s = h.sum()
    if s > n/2.0:
        ones = np.where(h==1.0)[0]
        rng_fix = np.random.RandomState(0)
        rng_fix.shuffle(ones)
        for i in range(int(s - n/2.0)):
            h[ones[i]] = 0.0
    elif s < n/2.0:
        zeros = np.where(h==0.0)[0]
        rng_fix = np.random.RandomState(0)
        rng_fix.shuffle(zeros)
        for i in range(int(n/2.0 - s)):
            h[zeros[i]] = 1.0
    return h

def to_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    s = h_b.sum()
    target = n // 2
    if s > target:
        ones = list(np.where(h_b==1.0)[0])
        rng.shuffle(ones)
        for i in range(int(s - target)):
            h_b[ones[i]] = 0.0
    elif s < target:
        zeros = list(np.where(h_b==0.0)[0])
        rng.shuffle(zeros)
        for i in range(int(target - s)):
            h_b[zeros[i]] = 1.0
    return h_b

def binary_ls(h, n, deadline, rng, max_evals=1000000):
    """Binary local search with 1-swaps and occasional 2-swaps"""
    h_cur = to_binary(h, n, rng)
    c_cur = compute_c5(h_cur, n)
    best_c = c_cur
    best_h = h_cur.copy()
    
    ones = list(np.where(h_cur==1.0)[0])
    zeros = list(np.where(h_cur==0.0)[0])
    
    evals = 0
    stalls = 0
    while time.time() < deadline and evals < max_evals:
        improved = False
        # Try 1-swaps
        n_try = 30
        for _ in range(n_try):
            if time.time() > deadline or evals >= max_evals:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            pi, pj = ones[i], zeros[j]
            h_cur[pi] = 0.0
            h_cur[pj] = 1.0
            c_new = compute_c5(h_cur, n)
            evals += 1
            if c_new < c_cur - 1e-14:
                c_cur = c_new
                ones[i] = pj
                zeros[j] = pi
                improved = True
            else:
                h_cur[pi] = 1.0
                h_cur[pj] = 0.0
        # Occasional 2-swap
        if not improved and len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.3:
            i1 = rng.randint(len(ones))
            i2 = rng.randint(len(ones))
            if i1 == i2: i2 = (i2+1) % len(ones)
            j1 = rng.randint(len(zeros))
            j2 = rng.randint(len(zeros))
            if j1 == j2: j2 = (j2+1) % len(zeros)
            p1, p2 = ones[i1], ones[i2]
            q1, q2 = zeros[j1], zeros[j2]
            h_cur[p1] = 0.0; h_cur[p2] = 0.0
            h_cur[q1] = 1.0; h_cur[q2] = 1.0
            c_new = compute_c5(h_cur, n)
            evals += 1
            if c_new < c_cur - 1e-14:
                c_cur = c_new
                ones[i1] = q1; ones[i2] = q2
                zeros[j1] = p1; zeros[j2] = p2
                improved = True
            else:
                h_cur[p1] = 1.0; h_cur[p2] = 1.0
                h_cur[q1] = 0.0; h_cur[q2] = 0.0
        
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
        
        if not improved:
            stalls += 1
            if stalls > 200:
                break
        else:
            stalls = 0
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=30, deltas=None):
    """Directed pair search for continuous h"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if best_h[i] + d <= 1.0 + 1e-15 and best_h[j] - d >= -1e-15:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if best_h[i] - d >= -1e-15 and best_h[j] + d <= 1.0 + 1e-15:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return project_feasible(best_h, n), best_c

def run_lbfgs_full(h, n, deadline, alphas, mu=0.01):
    """Full smooth-max L-BFGS"""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)]*n
    for alpha in alphas:
        if time.time() > deadline:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(full_obj, best_h, jac=full_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-20,
                                  'gtol': 1e-14, 'maxfun': maxit*8,
                                  'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_h = h_new
                best_c = c_new
        except:
            pass
    return best_h, best_c

def run_lbfgs_active(h, n, deadline, top_k=3, alphas=None, rounds=10):
    """Active-set L-BFGS (top-k correlations)"""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)]*n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000, 20000000]
    
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thr = max_val - max(1e-8, 0.002*max_val)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(100, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                            args=(n, active, alpha, 0.001), method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxiter': maxit, 'ftol': 1e-20,
                                     'gtol': 1e-14, 'maxfun': maxit*6,
                                     'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_h = h_new
                    best_c = c_new
            except:
                pass
    return best_h, best_c

        # Compute average gradient over active set
        dx = 2.0/n
        grad = np.zeros(n)
        for mm in np.where(active)[0]:
            k = mm - (n-1)
            if k >= 0:
                nv = n-k
                grad[0:nv] += (1.0 - h_cur[k:n]) * dx
                grad[k:n] += (-h_cur[0:nv]) * dx
            else:
                kk = -k
                nv = n-kk
                grad[kk:n] += (1.0 - h_cur[0:nv]) * dx
                grad[0:nv] += (-h_cur[kk:n]) * dx
        grad /= active.sum()
        
        # Move
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h_cur = h_new
        else:
            # Don't update h_cur, just decay step
            pass
        
        step *= decay
        if step < 1e-8:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def active_obj(x, n, act, alpha, mu):
    c = compute_c5_vec(x, n)
    v = c[act]
    m = v.max()
    val = m + np.log(np.sum(np.exp(alpha * (v - m)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, act, alpha, mu):
    c = compute_c5_vec(x, n)
    v = c[act]
    m = v.max()
    w = np.exp(alpha * (v - m))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, act):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_beatty(n, alpha):
    h = np.zeros(n)
    for k in range(n):
        h[k] = 1.0 if (k * alpha) % 1.0 < 0.5 else 0.0
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = list(np.where(h == 1.0)[0])
        np.random.RandomState(0).shuffle(ones)
        for i in range(s - target):
            h[ones[i]] = 0.0
    elif s < target:
        zeros = list(np.where(h == 0.0)[0])
        np.random.RandomState(0).shuffle(zeros)
        for i in range(target - s):
            h[zeros[i]] = 1.0
    return h

def to_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    s = int(h_b.sum())
    target = n // 2
    if s > target:
        ones = list(np.where(h_b == 1.0)[0])
        rng.shuffle(ones)
        for i in range(s - target):
            h_b[ones[i]] = 0.0
    elif s < target:
        zeros = list(np.where(h_b == 0.0)[0])
        rng.shuffle(zeros)
        for i in range(target - s):
            h_b[zeros[i]] = 1.0
    return h_b

def binary_ls(h, n, deadline, rng, max_evals=1500000):
    h_cur = to_binary(h, n, rng)
    c_cur = compute_c5(h_cur, n)
    best_c = c_cur
    best_h = h_cur.copy()
    ones = list(np.where(h_cur == 1.0)[0])
    zeros = list(np.where(h_cur == 0.0)[0])
    evals = 0
    stalls = 0
    while time.time() < deadline and evals < max_evals:
        improved = False
        n_try = 40
        for _ in range(n_try):
            if time.time() > deadline or evals >= max_evals:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            pi, pj = ones[i], zeros[j]
            h_cur[pi] = 0.0
            h_cur[pj] = 1.0
            c_new = compute_c5(h_cur, n)
            evals += 1
            if c_new < c_cur - 1e-14:
                c_cur = c_new
                ones[i] = pj
                zeros[j] = pi
                improved = True
            else:
                h_cur[pi] = 1.0
                h_cur[pj] = 0.0
        if not improved and len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.4:
            i1 = rng.randint(len(ones))
            i2 = rng.randint(len(ones))
            if i1 == i2:
                i2 = (i2 + 1) % len(ones)
            j1 = rng.randint(len(zeros))
            j2 = rng.randint(len(zeros))
            if j1 == j2:
                j2 = (j2 + 1) % len(zeros)
            p1, p2 = ones[i1], ones[i2]
            q1, q2 = zeros[j1], zeros[j2]
            h_cur[p1] = 0.0
            h_cur[p2] = 0.0
            h_cur[q1] = 1.0
            h_cur[q2] = 1.0
            c_new = compute_c5(h_cur, n)
            evals += 1
            if c_new < c_cur - 1e-14:
                c_cur = c_new
                ones[i1] = q1
                ones[i2] = q2
                zeros[j1] = p1
                zeros[j2] = p2
                improved = True
            else:
                h_cur[p1] = 1.0
                h_cur[p2] = 1.0
                h_cur[q1] = 0.0
                h_cur[q2] = 0.0
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
        if not improved:
            stalls += 1
            if stalls > 300:
                break
        else:
            stalls = 0
    return best_h, best_c

def pair_search(h, n, deadline, radius=40, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-14 and best_h[j] - d >= -1e-14:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= -1e-14 and best_h[j] + d <= 1.0 + 1e-14:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return project_feasible(best_h, n), best_c

def subgrad_descent(h, n, deadline, step_init=0.005, n_max=50000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h_cur = best_h.copy()
    step = step_init
    dx = 2.0 / n
    for it in range(n_max):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h_cur, n)
        m = c.max()
        tol = max(1e-7, 0.005 * m)
        active = c >= (m - tol)
        n_act = active.sum()
        if n_act < 1:
            break
        grad = np.zeros(n)
        idxs = np.where(active)[0]
        for mm in idxs:
            k = mm - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h_cur[k:n]) * dx
                grad[k:n] += (-h_cur[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h_cur[0:nv]) * dx
                grad[0:nv] += (-h_cur[kk:n]) * dx
        grad /= n_act
        h_new = project_feasible(h_cur - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h_cur = h_new
        step *= 0.9995
        if step < 1e-9:
            break
    return best_h, best_c

def run_lbfgs_full(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(full_obj, best_h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c

def run_lbfgs_active(h, n, deadline, top_k=3, rounds=10, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thr = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(100, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_h = h_new
                    best_c = c_new
            except Exception:
                pass
    return best_h, best_c
