                            best_h = h_interp
                            break
            else:
                break
        except Exception:
            break
        
        # Convergence check
        if abs(cur_c - best_c) < 1e-10:
            break
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_radius=50, step_sizes=None):
    """Fine coordinate descent: try small shifts between pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if step_sizes is None:
        step_sizes = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = np.clip(best_h, 0, 1)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = np.clip(best_h, 0, 1)
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = np.clip(best_h, 0, 1)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c

Wait, the coordinate descent with radius 50 and n=500 means for each i, we check ~100 j's, so ~50000 pairs per sweep. Each evaluation is O(n) for the correlation. So one sweep is 50000 * 500 = 25M operations. With multiple step sizes and sweeps, this could be slow.

Let me optimize the coordinate descent. Instead of full correlation each time, I can compute only the affected c_k values.

When we change h[i] by +d and h[j] by -d, which c_k values change?
- c_k involves h[a] * (1-h[b]) where b = a + k (or a = b + k)
- Changing h[i] affects all c_k where i is involved: k such that i appears as a or b
  - As a: k = b - a for various b → all k from -(n-1) to n-1 where i is in range
  - This is basically all 2n-1 values!

So we can't easily avoid the full recomputation. But np.correlate is implemented in C and should be fast for n=500.

Actually, np.correlate for n=500 arrays should be very fast (microseconds). Let me estimate: 50000 pairs * 50 microseconds per correlation = 2.5 seconds per sweep. With 7 step sizes and a few sweeps, that's ~50 seconds. Acceptable.

But for n=2000, it would be 4x slower per correlation and 4x more pairs (radius 50 → 200000 pairs). That's 200000 * 200us = 40 seconds per sweep. Too slow.

For high resolution, I'll skip the coordinate descent or reduce the radius.

Phase 1 (200s): Start with parent (n=500), L-BFGS with smooth max
Phase 2 (150s): Bilinear LP cutting plane at n=500
Phase 3 (200s): Upsample to n=2000, L-BFGS

Phase 6 (100s): Coordinate descent / pair search
Phase 7 (50s): Final polish

Actually, let me reconsider. The bilinear LP at n=2000 would have 2001 variables and 3999 constraints. The LP solve should still be fast (maybe 10-50ms per solve). With 200 iterations, that's 10-10 seconds total. Very feasible.

But computing the gradients (A_ub matrix) for n=2000: that's 3999 gradients, each of length 2000. Computing them naively is O(n^2) = 4M. Fine.

Let me also think about whether the LP formulation is correct.

The true objective is: max_k f_k(h) where f_k(h) = sum_i h[i] * (1-h[i+k]) * dx (for valid i, non-periodic)

At the optimum h*, the max is achieved at some set of k values. For the LP cutting plane:
- At iteration t, we have current point h^t
- We linearize: f_k(h) ≈ f_k(h^t) + ∇f_k(h^t)^T (h - h^t)
- We solve: min_h max_k [f_k(h^t) + ∇f_k(h^t)^T (h - h^t)] s.t. sum(h) = n/2, 0≤h≤1
- This LP gives h^{t+1}

The key question: does this converge? For convex f_k, yes (this is the classical cutting plane for convex minimization). But f_k is bilinear (non-convex), so the linearization might overestimate or underestimate.

For bilinear f(x,y) = x*y, at point (x0, y0): linearization is x0*y + y0*x - x0*y0. The true value could be above or below this.

So the LP solution might be optimistic (underestimate the true max). We need to check the true objective and only accept improvements.

If the LP solution doesn't improve the true objective, we can:
1. Try interpolation between h^t and h^{t+1}
2. Try a different step size
3. Fall back to gradient-based methods

This is essentially a trust-region / line-search approach applied to the LP solution.

Let me also add a "random restart" component: periodically perturb the current best and see if we get a better solution.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c_k(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n]) * dx
        g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv]) * dx
        g[0:nv] += (-h[kk:n]) * dx
    return g

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def bilinear_lp_refine(h, n, deadline, max_iters=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    prev_c = best_c
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        cur_c = c_vec.max()
        
        m = 2 * n - 1
        A_ub = np.zeros((m, n + 1))
        A_ub[:, 0] = -1.0
        b_ub = np.zeros(m)
        
        for j in range(m):
            k = j - (n - 1)
            gj = grad_c_k(best_h, n, k)
            A_ub[j, 1:] = gj
            b_ub[j] = np.dot(gj, best_h) - c_vec[j]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([n / 2.0])
        
        c_obj = np.zeros(n + 1)
        c_obj[0] = 1.0
        
        bounds = [(-1.0, 1.0)] + [(0.0, 1.0)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                break
            h_new = res.x[1:]
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            
            improved = False
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                improved = True
            else:
                for alpha_ls in [0.5, 0.75, 0.25, 0.6, 0.4, 0.9, 0.1]:
                    h_interp = alpha_ls * h_new + (1.0 - alpha_ls) * best_h
                    h_interp = project_feasible(h_interp, n)
                    c_interp = compute_c5(h_interp, n)
                    if c_interp < best_c - 1e-14:
                        best_c = c_interp
                        best_h = h_interp.copy()
                        improved = True
                        break
            
            if abs(prev_c - best_c) < 1e-12 and it > 10:
                break
            prev_c = best_c
        except Exception:
            break
        
        if it % 50 == 49 and not improved:
            break
    
    return best_h, best_c

def lbfgs_refine(h, n, deadline, alphas=None, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 500000]
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 3))
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def active_set_lp(h, n, deadline, top_k=10, max_rounds=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])[:top_k]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        m_a = len(active)
        A_ub = np.zeros((m_a, n + 1))
        A_ub[:, 0] = -1.0
        b_ub = np.zeros(m_a)
        
        for ii, j in enumerate(active):
            k = j - (n - 1)
            gj = grad_c_k(best_h, n, k)
            A_ub[ii, 1:] = gj
            b_ub[ii] = np.dot(gj, best_h) - c_vec[j]
        
        # Also add all constraints that are near-active
        all_idx = set(active.tolist())
        near_mask = c_vec >= max_val - 0.01 * max_val
        for j in range(len(c_vec)):
            if near_mask[j] and j not in all_idx:
                A_ub = np.vstack([A_ub, np.zeros(n+1)])
                A_ub[-1, 0] = -1.0
                k = j - (n - 1)
                gj = grad_c_k(best_h, n, k)
                A_ub[-1, 1:] = gj
                b_ub = np.append(b_ub, np.dot(gj, best_h) - c_vec[j])
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([n / 2.0])
        c_obj = np.zeros(n + 1)
        c_obj[0] = 1.0
        bounds = [(-1.0, 1.0)] + [(0.0, 1.0)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                continue
            h_new = project_feasible(res.x[1:], n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
            else:
                for a in [0.5, 0.7, 0.3]:
                    hi = a * h_new + (1-a) * best_h
                    hi = project_feasible(hi, n)
                    ci = compute_c5(hi, n)
                    if ci < best_c - 1e-14:
                        best_c = ci
                        best_h = hi.copy()
                        break
        except Exception:
            continue
    
    return best_h, best_c

def coord_descent(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0, 1)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = np.clip(h, 0, 1)
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0, 1)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = np.clip(h, 0, 1)
    
    return best_h, best_c

def upsample_to(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_perturbed_start(h, n, rng, strength=0.05):
    h_new = h.copy()
    noise = rng.randn(n) * strength
    h_new += noise
    h_new = project_feasible(h_new, n)
    return h_new
