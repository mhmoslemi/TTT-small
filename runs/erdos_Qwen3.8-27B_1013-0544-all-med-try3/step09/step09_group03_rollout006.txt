def coord_descent(h, n, max_rounds, deadline):
    """Cycle through coordinates, try small perturbations, project, keep improvement."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(777)
    deltas = [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.01, -0.01, 0.003, -0.003]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.argsort(rng.rand(n))
        for j in order:
            if time.time() > deadline:
                break
            h_save = best_h[j]
            for d in deltas:
                nv = best_h[j] + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                best_h[j] = nv
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:

                    improved = True
                    break

                    # Re-project after restoring
                    best_h = project_feasible(best_h, n)
        if not improved and rnd > 20:
            break
    return best_h, best_c

        t7 = min(time.time() + 80, deadline - 150)
        best_h, best_c = perturbation_search(best_h, n_base, t7,
            n_trials=8, mags=[0.02, 0.04, 0.01, 0.06, 0.03, 0.08, 0.015, 0.05],
            alpha=50000, mu=10000.0)
    
    # Phase 8: Upsample to n=1000 (150s)
    n_hi = 1000
    if time.time() < deadline - 200:
        h_hi = interpolate_to(best_h, n_base, n_hi)
        t8 = min(time.time() + 80, deadline - 100)
        if t8 > time.time() + 10:
            h_hi, c_hi = run_lbfgs(h_hi, n_hi, t8,
                configs=[(2000, 500.0), (5000, 2000.0), (15000, 5000.0),
                         (30000, 10000.0), (50000, 20000.0)])
            # Coordinate descent at high res
            h_hi, c_hi = coord_descent(h_hi, n_hi, max_rounds=100, deadline=deadline - 60)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi
                n_base = n_hi
    
    # Phase 9: Final SLSQP polish (50s)
    if time.time() < deadline - 40:
        h9, c9 = slqp_opt(best_h, len(best_h), alpha=100000, maxiter=2000,
                         deadline=deadline - 10)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final coordinate descent (20s)
    if time.time() < deadline - 5:
        best_h, best_c = coord_descent(best_h, len(best_h), max_rounds=50,
                                       deadline=deadline - 2)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
