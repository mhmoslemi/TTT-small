                    best_c = c
                    best_h = h_proj.copy()
                    h = h_proj.copy()
                    found_improve = True
                    break
                else:
                    h = best_h.copy()
            if found_improve:
                break
        if not found_improve and round_num > 50:
            break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_optimize(h, n, deadline, n_trials=6, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    if mags is None:
        mags = [0.03, 0.05, 0.02, 0.07, 0.04, 0.06]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(lse_obj, h_p, jac=lse_grad,
                          args=(n, 5000, 100.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 25
    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h)
    except Exception:
        n_base = 500
        h = np.full(n_base, 0.5)

    h = project_feasible(h, n_base)
    best_h = h.copy()
    best_c = compute_c5(h, n_base)

    # Phase 1: L-BFGS warmup at base (70s)
    t1 = min(t0 + 70, deadline - 750)
    if time.time() < t1:
        h, c = run_lbfgs(h, n_base, t1,
            alphas=[100, 500, 2000, 8000, 20000], maxiter_per=6000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 2: SLSQP polish at base (50s)
    t2 = min(time.time() + 50, deadline - 650)
    if time.time() < t2:
        h, c = slqp_optim(best_h, n_base, t2, alpha=50000, maxiter=3000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Binary nudge + L-BFGS restart (60s)
    t3 = min(time.time() + 60, deadline - 550)
    if time.time() < t3:
        h_nudged = binary_nudge(best_h, n_base)
        h, c = run_lbfgs(h_nudged, n_base, t3,
            alphas=[2000, 10000, 50000], maxiter_per=4000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 4: Perturbation search at base (60s)
    t4 = min(time.time() + 60, deadline - 450)
    if time.time() < t4:
        best_h, best_c = perturb_optimize(best_h, n_base, t4,
            n_trials=5, mags=[0.03, 0.05, 0.02, 0.06, 0.04])

    # Phase 5: SLSQP after perturbation (40s)
    t5 = min(time.time() + 40, deadline - 380)
    if time.time() < t5:
        h, c = slqp_optim(best_h, n_base, t5, alpha=100000, maxiter=2000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 6: Upsample to n=1000, thorough optimization (200s)
    n1 = 1000
    if time.time() < deadline - 300:
        h1000 = interpolate_to(best_h, n_base, n1)
        t6 = min(time.time() + 120, deadline - 160)
        if time.time() < t6:
            h1000, c1000 = run_lbfgs(h1000, n1, t6,
                alphas=[200, 1000, 5000, 20000, 80000], maxiter_per=6000)
            if c1000 < best_c:
                best_c, best_h = c1000, h1000

        t6b = min(time.time() + 80, deadline - 80)
        if time.time() < t6b:
            h, c = slqp_optim(best_h, len(best_h), t6b, alpha=200000, maxiter=5000)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 7: Coord descent at current resolution (50s)
    if time.time() < deadline - 80:
        n_cd = len(best_h)
        t7 = min(time.time() + 50, deadline - 40)
        if time.time() < t7:
            best_h, best_c = coord_descent(best_h, n_cd, t7, n_sweeps=200)

    # Phase 8: Upsample to n=1500 (100s)
    n2 = 1500
    if time.time() < deadline - 120:
        h1500 = interpolate_to(best_h, len(best_h), n2)
        t8 = min(time.time() + 80, deadline - 50)
        if time.time() < t8:
            h1500, c1500 = run_lbfgs(h1500, n2, t8,
                alphas=[500, 3000, 15000, 60000], maxiter_per=5000)
            if c1500 < best_c:
                best_c, best_h = c1500, h1500

        t8b = min(time.time() + 40, deadline - 15)
        if time.time() < t8b:
            h, c = slqp_optim(best_h, len(best_h), t8b, alpha=400000, maxiter=2000)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 9: Final targeted refinement
    if time.time() < deadline - 10:
        n_fin = len(best_h)
        t9 = deadline - 10
        best_h, best_c = targeted_deltas(best_h, n_fin, t9)

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
