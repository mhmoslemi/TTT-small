def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Smooth max objective and gradient for L-BFGS
def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-14:
            continue
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, max_iters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        if max_iters is not None:
            maxit = max_iters
        else:
            maxit = max(500, int((deadline - time.time()) * 2))
        
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-20,
                                  'gtol': 1e-16, 'maxfun': maxit*8,
                                  'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def swap_delta_vec(h, n, i, j):
    """Compute dx*delta_c for swapping h[i] and h[j]. Vectorized O(n)."""
    dx = 2.0 / n
    da = h[j] - h[i]
    if abs(da) < 1e-14:
        return np.zeros(2*n-1)
    db = -da
    
    delta_c = np.zeros(2*n-1)
    
    # Position i, change da:
    # Term 1: da*(1-h[x]) where x = i-d, valid 0≤x<n
    #   d = i-x, k = d+(n-1) = i-x+n-1
    #   x from 0 to n-1: k from i+n-1 to i
    #   So k ranges [i, i+n-1] clipped to [0, 2n-2]
    lo = max(0, i)
    hi = min(2*n-2, i+n-1)
    if lo <= hi:
        # x = i - (k-(n-1)) = n-1+i-k
        # When k=lo: x = n-1+i-lo; when k=hi: x = n-1+i-hi
        x_start = n-1+i-hi
        x_end = n-1+i-lo
        delta_c[lo:hi+1] += da * (1.0 - h[x_start:x_end+1])
    
    # Term 2: -da*h[x] where x = i+d, valid 0≤x<n
    #   d = x-i, k = x-i+n-1
    #   x from 0 to n-1: k from n-1-i to 2n-2-i
    lo = max(0, n-1-i)
    hi = min(2*n-2, 2*n-2-i)
    if lo <= hi:
        # x = k-(n-1)+i
        x_start = lo-(n-1)+i
        x_end = hi-(n-1)+i
        delta_c[lo:hi+1] -= da * h[x_start:x_end+1]
    
    # Position j, change db:
    lo = max(0, j)
    hi = min(2*n-2, j+n-1)
    if lo <= hi:
        x_start = n-1+j-hi
        x_end = n-1+j-lo
        delta_c[lo:hi+1] += db * (1.0 - h[x_start:x_end+1])
    
    lo = max(0, n-1-j)
    hi = min(2*n-2, 2*n-2-j)
    if lo <= hi:
        x_start = lo-(n-1)+j
        x_end = hi-(n-1)+j
        delta_c[lo:hi+1] -= db * h[x_start:x_end+1]
    
    return dx * delta_c

def local_search(h, n, deadline, n_trials=2000000, seed=42):
    h = h.copy()
    c = compute_c5_vec(h, n)
    c_max = float(c.max())
    best_c = c_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    for trial in range(n_trials):
        if time.time() > deadline - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        a, b = h[i], h[j]
        if abs(a-b) < 1e-10:
            continue
        
        delta_c = swap_delta_vec(h, n, i, j)
        new_c = c + delta_c
        new_c_max = float(new_c.max())
        
        if new_c_max < c_max - 1e-15:
            h[i], h[j] = b, a
            c = new_c
            c_max = new_c_max
            if c_max < best_c:
                best_c = c_max
                best_h = h.copy()
    
    return best_h, best_c
