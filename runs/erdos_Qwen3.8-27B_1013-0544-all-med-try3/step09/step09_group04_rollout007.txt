def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_softmax_obj(x, n, k, alpha, mu):
    c = compute_c5_vec(x, n)
    # Get top-k values
    topk_idx = np.argpartition(c, -k)[-k:]
    vals = c[topk_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    topk_idx = np.argpartition(c, -k)[-k:]
    vals = c[topk_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, topk_idx):
        kk = idx - (n - 1)
        wm = wi * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softobj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    if active is not None:
        vals = c[active]
    else:
        vals = c
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softgrad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    if active is not None:
        vals = c[active]
    else:
        vals = c
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    if active is not None:
        for wi, idx in zip(w, active):
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    else:
        for mm in range(len(vals)):
            if w[mm] < 1e-15:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.005, maxiters=None, active_list=None):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    n_alphas = len(alphas)
    for i in range(n_alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        alpha = alphas[i]
        active = None
        if active_list is not None and i < len(active_list):
            active = active_list[i]
        if maxiters is not None and i < len(maxiters):
            mxi = maxiters[i]
        else:
            mxi = max(100, int(remaining * 1.5))
        try:
            res = minimize(softobj, best_h, jac=softgrad,
                           args=(n, active, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': mxi, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': mxi * 10,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_loop(h, n, deadline, max_rounds=30, top_k=3, margin=0.005):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - margin * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        alphas = [200000, 1000000, 5000000, 20000000]
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 10:
                break
            mxi = max(80, int(rem2 * 0.8))
            try:
                res = minimize(softobj, best_h, jac=softgrad,
                               args=(n, active, alpha, 0.003), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': mxi, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': mxi * 8,
                                        'maxls': 35, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb(h, n, rng, mode='gauss', scale=0.01):
    h_new = h.copy()
    if mode == 'gauss':
        noise = rng.normal(0, scale, n)
        h_new += noise
    elif mode == 'fft':
        freq = rng.integers(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        noise = scale * np.sin(2 * np.pi * freq * x / n + phase)
        h_new += noise
    elif mode == 'swap':
        i, j = rng.integers(0, n, 2)
        if i != j:
            h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_feasible(h_new, n)

def coord_descent_step(h, n, i, c_vec):
    """Find optimal h[i] given all other h fixed. Uses the fact that
    c_k is linear in h[i] for fixed other variables."""
    old_val = h[i]
    dx = 2.0 / n
    # Compute effect of h[i] on each c[m]
    # c[m] includes: h[i]*(1-h[i+k]) and h[i-k]*(1-h[i]) terms
    # where k = m - (n-1)
    slopes = np.zeros(2 * n - 1)
    intercepts = c_vec - h[i] * 0  # will compute properly
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            # h[i] appears as h[i]*(1-h[i+k]) if i+k < n
            # h[i] appears as h[i-k]*(1-h[i]) if i-k >= 0
            slope = 0.0
            if i + k < n:
                slope += (1.0 - h[i + k])
            if i - k >= 0:
                slope += (-h[i - k])
            slopes[m] = slope * dx
        else:
            kk = -k
            slope = 0.0
            if i - kk >= 0:
                slope += (1.0 - h[i - kk])
            if i + kk < n:
                slope += (-h[i + kk])
            slopes[m] = slope * dx
    # c[m] = slopes[m]*h[i] + (c[m] - slopes[m]*h[i])
    intercepts = c_vec - old_val * slopes
    # Minimize max_m (slopes[m]*t + intercepts[m]) over t in [0,1]
    # This is a 1D convex piecewise linear problem
    # Evaluate at key points
    candidates = [0.0, 1.0, old_val]
    # Find crossing of top-2 at t=0 and t=1
    vals0 = slopes * 0 + intercepts
    vals1 = slopes * 1 + intercepts
    idx0 = np.argmax(vals0)
    idx1 = np.argmax(vals1)
    if idx0 != idx1:
        a0, a1 = slopes[idx0], slopes[idx1]
        b0, b1 = intercepts[idx0], intercepts[idx1]
        if abs(a0 - a1) > 1e-15:
            t_cross = (b1 - b0) / (a0 - a1)
            if 0.0 < t_cross < 1.0:
                candidates.append(t_cross)
    # Also check crossing of top-3
    idxs0 = np.argsort(-vals0)[:3]
    idxs1 = np.argsort(-vals1)[:3]
    for ia in idxs0:
        for ib in idxs1:
            if ia != ib:
                a0, a1 = slopes[ia], slopes[ib]
                b0, b1 = intercepts[ia], intercepts[ib]
                if abs(a0 - a1) > 1e-15:
                    tc = (b1 - b0) / (a0 - a1)
                    if 0.0 < tc < 1.0:
                        candidates.append(tc)
    # Evaluate and pick best
    best_t = old_val
    best_v = np.inf
    for t in candidates:
        t = max(0.0, min(1.0, t))
        v = np.max(slopes * t + intercepts)
        if v < best_v:
            best_v = v
            best_t = t
    return best_t, best_v

def coord_descent(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        improved = False
        for i in range(n):
            t_new, v_new = coord_descent_step(best_h, n, i, c_vec)
            if abs(t_new - best_h[i]) > 1e-8:
                best_h[i] = t_new
                # Recompute c_vec (expensive but necessary)
                if i % 10 == 0:
                    c_vec = compute_c5_vec(best_h, n)
                improved = True
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
        if not improved:
            break
    return best_h, best_c

        h_mid = upsample(best_h, best_n, n_mid)
        p2_end = min(time.time() + 180, deadline - 400)
        h2, c2 = lbfgs_phase(h_mid, n_mid, p2_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxiters=[500, 700, 1000, 1200, 1500, 1800, 2000, 2500])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_mid

    # Phase 3: Active set loop at current resolution (200s)
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 200, deadline - 250)
        h3, c3 = active_set_loop(best_h, best_n, p3_end,
                                  max_rounds=25, top_k=3, margin=0.003)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Perturbation restarts (150s)
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 150, deadline - 150)
        modes = ['gauss', 'fft', 'gauss', 'fft', 'swap', 'gauss']
        scales = [0.01, 0.015, 0.02, 0.025, 0.0, 0.03]
        for mi in range(len(modes)):
            if time.time() > p4_end:
                break
            rem = p4_end - time.time()
            if rem < 25:
                break
            h_pert = perturb(best_h, best_n, rng, mode=modes[mi], scale=scales[mi])
            # Quick optimize
            sub_end = time.time() + min(25, rem - 5)
            h_p2, c_p2 = lbfgs_phase(h_pert, best_n, sub_end,
                alphas=[5000, 50000, 500000], mu=0.01,
                maxiters=[300, 500, 600])
            if c_p2 < best_c:
                best_c = c_p2
                best_h = h_p2

    # Phase 5: Upsample to n=2000 (250s)

            h5b, c5b = active_set_loop(h5, n_high, p5b_end,

                best_c = c5b
                best_h = h5b
                best_n = n_high
            elif c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_high

    if time.time() < deadline - 60:
        p6_end = min(time.time() + 100, deadline - 40)
        h6, c6 = active_set_loop(best_h, best_n, p6_end,
                                  max_rounds=20, top_k=3, margin=0.002)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Top-2 constraint dive with very high alpha (50s)
    if time.time() < deadline - 30:
        c_vec = compute_c5_vec(best_h, best_n)
        top2 = np.argsort(-c_vec)[:2]
        p7_end = min(time.time() + 50, deadline - 20)
        sub_deadline = p7_end
        for alpha in [5000000, 20000000, 100000000]:
            if time.time() > sub_deadline:
                break
            try:
                res = minimize(softobj, best_h, jac=softgrad,
                               args=(best_n, top2, alpha, 0.002), method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * best_n,
                               options={'maxiter': 300, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': 2500})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    c_vec = compute_c5_vec(best_h, best_n)
                    top2 = np.argsort(-c_vec)[:2]
            except Exception:
                pass

    # Phase 8: Coordinate descent (30s)
    if time.time() < deadline - 5:

        h8, c8 = coord_descent(best_h, best_n, p8_end, max_sweeps=100)

    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
