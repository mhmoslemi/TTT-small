        hp = h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        c_try = compute_c5(hp, n)
        # Quick subgradient
        alloc = min(40, (deadline - time.time()) * 0.3)
        if alloc < 10:
            break
        h_opt, c_opt = subgradient_method(hp, n, time.time() + alloc, max_iter=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP polishing with various alpha and active sets
    if time.time() < deadline - 60:
        slsqp_alloc = min(80, (deadline - t0) * 0.15)
        slsqp_deadline = t0 + slsqp_alloc
        for alpha in [50000, 200000]:
            if time.time() > slsqp_deadline:
                break
            for top_m in [3, 5, 8, None]:
                if time.time() > slsqp_deadline:
                    break
                h_new = slsqp_polish(h, n, alpha, maxiter=3000, top_m=top_m)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new

    # Phase 2: Subgradient descent
    if time.time() < deadline - 60:
        sd_alloc = min(120, (deadline - time.time()) * 0.3)
        h_new, c_new = subgradient_method(h, n, time.time() + sd_alloc, max_iter=100000)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new

    # Phase 3: Local search with bisection
    if time.time() < deadline - 30:
        ls_alloc = min(200, (deadline - time.time()) * 0.4)
        h_new, c_new = local_search_bisection(h, n, time.time() + ls_alloc, max_rounds=300)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new

    # Phase 4: SLSQP again (sharper)
    if time.time() < deadline - 20:
        slsqp_deadline2 = time.time() + min(60, (deadline - time.time()) * 0.3)
        for alpha in [200000, 500000, 1000000]:
            if time.time() > slsqp_deadline2:
                break
            for top_m in [2, 3, 5]:
                if time.time() > slsqp_deadline2:
                    break
                h_new = slsqp_polish(h, n, alpha, maxiter=5000, top_m=top_m)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new

    # Phase 5: Another round of local search
    if time.time() < deadline - 15:
        ls_alloc2 = min(80, (deadline - time.time()) * 0.5)
        h_new, c_new = local_search_bisection(h, n, time.time() + ls_alloc2, max_rounds=200)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new

    # Phase 6: Multi-start perturbation (if time remains)
    if time.time() < deadline - 40:
        h_new, c_new = multi_start_perturb(h, n, deadline - 10, rng, n_starts=3)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 8:
        h_new = slsqp_polish(h, n, 500000, maxiter=3000, top_m=3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 8: Final local search
    if time.time() < deadline - 3:
        h_new, c_new = local_search_bisection(h, n, deadline - 1, max_rounds=100)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
