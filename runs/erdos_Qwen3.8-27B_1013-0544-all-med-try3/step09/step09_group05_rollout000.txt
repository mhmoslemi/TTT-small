                break
            to = min(15, t1_end - time.time() - 1)
            if to > 3:
                h_new = lbfgs_top(best_h, n_init, top_idx, alpha, 10.0, 2000, to)
                c_new = compute_c5(h_new, n_init)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                top_idx = find_active_shifts(best_h, n_init, top_k=10)

    # Phase 2: Upsample to n=1000
    n_target = 1000
    h_1000 = upsample(best_h, n_init, n_target)
    c_1000 = compute_c5(h_1000, n_target)
    if c_1000 < best_c:
        best_c = c_1000
        best_h = h_1000
    else:
        best_h = h_1000
        best_c = c_1000

    # Phase 3: Main L-BFGS optimization at n=1000 (400s)
    t3_start = time.time()
    t3_end = t3_start + 400
    n = n_target
    if t3_end > deadline - 350:
        t3_end = deadline - 350

    for i, (alpha, mu) in enumerate([(1000, 50.0), (5000, 100.0), (20000, 500.0),
                                       (100000, 2000.0), (500000, 10000.0),
                                       (2000000, 50000.0), (5000000, 200000.0)]):
        if time.time() > t3_end - 5:
            break
        top_idx = find_active_shifts(best_h, n, top_k=8, frac=0.003)
        to = min(60, t3_end - time.time() - 3)
        if to < 5:
            break
        h_new = lbfgs_top(best_h, n, top_idx, alpha, mu, 5000, to)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Full optimization at moderate alpha (150s)
    if time.time() < deadline - 250:
        t4_end = min(time.time() + 150, deadline - 250)
        for alpha in [2000, 10000, 50000]:
            if time.time() > t4_end - 5:
                break
            to = min(45, t4_end - time.time() - 3)
            if to < 5:
                break
            h_new = lbfgs_full(best_h, n, alpha, 100.0, 3000, to)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 5: Top-shifts focused with very sharp alpha (200s)
    if time.time() < deadline - 150:
        t5_end = min(time.time() + 200, deadline - 150)
        for it in range(20):
            if time.time() > t5_end - 5:
                break
            top_idx = find_active_shifts(best_h, n, top_k=5, frac=0.001)
            to = min(15, t5_end - time.time() - 2)
            if to < 3:
                break
            alpha = 100000 * (it + 1)
            h_new = lbfgs_top(best_h, n, top_idx, alpha, 1000.0, 3000, to)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 6: Gradient descent local refinement (100s)
    if time.time() < deadline - 80:
        t6_end = min(time.time() + 100, deadline - 80)
        best_h, best_c = gradient_descent_local(best_h, n, t6_end, top_k=5)

    # Phase 7: Random perturb + re-optimization (80s)
    if time.time() < deadline - 60:
        t7_end = min(time.time() + 80, deadline - 60)
        for mag in [0.03, 0.01, 0.005]:
            if time.time() > t7_end - 10:
                break
            h_pert = random_perturb(best_h, n, mag)
            c_pert = compute_c5(h_pert, n)
            start_h = best_h if c_pert > best_c else h_pert
            if c_pert < best_c:
                best_h = h_pert
                best_c = c_pert
            top_idx = find_active_shifts(start_h, n, top_k=5)
            to = min(15, t7_end - time.time() - 2)
            if to > 3:
                h_new = lbfgs_top(start_h, n, top_idx, 500000, 5000.0, 2000, to)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new

    # Phase 8: Fine local pair search (60s)
    if time.time() < deadline - 30:
        t8_end = min(time.time() + 60, deadline - 30)
        best_h, best_c = local_pair_search(best_h, n, t8_end,
                                            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                            radius=30)

    # Phase 9: Final top-shifts polish (30s)
    if time.time() < deadline - 5:
        t9_end = time.time() + min(30, deadline - time.time() - 2)
        top_idx = find_active_shifts(best_h, n, top_k=3, frac=0.0005)
        h_new = lbfgs_top(best_h, n, top_idx, 5000000, 50000.0, 3000, max(3, t9_end - time.time() - 1))
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
