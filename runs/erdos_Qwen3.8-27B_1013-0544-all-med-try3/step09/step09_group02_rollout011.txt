def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

        k = mm - (n - 1)

    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)

        hi = min(1.0 - h[i], rho)

        if res.success:

        pass

        importance += np.abs(shift_grad_single(h, n, idx))
    if len(active) > 0:
        importance /= len(active)
    top_count = min(60, n)
    top_idx = np.argsort(-importance)[:top_count]
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - 0.002 * max_val
            active = np.where(c_vec >= thresh)[0]
            importance = np.zeros(n)
            for idx in active:
                importance += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                importance /= len(active)
            top_idx = np.argsort(-importance)[:top_count]
            for i_pos in range(len(top_idx)):
                if time.time() > deadline:
                    return best_h, best_c
                i = int(top_idx[i_pos])
                for j_pos in range(len(top_idx)):
                    if i_pos == j_pos:
                        continue
                    j = int(top_idx[j_pos])
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def stochastic_hclimb(h, n, deadline, n_restart=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(n_restart):
        if time.time() > deadline:
            break
        pert_scale = 0.02 - 0.01 * (r / max(n_restart - 1, 1))
        h_try = h + np.random.randn(n) * pert_scale
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            h = h_try
        h2, c2 = lp_optimize(h_try, n, min(time.time() + 15, deadline - 5),
                             max_iter=200, rho_init=0.08, max_active=60)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            h = h2
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 50.0), (2000, 500.0), (8000, 2000.0), (20000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def slqp_obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def slqp_sq_func(x, n):
        return float(x.sum() - n / 2.0)
    def slqp_sq_jac(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_sq_func, 'jac': slqp_sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup (20s)
    t1 = min(t0 + 20, deadline - 850)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 50.0), (2000, 500.0), (8000, 2000.0), (20000, 5000.0)])

    # Phase 2: LP cutting plane - main (300s)
    t2 = min(t0 + 320, deadline - 550)
    if t2 > time.time() + 10:
        best_h, best_c = lp_optimize(best_h, n_orig, t2,
                                     max_iter=3000, rho_init=0.15, max_active=100)

    # Phase 3: Local search (80s)
    t3 = min(t0 + 400, deadline - 450)
    if t3 > time.time() + 5:
        best_h, best_c = local_search(best_h, n_orig, t3)

    # Phase 4: Second LP round (100s)
    t4 = min(t0 + 500, deadline - 350)
    if t4 > time.time() + 10:
        best_h, best_c = lp_optimize(best_h, n_orig, t4,
                                     max_iter=2000, rho_init=0.05, max_active=120)

    # Phase 5: Third local search (50s)
    t5 = min(t0 + 550, deadline - 300)
    if t5 > time.time() + 5:
        best_h, best_c = local_search(best_h, n_orig, t5)

    # Phase 6: SLSQP polish (40s)
    t6 = min(t0 + 590, deadline - 250)
    if t6 > time.time() + 10:
        h6, c6 = slqp_polish(best_h, n_orig, t6, alpha=5000000, maxiter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Stochastic restarts + LP re-optimize (80s)
    t7 = min(t0 + 670, deadline - 180)
    if t7 > time.time() + 10:
        best_h, best_c = stochastic_hclimb(best_h, n_orig, t7, n_restart=4)

    # Phase 8: Upsample to n=1000 and optimize (200s)
    n_high = 1000
    t8_start = time.time()
    if t8_start < deadline - 100:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t8a = min(t8_start + 50, deadline - 60)
        if t8a > time.time() + 3:
            h_high, c_high = run_lbfgs(h_high, n_high, t8a,
                configs=[(1000, 100.0), (5000, 500.0), (15000, 2000.0)])

        t8b = min(t8_start + 120, deadline - 30)
        if t8b > time.time() + 10:
            h_high, c_high = lp_optimize(h_high, n_high, t8b,
                                         max_iter=1000, rho_init=0.1, max_active=80)

        t8c = min(t8_start + 170, deadline - 10)
        if t8c > time.time() + 5:
            h_high, c_high = local_search(h_high, n_high, t8c)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Phase 9: Final polish at whatever resolution we have (30s)
    if time.time() < deadline - 5:
        h_final, c_final = lp_optimize(best_h, len(best_h), deadline - 3,
                                       max_iter=500, rho_init=0.03, max_active=100)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
