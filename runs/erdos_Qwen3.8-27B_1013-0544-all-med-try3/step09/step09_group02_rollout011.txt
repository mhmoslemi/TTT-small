    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def huber_obj_grad(x, n, target_C, mu):
    """
    Objective: sum_k max(0, c_k - target_C)^2 + mu*(sum(x) - n/2)^2
    where c_k = (correlate(x, 1-x))[k] * dx
    """
    dx = 2.0 / n
    c = compute_c5_vec(x, n)
    excess = np.maximum(0, c - target_C)
    obj = float(np.sum(excess**2) / (2*n) + mu * (x.sum() - n/2)**2)
    
    # Gradient
    grad_c = np.zeros_like(excess)
    grad_c[excess > 0] = excess[excess > 0]  # d/dx of excess^2/2 = excess

            continue
        wm = grad_c[k] * dx

            grad[lag:n] += wm * (-x[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

        h_try = project_feasible(h_try, n)

            h = h_try

                h = best_h.copy() + np.random.randn(n) * 0.001
                h = project_feasible(h, n)
                lr = 0.01
                break
    return best_h, best_c

                no_improve += 1
                rho *= 0.5
        except Exception:
            no_improve += 1
            rho *= 0.5

        if rho < 1e-9 or no_improve > 25:
            break

    return best_h, best_c

def pairwise_fine_tune(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    step_sizes = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]

    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            top_k = np.argsort(-c_vec)[:8]
            grad = np.zeros(n)
            for idx in top_k:
                grad += shift_grad_single(h, n, idx)
            grad /= len(top_k)

            n_cand = min(40, n)
            donate = np.argsort(-grad)[:n_cand]
            receive = np.argsort(grad)[:n_cand]

            for i in receive:
                if time.time() > deadline:
                    return best_h, best_c
                for j in donate:
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
            h = project_feasible(h, n)

    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=3000):
    def obj_fn(x):
        c = compute_c5_vec(x, len(x))
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha

    def sq_fn(x):
        return float(x.sum() - len(x) / 2.0)

    def sq_jac_fn(x):
        return np.ones(len(x))

    b = [(0.0, 1.0)] * len(x)
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=b, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS warmup (50s)
    t1 = min(t0 + 50, deadline - 700)
    if t1 > time.time() + 5:
        bounds = [(0.0, 1.0)] * n_orig
        configs = [(300, 30.0), (1000, 100.0), (4000, 500.0)]
        for alpha, mu in configs:
            if time.time() > t1 - 3:
                break
            try:
                res = minimize(softmax_obj, best_h, jac=softmax_grad,
                              args=(n_orig, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxfun': 30000, 'maxls': 40})
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: SQP (150s)
    t2 = min(t0 + 200, deadline - 450)
    if t2 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=400, rho_init=0.04)

    # Phase 3: Projected gradient - medium alpha (100s)
    t3 = min(t0 + 300, deadline - 300)
    if t3 > time.time() + 10:
        best_h, best_c = projected_grad_opt(best_h, n_orig, t3, alpha=5000, mu=50.0)

    # Phase 4: Pairwise fine-tune (80s)
    t4 = min(t0 + 380, deadline - 250)
    if t4 > time.time() + 5:
        best_h, best_c = pairwise_fine_tune(best_h, n_orig, t4)

    # Phase 5: SQP again with tighter params (80s)
    t5 = min(t0 + 460, deadline - 200)
    if t5 > time.time() + 10:
        best_h, best_c = sqp_optimize(best_h, n_orig, t5, max_iter=300, rho_init=0.02)

    # Phase 6: Projected gradient - high alpha (80s)
    t6 = min(t0 + 540, deadline - 150)
    if t6 > time.time() + 10:
        best_h, best_c = projected_grad_opt(best_h, n_orig, t6, alpha=50000, mu=100.0)

    # Phase 7: SLSQP polish (50s)
    t7 = min(t0 + 590, deadline - 80)
    if t7 > time.time() + 10:
        h7, c7 = slqp_polish(best_h, n_orig, t7, alpha=8000000, maxiter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Try higher resolution n=800
    n_high = 800
    if time.time() < deadline - 120:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)

        t8 = min(time.time() + 50, deadline - 60)
        if t8 > time.time() + 5:
            bounds_h = [(0.0, 1.0)] * n_high
            try:
                res = minimize(softmax_obj, h_high, jac=softmax_grad,
                             args=(n_high, 3000, 80.0), method='L-BFGS-B',
                             bounds=bounds_h,
                             options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12})
                h_high = project_feasible(res.x, n_high)
                c_high = compute_c5(h_high, n_high)
            except Exception:
                pass

        t9 = min(time.time() + 40, deadline - 30)
        if t9 > time.time() + 5:
            h_high, c_high = projected_grad_opt(h_high, n_high, t9, alpha=20000, mu=80.0)

        t10 = min(time.time() + 25, deadline - 10)
        if t10 > time.time() + 3:
            h_high, c_high = pairwise_fine_tune(h_high, n_high, t10)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high

    # Phase 9: Final pairwise fine-tune (remaining time)
    if time.time() < deadline - 8:
        n_fin = len(best_h)
        best_h, best_c = pairwise_fine_tune(best_h, n_fin, deadline - 3)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
