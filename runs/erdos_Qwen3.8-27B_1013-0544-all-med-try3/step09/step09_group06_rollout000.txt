    """Simulated annealing on binary solutions"""
    # Convert to binary
    h = np.zeros(n)
    idx = np.argsort(h0)[::-1][:n//2]
    h[idx] = 1.0
    
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    best_c = c.max()
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(42)
    
    for step in range(steps):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (step / steps)
        
        # Pick random swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        a = ones[i]  # was 1, becomes 0

        h[a] = 0
        h[b] = 1
        c_new = np.correlate(h, 1-h, mode='full') * dx
        new_c = c_new.max()
        
        # Update
        if new_c < best_c or rng.random() < np.exp(-(new_c - best_c) / max(T, 1e-10)):

                best_c = new_c

            # Reject
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c
