            if tot < 1e-14:
                break
            h += (diff / tot) * h

    """Compute c[k]*dx for all shifts using FFT."""
    if N is None:
        N = 1
        while N < 2 * n - 1:
            N *= 2
    H = fft(h, N)
    G = fft(1.0 - h, N)
    R = np.real(ifft(H * np.conj(G)))
    dx = 2.0 / n
    c = np.empty(2 * n - 1)
    c[n - 1] = R[0] * dx
    c[n:] = R[N-1 : N-n : -1] * dx
    c[:n-1] = R[1:n][::-1] * dx
    return c

    """Compute C5 = max of correlation."""
    c = fft_correlation(h, n, N_cache)
    return float(c.max())

def active_gradient(h, n, c, alpha, top_k=30, N_cache=None):
    """Compute gradient of smooth log-sum-exp objective."""
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Get indices with significant weight
    significant = np.where(w > 1e-8)[0]
    if len(significant) == 0:
        return grad
    
    # Sort by weight, take top_k
    if len(significant) > top_k:
        order = np.argsort(-w[significant])
        significant = significant[order[:top_k]]
    
    for idx in significant:
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    
    return grad

def adam_optimize(h, n, deadline, alpha, lr, top_k=30, mu=0.01, max_steps=50000, 
                  patience=2000, verbose=False):
    """Adam optimizer for the smooth objective."""
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    best_h = h.copy()
    best_c = compute_c5(h, n, N)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    h = h.copy()
    step_count = 0
    stale = 0
    lr_current = lr
    
    while step_count < max_steps:
        if time.time() > deadline:
            break
        
        c = fft_correlation(h, n, N)
        grad = active_gradient(h, n, c, alpha, top_k, N)
        
        # Add constraint penalty
        sum_diff = h.sum() - n / 2.0
        if abs(sum_diff) > 1e-10:
            grad += mu * sum_diff * np.ones(n)
        
        step_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        
        h_new = h - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n, N)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        
        h = h_new
        
        # Learning rate decay
        if step_count % 1000 == 0:
            lr_current *= 0.95
            top_k = max(5, top_k - 1)  # Focus more on the max
        
        if stale > patience:
            break
    
    return best_h, best_c
