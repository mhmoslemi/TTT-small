```python
def linearized_active_step(h, n, active, max_iter=100, deadline=0):
    """Successive linearization of the active-set problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(best_h, n)
        max_val = c.max()
        
        # Update active set
        new_active = np.where(c >= max_val - 1e-8)[0]
        if len(new_active) > 20:
            idx = np.argsort(-c)
            new_active = idx[0:20]
        
        # Compute c_k and gradients for active shifts
        c_active = c[new_active]
        dxx = 2.0 / n
        
        # Build LP: min t s.t. c_k + grad_k · (h' - h) <= t, 0 <= h' <= 1, sum(h') = n/2
        # Variables: [h'_0, ..., h'_{n-1}, t]
        # Objective: [0, 0, ..., 0, 1] (minimize t)
        
        cobj = np.zeros(n + 1)
        cobj[-1] = 1.0  # minimize t
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active shift k: grad_k · h' - t <= c_k - grad_k · h
        A_rows = []
        b_rows = []
        for idx, mm in enumerate(new_active):
            k = int(mm) - (n - 1)
            grad = np.zeros(n)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] = (1.0 - best_h[k:n]) * dxx
                    grad[k:n] += (-best_h[0:nv]) * dxx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] = (1.0 - best_h[0:nv]) * dxx
                    grad[0:nv] += (-best_h[kk:n]) * dxx
            # grad · h' - t <= c_k - grad · h
            row = np.zeros(n + 1)
            row[0:n] = grad
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(c_active[idx] - np.dot(grad, best_h))
        
        if len(A_rows) > 0:
            A_ub = np.array(A_rows)
            b_ub = np.array(b_rows)
        else:
            A_ub = None
            b_ub = None
        
        # Equality constraint: sum(h') = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h'_i <= 1, -inf <= t <= inf (or 0 <= t)
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            from scipy.optimize import linprog
            res = linprog(cobj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
            if res.status == 0 and res.x is not None:
                h_new = project_feasible(res.x[0:n], n)
                c_new = compute_c5(h_new, n)
```

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
