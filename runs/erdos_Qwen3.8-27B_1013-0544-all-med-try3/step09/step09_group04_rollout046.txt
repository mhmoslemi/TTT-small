        val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)


def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nh = n - 1
    for lag in range(2 * n - 1):
        if w[lag] < 1e-12:
            continue
        k = lag - nh
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - 0.5 * n)
    return grad


def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)


def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nh = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-12:
            continue
        k = idx - nh
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - 0.5 * n)
    return grad


def lbfgs_polish(h, n, deadline, alphas, mu=0.005, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        rem = deadline - time.time()
        if rem < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(rem * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 8,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def active_opt(h, n, deadline, top_k=6, alphas=None, rounds=30):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000]
    for rnd in range(rounds):
        rem = deadline - time.time()
        if rem < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(100, int(rem2 * 1.2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': maxit * 8,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c


def de_population(h0, n, pop_size, rng, radius=0.3):
    pop = [project_feasible(h0.copy(), n)]
    for _ in range(pop_size - 1):
        idx = rng.integers(0, len(pop))
        pert = pop[idx] + radius * rng.standard_normal(n)
        pop.append(project_feasible(pert, n))
    return pop


def run_de(h0, n, deadline, seed, pop_size=40, F_vals=(0.5, 0.7, 0.9),
           CR_vals=(0.7, 0.8, 0.9)):
    rng = np.random.default_rng(seed)
    pop = de_population(h0, n, pop_size, rng, radius=0.4)
    fvals = np.array([compute_c5(p, n) for p in pop])
    best_idx = int(fvals.argmin())
    best_h = pop[best_idx].copy()
    best_f = fvals[best_idx]
    no_improve = 0
    gen = 0
    while time.time() < deadline:
        F = F_vals[gen % len(F_vals)]
        CR = CR_vals[gen % len(CR_vals)]
        if no_improve > 100:
            F = min(1.0, F + 0.2)
        for i in range(pop_size):
            if time.time() > deadline:
                break
            idxs = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = project_feasible(mutant, n)
            j_rand = rng.integers(0, n)
            mask = rng.random(n) <= CR
            mask[j_rand] = True
            child = np.where(mask, mutant, pop[i])
            child = project_feasible(child, n)
            f_child = compute_c5(child, n)
            if f_child < fvals[i]:
                pop[i] = child
                fvals[i] = f_child
                if f_child < best_f:
                    best_f = f_child
                    best_h = child.copy()
                    no_improve = 0
            else:
                no_improve += 1
        gen += 1
    return best_h, best_f


def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)


def structured_perturb(h, n, rng, strength=0.05):
    h2 = h.copy()
    strategy = rng.integers(0, 4)
    if strategy == 0:
        b = int(n * rng.uniform(0.1, 0.3))
        start = rng.integers(0, max(1, n - b))
        h2[start:start + b] += strength * rng.standard_normal(b)
    elif strategy == 1:
        t = np.linspace(0, 4 * np.pi, n)
        h2 += strength * np.sin(t)
    elif strategy == 2:
        for _ in range(3):
            i = rng.integers(0, n - 1)
            h2[i], h2[i + 1] = h2[i + 1], h2[i]
    else:
        idx = rng.permutation(n)[:n // 4]
        h2[idx] = h[idx[rng.permutation(len(idx))]]
    return project_feasible(h2, n)


def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c


def subgrad_refine(h, n, deadline, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        nh = n - 1
        for idx in active:
            k = idx - nh
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= max(1, len(active))
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.015)
        else:
            eta *= 0.96
        h = h_new
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    rng = np.random.default_rng(seed)

    # Phase 1: DE exploration at n=400 (global search)
    n_de = 400
    if best_n != n_de:
        h_de_start = upsample(h0, best_n, n_de)
    else:
        h_de_start = h0.copy()
    p1_end = t0 + 100
    h_de, c_de = run_de(h_de_start, n_de, p1_end, seed=seed + 1,
                        pop_size=45, F_vals=(0.4, 0.6, 0.8),
                        CR_vals=(0.6, 0.8, 0.9))
    if c_de < best_c:
        best_c = c_de
        best_h = h_de
        best_n = n_de

    # Phase 2: Quick L-BFGS polish at n=500
    n_work = 500
    if time.time() < deadline - 500:
        h_work = upsample(best_h, best_n, n_work)
        p2_end = min(time.time() + 120, deadline - 380)
        h2, c2 = lbfgs_polish(h_work, n_work, p2_end,
                              alphas=[20, 100, 500, 2000, 10000, 50000],
                              mu=0.005,
                              maxiters=[600, 800, 1200, 1500, 2000, 2500])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_work

    # Phase 3: Active set optimization
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 100, deadline - 250)
        h3, c3 = active_opt(best_h, best_n, p3_end, top_k=5,
                            alphas=[50000, 200000, 1000000, 5000000],
                            rounds=25)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: DE at n=500 with structured perturbations
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 80, deadline - 200)
        pop_init = [best_h.copy()]
        for _ in range(15):
            pop_init.append(structured_perturb(best_h, best_n, rng, 0.08))
        h4, c4 = run_de(best_h, best_n, p4_end, seed=seed + 100,
                        pop_size=20, F_vals=(0.3, 0.5, 0.7),
                        CR_vals=(0.5, 0.7, 0.9))
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Upsample to n=800 and polish
    n_high = 800
    if time.time() < deadline - 200:
        h_high = upsample(best_h, best_n, n_high)
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = lbfgs_polish(h_high, n_high, p5_end,
                              alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
                              mu=0.003,
                              maxiters=[500, 700, 1000, 1200, 1500, 2000, 2500])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_high

    # Phase 6: Active set at current best
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 80)
        h6, c6 = active_opt(best_h, best_n, p6_end, top_k=4,
                            alphas=[100000, 500000, 2000000, 10000000],
                            rounds=20)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Iterated local search with perturbations
    if time.time() < deadline - 80:
        p7_end = min(time.time() + 50, deadline - 50)
        h_ils = best_h.copy()
        c_ils = best_c
        for it in range(15):
            if time.time() > p7_end:
                break
            pert = structured_perturb(h_ils, best_n, rng, 0.03)
            h_tmp, c_tmp = lbfgs_polish(pert, best_n,
                                        min(time.time() + 8, p7_end),
                                        alphas=[5000, 50000, 500000],
                                        mu=0.01, maxiters=[200, 400, 600])
            if c_tmp < c_ils:
                c_ils = c_tmp
                h_ils = h_tmp
        if c_ils < best_c:
            best_c = c_ils
            best_h = h_ils

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 20, deadline - 15)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=5000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Pair search
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: active-set push on worst constraint
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = int(np.argmax(c_vec))
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 10000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 300, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': 800})
            h_f = project_feasible(res.x, best_n)
            c_f = compute_c5(h_f, best_n)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        except Exception:
            pass

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
