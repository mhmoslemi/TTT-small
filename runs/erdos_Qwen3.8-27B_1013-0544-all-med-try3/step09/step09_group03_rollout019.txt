```python
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for idx in range(n2m1):
        if w[idx] < 1e-15:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[0:nv])
        else:
            m2 = -k
            nv = n - m2
            if nv > 0:
                grad[m2:] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[m2:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 25000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def binary_local_search(h, n, deadline, n_passes=200):
    """Greedy binary swap local search."""
    h_b = np.where(h > 0.5, 1.0, 0.0)
    target = n / 2
    # Ensure exactly n/2 ones
    if h_b.sum() != target:
        if h_b.sum() > target:
            # Turn some 1s to 0s
            ones = np.where(h_b == 1)[0]
            # Turn off the ones where h is closest to 0.5
            scores = h[ones] - 0.5  # smallest first
            order = np.argsort(scores)
            for i in order:
                h_b[ones[i]] = 0
                if h_b.sum() == target:
                    break
        else:
            zeros = np.where(h_b == 0)[0]
            scores = 0.5 - h[zeros]  # smallest first (closest to 0.5)
            order = np.argsort(scores)
            for i in order:
                h_b[zeros[i]] = 1
                if h_b.sum() == target:
                    break
    
    best_c = compute_c5(h_b, n)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        # Randomize order for diversity
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            if time.time() > deadline:
                return h_b, best_c
            for j in zeros:
                # Try swap: h[i] 1->0, h[j] 0->1
                h_b[i] = 0
                h_b[j] = 1
                c = compute_c5(h_b, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    # Recompute zeros (j is no longer zero)
                    break
                else:
                    h_b[i] = 1
                    h_b[j] = 0
            if improved:
                zeros = np.where(h_b == 0.0)[0]
                improved = False
    return h_b, best_c
```

```python
def binary_sa(h, n, deadline, n_trials=5000):
    """Simulated annealing on binary patterns."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2
    # Fix sum
    if h_b.sum() > target:
        ones_idx = np.where(h_b == 1)[0]
        np.random.shuffle(ones_idx)
        for idx in ones_idx:
            h_b[idx] = 0
            if h_b.sum() == target:
                break
    elif h_b.sum() < target:
        zeros_idx = np.where(h_b == 0)[0]
        np.random.shuffle(zeros_idx)
        for idx in zeros_idx:
            h_b[idx] = 1
            if h_b.sum() == target:
                break
    
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    temp = 0.01
    cooling = 0.9995
    
    for t in range(n_trials):
        if time.time() > deadline:
            break
        # Random swap
        ones_idx = np.where(h_b == 1.0)[0]
        zeros_idx = np.where(h_b == 0.0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        
        h_b[i] = 0.0
        h_b[j] = 1.0
        c_new = compute_c5(h_b, n)
        
        if c_new <= best_c or np.random.rand() < np.exp((best_c - c_new) / temp):
            best_c = min(best_c, c_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_b.copy()
        else:
            h_b[j] = 0.0
            h_b[i] = 1.0
        
        temp *= cooling
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for idx in range(n2m1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[0:nv])
        else:
            m2 = -k
            nv = n - m2
            if nv > 0:
                grad[m2:] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[m2:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if time.time() > deadline:
        return h, best_c
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 25000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros_idx = np.where(h_b == 0.0)[0]
            order = np.argsort(h[zeros_idx])[::-1]

    h_b = project_feasible(h_b, n)

        ones_idx = np.where(h_b == 1.0)[0]

def slqp_polish(h, n, deadline, alpha=300000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(30):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        found = False
        if k_worst >= 0:
            affected = list(range(max(0, n - k_worst - 20), min(n, n - k_worst + 20)))
        else:
            affected = list(range(max(0, -k_worst - 20), min(n, -k_worst + 20)))
        for i in affected:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = best_h[i]
            if found:
                h = best_h.copy()
        if not found and round_num > 10:
            break
    return best_h, best_c

    h_coarse = interpolate_to(best_h, n_start, n_coarse)
    t_coarse_end = min(time.time() + 40, deadline - 300)
    if t_coarse_end > time.time() + 10:
        h_coarse, c_coarse = run_lbfgs(h_coarse, n_coarse, t_coarse_end,
            configs=[(100, 50.0), (500, 200.0), (2000, 1000.0)])
        if c_coarse < best_c:
            best_c = c_coarse
            best_h = interpolate_to(h_coarse, n_coarse, n_start)
    n_mid = max(n_start // 2, 100)
    h_mid = interpolate_to(best_h, n_start, n_mid)
    t_mid_end = min(time.time() + 80, deadline - 150)
    if t_mid_end > time.time() + 10:
        h_mid, c_mid = run_lbfgs(h_mid, n_mid, t_mid_end,
            configs=[(500, 200.0), (2000, 1000.0), (10000, 5000.0)])
        if c_mid < best_c:
            best_c = c_mid
            best_h = interpolate_to(h_mid, n_mid, n_start)
    return best_h, best_c

def random_binary_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary SA at base resolution
    t1 = min(t0 + 35, deadline - 500)
    if t1 > time.time() + 5:
        h_sa, c_sa = binary_sa(best_h, n, t1, n_trials=4000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 2: Binary greedy from SA result
    t2 = min(t0 + 60, deadline - 450)
    if t2 > time.time() + 5:
        h_g, c_g = binary_greedy(best_h, n, t2, n_passes=40)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    # Phase 3: L-BFGS from binary
    t3 = min(t0 + 140, deadline - 350)
    if t3 > time.time() + 10:
        h3, c3 = run_lbfgs(best_h, n, t3,
            configs=[(200, 100.0), (1000, 500.0), (5000, 2000.0),
                     (20000, 10000.0), (100000, 50000.0)])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: L-BFGS from original continuous
    t4 = min(t0 + 220, deadline - 300)
    if t4 > time.time() + 10:
        h4, c4 = run_lbfgs(h_init.copy(), n, t4,
            configs=[(200, 100.0), (1000, 500.0), (5000, 2000.0),
                     (20000, 10000.0), (100000, 50000.0)])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: SLSQP polish at base
    t5 = min(t0 + 290, deadline - 200)
    if t5 > time.time() + 10:
        h5, c5 = slqp_polish(best_h, n, t5, alpha=500000, maxiter=4000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Multi-scale (coarse to fine)
    t6 = min(t0 + 420, deadline - 200)
    if t6 > time.time() + 30:
        h6, c6 = multi_scale_optimize(best_h, n, t6)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation search at base
    t7 = min(t0 + 480, deadline - 150)
    if t7 > time.time() + 10:
        best_h, best_c = perturbation_search(best_h, n, t7,
            n_trials=8, mags=[0.02, 0.04, 0.01, 0.06, 0.03, 0.08, 0.015, 0.05])

    # Phase 8: Quick SLSQP after perturbation
    t8 = min(t0 + 530, deadline - 120)
    if t8 > time.time() + 5:
        h8, c8 = slqp_polish(best_h, n, t8, alpha=800000, maxiter=3000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Upsample to 2n and optimize
    n2 = 2 * n
    t9 = min(t0 + 800, deadline - 80)
    if t9 > time.time() + 60:
        h2x = interpolate_to(best_h, n, n2)
        h2x, c2x = run_lbfgs(h2x, n2, t9 - 40,
            configs=[(500, 200.0), (2000, 1000.0), (8000, 5000.0)])
        if c2x < best_c:
            best_c = c2x
            best_h = h2x
        t9b = min(t9 - 40 + 30, deadline - 30)
        if t9b > time.time() + 5:
            h9b, c9b = slqp_polish(best_h, len(best_h), t9b,
                                   alpha=1000000, maxiter=2000)
            if c9b < best_c:
                best_c = c9b
                best_h = h9b

    # Phase 10: Perturbation at high resolution
    t10 = min(t0 + 900, deadline - 50)
    if t10 > time.time() + 15:
        best_h, best_c = perturbation_search(best_h, len(best_h), t10,
            n_trials=4, mags=[0.01, 0.02, 0.015, 0.03])

    # Phase 11: Targeted refinement
    t11 = deadline - 5
    if t11 > time.time() + 5:
        best_h, best_c = targeted_refine(best_h, len(best_h), t11)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
```
