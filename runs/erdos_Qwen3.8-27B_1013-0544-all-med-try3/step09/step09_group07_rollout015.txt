```python
def compute_delta_c(h, n, i, j, d):
    """Compute the change in all lags when transferring d from position i to position j.
    h[i] -= d, h[j] += d."""
    # c(m) for offset m = k - (n-1), k = 0, ..., 2n-2
    # delta_c(k) for each k
    
    # For position i (decreasing by d):
    # First index: contributes -d * (1-h[i+m]) if 0 <= i+m < n
    # Second index: contributes +d * h[i-m] if 0 <= i-m < n
    # For position j (increasing by d):
    # First index: contributes +d * (1-h[j+m]) if 0 <= j+m < n
    # Second index: contributes -d * h[j-m] if 0 <= j-m < n
    
    delta = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        m = k - (n - 1)
        dc = 0.0
        # i as first index
        if 0 <= i + m < n:
            dc += -d * (1.0 - h[i + m])
        # i as second index
        if 0 <= i - m < n:
            dc += d * h[i - m]
        # j as first index
        if 0 <= j + m < n:
            dc += d * (1.0 - h[j + m])
        # j as second index
        if 0 <= j - m < n:
            dc += -d * h[j - m]
        delta[k] = dc
    return delta
```

```python
def compute_delta_c_vec(h, n, i, j, d):
    m_vals = np.arange(-(n-1), n)  # all possible offsets
    # i as first: 0 <= i+m < n
    mask_i_first = (i + m_vals >= 0) & (i + m_vals < n)
    # i as second: 0 <= i-m < n
    mask_i_second = (i - m_vals >= 0) & (i - m_vals < n)
    # j as first
    mask_j_first = (j + m_vals >= 0) & (j + m_vals < n)
    # j as second
    mask_j_second = (j - m_vals >= 0) & (j - m_vals < n)
    
    delta = np.zeros(2*n-1)
    delta[mask_i_first] += -d * (1.0 - h[i + m_vals[mask_i_first]])
    delta[mask_i_second] += d * h[i - m_vals[mask_i_second]]
    delta[mask_j_first] += d * (1.0 - h[j + m_vals[mask_j_first]])
    delta[mask_j_second] += -d * h[j - m_vals[mask_j_second]]
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(c, h, n, alpha):
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n-lag])
            grad[0:n-lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n-idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n-idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    penalty_coeff = 5e6

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(c, x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 80,
                                'iprint': -1, 'gtol': 1e-12})
    except Exception:
        return project_feasible(h.copy(), n)
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    target_sum = n / 2.0

    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        return smooth_max_value(c, alpha)

    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        return smooth_max_grad(c, x, n, alpha)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    x0 = h.copy()
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad, bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'iprint': 0, 'ftol': 1e-12})
    except Exception:
        return project_feasible(h.copy(), n)
    return project_feasible(np.clip(res.x, 0, 1), n)

def delta_c_compute(h, n, i, j, d):
    m_vals = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    mask_i1 = (i + m_vals >= 0) & (i + m_vals < n)
    mask_i2 = (i - m_vals >= 0) & (i - m_vals < n)
    mask_j1 = (j + m_vals >= 0) & (j + m_vals < n)
    mask_j2 = (j - m_vals >= 0) & (j - m_vals < n)
    if np.any(mask_i1):
        idx = np.where(mask_i1)[0]
        delta[idx] += -d * (1.0 - h[i + m_vals[idx]])
    if np.any(mask_i2):
        idx = np.where(mask_i2)[0]
        delta[idx] += d * h[i - m_vals[idx]]
    if np.any(mask_j1):
        idx = np.where(mask_j1)[0]
        delta[idx] += d * (1.0 - h[j + m_vals[idx]])
    if np.any(mask_j2):
        idx = np.where(mask_j2)[0]
        delta[idx] += -d * h[j - m_vals[idx]]
    return delta

def fast_local_search(h, n, deadline, topK=6):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    c_all = compute_all_lags(h, n)
    best_c = float(c_all.max())
    
    for sweep in range(200000):
        if time.time() > deadline:
            break
        top_idx = np.argsort(c_all)[-topK:]
        improved_this_sweep = False
        
        for k_idx in np.random.permutation(top_idx):
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            start = 0 if lag > 0 else -lag
            end = n - m
            for i in range(start, end):
                if time.time() > deadline:
                    break
                j = i + m
                if abs(h[i] - h[j]) < 1e-9:
                    continue
                if h[i] > h[j]:
                    d = min(0.03, (h[i] - h[j]) * 0.5)
                    if d < 1e-10:
                        continue
                else:
                    d = -min(0.03, (h[j] - h[i]) * 0.5)
                    if abs(d) < 1e-10:
                        continue
                
                dc = delta_c_compute(h, n, i, j, d)
                new_c = c_all + dc
                new_max = float(new_c.max())
                
                if new_max < best_c - 1e-14:
                    h[i] -= d if d > 0 else (-d)
                    h[j] += d if d > 0 else (-d)
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    h = project_feasible(h, n)
                    c_all = compute_all_lags(h, n)
                    new_max = float(c_all.max())
                    if new_max < best_c:
                        best_c = new_max
                        best_h = h.copy()
                        improved_this_sweep = True
                    else:
                        best_h2 = h.copy()
                        h = project_feasible(best_h, n)
                        c_all = compute_all_lags(h, n)
                else:
                    pass
        
        if not improved_this_sweep:
            try_fine(h, n, c_all, deadline, topK, 5)
            c_all_new = compute_all_lags(h, n)
            c_max_new = float(c_all_new.max())
            if c_max_new < best_c:
                best_c = c_max_new
                best_h = h.copy()
            if time.time() > deadline:
                break
            # If still stuck, try random perturbations
            stuck = True
            for _ in range(20):
                if time.time() > deadline:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                if h[i] > h[j] + 1e-8:
                    d = min(0.01, (h[i]-h[j])*0.3)
                elif h[j] > h[i] + 1e-8:
                    d = -min(0.01, (h[j]-h[i])*0.3)
                else:
                    continue
                if abs(d) < 1e-10:
                    continue
                dc = delta_c_compute(h, n, i, j, d)
                new_c = c_all + dc
                if new_c.max() < best_c - 1e-14:
                    h[i] += -d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_all = compute_all_lags(h, n)
                    if c_all.max() < best_c:
                        best_c = float(c_all.max())
                        best_h = h.copy()
                    stuck = False
                    break
            if stuck:
                break
    
    return best_h, best_c

def try_fine(h, n, c_all, deadline, topK, max_rounds):
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        top_idx = np.argsort(c_all)[-topK:]
        any_improve = False
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n-1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 1e-10:
                    continue
                d = min(0.005, abs(diff) * 0.3) * (1 if diff > 0 else -1)
                dc = delta_c_compute(h, n, i, j, d)
                new_c = c_all + dc
                if new_c.max() < c_all.max() - 1e-14:
                    h[i] += -d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_all = compute_all_lags(h, n)
                    any_improve = True
                    break
        if not any_improve:
            break

def binary_local_search(h_vals, n, deadline, rng):
    h = (h_vals > 0.5).astype(np.float64)
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        one_idx = np.where(h == 1)[0]
        order = np.argsort(h_vals[one_idx])
        for idx in order[:num_ones - target]:
            h[one_idx[idx]] = 0.0
    elif num_ones < target:
        zero_idx = np.where(h == 0)[0]
        order = np.argsort(-h_vals[zero_idx])
        for idx in order[:target - num_ones]:
            h[zero_idx[idx]] = 1.0
    
    c_all = compute_all_lags(h, n)
    best_c = float(c_all.max())
    best_h = h.copy()
    
    for sweep in range(500000):
        if time.time() > deadline:
            break
        top_idx = np.argsort(c_all)[-10:]
        improved = False
        
        for k_idx in np.random.permutation(top_idx):
            if time.time() > deadline:
                break
            lag = k_idx - (n-1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] == h[j]:
                    continue
                # Swap h[i] and h[j]
                h[i], h[j] = h[j], h[i]
                new_c = compute_all_lags(h, n)
                new_max = float(new_c.max())
                if new_max < best_c - 1e-15:
                    best_c = new_max
                    best_h = h.copy()
                    c_all = new_c
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            # Try random swaps
            any_swaps = False
            for _ in range(50):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c = compute_all_lags(h, n)
                if new_c.max() < best_c - 1e-15:
                    best_c = float(new_c.max())
                    best_h = h.copy()
                    c_all = new_c
                    any_swaps = True
                else:
                    h[i], h[j] = h[j], h[i]
            if not any_swaps:
                break
    
    return best_h, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    
    # Phase 1: L-BFGS-B with increasing alpha
    alpha_list = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alpha_list:
        if time.time() > deadline - 400:
            break
        remaining_for_phase = deadline - 400 - time.time()
        if remaining_for_phase < 5:
            break
        t_phase = min(20, remaining_for_phase * 0.15)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t0 + time.time() - t0 + t_phase, maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: SLSQP for exact constraint handling
    if time.time() < deadline - 300:
        h_slsqp = slsqp_optimize(best_h, n, 50000, deadline - 280, maxiter=200)
        c_slsqp = compute_c5(h_slsqp, n)
        if c_slsqp < best_c:
            best_c = c_slsqp
            best_h = h_slsqp
    
    # Phase 3: Binary conversion + binary local search
    if time.time() < deadline - 350:
        bl_deadline = min(time.time() + 200, deadline - 200)
        h_bin, c_bin = binary_local_search(best_h, n, bl_deadline, rng)
        c_bin_cont = compute_c5(h_bin, n)
        if c_bin_cont < best_c:
            best_c = c_bin_cont
            best_h = h_bin.copy()
        # Also try: L-BFGS-B starting from binary
        if time.time() < deadline - 200:
            h_relaxed = lbfgsb_optimize(h_bin, n, 200000, deadline - 180, maxiter=2000)
            c_relaxed = compute_c5(h_relaxed, n)
            if c_relaxed < best_c:
                best_c = c_relaxed
                best_h = h_relaxed
    
    # Phase 4: L-BFGS-B polish after binary
    if time.time() < deadline - 200:
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > deadline - 180:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, min(time.time()+25, deadline-160), maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Intensive fast local search
    if time.time() < deadline - 100:
        ls_deadline = min(time.time() + 300, deadline - 60)
        h_opt, c_opt = fast_local_search(best_h, n, ls_deadline, topK=6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Perturbation + reoptimize cycle
    if time.time() < deadline - 150:
        for pert_round in range(4):
            if time.time() > deadline - 100:
                break
            # Add structured perturbation
            noise_scale = 0.05 / (pert_round + 1)
            h_pert = best_h + rng.normal(0, noise_scale, n)
            h_pert = project_feasible(h_pert, n)
            
            # Quick L-BFGS-B
            if time.time() < deadline - 80:
                h_pert = lbfgsb_optimize(h_pert, n, 100000, min(time.time()+20, deadline-70), maxiter=800)
                c_pert = compute_c5(h_pert, n)
                if c_pert < best_c:
                    best_c = c_pert
                    best_h = h_pert
                # Quick local search
                if time.time() < deadline - 60:
                    h_ls, c_ls = fast_local_search(h_pert, n, min(time.time()+30, deadline-50), topK=5)
                    if c_ls < best_c:
                        best_c = c_ls
                        best_h = h_ls
    
    # Phase 7: Try higher resolution
    for n_try in [700, 1000]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - 60 - time.time()
        if remaining < 30:
            break
        h_interp = expand_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 200000, min(time.time()+20, deadline-50), maxiter=1500)
        h_opt = lbfgsb_optimize(h_opt, n_try, 5000000, min(time.time()+15, deadline-40), maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Final SLSQP
    if time.time() < deadline - 40:
        h_slsqp = slsqp_optimize(best_h, len(best_h), 1000000, deadline - 25, maxiter=200)
        c_slsqp = compute_c5(h_slsqp, len(best_h))
        if c_slsqp < best_c:
            best_c = c_slsqp
            best_h = h_slsqp
    
    # Phase 9: Final local search
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt, c_opt = fast_local_search(best_h, n_final, deadline - 2, topK=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    if best_h is None or len(best_h) == 0:
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    # Verify sum constraint precisely
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_feasible(best_h, n_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
```
