import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    # Bisection on lambda for sum(clip(h-lambda, 0, 1)) = target
    lo = min(h.min() - 1.0, -1.0)
    hi = max(h.max() + 1.0, 1.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        interior = np.where((h_new > 1e-14) & (h_new < 1.0 - 1e-14))[0]
        if len(interior) > 0:
            h_new[interior] += diff / len(interior)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def weighted_grad(h, n, c_vec, weights):
    """Compute weighted sum of gradients of c[s] for given weights."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, w in zip(range(len(c_vec)), weights):
        if w < 1e-15:
            continue
        s = idx - (n - 1)
        wm = w * dx
        if s >= 0:
            nv = n - s
            grad[0:nv] += wm * (1.0 - h[s:n])
            grad[s:n] += wm * (-h[0:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def nesterov_optimize(h, n, deadline, alpha_start=100, alpha_end=1000000, 
                       n_outer=50, beta=0.7, eta0=0.01, eta_min=1e-8):
    """Nesterov-accelerated proximal gradient with softmax smoothing."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    
    # Warm start
    m = np.zeros(n)
    h_curr = best_h.copy()
    eta = eta0
    
    for outer in range(n_outer):
        if time.time() > deadline - 5:
            break
        # Anneal alpha
        frac = outer / max(1, n_outer - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        c_vec = compute_c5_vec(h_curr, n)
        c_max = c_vec.max()
        sh = alpha * (c_vec - c_max)
        sh_max = sh.max()
        w = np.exp(sh - sh_max)
        w /= w.sum()
        
        # Gradient
        g = weighted_grad(h_curr, n, c_vec, w)
        
        # Nesterov momentum
        m = beta * m + g
        
        # Backtracking line search
        f_curr = float(w @ c_vec)  # soft-max objective (approx)
        step = eta
        found = False
        for _ in range(20):
            h_trial = project_feasible(h_curr - step * m, n)
            c_trial_vec = compute_c5_vec(h_trial, n)
            c_trial = c_trial_vec.max()
            if c_trial < best_c - 1e-16:

                best_h = h_trial.copy()
                h_curr = h_trial
                found = True
                break

                h_curr = h_trial
                found = True
                break
            step *= 0.5
            if step < eta_min:
                break
        
        if found:
            eta = min(eta * 1.05, 0.05)
        else:
            eta *= 0.7
    
    return best_h, best_c

    target = n / 2.0

def pair_descent_efficient(h, n, deadline, radius=20):
    """Efficient pair descent: only try pairs near boundaries and critical indices."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    target = n / 2.0
    
    # Identify "interesting" indices: near 0 or 1, or at boundaries
    interesting = np.where((best_h < 0.3) | (best_h > 0.7))[0]
    if len(interesting) > 200:
        interesting = np.where((best_h < 0.1) | (best_h > 0.9))[0]
    
    deltas = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in interesting:
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try moving mass from i to j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    lo = min(float(h.min()) - 1.0, -1.0)
    hi = max(float(h.max()) + 1.0, 1.0)

        val = np.clip(h - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        interior = np.where((h_new > 1e-14) & (h_new < 1.0 - 1e-14))[0]
        if len(interior) > 0:
            h_new[interior] += diff / len(interior)
            h_new = np.clip(h_new, 0.0, 1.0)

def weighted_grad(h, n, weights):

    grad = np.zeros(n)
    for idx in range(len(weights)):
        wi = weights[idx]
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        wm = wi * dx
        if s >= 0:
            nv = n - s
            grad[0:nv] += wm * (1.0 - h[s:n])
            grad[s:n] += wm * (-h[0:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    h1, c1 = nesterov_phase(best_h, n, p1_end,
                            alpha_start=50, alpha_end=200000,
                            n_iters=5000, beta=0.7, eta0=0.03)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: L-BFGS with multiple alphas
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 200, deadline - 300)
        h2, c2 = lbfgs_phase(best_h, n, p2_end,
                             alphas=[2000, 10000, 50000, 200000, 500000, 2000000],
                             maxiter_list=[2000, 3000, 3000, 4000, 4000, 4000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Active-shift focused
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 100, deadline - 200)
        h3, c3 = active_shift_phase(best_h, n, p3_end, top_k=6,
                                    alphas=[10000, 100000, 500000, 2000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Second Nesterov with higher alpha (sharper)
    if time.time() < deadline - 120:
        p4_end = min(time.time() + 80, deadline - 120)
        h4, c4 = nesterov_phase(best_h, n, p4_end,
                                alpha_start=50000, alpha_end=5000000,
                                n_iters=3000, beta=0.8, eta0=0.008)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Perturbation restarts
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 60, deadline - 80)
        h5, c5 = perturbation_restart(best_h, n, p5_end, rng, n_restarts=4)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: SLSQP polish
    if time.time() < deadline - 50:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 40:
                break
            h6, c6 = slqp_polish(best_h, n, min(time.time() + 20, deadline - 35), alpha, 1000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Fine coordinate descent
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 15, deadline - 15)
        h7, c7 = fine_coord_descent(best_h, n, p7_end)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Pair descent (limited)
    if time.time() < deadline - 3:
        h8, c8 = pair_descent_limited(best_h, n, deadline - 1, max_pairs=8000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
