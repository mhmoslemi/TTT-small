        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()

            if w[idx] < 1e-12:
                continue
            k = idx - (n - 1)
            dx = 2.0 / n
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h_ylm1[k:n])
                grad[k:n] += wm * (-h_ylm1[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h_ylm1[0:nv])
                grad[0:nv] += wm * (-h_ylm1[kk:n])
        
        # Nesterov momentum
        m_vec = momentum * m_vec - lr * grad
        h_new = project_feasible(h_curr + m_vec, n)
        
        # Update best
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

        # Decrease learning rate
        lr = lr_init / np.sqrt(t + 1)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def shift_grad(h, n, lag_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
        if nv > 0 and k < n:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smoothing_obj(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha


def smoothing_grad(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad


def cutting_plane_opt(h, n, deadline, max_iter=400, max_active=25, step_frac=1.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_all(h, n)
        curr_max = c_vec.max()
        top_idx = np.argsort(-c_vec)
        active = []
        for idx in top_idx[:max_active]:
            if c_vec[int(idx)] >= curr_max - 0.003 * max(1e-8, curr_max):
                active.append(int(idx))
        if not active:
            active = [int(top_idx[0])]
        n_act = len(active)
        grads = np.array([shift_grad(h, n, lag) for lag in active])
        c_vals = c_vec[active]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.zeros((n_act + 2 * n, n_var))
        b_ub = np.zeros(n_act + 2 * n)
        row = 0
        for j in range(n_act):
            A_ub[row, 0:n] = grads[j]
            A_ub[row, n] = -1.0
            b_ub[row] = c_vals[j] - np.dot(grads[j], h)
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = 0.0
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[0:n]
                for frac in [step_frac, 0.7, 0.4, 0.2]:
                    h_try = h + frac * (h_lp - h)
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_h = h_try.copy()
                        h = h_try
                        break
                if abs(best_c - compute_c5(h, n)) < 1e-12:
                    h = best_h.copy()
        except Exception:
            pass
    return best_h, best_c


def lbfgs_opt(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 500, 1500, 5000, 15000, 50000]
    h_curr = project_feasible(h, n)
    best_c = compute_c5(h_curr, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smoothing_obj, h_curr, jac=smoothing_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-17,
                                   'gtol': 1e-15, 'maxfun': 40000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h_curr = h_new
        except Exception:
            pass
    return h_curr, best_c


def nesterov_subgrad(h, n, deadline, alpha=800, lr_init=0.003, momentum=0.85):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    m_vec = np.zeros(n)
    h_old = h.copy()
    for t in range(20000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_all(h_curr, n)
        m_val = c_vec.max()
        sh = alpha * (c_vec - m_val)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in range(2 * n - 1):
            if w[idx] < 1e-12:
                continue
            k = idx - (n - 1)
            wm = w[idx] * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - h_curr[k:n])
                    grad[k:n] += wm * (-h_curr[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - h_curr[0:nv])
                    grad[0:nv] += wm * (-h_curr[kk:n])
        m_vec = momentum * m_vec - lr * grad
        h_new = project_feasible(h_curr + m_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_old = h_curr.copy()
        beta = (1 + np.sqrt(1 + 4.0 * t * t)) / 2.0
        y_next = h_new + (t - 1) / beta * (h_new - h_old)
        h_curr = project_feasible(y_next, n)
        h_old = h_new
        lr = lr_init / np.sqrt(t + 1.0)
    return best_h, best_c


def pairwise_swap_search(h, n, deadline, radius=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                candidates = np.arange(lo, hi)
                np.random.shuffle(candidates)
                for j in candidates:
                    if i == j:

                    if best_h[i] + step <= 1.0 + 1e-12 and best_h[j] - step >= -1e-12:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    if best_h[i] - step >= -1e-12 and best_h[j] + step <= 1.0 + 1e-12:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
    return best_h, best_c

def als_iteration(h, n, deadline, max_iter=1000):
    """Alternating direction: fix lambda, solve QP approximately via CD, update lambda."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_all(h, n)
    m_val = c_vec.max()
    alpha_l = 500.0
    sh = alpha_l * (c_vec - m_val)
    mx = sh.max()
    lam = np.exp(sh - mx)
    lam /= lam.sum()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Solve max h^T Q h - linear s.t. 0<=h<=1, sum=n/2
        # via coordinate descent with momentum
        for coord in range(n):
            if time.time() > deadline:
                return best_h, best_c
            h_save = h[coord]
            # Compute gradient component at coord
            grad_c = np.zeros(n)
            for idx in range(2 * n - 1):
                if lam[idx] < 1e-12:
                    continue
                k = idx - (n - 1)
                dx = 2.0 / n
                wml = lam[idx] * dx
                if k >= 0:
                    nv = n - k
                    if 0 <= coord < nv:
                        grad_c[coord] += wml * (1.0 - h[k + coord] - h[coord] * 2.0)
                    if k <= coord < n:
                        grad_c[coord] += wml * (-h[coord - k])
                else:
                    kk = -k
                    nv = n - kk
                    if kk <= coord < n:
                        grad_c[coord] += wml * (1.0 - h[coord - kk] - h[coord] * 2.0)
                    if 0 <= coord < nv:
                        grad_c[coord] += wml * (-h[coord + kk])
            # Move in negative gradient direction (we want to minimize, not maximize)
            # Actually we want to minimize sum lam_k f_k, which is linear - quadratic
            # The gradient of sum lam_k f_k w.r.t h[coord] is what we computed
            step = 0.01
            h_new_val = np.clip(h_save - step * grad_c[coord], 0.0, 1.0)
            # Renormalize approximately
            diff = h_new_val - h_save
            if abs(diff) > 1e-12:
                h[coord] = h_new_val
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
        # Update lambda

        m_val = c_vec.max()
        sh = alpha_l * (c_vec - m_val)
        mx = sh.max()
        lam_new = np.exp(sh - mx)
        lam_new /= lam_new.sum()
        lam = 0.7 * lam + 0.3 * lam_new
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_curr = n_init

    # Phase 1: Threshold + cutting plane on initial (100s)
    t1 = min(t0 + 100, deadline - 600)
    if t1 > time.time() + 10:
        best_h, best_c = threshold_try(best_h, n_curr, min(time.time() + 10, t1))
        best_h, best_c = cutting_plane_opt(best_h, n_curr, t1, max_iter=200, max_active=25, step_frac=0.8)

    # Phase 2: L-BFGS smooth (80s)
    t2 = min(t0 + 180, deadline - 500)
    if t2 > time.time() + 10:
        best_h, best_c = lbfgs_opt(best_h, n_curr, t2, alphas=[300, 1000, 3000, 10000])

    # Phase 3: Cutting plane round 2 (80s)
    t3 = min(t0 + 260, deadline - 420)
    if t3 > time.time() + 10:
        best_h, best_c = cutting_plane_opt(best_h, n_curr, t3, max_iter=200, max_active=30, step_frac=1.0)

    # Phase 4: Pairwise swap local search (80s)
    t4 = min(t0 + 340, deadline - 350)
    if t4 > time.time() + 10:
        best_h, best_c = pairwise_swap_search(best_h, n_curr, t4, radius=50)

    # Phase 5: Nesterov subgradient (60s)
    t5 = min(t0 + 400, deadline - 300)
    if t5 > time.time() + 10:
        h5, c5 = nesterov_subgrad(best_h, n_curr, t5, alpha=1000, lr_init=0.002)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Cutting plane round 3 (60s)
    t6 = min(t0 + 460, deadline - 250)
    if t6 > time.time() + 10:
        best_h, best_c = cutting_plane_opt(best_h, n_curr, t6, max_iter=150, max_active=20, step_frac=0.6)

    # Phase 7: L-BFGS polish (50s)
    t7 = min(t0 + 510, deadline - 200)
    if t7 > time.time() + 10:
        best_h, best_c = lbfgs_opt(best_h, n_curr, t7, alphas=[5000, 20000, 80000])

    # Phase 8: Pairwise swap fine (50s)
    t8 = min(t0 + 560, deadline - 150)
    if t8 > time.time() + 10:
        best_h, best_c = pairwise_swap_search(best_h, n_curr, t8, radius=30)

    # Phase 9: Try upsample to higher resolution
    n_up = 800
    t9_start = min(time.time() + 5, deadline - 120)
    if t9_start > time.time() + 10 and n_up > n_curr:
        x_old = np.linspace(0, 2.0, n_curr, endpoint=False)
        x_new = np.linspace(0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up < best_c + 0.002:
            t_lo = min(time.time() + 30, deadline - 60)
            if t_lo > time.time() + 5:
                h_up, c_up = lbfgs_opt(h_up, n_up, t_lo, alphas=[500, 2000, 8000])
            t_cp = min(time.time() + 25, deadline - 30)
            if t_cp > time.time() + 5:
                h_up, c_up = cutting_plane_opt(h_up, n_up, t_cp, max_iter=80, max_active=20, step_frac=0.7)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                n_curr = n_up

    # Phase 10: Try a random 0-1 restart at n_curr
    if time.time() < deadline - 40:
        h_rand = random_01_init(n_curr, rng)
        h_rand = project_feasible(h_rand, n_curr)
        c_rand = compute_c5(h_rand, n_curr)
        t_rs = min(time.time() + 30, deadline - 15)
        if t_rs > time.time() + 5:
            h_rand, c_rand = cutting_plane_opt(h_rand, n_curr, t_rs, max_iter=60, max_active=20, step_frac=0.7)
            if c_rand < best_c:
                best_c = c_rand
                best_h = h_rand

    # Phase 11: Final local search
    if time.time() < deadline - 5:
        best_h, best_c = pairwise_swap_search(best_h, n_curr, deadline - 3, radius=20)

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
