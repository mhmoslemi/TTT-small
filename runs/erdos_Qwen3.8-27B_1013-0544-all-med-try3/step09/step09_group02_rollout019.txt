import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_c_k(h, n, k_offset):
    """Gradient of c_k with respect to h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = k_offset - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def active_set_optimize(h, n, deadline, max_iter=5000, top_k=8):

    Active set optimization: focus on top_k shifts, solve small LP for step.
    """
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Get top shifts
        top_idx = np.argsort(-c_vec)[:top_k]

        grads = np.array([gradient_c_k(h, n, idx) for idx in top_idx])
        c_vals = c_vec[top_idx]

        # subject to: 0 ≤ h+delta ≤ 1, sum(delta) = 0, ||delta||∞ ≤ rho
        # LP: min t s.t. grads[j]·delta + c_vals[j] ≤ t, constraints
        
        rho = 0.03
        n_var = n + 1  # delta_0,...,delta_{n-1}, t
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0  # minimize t
        
        # Inequality constraints: grads[j]·delta - t ≤ -c_vals[j] for each j
        # Bounds: -h[i] ≤ delta[i] ≤ 1-h[i]
        # Step bound: -rho ≤ delta[i] ≤ rho
        # Equality: sum(delta) = 0
        
        n_ineq = top_k + 4 * n
        A_ub = np.zeros((n_ineq, n_var))
        b_ub = np.zeros(n_ineq)
        row = 0
        
        for j in range(top_k):

            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[j]
            row += 1

            b_ub[row] = h[i]
            row += 1
        # delta[i] ≤ 1-h[i]
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1

        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        # delta[i] ≤ rho
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        
        # Tighter bounds
        bounds = [(0.0, None)] * n + [(None, None)]
        
        try:

Coordinate descent idea:
- For each i, the "most impactful" change is to adjust h[i] in the direction that reduces the max c_k
- The direction is determined by the active constraints
- We can analytically compute the optimal h[i] given all other h[j] fixed

For fixed h[j] (j≠i), c_k as a function of h[i] is:
c_k = ∑_l h_l(1-h_{l+k})dx
The terms involving h[i]:
- If i is in the valid range for shift k: h_i(1-h_{i+k})dx appears (when the first index is i)
- Also: h_{i-k}(1-h_i)dx appears (when the second index is i, i.e., l+k=i, l=i-k)

So c_k(h_i) = [h_i * (1-h_{i+k}) - h_{i-k} * h_i] * dx + const
= h_i * [(1-h_{i+k}) - h_{i-k}] * dx + const

This is LINEAR in h_i! (For fixed other coordinates.)

So the optimal h_i (to minimize max_k c_k) is:
- For each active k, c_k is linear in h_i
- We want to minimize max_k (a_k * h_i + b_k) where a_k and b_k are constants
- This is a 1D convex optimization (min of max of lines)
- Solution: find the h_i where the maximum of the lines is minimized
- This is a classic "minimax" or "Chebyshev approximation" problem in 1D

This is very efficient! We can solve it in O(m log m) where m is the number of active constraints.

def coordinate_descent(h, n, deadline, top_k=10, sweeps=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(sweeps):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Get top shifts
        top_idx = np.argsort(-c_vec)[:top_k]
        
        # Random order for coordinate descent
        order = np.random.permutation(n)
        
        for idx in order:
            i = int(idx)
            if time.time() > deadline:
                return h, best_c
            
            # Compute c_k as linear function of h[i] for each active k
            # c_k = a_k * h_i + b_k
            # where a_k = contribution coefficient of h_i to c_k
            # and b_k = c_k with h_i set to 0 (or some base)
            
            a_k = np.zeros(top_k)
            b_k = np.zeros(top_k)
            
            for jj in range(top_k):
                k = top_idx[jj] - (n - 1)
                # c_k = sum_l h_l (1-h_{l+k}) dx
                # Terms with h[i]:
                # l = i: h[i](1-h[i+k])dx, valid if 0≤i<n and 0≤i+k<n
                # l+k = i → l = i-k: h[i-k](1-h[i])dx, valid if 0≤i-k<n and 0≤i<n
                coeff = 0.0
                const = 0.0
                
                # h[i] appears in term l=i: h[i](1-h[i+k])
                if 0 <= i < n and 0 <= i+k < n:
                    coeff += (1 - h[i+k])
                # h[i] appears in term l=i-k: h[i-k](1-h[i])
                if 0 <= i-k < n and 0 <= i < n:
                    coeff += (-h[i-k])
                
                a_k[jj] = coeff * dx
                # b_k is c_k with h_i contribution removed
                b_k[jj] = c_vec[top_idx[jj]] - h[i] * a_k[jj]
            
            # Now minimize max_j (a_k[j] * x + b_k[j]) over x ∈ [0,1]
            # This is a 1D minimax problem
            x_opt = _minimax_1d(a_k, b_k, 0.0, 1.0)
            
            new_val = x_opt
            if abs(new_val - h[i]) > 1e-14:
                h[i] = new_val
                # Check improvement
                # (skip full recompute for speed, do it at end of sweep)
        
        # End of sweep: check
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
        # Also fix the sum constraint
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

def _minimax_1d(a, b, lo, hi):
    """Minimize max_j (a[j]*x + b[j]) over x in [lo, hi]"""
    # The function f(x) = max_j (a[j]*x + b[j]) is convex in x
    # Minimum occurs where the active constraints cross, or at boundary
    # Use bisection on the derivative (subgradient)
    
    def f(x):
        return np.max(a * x + b)
    
    # Check boundaries
    f_lo = f(lo)
    f_hi = f(hi)
    
    # Subgradient at x: set of {a[j] : j in argmax}
    # Find x where 0 is in the subgradient
    # Or use ternary search (since f is convex)
    
    # Ternary search
    for _ in range(100):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2
    return np.clip(x_opt, lo, hi)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_c_k(h, n, k_offset):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = k_offset - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def minimax_1d(a, b, lo, hi):
    """Minimize max_j (a[j]*x + b[j]) over x in [lo, hi] via ternary search."""
    lo = max(lo, 0.0)
    hi = min(hi, 1.0)
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        vals1 = np.max(a * m1 + b)
        vals2 = np.max(a * m2 + b)
        if vals1 < vals2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coordinate_descend(h, n, deadline, top_k=12, max_sweeps=50000):
    """Coordinate descent with 1D minimax subproblems."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Get top shifts (sorted by value descending)
        top_idx = np.argsort(-c_vec)[:top_k]
        
        # Process coordinates in order of impact
        # Compute gradient magnitudes for ordering
        # (skip for speed - use random order)
        order = np.random.permutation(n)
        
        for idx in order:
            i = int(idx)
            
            # Compute c_k as linear function of h[i]: c_k = a_k * x + b_k
            a_k = np.empty(top_k)
            b_k = np.empty(top_k)
            
            for jj in range(top_k):
                k = int(top_idx[jj]) - (n - 1)
                coeff = 0.0
                # h[i] in term l=i: h[i]*(1-h[i+k])*dx if valid
                if 0 <= i + k < n:
                    coeff += (1.0 - h[i + k])
                # h[i] in term l=i-k: h[i-k]*(1-h[i])*dx if valid
                if 0 <= i - k < n:
                    coeff += (-h[i - k])
                a_k[jj] = coeff * dx
                b_k[jj] = c_vec[int(top_idx[jj])] - h[i] * a_k[jj]
            
            # Solve 1D minimax
            x_opt = minimax_1d(a_k, b_k, 0.0, 1.0)
            
            if abs(x_opt - h[i]) > 1e-12:
                h[i] = x_opt
        
        # Project to feasible set (maintain sum constraint)
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Try a perturbation to escape
                n_pert = np.random.randint(1, 10)
                for _ in range(n_pert):
                    pi = np.random.randint(n)
                    h[pi] = np.clip(h[pi] + np.random.randn() * 0.02, 0, 1)
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    h = best_h.copy()
                    break
    
    return best_h, best_c

def lp_active_step(h, n, top_idx, c_vec, rho):
    """Solve LP to find best step reducing active constraints."""
    top_k = len(top_idx)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    n_ineq = top_k + 4 * n
    A_ub = np.zeros((n_ineq, n_var))
    b_ub = np.zeros(n_ineq)
    row = 0
    
    for jj in range(top_k):
        g = gradient_c_k(h, n, int(top_idx[jj]))
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[int(top_idx[jj])]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[0:n]
    except Exception:
        pass
    return None

def lp_polish(h, n, deadline, max_iter=2000, top_k=6, rho_init=0.02):
    """LP-based active set polishing."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        top_idx = np.argsort(-c_vec)[:top_k]
        
        delta = lp_active_step(h, n, top_idx, c_vec, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        
        if np.max(np.abs(delta)) < 1e-13:
            break
        
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            if rho < 0.04:
                rho = min(rho * 1.3, 0.06)
        else:
            stall += 1
            rho *= 0.6
            if rho < 1e-9:
                break
            if stall > 50:
                break
    
    return best_h, best_c

def smooth_obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_phase(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu, maxit in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj_fn, h, jac=smooth_grad_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-15, 'maxfun': maxit * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:

def pair_swap_search(h, n, deadline, top_k=8):
    """Targeted pair-swap local search preserving sum automatically."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            
            # Find high-impact coordinates
            # A coordinate i affects shifts where i or i+k is in range
            impact = np.zeros(n)
            for jj in range(top_k):
                k = int(np.argsort(-c_vec)[jj]) - (n - 1)
                if k >= 0:
                    impact[0:n-k] += 1
                    if n-k > 0:
                        impact[k:n] += 1
                else:
                    kk = -k
                    impact[kk:n] += 1
                    impact[0:n-kk] += 1
            impact /= top_k
            
            top_coords = np.argsort(-impact)[:min(60, n)]
            
            for i_val in top_coords:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                # Try swapping with j that's less impactful
                j_candidates = np.argsort(impact)[:min(30, n)]
                for j_val in j_candidates:
                    j = int(j_val)
                    if i == j:
                        continue
                    
                    # Try h[i]+step, h[j]-step
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    
                    # Try h[i]-step, h[j]+step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:

                        h[j] += step
                        c = compute_c5(h, n)

                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step

    return best_h, best_c

def slqp_final(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, nn, aa):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = aa * (c - m)
        mx = sh.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / aa)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / aa)
    
    def eq_fun(x, nn):
        return float(x.sum() - nn / 2.0)
    
    def eq_jac(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
