Let me implement the FFT gradient. The key insight:

grad[i] = dx * ∑_s w[s+n-1] * [(1-h[i+s])*valid1 - h[i-s]*valid2]

If I ignore the validity masks (i.e., treat it as a circular convolution), I get:
grad_circ[i] = dx * ∑_s w[s+n-1] * [(1-h[i+s mod n]) - h[(i-s) mod n]]

This is a circular cross-correlation of w with (1-h) minus circular cross-correlation of w with h.

The validity masks only matter at the boundaries. For large n, the boundary effects are small (O(1) out of O(n) terms). So the circular approximation is good for most of the optimization, and I can switch to the exact computation for the final evaluations.

But actually, for the optimization to be accurate, I need the exact gradient. The boundary effects, while small per-point, accumulate over many iterations and could lead to a slightly suboptimal solution.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_loss_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for s in range(n):
        if s == 0:
            grad += w[center] * (1.0 - 2.0 * x) * dx
        else:
            grad[0:n-s] += (w[center + s] * (1.0 - x[s:n]) - w[center - s] * x[s:n]) * dx
            grad[s:n] += (w[center - s] * (1.0 - x[0:n-s]) - w[center + s] * x[0:n-s]) * dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu_base=1000.0, maxiter_per=3000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_loss_fn, h.copy(), jac=smooth_grad_fn,
                          args=(n, alpha, mu_base), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxiter_per * 3, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_loss_fn, h, jac=smooth_grad_fn,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_swap_search(h, n, deadline, max_iters=200000):
    """Hill-climbing on binary swaps to reduce C5."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    if abs(h_b.sum() - target) > 1e-10:
        h_b = project_feasible(h_b, n)
        h_b = np.where(h_b > 0.5, 1.0, 0.0).astype(float)
    best_c = compute_c5(h_b, n)
    rng = np.random.RandomState(777)
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        if i1 == i0:
            continue
        h_b[i1] = 0.0
        h_b[i0] = 1.0
        c = compute_c5(h_b, n)
        if c < best_c - 1e-15:
            best_c = c
            ones = np.where(h_b == 1.0)[0]
            zeros = np.where(h_b == 0.0)[0]
        else:
            h_b[i1] = 1.0
            h_b[i0] = 0.0
    return h_b, best_c

def coordinate_refine(h, n, deadline, n_rounds=100):
    """Greedy single-coordinate improvements."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        # Try adjusting coordinates that affect the worst shift
        if k_worst >= 0:
            affected = list(range(0, n - k_worst)) + list(range(k_worst, n))
        else:
            kk = -k_worst
            affected = list(range(kk, n)) + list(range(0, n - kk))
        rng = np.random.RandomState(rnd + 1)
        rng.shuffle(affected)
        for i in affected:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002, 0.0001, -0.0001]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = best_h[i]
        if not improved and rnd > 20:
            break
    return best_h, best_c

def diverse_start(n, rng):
    """Generate a diverse starting pattern."""
    method = rng.randint(4)
    if method == 0:
        # Random binary
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif method == 1:
        # Structured: blocks of varying size
        h = np.zeros(n)
        pos = 0
        while pos < n:
            block_len = rng.randint(1, 5)
            for j in range(block_len):
                if pos + j < n:
                    h[pos + j] = 1.0 if (j % 2 == 0) else 0.0
            pos += block_len
        # Fix to have n/2 ones
        h = project_feasible(h, n)
    elif method == 2:
        # Smooth random
        t = np.linspace(0, 2, n)
        freqs = rng.choice([1, 2, 3, 5, 7, 11], 4)
        h = np.zeros(n)
        for f in freqs:
            ph = rng.uniform(0, 2 * np.pi)
            h += np.sin(f * t + ph) + 0.5
        h = np.clip(h / 4.0, 0, 1)
        h = project_feasible(h, n)
    else:
        # Alternating with noise
        h = np.tile([1.0, 0.0], n // 2 + 1)[:n]
        h += (rng.rand(n) - 0.5) * 0.3
        h = project_feasible(h, n)
    return h
