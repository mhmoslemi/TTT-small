def correlation_gradient_single(h, n, m):
    """Gradient of c[m] (the m-th full correlation value) w.r.t. h."""
    s = m - (n - 1)  # actual shift
    grad = np.zeros(n)
    if s >= 0:
        # c[m] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
        valid_i = slice(0, n - s)
        grad[s:] += (1 - h[valid_i])  # from h[i+s] term, j = i+s
        grad[valid_i] += -h[s:]  # from (1-h[i]) term, j = i
    else:
        ss = -s  # positive
        # c[m] = sum_{i=0}^{n-1-ss} h[i] * (1-h[i+ss])
        valid_i = slice(0, n - ss)
        grad[valid_i] += (1 - h[ss:])  # from (1-h[i+ss]) term, j = i
        grad[ss:] += -h[valid_i]  # from h[i] term, j = i+ss
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_grad_multi(h, n, active_shifts):
    """Compute weighted gradient for a set of active correlation shifts."""
    grad = np.zeros(n)
    for m in active_shifts:
        dm = m - (n - 1)
        s = abs(dm)
        if s == 0:
            # c = sum(h[i]*(1-h[i])) = sum(h) - sum(h^2), grad = 1 - 2*h
            grad += (1.0 - 2.0 * h)
        else:
            nv = n - s
            if dm >= 0:
                # k > 0: c = sum_{i=0}^{nv-1} h[i]*(1-h[i+s])
                # d/dh[j] = (1-h[j+s]) for j in [0,nv-1] + (-h[j-s]) for j in [s,n-1]

                grad[s:] += (-h[0:nv])

                grad[s:] += (1.0 - h[0:nv])
                grad[:nv] += (-h[s:n])
    return grad

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)

    if configs is None:

def coordinate_descent_fine(h, n, deadline, n_iter_max=50000):

def sim_annealing(h, n, deadline, t_init=0.01, t_min=0.0001, cooling=0.999):
    """Simple simulated annealing with 2-point swaps."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t = t_init
    
    while time.time() < deadline and t > t_min:
        # Pick random pair
        i, j = np.random.choice(n, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        if h[i] + delta >= 0 and h[i] + delta <= 1 and h[j] - delta >= 0 and h[j] - delta <= 1:
            old_c = compute_c5(h, n)
            h[i] += delta
            h[j] -= delta
            new_c = compute_c5(h, n)
            
            if new_c < old_c or np.random.random() < np.exp(-(new_c - old_c) / t):
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
        
        t *= cooling
    
    return best_h, best_c

        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # === Phase 4: Upsample to n=600 ===
    n600 = 600
    t4_end = min(t0 + 420, deadline - 250)
    if t4_end > time.time() + 20:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n600)
        h600 = np.interp(x_new, x_old, best_h)
        h600 = project_feasible(h600, n600)
        c600 = compute_c5(h600, n600)
        
        # Optimize at n=600
        t_mid = time.time() + (t4_end - time.time()) * 0.5
        h600, c600 = nesterov_pgd(h600, n600, t_mid,
                                   beta=0.9, eta_init=0.003, top_k=30)
        if time.time() < t4_end - 5:
            h600, c600 = lbfgs_optimize(h600, n600, t4_end - 5,
                                         configs=[(500, 100.0), (2000, 500.0), (20000, 5000.0)])
        
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = n600
        else:
            best_n = n_init
    
    # === Phase 5: Upsample to n=800 ===
    n800 = 800
    t5_end = min(t0 + 600, deadline - 200)
    if t5_end > time.time() + 30:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n800)
        h800 = np.interp(x_new, x_old, best_h)
        h800 = project_feasible(h800, n800)
        c800 = compute_c5(h800, n800)
        
        t_mid5 = time.time() + (t5_end - time.time()) * 0.4
        h800, c800 = nesterov_pgd(h800, n800, t_mid5,
                                   beta=0.9, eta_init=0.002, top_k=35)
        if time.time() < t5_end - 10:
            h800, c800 = lbfgs_optimize(h800, n800, t5_end - 10,
                                         configs=[(200, 50.0), (1000, 200.0), (5000, 2000.0),
                                                  (20000, 10000.0)])
        
        if c800 < best_c:
            best_c = c800
            best_h = h800
            best_n = n800
    
    # === Phase 6: Top-k refinement at best resolution ===
    n_cur = len(best_h)
    t6_end = min(t0 + 750, deadline - 150)
    if t6_end > time.time() + 15:
        h6, c6 = top_k_lbfgs(best_h, n_cur, t6_end, k=6, alphas=[5000, 50000, 500000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # === Phase 7: Nesterov PGD continuation ===
    t7_end = min(t0 + 880, deadline - 80)
    if t7_end > time.time() + 10:
        h7, c7 = nesterov_pgd(best_h, n_cur, t7_end,
                               beta=0.91, eta_init=0.0015, top_k=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # === Phase 8: SLSQP polish ===
    t8_end = min(t0 + 960, deadline - 40)
    if t8_end > time.time() + 10:
        h8, c8 = slqp_polish(best_h, n_cur, t8_end, alpha=100000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    if time.time() < deadline - 20:
        h8b, c8b = slqp_polish(best_h, n_cur, deadline - 15, alpha=500000)
        if c8b < best_c:
            best_c = c8b
            best_h = h8b
    
    # === Phase 9: Fine coordinate descent ===
    t9_end = deadline - 5
    if t9_end > time.time() + 5:
        h9, c9 = coordinate_descent_fine(best_h, n_cur, t9_end)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # === Final: also try at a slightly different resolution ===
    for n_try in [n_cur - 20, n_cur + 20, n_cur + 50]:
        if n_try < 50 or n_try > 2000:
            continue
        if time.time() > deadline - 3:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    
    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
