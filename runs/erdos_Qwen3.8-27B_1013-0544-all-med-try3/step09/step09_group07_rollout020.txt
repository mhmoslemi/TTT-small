import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 60,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

Let me try a different angle: what if I use **CVXPY** to solve a sequence of convex relaxations?

The problem min max_k c_k(h) subject to 0 ≤ h ≤ 1, sum(h) = n/2 is a **min-max** problem. It can be written as:

min t
s.t. c_k(h) ≤ t for all k
     0 ≤ h_i ≤ 1 for all i
     sum(h) = n/2

But c_k(h) = sum_i h[i] * (1-h[i+k]) is **bilinear** in h (not convex in h). So the problem is non-convex.

However, if we fix some variables, it becomes linear. This suggests a **benders decomposition** or **cutting plane** approach, but those are usually for the dual.

Alternatively, we can use **successive linear programming**:
1. Linearize c_k(h) around the current point h₀
2. Solve the resulting linear program
3. Update and repeat

The linearization of c_k at h₀:
c_k(h) ≈ c_k(h₀) + Σ_i ∂c_k/∂h[i] * (h[i] - h₀[i])

This gives a linear objective for each k, so the problem becomes:
min t
s.t. c_k(h₀) + Σ_i ∂c_k/∂h[i] * (h[i] - h₀[i]) ≤ t for all k
     0 ≤ h_i ≤ 1
     sum(h) = n/2

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 60,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def slp_solve(h_current, n, max_iters=80, deadline=None):
    """Successive Linear Programming: linearize the min-max problem and solve LPs."""
    h = h_current.copy()

    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break

        # Build the LP: min t s.t. A[k] @ h_var <= t + offset[k] for all k
        n_lags = 2 * n - 1
        A = np.zeros((n_lags, n))
        offset = np.zeros(n_lags)

        for k in range(n_lags):
            lag = k - (n - 1)
            if lag == 0:
                dc = 1.0 - 2.0 * h
                c_val = np.sum(h * (1.0 - h))
            elif lag > 0:
                dc = np.zeros(n)
                dc[0:n-lag] = (1.0 - h[lag:n])
                dc[lag:n] = (-h[0:n-lag])
                c_val = np.sum(h[0:n-lag] * (1.0 - h[lag:n]))
            else:
                idx = -lag
                dc = np.zeros(n)
                dc[idx:n] = (1.0 - h[0:n-idx])
                dc[0:n-idx] = (-h[idx:n])
                c_val = np.sum(h[idx:n] * (1.0 - h[0:n-idx]))
            A[k] = dc
            offset[k] = dc @ h - c_val

        t = cp.Variable()
        h_var = cp.Variable(n)

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0,
            A @ h_var - t <= offset
        ]

        prob = cp.Problem(cp.Minimize(t), constraints)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break

        if h_var.value is None:
            break

        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, n)

        if np.max(np.abs(h_new - h)) < 1e-11:
            h = h_new
            break
        h = h_new

    return h, compute_c5(h, n)

def gradient_local_search(h, n, deadline, rng, alpha=100000, max_rounds=50000):
    """Local search guided by the smooth-max gradient."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stuck = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        c_all = compute_all_lags(h, n)
        grad = smooth_max_grad(h, n, alpha)

        # Find the coordinate to decrease (most positive gradient) and increase (most negative)
        # Add some randomness to avoid cycling
        prob_dec = np.maximum(grad, 0)
        prob_dec += 1e-10
        prob_inc = np.maximum(-grad, 0)
        prob_inc += 1e-10

        if time.time() > deadline - 0.5:
            break

        # Pick with some randomness: mix of greedy and random
        r = rng.random()
        if r < 0.7:
            # Weighted random
            dec_probs = prob_dec / prob_dec.sum()
            inc_probs = prob_inc / prob_inc.sum()
            i = rng.choice(n, p=dec_probs)
            j = rng.choice(n, p=inc_probs)
        else:
            # Pure greedy
            i = np.argmax(grad)
            j = np.argmin(grad)

        if i == j:
            stuck += 1
            continue

        if grad[i] <= 0 and grad[j] >= 0:
            stuck += 1
            continue

        # Transfer mass from i to j
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-8:
            stuck += 1
            continue

        # Adaptive step size
        step_scale = 0.3 if stuck < 100 else (0.15 if stuck < 500 else 0.05)
        d = min(max_d * step_scale, 0.05)
        d = max(d, 1e-5)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)

        c_test = compute_c5(h, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h.copy()
            stuck = 0
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            stuck += 1

            if stuck > 5000:
                # Perturb and restart
                h = best_h + rng.normal(0, 0.02, n)
                h = project_feasible(h, n)
                stuck = 0

    return best_h, best_c

def iterated_refinement(h, n, deadline, rng, max_iters=20):
    """Iterate: perturb -> L-BFGS-B -> SLP, keep best."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for it in range(max_iters):
        if time.time() > deadline - 60:
            break

        # Perturb
        scale = 0.05 * (1.0 + 0.5 * np.cos(it * 0.3))
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)

        # L-BFGS-B
        alloc = min(20, (deadline - 60 - time.time()) * 0.3)
        if alloc < 5:
            break
        h_opt = lbfgsb_optimize(h_pert, n, 200000, time.time() + alloc, maxiter=1000)
        c_opt = compute_c5(h_opt, n)

        if c_opt < best_c:
            # Polish with SLP
            alloc_slp = min(30, deadline - 60 - time.time())
            if alloc_slp > 5:
                h_slp, c_slp = slp_solve(h_opt, n, max_iters=40, deadline=time.time() + alloc_slp)
                if c_slp < c_opt:
                    h_opt, c_opt = h_slp, c_slp
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c

    idx = rng.choice(n_init, n_init // 2, replace=False)
    h_rand[idx] = 1.0
    starts.append(h_rand)

    # Golden ratio pattern
    golden = (1 + np.sqrt(5)) / 2
    h_gr = np.zeros(n_init)
    for i in range(n_init):
        if ((i * golden) % 1.0) < 0.5:
            h_gr[i] = 1.0
    h_gr = project_feasible(h_gr, n_init)
    starts.append(h_gr)

    # Alternating blocks
    h_blk = np.zeros(n_init)
    bs = max(1, n_init // 8)
    for b in range(8):
        if b % 2 == 0:
            h_blk[b*bs:min((b+1)*bs, n_init)] = 1.0
    h_blk = project_feasible(h_blk, n_init)
    starts.append(h_blk)

    for h_s in starts:
        if time.time() > deadline - 300:
            break
        n_s = len(h_s)
        alloc = min(25, (deadline - 300 - time.time()) * 0.25)
        if alloc < 5:
            break
        h_opt = lbfgsb_optimize(h_s, n_s, 5000, time.time() + alloc, maxiter=600)
        c_opt = compute_c5(h_opt, n_s)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 2: L-BFGS-B with increasing alpha on best solution
    alpha_schedule = [2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 400:
            break
        remaining = deadline - 400 - time.time()
        if remaining < 10:
            break
        alloc = min(30, remaining * 0.25)
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + alloc, maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: SLP polish
    if time.time() < deadline - 250:
        alloc = min(60, deadline - 250 - time.time())
        h_slp, c_slp = slp_solve(best_h, n, max_iters=60, deadline=time.time() + alloc)
        if c_slp < best_c:
            best_c = c_slp
            best_h = h_slp

    # Phase 4: Gradient-guided local search
    if time.time() < deadline - 200:
        alloc = min(120, deadline - 200 - time.time())
        h_ls, c_ls = gradient_local_search(best_h, n, time.time() + alloc, rng, alpha=200000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 5: Lag-focused local search
    if time.time() < deadline - 150:
        alloc = min(100, deadline - 150 - time.time())
        h_ls2, c_ls2 = lag_focused_search(best_h, n, time.time() + alloc, rng)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2

    # Phase 6: SLP again after local search
    if time.time() < deadline - 100:
        alloc = min(50, deadline - 100 - time.time())
        h_slp2, c_slp2 = slp_solve(best_h, n, max_iters=50, deadline=time.time() + alloc)
        if c_slp2 < best_c:
            best_c = c_slp2
            best_h = h_slp2

    # Phase 7: Iterated refinement
    if time.time() < deadline - 80:
        alloc = min(150, deadline - 80 - time.time())
        h_iter, c_iter = iterated_refinement(best_h, n, time.time() + alloc, rng, max_iters=15)
        if c_iter < best_c:
            best_c = c_iter
            best_h = h_iter

    # Phase 8: Try higher resolution
    for n_try in [1000]:
        if time.time() > deadline - 120:
            break
        h_interp = expand_to_resolution(best_h, n, n_try)
        alloc1 = min(30, (deadline - 120 - time.time()) * 0.4)
        if alloc1 < 10:
            break
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, time.time() + alloc1, maxiter=2000)
        alloc2 = min(30, deadline - 120 - time.time())
        if alloc2 > 10:
            h_opt = slp_solve(h_opt, n_try, max_iters=30, deadline=time.time() + alloc2)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try

    # Phase 9: Final SLP + local search
    if time.time() < deadline - 30:
        alloc = min(40, deadline - 30 - time.time())
        h_final, c_final = slp_solve(best_h, n, max_iters=50, deadline=time.time() + alloc)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Phase 10: Final gradient local search
    if time.time() < deadline - 5:
        alloc = max(3, deadline - 5 - time.time())
        h_fs, c_fs = gradient_local_search(best_h, n, time.time() + alloc, rng, alpha=500000, max_rounds=10000)
        if c_fs < best_c:
            best_c = c_fs
            best_h = h_fs

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
