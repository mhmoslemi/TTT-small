import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def logsumexp_smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

    bounds = [(0.0, 1.0)] * n

        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:

        try:
            res = minimize(logsumexp_smooth_obj, h, jac=logsumexp_smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:

def generate_periodic(n, freq, phase=0.0, duty=0.5):
    """Generate a periodic binary pattern."""
    x = np.arange(n)
    t = (x / n * 2.0 * freq + phase) % 1.0
    h = (t < duty).astype(float)
    # Fix the count
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        order = np.argsort(-((t - duty) ** 2))
        count = 0
        for i in range(len(order)):
            if h[order[i]] == 0.0 and count < int(diff) + 1:
                h[order[i]] = 1.0
                count += 1
                if h.sum() >= target:
                    break
    elif diff < 0:
        order = np.argsort(-((t - duty) ** 2))
        count = 0
        for i in range(len(order)):
            if h[order[i]] == 1.0 and count < int(-diff) + 1:
                h[order[i]] = 0.0
                count += 1
                if h.sum() <= target:
                    break
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_run(h, n, deadline, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def to_binary(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
                diff -= 1.0
                if abs(diff) < 1e-12:
                    break
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
                diff += 1.0
                if abs(diff) < 1e-12:
                    break

def swap_search(h_bin, n, deadline, n_trials_per_round=80, max_stall=500):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    one_idx = np.where(h == 1.0)[0]
    zero_idx = np.where(h == 0.0)[0]
    rng = np.random.RandomState(42)
    stall = 0
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        found = False
        best_trial_c = best_c
        best_a, best_b = -1, -1
        
        for _ in range(n_trials_per_round):
            a = one_idx[rng.randint(len(one_idx))]
            b = zero_idx[rng.randint(len(zero_idx))]
            if a == b:
                continue
            h[a] = 0.0
            h[b] = 1.0
            c = compute_c5(h, n)
            if c < best_trial_c:
                best_trial_c = c
                best_a, best_b = a, b
            h[a] = 1.0
            h[b] = 0.0
        
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            best_c = best_trial_c
            one_idx = np.where(h == 1.0)[0]
            zero_idx = np.where(h == 0.0)[0]
            stall = 0
            found = True
        else:
            stall += 1
    
    return h, best_c

def ga_binary(n, deadline, pop_size=30, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    
    pop = np.zeros((pop_size, n))
    for i in range(pop_size):
        idx = rng.choice(n, size=target, replace=False)
        pop[i, idx] = 1.0
    
    c5s = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    
    while time.time() < deadline - 5:
        order = np.argsort(c5s)
        pop = pop[order]
        c5s = c5s[order]
        n_keep = pop_size // 2
        
        new_pop = np.zeros((pop_size, n))
        new_pop[:n_keep] = pop[:n_keep]
        
        for i in range(n_keep, pop_size):
            p1 = pop[rng.randint(0, n_keep)]
            p2 = pop[rng.randint(0, n_keep)]
            mask = rng.rand(n) < 0.5
            child = np.where(mask, p1, p2).astype(float)
            
            n_ones = int(child.sum())
            if n_ones != target:
                one_idx = np.where(child == 1.0)[0]
                zero_idx = np.where(child == 0.0)[0]
                if n_ones > target:
                    rem = rng.choice(one_idx, size=n_ones - target, replace=False)
                    child[rem] = 0.0
                    add = rng.choice(np.where(child == 0.0)[0], size=n_ones - target, replace=False)
                    child[add] = 1.0
                else:
                    add = rng.choice(zero_idx, size=target - n_ones, replace=False)
                    child[add] = 1.0
                    rem = rng.choice(np.where(child == 1.0)[0], size=target - n_ones, replace=False)
                    child[rem] = 0.0
            
            for _ in range(rng.randint(1, 4)):
                oi = np.where(child == 1.0)[0]
                zi = np.where(child == 0.0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    child[oi[rng.randint(len(oi))]] = 0.0
                    child[zi[rng.randint(len(zi))]] = 1.0
            
            new_pop[i] = child
        
        pop = new_pop
        c5s = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    
    best_i = np.argmin(c5s)
    return pop[best_i]

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=2000):
    def obj_fn(x, na, al):
        return lse_obj(x, na, al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def perturbation_optimize(h, n, deadline, n_trials=5, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    if mags is None:
        mags = [0.02, 0.04, 0.06, 0.01, 0.08]
    
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mag = mags[t % len(mags)]
        h_p = project_feasible(h + mag * (rng.rand(n) - 0.5), n)
        try:
            res = minimize(lse_obj, h_p, jac=lse_grad, args=(n, 5000),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 5000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
