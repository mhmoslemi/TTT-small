```python
def compute_grad(h, n, weights):
    """weights: (2n-1,) array of softmax weights for each shift"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(weights)):
        wi = weights[idx] * dx
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += wi * (1.0 - h[k:])
            grad[k:] += wi * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wi * (1.0 - h[:nv])
            grad[:nv] += wi * (-h[kk:])
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)

    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(active)):

        k = idx - (n - 1)

def slqp_obj(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)

    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 4))
        
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-16:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):

def coordinate_descent(h, n, deadline, max_rounds=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for s in steps:
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if best_h[i] >= s:
                    h_trial = best_h.copy()
                    h_trial[i] -= s
                    j = (i + 1) % n
                    while j == i:
                        j = (j + 1) % n
                    if h_trial[j] <= 1.0 - s:
                        h_trial[j] += s
                    h_trial = project_feasible(h_trial, n)
                    c_new = compute_c5(h_trial, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_trial
                        improved = True
                if best_h[i] <= 1.0 - s:
                    h_trial = best_h.copy()
                    h_trial[i] += s
                    j = (i + 1) % n
                    if h_trial[j] >= s:
                        h_trial[j] -= s
                    h_trial = project_feasible(h_trial, n)
                    c_new = compute_c5(h_trial, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_trial
                        improved = True
            if not improved:
                break
    return best_h, best_c

def pair_search(h, n, deadline, radius=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi):
                    if abs(i - j) < 3:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def subgradient_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.0005
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 50:
            order = np.argsort(-c_vec[active])
            active = active[order[:50]]
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
    
    return best_h, best_c

def thue_morse_start(n):
    h = np.zeros(n)
    hm = max(1, n.bit_length())
    length = 1 << hm
    tm = np.zeros(length, dtype=int)
    for i in range(1, length):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h[:n] = (tm[:n] % 2).astype(float)
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        idxs = np.where(h < 1.0)[0]
        for i in idxs:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff < 1e-12:
                break
    elif diff < 0:
        idxs = np.where(h > 0.0)[0]
        for i in idxs:
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return project_feasible(h, n)

def random_start(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.random(n)

    target = n / 2.0

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
