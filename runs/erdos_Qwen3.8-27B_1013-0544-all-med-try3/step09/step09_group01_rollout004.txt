from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    ncorr = len(c)  # 2n-1

        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    w /= s
    return w

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = sm_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = sm_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(active))
        w[np.argmax(vals)] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu, maxit_list, bounds):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        maxit = maxit_list[i] if i < len(maxit_list) else 2000
        maxit = min(maxit, max(500, int(remaining * 4)))
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_phase(h, n, deadline, top_k, alphas, mu, rounds):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            remaining = deadline - time.time()
            if remaining < 15:
                break
            maxit = max(800, int(remaining * 3))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def gradient_descent_polish(h, n, deadline, n_steps=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    for it in range(n_steps):
        if time.time() > deadline:
            break
        if (it % 500 == 0):
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
        else:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 15:
            active = np.array(np.argsort(-c_vec[active])[:15])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        for step_scale in [eta, eta * 0.5, eta * 0.1]:
            h_try = best_h - step_scale * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                best_c = c_try
                best_h = h_try
                eta = min(eta * 1.02, 0.1)
                break
        else:
            eta *= 0.95
        if eta < 1e-7:
            break
    return best_h, best_c

def coordinate_polish(h, n, deadline, n_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(12345)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        step = max(1e-6, 0.005 * (1 - r / n_rounds))
        for _ in range(5):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for sign in [1, -1]:
                di = sign * step
                dj = -sign * step
                if 0 <= best_h[i] + di <= 1 and 0 <= best_h[j] + dj <= 1:
                    h_try = best_h.copy()
                    h_try[i] += di
                    h_try[j] += dj
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-16:
                        best_c = c_try
                        best_h = h_try
    return best_h, best_c

def perturbation_restart(h, n, deadline, rng, n_tries=5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for t in range(n_tries):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.randn(n) * 0.01
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        bounds = [(0.0, 1.0)] * n
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = min(3000, max(1000, int(remaining * 2)))
        try:
            res = minimize(obj_full, h_pert, jac=grad_full,
                          args=(n, 50000, 0.005), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 5,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS at parent resolution (n=500)
    p1_end = t0 + 60
    bounds0 = [(0.0, 1.0)] * n0
    h1, c1 = lbfgs_phase(best_h, n0, p1_end,
                         alphas=[100, 1000, 10000, 100000],
                         mu=0.01,
                         maxit_list=[2000, 3000, 3000, 3000],
                         bounds=bounds0)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=2000 and coarse optimization
    n_main = 2000
    x_fine = np.linspace(0, 2, n_main)
    x_coarse = np.linspace(0, 2, best_n)
    h_main = np.interp(x_fine, x_coarse, best_h)
    h_main = project_feasible(h_main, n_main)
    c_main = compute_c5(h_main, n_main)

    p2_end = t0 + 220
    bounds_main = [(0.0, 1.0)] * n_main
    h2, c2 = lbfgs_phase(h_main, n_main, p2_end,
                         alphas=[50, 500, 5000, 50000],
                         mu=0.005,
                         maxit_list=[2000, 3000, 3000, 4000],
                         bounds=bounds_main)
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = n_main

    # Phase 3: Active-set heavy optimization at n=2000
    p3_end = t0 + 650
    h3, c3 = active_phase(best_h, best_n, p3_end,
                         top_k=10,
                         alphas=[50000, 200000, 1000000, 5000000],
                         mu=0.003, rounds=10)
    if c3 < best_c:
        best_c = c3
        best_h = h3
        best_n = best_n

    # Phase 4: Try higher resolution n=4000 if time permits
    elapsed = time.time() - t0
    if elapsed < budget_s - 350:
        n_hi = 4000
        x_hi = np.linspace(0, 2, n_hi)
        x_cur = np.linspace(0, 2, best_n)
        h_hi = np.interp(x_hi, x_cur, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)

        p4_end = min(t0 + 900, deadline - 150)
        bounds_hi = [(0.0, 1.0)] * n_hi
        h4, c4 = lbfgs_phase(h_hi, n_hi, p4_end,
                             alphas=[20, 200, 2000, 20000, 200000],
                             mu=0.003,
                             maxit_list=[1500, 2000, 2500, 2500, 3000],
                             bounds=bounds_hi)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_hi
        # Quick active-set on n_hi
        if time.time() < deadline - 120:
            p4b_end = min(time.time() + 150, deadline - 100)
            h4b, c4b = active_phase(best_h, best_n, p4b_end,
                                   top_k=8,
                                   alphas=[100000, 500000, 2000000],
                                   mu=0.002, rounds=5)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b
            best_n = len(best_h)

    # Phase 5: Gradient descent polish
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 60, deadline - 40)
        h5, c5 = gradient_descent_polish(best_h, best_n, p5_end, n_steps=8000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = len(best_h)

    # Phase 6: Perturbation restarts
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 50, deadline - 20)
        h6, c6 = perturbation_restart(best_h, best_n, p6_end, rng, n_tries=3)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = len(best_h)

    # Phase 7: Coordinate polish
    if time.time() < deadline - 5:
        p7_end = deadline - 2
        h7, c7 = coordinate_polish(best_h, best_n, p7_end, n_rounds=500)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = len(best_h)

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
