    dx = 2.0/n
    c = np.correlate(h, 1-h, 'full') * dx
    best_c = c.max()
    best_h = h.copy()
    
    while time.time() < deadline:
        improved = False
        # Get top-10 worst shifts
        worst_idx = np.argsort(-c)[:10]
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(max(0,i-radius), min(n, i+radius+1)):
                if i == j:
                    continue
                # Try moving 0.01 from j to i
                if h[i] < 1 and h[j] > 0:
                    h[i] += 0.01
                    h[j] -= 0.01
                    c_new = np.correlate(h, 1-h, 'full') * dx
                    max_new = c_new.max()
                    if max_new < best_c:

                        best_h = h.copy()
                        c = c_new
                        improved = True
                    else:

                        h[j] += 0.01
        if not improved:

    return best_h, best_c

def local_search_fast(h, n, deadline, radius=100, delta=0.01):
    dx = 2.0/n
    c = np.correlate(h, 1-h, 'full') * dx
    best_c = c.max()
    best_h = h.copy()
    best_c_val = best_c
    
    # Precompute affected index ranges for each position
    # For position p, affected m values:
    # lag = m - (n-1)
    # First factor (a=p): 0 <= p+lag < n -> -p <= lag <= n-1-p -> n-1-p <= m <= 2n-2-p
    # Second factor (b=p, a=p-lag): 0 <= p-lag < n -> p-n+1 <= lag <= p -> p <= m <= n-1+p
    # Union: [min(n-1-p, p), max(2n-2-p, n-1+p)]
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(max(0, i-radius), min(n, i+radius+1)):
                if i == j:
                    continue
                for sign in [1, -1]:
                    di = sign * delta
                    dj = -sign * delta
                    if h[i] + di < 0 or h[i] + di > 1:
                        continue
                    if h[j] + dj < 0 or h[j] + dj > 1:
                        continue
                    # Check if max correlation decreases
                    # Affected m values
                    m_lo_i = min(n-1-i, i)
                    m_hi_i = max(2*n-2-i, n-1+i)
                    m_lo_j = min(n-1-j, j)
                    m_hi_j = max(2*n-2-j, n-1+j)
                    m_lo = max(0, min(m_lo_i, m_lo_j))
                    m_hi = min(2*n-2, max(m_hi_i, m_hi_j))
                    
                    # Compute delta c for affected m
                    lag_vals = np.arange(m_lo, m_hi+1) - (n-1)
                    # Contribution from i:
                    # δ*(1-h[i+lag]) where 0<=i+lag<n
                    # -δ*h[i-lag] where 0<=i-lag<n
                    contrib_i = np.zeros(m_hi - m_lo + 1)
                    valid1_i = (0 <= i + lag_vals) & (i + lag_vals < n)
                    valid2_i = (0 <= i - lag_vals) & (i - lag_vals < n)
                    contrib_i[valid1_i] = di * (1 - h[i + lag_vals[valid1_i]])
                    contrib_i[valid2_i] += di * (-h[i - lag_vals[valid2_i]])
                    
                    # Similarly for j (with -di = dj)
                    contrib_j = np.zeros(m_hi - m_lo + 1)
                    valid1_j = (0 <= j + lag_vals) & (j + lag_vals < n)
                    valid2_j = (0 <= j - lag_vals) & (j - lag_vals < n)
                    contrib_j[valid1_j] = dj * (1 - h[j + lag_vals[valid1_j]])
                    contrib_j[valid2_j] += dj * (-h[j - lag_vals[valid2_j]])
                    
                    total_dc = contrib_i + contrib_j
                    new_affected = c[m_lo:m_hi+1] + total_dc * dx
                    
                    # Check if max decreases
                    new_max = max(new_affected.max(), 
                                  c[:m_lo].max() if m_lo > 0 else 0,
                                  c[m_hi+1:].max() if m_hi < 2*n-2 else 0)
                    
                    if new_max < best_c_val - 1e-15:
                        # Accept
                        h[i] += di
                        h[j] += dj
                        c[m_lo:m_hi+1] = new_affected
                        best_c_val = new_max
                        best_h = h.copy()
                        improved = True
                    # else: don't accept (since we didn't modify h, no revert needed)
    
    return best_h, best_c_val
