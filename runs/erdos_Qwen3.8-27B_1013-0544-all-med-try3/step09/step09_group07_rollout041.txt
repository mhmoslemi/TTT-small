import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    penalty_coeff = 1e10
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                                'maxls': 100, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate(h_src, n_src, n_dst):
    x_src = np.arange(n_src) / n_src * 2.0
    x_dst = np.arange(n_dst) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def lag_targeted_search(h, n, deadline, rng, num_lags=5, max_rounds=50000):

        improved = False

            m = abs(lag)
            if m >= n:
                continue

            # For lag > 0: c_m = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

            # for pairs where h[i] is high and h[i+m] is low

                    else:

                        h[j] -= d
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:

                    pass

def binary_round_relax(h, n, deadline, rng):
    """Round to binary, then relax back."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    
    # Optimize from binary
    h_relaxed = lbfgsb_optimize(h_bin, n, 1000000, min(deadline, time.time() + 30), maxiter=2000)
    return h_relaxed

def simulated_annealing_fast(h, n, deadline, rng, alpha_sa=50000, t_init=0.005, t_min=0.00002, cool=0.9999):
    """Faster SA with vectorized operations."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stagnant = 0
    
    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.5:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-3:]
            k_star = top_k[rng.randint(3)]
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                T *= cool
                continue
            i = rng.randint(0, n - abs(lag))
            j = i + abs(lag)
        
        if i == j:
            continue
        if h[i] <= h[j]:
            i, j = j, i
        
        d = rng.uniform(0.002, 0.05)
        d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
        
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            stagnant = 0
        else:
            # Check acceptance
            de = c_new - best_c
            ratio = -de / T if T > 0 else float('inf')
            if ratio > 0 and ratio < 20 and rng.random() < np.exp(-ratio):
                stagnant = 0
            else:
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                stagnant += 1
        
        T *= cool
        if stagnant > 5000:
            h = best_h + rng.normal(0, 0.01, n)
            h = project_feasible(h, n)
            stagnant = 0
            T = max(T, 0.003)
    
    return best_h, best_c

    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def gradient_at_lag(h, n, lag):
    """Compute gradient of the overlap at a specific lag."""
    dx = 2.0 / n
    if lag == 0:
        return (1.0 - 2.0 * h) * dx
    elif lag > 0:
        m = lag
        g = np.zeros(n)
        g[:n-m] += (1.0 - h[m:])
        g[m:] += (-h[:n-m])
        return g * dx
    else:
        m = -lag
        g = np.zeros(n)
        g[:n-m] += (1.0 - h[m:])
        g[m:] += (-h[:n-m])
        return g * dx

def active_set_subgradient(h, n, deadline, lr_init=0.005, decay=0.9995, epsilon=1e-6, max_stagnant=2000):
    """Subgradient method using the active set of near-max lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stagnant = 0
    
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        max_val = c_all.max()
        
        # Identify active set: lags within epsilon of max
        active_mask = c_all >= (max_val - epsilon)
        active_frac = active_mask.sum() / len(c_all)
        
        # Compute subgradient as weighted average of active lag gradients
        grad = np.zeros(n)
        num_active = max(1, int(active_mask.sum()))
        
        for k in range(2 * n - 1):
            if not active_mask[k]:
                continue
            lag = k - (n - 1)
            grad += gradient_at_lag(h, n, lag)
        grad /= num_active
        
        # Project to maintain sum constraint
        grad -= grad.mean()
        
        # Take step (minimize, so go against gradient)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            # Slightly increase lr if making progress
            lr = min(lr * 1.01, lr_init * 2)
        else:
            # Revert
            stagnant += 1
            if stagnant < 200:
                lr *= decay
            elif stagnant < max_stagnant:
                # Try smaller step
                h_try = h - lr * 0.1 * grad
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    stagnant = 0
                    lr *= 0.5
                else:
                    lr *= decay
            else:
                # Restart with noise
                h = best_h + np.random.RandomState(int(time.time()*1000) % 10000).normal(0, 0.008, n)
                h = project_feasible(h, n)
                stagnant = 0
                lr = lr_init
        
        if lr < 1e-12:
            break
    
    return best_h, best_c

def lbfgsb_optimize(h, n, alpha, maxiter=2000, time_limit=None):
    target_sum = n / 2.0
    penalty_coeff = 1e12
    t_start = time.time()
    
    def obj_and_grad(x):
        if time_limit is not None and time.time() - t_start > time_limit:
            raise Exception("time limit exceeded")
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                                'maxls': 100, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_to(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src) * 2.0 / n_src
    x_dst = np.arange(n_dst) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def targeted_pair_search(h, n, deadline, rng, num_lags=5, max_rounds=100000):
    """Targeted search: for each top lag, try structured pair modifications."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds = 0
    no_improve = 0
    
    while time.time() < deadline and rounds < max_rounds:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-num_lags:]
        
        round_improved = False
        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # For this lag, the overlap is sum h[i]*(1-h[i+m]) (or similar)
            # To reduce it, move mass from high h to low h at paired positions
            if lag > 0:
                # Pairs (i, i+lag), i in [0, n-1-lag)
                # We want to reduce h[i]*(1-h[i+lag])
                # Best: where h[i] is high and h[i+lag] is low
                contribs = h[:n-m] * (1.0 - h[m:n])
                # Try top contributing pairs with various step sizes
                top_pairs = np.argsort(contribs)[-15:]
                for pi in top_pairs:
                    if time.time() > deadline:
                        break
                    i = pi
                    j = pi + m
                    if h[i] > h[j] + 1e-7:
                        for d_frac in [0.3, 0.15, 0.07, 0.03]:
                            d = d_frac * (h[i] - h[j])
                            d = min(d, 0.05)
                            if d < 1e-8:
                                continue
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
                            c_test = compute_c5(h, n)
                            if c_test < best_c - 1e-15:
                                best_c = c_test
                                best_h = h.copy()
                                round_improved = True
                                break
                            else:
                                h[i] += d
                                h[j] -= d
                                h = project_feasible(h, n)
            else:
                m2 = -lag
                # Pairs (i, i+m2), i in [0, n-1-m2), but the overlap is h[i+m2]*(1-h[i])
                # To reduce: where h[i+m2] is high and h[i] is low
                contribs = h[m2:] * (1.0 - h[:n-m2])
                top_pairs = np.argsort(contribs)[-15:]
                for pi in top_pairs:
                    if time.time() > deadline:
                        break
                    i = pi
                    j = pi + m2
                    if h[j] > h[i] + 1e-7:
                        for d_frac in [0.3, 0.15, 0.07, 0.03]:
                            d = d_frac * (h[j] - h[i])
                            d = min(d, 0.05)
                            if d < 1e-8:
                                continue
                            h[j] -= d
                            h[i] += d
                            h = project_feasible(h, n)
                            c_test = compute_c5(h, n)
                            if c_test < best_c - 1e-15:
                                best_c = c_test
                                best_h = h.copy()
                                round_improved = True
                                break
                            else:
                                h[j] += d
                                h[i] -= d
                                h = project_feasible(h, n)
        
        rounds += 1
        if not round_improved:
            no_improve += 1
            if no_improve > 20:
                # Add perturbation
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                no_improve = 0
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, t_init=0.004, t_min=0.00001, cool=0.9999):
    """SA with focus on structured moves at worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stagnant = 0
    
    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-4:]
        
        r = rng.random()
        if r < 0.6:
            # Structured move at a top lag
            k_star = top_idx[rng.randint(4)]
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                T *= cool
                continue
            m = abs(lag)
            i = rng.randint(0, n - m)
            j = i + m
            if lag > 0:
                high, low = (i, j) if h[i] > h[j] else (j, i)
            else:
                high, low = (j, i) if h[j] > h[i] else (i, j)
            if h[high] - h[low] < 1e-7:
                T *= cool
                continue
            d = rng.uniform(0.002, 0.04)
            d = min(d, (h[high] - h[low]) * 0.4)
        else:
            # Random pair
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            high, low = (i, j) if h[i] > h[j] else (j, i)
            if h[high] - h[low] < 1e-7:
                T *= cool
                continue
            d = rng.uniform(0.001, 0.03)
            d = min(d, (h[high] - h[low]) * 0.3)
        
        h[high] -= d
        h[low] += d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            stagnant = 0
        else:
            ratio = (c_new - best_c) / T
            if ratio < 15 and rng.random() < np.exp(-ratio):
                stagnant = 0
            else:
                h = best_h if c_new > best_c else h
                h[low] -= d
                h[high] += d
                h = project_feasible(h, n)
                stagnant += 1
        
        T *= cool
        if stagnant > 3000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            stagnant = 0
            T = max(T, 0.002)
    
    return best_h, best_c

def binary_round_and_relax(h, n, deadline):
    """Round to binary, then relax back via L-BFGS-B."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    
    h_relaxed = lbfgsb_optimize(h_bin, n, 2000000, maxiter=5000,
                                  time_limit=max(5, deadline - time.time()))
    return h_relaxed
