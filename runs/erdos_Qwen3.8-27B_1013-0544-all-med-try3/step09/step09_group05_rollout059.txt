def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    """Log-sum-exp smooth max with constraint penalty."""
    c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return val

    m = c.max()

    w = np.exp(sh - mx)

    dx = 2.0 / n

    best_c = compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_single_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] = (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] = (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

def lse_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -200, 400)
    smx = sh.max()
    logsum = smx + np.log(np.sum(np.exp(sh - smx)))
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu, top_k=25):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -200, 400)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    if top_k < len(w):
        top_idx = np.argsort(-w)[:top_k]
        w_sub = w[top_idx].copy()
        w_sub /= w_sub.sum()
    else:
        top_idx = np.arange(len(w))
        w_sub = w
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w_sub, top_idx):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_phase(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu, top_k, maxiter in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu, top_k),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxiter * 15,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def mshift_refine(h, n, deadline, top_k=3, max_iter=80000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

            best_h = h.copy()
        threshold = max_val * (1.0 - 0.003)

        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]

            break
        g = np.zeros(n)

            g += shift_single_grad(h, n, idx)
        g /= len(active)
        improved = False
        for lr in [0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)

                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
