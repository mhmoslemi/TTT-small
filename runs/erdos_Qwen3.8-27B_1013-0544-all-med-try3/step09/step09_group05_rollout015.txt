    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):

        k = mm - (n - 1)

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

                h = h_try
                stagnant = 0
                break
        if best_impr == 0:
            stagnant += 1
            step *= 0.5
            if step < 1e-7 or stagnant > 30:
                break
    return best_h, best_c

def multi_start_evaluate(h_candidates, n, deadline, seed=0):
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    bounds = [(0.0, 1.0)] * n
    for h0 in h_candidates:
        if time.time() > deadline - 20:
            break
        h0 = project_feasible(h0, n)
        c0 = compute_c5(h0, n)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
        for alpha in [200, 2000, 20000]:
            if time.time() > deadline - 10:
                break
            try:
                res = minimize(obj_sm, h0, jac=grad_sm,
                              args=(n, alpha, 0.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxfun': 8000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                h0 = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick polish at original resolution
    if n_orig <= 600:
        t1 = min(t0 + 40, deadline - 500)
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            alphas=[100, 1000, 5000], mu=1.0, maxiter=3000)

    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_1000 = np.interp(x_new, x_old, best_h)
    h_1000 = project_feasible(h_1000, n)
    c_1000 = compute_c5(h_1000, n)
    if c_1000 < best_c:
        best_c = c_1000
    h_work = h_1000.copy()

    # Phase 3: Multi-stage L-BFGS at n=1000
    t3 = min(t0 + 220, deadline - 400)
    h_work, c_work = run_lbfgs(h_work, n, t3,
        alphas=[50, 200, 1000, 5000, 20000, 100000], mu=5.0, maxiter=4000)
    if c_work < best_c:
        best_c = c_work
        best_h = h_work.copy()

    # Phase 4: Top-k focused optimization
    if time.time() < deadline - 300:
        t4 = min(time.time() + 120, deadline - 300)
        h_work, c_work = run_topk_opt(h_work, n, t4, top_k=5)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 5: Penalty-based descent
    if time.time() < deadline - 250:
        current_c = compute_c5(h_work, n)
        targets = [current_c - 1e-4, current_c - 3e-4, current_c - 5e-4,
                   current_c - 8e-4, current_c - 1.2e-3]
        t5 = min(time.time() + 130, deadline - 250)
        h_pen, c_pen = penalty_descent(h_work, n, t5, current_c, targets)
        if c_pen < best_c:
            best_c = c_pen
            best_h = h_pen.copy()
            h_work = h_pen.copy()

    # Phase 6: Another round of top-k + L-BFGS
    if time.time() < deadline - 150:
        t6 = min(time.time() + 60, deadline - 150)
        h_work, c_work = run_topk_opt(h_work, n, t6, top_k=3)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
        if time.time() < deadline - 80:
            t6b = min(time.time() + 50, deadline - 80)
            h_work, c_work = run_lbfgs(h_work, n, t6b,
                alphas=[50000, 200000, 500000], mu=1.0, maxiter=3000)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

    # Phase 7: Gradient descent fine-tuning
    if time.time() < deadline - 80:
        t7 = min(time.time() + 50, deadline - 80)
        h_work, c_work = gradient_descent_fine(h_work, n, t7, base_step=0.003)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 8: Simulated annealing for exploration
    if time.time() < deadline - 60:
        t8 = min(time.time() + 50, deadline - 60)
        h_sa, c_sa = simulated_anneal(h_work, n, t8, seed=seed)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()

    # Phase 9: Final L-BFGS polish from best
    if time.time() < deadline - 10:
        t9 = min(time.time() + 40, deadline - 5)
        h_final, c_final = run_lbfgs(best_h, n, t9,
            alphas=[100000, 500000], mu=1.0, maxiter=3000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
