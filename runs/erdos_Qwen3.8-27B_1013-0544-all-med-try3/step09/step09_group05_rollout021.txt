def swap_update_corr_fast(corr, h, i, j, n):
    """Vectorized update of correlation after swap i:1→0, j:0→1."""
    new_corr = corr.copy()
    
    # Compute affected range
    m_candidates_min = [n-1-i, i, n-1-j, j]
    m_candidates_max = [2*n-2-i, n-1+i, 2*n-2-j, n-1+j]
    m_lo = max(0, min(m_candidates_min))
    m_hi = min(2*n-2, max(m_candidates_max))
    
    if m_lo > m_hi:
        return new_corr
    
    m_range = np.arange(m_lo, m_hi + 1)
    shifts = m_range - (n - 1)
    
    # i+shift
    ip = i + shifts
    mask_ip = (ip >= 0) & (ip < n)
    # Contribution: -(1 - h[ip]) = -1 if h[ip]=0, 0 if h[ip]=1 (for binary)
    delta_ip = np.where(mask_ip, -(1.0 - h[ip[mask_ip]]), 0.0)
    
    # i-shift
    im_arr = i - shifts
    mask_im = (im_arr >= 0) & (im_arr < n)
    delta_im = np.where(mask_im, h[im_arr[mask_im]], 0.0)
    
    # j+shift
    jp = j + shifts
    mask_jp = (jp >= 0) & (jp < n)
    delta_jp = np.where(mask_jp, 1.0 - h[jp[mask_jp]], 0.0)
    
    # j-shift
    jm_arr = j - shifts
    mask_jm = (jm_arr >= 0) & (jm_arr < n)
    delta_jm = np.where(mask_jm, -h[jm_arr[mask_jm]], 0.0)
    
    new_corr[m_lo:m_hi+1] += delta_ip + delta_im + delta_jp + delta_jm
    return new_corr
