def smooth_max_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

            m_pos = n - 1 + k_arr

            term1 = np.zeros(n)

def steepest_descend(h, n, deadline, n_steps=50000):
    """Steepest descent on the worst constraint only."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = int(np.argmax(c_vec))
        k = worst - (n - 1)
        
        # Gradient of O(k) w.r.t. h
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - best_h[k:n]) * dx
            grad[k:n] += (-best_h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - best_h[0:nv]) * dx
            grad[0:nv] += (-best_h[kk:n]) * dx
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            no_improve = 0
            eta = min(eta * 1.02, 0.05)
        else:
            no_improve += 1
            eta *= 0.95
            if no_improve > 200:
                break
    
    return best_h, best_c

                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_round = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_round = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = t0 + 40
    h1, c1 = lbfgs_opt(best_h, best_n, p1_end,
                       alphas=[100, 1000, 10000, 50000], mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Upsample to n=1000 and optimize
    n_mid = 1000
    if time.time() < deadline - 400:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c + 0.002:
            p2_end = min(time.time() + 150, deadline - 350)
            h2, c2 = lbfgs_opt(h_mid, n_mid, p2_end,
                               alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                               mu=0.003)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_mid
    
    # Phase 3: Coordinate descent at current resolution
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 100, deadline - 200)
        h3, c3 = coord_optimize(best_h, best_n, p3_end, n_passes=200)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Active set optimization
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 80, deadline - 150)
        h4, c4 = active_set_opt(best_h, best_n, p4_end, top_k=5, n_rounds=30)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try n=1500
    n_high = 1500
    if time.time() < deadline - 200 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.002:
            p5_end = min(time.time() + 100, deadline - 150)
            h5, c5 = lbfgs_opt(h_high, n_high, p5_end,
                               alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                               mu=0.002)
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_high
    
    # Phase 6: More coordinate descent
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 60, deadline - 60)
        h6, c6 = coord_optimize(best_h, best_n, p6_end, n_passes=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Steepest descent on worst constraint
    if time.time() < deadline - 80:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = steepest_descend(best_h, best_n, p7_end, n_steps=50000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final active set push
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 25, deadline - 15)
        h8, c8 = active_set_opt(best_h, best_n, p8_end, top_k=3, n_rounds=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    if time.time() < deadline - 5:

        if c9 < best_c:

            best_h = h9
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
