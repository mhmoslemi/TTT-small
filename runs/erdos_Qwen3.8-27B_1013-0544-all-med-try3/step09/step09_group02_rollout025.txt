**Key insight**: The problem is to minimize the maximum of 2n-1 bilinear forms. This is a non-convex problem, but we can use the following approach:

1. **Alternating minimization**: Fix the weights on the active set, solve a convex subproblem, update weights.

2. **Benders decomposition**: Use cuts from the max to build a piecewise linear approximation.

3. **Trust region**: Use a trust region around the current point, solve a simplified problem within the region.

For the trust region approach:
- Current point h₀, current max value M
- Active set: indices where c_k ≈ M
- Solve: min t subject to c_k(h) ≤ t + ε for k in active set, ||h - h₀|| ≤ Δ, 0 ≤ h ≤ 1, sum(h) = n/2
- This is a quadratic program (since c_k is bilinear, within a small trust region it's approximately linear)

Actually, within a trust region, c_k(h) ≈ c_k(h₀) + ∇c_k(h₀)·(h-h₀), so the subproblem is:
min t
s.t. c_k(h₀) + ∇c_k(h₀)·(h-h₀) ≤ t for k in active set
     ||h - h₀||₂ ≤ Δ
     0 ≤ h ≤ 1
     sum(h) = n/2

This is a linear program (LP) in variables (h, t) with trust region constraint. The trust region makes it non-linear, but we can approximate it with infinity norm: |h_j - h₀_j| ≤ Δ for all j, which is linear.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_project(h, n):
    target = n // 2
    h_sorted = np.argsort(-h)
    binary = np.zeros(n)
    binary[h_sorted[:target]] = 1.0
    return binary

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

        k = mm - (n - 1)

                            h[i] += step
                            h[j] -= step
    return h, best_c

def binary_local_search(h_bin, n, deadline, max_iter=300):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        best_improvement = 0
        best_swap = None
        n_try = min(3000, len(ones) * len(zeros) // 4)
        for _ in range(n_try):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            c = compute_c5(h, n)
            if c < best_c - 1e-10:
                imp = best_c - c
                if imp > best_improvement:
                    best_improvement = imp
                    best_swap = (i, j)
            h[i] = 1
            h[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        ones.remove(i)
        zeros.append(i)
        zeros.remove(j)
        ones.append(j)
        best_c = compute_c5(h, n)
    return h, best_c

def block_local_search(h, n, deadline, block_size=2):
    best_c = compute_c5(h, n)
    target = n / 2.0
    for it in range(500):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(2000):
            i = np.random.randint(0, n - block_size + 1)
            j = np.random.randint(0, n - block_size + 1)
            if abs(i - j) < block_size:
                continue
            block_i = h[i:i+block_size].copy()
            block_j = h[j:j+block_size].copy()
            h[i:i+block_size] = block_j
            h[j:j+block_size] = block_i
            if h.min() >= -1e-12 and h.max() <= 1 + 1e-12:
                c = compute_c5(h, n)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
            else:
                h[i:i+block_size] = block_i
                h[j:j+block_size] = block_j
            h[i:i+block_size] = block_i
            h[j:j+block_size] = block_j
        if not improved:
            break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS at original resolution (60s)
    t1 = t0 + 50
    if t1 < deadline - 300:
        best_h, best_c = lbfgs_optimize(best_h, n_orig, t1,
            alphas=[2000, 10000, 50000], mu=500.0)
    
    # Phase 2: SQP at original resolution (40s)
    t2 = time.time() + 35
    if t2 < deadline - 250:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=150, rho_init=0.04)
    
    # Phase 3: Upsample to n=1000 and optimize (150s)
    n_mid = 1000
    h_mid = upsample(best_h, n_orig, n_mid)
    c_mid = compute_c5(h_mid, n_mid)
    
    t3 = time.time() + 80
    if t3 < deadline - 150:
        h_mid, c_mid = lbfgs_optimize(h_mid, n_mid, t3,
            alphas=[2000, 10000, 50000, 200000], mu=500.0)
    
    t4 = time.time() + 50
    if t4 < deadline - 80:
        h_mid, c_mid = sqp_optimize(h_mid, n_mid, t4, max_iter=150, rho_init=0.03)
    
    if c_mid < best_c:
        best_c = c_mid
        best_h = h_mid
    
    # Phase 4: Directed local search at n=1000 (50s)
    t5 = time.time() + 40
    if t5 < deadline - 50:
        h_ls, c_ls = directed_local_search(best_h.copy(), len(best_h), t5, radius=40)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    # Phase 5: Binary projection + local search (40s)
    if time.time() < deadline - 40:
        n_curr = len(best_h)
        h_bin = binary_project(best_h, n_curr)
        c_bin = compute_c5(h_bin, n_curr)
        t6 = time.time() + 30
        if t6 < deadline - 10:
            h_bin, c_bin = binary_local_search(h_bin, n_curr, t6, max_iter=150)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    
    # Phase 6: Upsample to n=2000 and optimize (120s)
    n_high = 2000
    h_high = upsample(best_h, len(best_h), n_high)
    c_high = compute_c5(h_high, n_high)
    
    t7 = time.time() + 60
    if t7 < deadline - 60:
        h_high, c_high = lbfgs_optimize(h_high, n_high, t7,
            alphas=[5000, 20000, 100000], mu=500.0)
    
    t8 = time.time() + 40
    if t8 < deadline - 30:
        h_high, c_high = sqp_optimize(h_high, n_high, t8, max_iter=100, rho_init=0.03)
    
    if c_high < best_c:
        best_c = c_high
        best_h = h_high
    
    # Phase 7: Directed local search at high res (40s)
    t9 = time.time() + 35
    if t9 < deadline - 20:
        h_ls2, c_ls2 = directed_local_search(best_h.copy(), len(best_h), t9, radius=30)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2
    
    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 15:
        h_fin, c_fin = slqp_polish(best_h, len(best_h), deadline - 3, alpha=1000000, maxiter=1500)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
