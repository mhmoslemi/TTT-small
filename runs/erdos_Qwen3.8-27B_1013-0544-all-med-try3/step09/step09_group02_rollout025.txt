def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

def project_feasible(h, n):

        if abs(diff) < 1e-15:

            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            if tot < 1e-14:
                break
            h += (diff / tot) * h

def shift_grad_single(h, n, shift_idx):

    grad = np.zeros(n)
    k = shift_idx - (n - 1)
