            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def top_active_gradient(h, n, w, k_active=150):
    n_corr = len(w)
    if k_active >= n_corr:
        idxs = np.arange(n_corr)
        ws = w.copy()
    else:
        top = np.argsort(-w)[:k_active]
        idxs = top
        ws = w[top]
        ws = ws / ws.sum()
    g = np.zeros(n)
    for i in range(len(idxs)):
        if ws[i] < 1e-14:
            continue
        g += ws[i] * shift_grad_single(h, n, int(idxs[i]))
    return g

def lbfqq_run(h, n, deadline, alphas=None):
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [3000, 15000, 80000, 300000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 40:
            break
        def obj_fn(x, n_x, a):
            c = compute_c5_vec(x, n_x)
            val = smooth_max_val(c, a)
            val += 1e8 * (x.sum() - n_x / 2.0) ** 2
            return val
        def grad_fn(x, n_x, a):
            c = compute_c5_vec(x, n_x)
            w = smooth_max_weights(c, a)
            g = top_active_gradient(x, n_x, w, 150)
            g += 2e8 * (x.sum() - n_x / 2.0)
            return g
        try:
            res = minimize(lambda x: obj_fn(x, n, alpha), h,
                          jac=lambda x: grad_fn(x, n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def sqp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds_lp = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.06):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-12, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 25:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:25]]
        delta, t_opt = sqp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.5, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 30:
                break
    return best_h, best_c

def gradient_descent_run(h, n, deadline, max_iter=50000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha_smooth = 50000
    step_size = 0.5
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        w = smooth_max_weights(c_vec, alpha_smooth)
        g = top_active_gradient(h, n, w, 150)
        g += 2e8 * (h.sum() - n / 2.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        for ls in range(30):
            step = step_size * (0.7 ** ls)
            h_trial = project_feasible(h - step * g, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                h = h_trial
                best_c = c_trial
                best_h = h.copy()
                step_size = min(step * 1.5, 5.0)
                break
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return best_h, best_c

def local_search(h, n, deadline, center_radius=60):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline - 5:
            break
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad_abs = np.zeros(n)
            for idx in active[:50]:
                grad_abs += np.abs(shift_grad_single(h, n, int(idx)))
            grad_abs /= min(len(active), 50)
            top_pos = np.argsort(-grad_abs)[:min(center_radius * 3, n)]
            for i_val in top_pos:
                if time.time() > deadline - 5:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x, n_x, a):
        c = compute_c5_vec(x, n_x)
        return smooth_max_val(c, a)
    def sq_fn(x, n_x):
        return float(x.sum() - n_x / 2.0)
    def sq_jac_fn(x, n_x):
        return np.ones(n_x)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(lambda x: obj_fn(x, n, alpha), h,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    # Work at n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_parent)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, h_parent)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS warmup (100s)
    t1 = min(t0 + 100, deadline - 800)
    if time.time() < t1 - 5:
        h, c = lbfqq_run(h, n, t1, alphas=[3000, 20000, 100000])
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SQP (150s)
    t2 = min(t0 + 250, deadline - 650)
    if time.time() < t2 - 10:
        h, c = sqp_optimize(h, n, t2, max_iter=500, rho_init=0.06)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Gradient descent (80s)
    t3 = min(t0 + 330, deadline - 550)
    if time.time() < t3 - 5:
        h, c = gradient_descent_run(h, n, t3, max_iter=30000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Local search (100s)
    t4 = min(t0 + 430, deadline - 450)
    if time.time() < t4 - 5:
        h, c = local_search(h, n, t4, center_radius=50)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: SQP again with tighter active set (80s)
    t5 = min(t0 + 510, deadline - 370)
    if time.time() < t5 - 10:
        h, c = sqp_optimize(h, n, t5, max_iter=400, rho_init=0.03)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Another L-BFGS round with very high alpha (80s)
    t6 = min(t0 + 590, deadline - 300)
    if time.time() < t6 - 5:
        h, c = lbfqq_run(h, n, t6, alphas=[200000, 500000])
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Local search again (60s)
    t7 = min(t0 + 650, deadline - 240)
    if time.time() < t7 - 5:
        h, c = local_search(h, n, t7, center_radius=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 8: Try n=2000 (200s)
    n_high = 2000
    t8 = min(t0 + 850, deadline - 150)
    if time.time() < t8 - 20:
        x_old2 = np.linspace(0, 2, n)
        x_new2 = np.linspace(0, 2, n_high)
        h_hi = np.interp(x_new2, x_old2, best_h)
        h_hi = project_feasible(h_hi, n_high)
        c_hi = compute_c5(h_hi, n_high)

        t8a = min(time.time() + 100, deadline - 100)
        if time.time() < t8a - 10:
            h_hi, c_hi = lbfqq_run(h_hi, n_high, t8a, alphas=[10000, 50000, 200000])

        t8b = min(time.time() + 80, deadline - 50)
        if time.time() < t8b - 10:
            h_hi, c_hi = sqp_optimize(h_hi, n_high, t8b, max_iter=150, rho_init=0.04)

        t8c = min(time.time() + 50, deadline - 20)
        if time.time() < t8c - 5:
            h_hi, c_hi = local_search(h_hi, n_high, t8c, center_radius=40)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            n = n_high

    # Phase 9: Final polish
    if time.time() < deadline - 15:
        h_f, c_f = slqp_polish(best_h, len(best_h), deadline - 3, alpha=1000000, maxiter=1000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
