import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    """Gradient of c[shift_idx] with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def topk_obj(x, n, k):
    """Sum of squares of top-k overlap values."""
    c = compute_c5_vec(x, n)
    idx = np.argsort(-c)[:k]
    return float(np.sum(c[idx]**2))

def topk_grad(x, n, k):
    """Gradient of sum of squares of top-k overlap values."""
    c = compute_c5_vec(x, n)
    idx = np.argsort(-c)[:k]
    grad = np.zeros(n)
    for i in idx:
        grad += 2.0 * c[i] * shift_grad_single(x, n, i)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    """SQP step using LP subproblem."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:

                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            return delta
    except Exception:
        pass
    return None

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(5000):

        c_vec = compute_c5_vec(h, n)

        if c_max < best_c:

def local_search_swap(h, n, deadline):
    """Efficient swap-based local search."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline - 2:
            break
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            # Focus on shifts near the max
            thresh = max_val - max(1e-8, 0.005 * max_val)
            active_set = set(np.where(c_vec >= thresh)[0].tolist())
            
            # For each active shift, find the pair that most reduces that shift
            # Shift k affects pairs (i, i+k) for k>=0 or (i-k, i) for k<0
            # Moving mass from i to j changes c_k by:
            # For each active k: delta_c_k = step * [grad_k[i] - grad_k[j]]
            
            # Use gradient to find good pairs
            grad = np.zeros(n)
            active_list = list(active_set)
            for idx in active_list:
                g = shift_grad_single(h, n, idx)
                # Weight by how close c[idx] is to max
                w = max(0, c_vec[idx] - (max_val - 0.01))
                grad += w * g
            if len(active_list) > 0:
                grad /= len(active_list)
            
            # Find best pair: maximize -grad[i] + grad[j] (i gives, j receives)
            # i should have high grad (giving mass reduces active overlaps)
            # j should have low grad (receiving mass doesn't increase them)
            # But we also need h[i] - step >= 0 and h[j] + step <= 1
            
            candidates = np.argsort(-grad)[:min(50, n)]
            donors = np.argsort(grad)[:min(50, n)]  # low grad = good recipient
            
            for i_val in candidates:
                i = int(i_val)
                if h[i] < step + 1e-12:
                    continue
                for j_val in donors:
                    j = int(j_val)
                    if i == j:
                        continue
                    if h[j] + step > 1.0 + 1e-12:
                        continue
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] += step
                        h[j] -= step
                    if time.time() > deadline - 3:
                        return h, best_c
    return h, best_c

def try_binary(h, n):
    """Try binary projection and local search."""
    # Project to binary: keep the largest n/2 values
    thresholds = np.sort(h)[::-1]
    target = int(n // 2)

    if cnt > target:

        # Remove the ones with smallest h values
        order = np.argsort(h[ones_idx])[:cnt - target]
        for idx in order:
            h_bin[ones_idx[idx]] = 0
    elif cnt < target:

        for idx in order:
            h_bin[zeros_idx[idx]] = 1
    c_bin = compute_c5(h_bin, n)
    return h_bin, c_bin

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=0,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n]
    except Exception:
        pass
    return None

def sqp_phase(h, n, deadline, max_iter, rho_init, max_active, label):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_imp = 0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-11, 0.0003 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if len(active) > max_active:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:max_active]]
        if not active:
            break
        delta = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            no_imp += 1
            if no_imp > 15:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_imp = 0
            rho = min(rho * 1.15, 0.15)
        else:
            no_imp += 1
            rho *= 0.5
            if rho < 1e-11 or no_imp > 20:
                break
    return best_h, best_c

def topk_gradient_step(h, n, k, lr, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(3000):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(h, n)
        idx = np.argsort(-c_vec)[:k]
        c_max = c_vec[idx[0]]

            best_c = c_max
            best_h = h.copy()
        grad = np.zeros(n)
        for i in idx:
            grad += 2.0 * c_vec[i] * shift_grad_single(h, n, i)
        grad += 0.05 * (h.sum() - n / 2.0)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_max - 1e-16:
            h = h_new
            lr = min(lr * 1.03, 0.05)

            lr *= 0.7
            if lr < 1e-9:
                break

def directed_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:
        if time.time() > deadline - 3:
            break
        improved = True
        while improved and time.time() < deadline - 3:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.003 * max_val)
            active_set = np.where(c_vec >= thresh)[0]
            if len(active_set) == 0:
                break
            grad = np.zeros(n)
            for idx in active_set:
                w = c_vec[idx]
                grad += w * shift_grad_single(h, n, idx)
            grad /= (len(active_set) * max_val + 1e-15)
            top_donors = np.argsort(-grad)[:min(30, n)]
            top_recv = np.argsort(grad)[:min(30, n)]
            found = False
            for i_val in top_donors:
                if found:
                    break
                i = int(i_val)
                if h[i] < step + 1e-12:
                    continue
                for j_val in top_recv:
                    j = int(j_val)
                    if i == j:
                        continue
                    if h[j] + step > 1.0 + 1e-12:
                        continue
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        found = True
                    else:
                        h[i] += step
                        h[j] -= step
                    if time.time() > deadline - 4:
                        return h, best_c
    return h, best_c
