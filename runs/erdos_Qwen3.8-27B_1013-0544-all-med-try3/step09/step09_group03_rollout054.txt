            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            new_cvec = np.correlate(h_bin, 1 - h_bin, mode='full')
            new_max = new_cvec.max()
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                best_c = float(new_max * 2.0 / n)
                improved = True
                c_vec = new_cvec
                ones = np.where(h_bin == 1.0)[0]
                zeros = np.where(h_bin == 0.0)[0]
            else:
                h_bin[a] = 1.0
                h_bin[b] = 0.0
        if not improved and rnd > 20:
            break
    return h_bin, best_c

def targeted_refine_fast(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        top_idx = np.argsort(-c_vec)[:10]
        improved = False
        center = n - 1
        for idx in top_idx:
            k = idx - center
            if k >= 0:
                affected = list(range(k, n)) + list(range(0, n - k))
            else:
                kk = -k
                affected = list(range(0, n - kk)) + list(range(kk, n))
            for i in affected[:50]:
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]:
                    nv = best_h[i] + delta
                    if nv < -1e-10 or nv > 1.0 + 1e-10:
                        continue
                    saved = best_h[i]
                    best_h[i] = nv
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_h = best_h.copy()
                        best_h[i] = saved
                        best_h = project_feasible(best_h, n)
                    if improved:
                        break
            if improved:
                break
        if not improved and rnd > 30:
            break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search_fast(h, n, deadline, n_trials, mags):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(99999)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_objective, h_p, jac=smooth_gradient,
                          args=(n, 5000, 20000.0, 20), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: L-BFGS at base resolution with increasing alpha (250s)
    t1 = t0 + 250
    if t1 < deadline - 500:
        configs_500 = [
            (100, 10.0, None, 3000),
            (500, 50.0, None, 3000),
            (2000, 200.0, 30, 4000),
            (10000, 1000.0, 20, 4000),
            (50000, 5000.0, 15, 3000),
            (200000, 20000.0, 10, 2000),
        ]
        best_h, best_c = run_lbfgs_fast(best_h, n_base, t1, configs_500)
    
    # Phase 2: SLSQP polish at base (60s)
    t2 = min(time.time() + 60, deadline - 450)
    if t2 > time.time() + 10:
        h2, c2 = slqp_optimize(best_h, n_base, t2, alpha=500000, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Binary local search at base (60s)
    t3 = min(time.time() + 60, deadline - 400)
    if t3 > time.time() + 5:
        h_bin, c_bin = binary_local_search(best_h, n_base, t3, max_rounds=150)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Continue from binary result with L-BFGS
        if time.time() < deadline - 350:
            t3b = min(time.time() + 40, deadline - 300)
            h3b, c3b = run_lbfgs_fast(best_h, n_base, t3b,
                [(5000, 20000.0, 15, 2000), (50000, 200000.0, 10, 1500)])
            if c3b < best_c:
                best_c = c3b
                best_h = h3b
    
    # Phase 4: Perturbation at base (60s)
    t4 = min(time.time() + 60, deadline - 300)
    if t4 > time.time() + 5:
        best_h, best_c = perturbation_search_fast(best_h, n_base, t4,
            n_trials=8, mags=[0.02, 0.04, 0.06, 0.01, 0.08, 0.03, 0.10, 0.025])
    
    # Phase 5: SLSQP after perturbation (40s)
    t5 = min(time.time() + 40, deadline - 250)
    if t5 > time.time() + 5:
        h5, c5 = slqp_optimize(best_h, n_base, t5, alpha=1000000, maxiter=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Upsample to n=1000 and optimize (200s)
    n1 = 1000
    if time.time() < deadline - 250:
        h1000 = interpolate_to(best_h, n_base, n1)
        configs_1000 = [
            (500, 50.0, None, 2000),
            (2000, 200.0, 30, 2500),
            (10000, 1000.0, 20, 2000),
            (50000, 5000.0, 15, 1500),
            (200000, 20000.0, 10, 1000),
        ]
        t6 = min(time.time() + 130, deadline - 120)
        h1000, c1000 = run_lbfgs_fast(h1000, n1, t6, configs_1000)
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        
        t7 = min(time.time() + 50, deadline - 70)
        if t7 > time.time() + 5:
            h7, c7 = slqp_optimize(best_h, n1, t7, alpha=2000000, maxiter=2000)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 7: Binary local search at n=1000 (40s)
    if time.time() < deadline - 100:
        t8 = min(time.time() + 40, deadline - 60)
        h_bin1k, c_bin1k = binary_local_search(best_h, n1, t8, max_rounds=100)
        if c_bin1k < best_c:
            best_c = c_bin1k
            best_h = h_bin1k
        # Quick L-BFGS from binary
        if time.time() < deadline - 50:
            t8b = min(time.time() + 25, deadline - 30)
            h8b, c8b = run_lbfgs_fast(best_h, n1, t8b,
                [(10000, 50000.0, 10, 1000), (100000, 500000.0, 8, 500)])
            if c8b < best_c:
                best_c = c8b
                best_h = h8b
    
    # Phase 8: Upsample to n=2000 (100s)
    n2 = 2000
    if time.time() < deadline - 100:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        configs_2000 = [
            (1000, 100.0, None, 1000),
            (5000, 500.0, 30, 1000),
            (20000, 2000.0, 20, 800),
            (100000, 10000.0, 10, 500),
        ]
        t9 = min(time.time() + 70, deadline - 30)
        h2000, c2000 = run_lbfgs_fast(h2000, n2, t9, configs_2000)
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        
        t10 = min(time.time() + 25, deadline - 10)
        if t10 > time.time() + 5:
            h10, c10 = slqp_optimize(best_h, n2, t10, alpha=5000000, maxiter=1000)
            if c10 < best_c:
                best_c = c10
                best_h = h10
    
    # Phase 9: Final targeted refinement (20s)
    if time.time() < deadline - 5:
        best_h, best_c = targeted_refine_fast(best_h, len(best_h), deadline - 2)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
