import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def adam_optimize(h, n, deadline, lr=0.01, beta1=0.9, beta2=0.999, alpha=500):
    """Adam optimizer for the smooth max objective"""
    h_curr = h.copy().astype(float)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    patience = 0
    lr_current = lr
    
    while time.time() < deadline:
        t += 1
        c = compute_all_lags(h_curr, n)
        # Smooth max gradient
        m_c = c.max()
        shifted = alpha * (c - m_c)
        mx_s = shifted.max()
        if mx_s > 500:
            w = np.exp(shifted - mx_s)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * _lag_gradient(h_curr, n, lag)
        grad *= (2.0 / n)
        
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        update = m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_curr - lr_current * update, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > 500:
            lr_current *= 0.8
            patience = 0
            alpha = max(100, alpha * 0.9)
        if t % 500 == 0:
            lr_current = max(0.001, lr_current * 0.95)
    
    return best_h, best_c

def lag_targeted_gradient(h, n, top_k=1):
    """Compute gradient focusing only on top-k lags"""
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    top_idx = np.sort(top_idx)
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        grad += _lag_gradient(h, n, lag)
    grad /= top_k
    return grad * (2.0 / n)

def lag_targeted_descent(h, n, deadline, top_k=1, lr=0.05):
    """Gradient descent focusing on top-k lags"""
    h_curr = h.copy().astype(float)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    while time.time() < deadline:
        grad = lag_targeted_gradient(h_curr, n, top_k)
        h_new = project_feasible(h_curr - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        else:
            # Try smaller step
            h_new = project_feasible(h_curr - lr * 0.5 * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
            else:
                lr *= 0.9
                h_new = project_feasible(h_curr, n)
        
        h_curr = h_new
    
    return best_h, best_c

def top_lag_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, top_m=3, alpha=500000, maxiter=2000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(top_lag_smooth_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb_and_refine(h, n, deadline, rng, n_trials=10):
    """Perturb the solution and try to refine"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        scale = 0.01 + 0.02 * trial / max(1, n_trials - 1)
        hp = h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        
        # Quick Adam optimization
        ad = deadline - max(5, (deadline - time.time()) / (n_trials - trial))
        if time.time() > ad:
            ad = time.time() + 5
        h_new, c_new = adam_optimize(hp, n, ad, lr=0.05, alpha=200)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def top_lag_gradient(h, n, top_k=2):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        grad += _lag_gradient(h, n, lag)
    grad /= float(top_k)
    return grad * (2.0 / n), c

def slsqp_polish(h, n, top_m=3, alpha=300000, maxiter=2000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(top_lag_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def full_smooth_slsqp(h, n, alpha=500000, maxiter=1000):

    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
