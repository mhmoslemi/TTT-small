    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def _smooth_max(c_active, alpha):
    m_val = c_active.max()
    shifted = alpha * (c_active - m_val)
    mx = shifted.max()
    if mx > 500:
        return float(m_val + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m_val + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_max_grad_weights(c_active, alpha):
    m_val = c_active.max()
    shifted = alpha * (c_active - m_val)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def _grad_smooth_max(h, n, top_idx, alpha):
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    active = c_all[top_idx]
    w = _smooth_max_grad_weights(active, alpha)
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] < 1e-10:
            continue
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad[:] += w[i] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[i] * (1.0 - h[:n - lag])
            grad[:n - lag] += w[i] * (-h[lag:n])
        else:
            grad[:n + lag] += w[i] * (1.0 - h[-lag:])
            grad[-lag:] += w[i] * (-h[:n + lag])
    return grad * dx

def multi_lag_descent(h, n, deadline, max_iter=30000, top_m=8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    step_base = 0.03
    it_since_improve = 0
    alpha_base = 100.0
    rng_local = np.random.RandomState(12345)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c_all)[-top_m:]
        alpha = alpha_base / (1.0 + it * 0.0003)
        g = _grad_smooth_max(h_curr, n, top_idx, alpha)
        step = step_base / (1.0 + it * 0.002) ** 0.5
        h_try = project_feasible(h_curr - step * g, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h_curr = h_try
            it_since_improve = 0
        else:
            it_since_improve += 1
            # Try smaller step
            for s_mult in [0.5, 0.25, 0.1]:
                h_try2 = project_feasible(h_curr - step * s_mult * g, n)
                c_try2 = compute_c5(h_try2, n)
                if c_try2 < best_c - 1e-15:
                    best_c = c_try2
                    best_h = h_try2.copy()
                    h_curr = h_try2
                    it_since_improve = 0
                    break
            else:
                if it_since_improve > 800:
                    h_curr = best_h + rng_local.randn(n) * 0.015
                    h_curr = project_feasible(h_curr, n)
                    it_since_improve = 0
                    step_base = max(0.005, step_base * 0.8)
        if it % 500 == 499:
            step_base = max(0.003, step_base * 0.9)

    return best_h, best_c

def fine_local_search(h, n, deadline, top_m=3, rounds_limit=2000):
    h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0

    for rnd in range(rounds_limit):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[-top_m:]

        improved_this_round = False
        for k_idx in top_lags:
            if time.time() > deadline:
                break
            lag = int(k_idx) - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for d in [0.001, 0.0005, 0.0002]:
                for i in range(0, n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    diff = h[i] - h[j]
                    if abs(diff) < 1e-11:
                        continue
                    if diff > 0:
                        delta = min(d, diff * 0.35)
                        if delta < 1e-13:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved_this_round = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    else:
                        delta = min(d, (-diff) * 0.35)
                        if delta < 1e-13:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved_this_round = True
                        else:
                            h[i] -= delta
                            h[j] += delta

        if improved_this_round:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return h, best_c

def slsqp_polish(h, n, deadline, top_m=8, maxiter=5000):
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    alpha = 50000

    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return _smooth_max(active, alpha)

    def con_fun(x):
        return float(np.sum(x) - n / 2.0)

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def cutting_plane_opt(h, n, deadline, max_iters=200, top_m=15):
    """Cutting plane for the convex min-max problem"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    cuts_grads = []
    cuts_vals = []
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute current objective and subgradient
        c_all = compute_all_lags(h_curr, n)
        cur_val = float(np.max(c_all))
        top_idx = np.argsort(c_all)[-top_m:]
        active = c_all[top_idx]
        
        # Weighted subgradient (smooth max gradient approx)
        alpha_cp = 200.0
        w = _smooth_max_grad_weights(active, alpha_cp)
        g = _grad_smooth_max(h_curr, n, top_idx, alpha_cp)
        
        # Add cut: f(x) >= cur_val + g · (x - h_curr)
        cut_g = g.copy()
        cut_b = cur_val - float(np.dot(g, h_curr))
        cuts_grads.append(cut_g)
        cuts_vals.append(cut_b)
        
        # Keep only top 50 cuts (most violated)
        if len(cuts_grads) > 50:
           # Evaluate all cuts at current point, keep most relevant
            vals = np.array([float(np.dot(cg, h_curr)) + cb for cg, cb in zip(cuts_grads, cuts_vals)])
            order = np.argsort(vals)[-40:]
            cuts_grads = [cuts_grads[i] for i in order]
            cuts_vals = [cuts_vals[i] for i in order]
        
        # Solve the LP: min t s.t. g_k·x + b_k <= t, 0<=x<=1, sum(x)=n/2
        # Use SLSQP as surrogate (we might not have a fast LP solver)
        m_cuts = len(cuts_grads)
        stag = np.array(cuts_grads)  # (m_cuts, n)
        sb = np.array(cuts_vals)
        
        def lp_obj(x):
            # Maximize the max cut value
            vals = stag @ x + sb
            return float(np.max(vals))
        
        def lp_con_fun(x):
            return float(np.sum(x) - n / 2.0)
        
        def lp_con_jac(x):
            return np.ones(n)
        
        def lp_obj_grad(x):
            vals = stag @ x + sb
            k_max = int(np.argmax(vals))
            return stag[k_max]
        
        bounds = [(0.0, 1.0)] * n
        cons_lp = [{'type': 'eq', 'fun': lp_con_fun, 'jac': lp_con_jac}]
        
        try:
            res = minimize(lp_obj, h_curr, method='SLSQP', jac=lp_obj_grad,
                          bounds=bounds, constraints=cons_lp,
                          options={'maxiter': 200, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
        except Exception:
            h_new = h_curr
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            # Nudge
            h_curr = 0.7 * h_new + 0.3 * h_curr
            h_curr = project_feasible(h_curr, n)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    # Get parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Quick optimization at parent resolution
    ph1_end = t0 + min(60, (deadline - t0) * 0.06)
    h1, c1 = multi_lag_descent(h_init, n_init, ph1_end, max_iter=8000, top_m=8)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    h1, c1 = fine_local_search(h1, n_init, ph1_end + 30, top_m=3, rounds_limit=500)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Interpolate to n=1000 and optimize
    n2 = 1000
    if time.time() < deadline - 150:
        h2 = interp_h(best_h, len(best_h), n2)
        c2 = compute_c5(h2, n2)
        
        ph2_sub_end = time.time() + min(120, (deadline - time.time()) * 0.25)
        h2a, c2a = multi_lag_descent(h2, n2, ph2_sub_end, max_iter=20000, top_m=8)
        if c2a < c2:
            h2, c2 = h2a, c2a
        
        # SLSQP polish
        if time.time() < deadline - 80:
            h2b = slsqp_polish(h2, n2, deadline - 80, top_m=8, maxiter=4000)
            c2b = compute_c5(h2b, n2)
            if c2b < c2:
                h2, c2 = h2b, c2b
        
        # Fine local search
        ph2_ls_end = min(deadline - 60, time.time() + 120)
        h2c, c2c = fine_local_search(h2, n2, ph2_ls_end, top_m=4, rounds_limit=3000)
        if c2c < c2:
            h2, c2 = h2c, c2c
        
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Try n=2000
    n3 = 2000
    if time.time() < deadline - 120:
        h3 = interp_h(best_h, len(best_h), n3)
        c3 = compute_c5(h3, n3)
        
        remaining3 = deadline - time.time() - 30
        if remaining3 > 80:
            ph3_sub_end = time.time() + min(100, remaining3 * 0.35)
            h3a, c3a = multi_lag_descent(h3, n3, ph3_sub_end, max_iter=10000, top_m=6)
            if c3a < c3:
                h3, c3 = h3a, c3a
            
            if time.time() < deadline - 40:
                ph3_ls_end = min(deadline - 10, time.time() + min(150, remaining3 * 0.4))
                h3b, c3b = fine_local_search(h3, n3, ph3_ls_end, top_m=3, rounds_limit=2000)
                if c3b < c3:
                    h3, c3 = h3b, c3b
            
            h3 = project_feasible(h3, n3)
            c3 = compute_c5(h3, n3)
            if c3 < best_c:
                best_c = c3
                best_h = h3
    
    # Phase 4: SLSQP final polish at best resolution
    if time.time() < deadline - 15:
        n_best = len(best_h)
        h_pol = slsqp_polish(best_h, n_best, deadline - 10, top_m=10, maxiter=3000)
        c_pol = compute_c5(h_pol, n_best)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol
    
    # Phase 5: Final fine local search
    if time.time() < deadline - 2:
        h_fin, c_fin = fine_local_search(best_h, len(best_h), deadline - 1, top_m=3, rounds_limit=500)
        best_h = project_feasible(h_fin, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
