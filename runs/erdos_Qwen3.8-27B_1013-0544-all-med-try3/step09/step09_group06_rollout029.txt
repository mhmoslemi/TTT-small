1. Compute the gradient of the worst lag
2. Project it onto the tangent space (remove the component along the sum constraint)
3. If the projected gradient is small, we're at a KKT point
4. If not, take a step in the negative projected gradient direction

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def fw_oracle(g, n):
    target = n / 2.0
    idx = np.argsort(g)
    s = np.zeros(n)
    k_full = int(target)
    s[idx[:k_full]] = 1.0
    frac = target - k_full
    if frac > 1e-10:
        s[idx[k_full]] = frac
    return s
