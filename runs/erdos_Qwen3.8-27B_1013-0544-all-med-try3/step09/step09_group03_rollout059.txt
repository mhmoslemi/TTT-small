    h_b = (h >= 0.5).astype(np.float64)
    target = n // 2
    diff = target - int(h_b.sum())
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] == 0:
                h_b[i] = 1.0
                diff -= 1
                if diff <= 0:
                    break
    elif diff < 0:
        order = np.argsort(h)
        for i in order:
            if h_b[i] == 1:
                h_b[i] = 0.0
                diff += 1
                if diff >= 0:
                    break
    return h_b

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def local_search_binary(h, n, deadline, n_trials_per_iter=100, max_iters=500, sa_phases=5, rng_seed=12345):
    """Binary local search with simulated annealing."""
    best = h.copy()
    dx = 2.0 / n
    best_c = float(np.max(np.correlate(best, 1 - best, mode='full')) * dx)
    rng = np.random.RandomState(rng_seed)
    
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    
    no_improve = 0
    total_iters = 0
    
    for phase in range(sa_phases):
        # Temperature for this phase
        T = dx * (0.5 + 0.3 * (sa_phases - 1 - phase) / max(1, sa_phases - 1))
        if phase == sa_phases - 1:
            T = 0  # Pure greedy at the end
        
        for it in range(max_iters // sa_phases):
            if time.time() > deadline:
                return best, best_c
            total_iters += 1
            found_improve = False
            
            for _ in range(n_trials_per_iter):
                if time.time() > deadline:
                    return best, best_c
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                
                best[i] = 0.0
                best[j] = 1.0
                c = float(np.max(np.correlate(best, 1 - best, mode='full')) * dx)
                
                if c < best_c - 1e-15:
                    best_c = c
                    found_improve = True
                    ones = np.where(best == 1)[0]
                    zeros = np.where(best == 0)[0]
                elif c - best_c < T and rng.rand() < np.exp(-(c - best_c) / max(T, 1e-15)):
                    # Accept worsening move (simulated annealing)
                    best_c = c
                    ones = np.where(best == 1)[0]
                    zeros = np.where(best == 0)[0]
                    found_improve = True
                else:
                    best[i] = 1.0
                    best[j] = 0.0
            
            if not found_improve:
                no_improve += 1
                if no_improve >= 3:
                    break
            else:
                no_improve = 0
    
    return best, best_c

def swap_local_search(h, n, deadline, n_candidates=200, max_rounds=200, rng_seed=999):

def best_improve_search(h, n, deadline, n_candidates=150, max_rounds=300, rng_seed=7777):
    """Proper best-improvement: restore after each trial, apply best at end of round."""
    cur = h.copy()
    dx = 2.0 / n
    cur_c = float(np.max(np.correlate(cur, 1 - cur, mode='full')) * dx)
    rng = np.random.RandomState(rng_seed)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        nc = min(n_candidates, len(ones))
        
        best_delta = 0.0
        best_i = -1
        best_j = -1
        
        for _ in range(nc):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0.0
            cur[j] = 1.0
            c = float(np.max(np.correlate(cur, 1 - cur, mode='full')) * dx)
            delta = c - cur_c
            if delta < best_delta - 1e-15:
                best_delta = delta
                best_i = i
                best_j = j
            cur[i] = 1.0
            cur[j] = 0.0
        
        if best_i == -1:
            break
        cur[best_i] = 0.0
        cur[best_j] = 1.0
        cur_c += best_delta
    
    return cur, cur_c

def multi_start_sa(h, n, deadline, n_starts=4, rng_seed=555):
    """Multiple starts: perturb and run SA."""
    best_h = h.copy()
    best_c = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
    rng = np.random.RandomState(rng_seed)
    time_per_start = (deadline - time.time()) / max(1, n_starts)
    
    for s in range(n_starts):
        t_end = time.time() + time_per_start
        # Create perturbed version
        h_p = h.copy()
        n_swaps = n // 50
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        for _ in range(n_swaps):
            if datetime_check(t_end):
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_p[i], h_p[j] = h_p[j], h_p[i]
        
        h_p, c_p = local_search_binary(h_p, n, t_end, n_trials_per_iter=80, max_iters=300, sa_phases=4, rng_seed=rng_seed + s * 1000)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
    
    return best_h, best_c

def datetime_check(t_end):
    return time.time() > t_end

def lbfgs_optimize(h, n, deadline, configs=None):
    """L-BFGS smooth optimization."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (20000, 10000.0), (50000, 30000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    """SLSQP with equality constraint."""
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def targeted_refine(h, n, deadline, n_rounds=30):
    """Coordinate descent on individual values."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for round_num in range(n_rounds):
        if time.time() > deadline:
            break
        # Find worst shift
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        # Determine which indices affect this shift
        if k_worst >= 0:
            idx_range = range(0, n - k_worst)
        else:
            idx_range = range(-k_worst, n)
        
        improved = False
        for i in idx_range:
            if time.time() > deadline - 5:
                return best_h, best_c
            # Try small adjustments
            for delta in [0.003, -0.003, 0.001, -0.001, 0.005, -0.005]:
                nv = h[i] + delta
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = best_h[i]
            if improved:
                h = best_h.copy()
    
    return best_h, best_c

def perturbation_then_optimize(h, n, deadline, n_trials=6, mags=None):
    """Random perturbations + quick L-BFGS."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(31415)
    if mags is None:
        mags = [0.03, 0.05, 0.02, 0.07, 0.04, 0.10]
    
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        t_end = min(time.time() + 60, deadline - 10)
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 2000, 3000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(1000, 0.5)
        n0 = 1000
    
    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: Binarize and local search at n0 (~150s)
    h_bin0 = binarize(best_h, n0)
    c_bin0 = compute_c5(h_bin0, n0)
    t1 = min(t0 + 120, deadline - 700)
    h_bin0, c_bin0 = best_improve_search(h_bin0, n0, t1, n_candidates=200, max_rounds=500, rng_seed=111)
    
    # SA at n0
    t1b = min(t0 + 200, deadline - 600)
    h_bin0, c_bin0 = local_search_binary(h_bin0, n0, t1b, n_trials_per_iter=80, max_iters=400, sa_phases=5, rng_seed=222)
    
    if c_bin0 < best_c:
        best_c = c_bin0
        best_h = h_bin0.copy()
    
    # Phase 2: Upsample to n=1000, binarize, optimize (~250s)
    n1 = 1000
    if time.time() < deadline - 400:
        h1 = interpolate_to(best_h, len(best_h), n1)
        h1_bin = binarize(h1, n1)
        
        t2 = min(t0 + 400, deadline - 350)
        h1_bin, c1 = best_improve_search(h1_bin, n1, t2, n_candidates=150, max_rounds=400, rng_seed=333)
        
        # SA
        t2b = min(t0 + 500, deadline - 280)
        h1_bin, c1 = local_search_binary(h1_bin, n1, t2b, n_trials_per_iter=60, max_iters=300, sa_phases=4, rng_seed=444)
        
        if c1 < best_c:
            best_c = c1
            best_h = h1_bin.copy()
    
    # Phase 3: Multi-start SA at n=1000 (~150s)
    if time.time() < deadline - 250:
        t3 = min(t0 + 650, deadline - 200)
        h3, c3 = multi_start_sa(best_h, n1, t3, n_starts=3, rng_seed=555)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: Upsample to n=2000, binarize, optimize (~200s)
    n2 = 2000
    if time.time() < deadline - 200:
        h2 = interpolate_to(best_h, len(best_h), n2)
        h2_bin = binarize(h2, n2)
        
        t4 = min(time.time() + 120, deadline - 80)
        h2_bin, c2 = best_improve_search(h2_bin, n2, t4, n_candidates=100, max_rounds=200, rng_seed=666)
        
        t4b = min(time.time() + 50, deadline - 30)
        h2_bin, c2 = local_search_binary(h2_bin, n2, t4b, n_trials_per_iter=50, max_iters=200, sa_phases=3, rng_seed=777)
        
        if c2 < best_c:
            best_c = c2
            best_h = h2_bin.copy()
    
    # Phase 5: Continuous L-BFGS polish at best resolution (~100s)
    n_final = len(best_h)
    if time.time() < deadline - 60:
        t5 = min(time.time() + 60, deadline - 20)
        configs = [(500, 200.0), (2000, 1000.0), (8000, 5000.0)]
        best_h, c5 = lbfgs_optimize(best_h, n_final, t5, configs=configs)
        if c5 < best_c:
            best_c = c5
    
    # Phase 6: SLSQP polish (~30s)
    if time.time() < deadline - 25:
        t6 = min(time.time() + 20, deadline - 5)
        h6, c6 = slqp_polish(best_h, n_final, t6, alpha=3000000, maxiter=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        else:
            best_c = compute_c5(best_h, n_final)
    
    # Phase 7: Final targeted refinement
    if time.time() < deadline - 10:
        best_h, best_c = targeted_refine(best_h, n_final, deadline - 3, n_rounds=20)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
