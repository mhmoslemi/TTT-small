from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=(100, 500, 2000, 10000, 50000, 200000)):
    """L-BFGS with adaptive alpha, no mu penalty (use projection instead)."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        h = project_feasible(h, n)
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    
    return h, best_c

def binary_local_search(h, n, deadline, max_iters=50000):

        j = zeros[rng.randint(len(zeros))]

        # For each shift k, the change in c[k] depends on h[i], h[j], h[i±k], h[j±k]

            ones = np.append(ones, j)

            zeros = zeros[zeros != j]

            accepted += 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(x, n, alpha, top_k=None):
    """Gradient of log-sum-exp smoothed max. If top_k given, only use top-k shifts."""
    c = compute_c5_vec(x, n)
    m = c.max()
    if top_k is not None and top_k < len(c):
        # Zero out all but top-k
        thresh = np.sort(c)[-top_k]
        mask = c >= thresh - 1e-12
        c_work = np.where(mask, c, m - 1.0/alpha)  # push non-top-k far below max
    else:
        c_work = c
    sh = alpha * (c_work - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None, top_k=None):
    if alphas is None:
        alphas = (200, 1000, 5000, 20000, 100000, 500000)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h = project_feasible(h, n)
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, top_k), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    
    return h, best_c

def subgradient_polish(h, n, deadline, top_k=5, n_steps=5000):
    """Projected subgradient method focusing on top-k worst shifts."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for step in range(n_steps):
        if time.time() > deadline - 5:
            break
        
        c = compute_c5_vec(h, n)
        # Get top-k shifts
        top_indices = np.argsort(c)[-top_k:]
        
        # Compute average gradient of top-k shifts
        grad = np.zeros(n)
        for idx in top_indices:
            k = idx - (n - 1)
            grad += smooth_grad_single_shift(h, n, k)
        grad /= top_k
        
        # Adaptive step size
        step_size = 0.01 / (1 + step * 0.001)
        
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = best_h  # revert
    
    return best_h, best_c

def smooth_grad_single_shift(h, n, k):
    """Gradient of c_k (overlap at shift k) with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += dx * (1.0 - h[0:nv])
            grad[0:nv] += dx * (-h[kk:n])
    return grad

def binary_search_polish(h, n, deadline, max_iters=100000):
    """Binary local search: swap 1s and 0s to reduce C5."""
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:n//2]] = 1.0
    h_bin = project_feasible(h_bin, n)
    
    dx = 2.0 / n
    best_c = compute_c5(h_bin, n)
    c_vec = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_max = c_vec.max()
    
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    
    rng = np.random.RandomState(123)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        
        c_new_vec = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_max = c_new_vec.max()
        
        if new_max < best_max - 1e-15:
            best_max = new_max
            best_c = new_max * dx
            # Update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # Revert
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return h_bin, best_c

def semibinary_push(h, n, t=0.3):
    """Push values toward 0 and 1 with parameter t (0=preserve, 1=hard binary)."""
    # For t in (0,1): h_new = (1-t)*h + t*binary(h)
    h_bin = (h > 0.5).astype(float)
    # Adjust to maintain sum
    target = n / 2.0
    h_new = (1 - t) * h + t * h_bin
    h_new = project_feasible(h_new, n)
    return h_new

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
