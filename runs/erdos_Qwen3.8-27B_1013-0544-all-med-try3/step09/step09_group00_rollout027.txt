def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary, maintaining sum = n/2."""
    hb = np.zeros(n)
    # Sort indices by h value, assign 1 to top n/2
    idx = np.argsort(-h)
    n_ones = n // 2
    hb[idx[:n_ones]] = 1.0
    # Handle rounding: if n is odd, we need sum = n/2 which might not be integer
    # But n should be even for this to work cleanly
    if abs(hb.sum() - n/2) > 1e-10:
        # Adjust
        diff = n/2.0 - hb.sum()
        if diff > 0:
            # Need more ones
            zeros = np.where(hb == 0)[0]
            if len(zeros) > 0:
                hb[zeros[0]] = diff  # partial fill
        else:
            ones = np.where(hb > 0)[0]
            if len(ones) > 0:
                hb[ones[-1]] = max(0, hb[ones[-1]] + diff)
    return hb

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    # If not enough (shouldn't happen for irrational alpha), fill randomly
    if len(positions) < n_ones:
        remaining = n_ones - len(positions)
        zeros = np.where(h == 0)[0]
        extra = np.random.choice(zeros, size=min(remaining, len(zeros)), replace=False)
        for e in extra:
            positions.add(e)
    for pos in positions:
        h[pos] = 1.0
    return h

**Smarter local search**: Instead of trying all swaps, use the gradient to identify promising swaps.

The gradient of c[k] with respect to h:
∂c[k]/∂h[j] = (1-h[j+k]) - h[j-k] (for valid indices)

So for a given active set of k values (those near the max), the combined gradient is:
grad[j] = sum_{k in active} w_k * ((1-h[j+k]) - h[j-k])

This tells us:
- If grad[j] > 0: decreasing h[j] reduces the objective
- If grad[j] < 0: increasing h[j] reduces the objective

So the best swap is approximately: take i from where grad is most positive (decrease h[i]) and j from where grad is most negative (increase h[j]).

This is O(n * |active|) per iteration, which is much faster.

def grad_guided_swap(h, n, deadline, max_iters=100000):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    iters = 0
    no_improve = 0

        c_max = c_vec.max()

        threshold = c_max - max(1e-9, 0.005 * c_max)

        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)

        pos_mask = best_h > 0.5  # currently 1, want to decrease

        best_j = -1

        # Change in objective ≈ -grad[i] (decreasing h[i] by 1) + grad[j] (increasing h[j] by 1)
        # Wait: if we decrease h[i], the change is -grad[i] (times 1 for the full step)
        # If we increase h[j], the change is +grad[j]
        # Total change ≈ -grad[i] + grad[j]
        # We want this to be negative, so grad[i] > grad[j], i.e., grad[i] - grad[j] > 0

        candidates_j = np.where(neg_mask)[0]

            break

        score_i = grad[candidates_i]  # want large positive
        score_j = grad[candidates_j]  # want large negative (small)
        
        # Best: max score_i - min score_j
        # But we should check actual improvement
        top_i = candidates_i[np.argsort(-score_i)[:50]]
        bot_j = candidates_j[np.argsort(score_j)[:50]]

            for j in bot_j:

                    best_i = i
                    best_j = j

            no_improve += 1
            if no_improve > 50:
                break
            # Random perturbation
            i = candidates_i[np.random.randint(len(candidates_i))]
            j = candidates_j[np.random.randint(len(candidates_j))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
            best_c = compute_c5(best_h, n)

        best_c = compute_c5(best_h, n)
        no_improve = 0
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_flat(c, alpha, x, n):
    """Gradient of smooth max of c with respect to x."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    grad = smooth_max_grad_flat(c, alpha, x, n)

    return grad

def grad_swap_search(h, n, deadline, score_top=25, max_stall=40):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    iters = 0
    no_improve = 0
    max_iters = 50000

    while time.time() < deadline and iters < max_iters:
        iters += 1
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()

        threshold = c_max - max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 20:
            active = active[np.argsort(-c_vec[active])[:20]]

        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)

        pos_cand = np.where(best_h > 0.4)[0]
        neg_cand = np.where(best_h < 0.6)[0]
        if len(pos_cand) == 0 or len(neg_cand) == 0:
            break

        top_i = pos_cand[np.argsort(-grad[pos_cand])[:score_top]]
        bot_j = neg_cand[np.argsort(grad[neg_cand])[:score_top]]

        best_gain = 0
        best_i = -1
        best_j = -1

        for i in top_i:
            gi = grad[i]
            if gi <= 0:
                continue
            for j in bot_j:
                if i == j:
                    continue
                gj = grad[j]
                if gi - gj < 0.001:
                    continue
                old_i, old_j = best_h[i], best_h[j]
                best_h[i], best_h[j] = old_j, old_i
                c_new = compute_c5(best_h, n)
                best_h[i], best_h[j] = old_i, old_j
                gain = c_max - c_new
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j

        if best_i < 0:
            no_improve += 1
            if no_improve > max_stall:
                break
            if no_improve % 10 == 0:
                kick_n = min(3, len(pos_cand), len(neg_cand))
                for _ in range(kick_n):
                    ii = pos_cand[np.random.randint(len(pos_cand))]
                    jj = neg_cand[np.random.randint(len(neg_cand))]

def sim_anneal(h, n, deadline, max_iters=30000):

    while time.time() < deadline and iters < max_iters:
        iters += 1
        frac = iters / max_iters
        t = t_start * (t_end / t_start) ** frac

        pos = np.where(cur_h > 0.5)[0]
        neg = np.where(cur_h < 0.5)[0]
        if len(pos) == 0 or len(neg) == 0:
            break

        i = pos[np.random.randint(len(pos))]
        j = neg[np.random.randint(len(neg))]
        if i == j:
            continue

        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h, n)

                best_h = cur_h.copy()

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 20 * n):

        positions.add(pos)
        if len(positions) >= n_ones:
            break
    if len(positions) < n_ones:
        zeros = np.where(h == 0)[0]
        extra = np.random.RandomState(0).choice(zeros, size=min(n_ones - len(positions), len(zeros)), replace=False)
        for e in extra:
            positions.add(e)
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = make_beatty_start(500, (1.0 - 1.0 / np.sqrt(5.0)) / 2.0)
        n = len(h0)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Try a few Beatty starts to see if any beat initial
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alternates = [
        make_beatty_start(n, 1.0 / golden),
        make_beatty_start(n, np.sqrt(2.0) / 2.0),
        make_beatty_start(n, (np.sqrt(5.0) - 1.0) / 4.0),
    ]
    for ha in alternates:
        ca = compute_c5(ha, n)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 1: L-BFGS warmup (~80s)
    p1_end = t0 + 80
    h1, c1 = run_lbfgs(best_h, n, p1_end,
        alphas=[10, 50, 200, 1000, 5000, 50000, 200000],
        mu=0.01, maxit_list=[800, 1000, 1200, 1500, 2000, 2500, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Multi-active set L-BFGS (~150s)
    if time.time() < deadline - 450:
        p2_end = min(time.time() + 150, deadline - 430)
        h2, c2 = multi_active_opt(best_h, n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Penalty method (~100s)
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 100, deadline - 340)
        h3, c3 = penalty_optimize(best_h, n, p3_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Gradient-guided binary swap (~180s)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 180, deadline - 240)
        h4, c4 = grad_swap_search(best_h, n, p4_end, score_top=30, max_stall=50)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: L-BFGS refinement from binary (~60s)
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 60, deadline - 175)
        h5, c5 = run_lbfgs(best_h, n, p5_end,
            alphas=[100000, 500000, 2000000], mu=0.005,
            maxit_list=[1000, 1500, 2000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: More swap search (~100s)
    if time.time() < deadline - 130:
        p6_end = min(time.time() + 100, deadline - 125)
        h6, c6 = grad_swap_search(best_h, n, p6_end, score_top=20, max_stall=30)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Simulated annealing (~80s)
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 80, deadline - 55)
        h7, c7 = sim_anneal(best_h, n, p7_end, max_iters=20000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final L-BFGS polish (~30s)
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 30, deadline - 15)
        h8, c8 = run_lbfgs(best_h, n, p8_end,
            alphas=[500000, 2000000], mu=0.003, maxit_list=[800, 1000])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Fine coordinate search (~15s)
    if time.time() < deadline - 3:
        p9_end = deadline - 2
        h9, c9 = fine_coord_search(best_h, n, p9_end, radius=20,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
