    if mx > 50:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() < target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = hp < 0.99999999
        else:
            mask = hp > 1e-8
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp


def sms_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500.0:
        s = s - mx
    es = np.exp(s)
    return float(m + np.log(es.sum()) / alpha)


def sms_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500.0:
        s = s - mx
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum

    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - nm1
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        elif -n < lag < 0:
            m2 = -lag
            grad[0:n - m2] += wk * (1.0 - h[m2:n])
            grad[m2:n] += wk * (-h[0:n - m2])
    return grad


def lbfgs_opt(h, n, alpha, deadline, maxiter=2000, penalty=1e12):
    target = n / 2.0

    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full')
        v = sms_val(c, alpha)
        d = x.sum() - target
        v += penalty * d * d
        g = sms_grad(x, n, alpha)
        g += 2.0 * penalty * d
        return v, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'maxls': 50, 'iprint': -1})
        out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        out = h.copy()
    return project_feasible(out, n)


def interp_to(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def local_search(h, n, deadline, rng, n_iterations=80000):
    h = project_feasible(h.copy(), n)

    it = 0

    max_stuck = 20000

        m = abs(lag)

        i = rng.randint(0, n - m)

            d = min(rng.uniform(0.002, 0.06), (h[i] - h[j]) * 0.6)
            if d < 1e-10:

            h[i] -= d
            h[j] += d
            sign_flag = 'ij'
        elif h[j] > h[i] + 1e-9:

            h[i] += d
            h[j] -= d
            sign_flag = 'ji'
        else:

        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)

            c_all = c_new

                h[j] -= d
            else:

                h[j] += d
            h = project_feasible(h, n)

            h = project_feasible(h, n)
            c_all = compute_all_lags(h, n)
            stuck = 0
