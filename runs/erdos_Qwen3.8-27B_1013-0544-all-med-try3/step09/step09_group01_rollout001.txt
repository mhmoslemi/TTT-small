    c_vec = compute_c5_vec(best_h, n)
    c_max = c_vec.max()
    active = np.where(c_vec >= c_max - 0.001)[0]
    
    important_coords = set()
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            important_coords.update(range(0, n - k))
            important_coords.update(range(k, n))
        else:
            kk = -k
            important_coords.update(range(0, n - kk))
            important_coords.update(range(kk, n))
    
    imp_list = list(important_coords)
    rng = np.random.RandomState(12345)
    
    delta_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in delta_schedule:
        n_trials = int(min(200000, max(5000, (deadline - time.time()) * 500)))
        for trial in range(n_trials):
            if time.time() > deadline:
                break
            if len(imp_list) < 2:
                break
            i = imp_list[rng.randint(len(imp_list))]
            j = imp_list[rng.randint(len(imp_list))]
            if i == j:
                continue
            if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                best_h[i] += d
                best_h[j] -= d
                c = compute_c5(best_h, n)
                if c < best_c - 1e-16:
                    best_c = c
                else:
                    best_h[i] -= d
                    best_h[j] += d
            if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                best_h[i] -= d
                best_h[j] += d
                c = compute_c5(best_h, n)
                if c < best_c - 1e-16:
                    best_c = c
                else:
                    best_h[i] += d
                    best_h[j] -= d
            if trial % 50000 == 0 and trial > 0:
                c_vec = compute_c5_vec(best_h, n)
                c_max = c_vec.max()
                active = np.where(c_vec >= c_max - 0.001)[0]
                important_coords = set()
                for idx in active:
                    k = idx - (n - 1)
                    if k >= 0:
                        important_coords.update(range(0, n - k))
                        important_coords.update(range(k, n))
                    else:
                        kk = -k
                        important_coords.update(range(0, n - kk))
                        important_coords.update(range(kk, n))
                imp_list = list(important_coords)
    
    return best_h, best_c

def subgrad_line(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        weights = c_vec[active] - (c_max - tol)
        weights = np.exp(weights * 1000)
        weights /= weights.sum()
        
        g = grad_direct(best_h, n, c_vec, np.zeros(2*n-1))
        w_full = np.zeros(2 * n - 1)
        w_full[active] = weights
        g = grad_direct(best_h, n, c_vec, w_full)
        
        h_new = best_h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.01)
        else:
            eta *= 0.97
        if eta < 1e-6:
            break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def jac(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-15:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        return grad
    
    def coneq(x):
        return x.sum() - n / 2.0
    def coneq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': coneq, 'jac': coneq_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=cons,
                      options={'maxiter': 1000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 3))
        
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: Quick L-BFGS at original resolution (200s)
    p1_end = t0 + 180
    if time.time() < p1_end:
        h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
            alphas=[500, 5000, 50000, 200000, 1000000],
            mu=0.01, maxiters=[1500, 2000, 2500, 3000, 3000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Active-set at original resolution (150s)
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 150, deadline - 500)
        h2, c2 = active_set_optimize(best_h, best_n, p2_end, top_k=8,
            alphas=[10000, 50000, 200000, 500000, 2000000, 5000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to 2x resolution and optimize (300s)
    n_fine = n0 * 2
    if n_fine > 4000:
        n_fine = 4000
    if time.time() < deadline - 400:
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p3_end = min(time.time() + 300, deadline - 350)
        h3, c3 = lbfgs_phase(h_fine, n_fine, p3_end,
            alphas=[100, 1000, 5000, 50000, 200000, 1000000],
            mu=0.005, maxiters=[2000, 2500, 3000, 3000, 4000, 4000])
        
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_fine
    
    # Phase 4: Active-set at best resolution (200s)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 200, deadline - 250)
        h4, c4 = active_set_optimize(best_h, best_n, p4_end, top_k=8,
            alphas=[50000, 200000, 500000, 2000000, 5000000, 10000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Subgradient refinement (100s)
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = subgrad_line(best_h, best_n, p5_end, n_iter=8000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: SLSQP polish (80s)
    if time.time() < deadline - 80:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 75:
                break
            p6_end = min(time.time() + 35, deadline - 70)
            h6, c6 = slqp_polish(best_h, best_n, p6_end, alpha=alpha)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Aggressive local coordinate search (remaining)
    if time.time() < deadline - 10:
        p7_end = deadline - 5
        h7, c7 = aggressive_local(best_h, best_n, p7_end)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final coordinate descent pass
    if time.time() < deadline - 3:
        h8, c8 = coordinate_descent(best_h, best_n, deadline - 2, radius=50, n_passes=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final check
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
