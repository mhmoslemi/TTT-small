    val, _ = softmax_value_and_weights(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_smooth_jac(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val, _ = softmax_value_and_weights(c, alpha)
    g = grad_from_weights(x, n, _)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return float(val), g

def topk_obj_jac(x, n, topk, alpha):
    c = compute_c5_vec(x, n)
    top_idx = np.argsort(c)[::-1][:topk]
    top_vals = c[top_idx]
    m = top_vals.max()
    sh = alpha * (top_vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    else:
        val = m + float(np.log(np.sum(np.exp(sh)))) / alpha
    w_top = np.exp(sh - mx)
    w_top /= w_top.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w_full[idx] = w_top[i]
    g = grad_from_weights(x, n, w_full)
    return float(val), g

def hard_max_step(h, n, eta, deadline):
    """Take a gradient step on the single worst shift."""
    c = compute_c5_vec(h, n)
    k_star = int(np.argmax(c))
    w = np.zeros(2 * n - 1)
    w[k_star] = 1.0
    g = grad_from_weights(h, n, w)
    for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 0.05]:
        h_new = h - trial_eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new
    return h, compute_c5(h, n)

def run_lbfgs_aggressive(h, n, deadline, alphas=(10000.0, 50000.0, 200000.0, 500000.0, 1000000.0, 5000000.0)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu = 10000.0
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(obj_smooth_jac, best_h, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run_topk(h, n, deadline, topks=(1, 2, 3, 5, 10, 20)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for topk in topks:
        if time.time() > deadline - 20:
            break
        for alpha in [5000.0, 20000.0, 100000.0]:
            if time.time() > deadline - 15:
                break
            try:
                res = minimize(topk_obj_jac, best_h, args=(n, topk, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-13, 'maxfun': 20000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000.0, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        val, _ = softmax_value_and_weights(c, al)
        return val
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_perturb_and_refine(h, n, deadline, n_trials=12):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(99)
    mags = [0.01, 0.02, 0.03, 0.05, 0.015, 0.025, 0.04, 0.06, 0.01, 0.035, 0.05, 0.02]
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        # Structured perturbation: perturb in blocks
        block_size = max(1, n // 50)
        h_p = h.copy()
        for b in range(0, n, block_size):
            h_p[b:b+block_size] += mag * (rng.rand(block_size) - 0.5) * 2
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(obj_smooth_jac, h_p, args=(n, 50000.0, 10000.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-16, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def worst_shift_descent(h, n, deadline, n_steps=200):
    """Iteratively reduce the worst shift."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    eta = 0.01
    for step in range(n_steps):
        if time.time() > deadline - 10:
            break
        c = compute_c5_vec(best_h, n)
        k_star = int(np.argmax(c))
        w = np.zeros(2 * n - 1)
        w[k_star] = 1.0
        g = grad_from_weights(best_h, n, w)
        # Line search
        best_h_trial = best_h
        best_c_trial = best_c
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = best_h - eta * factor * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_trial:
                best_h_trial = h_try
                best_c_trial = c_try
        if best_c_trial < best_c:
            best_h = best_h_trial
            best_c = best_c_trial
            eta = min(eta * 1.2, 0.1)
        else:
            eta = eta * 0.8
            if eta < 1e-6:
                break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_start_search(h_best, n, deadline, n_starts=6):
    """Try different starting points near the best solution."""
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(777)
    best_h = h_best.copy()
    best_c = compute_c5(h_best, n)
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        # Different perturbation strategies
        if s == 0:
            # Small random
            h_try = h_best + 0.01 * (rng.rand(n) - 0.5)
        elif s == 1:
            # Medium random
            h_try = h_best + 0.05 * (rng.rand(n) - 0.5)
        elif s == 2:
            # Binary push
            h_try = np.where(h_best > 0.5, 1.0, 0.0).astype(float)
        elif s == 3:
            # Oscillating perturbation
            phase = rng.rand() * 2 * np.pi
            h_try = h_best + 0.03 * np.sin(10 * np.linspace(0, 2, n) + phase)
        elif s == 4:
            # Complementary pattern
            h_try = 1.0 - h_best
        else:
            # Large random restart
            h_try = rng.rand(n)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
        # Quick L-BFGS from this start
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(obj_smooth_jac, h_try, args=(n, 20000.0, 5000.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-15, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_rounds=50):
    """Focused coordinate descent on gradient-suggested coordinates."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_c5_vec(best_h, n)
        k_star = int(np.argmax(c))
        w = np.zeros(2 * n - 1)
        w[k_star] = 1.0
        g = grad_from_weights(best_h, n, w)
        # Top coordinates to try
        top_coords = np.argsort(-np.abs(g))[:20]
        improved = False
        for j in top_coords:
            if time.time() > deadline - 3:
                return best_h, best_c
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                nv = best_h[j] + delta
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                best_h[j] = nv
                c_try = compute_c5(best_h, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    improved = True
                    break
                else:
                    best_h[j] = best_h[j] - delta
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 25
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Aggressive L-BFGS at base resolution (200s)
    t1 = min(t0 + 180, deadline - 700)
    if time.time() < t1 - 10:
        best_h, best_c = run_lbfgs_aggressive(best_h, n_base, t1,
            alphas=(10000.0, 50000.0, 200000.0, 1000000.0, 5000000.0))
    
    # Phase 2: Top-k active set (120s)
    t2 = min(t0 + 300, deadline - 550)
    if time.time() < t2 - 10:
        best_h, best_c = run_topk(best_h, n_base, t2, topks=(1, 2, 3, 5, 10))
    
    # Phase 3: SLSQP polish (60s)
    t3 = min(t0 + 360, deadline - 450)
    if time.time() < t3 - 10:
        h3, c3 = slqp_polish(best_h, n_base, t3, alpha=5000000.0, maxiter=3000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Worst-shift descent (80s)
    t4 = min(t0 + 440, deadline - 380)
    if time.time() < t4 - 10:
        best_h, best_c = worst_shift_descent(best_h, n_base, t4, n_steps=1000)
    
    # Phase 5: Multi-start search (100s)
    t5 = min(t0 + 540, deadline - 300)
    if time.time() < t5 - 10:
        best_h, best_c = multi_start_search(best_h, n_base, t5, n_starts=5)
    
    # Phase 6: Binary perturbation + refine (80s)
    t6 = min(t0 + 620, deadline - 220)
    if time.time() < t6 - 10:
        best_h, best_c = binary_perturb_and_refine(best_h, n_base, t6, n_trials=8)
    
    # Phase 7: Additional aggressive optimization (100s)
    t7 = min(t0 + 720, deadline - 150)
    if time.time() < t7 - 10:
        best_h, best_c = run_lbfgs_aggressive(best_h, n_base, t7,
            alphas=(100000.0, 500000.0, 2000000.0, 10000000.0))
    
    # Phase 8: Upscale to n=1000 (120s)
    n1 = 1000
    if time.time() < deadline - 120:
        h1000 = interpolate_to(best_h, n_base, n1)
        c1000 = compute_c5(h1000, n1)
        t8 = min(time.time() + 80, deadline - 60)
        if time.time() < t8 - 10:
            h1000, c1000 = run_lbfgs_aggressive(h1000, n1, t8,
                alphas=(20000.0, 100000.0, 500000.0, 2000000.0))
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
    
    # Phase 9: Final SLSQP on best resolution (50s)
    n_final_try = len(best_h)
    if time.time() < deadline - 30:
        t9 = min(time.time() + 40, deadline - 10)
        h_final, c_final = slqp_polish(best_h, n_final_try, t9,
                                        alpha=10000000.0, maxiter=2000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Phase 10: Coordinate descent polish (20s)
    if time.time() < deadline - 5:
        best_h, best_c = coordinate_descent(best_h, len(best_h), deadline - 2, max_rounds=30)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
