```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            m = n - lag
            grad[lag:n] += wk * (1.0 - h[0:m]) * dx
            grad[0:m] += wk * (-h[lag:n]) * dx
        else:
            m = -lag
            grad[0:n-m] += wk * (1.0 - h[m:n]) * dx
            grad[m:n] += wk * (-h[0:n-m]) * dx
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e6):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 100,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_to_resolution(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def targeted_pair_search(h, n, deadline, rng, top_K=8, max_rounds=100000):

    contributes most and try to fix it."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds = 0
    stuck = 0

    while rounds < max_rounds and time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_K:]
        max_val = c_all.max()

        improved_this_round = False
        for k_idx in top_idx:
            if time.time() > deadline - 2:
                return best_h, best_c
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            # Compute per-pair contributions for this lag
            if lag > 0:
                n_pairs = n - m
                contrib = h[0:n_pairs] * (1.0 - h[m:n])
            else:
                n_pairs = n - m
                contrib = h[m:n] * (1.0 - h[0:n_pairs])

            # Find top contributing pairs
            top_pairs = np.argsort(contrib)[-min(20, n_pairs):]
            for pi in top_pairs:
                if time.time() > deadline - 2:
                    return best_h, best_c
                if lag > 0:
                    i_h = pi
                    i_1h = pi + m
                else:
                    i_h = pi + m
                    i_1h = pi

                # Try increasing the (1-h) side
                d = rng.uniform(0.001, 0.015)
                d = min(d, 1.0 - h[i_1h])
                if d < 1e-8:
                    d = 0.001
                    if h[i_1h] >= 1.0:
                        continue
                save_i1h = h[i_1h]
                h[i_1h] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    improved_this_round = True
                    stuck = 0
                    break
                else:
                    h[i_1h] = save_i1h
                    h = project_feasible(h, n)

                d = rng.uniform(0.001, 0.015)
                d = min(d, h[i_h])

                    d = 0.001
                    if h[i_h] <= 0.0:

                h[i_h] -= d

                    improved_this_round = True
                    stuck = 0
                    break

def random_pair_sa(h, n, deadline, rng, t_init=0.01, t_min=0.00001, cool=0.9999, max_iters=50000):
    """Simulated annealing with random pair swaps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it = 0
    stuck = 0

    while it < max_iters and time.time() < deadline and T > t_min:
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            T *= cool
            it += 1
            continue

        max_d = min(0.05, 0.002 + T * 3)
        if h[i] > h[j]:
            d = rng.uniform(0.002, max_d)
            d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            if d < 1e-9:
                T *= cool
                it += 1
                continue
            h[i] -= d
            h[j] += d
        else:
            d = rng.uniform(0.002, max_d)
            d = min(d, h[j] * 0.3, (1.0 - h[i]) * 0.3)
            if d < 1e-9:
                T *= cool
                it += 1
                continue
            h[i] += d
            h[j] -= d

        h = project_feasible(h, n)
        c_new = compute_c5(h, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            stuck = 0
        else:
            # Accept with probability
            if c_new > best_c:
                ratio = (c_new - best_c) / max(T, 1e-10)
                if ratio < 25 and rng.random() < np.exp(-ratio):
                    stuck = 0
                else:
                    h[i] += d if h[i] < 0.5 else -d
                    # Actually revert properly
                    if h[i] + d > 1 or h[i] - d < 0:
                        pass
                    h[i] = h[i] + d  # revert
                    h[j] = h[j] - d
                    h = project_feasible(h, n)
                    stuck += 1
            else:
                stuck = 0

        T *= cool
        it += 1

        if stuck > 3000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            stuck = 0
            T = max(T, 0.003)

    return best_h, best_c

        # Targeted local search at n=2000
        if time.time() < deadline - 50:
            ls_dl = min(deadline - 30, time.time() + 40)
            h_opt, c_opt = targeted_pair_search(h3, n3, ls_dl, rng, top_K=12)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h3 = h_opt

        # Final polish at n=2000
        if time.time() < deadline - 20:
            h_opt = lbfgsb_optimize(best_h, len(best_h), 100000000,
                                     min(deadline - 8, time.time() + 25), maxiter=2000)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # === Final: one more targeted search at best resolution ===
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt, c_opt = targeted_pair_search(best_h, n_final, deadline - 2, rng, top_K=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation and return
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
