def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    if smax > 400:
        return float(m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smax = shifted.max()
    if smax > 400:
        val = mx + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):

            grad[:nv] += wm * (-x[kk:])

    return np.clip(h, 0.0, 1.0)

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smax = shifted.max()
    if smax > 400:
        val = mx + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    shifted = alpha * (vals - mx)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx_val
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smooth_max_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    if smax > 400:
        val = m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    count = 0
    target = n // 2
    for k in range(1, 6 * n):
        pos = int(k * alpha_param) % n
        if h[pos] == 0 and count < target:
            h[pos] = 1.0
            count += 1
            if count >= target:
                break
    if count < target:
        zeros = np.where(h == 0)[0]
        need = target - count
        if len(zeros) >= need:
            h[zeros[:need]] = 1.0
    return project_feasible(h, n)

def make_fibonacci_start(n):
    h = np.zeros(n)
    target = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.zeros(target, dtype=int)
    for k in range(1, target + 1):
        pos = int(n * ((k * phi) % 1.0))
        if pos >= n:
            pos = n - 1
        positions[k-1] = pos
    # Deduplicate
    seen = set()
    count = 0
    for pos in positions:
        if pos not in seen:
            h[pos] = 1.0
            seen.add(pos)
            count += 1
    if count < target:
        zeros = np.where(h == 0)[0]
        picks = zeros[:target - count]
        h[picks] = 1.0
    return project_feasible(h, n)

def active_set_rounds(h, n, deadline, rounds=8):
    h_curr = project_feasible(h, n)
    c_curr = compute_c5(h_curr, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000, 200000000]
    top_ks = [10, 8, 7, 6, 5, 4, 3, 3]
    
    for rnd in range(rounds):
        if time.time() > deadline - 5:
            break
        alpha = alphas[rnd] if rnd < len(alphas) else 50000000
        top_k = top_ks[rnd] if rnd < len(top_ks) else 3
        
        c_vec = compute_c5_vec(h_curr, n)
        max_val = c_vec.max()
        threshold = max_val - max(0.0005, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])[:top_k]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:3]
        
        maxit = max(400, int(min(5000, (deadline - time.time()) * 4)))
        try:
            res = minimize(topK_obj, h_curr, jac=topK_grad,
                         args=(n, active, alpha, 0.001),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-22,
                                  'gtol': 1e-16, 'maxfun': maxit * 8,
                                  'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_curr - 1e-16:
                h_curr = h_new
                c_curr = c_new
        except Exception:
            pass
    
    return h_curr, c_curr

def binary_swap_anneal(h, n, deadline, seed_sw=777, max_iters=80000):
    h_curr = np.where(h > 0.5, 1.0, 0.0).copy()
    h_curr = project_feasible(h_curr, n)
    c_curr = compute_c5(h_curr, n)
    
    ones_list = list(np.where(h_curr > 0.5)[0])
    zeros_list = list(np.where(h_curr < 0.5)[0])
    
    rng = np.random.RandomState(seed_sw)
    T = 0.003
    T_min = 5e-7
    cool = 0.997
    
    best_c = c_curr
    best_h = h_curr.copy()
    iters = 0
    no_improve = 0
    
    while time.time() < deadline and iters < max_iters:
        iters += 1
        i = int(rng.randint(len(ones_list)))
        j = int(rng.randint(len(zeros_list)))
        ip = ones_list[i]
        jp = zeros_list[j]
        
        h_curr[ip] = 0.0
        h_curr[jp] = 1.0
        c_new = compute_c5(h_curr, n)
        
        accepted = False
        if c_new < c_curr - 1e-15:
            accepted = True
            c_curr = c_new
            no_improve = 0
        elif c_new < c_curr + T:
            if rng.random() < np.exp((c_curr - c_new) / max(T, 1e-10)):
                accepted = True
                c_curr = c_new
                no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if accepted:
            ones_list[i] = jp
            zeros_list[j] = ip
        else:
            h_curr[ip] = 1.0
            h_curr[jp] = 0.0
        
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        
        if iters % 2000 == 0:
            T = max(T * cool, T_min)
        
        if no_improve > 30000 and T <= T_min * 2:
            break
    
    best_h = np.where(best_h > 0.5, 1.0, 0.0)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def smart_swap_pass(h, n, deadline, max_rounds=50):
    """Try targeted swaps: for each 1, find best 0 to swap with."""
    h_curr = np.where(h > 0.5, 1.0, 0.0).copy()
    h_curr = project_feasible(h_curr, n)
    c_curr = compute_c5(h_curr, n)
    c_vec = compute_c5_vec(h_curr, n)
    max_c = c_vec.max()
    
    ones_set = set(np.where(h_curr > 0.5)[0].tolist())
    zeros_set = set(np.where(h_curr < 0.5)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones_list = list(ones_set)
        zeros_list = list(zeros_set)
        np.random.shuffle(ones_list)
        
        for i in ones_list[:min(len(ones_list), 100)]:
            if time.time() > deadline:
                break
            best_j = -1
            best_c_val = c_curr
            h_curr[i] = 0.0
            for j in zeros_list[:min(len(zeros_list), 30)]:
                h_curr[j] = 1.0
                c_test = compute_c5(h_curr, n)
                if c_test < best_c_val - 1e-15:
                    best_c_val = c_test
                    best_j = j
                h_curr[j] = 0.0
            h_curr[i] = 1.0
            
            if best_j >= 0:
                h_curr[i] = 0.0
                h_curr[best_j] = 1.0
                h_curr = project_feasible(h_curr, n)
                c_curr = best_c_val
                ones_set.remove(i)
                ones_set.add(best_j)
                zeros_set.remove(best_j)
                zeros_set.add(i)
                improved = True
    
        if not improved:
            break
    
    h_curr = np.where(h_curr > 0.5, 1.0, 0.0)
    h_curr = project_feasible(h_curr, n)
    c_curr = compute_c5(h_curr, n)
    return h_curr, c_curr

def continuous_optimize(h, n, deadline):
    h_curr = project_feasible(h, n)
    c_curr = compute_c5(h_curr, n)
    bounds = [(0.0, 1.0)] * n
    
    # Smooth max L-BFGS at various alphas
    for alpha in [100, 500, 2000, 10000, 50000]:
        if time.time() > deadline - 5:
            break
        maxit = max(500, int(min(3000, (deadline - time.time()) * 3)))
        try:
            res = minimize(smooth_max_func, h_curr, jac=smooth_max_grad,
                         args=(n, alpha, 0.01),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-22,
                                  'gtol': 1e-14, 'maxfun': maxit * 10,
                                  'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_curr:
                h_curr = h_new
                c_curr = c_new
        except Exception:
            pass
    
    # Active set refinement
    h_curr, c_curr = active_set_rounds(h_curr, n, deadline, rounds=10)
    
    return h_curr, c_curr

def multi_start_optimize(n, t0, deadline, seed=42):
    best_h = None
    best_c = float('inf')
    
    starts = []
    # Beatty starts with different irrationals
    beatty_alphas = [
        (1.0 + np.sqrt(5.0)) / 2.0 / 2.0,
        np.sqrt(2.0) / 2.0,
        np.sqrt(3.0) / 2.0,
        0.381966,
        0.618034,
        0.414214,
        (3.0 - np.sqrt(5.0)) / 2.0,
    ]
    for a in beatty_alphas[:5]:
        starts.append(make_beatty_start(n, a))
    
    # Fibonacci start
    starts.append(make_fibonacci_start(n))
    
    # Random starts
    rng = np.random.RandomState(seed)
    for _ in range(2):
        h_r = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_r[idx] = 1.0
        starts.append(project_feasible(h_r, n))
    
    # Score and sort
    scored = []
    for h_s in starts:
        c_s = compute_c5(h_s, n)
        scored.append((c_s, h_s))
    scored.sort(key=lambda x: x[0])
    
    time_budget_per = (deadline - t0 - 20) / float(len(scored))
    
    for c_s, h_s in scored:
        if time.time() > deadline - 15:
            break
        alloc = min(time.time() + time_budget_per * 1.5, deadline - 10)
        
        if c_s >= best_c and best_h is not None:
            alloc = min(alloc, time.time() + 20)
        
        # Continuous optimization
        h_opt, c_opt = continuous_optimize(h_s, n, alloc)
        
        # Binary rounding and swap
        h_bin = np.where(h_opt > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < c_opt:
            h_opt = h_bin
            c_opt = c_bin
        
        if time.time() < deadline - 15:
            h_swap, c_swap = binary_swap_anneal(h_opt, n, deadline - 10,
                                               seed_sw=seed + n)
            if c_swap < c_opt:
                h_opt = h_swap
                c_opt = c_swap
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 500
    
    overall_best_h = None
    overall_best_c = float('inf')
    overall_best_n = 0
    
    # Process parent at its resolution
    if h_parent is not None:
        h_p = project_feasible(h_parent, n_parent)
        c_p = compute_c5(h_p, n_parent)
        overall_best_c = c_p
        overall_best_h = h_p.copy()
        overall_best_n = n_parent
        
        # Optimize parent
        if time.time() < deadline - 200:
            h_opt, c_opt = continuous_optimize(h_p, n_parent, deadline - 150)
            h_bin = np.where(h_opt > 0.5, 1.0, 0.0)
            h_bin = project_feasible(h_bin, n_parent)
            c_bin = compute_c5(h_bin, n_parent)
            if c_bin < c_opt:
                h_opt = h_bin
                c_opt = c_bin
            if time.time() < deadline - 30:
                h_swap, c_swap = binary_swap_anneal(h_opt, n_parent, deadline - 10,
                                                   seed_sw=seed)
                if c_swap < c_opt:
                    h_opt = h_swap
                    c_opt = c_swap
            if c_opt < overall_best_c:
                overall_best_c = c_opt
                overall_best_h = h_opt.copy()
                overall_best_n = n_parent
    
    # Try n=1000 (higher resolution)
    if time.time() < deadline - 150:
        n_try = 1000
        # Build start from parent if available
        if h_parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 2, n_parent)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, h_parent)
            h_interp = project_feasible(h_interp, n_try)
            c_interp = compute_c5(h_interp, n_try)
            if c_interp < overall_best_c:
                overall_best_c = c_interp
                overall_best_h = h_interp.copy()
                overall_best_n = n_try
        
        # Multi-start at n=1000
        h_1000, c_1000 = multi_start_optimize(n_try, t0, deadline - 30, seed=seed + 1)
        if h_1000 is not None and c_1000 < overall_best_c:
            overall_best_c = c_1000
            overall_best_h = h_1000.copy()
            overall_best_n = n_try
    
    # Try n=500 with multiple starts (if not already done well)
    if time.time() < deadline - 80:
        h_500, c_500 = multi_start_optimize(500, t0, deadline - 10, seed=seed + 2)
        if h_500 is not None and c_500 < overall_best_c:
            overall_best_c = c_500
            overall_best_h = h_500.copy()
            overall_best_n = 500
    
    # Final polish on best
    if overall_best_h is not None and time.time() < deadline - 8:
        n_fin = len(overall_best_h)
        h_fin = overall_best_h.copy()
        c_fin = overall_best_c
        
        # Quick active set
        if time.time() < deadline - 5:
            h_fin, c_fin = active_set_rounds(h_fin, n_fin, deadline - 3, rounds=4)
        
        # Try binary
        h_bin_f = np.where(h_fin > 0.5, 1.0, 0.0)
        h_bin_f = project_feasible(h_bin_f, n_fin)
        c_bin_f = compute_c5(h_bin_f, n_fin)
        if c_bin_f < c_fin:
            h_fin = h_bin_f
            c_fin = c_bin_f
        
        # Smart swap
        if time.time() < deadline - 3:
            h_sw, c_sw = smart_swap_pass(h_fin, n_fin, deadline - 1)
            if c_sw < c_fin:
                h_fin = h_sw
                c_fin = c_sw
        
        if c_fin < overall_best_c:
            overall_best_c = c_fin
            overall_best_h = h_fin
            overall_best_n = n_fin
    
    # Final safety
    if overall_best_h is None:
        overall_best_h = make_beatty_start(500, (1.0 + np.sqrt(5.0)) / 4.0)
        overall_best_n = 500
    
    overall_best_h = project_feasible(overall_best_h, overall_best_n)
    overall_best_c = compute_c5(overall_best_h, overall_best_n)
    
    return overall_best_h, overall_best_c, overall_best_n
