def coord_descent(h, n, deadline, step_sizes=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for ds in step_sizes:
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                for sign in [1, -1]:
                    nl = best_h[i] + sign * ds
                    if nl < 0 or nl > 1:
                        continue
                    # Find a j to transfer to/from
                    if sign > 0:
                        # h[i] increases, need to decrease some h[j]
                        j_candidates = np.where(best_h > ds + 1e-12)[0]
                        if len(j_candidates) == 0:
                            continue
                        j = j_candidates[np.argmin(best_h[j_candidates])]
                    else:
                        j_candidates = np.where(best_h < 1.0 - ds - 1e-12)[0]
                        if len(j_candidates) == 0:
                            continue
                        j = j_candidates[np.argmax(best_h[j_candidates])]
                    test_h = best_h.copy()
                    test_h[i] += sign * ds
                    test_h[j] -= sign * ds
                    test_h = project_feasible(test_h, n)
                    c_new = compute_c5(test_h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        best_h = test_h
                        improved = True
                        break
    return best_h, best_c

def make_periodic_start(n, period, ones_per_period):

    # Phase 1: Quick optimization at n0 (up to 80s)
    p1_end = t0 + min(80, (deadline - t0) * 0.08)
    if n0 <= 600:
        h1, c1 = run_lbfgs(best_h, n0, p1_end,
            alphas=[100, 1000, 10000, 100000, 1000000],
            mu=0.01, maxiter_per=[1000, 1500, 2000, 2000, 2500])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=1000 and main optimization
    n_main = 1000
    if time.time() < deadline - 500:
        h_main = upsample_h(best_h, n_main)
        c_main = compute_c5(h_main, n_main)
        best_h_main = h_main.copy()
        best_c_main = c_main

        # L-BFGS at n=1000 with multiple alpha levels
        p2_alphas = [50, 200, 1000, 5000, 50000, 500000, 5000000]
        p2_iters = [1000, 1500, 2000, 2500, 3000, 3000, 3000]
        p2_end = min(time.time() + 350, deadline - 250)
        h2, c2 = run_lbfgs(best_h_main, n_main, p2_end,
            alphas=p2_alphas, mu=0.005, maxiter_per=p2_iters)
        if c2 < best_c_main:
            best_c_main = c2
            best_h_main = h2

        # Active set refinement at n=1000
        p3_end = min(time.time() + 150, deadline - 100)
        h3, c3 = run_active_refine(best_h_main, n_main, p3_end,
            top_k=8, rounds=6)
        if c3 < best_c_main:
            best_c_main = c3
            best_h_main = h3

        if best_c_main < best_c:
            best_c = best_c_main
            best_h = best_h_main.copy()
            best_n = n_main

    # Phase 3: Try different initializations at n_main
    if time.time() < deadline - 200:
        n_try = best_n if best_n >= 800 else 1000
        starts = []
        # Binary start from best
        starts.append(make_binary_start(best_h, n_try))
        # Periodic starts
        for period, ones in [(7,3), (7,4), (9,4), (9,5), (11,5), (13,6), (13,7)]:
            if n_try % period == 0:
                starts.append(make_periodic_start(n_try, period, ones))
        # Smoothed binary
        h_sb = make_binary_start(best_h, n_try)
        # Blur it
        from scipy.ndimage import uniform_filter1d
        h_blur = uniform_filter1d(h_sb, size=5)
        h_blur = project_feasible(h_blur, n_try)
        starts.append(h_blur)

        for hs in starts[:4]:
            if time.time() > deadline - 60:
                break
            c_s = compute_c5(hs, n_try)
            if c_s >= best_c + 0.01:
                continue
            # Quick optimization
            p_end = min(time.time() + 60, deadline - 40)
            h_opt, c_opt = run_lbfgs(hs, n_try, p_end,
                alphas=[1000, 50000, 1000000],
                mu=0.005, maxiter_per=[1000, 2000, 2000])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 4: More active-set refinement at best resolution
    if time.time() < deadline - 120:
        p4_end = min(time.time() + 100, deadline - 80)
        h4, c4 = run_active_refine(best_h, best_n, p4_end,
            top_k=6, rounds=8)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Subgradient refinement
    if time.time() < deadline - 60:
        p5_end = min(time.time() + 50, deadline - 40)
        h5, c5 = subgrad_refine(best_h, best_n, p5_end, n_iter=5000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another L-BFGS round with very high alpha
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 30, deadline - 25)
        h6, c6 = run_lbfgs(best_h, best_n, p6_end,
            alphas=[10000000, 50000000],
            mu=0.001, maxiter_per=[2000, 1500])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final active-set polish
    if time.time() < deadline - 15:
        p7_end = min(time.time() + 12, deadline - 5)
        h7, c7 = run_active_refine(best_h, best_n, p7_end,
            top_k=5, rounds=3)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final: reproject and verify
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    # Ensure sum constraint is met exactly
    target = final_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_feasible(best_h, final_n)

    return best_h, best_c, final_n
