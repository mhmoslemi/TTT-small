def subgradient_method(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.005
    active_frac = 0.02
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        if max_val < best_c - 1e-14:
            best_c = max_val
            best_h = h.copy()
        
        threshold = max_val - active_frac * max_val
        active = np.where(c_vec >= threshold)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            dx = 2.0 / n
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Try several step sizes
        improved = False

                improved = True
                break

            alpha *= 0.9
            if alpha < 1e-6:

def local_pair_search(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline:
            break
        changed = True
        while changed:
            changed = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step; h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; changed = True
                        else:
                            h[i] += step; h[j] -= step
                    # Try moving mass from j to i
                    if h[j] - step >= 0.0 and h[i] + step <= 1.0:
                        h[j] -= step; h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; changed = True
                        else:
                            h[j] += step; h[i] -= step
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_shifts(h, n, frac=0.01, top_k=20):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-6, frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def smooth_max_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if k >= 0:
            nv = n - k

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n

        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def opt_active_set(h, n, deadline, active, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, active, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 5,
                               'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

    except Exception:

def subgradient_method(h, n, deadline, max_iter=50000, active_frac=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.003
    dx_val = 2.0 / n

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()

        if max_val < best_c - 1e-14:
            best_c = max_val
            best_h = h.copy()

        threshold = max_val - active_frac * max_val
        active = np.where(c_vec >= threshold)[0]

        if len(active) == 0:
            break

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx_val
                grad[k:n] += (-h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)

        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = alpha * mult
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < max_val - 1e-16:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.92
            if alpha < 1e-8:
                break
        else:
            alpha = min(alpha * 1.02, 0.01)

    return best_h, best_c

def local_pair_search_fast(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]

    for step in steps:
        if time.time() > deadline:
            break
        changed = True
        while changed:
            changed = False
            if time.time() > deadline:
                return h, best_c
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            changed = True
                        else:
                            h[i] += step
                            h[j] -= step
                    elif h[j] - step >= 0.0 and h[i] + step <= 1.0:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            changed = True
                        else:
                            h[j] += step
                            h[i] -= step
    return h, best_c

def gradient_driven_local(h, n, deadline, base_step=0.001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    active_frac = 0.005
    dx_val = 2.0 / n

    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        if max_val < best_c - 1e-14:
            best_c = max_val
            best_h = h.copy()

        threshold = max_val - active_frac * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break

        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx_val
                grad[k:n] += (-h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)

        top_idx = np.argsort(-np.abs(grad))[:80]
        found = False

        for ii in range(len(top_idx)):
            if time.time() > deadline:
                return best_h, best_c
            i = top_idx[ii]
            for jj in range(len(top_idx)):
                if ii == jj:
                    continue
                j = top_idx[jj]
                gi, gj = grad[i], grad[j]
                if gi > gj:
                    di, dj = -1, 1
                else:
                    di, dj = 1, -1

                for mult in [1.0, 0.5, 0.25, 0.1]:
                    step = base_step * mult
                    if di == -1:
                        if h[i] - step >= 0 and h[j] + step <= 1:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                found = True
                            else:
                                h[i] += step
                                h[j] -= step
                    else:
                        if h[i] + step <= 1 and h[j] - step >= 0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                found = True
                            else:
                                h[i] -= step
                                h[j] += step
                    if found:
                        break
            if found:
                break

        if not found:
            base_step *= 0.5
            if base_step < 1e-7:
                break

    return best_h, best_c

def multi_restart(h, n, deadline, n_restarts=4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        rng = np.random.RandomState(1234 + r * 7919)
        scale = 0.015 * (1.0 + 0.5 * r)
        pert = rng.normal(0, scale, n)
        h_pert = project_feasible(h + pert, n)

        for alpha, mu in [(500, 10.0), (5000, 100.0)]:
            if time.time() > deadline - 40:
                break
            try:
                res = minimize(full_smooth_obj, h_pert, jac=full_smooth_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': 10000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h_pert = h_new
            except Exception:
                pass

    return best_h, best_c

def slqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n

    def slqp_obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        if smx > 500:
            val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)

    def slqp_constraint(x):
        return float(x.sum() - n / 2.0)

    def slqp_constraint_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': slqp_constraint, 'jac': slqp_constraint_jac}]
    try:
        res = minimize(slqp_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick optimization at original resolution
    if time.time() < deadline - 500:
        for alpha, mu in [(500, 50.0), (5000, 500.0), (20000, 5000.0)]:
            if time.time() > deadline - 500:
                break
            h_new, c_new = opt_full(best_h, n_orig, deadline - 500, alpha, mu, 3000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_hi = np.interp(x_new, x_old, best_h)
    h_hi = project_feasible(h_hi, n)
    c_hi = compute_c5(h_hi, n)
    best_h = h_hi
    best_c = min(best_c, c_hi)

    # Phase 3: Full L-BFGS at n=1000
    if time.time() < deadline - 400:
        for alpha, mu in [(100, 10.0), (500, 100.0), (2000, 1000.0), (10000, 10000.0)]:
            if time.time() > deadline - 400:
                break
            h_new, c_new = opt_full(best_h, n, deadline - 400, alpha, mu, 4000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 4: Active-set focused optimization (tighter sets)
    if time.time() < deadline - 300:
        for _ in range(6):
            if time.time() > deadline - 300:
                break
            active = get_active_shifts(best_h, n, frac=0.005, top_k=12)
            for alpha in [5000, 50000, 200000]:
                if time.time() > deadline - 300:
                    break
                h_new, c_new = opt_active_set(best_h, n, deadline - 300, active, alpha, 3000)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                active = get_active_shifts(best_h, n, frac=0.005, top_k=12)

    # Phase 5: Subgradient descent
    if time.time() < deadline - 250:
        t5 = min(time.time() + 150, deadline - 250)
        h_new, c_new = subgradient_method(best_h, n, t5, max_iter=40000, active_frac=0.015)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 6: More active-set with very tight sets
    if time.time() < deadline - 180:
        for _ in range(5):
            if time.time() > deadline - 180:
                break
            active = get_active_shifts(best_h, n, frac=0.002, top_k=6)
            for alpha in [50000, 500000, 2000000]:
                if time.time() > deadline - 180:
                    break
                h_new, c_new = opt_active_set(best_h, n, deadline - 180, active, alpha, 4000)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            active = get_active_shifts(best_h, n, frac=0.001, top_k=4)
            h_new, c_new = opt_active_set(best_h, n, deadline - 180, active, 5000000, 5000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 7: Gradient-driven local search
    if time.time() < deadline - 100:
        t7 = min(time.time() + 70, deadline - 100)
        h_new, c_new = gradient_driven_local(best_h, n, t7, base_step=0.0015)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 8: Multi-restart
    if time.time() < deadline - 80:
        t8 = min(time.time() + 50, deadline - 80)
        h_new, c_new = multi_restart(best_h, n, t8, n_restarts=3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 9: Subgradient polish with very tight active set
    if time.time() < deadline - 50:
        t9 = min(time.time() + 35, deadline - 50)
        h_new, c_new = subgradient_method(best_h, n, t9, max_iter=60000, active_frac=0.005)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 10: Fine pair search
    if time.time() < deadline - 15:
        t10 = min(time.time() + 12, deadline - 15)
        h_new, c_new = local_pair_search_fast(best_h, n, t10, radius=40)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 11: Final active-set micro-optimization
    if time.time() < deadline - 3:
        active = get_active_shifts(best_h, n, frac=0.001, top_k=3)
        h_new, c_new = opt_active_set(best_h, n, deadline - 1, active, 10000000, 3000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
