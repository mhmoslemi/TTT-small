If we fix h and only vary h[j], then c_k(h) is a quadratic in h[j]:
c_k = a_k * h[j]² + b_k * h[j] + c_k_other

So for each j, we can find the value of h[j] that minimizes max_{k∈A} (a_k * h[j]² + b_k * h[j] + c_k_other) via a 1D problem (minimize max of quadratics on [0,1]).

This is a proper coordinate descent! Let me work out the formula.

For a given j and shift k:
c_k(h) contains terms involving h[j]:
- From the "h[i]*(1-h[i+k])" structure:
  - h[j] appears in the h[i] slot: i=j, contributing h[j]*(1-h[j+k])*dx
  - h[j] appears in the (1-h[i+k]) slot: i+k=j, i.e., i=j-k, contributing h[j-k]*(1-h[j])*dx

So: c_k = ... + (1-h[j+k])*dx*h[j] - h[j-k]*dx*h[j] + ... (for the relevant j)
     = ... + h[j]*dx*(1-h[j+k]-h[j-k]) + ...

Wait, but for k=0: h[j] appears twice: h[j]*(1-h[j]) = h[j] - h[j]².
So c_0 contains -h[j]²*dx + h[j]*dx.

For k≠0:
c_k contains (1-h[j+k])*dx*h[j] from the i=j term (if valid)
and h[j-k]*dx*(1-h[j]) from the i=j-k term (if valid)

So c_k = Q_k + L_k*h[j] + (possibly) (-h[j]²*dx) for k=0

For k≠0, c_k is LINEAR in h[j]!
For k=0, c_k is QUADRATIC in h[j]: c_0 = -dx*h[j]² + dx*h[j] + Q_0

This is a key insight. For k≠0, changing h[j] changes c_k linearly. So the "max of linear functions" in h[j] is a piecewise linear function, and we can find the minimum by checking the breakpoints.

For k=0, we have one quadratic mixing in.

So the 1D subproblem for h[j] is:
min_{t∈[0,1]} max( -dx*t² + dx*t + Q_0, max_{k≠0 active} (a_k*t + b_k) )

2. The max of these is a convex piecewise linear function
3. Add the quadratic (which is concave on [0,1] since -dx*t²+dx*t has max at t=0.5)
4. The max of a concave and convex function... hmm, this is trickier.

Actually, -dx*t² + dx*t is concave (opens downward). So we're maximizing... no, we're minimizing the max.

min_t max(concave_quad(t), convex_piecewise_linear(t))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c_k_grad(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] += (1.0 - h[k:k+nv])
        if nv > 0:
            grad[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv])
        if nv > 0:
            grad[0:nv] += (-h[kk:kk+nv])
    return grad * dx

def subgradient_step(h, n, lr, max_active=30):
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    thresh = c_max - max(1e-12, 0.005 * c_max)
    active = np.where(c_vec >= thresh)[0]
    if len(active) == 0:
        return h, c_max
    if len(active) > max_active:
        idx = active[np.argsort(-c_vec[active])[:max_active]]
    else:
        idx = active
    w = c_vec[idx] - (c_max - thresh)
    w = np.maximum(w, 1e-10)
    w /= w.sum()
    grad = np.zeros(n)
    for j, kk in enumerate(idx):
        k_shift = int(kk) - (n - 1)
        grad += w[j] * c_k_grad(h, n, k_shift)
    h_new = project_feasible(h - lr * grad, n)
    return h_new, c_max

    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0

    b_list = []

        k_shift = int(idx) - (n - 1)
        g = c_k_grad(h, n, k_shift)
        A_list.append(np.append(g, -1.0))
        b_list.append(-c_vec[idx])

        b_list.append(h[i])

        r[i] = 1.0

    A_eq = np.zeros((1, n_var))

def fast_local_search(h, n, deadline, max_iter=5000):

    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for si, step in enumerate(steps):
        it = 0
        while it < max_iter and time.time() < deadline:
            it += 1
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            thresh = c_max - max(1e-9, 0.003 * c_max)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            if len(active) > 8:
                active = active[np.argsort(-c_vec[active])[:8]]
            improved = False
            for idx in active:
                if time.time() > deadline:
                    return best_h, best_c
                k_shift = int(idx) - (n - 1)
                g = c_k_grad(h, n, k_shift)
                # Find best single-pair move: increase h[a] by step, decrease h[b] by step
                # This reduces c_k if g[a] < g[b] (movement against gradient)
                # We want to move in direction -g, so increase where g is most negative, decrease where g is most positive
                pos_cand = np.argmin(g)
                neg_cand = np.argmax(g)
                a = int(pos_cand)
                b = int(neg_cand)
                if h[a] + step > 1.0 + 1e-12 or h[b] - step < -1e-12:
                    # Try adjusting step
                    s = min(step, 1.0 - h[a], h[b])
                    if s < 1e-10:
                        continue
                else:
                    s = step
                old_ha, old_hb = h[a], h[b]
                h[a] += s
                h[b] -= s
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[a] = old_ha
                    h[b] = old_hb
                    # Try reverse
                    if h[b] + s <= 1.0 + 1e-12 and h[a] - s >= -1e-12:
                        h[b] += s
                        h[a] -= s
                        h = project_feasible(h, n)

                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:

                            h[a] = old_ha
            if not improved:
                break
    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alpha, mu, max_iter=3000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)

    def grad_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(len(w)):
            if w[j] < 1e-14:
                continue
            k_shift = j - (n - 1)
            g = c_k_grad(x, n, k_shift)
            grad += w[j] * g
        grad += 2.0 * mu * (x.sum() - n / 2.0)
        return grad

    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-20,
                                'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

    t1 = min(t0 + 30, deadline - 300)
    if time.time() < t1:
        h, c = lbfgs_smooth(h, n0, t1, alpha=500, mu=100.0, max_iter=1000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Multi-start SQP + subgradient (120s)
    t2 = min(t0 + 150, deadline - 250)
    if time.time() < t2:
        h, c = multi_start_optimize(h, n0, t2, max_iter=400, rho_init=0.05)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Fast local search (150s)
    t3 = min(t0 + 300, deadline - 200)
    if time.time() < t3:
        h, c = fast_local_search(best_h, n0, t3, max_iter=8000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Another SQP round (60s)
    t4 = min(t0 + 360, deadline - 150)
    if time.time() < t4:
        h, c = sqp_optimize(best_h, n0, t4, max_iter=300, rho_init=0.03)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: More local search with smaller steps (100s)
    t5 = min(t0 + 460, deadline - 100)
    if time.time() < t5:
        h, c = fast_local_search(best_h, n0, t5, max_iter=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Try upsampling to n=1000 (250s)
    n_hi = 1000
    t6 = time.time()
    remaining = deadline - t6 - 50
    if remaining > 200:
        h_hi = upsample(best_h, n0, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        t6a = t6 + min(80, remaining * 0.3)
        if time.time() < t6a:
            h_hi, c_hi = lbfgs_smooth(h_hi, n_hi, t6a, alpha=1000, mu=500.0, max_iter=2000)
        t6b = t6 + min(120, remaining * 0.5)
        if time.time() < t6b:
            h_hi, c_hi = sqp_optimize(h_hi, n_hi, t6b, max_iter=300, rho_init=0.04)
        t6c = t6 + min(160, remaining * 0.7)
        if time.time() < t6c:
            h_hi, c_hi = fast_local_search(h_hi, n_hi, t6c, max_iter=5000)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            n_best = n_hi

    # Phase 7: Try alternative starting point at lower resolution (100s)
    t7 = time.time()
    remaining = deadline - t7 - 30
    if remaining > 100:
        n_alt = 300
        h_alt = upsample(best_h, n_best, n_alt)
        # Perturb
        rng = np.random.RandomState(seed + 1)
        h_alt += rng.randn(n_alt) * 0.02
        h_alt = project_feasible(h_alt, n_alt)
        t7a = t7 + min(40, remaining * 0.4)
        if time.time() < t7a:
            h_alt, c_alt = multi_start_optimize(h_alt, n_alt, t7a, max_iter=200, rho_init=0.05)
        t7b = t7 + min(70, remaining * 0.7)
        if time.time() < t7b:
            h_alt, c_alt = fast_local_search(h_alt, n_alt, t7b, max_iter=5000)
        # Upsample back to n_best and compare
        if c_alt < best_c + 0.001:
            h_alt_hi = upsample(h_alt, n_alt, n_best)
            c_alt_hi = compute_c5(h_alt_hi, n_best)
            if c_alt_hi < best_c:
                best_c = c_alt_hi
                best_h = h_alt_hi.copy()

    # Phase 8: Final SQP polish (80s)
    t8 = time.time()
    if time.time() < deadline - 10:
        h_fin, c_fin = sqp_optimize(best_h, n_best, deadline - 10, max_iter=200, rho_init=0.02, max_active=10)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    # Phase 9: Final local search
    if time.time() < deadline - 5:
        h_fin, c_fin = fast_local_search(best_h, n_best, deadline - 3, max_iter=5000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, n_best
