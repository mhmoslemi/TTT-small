import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    np.clip(h, 0.0, 1.0, out=h)
    return h

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

            h_new = project_feasible(res.x, n)

def exact_line_search_pair(h, n, i, j):
    """Find optimal t for h[i]+=t, h[j]-=t to minimize max correlation."""
    t_max = min(1.0 - h[i], h[j])
    if t_max < 1e-12:
        return 0.0
    c_vec = compute_c5_vec(h, n)
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        if k >= 0:
            vi = 0.0
            if i < n - k:
                vi += 1.0 - h[i + k]
            if i >= k and (i - k) < (n - k):
                vi -= h[i - k]
            vj = 0.0
            if j < n - k:
                vj += 1.0 - h[j + k]
            if j >= k and (j - k) < (n - k):
                vj -= h[j - k]
        else:
            kk = -k
            vi = 0.0
            if i >= kk:
                vi += 1.0 - h[i - kk]
            if (i + kk) < (n - kk):
                vi -= h[i + kk]
            vj = 0.0
            if j >= kk:
                vj += 1.0 - h[j - kk]
            if (j + kk) < (n - kk):
                vj -= h[j + kk]
        d[mm] = (vi - vj) * dx
    f0 = c_vec.max()
    grid = np.linspace(0, t_max, 50)
    f_grid = np.zeros(50)
    for gi in range(50):
        f_grid[gi] = np.max(c_vec + grid[gi] * d)
    best_gi = np.argmin(f_grid)
    if f_grid[best_gi] >= f0 - 1e-14:
        return 0.0
    lo = grid[max(0, best_gi - 1)]
    hi = grid[min(49, best_gi + 1)]
    grid2 = np.linspace(lo, hi, 100)
    f_grid2 = np.zeros(100)
    for gi in range(100):
        f_grid2[gi] = np.max(c_vec + grid2[gi] * d)
    best_gi2 = np.argmin(f_grid2)
    return float(grid2[best_gi2])

def exact_local_search(h, n, deadline, max_iter=50000):
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(h, n, idx)
        grad /= len(active)
        top_neg = np.argsort(grad)[:15]
        top_pos = np.argsort(-grad)[:15]
        found = False
        for ii in top_neg:
            for jj in top_pos:
                if ii == jj:
                    continue
                t_opt = exact_line_search_pair(h, n, int(ii), int(jj))
                if t_opt > 1e-12:
                    h[int(ii)] += t_opt
                    h[int(jj)] -= t_opt
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        found = True
                        break
            if found:
                break
        if not found:
            # Try reverse direction
            for ii in top_pos:
                for jj in top_neg:
                    if ii == jj:
                        continue
                    t_opt = exact_line_search_pair(h, n, int(ii), int(jj))
                    if t_opt > 1e-12:
                        h[int(ii)] += t_opt
                        h[int(jj)] -= t_opt
                        h = project_feasible(h, n)

                        if c_new < best_c - 1e-15:

def window_coordinate_descent(h, n, deadline, win_size=8, max_rounds=200):
    """Optimize small windows using coordinate descent with exact line search."""
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    # Focus on regions near active shifts
    active_idx = np.where(c_vec >= max_val - 0.002 * max_val)[0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for idx in active_idx:
            k = idx - (n - 1)
            if k >= 0:
                center = n // 2 + k // 2
            else:
                center = n // 2 + k // 2
            lo = max(0, center - win_size)
            hi = min(n, center + win_size + 1)
            for i in range(lo, hi):
                for j in range(i + 1, hi):
                    if j - i > win_size:
                        continue
                    t_opt = exact_line_search_pair(h, n, i, j)
                    if t_opt > 1e-12:
                        h[i] += t_opt
                        h[j] -= t_opt
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

            best_c = c_high
            best_h = h_high
            n = n_high

    # Phase 9: Final L-BFGS polish with very high alpha (40s)
    if time.time() < deadline - 5:
        h_fin, c_fin = run_lbfgs_adaptive(best_h, n, deadline - 2, start_alpha=500000, max_alpha=500000, mu=100.0)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
