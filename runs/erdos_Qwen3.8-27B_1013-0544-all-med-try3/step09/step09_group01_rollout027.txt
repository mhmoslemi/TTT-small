import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 3))
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def binary_to_h(binary_idx, n):
    """Create a binary h from a set of indices that are 1."""
    h = np.zeros(n)
    h[binary_idx] = 1.0
    return h

def binary_c5(binary_h, n):
    """Compute C5 for a binary h (sum of 1s should be n/2)."""
    return compute_c5(binary_h, n)

def simulated_annealing_binary(n, h_start, deadline, max_iters=None):
    """Simulated annealing on binary configurations."""
    # Start with binary version of h_start
    n_ones = n // 2
    if n % 2 == 1:
        n_ones = n // 2
    
    # Create initial binary from h_start
    if h_start is not None:
        sorted_idx = np.argsort(-h_start)
        ones = set(sorted_idx[:n_ones].tolist())
    else:
        ones = set(np.random.choice(n, n_ones, replace=False).tolist())
    
    h = np.zeros(n)
    h[list(ones)] = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    t = 2.0
    t_min = 0.001
    cooling = 0.9995
    
    if max_iters is None:
        max_iters = 200000
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose a move: move a 1 to a 0 position
        ones_list = list(ones)
        if len(ones_list) < n_ones:
            break
        i = ones_list[rng.randint(len(ones_list))]
        # Find a 0 to swap with
        zeros_list = np.array([j for j in range(n) if j not in ones])
        if len(zeros_list) == 0:
            break
        j = zeros_list[rng.randint(len(zeros_list))]
        
        # Apply move
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        ones.add(j)
        
        c_new = compute_c5(h, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            t *= cooling
        else:
            delta = c_new - best_c
            if rng.expovariate(1.0) < np.exp(-delta / t):
                # Accept
                t *= cooling
            else:
                # Reject
                h[i] = 1.0
                h[j] = 0.0
                ones.add(i)
                ones.remove(j)
        
        t = max(t, t_min)
    
    return best_h, best_c

            if tot < 1e-15:

            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_full(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(500, int(remaining * 2))
        try:
            res = minimize(smo_obj, best_h, jac=smo_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_shifts(h, n, deadline, top_k=8):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [5000, 20000, 100000, 500000, 2000000]
    
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int(remaining * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n//2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h
