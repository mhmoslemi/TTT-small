def compute_c5_fft(h, n):
    """Compute correlation using FFT for speed."""
    dx = 2.0 / n
    # Use FFT for cross-correlation
    # correlate(h, 1-h, mode='full')
    a = h
    b = 1 - h
    # Full correlation: length 2n-1
    # Use: correlate(a,b) = convolve(a, flip(b))
    # Actually, let's just use np.correlate for n <= 2000, FFT for larger
    if n <= 2000:
        return np.correlate(h, 1-h, mode='full') * dx
    else:

        B = fft(1-h, N)
        corr = ifft(A * np.conj(B)).real  # This gives circular correlation
        # Need to adjust for linear correlation
        # Actually, let's use scipy.signal.fftconvolve
        from scipy.signal import fftconvolve
        corr = fftconvolve(h, (1-h)[::-1], mode='full')
        return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

        x = i
        pc = 0
        while x:
            pc ^= 1
            x >>= 1

import numpy as np

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_array(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, seed=0):
    """Convert to binary with exactly n//2 ones."""
    h = np.clip(h, 0, 1)
    n_ones = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_ones]] = 1.0
    return h_bin

def make_thue_morse(n, seed=0):
    h = np.zeros(n)
    for i in range(n):
        x = i
        pc = 0
        while x:
            pc ^= 1
            x >>= 1
        h[i] = 1.0 if pc == 1 else 0.0
    # Adjust to exactly n//2 ones
    n_ones = int(h.sum())
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        h[to_flip] = 1.0
    return h

def sa_binary(h, n, deadline, T0=2.0, T1=0.01, max_iter=500000, seed=42):
    """Simulated annealing on binary sequence with swap moves."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = best_c
    rng = np.random.RandomState(seed)
    
    # Track one and zero positions for fast sampling
    one_pos = np.where(h == 1.0)[0]
    zero_pos = np.where(h == 0.0)[0]
    
    eval_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            # Check time more frequently
            if time.time() > deadline:
                break
        
        # Temperature
        frac = min(1.0, it / max(1, max_iter - 1))
        T = T0 * (T1 / T0) ** frac
        
        # Random swap
        i_idx = rng.randint(len(zero_pos))
        j_idx = rng.randint(len(one_pos))
        i = zero_pos[i_idx]
        j = one_pos[j_idx]
        
        # Perform swap
        h[i] = 1.0
        h[j] = 0.0
        zero_pos[i_idx] = j
        one_pos[j_idx] = i
        
        new_c = compute_c5(h, n)
        eval_count += 1
        
        if new_c < c:
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            delta = new_c - c

                c = new_c
            else:

                h[i] = 0.0
                h[j] = 1.0
                zero_pos[i_idx] = i
                one_pos[j_idx] = j
    
    return best_h, best_c, eval_count

def smooth_max_obj(x, n, alpha):
    c = compute_c5_array(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_array(x, n) / (2.0 / n)  # raw correlation without dx
    dx = 2.0 / n
    m_raw = (c * dx).max()
    c_raw = np.correlate(x, 1 - x, mode='full')
    m = c_raw.max()
    sh = alpha * (c_raw - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def continuous_refine(h, n, deadline, alpha=100000):
    """Refine using L-BFGS-B on the continuous problem."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    try:
        maxit = max(500, int((deadline - time.time()) * 2))
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                       args=(n, alpha), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-20,
                                'gtol': 1e-16, 'maxfun': maxit * 5,
                                'maxls': 30, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    
    return h, best_c

        best_c = c_bin
        best_h = h_bin.copy()
        best_n = n
    start_h = h_bin.copy()
    
    # Phase 2: Main SA rounds (multiple restarts)
    n_sa_rounds = 3
    sa_time = 180  # seconds per SA round
    for r in range(n_sa_rounds):
        t_end = min(t0 + (r + 1) * sa_time, deadline - 100)
        if time.time() > t_end - 10:
            break
        # For restarts, perturb the best solution
        if r == 0:
            h_start = start_h.copy()
        else:
            h_start = best_h.copy()
            # Random perturb: flip 5% of bits
            rng_p = np.random.RandomState(seed + r * 1000)
            n_flip = max(3, n // 20)
            pos = rng_p.choice(n, 2 * n_flip, replace=False)
            for p in pos:
                h_start[p] = 1.0 - h_start[p]
            # Rebalance
            n_ones = int(h_start.sum())
            target = n // 2
            if n_ones > target:
                ones = np.where(h_start == 1.0)[0]
                to_flip = rng_p.choice(ones, n_ones - target, replace=False)
                h_start[to_flip] = 0.0
            elif n_ones < target:
                zeros = np.where(h_start == 0.0)[0]
                to_flip = rng_p.choice(zeros, target - n_ones, replace=False)
                h_start[to_flip] = 1.0
        
        T0_val = 3.0 / (1 + r)
        T1_val = 0.005
        h_sa, c_sa, evals = sa_binary(h_start, n, t_end,
                                       T0=T0_val, T1=T1_val,
                                       max_iter=1000000, seed=seed + r)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
            best_n = n
    
    # Phase 3: Local search (multiple passes)
    if time.time() < deadline - 80:
        ls_end = min(time.time() + 60, deadline - 60)
        h_ls, c_ls = local_search_binary(best_h, best_n, ls_end,
                                          max_rounds=500, rng_seed=seed)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    # Phase 4: Guided local search
    if time.time() < deadline - 60:
        g_end = min(time.time() + 40, deadline - 40)
        h_g, c_g = guided_local_search(best_h, best_n, g_end, rng_seed=seed + 7)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
    
    # Phase 5: Try Thue-Morse based start
    if time.time() < deadline - 150:
        h_tm = make_thue_morse(best_n, seed=seed)
        c_tm = compute_c5(h_tm, best_n)
        tm_end = min(time.time() + 60, deadline - 80)
        h_tm_sa, c_tm_sa, _ = sa_binary(h_tm, best_n, tm_end,
                                         T0=2.0, T1=0.01,

        if c_tm_sa < best_c:
