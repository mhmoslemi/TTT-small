Let me also add a "gradient-based pair search" that's more efficient than the previous O(n²) approach:
- Compute the gradient g of the smoothed max objective
- For the "increase" side: positions where g[i] < 0 (decreasing h[i] helps... wait, the gradient points in the direction of increase)
- Actually: if g[i] > 0, increasing h[i] increases the objective → we want to decrease h[i]
- If g[i] < 0, increasing h[i] decreases the objective → we want to increase h[i]
- So: move mass from high-g positions to low-g positions
- But we also need to respect [0,1] bounds and sum constraint

The "optimal" pair move is: pick i with largest g[i] (and h[i] > 0) and j with smallest g[j] (and h[j] < 1), and move δ from i to j.

The optimal δ is limited by:
- h[i] - δ ≥ 0 → δ ≤ h[i]
- h[j] + δ ≤ 1 → δ ≤ 1 - h[j]
- The objective should decrease (for the linear approximation): δ*(g[i] - g[j]) < 0, which is satisfied since g[i] > g[j]

For the actual (nonlinear) change, we should do a line search or just use small δ.

This is O(n log n) per step (sorting) + O(n) for the gradient = very fast!

                grad[0:nv] += wm * (-best_h[kk:n])

        # Subject to bounds
        candidates_src = np.where(best_h > 0.01)[0]
        candidates_dst = np.where(best_h < 0.99)[0]

            break
        
        # Top source (highest gradient) and top destination (lowest gradient)
        src_candidates = sorted(candidates_src, key=lambda i: -grad[i])[:20]
        dst_candidates = sorted(candidates_dst, key=lambda i: grad[i])[:20]
        
        best_improvement = 0
        best_move = None
        for i in src_candidates:
            for j in dst_candidates:
                if i == j:
                    continue
                delta = min(best_h[i], 1.0 - best_h[j], 0.01)
                if delta < 1e-6:
                    continue
                best_h[i] -= delta
                best_h[j] += delta
                new_c = compute_c5(best_h, n)
                if new_c < best_c - 1e-15:

                        best_move = (i, j, delta, new_c)

            best_h[i] -= delta
            best_h[j] += delta
            best_c = new_c
        else:

            break

        src_sorted = src_sorted[np.argsort(-grad[src_sorted])][:10]
        dst_sorted = np.where(dst_mask)[0]
        dst_sorted = dst_sorted[np.argsort(grad[dst_sorted])][:10]
        best_improve = 0.0
        best_move = None
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for i in src_sorted:
            for j in dst_sorted:
                if i == j:
                    continue
                for d in deltas:
                    if best_h[i] < d or best_h[j] > 1.0 - d:
                        continue
                    nd = min(d, best_h[i], 1.0 - best_h[j])
                    if nd < 1e-7:
                        continue
                    best_h[i] -= nd
                    best_h[j] += nd
                    nc = compute_c5(best_h, n)
                    best_h[i] += nd
                    best_h[j] -= nd
                    if nc < best_c - 1e-15:
                        imp = best_c - nc
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, nd)
        if best_move is not None:
            i, j, nd = best_move
            best_h[i] -= nd
            best_h[j] += nd
            best_c = compute_c5(best_h, n)
        else:
            break
    return best_h, best_c


def binary_local_search(h, n, deadline):
    h_bin = (h >= 0.5).astype(float)
    n_ones = n // 2
    cur_ones = int(h_bin.sum())
    if cur_ones > n_ones:
        idx = np.where(h_bin == 1)[0]
        for t in range(cur_ones - n_ones):
            h_bin[idx[t * 3 + 1]] = 0.0
    elif cur_ones < n_ones:
        idx = np.where(h_bin == 0)[0]
        for t in range(n_ones - cur_ones):
            h_bin[idx[t * 3 + 2]] = 1.0
    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    dx = 2.0 / n
    ones_idx = list(np.where(h_bin == 1)[0])
    zeros_idx = list(np.where(h_bin == 0)[0])
    for it in range(20000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        top_shifts = np.argsort(-c_vec)[:5]
        old_ones = set(int(x) for x in ones_idx)
        best_improve = 0.0
        best_flip = None
        sample_o = ones_idx[np.random.randint(len(ones_idx), size=min(50, len(ones_idx)))]
        sample_z = zeros_idx[np.random.randint(len(zeros_idx), size=min(50, len(zeros_idx)))]
        for a in sample_o:
            for b in sample_z:
                best_h[a] = 0.0
                best_h[b] = 1.0
                nc = compute_c5(best_h, n)
                best_h[a] = 1.0
                best_h[b] = 0.0
                if nc < best_c - 1e-14:
                    imp = best_c - nc
                    if imp > best_improve:
                        best_improve = imp
                        best_flip = (int(a), int(b))
        if best_flip is not None:
            a, b = best_flip
            best_h[a] = 0.0
            best_h[b] = 1.0
            ones_idx.remove(a)
            ones_idx.append(b)
            zeros_idx.remove(b)
            zeros_idx.append(a)
            best_c = compute_c5(best_h, n)
        else:
            break
    return best_h, best_c


def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def periodic_init(n, fill_frac=0.5, n_blocks=1):
    h = np.zeros(n)
    block_size = n // (2 * n_blocks)
    for b in range(n_blocks):
        start = b * 2 * block_size
        h[start:start + block_size] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in np.where(h < 1)[0]:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff < 1e-10:
                break
    elif diff < 0:
        for i in np.where(h > 0)[0][::-1]:
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
            if diff > -1e-10:
                break
    return project_feasible(h, n)


def run_lbfgs_short(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 8,
                                    'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Optimize at original resolution n0
    p1_end = t0 + 120
    h1, c1 = phase_lbfgs(h0, n0, p1_end,
                         alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
                         mu=0.01, maxiters=[800, 800, 1000, 1500, 2000, 2000, 3000])
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, n0

    # Phase 2: Active-set at n0
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 100, deadline - 500)
        h2, c2 = phase_active(best_h, best_n, p2_end, top_k=5,
                              alphas=[100000, 500000, 2000000, 10000000], rounds=10)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, best_n

    # Phase 3: Upsample to n=1000 and optimize
    n_hi = 1000
    if time.time() < deadline - 400:
        h_hi = upsample_h(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c:
            best_h, best_c, best_n = h_hi, c_hi, n_hi

        p3_end = min(time.time() + 300, deadline - 250)
        h3, c3 = phase_lbfgs(h_hi, n_hi, p3_end,
                             alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                             mu=0.005, maxiters=[800, 800, 1000, 1500, 2000, 2000, 2500, 3000])
        if c3 < best_c:
            best_h, best_c, best_n = h3, c3, n_hi

        if time.time() < deadline - 200:
            p3b_end = min(time.time() + 150, deadline - 200)
            h3b, c3b = phase_active(best_h, best_n, p3b_end, top_k=4,
                                    alphas=[500000, 2000000, 10000000, 50000000], rounds=12)
            if c3b < best_c:
                best_h, best_c, best_n = h3b, c3b, best_n

    # Phase 4: Greedy mass transfer
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = greedy_mass_transfer(best_h, best_n, p4_end, max_iters=4000)
        if c4 < best_c:
            best_h, best_c, best_n = h4, c4, best_n

    # Phase 5: Binary local search
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 80, deadline - 150)
        h5, c5 = binary_local_search(best_h, best_n, p5_end)
        if c5 < best_c:
            best_h, best_c, best_n = h5, c5, best_n
        else:
            # Try binary from scratch with periodic structure
            h5b = periodic_init(best_n, fill_frac=0.5, n_blocks=1)
            c5b = compute_c5(h5b, best_n)
            if c5b < best_c:
                best_h, best_c, best_n = h5b, c5b, best_n

    # Phase 6: Try n=2000
    n_very = 2000
    if time.time() < deadline - 150:
        h_v = upsample_h(best_h, best_n, n_very)
        c_v = compute_c5(h_v, n_very)
        if c_v < best_c:
            best_h, best_c, best_n = h_v, c_v, n_very

        p6_end = min(time.time() + 100, deadline - 100)
        h6, c6 = phase_lbfgs(h_v, n_very, p6_end,
                             alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
                             mu=0.005, maxiters=[500, 800, 1000, 1200, 1500, 1500, 2000])
        if c6 < best_c:
            best_h, best_c, best_n = h6, c6, n_very

        if time.time() < deadline - 80:
            p6b_end = min(time.time() + 70, deadline - 80)
            h6b, c6b = phase_active(best_h, best_n, p6b_end, top_k=4,
                                    alphas=[500000, 5000000, 50000000], rounds=8)
            if c6b < best_c:
                best_h, best_c, best_n = h6b, c6b, best_n

    # Phase 7: Multi-start exploration
    if time.time() < deadline - 100:
        starts = []
        # Binary threshold of best
        hb = (best_h >= 0.5).astype(float)
        hb = project_feasible(hb, best_n)
        starts.append(hb)
        # Periodic with 2 blocks
        hp = periodic_init(best_n, fill_frac=0.5, n_blocks=2)
        starts.append(hp)
        # Random perturbation
        hr = best_h + 0.05 * rng.randn(best_n)
        hr = project_feasible(hr, best_n)
        starts.append(hr)

        p7_end = deadline - 60
        for start_h in starts:
            if time.time() > p7_end:
                break
            sc = compute_c5(start_h, best_n)
            if sc < best_c:
                best_h, best_c = start_h.copy(), sc
            short_end = min(time.time() + 30, p7_end)
            sh, sc2 = run_lbfgs_short(start_h, best_n, short_end,
                                      alphas=[1000, 10000, 100000, 1000000], mu=0.01)
            if sc2 < best_c:
                best_h, best_c, best_n = sh, sc2, best_n

    # Phase 8: Final active-set polish
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 25, deadline - 15)
        h8, c8 = phase_active(best_h, best_n, p8_end, top_k=3,
                              alphas=[5000000, 50000000, 500000000], rounds=5)
        if c8 < best_c:
            best_h, best_c, best_n = h8, c8, best_n

    # Phase 9: Very fine mass transfer
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = greedy_mass_transfer(best_h, best_n, p9_end, max_iters=2000)
        if c9 < best_c:
            best_h, best_c, best_n = h9, c9, best_n

    # Final
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    # Ensure exact sum
    assert abs(best_h.sum() - final_n / 2.0) < 1e-10, f"Sum constraint violated: {best_h.sum()} vs {final_n/2}"
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1 + 1e-15), "Bounds violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
