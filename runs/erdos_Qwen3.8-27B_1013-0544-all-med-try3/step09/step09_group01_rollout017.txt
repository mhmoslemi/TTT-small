    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for i, idx in enumerate(active):
        wi = w[i]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[0:n-s] += wi * dx * (1.0 - h[s:n])
            grad[s:n] -= wi * dx * h[0:n-s]
        else:
            ss = -s
            grad[ss:n] += wi * dx * (1.0 - h[0:n-ss])
            grad[0:n-ss] -= wi * dx * h[ss:n]
    if mu > 0:
        grad += mu * (h.sum() - n/2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = min(5000, max(500, int(remaining * 3)))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-18,
                                  'gtol': 1e-15, 'maxfun': maxit * 10,
                                  'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_lbfgs(h, n, deadline, top_k=3, alphas=None):

    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

        if remaining < 20:
            break

        max_val = c_vec.max()

        sorted_idx = np.argsort(-c_vec)
        active = sorted_idx[:top_k]
        # Also include nearby values
        threshold = max_val - 0.001 * max_val
        active_all = np.where(c_vec >= threshold)[0]
        if len(active_all) > top_k:
            active_all = sorted_idx[:top_k]
        active = sorted_idx[:top_k]
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = min(3000, max(500, int((deadline - time.time()) * 2)))

                             args=(n, active, alpha, 0.001), method='L-BFGS-B',

                             options={'maxiter': maxit, 'ftol': 1e-18,
                                      'gtol': 1e-15, 'maxfun': maxit * 10,

            except Exception:
                pass

def subgrad_descent(h, n, deadline, n_iter=10000, eta_init=0.003):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = eta_init
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        # Use top-K active
        tol = max(1e-8, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > 10:
            sorted_idx = np.argsort(-c_vec)
            active = sorted_idx[:10]
        
        # Average gradient over active
        grad = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            if s >= 0:
                grad[0:n-s] += (1.0 - h[s:n])
                grad[s:n] -= h[0:n-s]
            else:
                ss = -s
                grad[ss:n] += (1.0 - h[0:n-ss])
                grad[0:n-ss] -= h[ss:n]
        grad *= dx / len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.02)
        else:
            eta *= 0.97
        h = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, 0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
