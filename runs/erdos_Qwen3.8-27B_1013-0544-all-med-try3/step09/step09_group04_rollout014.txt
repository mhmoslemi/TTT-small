import numpy as np
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def power_sum_obj(x, n, p, mu):
    """Power-sum approximation to max"""
    c = compute_c5_vec(x, n)
    # Scale to avoid overflow
    cm = c.max()
    if cm < 1e-15:
        val = 0.0
    else:
        scaled = c / cm
        val = cm * (np.sum(scaled**p))**(1.0/p)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def power_sum_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    cm = c.max()
    if cm < 1e-15:
        return np.zeros(n)
    scaled = c / cm
    # d/dx of cm * (sum(scaled^p))^(1/p)
    # = cm * (1/p) * (sum(scaled^p))^(1/p - 1) * p * sum(scaled^(p-1) * dscaled/dx)
    # = (sum(scaled^p))^(1/p - 1) * sum(scaled^(p-1) * dscaled/dx)
    # where dscaled/dx = dc/dx / cm
    
    sp = np.sum(scaled**p)
    factor = sp**(1.0/p - 1.0) / cm  # (sum(scaled^p))^(1/p-1) / cm

    dx = 2.0 / n

def pair_search(h, n, deadline, radius=50):

        while improved and time.time() < deadline:

            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)

                            best_h = h.copy()

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def power_sum_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    cm = c.max()
    if cm < 1e-12:
        val = 0.0
    else:
        scaled = c / cm
        val = cm * (np.sum(scaled ** p)) ** (1.0 / p)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def power_sum_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    cm = c.max()
    grad = np.zeros(n)
    if cm < 1e-12:
        if mu > 0:
            grad += 2.0 * mu * (x.sum() - n / 2.0)
        return grad
    scaled = c / cm
    sp = np.sum(scaled ** p)
    factor = sp ** (1.0 / p - 1.0) / cm
    dx = 2.0 / n
    for mm in range(2 * n - 1):
        w = scaled[mm] ** (p - 1) * factor
        if w < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pair_search(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

        p10_end = deadline - 3
        h10, c10 = pair_search(best_h, best_n, p10_end, radius=30)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final: one more tight active-set push
    if time.time() < deadline - 1:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 50000000, 0.0005),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 200, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 500})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
