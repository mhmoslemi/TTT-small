            if tot < 1e-18:

            h += (diff / tot) * h

def smooth_max_obj(x, n, alpha, mu, beta):
    c = compute_c5_vec(x, n)
    m = c.max()
    # Log-sum-exp approximation
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    val = m + lse / alpha
    if beta > 0:
        val += beta * c.var()
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()

    mx = shifted.max()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def lse_alpha(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha


def obj_sm(x, n, alpha, mu, beta):
    c = compute_c5_vec(x, n)
    val = lse_alpha(c, alpha)
    if beta > 0:
        val += beta * float(c.var())
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def grad_sm(x, n, alpha, mu, beta):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nn2 = 2 * n - 1
    grad = np.zeros(n)
    for mm in range(nn2):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if beta > 0:
        cmean = c.mean()
        scale = beta * 2.0 / nn2
        for mm in range(nn2):
            ck = c[mm] - cmean
            if abs(ck) < 1e-16:
                continue
            k = mm - (n - 1)
            wm = scale * ck * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs(h, n, deadline, alphas, mu=0.005, beta=0.0, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(300, int(remaining * 1.5))
        try:
            res = minimize(obj_sm, best_h, jac=grad_sm,
                           args=(n, alpha, mu, beta), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': maxit * 15,
                                    'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def run_active_set(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000, 100000000]
    for rnd in range(100):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-10, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 15:
                break
            maxit = max(200, int(rem2 * 2))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                               args=(n, active, alpha, 0.001),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-25,
                                        'gtol': 1e-20, 'maxfun': maxit * 12,
                                        'maxls': 50, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c


def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old + 1)
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_new = np.linspace(0, 2, n_new + 1)
    h_new = np.interp(x_new, x_old, h_ext)
    h_new = h_new[:n_new]
    return project_feasible(h_new, n_new)


def coord_descent_refine(h, n, deadline, n_passes=50):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    c_vec = compute_c5_vec(best_h, n)
    sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in sizes:
                for sign in [1, -1]:
                    t = best_h[i] + sign * d
                    if t < 0 or t > 1:
                        continue
                    old = best_h[i]
                    best_h[i] = t
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved_any = True
                        c_vec = compute_c5_vec(best_h, n)
                    else:
                        best_h[i] = old
                        break
        if not improved_any:
            break
    return best_h, best_c


def random_perturb(h, n, eps):
    h_p = h.copy()
    h_p += np.random.randn(n) * eps
    return project_feasible(h_p, n)
