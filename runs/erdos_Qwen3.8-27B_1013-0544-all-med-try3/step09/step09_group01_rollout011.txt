def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    if alpha * m > 500:
        mx2 = sh.max()
        val = m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

This is efficient: O(n * k_active) to compute gradients, then O(n² * k_active) for the pair search.

- h[j] - d ≥ 0

We want all active c_k to decrease. The maximum d we can use (before any c_k increases) is:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    if abs(alpha) * abs(m) > 500:
        mx2 = sh.max()
        val = m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alphas, mu, maxiters):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, (alpha, maxit) in enumerate(zip(alphas, maxiters)):
        if time.time() > deadline - 10:
            break
        remaining = deadline - time.time()
        actual_maxit = min(maxit, max(200, int(remaining * 4)))
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': actual_maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': actual_maxit * 10,
                                   'maxls': 60, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(30):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-6, 0.0003 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.002),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new

            except Exception:
                pass
    return best_h, best_c

def run_slqp(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def gradient_local_search(h, n, deadline, radii=(30, 20, 10, 5)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for radius in radii:
        for d in deltas:
            while time.time() < deadline:
                improved = False
                c_vec = compute_c5_vec(best_h, n)
                c_max = c_vec.max()
                active = np.where(c_vec >= c_max - 2e-4)[0]
                if len(active) > 10:
                    active = np.array(np.argsort(-c_vec[active])[:10])
                # Compute gradients for active
                grads = np.zeros((len(active), n))
                for ai, idx in enumerate(active):
                    k = idx - (n - 1)
                    if k >= 0:
                        nv = n - k
                        grads[ai, 0:nv] = dx * (1.0 - best_h[k:n])
                        grads[ai, k:n] += dx * (-best_h[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        grads[ai, kk:n] = dx * (1.0 - best_h[0:nv])
                        grads[ai, 0:nv] += dx * (-best_h[kk:n])
                best_delta = 0.0
                best_pair = None
                for i in range(n):
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    for j in range(lo, hi):
                        if i == j:
                            continue
                        if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                            delta = d * (grads[:, i] - grads[:, j])
                            new_max = (c_vec[active] + delta).max()
                            if new_max < best_c - 1e-16 and (best_c - new_max) > best_delta:
                                best_delta = best_c - new_max
                                best_pair = (i, j, 1)
                        if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                            delta = -d * (grads[:, i] - grads[:, j])
                            new_max = (c_vec[active] + delta).max()
                            if new_max < best_c - 1e-16 and (best_c - new_max) > best_delta:
                                best_delta = best_c - new_max
                                best_pair = (i, j, -1)
                if best_pair is not None:

                    best_h = project_feasible(best_h, n)
                    best_c = compute_c5(best_h, n)

    return best_h, best_c

def subgradient_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-8, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.985
        h = h_new
    return best_h, best_c

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    if n % 2 != 0:
        n_ones = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def periodic_start(n, period, ones_per_period, phase=0):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[(phase + p) % period::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)
