import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_smooth(h, n, deadline, alphas, mu=0.005, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-24,
                                    'gtol': 1e-20, 'maxfun': maxit * 12,
                                    'maxls': 40, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        # Normalize gradient for stable step
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn * min(gn, 0.1)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.005)
        else:
            eta *= 0.98
    return best_h, best_c

def coordinate_descent(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    # Try h[i]+d, h[j]-d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i]-d, h[j]+d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_init(n, pattern='alt'):
    h = np.zeros(n)
    if pattern == 'alt':
        h[::2] = 1.0
    elif pattern == 'blocks':
        block = n // 4
        h[0:2*block] = 1.0
    elif pattern == 'four':
        # Four blocks of 1s and 0s
        seg = n // 8
        h[0:2*seg] = 1.0
        h[4*seg:6*seg] = 1.0
    else:
        h[:n//2] = 1.0
    return project_feasible(h, n)

    # Phase 2: Upsample to n=2000 and optimize
    n_main = 2000
    h_main = upsample(best_h, best_n, n_main)
    c_main = compute_c5(h_main, n_main)
    best_h_main = h_main.copy()
    best_c_main = c_main

    p2_end = min(t0 + 350, deadline - 400)
    h2, c2 = run_lbfgs_smooth(h_main, n_main, p2_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.003, maxiters=[600, 800, 1000, 1500, 2000, 2500, 3000])
    if c2 < best_c_main:
        best_c_main = c2
        best_h_main = h2

    # Phase 3: Active set optimization at n=2000
    p3_end = min(time.time() + 120, deadline - 250)
    h3, c3 = run_active_set(best_h_main, n_main, p3_end, top_k=5,
        alphas=[100000, 500000, 2000000, 10000000], n_rounds=20)
    if c3 < best_c_main:
        best_c_main = c3
        best_h_main = h3

    # Phase 4: Squeeze - focus on worst lags
    p4_end = min(time.time() + 80, deadline - 150)
    h4, c4 = run_squeeze(best_h_main, n_main, p4_end)
    if c4 < best_c_main:
        best_c_main = c4
        best_h_main = h4

    # Track global best
    if best_c_main < best_c:
        best_c = best_c_main
        best_h = best_h_main
        best_n = n_main

    # Phase 5: Try n=2500 if time allows
    n_high = 2500
    if time.time() < deadline - 200:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.001:
            p5_end = min(time.time() + 120, deadline - 120)
            h5, c5 = run_lbfgs_smooth(h_high, n_high, p5_end,
                alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                mu=0.002, maxiters=[400, 600, 800, 1000, 1200, 1500, 1800, 2000])
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_high
            # Active set at high resolution
            if time.time() < deadline - 60:
                p5b_end = min(time.time() + 50, deadline - 30)
                h5b, c5b = run_active_set(h5, n_high, p5b_end, top_k=4,
                    alphas=[500000, 2000000, 10000000], n_rounds=10)
                if c5b < best_c:
                    best_c = c5b
                    best_h = h5b

    # Phase 6: Subgradient refinement at best resolution
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 30, deadline - 10)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Coordinate descent
    if time.time() < deadline - 10:
        p7_end = deadline - 2
        h7, c7 = coordinate_descent(best_h, best_n, p7_end, radius=25)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final squeeze if time remains
    if time.time() < deadline - 3 and best_n == n_main:
        h_f, c_f = run_squeeze(best_h, best_n, deadline - 1)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
