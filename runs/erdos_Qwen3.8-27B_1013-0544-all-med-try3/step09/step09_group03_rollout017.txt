```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    n2m1 = 2 * n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': 500000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500):
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    n_ones = n // 2
    cur_ones = int(h_bin.sum())
    if cur_ones > n_ones:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])[:cur_ones - n_ones]
        for i in order:
            h_bin[ones_idx[i]] = 0.0
    elif cur_ones < n_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])[:n_ones - cur_ones]
        for i in order:
            h_bin[zeros_idx[i]] = 1.0
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Try random moves
        n_trials = min(500, len(ones_idx) * len(zeros_idx))
        best_move = None
        best_new_c = best_c
        
        for _ in range(n_trials):
            a = ones_idx[rng.randint(len(ones_idx))]
            b = zeros_idx[rng.randint(len(zeros_idx))]
            if a == b:
                continue
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            nc = compute_c5(h_bin, n)
            if nc < best_new_c - 1e-14:
                best_new_c = nc
                best_move = (a, b)
            h_bin[a] = 1.0
            h_bin[b] = 0.0
        
        if best_move is not None:
            a, b = best_move
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            best_c = best_new_c

        else:

def upsample_step(h, n_old, n_new):
    """Step-function upsample (nearest neighbor)."""
    indices = np.floor(np.linspace(0, n_old, n_new + 1)[:-1]).astype(int)
    indices = np.clip(indices, 0, n_old - 1)
    h_new = h[indices]
    return project_feasible(h_new, n_new)

def upsample_linear(h, n_old, n_new):

    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

            old_val = h[i]
            # Try a few perturbations
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                nv = old_val + delta
                if nv < 0 or nv > 1:

                h[i] = nv
                # Need to maintain sum: adjust another point
                # Actually, let's just try the perturbation and check if sum is still OK

                    # Maintain sum by adjusting nearest non-boundary point
                    diff = n / 2.0 - h.sum()
                    if abs(diff) > 1e-10:
                        # Find a point to adjust
                        if diff > 0:
                            candidates = np.where(h < 0.99)[0]
                        else:
                            candidates = np.where(h > 0.01)[0]
                        if len(candidates) > 0:
                            j = candidates[rng.randint(len(candidates))]

                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    best_c = c
                    improved = True
                    break
                else:
                    h[i] = old_val
            if time.time() > deadline:
                break
        if not improved and rnd > 20:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
