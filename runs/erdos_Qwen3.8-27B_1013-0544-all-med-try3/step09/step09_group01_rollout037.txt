import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiter_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_list and i < len(maxiter_list):

        else:

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

        if len(active) > 15:
            order = np.argsort(-c_vec[active])
            active = active[order[:15]]
        if len(active) > 0:
            bounds = [(0.0, 1.0)] * best_n
            for alpha in [5000000, 20000000]:
                if time.time() > p7_end - 2:
                    break
                maxit = max(200, int((p7_end - time.time()) * 3))
                try:
                    res = minimize(active_obj, best_h, jac=active_grad,
                                  args=(best_n, active, alpha, 0.0),
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-22,
                                           'gtol': 1e-18, 'maxfun': maxit * 8,
                                           'maxls': 50, 'maxcor': 40})
                    h7 = project_feasible(res.x, best_n)
                    c7 = compute_c5(h7, best_n)
                    if c7 < best_c:
                        best_c, best_h = c7, h7
                except Exception:
                    pass

    # Final: pair swap local search on the worst shifts
    if time.time() < deadline - 3:
        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        worst_shifts = np.argsort(-c_vec)[:5]
        n_act = best_n
        h_curr = best_h.copy()
        c_curr = best_c
        dx_val = 2.0 / n_act
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for step in steps:
            if time.time() > deadline - 3:
                break
            improved = True
            iter_count = 0
            while improved and time.time() < deadline - 3:
                improved = False
                iter_count += 1
                if iter_count > 200:
                    break
                for _ in range(50):
                    if time.time() > deadline - 3:
                        break
                    i = rng.randint(n_act)
                    j = rng.randint(n_act)
                    if i == j:
                        continue
                    if h_curr[i] + step <= 1.0 and h_curr[j] - step >= 0.0:
                        h_curr[i] += step
                        h_curr[j] -= step
                        cn = compute_c5(h_curr, n_act)
                        if cn < c_curr - 1e-15:
                            c_curr = cn
                            improved = True
                        else:
                            h_curr[i] -= step
                            h_curr[j] += step
                    elif h_curr[i] - step >= 0.0 and h_curr[j] + step <= 1.0:
                        h_curr[i] -= step
                        h_curr[j] += step
                        cn = compute_c5(h_curr, n_act)
                        if cn < c_curr - 1e-15:
                            c_curr = cn
                            improved = True
                        else:
                            h_curr[i] += step
                            h_curr[j] -= step
        if c_curr < best_c:
            best_c, best_h = c_curr, h_curr

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
