    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    val = m + log_sum / alpha

    return float(val)

        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def lbfgs_step(h, n, alpha, mu, maxiter, maxfun=None):
    bounds = [(0.0, 1.0)] * n
    if maxfun is None:
        maxfun = maxiter * 10
    try:
        res = minimize(sm_obj, h, jac=sm_grad,
                       args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-16, 'maxfun': maxfun,
                                'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)


def act_lbfgs_step(h, n, active, alpha, mu, maxiter, maxfun=None):
    bounds = [(0.0, 1.0)] * n
    if maxfun is None:
        maxfun = maxiter * 10
    try:
        res = minimize(act_obj, h, jac=act_grad,
                       args=(n, active, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-16, 'maxfun': maxfun,
                                'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)


def multi_alpha_lbfgs(h, n, deadline, alphas, mu=0.005, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):
            mi = maxiters[i]
        else:
            mi = max(200, int(remaining * 1.5))
        h_new, c_new = lbfgs_step(best_h, n, alpha, mu, mi)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c


def active_set_opt(h, n, deadline, top_k=5, alphas=None):
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for rnd in range(200):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            mi = max(100, int(rem2 * 1.2))
            h_new, c_new = act_lbfgs_step(best_h, n, active, alpha, 0.001, mi)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    return best_h, best_c


def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)


def random_binary(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:n_ones - target]] = 0
    return project_feasible(h, n)
