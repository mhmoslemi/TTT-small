def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return w / s

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])

            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return w / s

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e6):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:

            candidates = np.where(pos)[0]
            if len(candidates) > 30:
                order = np.argsort(-diff_h[candidates])[:30]
                candidates = candidates[order]

            # Adaptive step size
            step = step_base * (1.0 + 0.5 * (c_all[k_star] / c_max - 0.8))
            step = max(0.001, min(0.02, step))

            for ci in candidates:
                if time.time() > deadline:
                    break
                i = int(ci)
                j = i + m
                d = min(step, diff_h[ci] * 0.4)
                if d < 1e-8:
                    continue

                if lag_star > 0:
                    h[i] -= d
                    h[j] += d
                else:
                    h[i] += d
                    h[j] -= d

                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-13:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                else:
                    if lag_star > 0:
                        h[i] += d
                        h[j] -= d
                    else:
                        h[i] -= d
                        h[j] += d
                    h = project_feasible(h, n)

        if improved:
            step_base = max(0.001, step_base * 0.995)
        else:
            no_improve += 1
            if no_improve > 300:
                # Try smaller steps
                step_base = max(0.0005, step_base * 0.6)
                if no_improve > 1000:
                    # Perturb
                    h = best_h + rng.normal(0, 0.004, n)
                    h = project_feasible(h, n)
                    no_improve = 0
                    step_base = 0.005

        rounds += 1

    return best_h, best_c

def micro_local_search(h, n, deadline, rng):
    """Very fine-grained local search with tiny steps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top3 = np.argsort(c_all)[-3:]
        improved = False

        for k3 in top3:
            lag3 = k3 - (n - 1)
            if lag3 == 0 or abs(lag3) >= n:
                continue
            m3 = abs(lag3)

            # Systematic scan (coarse)
            stride = max(1, (n - m3) // 500)
            for i in range(0, n - m3, stride):
                if time.time() > deadline:
                    break
                j = i + m3
                if lag3 > 0:
                    if h[i] > h[j] + 1e-6:
                        d = min(0.001, (h[i] - h[j]) * 0.2)
                        h[i] -= d
                        h[j] += d
                    else:
                        continue
                else:
                    if h[j] > h[i] + 1e-6:
                        d = min(0.001, (h[j] - h[i]) * 0.2)
                        h[i] += d
                        h[j] -= d
                    else:
                        continue

                if d < 1e-9:
                    continue
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-13:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    if lag3 > 0:
                        h[i] += d
                        h[j] -= d
                    else:
                        h[i] -= d
                        h[j] += d
                    h = project_feasible(h, n)

        if not improved:
            # Random micro moves
            for _ in range(200):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j:
                    continue
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                if h[i] > h[j]:
                    d = min(0.0005, (h[i] - h[j]) * 0.1)
                    h[i] -= d
                    h[j] += d
                else:
                    d = min(0.0005, (h[j] - h[i]) * 0.1)
                    h[i] += d
                    h[j] -= d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-13:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += d if h[i] <= h[j] else -d
                    h[j] -= d if h[i] <= h[j] else -d
                    h = project_feasible(h, n)
            if not improved:
                break

    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, alpha_sa=100000, t_init=0.004, t_min=0.00001, cool=0.99992):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.65:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.1, 0.004 + T * 6)
            if h[i] > h[j] + 1e-6:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-6:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-5:]
            k_star = top_k[rng.randint(5)]
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                T *= cool
                continue
            m = abs(lag)
            i = rng.randint(0, n - m)
            j = i + m
            if lag > 0:
                if h[i] > h[j] + 1e-6:
                    d = rng.uniform(0.002, 0.04)
                    d = min(d, (h[i] - h[j]) * 0.35)
                elif h[j] > h[i] + 1e-6:
                    d = -rng.uniform(0.002, 0.04)
                    d = max(d, -(h[j] - h[i]) * 0.35)
                else:
                    T *= cool
                    continue
            else:
                if h[j] > h[i] + 1e-6:
                    d = rng.uniform(0.002, 0.04)
                    d = min(d, (h[j] - h[i]) * 0.35)
                elif h[i] > h[j] + 1e-6:
                    d = -rng.uniform(0.002, 0.04)
                    d = max(d, -(h[i] - h[j]) * 0.35)
                else:
                    T *= cool
                    continue

        if abs(d) < 1e-9:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = float(c_new.max())

        de = e_new - e_old
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            accept = ratio < 25 and rng.random() < np.exp(-ratio)
        else:
            accept = False

        if accept:
            if max_new < best_c - 1e-14:
                best_c = max_new
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool

        if it_since_improve > 40000:
            h = best_h + rng.normal(0, 0.01, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.002)

    return best_h, best_c

def binary_relax(h, n, deadline, rng, alpha=3000000, maxiter=2500):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    h_relaxed = lbfgsb_optimize(h_bin.astype(float), n, alpha, deadline, maxiter=maxiter)
    return h_relaxed

def subgradient_polish(h, n, deadline, n_steps=500):
    """Take subgradient steps toward reducing the true max."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.002

    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        c_max = c_all.max()
        # Subgradient: average of gradients at max lags
        max_lags = np.where(np.abs(c_all - c_max) < 1e-10)[0]
        if len(max_lags) == 0:
            max_lags = [np.argmax(c_all)]

        grad = np.zeros(n)
        for k in max_lags[:10]:
            lag = k - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += (1.0 - h[0:n - lag])
                grad[0:n - lag] += (-h[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += (1.0 - h[idx:n])
                grad[idx:n] += (-h[0:n - idx])
        grad /= max(1, len(max_lags))

        # Projected gradient step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            step_size *= 0.8
            if step_size < 1e-6:
                break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: Progressive alpha L-BFGS-B (150s)
    alpha_schedule = [50, 200, 800, 3000, 10000, 40000, 150000, 600000, 2000000, 8000000, 30000000, 100000000]
    maxiter_schedule = [200, 300, 300, 400, 400, 500, 500, 600, 700, 800, 1000, 1200]

    for alpha, maxit in zip(alpha_schedule, maxiter_schedule):
        if time.time() > deadline - 450:
            break
        remaining = deadline - 450 - time.time()
        if remaining < 8:
            break
        t_step = min(remaining * 0.25, 20)
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + t_step, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Targeted local search (250s)
    if time.time() < deadline - 300:
        h_opt, c_opt = targeted_local_search(best_h, n, min(deadline - 250, time.time() + 250), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Subgradient polish + L-BFGS-B (100s)
    if time.time() < deadline - 250:
        h_opt, c_opt = subgradient_polish(best_h, n, time.time() + 20, n_steps=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        for alpha in [5000000, 20000000, 100000000]:
            if time.time() > deadline - 200:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 15, maxiter=1500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Micro local search (100s)
    if time.time() < deadline - 200:
        h_opt, c_opt = micro_local_search(best_h, n, min(deadline - 150, time.time() + 100), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Binary relaxation (80s)
    if time.time() < deadline - 170:
        h_bin = binary_relax(best_h, n, time.time() + 35, rng, alpha=5000000, maxiter=2500)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        h_opt = lbfgsb_optimize(best_h, n, 50000000, time.time() + 15, maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h_opt, c_opt = micro_local_search(best_h, n, time.time() + 15, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try different resolutions (150s)
    for n_try in [600, 700, 400, 800]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - 100 - time.time()
        if remaining < 35:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 200000, time.time() + min(20, remaining * 0.4), maxiter=1200)
        h_opt = lbfgsb_optimize(h_opt, n_try, 10000000, time.time() + min(15, remaining * 0.3), maxiter=1500)
        h_opt, c_opt = micro_local_search(h_opt, n_try, time.time() + min(10, remaining * 0.2), rng)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: SA at best resolution (80s)
    if time.time() < deadline - 40:
        n_final = len(best_h)
        h_opt, c_opt = simulated_annealing(best_h, n_final, min(deadline - 25, time.time() + 80), rng,
                                           alpha_sa=200000, t_init=0.003, t_min=0.000005, cool=0.99993)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, time.time() + 10, maxiter=1000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final micro local search (25s)
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt, c_opt = micro_local_search(best_h, n_final, deadline - 2, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
