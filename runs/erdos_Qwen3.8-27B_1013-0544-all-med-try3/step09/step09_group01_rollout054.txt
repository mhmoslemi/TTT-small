    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def nesterov_smooth_max(h, n, alpha, max_iter, deadline, mu=0.01):
    """Nesterov-accelerated projected gradient for smooth max approximation."""
    x = h.copy()
    y = h.copy()
    momentum = 0.0
    best_x = x.copy()
    best_c = compute_c5(x, n)
    step = 0.01  # initial step size
    
    c_vec = compute_c5_vec(y, n)
    c_max = c_vec.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Nesterov step
        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * momentum**2))
        y = x + momentum * (x - y) + t_k * (x - y)
        # Actually standard Nesterov: y = x + (t_{k-1} - 1)/t_k * (x - y_prev)
        # Let me use a simpler momentum approach
        
        # Compute gradient at y
        c_vec = compute_c5_vec(y, n)
        c_max = c_vec.max()

        sh_max = sh.max()

        x = x_new

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

            continue

        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
