        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs_smooth(h, n, deadline, alphas, mu, maxiters_list):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if i < len(maxiters_list):
            maxit = maxiters_list[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-23,
                                    'gtol': 1e-19, 'maxfun': maxit * 10,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k, alphas, mu):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(80):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 10:
                break
            maxit = max(150, int(rem2 * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, mu), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-23,
                                        'gtol': 1e-19, 'maxfun': maxit * 8,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_single_lag(h, n, deadline, alpha, maxiters):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for it in range(300):
        rem = deadline - time.time()
        if rem < 8:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(active_obj, best_h, jac=active_grad,
                           args=(n, active, alpha, 0.0005), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiters, 'ftol': 1e-23,
                                    'gtol': 1e-18, 'maxfun': maxiters * 8,
                                    'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            else:
                break
        except Exception:
            break
    return best_h, best_c

def projected_grad_descent(h, n, deadline, n_iter=50000, eta_init=0.003):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta_init
    dx = 2.0 / n
    patience = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 10:
            order = np.argsort(-c_vec[active])
            active = active[order[:10]]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.02, 0.02)
            patience = 0
        else:
            eta *= 0.95
            patience += 1
            if patience > 500:
                eta = eta_init * 0.5
                patience = 0
        best_h = project_feasible(best_h, n)
    return best_h, best_c

def coord_descent(h, n, deadline, max_rounds=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = np.argmax(c_vec)
        k = worst - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - best_h[k:n]) * dx
            grad[k:n] += (-best_h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - best_h[0:nv]) * dx
            grad[0:nv] += (-best_h[kk:n]) * dx
        neg_idx = np.where((grad < -1e-12) & (best_h < 1.0 - 1e-12))[0]
        pos_idx = np.where((grad > 1e-12) & (best_h > 1e-12))[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            break
        if len(neg_idx) > 30:
            neg_idx = neg_idx[np.argsort(np.abs(grad[neg_idx]))[::-1][:30]]
        if len(pos_idx) > 30:
            pos_idx = pos_idx[np.argsort(np.abs(grad[pos_idx]))[::-1][:30]]
        improved = False
        for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            best_move_c = best_c
            best_move = None
            for i in neg_idx:
                if time.time() > deadline:
                    break
                for j in pos_idx:
                    if i == j:
                        continue
                    test_h = best_h.copy()
                    test_h[i] += delta
                    test_h[j] -= delta
                    c_test = compute_c5(test_h, n)
                    if c_test < best_move_c:
                        best_move_c = c_test
                        best_move = (i, j, delta)
            if best_move is not None:
                i, j, d = best_move
                best_h[i] += d
                best_h[j] -= d
                best_h = project_feasible(best_h, n)
                best_c = best_move_c
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def random_restart_optimize(h0, n, deadline, n_restarts, opt_time_each, rng):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        rem = deadline - time.time()
        if rem < opt_time_each + 10:
            break
        noise = rng.standard_normal(n) * 0.05
        h_init = project_feasible(h0 + noise, n)
        c_init = compute_c5(h_init, n)
        t_end = time.time() + opt_time_each
        alphas = [10000, 100000, 1000000]
        h_try = h_init
        c_try = c_init
        for alpha in alphas:
            rem2 = t_end - time.time()
            if rem2 < 5:
                break
            maxit = max(100, int(rem2 * 3))
            try:
                res = minimize(smooth_max_obj, h_try, jac=smooth_max_grad,
                               args=(n, alpha, 0.005), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 8,
                                        'maxls': 30, 'maxcor': 30})
                h_try = project_feasible(res.x, n)
                c_try = compute_c5(h_try, n)
            except Exception:
                pass
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

def binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if n_ones < n:
        h_bin[sorted_idx[n_ones]] += max(0, remaining)
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution (30s)
    p1_end = t0 + 30
    h1, c1 = run_lbfgs_smooth(best_h, best_n, p1_end,
        alphas=[500, 5000, 50000],
        mu=0.01, maxiters_list=[500, 800, 1000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=2000
    n_target = 2000
    h_up = upsample(best_h, best_n, n_target)
    c_up = compute_c5(h_up, n_target)
    if c_up < best_c + 0.01:
        best_h = h_up
        best_c = min(best_c, c_up)
        best_n = n_target

    # Phase 3: Progressive L-BFGS at high resolution (200s)
    p3_end = min(time.time() + 200, deadline - 400)
    h3, c3 = run_lbfgs_smooth(best_h, best_n, p3_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.005,
        maxiters_list=[500, 800, 1000, 1200, 1500, 2000, 2500])
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active set optimization (150s)
    p4_end = min(time.time() + 150, deadline - 250)
    h4, c4 = run_active_set(best_h, best_n, p4_end, top_k=5,
        alphas=[100000, 500000, 2000000, 10000000], mu=0.002)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Single worst lag targeting (100s)
    p5_end = min(time.time() + 100, deadline - 180)
    h5, c5 = run_single_lag(best_h, best_n, p5_end, alpha=5000000, maxiters=500)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Projected gradient descent (100s)
    p6_end = min(time.time() + 100, deadline - 150)
    h6, c6 = projected_grad_descent(best_h, best_n, p6_end,
        n_iter=100000, eta_init=0.002)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Random restarts (80s)
    p7_end = min(time.time() + 80, deadline - 100)
    h7, c7 = random_restart_optimize(best_h, best_n, p7_end,
        n_restarts=8, opt_time_each=8, rng=rng)
    if c7 < best_c:
        best_c = c7
        best_h = h7

    # Phase 8: Coordinate descent (80s)
    p8_end = min(time.time() + 80, deadline - 50)
    h8, c8 = coord_descent(best_h, best_n, p8_end, max_rounds=30000)
    if c8 < best_c:
        best_c = c8
        best_h = h8

    # Phase 9: Final active set push (30s)
    p9_end = min(time.time() + 30, deadline - 15)
    h9, c9 = run_active_set(best_h, best_n, p9_end, top_k=3,
        alphas=[5000000, 20000000], mu=0.001)
    if c9 < best_c:
        best_c = c9
        best_h = h9

    # Phase 10: Final single lag (10s)
    p10_end = deadline - 3
    if time.time() < p10_end:
        h10, c10 = run_single_lag(best_h, best_n, p10_end, alpha=10000000, maxiters=200)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
