            if tot < 1e-14:

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    return w / w.sum()

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c_raw, alpha)
    target_sum = n / 2.0
    grad_h = np.zeros(n)
    center = n - 1
    n_active = 0
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        n_active += 1
        lag = k - center
        wk = w[k] * dx
        if lag == 0:
            grad_h += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad_h[:n - lag] += wk * (1.0 - h[lag:])
            grad_h[lag:] += wk * (-h[:n - lag])
        else:
            m = -lag
            grad_h[m:] += wk * (1.0 - h[:n - m])
            grad_h[:n - m] += wk * (-h[m:])
    return grad_h

import numpy as np

# ... utility functions ...

def slp_optimize(h, n, max_iters=100, delta_init=0.1, deadline=None):
    """SLP optimization for the Erdős problem."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    delta = delta_init
    dx = 2.0 / n
    center = n - 1
    nlag = 2 * n - 1
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        c_max = c_all.max()
        
        # Build LP
        A_ub = np.zeros((nlag, n + 1))
        b_ub = np.zeros(nlag)
        
        for k in range(nlag):
            lag = k - center
            m = abs(lag)
            if m == 0:
                A_ub[k, :n] = dx * (1.0 - 2.0 * h)
            elif lag > 0:
                A_ub[k, :n-m] += dx * (1.0 - h[m:])
                A_ub[k, m:] -= dx * h[:n-m]
            else:
                A_ub[k, m:] += dx * (1.0 - h[:n-m])
                A_ub[k, :n-m] -= dx * h[m:]
            A_ub[k, n] = -1.0
            b_ub[k] = -dx * np.dot(h[:n-m], h[m:]) if m < n else 0.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        lb = np.maximum(0.0, h - delta)
        ub = np.minimum(1.0, h + delta)
        bounds = list(zip(lb, ub))
        bounds.append((0.0, None))
        
        try:
            res = linprog(np.zeros(n + 1) + np.array([0]*n + [1.0]),
                         A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[:n]
                # Damping
                h_new = 0.5 * h + 0.5 * h_lp  # conservative damping
                # Actually use λ=1.0 first, reduce if it doesn't help
                h_new = project_feasible(h_lp, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    delta *= 0.95  # tighten trust region
                else:
                    # Try with damping
                    for lam in [0.5, 0.3, 0.1]:
                        h_damp = project_feasible((1-lam)*h + lam*h_lp, n)

                            h = h_damp

                    delta *= 0.9
            else:
                break
        except Exception:
            break
        
        # Check convergence
        if delta < 1e-6:
            break
    
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c == c.max()] = 1.0
        return w / (c.max() - c.max() + 1)  # equal weight on max
    return w / s

def grad_smooth_max(h, n, alpha):

    c_raw = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c_raw, alpha)
    grad_h = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - center
        wk = w[k] * dx
        if lag == 0:
            grad_h += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad_h[:n - lag] += wk * (1.0 - h[lag:])
            grad_h[lag:] += wk * (-h[:n - lag])
        else:
            m = -lag
            grad_h[m:] += wk * (1.0 - h[:n - m])
            grad_h[:n - m] += wk * (-h[m:])
    return grad_h

def lbfgs_opt(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    def obj_and_grad(x):
        x = np.clip(x, 0, 1)
        c_raw = np.correlate(x, 1.0 - x, mode='full')
        obj = smooth_max_value(c_raw, alpha)
        diff = x.sum() - target_sum
        obj += 1e8 * diff * diff
        g = grad_smooth_max(x, n, alpha)
        g += 2.0e8 * diff * np.ones(n)
        return obj, g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
