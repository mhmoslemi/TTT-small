  - Try h[i] += eps, h[i+k] -= eps
  - Accept if C₅ decreases

With eps = 1e-4 initially, decreasing to 1e-6.

This is O(5 * n * 4) per full sweep = O(20n) ≈ 10000 operations. Very fast.

Multiple sweeps with decreasing eps should be effective.

Wait, one more important consideration: the `np.correlate(h, 1-h, mode='full')` for n=500 gives an array of length 999. The max over all these gives C₅/dx. This is computed O(n²) times during optimization. For n=500, each correlate is about 500*500 = 250K multiplications. With numpy, this is fast (~1ms).

For the gradient computation, we need to look at each of the 999 lags and update the gradient. With the sparse structure (each lag contributes to at most n positions), this is O(n²) per gradient evaluation. For n=500, that's 250K operations, also fast.

So the bottleneck in L-BFGS-B is the number of iterations, not the per-iteration cost. Good.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        vals = sh - mx
        vals = np.where(vals > -50, vals, -50)
        return float(m + (mx + np.log(np.sum(np.exp(vals)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lag_gradient(h, n, k_idx, dx):
    """Gradient of f_k(h) = c[k_idx] w.r.t. h."""
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        g[0:n-lag] = -dx * h[lag:n]
        g[lag:n] += dx * (1.0 - h[0:n-lag])
    else:
        idx = -lag
        g[0:n-idx] = -dx * h[idx:n]
        g[idx:n] += dx * (1.0 - h[0:n-idx])
    return g

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * lag_gradient(h, n, k, 1.0)  # dx applied later
    return grad * dx

def lbfgsb_opt(h, n, alpha, maxiter, deadline):
    target_sum = n / 2.0
    def og(x):
        x = np.clip(x, 0, 1)
        c = compute_all_lags(x, n)
        v = smooth_max_val(c, alpha)
        d = x.sum() - target_sum
        v += 1e8 * d * d
        g = smooth_max_grad(x, n, alpha)
        g += 2e8 * d
        return v, g
    try:
        res = minimize(og, h, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def slsqp_constrained(h_init, n, theta, lag_indices, maxiter=200):
    """SLSQP: minimize 0 s.t. c[k] <= theta for k in lag_indices, sum(h)=n/2, 0<=h<=1"""
    dx = 2.0 / n
    target_sum = n / 2.0
    nlags = len(lag_indices)
    
    def obj(x):
        return 0.0
    def obj_j(x):
        return np.zeros(n)
    
    cons = []
    for ki in lag_indices:
        lag = ki - (n - 1)
        def cf(x, ki=ki, lag=lag):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return theta - c[ki]
        def cj(x, ki=ki, lag=lag):
            # gradient of (theta - c[ki]) = -gradient of c[ki]
            return -lag_gradient(x, n, ki, dx)
        cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})
    
    def eqf(x):
        return x.sum() - target_sum
    def eqj(x):
        return np.ones(n)
    cons.append({'type': 'eq', 'fun': eqf, 'jac': eqj})
    
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=obj_j,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hn = np.clip(res.x, 0, 1)
        return hn, res.success, compute_c5(hn, n)
    except Exception:
        return h_init.copy(), False, float('inf')

def coord_descent(h, n, deadline, rng=None):
    """Coordinate descent on the top lags."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_lags = 2 * n - 1
    
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_k = np.argsort(c)[-5:]
        improved = False
        passes = 0
        while passes < 3:
            if time.time() > deadline:
                break
            passes += 1
            c = compute_all_lags(h, n)
            top_k = np.argsort(c)[-5:]
            any_improve = False
            for ki in top_k:
                if time.time() > deadline:
                    break
                lag = ki - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                if lag > 0:
                    for i in range(0, n - m):
                        # Move mass from i to i+m (or vice versa)
                        if h[i] > h[i + m] + 1e-8:
                            d = min(eps, (h[i] - h[i + m]) * 0.5)
                            if d < 1e-10:
                                continue
                            h[i] -= d
                            h[i + m] += d
                            h = project_feasible(h, n)
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h.copy()
                                any_improve = True
                                improved = True
                            else:
                                h[i] += d
                                h[i + m] -= d
                                h = project_feasible(h, n)
                        elif h[i + m] > h[i] + 1e-8:
                            d = min(eps, (h[i + m] - h[i]) * 0.5)
                            if d < 1e-10:
                                continue
                            h[i] += d
                            h[i + m] -= d
                            h = project_feasible(h, n)
