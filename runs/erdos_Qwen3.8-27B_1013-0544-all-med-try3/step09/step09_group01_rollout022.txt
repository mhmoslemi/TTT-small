def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha


def smooth_max_grad_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w, m


def full_grad(x, n, alpha, mu):

def gradient_guided_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for scale in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        n_iters = 0
        while improved and time.time() < deadline and n_iters < 200:
            improved = False
            n_iters += 1
            remaining = int(deadline - time.time())
            if remaining <= 0:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-7, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]

            for idx in active:
                if time.time() > deadline:
                    break
                k = idx - (n - 1)
                dx = 2.0 / n
                # Compute gradient contribution for this shift
                g = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    g[0:nv] = (1.0 - best_h[k:n]) * dx
                    g[k:n] = (-best_h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    g[nv:] = (1.0 - best_h[0:nv]) * dx
                    g[0:nv] = (-best_h[kk:]) * dx

                # Try moving along -gradient at best_h[i] and +gradient at best_h[j]
                # Find i where decreasing h[i] helps most, j where increasing h[j] helps most
                # For shift k >= 0: decreasing h[i] for i in [0,nv) helps, increasing h[i] for i in [k,n) helps
                # Actually let's just try pairs based on gradient
                desc_idx = np.argsort(g)  # most negative gradient = best to increase
                asc_idx = np.argsort(-g)  # most positive gradient = best to decrease

                tried = 0
                for ii in range(min(20, n)):
                    if time.time() > deadline:
                        break
                    for jj in range(min(20, n)):
                        tried += 1
                        if tried > 200:
                            break
                        i = desc_idx[ii]
                        j = asc_idx[jj]
                        if i == j:
                            continue
                        if best_h[i] + scale <= 1.0 and best_h[j] - scale >= 0.0:
                            best_h[i] += scale
                            best_h[j] -= scale

                                improved = True
                            else:
                                best_h[i] -= scale

                        if best_h[i] - scale >= 0.0 and best_h[j] + scale <= 1.0:

                            best_h[j] += scale
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                improved = True

def subgrad_refine(h, n, deadline, n_iter=5000):

    best_c = compute_c5(h, n)
    eta = 0.003

    for it in range(n_iter):
        if time.time() > deadline:

        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()

        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[nv:] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:]) * dx
        grad /= len(active)

        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)

def upsample(h_low, n_low, n_high):
    x_low = np.linspace(0, 2, n_low)
    x_high = np.linspace(0, 2, n_high)
    h_high = np.interp(x_high, x_low, h_low)
    return project_feasible(h_high, n_high)

            best_n = n_high

        # Active set at high res
        p4b_end = min(time.time() + 100, deadline - 150)
        if time.time() < p4b_end:
            h_act2, c_act2 = run_active_set_opt(best_h, best_n, p4b_end, top_k=6, n_rounds=15)
            if c_act2 < best_c:
                best_c = c_act2
                best_h = h_act2

    # Phase 5: Try even higher resolution if time allows (150s)
    p5_end = min(time.time() + 150, deadline - 100)
    n_higher = 2000
    if time.time() < p5_end and best_n < n_higher:
        h_up3 = upsample(best_h, best_n, n_higher)
        c_up3 = compute_c5(h_up3, n_higher)
        if c_up3 < best_c:
            best_c = c_up3
            best_h = h_up3
            best_n = n_higher

        alphas_hi2 = [5, 20, 100, 500, 2000]
        h_up4, c_up4 = run_lbfgs_anneal(h_up3, n_higher, p5_end, alphas_hi2, 0.003,
                                         [1000, 1500, 2000, 2000, 2500])
        if c_up4 < best_c:
            best_c = c_up4
            best_h = h_up4
            best_n = n_higher

    # Phase 6: Active set at final resolution (80s)
    p6_end = min(time.time() + 80, deadline - 50)
    if time.time() < p6_end:
        h_a3, c_a3 = run_active_set_opt(best_h, best_n, p6_end, top_k=5, n_rounds=20)
        if c_a3 < best_c:
            best_c = c_a3
            best_h = h_a3

    # Phase 7: Gradient-guided search (60s)
    p7_end = min(time.time() + 60, deadline - 20)
    if time.time() < p7_end:
        h_g, c_g = gradient_guided_search(best_h, best_n, p7_end)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    # Phase 8: Subgradient refinement (30s)
    p8_end = min(time.time() + 30, deadline - 5)
    if time.time() < p8_end:
        h_s, c_s = subgrad_refine(best_h, best_n, p8_end, n_iter=3000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Final: one more quick active set pass
    if time.time() < deadline - 2:
        h_f, c_f = run_active_set_opt(best_h, best_n, deadline - 1, top_k=4, n_rounds=5)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
