            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:k+nv])
            grad[k:k+nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h, n_new):
    """Upsample h from n_old to n_new using linear interpolation."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def equalize_step(h, n, top_k, step_size):
    """Try to equalize the top-k correlations by moving mass."""
    c = compute_c5_vec(h, n)
    top_idx = np.argsort(-c)[:top_k]
    c_max = c[top_idx[0]]
    c_min_top = c[top_idx[-1]]
    
    if c_max - c_min_top < 1e-10:
        return h, compute_c5(h, n)
    
    # For the max correlation shift, decrease it
    # For the min (among top-k) correlation shift, increase it
    # This is a heuristic: move mass from where it contributes to max 
    # to where it contributes to min
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    max_idx = top_idx[0]
    k_max = max_idx - (n - 1)
    
    # Find pairs (i, j) where increasing h[i] and decreasing h[j] 
    # would decrease c_max and increase c_min_top
    # This is complex, so just do random small moves and keep improvements
    
    for _ in range(100):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        d = step_size
        if h[i] + d <= 1 and h[j] - d >= 0:
            h[i] += d
            h[j] -= d
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                best_h = project_feasible(best_h, n)
            else:
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
        elif h[i] - d >= 0 and h[j] + d <= 1:
            h[i] -= d
            h[j] += d
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                best_h = project_feasible(best_h, n)
            else:
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    """Systematic pair search for fine improvement."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_per=None):
    """L-BFGS with smooth max, alpha annealing."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxit_per is not None and i < len(maxit_per):
            maxit = maxit_per[i]
        else:
            maxit = max(500, int(remaining * 3))
        
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active(h, n, deadline, top_k=8, alphas=None):

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                # Re-identify active set
                c_vec = compute_c5_vec(best_h, n)
                threshold = c_vec.max() - max(1e-8, 0.001 * c_vec.max())
                active = np.where(c_vec >= threshold)[0]
                if len(active) > top_k:
                    active = np.array(np.argsort(-c_vec[active])[:top_k])
            except Exception:
                pass
    
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    """Subgradient refinement for the max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        center = n - 1
        for idx in active:
            k = idx - center
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:k+nv]) * dx
                grad[k:k+nv] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:kk+nv] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:kk+nv]) * dx
        grad /= len(active)
        
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.01)
        else:
            eta *= 0.99
        
        h = h_new
    
    return best_h, best_c

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(40):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]].copy()
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.004), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                c_vec = compute_c5_vec(best_h, n)
                threshold = c_vec.max() - max(1e-8, 0.001 * c_vec.max())
                active = np.where(c_vec >= threshold)[0]
                if len(active) > top_k:
                    order = np.argsort(-c_vec[active])
                    active = active[order[:top_k]].copy()
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    tol_expand = 0.003
    center = n - 1
    dx_val = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - center
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:k+nv]) * dx_val
                grad[k:k+nv] += (-h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:kk+nv] += (1.0 - h[:nv]) * dx_val
                grad[:nv] += (-h[kk:kk+nv]) * dx_val
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.99
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, radius=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] = min(1.0, max(0.0, remaining))
    return project_feasible(h_bin, n)

def perturb_restart(h, n, rng, amplitude=0.02):
    h_pert = h.copy()
    mask = rng.random(n) < 0.3
    h_pert[mask] += amplitude * (rng.random(np.sum(mask)) * 2 - 1)
    return project_feasible(h_pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS at original resolution (n0=500)
    p1_end = t0 + min(80, budget_s * 0.08)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[200, 2000, 20000, 100000, 500000],
        mu=0.01, maxit_per=[1000, 1500, 2000, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000, main L-BFGS optimization
    n_main = 1000
    h_main = upsample_h(best_h, n_main)
    c_main = compute_c5(h_main, n_main)

    p2_end = t0 + min(400, budget_s * 0.38)
    h2, c2 = run_lbfgs(h_main, n_main, p2_end,
        alphas=[30, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000],
        mu=0.004, maxit_per=[800, 1000, 1500, 1500, 2000, 2500, 2500, 3000, 3000, 3000])

    if c2 < max(best_c, c_main):
        best_c = c2
        best_h = h2
        best_n = n_main

    # Phase 3: Active set refinement at main resolution
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 130, deadline - 120)
        h3, c3 = run_active_set(best_h, best_n, p3_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000, 20000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try n=2000 for finer resolution
    n_high = 2000
    if time.time() < deadline - 180:
        h_high = upsample_h(best_h, n_high)
        p4_end = min(time.time() + 160, deadline - 60)
        h4, c4 = run_lbfgs(h_high, n_high, p4_end,
            alphas=[30, 200, 2000, 20000, 200000, 2000000],
            mu=0.003, maxit_per=[500, 800, 1000, 1500, 2000, 2000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high

    # Phase 5: Subgradient refinement at best resolution
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 60, deadline - 50)
        h5, c5 = subgrad_refine(best_h, best_n, p5_end, n_iter=8000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another active set pass
    if time.time() < deadline - 50:
        p6_end = min(time.time() + 35, deadline - 25)
        h6, c6 = run_active_set(best_h, best_n, p6_end, top_k=4,
            alphas=[100000, 1000000, 10000000, 50000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation restart + quick re-optimize
    if time.time() < deadline - 40:
        h_pert = perturb_restart(best_h, best_n, rng, amplitude=0.03)
        c_pert = compute_c5(h_pert, best_n)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
        if time.time() < deadline - 25:
            p7_end = deadline - 20
            h7, c7 = run_lbfgs(h_pert, best_n, p7_end,
                alphas=[10000, 100000, 1000000],
                mu=0.005, maxit_per=[1000, 1500, 1500])
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Subgradient final
    if time.time() < deadline - 20:
        p8_end = deadline - 10
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=3000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Pair search for final fine-tuning
    if time.time() < deadline - 5:
        p9_end = deadline - 1
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
