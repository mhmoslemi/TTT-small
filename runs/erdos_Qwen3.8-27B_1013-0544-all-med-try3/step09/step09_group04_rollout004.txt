import time

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            step = min(1.0, diff / tot)
            h += step * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            step = min(1.0, -diff / tot)
            h -= step * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at(h, n, shift_idx):
    """Compute subgradient at a specific shift."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

    """Average subgradient at active shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += (1.0 - h[:nv]) * dx
            grad[:nv] += (-h[kk:]) * dx
    return grad / len(active_idx)

def minimax_subgrad(h, n, deadline, max_iter=20000):
    """Direct minimax with subgradient + backtracking."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 1.0 / n

        if time.time() > deadline:

        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        g = vec_subgrad(h, n, active)
