For c_m = sum_{s=0}^{n-1-m} h[s+m]*(1-h[s]):
If I do h[i] -= d, h[i+m] += d:
- Term s=i changes: (h[i+m]+d)(1-h[i]+d) - h[i+m](1-h[i]) = d(1-h[i]) + d*h[i+m] + d²
- Term s=i-m changes (if i>=m): (h[i]-d)(1-h[i-m]) - h[i](1-h[i-m]) = -d(1-h[i-m])

So Δc_m = d(1-h[i]) + d*h[i+m] + d² - d(1-h[i-m]) [if i>=m]
Or: Δc_m = d(1-h[i]) + d*h[i+m] + d² [if i<m]

For the reverse swap (h[i] += d, h[i+m] -= d):

Or: Δc_m = -d(1-h[i+m]) - d*h[i] - d² [if i<m]

- Term s=i: (h[i+m]-d)(1-(h[i]+d)) - h[i+m](1-h[i])

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_full(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_opt(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    pen = 1e8

    def f(x):
        x = np.clip(x, 0, 1)
        c = compute_all_lags(x, n)
        val = smooth_max(c, alpha) + pen * (x.sum() - target_sum) ** 2
        g = grad_full(x, n, alpha) + 2.0 * pen * (x.sum() - target_sum)
        return val, g

    bounds = [(0, 1)] * n
    try:
        res = minimize(f, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40, 'iprint': -1})
        if time.time() > deadline:
            return project_feasible(h.copy(), n)
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interp_h(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return project_feasible(np.interp(xd, xs, h_src), n_dst)

def fast_local_search(h, n, deadline, max_rounds=1000000):
    """O(n) per round local search using O(1) correlation delta formulas."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds = 0
    stall = 0
    target = 5

    while time.time() < deadline and rounds < max_rounds:
        c_all = compute_all_lags(h, n)
        max_val = c_all.max()

        if max_val < best_c - 1e-15:
            best_c = float(max_val)
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            break

        top_idx = np.argsort(c_all)[-target:]
        found = False

        for k_star in top_idx:
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            best_i = -1
            best_delta = -1e-18
            best_d = 0.0
            best_rev = False

            n_pairs = n - m
            for i in range(n_pairs):
                hi = h[i]
                hj = h[i + m]

                # Forward: h[i]-=d, h[i+m]+=d
                if i >= m:
                    B_f = hj + h[i - m] - hi
                else:
                    B_f = hj + 1.0 - hi
                if B_f < -1e-12:
                    d_max_f = min(hi, 1.0 - hj, 0.5)
                    if d_max_f > 1e-9:
                        d_opt = min(-B_f / 2.0, d_max_f)
                        if d_opt > 1e-9:
                            delta = d_opt * d_opt + B_f * d_opt
                            if delta < best_delta:
                                best_delta = delta
                                best_i = i
                                best_d = d_opt
                                best_rev = False

                # Reverse: h[i]+=d, h[i+m]-=d
                if i >= m:
                    B_r = hi + h[i - m] - hj - 1.0
                else:
                    B_r = hi - hj - 1.0
                if B_r < -1e-12:
                    d_max_r = min(1.0 - hi, hj, 0.5)
                    if d_max_r > 1e-9:
                        d_opt = min(-B_r / 2.0, d_max_r)
                        if d_opt > 1e-9:
                            delta = d_opt * d_opt + B_r * d_opt
                            if delta < best_delta:
                                best_delta = delta
                                best_i = i
                                best_d = d_opt
                                best_rev = True

            if best_i >= 0:
                if best_rev:
                    h[best_i] += best_d
                    h[best_i + m] -= best_d
                else:
                    h[best_i] -= best_d
                    h[best_i + m] += best_d
                h = project_feasible(h, n)
                found = True
                break

        if not found:
            # Try a broader scan with smaller target
            top_idx2 = np.argsort(c_all)[-min(20, 2 * n - 1):]
            for k_star in top_idx2:
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                best_i2 = -1
                best_delta2 = -1e-18
                best_d2 = 0.0
                best_rev2 = False
                step = max(1, (n - m) // 200)
                for i in range(0, n - m, step):
                    hi = h[i]
                    hj = h[i + m]
                    if i >= m:
                        B_f = hj + h[i - m] - hi
                    else:
                        B_f = hj + 1.0 - hi
                    if B_f < -1e-12:
                        d_max_f = min(hi, 1.0 - hj, 0.3)
                        if d_max_f > 1e-9:
                            d_opt = min(-B_f / 2.0, d_max_f)
                            if d_opt > 1e-9:
                                delta = d_opt * d_opt + B_f * d_opt
                                if delta < best_delta2:
                                    best_delta2 = delta
                                    best_i2 = i
                                    best_d2 = d_opt
                                    best_rev2 = False
                    if i >= m:
                        B_r = hi + h[i - m] - hj - 1.0
                    else:
                        B_r = hi - hj - 1.0
                    if B_r < -1e-12:
                        d_max_r = min(1.0 - hi, hj, 0.3)
                        if d_max_r > 1e-9:
                            d_opt = min(-B_r / 2.0, d_max_r)
                            if d_opt > 1e-9:
                                delta = d_opt * d_opt + B_r * d_opt
                                if delta < best_delta2:
                                    best_delta2 = delta
                                    best_i2 = i
                                    best_d2 = d_opt
                                    best_rev2 = True
                if best_i2 >= 0:
                    if best_rev2:
                        h[best_i2] += best_d2
                        h[best_i2 + m] -= best_d2
                    else:
                        h[best_i2] -= best_d2
                        h[best_i2 + m] += best_d2
                    h = project_feasible(h, n)
                    found = True
                    break

            if not found:
                break

        rounds += 1

    return best_h, best_c, rounds

def sa_lag_aware(h, n, deadline, rng, t_init=0.006, t_min=0.00003, cool=0.99992, alpha=100000):
    """SA with structured moves targeting worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    no_improve = 0

    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        e_old = smooth_max(c_all, alpha)
        max_old = c_all.max()

        r = rng.random()
        if r < 0.75:
            # Target a top lag
            top3 = np.argsort(c_all)[-3:]
            k_star = top3[rng.randint(3)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
        else:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue

        max_d = min(0.02, 0.003 + T * 3.0)
        if h[i] > h[j] + 1e-8:
            d = rng.uniform(0.001, max_d)
            d = min(d, h[i] * 0.4, (1.0 - h[j]) * 0.4)
        elif h[j] > h[i] + 1e-8:
            d = -rng.uniform(0.001, max_d)
            d = max(d, -h[j] * 0.4, -(1.0 - h[i]) * 0.4)
        else:
            d = rng.uniform(-0.004, 0.004)

        if abs(d) < 1e-10:
            T *= cool
            continue

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max(c_new, alpha)
        max_new = c_new.max()

        de = e_new - e_old
        accept = False
        if de < -1e-14:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            no_improve += 1

        T *= cool

        if no_improve > 40000:
            h = best_h + rng.normal(0, 0.004, n)
            h = project_feasible(h, n)
            no_improve = 0
            T = max(T, 0.002)

    return best_h, best_c

def multi_start_explore(h0, n, deadline, rng, num_starts=3):
    """Try a few perturbed starts and keep the best."""
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    for s in range(num_starts):
        if time.time() > deadline:
            break
        perturb_scale = [0.01, 0.03, 0.06][s % 3]
        h_try = h0 + rng.normal(0, perturb_scale, n)
        h_try = project_feasible(h_try, n)

        # Quick gradient optimization
        h_try = lbfgsb_opt(h_try, n, 50000, time.time() + 20, maxiter=1500)
        c_try = compute_c5(h_try, n)

        # Quick local search
        h_try, c_try, _ = fast_local_search(h_try, n, time.time() + min(30, max(0, deadline - time.time())))

        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n_best = 500

    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)

    # Phase 1: Gradient optimization at parent resolution (~60s)
    for alpha in [3000, 30000, 300000, 3000000]:
        if time.time() > deadline - 750:
            break
        h_opt = lbfgsb_opt(h_best, n_best, alpha, time.time() + 25, maxiter=2500)
        c_opt = compute_c5(h_opt, n_best)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 2: Fast local search at parent resolution (~150s)
    if time.time() < deadline - 500:
        ls_dead = min(deadline - 450, time.time() + 180)
        h_opt, c_opt, _ = fast_local_search(h_best, n_best, ls_dead)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 3: Expand to n=1000 and optimize (~120s)
    n2 = 1000
    if time.time() < deadline - 400:
        h2 = interp_h(h_best, n_best, n2)
        c2 = compute_c5(h2, n2)
        if c2 < c_best:
            c_best = c2
            h_best = h2
            n_best = n2

        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 350:
                break
            h_opt = lbfgsb_opt(h_best, n_best, alpha, time.time() + 35, maxiter=2000)
            c_opt = compute_c5(h_opt, n_best)
            if c_opt < c_best:
                c_best = c_opt
                h_best = h_opt

    # Phase 4: Intensive local search at best resolution (~200s)
    if time.time() < deadline - 250:
        ls_dead = min(deadline - 200, time.time() + 250)
        h_opt, c_opt, _ = fast_local_search(h_best, n_best, ls_dead)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 5: SA for diversification (~100s)
    if time.time() < deadline - 150:
        sa_dead = min(deadline - 120, time.time() + 130)
        h_opt, c_opt = sa_lag_aware(h_best, n_best, sa_dead, rng,
                                     t_init=0.005, t_min=0.00005, cool=0.99995, alpha=80000)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 6: Multi-start exploration (~80s)
    if time.time() < deadline - 100:
        ms_dead = min(deadline - 60, time.time() + 100)
        h_opt, c_opt = multi_start_explore(h_best, n_best, ms_dead, rng, num_starts=3)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 7: Another local search round (~50s)
    if time.time() < deadline - 50:
        ls_dead = min(deadline - 20, time.time() + 60)
        h_opt, c_opt, _ = fast_local_search(h_best, n_best, ls_dead)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 8: Final gradient polish (~20s)
    if time.time() < deadline - 10:
        h_opt = lbfgsb_opt(h_best, n_best, 50000000, time.time() + 15, maxiter=3000)
        c_opt = compute_c5(h_opt, n_best)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Phase 9: Very final local search
    if time.time() < deadline - 2:
        h_opt, c_opt, _ = fast_local_search(h_best, n_best, deadline - 1)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt

    # Final validation
    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)

    return h_best, c_best, n_best
