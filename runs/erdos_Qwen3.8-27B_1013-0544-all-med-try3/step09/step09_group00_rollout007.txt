            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_active_grads(h, n, active_shifts):
    """Compute gradients of overlap_k w.r.t. h for given shift indices in full correlation."""
    dx = 2.0 / n
    grads = np.zeros((len(active_shifts), n))
    for idx, s in enumerate(active_shifts):
        k = s - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            # h[j] as h[i+k]: j in [k, n-1], contributes (1-h[j-k])*dx
            g[k:] += (1.0 - h[:nv]) * dx
            # h[j] in (1-h[i]): j in [0, n-1-k], contributes -h[j+k]*dx
            g[:nv] += (-h[k:]) * dx
        else:
            kk = -k
            nv = n - kk
            # h[j] as h[i]: j in [kk, n-1], contributes (1-h[j-kk])*dx
            g[kk:] += (1.0 - h[:nv]) * dx
            # h[j] in (1-h[i-kk]): j in [0, n-1-kk], contributes -h[j+kk]*dx
            g[:nv] += (-h[kk:]) * dx
        grads[idx] = g
    return grads

def lp_direction(h, n, top_k, step_max):
    """Solve LP to find best descent direction for the max overlap."""
    c = compute_c5_vec(h, n)
    c_max = c.max()
    eps = max(1e-10, 0.003 * c_max)
    active_all = np.where(c >= c_max - eps)[0]
    if len(active_all) > top_k:
        active_all = np.array(np.argsort(-c[active_all])[:top_k])
    if len(active_all) == 0:
        return None
    
    grads = compute_active_grads(h, n, active_all)
    c_vals = c[active_all]
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0
    
    # Inequality: -t + g_k . d <= -c_k for each active k
    A_ub_row = np.zeros((len(active_all), n_vars))
    A_ub_row[:, 0] = -1.0
    A_ub_row[:, 1:] = grads
    b_ub = -c_vals
    
    # Bounds
    lb = np.zeros(n_vars)
    ub = np.full(n_vars, np.inf)
    lb[0] = -np.inf  # t is free
    for i in range(n):
        lb[i+1] = max(-h[i], -step_max)
        ub[i+1] = min(1.0 - h[i], step_max)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs')
        if res.status == 0 and res.x is not None:
            delta = res.x[1:]
            delta = project_feasible(h + delta, n) - h
            return delta
    except Exception:
        pass
    return None

def lp_optimize(h, n, deadline, top_k=20, initial_step=0.02, min_steps=50):
    """LP-based direction finding optimization."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step
    stall = 0
    
    while time.time() < deadline - 5:
        delta = lp_direction(best_h, n, top_k, step)
        if delta is None:
            break
        
        # Line search: try fraction of delta
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(best_h + frac * delta, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try
                improved = True
                break
        
        if improved:
            stall = 0
            if stall < min_steps:
                pass
        else:
            stall += 1
            step *= 0.7
            if step < 0.0005:
                break
            # Try reducing top_k for more focused direction
                top_k = max(5, top_k - 2)
    
    return best_h, best_c

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(lbfgs_smooth_obj, best_h, jac=lbfgs_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, deadline, deltas=None, max_passes=20):
    """Fine coordinate descent with pair swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            # Compute gradient to guide direction
            c_vec = compute_c5_vec(best_h, n)
            c_max = c_vec.max()
            tol = max(1e-8, 0.005 * c_max)
            active = np.where(c_vec >= c_max - tol)[0]
            
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            
            # Greedy: find best pair (increase low-grad, decrease high-grad)
            # Group by gradient sign
            pos = np.where((grad > 1e-8) & (best_h < 1.0 - d))[0]
            neg = np.where((grad < -1e-8) & (best_h > d))[0]
            
            if len(pos) == 0 or len(neg) == 0:
                break
            
            # Try top candidates
            pos_sorted = pos[np.argsort(-grad[pos])][:30]
            neg_sorted = neg[np.argsort(grad[neg])][:30]
            
            for i in pos_sorted:
                for j in neg_sorted:
                    if time.time() > deadline:
                        break
                    amt = min(d, best_h[j], 1.0 - best_h[i])
                    if amt < 1e-10:
                        continue
                    best_h[j] -= amt
                    best_h[i] += amt
                    best_h = project_feasible(best_h, n)
                    c_try = compute_c5(best_h, n)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        improved = True
                    else:
                        best_h[i] -= amt
                        best_h[j] += amt
                        best_h = project_feasible(best_h, n)
            if not improved:
                break
    return best_h, best_c

def multi_seed_optimize(h, n, deadline, n_seeds=4, sigma=0.02):
    """Try small random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    per_seed = (deadline - time.time()) / n_seeds - 5
    
    for s in range(n_seeds):
        if time.time() > deadline - 10:
            break
        t_end = time.time() + per_seed
        h_pert = best_h + rng.randn(n) * sigma
        h_pert = project_feasible(h_pert, n)
        h_new, c_new = phase_lbfgs(h_pert, n, t_end, alphas=[500, 5000, 50000])
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(5, int(deadline - time.time()))
        maxit = min(maxiter, max(100, rem * 5))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h, n_old, n_new):
    """Upsample from n_old to n_new points, preserving integral."""
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=None)
    h_new = project_feasible(h_new, n_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    rng = np.random.RandomState(seed)
    
    # Load initial or create default
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    n = n_orig
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Quick alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        make_beatty_start(n, golden / 2.0),
        make_beatty_start(n, np.sqrt(2.0) / 2.0),
        make_beatty_start(n, np.sqrt(3.0) / 2.0),
    ]
    for hs in starts:
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()
    
    # Phase 1: L-BFGS warmup (40s)
    p1_end = min(t0 + 40, deadline - 800)
    if p1_end > t0 + 5:
        h1, c1 = phase_lbfgs(best_h, n, p1_end,
            alphas=[30, 100, 500, 2000, 10000, 50000],
            mu=0.01, maxit_list=[500, 800, 1000, 1200, 1500, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: LP-based optimization (main - 350s)
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 350, deadline - 450)
        h2, c2 = lp_optimize(best_h, n, p2_end, top_k=20, initial_step=0.015)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: L-BFGS polish after LP
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 100, deadline - 300)
        h3, c3 = phase_lbfgs(best_h, n, p3_end,
            alphas=[5000, 20000, 100000, 500000, 2000000],
            mu=0.005, maxit_list=[800, 1000, 1500, 1500, 1500])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Second LP round with tighter active set
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 150, deadline - 200)
        h4, c4 = lp_optimize(best_h, n, p4_end, top_k=10, initial_step=0.008)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Coordinate descent
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 120, deadline - 150)
        h5, c5 = coord_descent(best_h, n, p5_end,
            deltas=[0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            max_passes=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Multi-seed perturbation
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 80, deadline - 100)
        h6, c6 = multi_seed_optimize(best_h, n, p6_end, n_seeds=3, sigma=0.015)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 60:
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > deadline - 25:
                break
            p7_end = time.time() + min(20, deadline - time.time() - 10)
            h7, c7 = run_slqp(best_h, n, p7_end, alpha=alpha, maxiter=800)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Final LP + coord descent
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 15, deadline - 15)
        h8, c8 = lp_optimize(best_h, n, p8_end, top_k=8, initial_step=0.005)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    if time.time() < deadline - 5:
        h9, c9 = coord_descent(best_h, n, deadline - 3,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001],
            max_passes=15)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
