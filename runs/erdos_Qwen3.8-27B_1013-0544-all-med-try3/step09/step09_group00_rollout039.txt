    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def alt_linearization(h, n, deadline, K_max=40, max_iters=3000, solver='ECOS'):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    last_c = best_c
    h_var = cp.Variable(n)
    t_var = cp.Variable()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 50 == 0 and it > 0:
            if abs(last_c - best_c) < 1e-10:
                stall += 1
            else:
                stall = 0
            if stall > 3:
                break
            last_c = best_c

        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()

        threshold = max_val - max(1e-9, 0.0002 * max_val)
        active_all = np.where(c_vec >= threshold)[0]
        K = min(K_max, len(active_all))
        if K == 0:
            break
        if len(active_all) > K:
            order = np.argsort(-c_vec[active_all])
            active = active_all[order[:K]]
        else:
            active = active_all

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

        for k_idx in active:
            k = k_idx - (n - 1)
            if k >= 0:
                w = 1.0 - best_h[k:n]
                valid = w > 1e-12
                if valid.sum() == 0:
                    continue
                idxs = np.where(valid)[0]
                c_lin = dx * cp.sum(cp.multiply(w[valid], h_var[idxs]))
            else:
                kk = -k
                w = 1.0 - best_h[:n - kk]
                valid = w > 1e-12
                if valid.sum() == 0:
                    continue
                idxs = np.where(valid)[0] + kk
                c_lin = dx * cp.sum(cp.multiply(w[valid], h_var[idxs]))
            constraints.append(t_var >= c_lin)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver='SCS', verbose=False)
            except Exception:
                try:
                    prob.solve(solver='GLPK', verbose=False)
                except Exception:
                    break

        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break

        new_h = np.asarray(h_var.value).flatten()
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)

        if new_c < best_c - 1e-12:
            best-h_ignore = None
            best_h = new_h
            best_c = new_c
        else:
            # Try damped step
            for damp in [0.5, 0.3, 0.1]:
                damped_h = project_feasible(best_h + damp * (new_h - best_h), n)
                damped_c = compute_c5(damped_h, n)
                if damped_c < best_c - 1e-12:
                    best_h = damped_h
                    best_c = damped_c
                    break

    return best_h, best_c

def alt_linearization_v2(h, n, deadline):
    """V2: adaptive K, with periodic L-BFGS restarts."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0

    for phase in range(8):
        if time.time() > deadline:
            break
        phase_deadline = min(deadline, time.time() + (deadline - time.time()) / (8 - phase))
        K_for_phase = 15 + phase * 5

        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, (0.0001 + phase * 0.0001) * max_val)
        active_all = np.where(c_vec >= threshold)[0]
        K = min(K_for_phase, max(5, len(active_all)))
        if K == 0:
            break
        if len(active_all) > K:
            order = np.argsort(-c_vec[active_all])
            active = active_all[order[:K]]
        else:
            active = active_all

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

        for k_idx in active:
            k = k_idx - (n - 1)
            if k >= 0:
                w = 1.0 - best_h[k:n]
                mask = w > 1e-10
                if mask.sum() < 1:
                    continue
                idxs = np.where(mask)[0]
                c_lin = dx * cp.sum(cp.multiply(w[mask], h_var[idxs]))
            else:
                kk = -k
                w = 1.0 - best_h[:n - kk]
                mask = w > 1e-10
                if mask.sum() < 1:
                    continue
                idxs = np.where(mask)[0] + kk
                c_lin = dx * cp.sum(cp.multiply(w[mask], h_var[idxs]))
            constraints.append(t_var >= c_lin)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            try:
                prob.solve(solver='SCS', verbose=False)
            except Exception:
                break

        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break

        new_h = np.asarray(h_var.value).flatten()
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)

        if new_c <= best_c + 1e-12:
            best_h = new_h
            best_c = new_c
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break

    return best_h, best_c

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad_full(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas=None, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(lm_obj, best_h, jac=lm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, top_k=8, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if alphas is None:
        alphas = [100000, 1000000, 5000000, 20000000]
    bounds = [(0.0, 1.0)] * n

    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            break

        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(100, int((deadline - time.time()) * 1.5))

            def _topk_obj(x):
                c = compute_c5_vec(x, n)
                vals = c[active]
                return smooth_max_val(vals, alpha) + 0.001 * (x.sum() - n / 2.0) ** 2

            def _topk_grad(x):
                c = compute_c5_vec(x, n)
                vals = c[active]
                m = vals.max()
                sh = alpha * (vals - m)
                smx = sh.max()
                w = np.exp(sh - smx)
                w /= w.sum()
                dx_val = 2.0 / n
                grad = np.zeros(n)
                for wi, idx in zip(w, active):
                    k = idx - (n - 1)
                    wm = wi * dx_val
                    if k >= 0:
                        nv = n - k
                        grad[0:nv] += wm * (1.0 - x[k:n])
                        grad[k:n] += wm * (-x[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        grad[kk:n] += wm * (1.0 - x[0:nv])
                        grad[0:nv] += wm * (-x[kk:n])
                grad += 0.002 * (x.sum() - n / 2.0)
                return grad

            try:
                res = minimize(_topk_obj, best_h, jac=_topk_grad,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=40, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def triple_flip_search(h, n, deadline, deltas=None):
    """Try flipping 2 positions simultaneously (one down, one up) with a 3rd adjustment."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002]
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    rng = np.random.RandomState(123)
    for d in deltas:
        while time.time() < deadline:
            found = False
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h[i] -= d
            best_h[j] += d
            best_h = project_feasible(best_h, n)
            c = compute_c5(best_h, n)
            if c < best_c - 1e-16:
                best_c = c
                found = True
            else:
                best_h[i] += d
                best_h[j] -= d
                best_h = project_feasible(best_h, n)
            if not found:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 500

    # Phase 1: Generate and evaluate diverse initial points
    p1_deadline = t0 + 40
    best_h = None
    best_c = np.inf
    best_n = 500

    if h0 is not None:
        h0 = project_feasible(h0, n0)
        c0 = compute_c5(h0, n0)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
            best_n = n0

    # Beatty sequences with various slopes
    n_options = [400, 500, 600, 800]
    alpha_list = [0.453, 0.472, 0.489, 0.500, 0.511, 0.524,
                  0.538, 0.553, 0.571, 0.585, 0.618, 0.651,
                  0.707, 0.724, 0.754, 0.786]

    for n_try in n_options:
        if time.time() > p1_deadline:
            break
        for alpha in alpha_list:
            if time.time() > p1_deadline:
                break
            h_b = make_beatty(n_try, alpha)
            c_b = compute_c5(h_b, n_try)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b.copy()
                best_n = n_try

    # Random binary starts
    rng = np.random.RandomState(seed)
    for _ in range(8):
        if time.time() > p1_deadline:
            break
        n_r = 500
        h_r = make_random_binary(n_r, rng)
        c_r = compute_c5(h_r, n_r)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            best_n = n_r

    # Phase 2: Alternating Linearization (main optimization)
    p2_deadline = min(time.time() + 500, deadline - 300)
    if time.time() < p2_deadline:
        h_al, c_al = alt_linearization(best_h, best_n, p2_deadline, K_max=30, max_iters=2000)
        if c_al < best_c:
            best_c = c_al
            best_h = h_al.copy()
            best_n = len(h_al)

    # Phase 2b: Second AL pass with different parameters
    p2b_deadline = min(time.time() + 200, deadline - 200)
    if time.time() < p2b_deadline:
        h_al2, c_al2 = alt_linearization_v2(best_h, best_n, p2b_deadline)
        if c_al2 < best_c:
            best_c = c_al2
            best_h = h_al2.copy()
            best_n = len(h_al2)

    # Phase 3: L-BFGS polish
    p3_deadline = min(time.time() + 150, deadline - 100)
    if time.time() < p3_deadline:
        h_lb, c_lb = phase_lbfgs(best_h, best_n, p3_deadline,
                                  alphas=[200, 2000, 20000, 100000, 500000, 2000000])
        if c_lb < best_c:
            best_c = c_lb
            best_h = h_lb.copy()
            best_n = len(h_lb)

    # Phase 4: Active set L-BFGS
    p4_deadline = min(time.time() + 100, deadline - 60)
    if time.time() < p4_deadline:
        h_as, c_as = active_set_lbfgs(best_h, best_n, p4_deadline, top_k=10,
                                       alphas=[500000, 2000000, 10000000, 50000000], rounds=10)
        if c_as < best_c:
            best_c = c_as
            best_h = h_as.copy()
            best_n = len(h_as)

    # Phase 5: Gradient-guided search
    p5_deadline = min(time.time() + 80, deadline - 30)
    if time.time() < p5_deadline:
        h_gg, c_gg = grad_guided_search(best_h, best_n, p5_deadline,
                                         step_sizes=[0.002, 0.001, 0.0005, 0.0002,
                                                     0.0001, 0.00005, 0.00002],
                                         max_steps=50000)
        if c_gg < best_c:
            best_c = c_gg
            best_h = h_gg.copy()
            best_n = len(h_gg)

    # Phase 6: Fine pair search
    p6_deadline = min(time.time() + 50, deadline - 5)
    if time.time() < p6_deadline:
        h_fp, c_fp = fine_pair_search(best_h, best_n, p6_deadline, radius=30,
                                       deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c_fp < best_c:
            best_c = c_fp
            best_h = h_fp.copy()
            best_n = len(h_fp)

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
