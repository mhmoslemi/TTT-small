import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def next_pow2(n):
    return 1 << (n - 1).bit_length()

def compute_c5_fft(h, n):
    """Compute C5 correlation using FFT for O(n log n)."""
    dx = 2.0 / n
    N = next_pow2(2 * n - 1)
    H = fft(h, N)
    G = fft((1.0 - h)[::-1], N)
    corr = ifft(H * G).real[:2*n-1]
    return corr * dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_c5_fft(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_fft(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def next_pow2(n):
    return 1 << (n - 1).bit_length()

def compute_c5_fft(h, n):
    dx = 2.0 / n
    N = next_pow2(2 * n - 1)
    H = fft(h, N)
    G = fft((1.0 - h)[::-1], N)
    corr = ifft(H * G).real[:2*n-1]
    return corr * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_fft(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_fft(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_fft(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(30, 5.0), (100, 50.0), (300, 200.0),
                   (1000, 1000.0), (3000, 5000.0), (10000, 20000.0),
                   (30000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_fft(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
                diff -= 1.0
                if abs(diff) < 1e-12:
                    break
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
                diff += 1.0
                if abs(diff) < 1e-12:
                    break
    return project_feasible(h_b, n)

def binary_local_search(h_bin, n, deadline, max_rounds=200):
    c_best = compute_c5(h_bin, n)
    h_best = h_bin.copy()
    rng = np.random.RandomState(777)
    for rnd in range(max_rounds):
        if time.time() > deadline - 30:
            break
        c_vec = compute_c5_fft(h_best, n)
        worst_idx = int(np.argmax(c_vec))
        # Get gradient of worst shift to determine best flip direction
        k_worst = worst_idx - (n - 1)
        # Compute which flips would reduce c_worst most
        if k_worst >= 0:
            # c_k = sum h[i]*(1-h[i+k]) for i=0..n-1-k
            # Flipping h[j] from 1->0: reduces c_k by (1-h[j+k]) if j < n-k
            #                        or by h[j-k] if j >= k (removes h[j] from second factor)
            # Flipping h[j] from 0->1: increases c_k by (1-h[j+k]) if j < n-k
            #                        or by -h[j-k] if j >= k (but h[j-k] might be 0)
            # We want to reduce max, so try flips that reduce c_worst
            # But also need to not increase other c's too much
            # Simple heuristic: try all flips, keep best
            pass
        # Random perturbation approach: flip random bits
        n_flips = max(1, n // 200)
        improved = False
        for _ in range(n_flips * 3):
            if time.time() > deadline - 30:
                break
            i = int(rng.randint(n))
            h_try = h_best.copy()
            h_try[i] = 1.0 - h_try[i]
            # Fix sum constraint
            target = n / 2.0
            diff = target - h_try.sum()
            if abs(diff) < 1e-10:
                pass
            else:
                # Find a bit to flip to fix
                if diff > 0:
                    candidates = np.where(h_try == 0)[0]
                else:
                    candidates = np.where(h_try == 1)[0]
                if len(candidates) == 0:
                    continue
                j = candidates[int(rng.randint(len(candidates)))]
                h_try[j] = 1.0 - h_try[j]
            c_try = compute_c5(h_try, n)
            if c_try < c_best - 1e-12:
                c_best = c_try
                h_best = h_try.copy()
                improved = True
    return h_best, c_best

def perturbation_search(h, n, deadline, n_trials=8, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    if mags is None:
        mags = [0.03, 0.05, 0.08, 0.02, 0.10, 0.04, 0.06, 0.12]
    for t in range(n_trials):
        if time.time() > deadline - 40:
            break
        mag = mags[t % len(mags)]

        h_p = project_feasible(h_p, n)

                          args=(n, 3000, 5000.0), method='L-BFGS-B',

                          options={'maxiter': 5000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
        h_bin = binary_push(h_p, n)
        try:
            res2 = minimize(smooth_obj, h_bin, jac=smooth_grad,
                           args=(n, 2000, 3000.0), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 3000, 'ftol': 1e-16, 'maxls': 50})
            h_new2 = project_feasible(res2.x, n)
            c_new2 = compute_c5(h_new2, n)

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_fft(h, n)
        worst_idx = int(np.argmax(c_vec))
        n_active = np.sum(c_vec > best_c - 1e-6)
        # Subgradient step: move in direction that reduces worst correlations
        grad_worst = np.zeros(n)
        center = n - 1
        dx = 2.0 / n
        active = np.where(c_vec > best_c - 1e-5)[0]
        for idx in active:
            k = idx - center
            wm = (c_vec[idx] - best_c + 1e-5) * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad_worst[0:nv] += wm * (1.0 - h[k:n])
                    grad_worst[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad_worst[kk:n] += wm * (1.0 - h[0:nv])
                    grad_worst[0:nv] += wm * (-h[kk:n])
        # Line search along -grad_worst
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.05]:
            h_try = h - step * grad_worst
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
        # Also try single coordinate adjustments at worst shift
        found = False
        k_worst = worst_idx - center
        if time.time() < deadline - 10:
            if k_worst >= 0:
                affected = list(range(0, min(n - k_worst, 50)))
            else:
                affected = list(range(-k_worst, min(n - k_worst, n - 50)))
            for i in affected:
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                    nv_val = h[i] + delta
                    if nv_val < -1e-12 or nv_val > 1.0 + 1e-12:
                        continue
                    h[i] = nv_val
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        found = True
                        break
                    else:
                        h[i] = best_h[i]
                if found:
                    h = best_h.copy()
                    break
        if not found and np.max(np.abs(grad_worst)) < 0.01:
            break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_binary_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def structured_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    # Try different structured patterns
    pattern_type = seed_val % 5
    if pattern_type == 0:
        # Sine wave thresholded
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.5 * np.sin(2*np.pi*freq*x + phase)
        h = project_feasible(h, n)
    elif pattern_type == 1:
        # Block pattern
        block_size = max(2, n // 20)
        n_blocks = n // block_size
        h = np.zeros(n)
        for b in range(n_blocks):
            val = rng.uniform(0, 1)
            h[b*block_size:(b+1)*block_size] = val
        h = project_feasible(h, n)
    elif pattern_type == 2:
        # Golden ratio quasi-periodic
        x = np.linspace(0, 2, n, endpoint=False)
        phi = (1 + 5**0.5) / 2
        h = (np.sin(2*np.pi*phi*x) > 0).astype(float)
        h = project_feasible(h, n)
    elif pattern_type == 3:
        # Two-scale pattern
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3*np.sin(2*np.pi*3*x) + 0.2*np.sin(2*np.pi*7*x + 1.0)
        h = project_feasible(h, n)
    else:
        # Smooth noise
        h = 0.5 + 0.3 * rng.randn(n)
        h = project_feasible(h, n)
    return h
