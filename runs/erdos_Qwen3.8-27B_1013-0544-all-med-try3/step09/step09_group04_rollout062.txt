    """
    Find direction d that minimizes max_m (g_m · d)
    subject to sum(d)=0, |d_i| ≤ max_step, -h_i ≤ d_i ≤ 1-h_i
    Returns (d, t) where t is the optimal value.
    """
    G = compute_all_grads(h, n)  # shape (2n-1, n)
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize t
    # s.t. G @ d - t <= 0  →  [G, -1] @ [d, t] <= 0
    #      sum(d) = 0
    #      -h_i <= d_i <= min(1-h_i, max_step)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0

        hi = min(1.0 - h[i], max_step)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free

    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(ncorr)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo = max(-h[i], -max_step)
        hi = min(1.0 - h[i], max_step)
        if lo >= hi:
            lo2 = min(lo, -1e-10)
            hi2 = max(hi, 1e-10)
            if lo2 > hi2:
                lo2, hi2 = 0.0, 0.0
            bounds.append((lo2, hi2))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def line_search(h, n, d, max_alpha=1.0):
    best_alpha = 0.0
    best_c = compute_c5(h, n)
    alphas = [max_alpha, max_alpha * 0.7, max_alpha * 0.5,
              max_alpha * 0.35, max_alpha * 0.2, max_alpha * 0.1]
    for a in alphas:
        h_new = h + a * d
        if np.any(h_new < -0.5) or np.any(h_new > 1.5):
            continue
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_alpha = a
    return best_alpha, best_c

def lp_optimize(h, n, deadline, max_step_init=0.05, min_step=0.0005):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    step = max_step_init
    stall = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 3:
            break
        d, t = lp_find_direction(best_h, n, max_step=step)
        if d is None or t >= -1e-10:
            stall += 1
            step *= 0.7
            if stall > 30 or step < min_step:
                break
            continue
        alpha, c_new = line_search(best_h, n, d, max_alpha=1.0)
        if alpha > 0 and c_new < best_c - 1e-16:
            best_h = project_feasible(best_h + alpha * d, n)
            best_c = c_new
            stall = 0
            step = min(step * 1.05, max_step_init)
        else:
            stall += 1
            step *= 0.8
            if stall > 40 or step < min_step:
                break
    return best_h, best_c

def active_set_lp(h, n, top_k=10):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    idx = np.argsort(-c_vec)
    active = idx[:top_k]
    inactive = idx[top_k:]
    G_active = np.array([compute_grad_row(h, n, m - (n - 1)) for m in active])
    G_inact = np.array([compute_grad_row(h, n, m - (n - 1)) for m in inactive])
    n_a = len(active)
    n_i = len(inactive)
    n_total = n_a + n_i
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n_total, n + 1))
    A_ub[:n_a, :n] = G_active
    A_ub[:n_a, -1] = -1.0
    A_ub[n_a:, :n] = G_inact
    A_ub[n_a:, -1] = -1.0
    b_ub = np.zeros(n_total)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    max_step = 0.03
    bounds = []
    for i in range(n):
        lo = max(-h[i], -max_step)
        hi = min(1.0 - h[i], max_step)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def active_lp_optimize(h, n, deadline, top_k=8, max_step=0.03):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    stall = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 3:
            break
        d, t = active_set_lp(best_h, n, top_k=top_k)
        if d is None or t >= -1e-8:
            stall += 1
            top_k = min(top_k + 2, 20)
            if stall > 25:
                break
            continue
        alpha, c_new = line_search(best_h, n, d, max_alpha=0.8)
        if alpha > 0 and c_new < best_c - 1e-16:
            best_h = project_feasible(best_h + alpha * d, n)
            best_c = c_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=3000):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += compute_grad_row(best_h, n, k)
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.02, 0.01)
        else:
            eta *= 0.96
    return best_h, best_c

def pair_search_fine(h, n, deadline, radius=25, deltas=None):
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001]
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 3):
                    if abs(i - j) < 5:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS at original resolution (~35s)
    p1_end = t0 + 35
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
                       alphas=[50, 500, 5000, 50000], mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000 and quick polish (~40s)
    n_target = 1000
    if time.time() < deadline - 500 and best_n < n_target:
        h_up = upsample(best_h, best_n, n_target)
        c_up = compute_c5(h_up, n_target)
        if c_up < best_c + 0.005:
            p2_end = min(time.time() + 40, deadline - 400)
            h2, c2 = run_lbfgs(h_up, n_target, p2_end,
                               alphas=[10, 50, 200, 1000, 5000], mu=0.005)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_target

    # Phase 3: Full LP direction optimization (~250s)
    p3_end = min(time.time() + 250, deadline - 300)
    if time.time() < p3_end:
        h3, c3 = lp_optimize(best_h, best_n, p3_end,
                             max_step_init=0.05, min_step=0.001)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: L-BFGS interlude (~40s)
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 40, deadline - 160)
        h4, c4 = run_lbfgs(best_h, best_n, p4_end,
                           alphas=[20000, 100000, 500000], mu=0.005)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active-set LP optimization (~150s)
    p5_end = min(time.time() + 150, deadline - 120)
    if time.time() < p5_end:
        h5, c5 = active_lp_optimize(best_h, best_n, p5_end,
                                    top_k=8, max_step=0.02)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Subgradient refinement (~50s)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 50, deadline - 30)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Small-step LP (~60s)
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 60, deadline - 15)
        h7, c7 = lp_optimize(best_h, best_n, p7_end,
                             max_step_init=0.01, min_step=0.0003)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search (~25s)
    if time.time() < deadline - 5:
        p8_end = min(time.time() + 25, deadline - 2)
        h8, c8 = pair_search_fine(best_h, best_n, p8_end,
                                  radius=20,
                                  deltas=[0.0005, 0.0002, 0.0001])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: quick L-BFGS
    if time.time() < deadline - 1:
        h9, c9 = run_lbfgs(best_h, best_n, deadline - 0.5,
                           alphas=[200000], mu=0.003)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
