        if not improved:
            break
    return best_h, best_c

def guided_swap_search(h, n, deadline, max_sweeps=200):
    best_h = project_feasible(h, n).copy()
    best_c = compute_c5(best_h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline - 3:
            break
        c = compute_c5_vec(best_h, n)
        max_val = c.max()
        threshold = max_val - max(1e-10, 0.005 * max_val)
        active = np.where(c >= threshold)[0]
        
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        improved = False
        
        for i in ones:
            if time.time() > deadline - 3:
                break
            for j in zeros:
                if time.time() > deadline - 3:
                    break
                best_h[i] = 0.0
                best_h[j] = 1.0
                c_new = compute_c5_vec(best_h, n)
                max_new = float(c_new.max())
                if max_new < best_c - 1e-14:
                    best_c = max_new
                    improved = True
                    ones = np.where(best_h > 0.5)[0]
                    zeros = np.where(best_h < 0.5)[0]
                    if i not in ones:
                        break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
        if not improved:
            break
    return best_h, best_c

def targeted_swap_search(h, n, deadline, max_rounds=100):
    best_h = project_feasible(h, n).copy()
    best_c = compute_c5(best_h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_c5_vec(best_h, n)
        max_val = c.max()
        tol = max(1e-9, 0.002 * max_val)
        active = np.where(c >= max_val - tol)[0]
        
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        for idx in active:
            if time.time() > deadline - 5:
                break
            k = idx - (n - 1)
            
            for i in ones:
                if time.time() > deadline - 5:
                    break
                if k >= 0:
                    if i + k >= n:
                        continue
                    if best_h[i + k] < 0.5:
                        for j in zeros:
                            if j == i:
                                continue
                            if k >= 0 and j + k < n and best_h[j + k] > 0.5:
                                best_h[i] = 0.0
                                best_h[j] = 1.0
                                c_new = compute_c5_vec(best_h, n)
                                max_new = float(c_new.max())
                                if max_new < best_c - 1e-14:
                                    best_c = max_new
                                    improved = True
                                    ones = np.where(best_h > 0.5)[0]
                                    zeros = np.where(best_h < 0.5)[0]
                                    if k >= 0 and i + k < n:
                                        break
                                else:
                                    best_h[i] = 1.0
                                    best_h[j] = 0.0
                            else:
                                best_h[i] = 1.0
                else:
                    kk = -k
                    if i - kk < 0:
                        continue
                    if best_h[i - kk] < 0.5:
                        for j in zeros:
                            if j == i:
                                continue
                            if j - kk >= 0 and best_h[j - kk] > 0.5:
                                best_h[i] = 0.0
                                best_h[j] = 1.0
                                c_new = compute_c5_vec(best_h, n)
                                max_new = float(c_new.max())
                                if max_new < best_c - 1e-14:
                                    best_c = max_new
                                    improved = True
                                    ones = np.where(best_h > 0.5)[0]
                                    zeros = np.where(best_h < 0.5)[0]
                                    if kk <= j:
                                        break
                                else:
                                    best_h[i] = 1.0
                                    best_h[j] = 0.0
                            else:
                                best_h[i] = 1.0
        if not improved:
            break
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(10, int(deadline - time.time()))
        maxit = min(maxiter, max(100, rem * 5))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-22})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def multi_start_optimize(h_starts, n, deadline, alphas_phase1, mu=0.01):
    best_h = None
    best_c = float('inf')
    for h_s in h_starts:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        h_opt, c_opt = run_lbfgs(h_s, n, min(deadline, time.time() + remaining * 0.8 / len(h_starts)),
                                  alphas_phase1, mu)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def continuous_coord_descent(h, n, deadline, max_rounds=50):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_c5_vec(best_h, n)
        max_val = c.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c >= max_val - tol)[0]
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx_val
                grad[k:] += (-best_h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx_val
                grad[:nv] += (-best_h[kk:]) * dx_val
        grad /= len(active)
        
        improved = False
        pos_mask = (grad > 1e-12) & (best_h > 1e-10)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-10)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        for step in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
            if time.time() > deadline - 5:
                break
            for i in pos_idx[:50]:
                if time.time() > deadline - 5:
                    break
                for j in neg_idx[:50]:
                    if time.time() > deadline - 5:
                        break
                    amt = min(step, best_h[i], 1.0 - best_h[j])
                    if amt < 1e-12:
                        continue
                    best_h[i] -= amt
                    best_h[j] += amt
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved = True
                        pos_idx = np.where((grad > 1e-12) & (best_h > 1e-10))[0]
                        neg_idx = np.where((grad < -1e-12) & (best_h < 1.0 - 1e-10))[0]
                        break
                    else:
                        best_h[i] += amt
                        best_h[j] -= amt
                        best_h = project_feasible(best_h, n)
                if time.time() > deadline - 5:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = None
        n = 500
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Generate alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = []
    alt_starts.append(make_beatty_start(n, golden / 2.0))
    alt_starts.append(make_beatty_start(n, np.sqrt(2.0) / 2.0))
    alt_starts.append(make_random_binary(n, rng))
    
    # Quick evaluation of alternatives
    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Phase 1: L-BFGS warmup (120s)
    p1_end = min(t0 + 120, deadline - 500)
    alphas_p1 = [10, 50, 200, 1000, 5000, 50000, 200000]
    h1, c1 = run_lbfgs(best_h, n, p1_end, alphas_p1, mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Exhaustive binary swap search (250s)
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 250, deadline - 300)
        # Round to binary first for swap search
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        
        h2, c2 = exhaustive_swap_search(h_bin, n, p2_end, max_sweeps=300)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        
        # Also try from the continuous solution rounded
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, n)
        h2b, c2b = exhaustive_swap_search(h_bin2, n, min(time.time() + 100, deadline - 300), max_sweeps=100)
        if c2b < best_c:
            best_c = c2b
            best_h = h2b
    
    # Phase 3: L-BFGS polish after swaps (100s)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 100, deadline - 180)
        alphas_p3 = [100000, 500000, 2000000, 5000000]
        h3, c3 = run_lbfgs(best_h, n, p3_end, alphas_p3, mu=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: More swap search (150s)
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 150, deadline - 120)
        h_bin4 = (best_h > 0.5).astype(float)
        h_bin4 = project_feasible(h_bin4, n)
        h4, c4 = exhaustive_swap_search(h_bin4, n, p4_end, max_sweeps=200)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        # Also try continuous (not rounded) for swap-based
        h4b, c4b = exhaustive_swap_search(best_h, n, min(time.time() + 60, deadline - 120), max_sweeps=50)
        if c4b < best_c:
            best_c = c4b
            best_h = h4b
    
    # Phase 5: Targeted swap + continuous descent (100s)
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 60, deadline - 80)
        h5, c5 = targeted_swap_search(best_h, n, p5_end, max_rounds=100)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 50:
            p5b_end = min(time.time() + 40, deadline - 40)
            h5b, c5b = continuous_coord_descent(best_h, n, p5b_end, max_rounds=30)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
    
    # Phase 6: SLSQP polish (60s)
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 55, deadline - 20)
        for alpha in [2000000, 10000000]:
            if time.time() > deadline - 25:
                break
            h6, c6 = run_slqp(best_h, n, p6_end, alpha=alpha, maxiter=500)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Final fine-tuning (remaining)
    if time.time() < deadline - 5:
        p7_end = min(time.time() + (deadline - 10 - time.time()), deadline - 3)
        # Tiny perturbations
        h7, c7 = continuous_coord_descent(best_h, n, p7_end, max_rounds=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # Final binary snap and one more swap pass
        h_bin7 = (best_h > 0.5).astype(float)
        h_bin7 = project_feasible(h_bin7, n)
        h7b, c7b = exhaustive_swap_search(h_bin7, n, min(time.time() + 30, deadline - 2), max_sweeps=50)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
