def de_mutation(pop, pop_idx, best_idx, F, strategy, rng):
    n = len(pop[0])
    if strategy == 'rand1':
        i, j, k = rng.choice(len(pop), 3, replace=False)
        return pop[i] + F * (pop[j] - pop[k])
    elif strategy == 'best1':
        j, k = rng.choice(len(pop), 2, replace=False)
        return pop[best_idx] + F * (pop[j] - pop[k])
    elif strategy == 'current_to_best1':
        j, k = rng.choice(len(pop), 2, replace=False)
        return pop[pop_idx] + F * (pop[best_idx] - pop[pop_idx]) + F * (pop[j] - pop[k])
    else:
        i, j, k = rng.choice(len(pop), 3, replace=False)
        return pop[i] + F * (pop[j] - pop[k])

def de_crossover(trial, current, CR, rng):
    n = len(current)
    j0 = rng.integers(0, n)
    bin_mask = rng.random(n) <= CR
    bin_mask[j0] = True
    child = current.copy()
    child[bin_mask] = trial[bin_mask]
    return child

def run_de(h_init, n, deadline, pop_size=50, max_gens=None):
    """Differential Evolution for the Erdős problem"""
    rng = np.random.RandomState(12345)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = best_h
    for i in range(1, pop_size):
        # Mix: some near-binary, some perturbed best, some random
        if i < pop_size // 4:
            # Near-binary perturbations of best
            pop[i] = best_h + rng.normal(0, 0.1, n)
        elif i < pop_size // 2:
            # Binary pattern
            binary = (rng.random(n) < 0.5).astype(float)
            pop[i] = binary + rng.normal(0, 0.05, n)
        else:
            # Interpolation between best and random
            t = rng.random()
            pop[i] = (1-t) * best_h + t * rng.random(n)
        pop[i] = project_feasible(pop[i], n)
    
    # Evaluate initial population
    fitness = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_c = fitness[best_idx]
    best_h = pop[best_idx].copy()
    
    F_list = [0.8, 0.6, 0.5, 0.4, 0.3]
    CR = 0.8
    strategies = ['rand1', 'best1', 'current_to_best1']
    
    no_improve = 0
    max_no_improve = 50
    
    gen = 0
    while time.time() < deadline - 30:
        gen += 1
        # Adaptive F
        F = F_list[min(gen // 50, len(F_list)-1)]
        
        for i in range(pop_size):
            if time.time() > deadline - 30:
                break
            
            # Mutation
            strategy = strategies[rng.randint(len(strategies))]
            trial = de_mutation(pop, i, best_idx, F, strategy, rng)
            
            # Crossover
            child = de_crossover(trial, pop[i], CR, rng)
            
            # Project
            child = project_feasible(child, n)
            
            # Evaluate
            child_c = compute_c5(child, n)
            
            # Selection
            if child_c < fitness[i]:
                pop[i] = child
                fitness[i] = child_c
                no_improve = 0
                if child_c < best_c - 1e-10:
                    best_c = child_c
                    best_h = child.copy()
                    best_idx = i
            else:
                no_improve += 1
        
        # elitism check
        actual_best = np.argmin(fitness)
        if fitness[actual_best] < best_c:
            best_c = fitness[actual_best]
            best_h = pop[actual_best].copy()
            best_idx = actual_best
        
        if no_improve > max_no_improve:
            # Restart some of the worst solutions
            worst_indices = np.argsort(-fitness)[-pop_size//5:]
            for wi in worst_indices:
                pop[wi] = best_h + rng.normal(0, 0.15, n)
                pop[wi] = project_feasible(pop[wi], n)
                fitness[wi] = compute_c5(pop[wi], n)
            no_improve = 0
    
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = max(200, maxiters[i])
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(2024)
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        freq = rng.randint(1, 10)
        phase = rng.random() * 2 * np.pi
        amp2 = rng.uniform(0.01, amp)
        trial = best_h + amp2 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        trial = project_feasible(trial, n)
        c = compute_c5(trial, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = trial
    return best_h, best_c

def single_dive(h, n, deadline, alphas_dive=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas_dive is None:
        alphas_dive = [2000000, 5000000, 20000000]
    for it in range(200):
        rem = deadline - time.time()
        if rem < 8:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        for a in alphas_dive:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, a, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-22,
                                        'gtol': 1e-16, 'maxfun': 1500,
                                        'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

t0_de_start = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_de_start
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < deadline - 350:
        p1_end = min(t0 + 45, deadline - 350)
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[100, 1000, 10000, 100000],
            mu=0.01, maxiters=[600, 1000, 1500, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=2000, L-BFGS polish
    n_hi = 2000
    if time.time() < deadline - 450:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        if c_hi < best_c + 0.01:
            p2_end = min(t0 + 130, deadline - 450)
            h2, c2 = run_lbfgs(h_hi, n_hi, p2_end,
                alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
                mu=0.005, maxiters=[400, 600, 800, 1200, 1500, 2000, 2500])
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_hi
            elif c_hi < best_c + 0.005:
                best_h = h_hi
                best_n = n_hi

    # Phase 3: Differential Evolution at best resolution
    if time.time() < deadline - 300 and best_n >= 500:
        p3_end = min(t0 + 500, deadline - 250)
        t0_de_start = time.time()
        h3, c3 = run_de(best_h, best_n, p3_end, pop_size=50, rng_seed=seed)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        # Quick L-BFGS on DE result
        if time.time() < deadline - 200:
            p3b_end = min(time.time() + 80, deadline - 180)
            h3b, c3b = run_lbfgs(best_h, best_n, p3b_end,
                alphas=[5000, 20000, 100000],
                mu=0.005, maxiters=[800, 1200, 2000])
            if c3b < best_c:
                best_c = c3b
                best_h = h3b

    # Phase 4: Active set optimization
    if time.time() < deadline - 200:
        p4_end = min(t0 + 700, deadline - 150)
        h4, c4 = run_active_opt(best_h, best_n, p4_end, top_k=5,
            alphas=[100000, 500000, 2000000, 10000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Frequency perturbation + flip search
    if time.time() < deadline - 150:
        p5_end = min(t0 + 780, deadline - 120)
        h5, c5 = freq_perturb(best_h, best_n, p5_end, n_tries=30, amp=0.04)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 80:
            p5b_end = min(time.time() + 40, deadline - 60)
            h5b, c5b = flip_search(best_h, best_n, p5b_end)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Second DE round (shorter, focused)
    if time.time() < deadline - 120:
        p6_end = min(t0 + 880, deadline - 80)
        h6, c6 = run_de(best_h, best_n, p6_end, pop_size=30, rng_seed=seed + 100)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try higher resolution n=3000
    n_very_hi = 3000
    if time.time() < deadline - 100 and best_n < n_very_hi:
        h_vh = upsample(best_h, best_n, n_very_hi)
        c_vh = compute_c5(h_vh, n_very_hi)
        p7_end = min(t0 + 980, deadline - 50)
        h7, c7 = run_lbfgs(h_vh, n_very_hi, p7_end,
            alphas=[10, 50, 200, 1000, 5000, 20000],
            mu=0.003, maxiters=[300, 500, 800, 1000, 1200, 1500])
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_very_hi

    # Phase 8: Single constraint dive
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 30, deadline - 10)
        h8, c8 = single_dive(best_h, best_n, p8_end)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final active set polish
    if time.time() < deadline - 10:
        h9, c9 = run_active_opt(best_h, best_n, deadline - 3, top_k=3,
            alphas=[500000, 5000000, 20000000])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
