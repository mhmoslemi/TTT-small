import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_c5(h, n, top_k=10):
    """Compute weighted subgradient of max_k O(k)"""
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    # Use top_k lags with softmax weighting
    top_idx = np.argsort(-c_vec)[:top_k]
    vals = c_vec[top_idx]
    # Softmax weights
    w = np.exp(-(vals - vals[0]) * 100)  # sharp softmax
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mi in zip(w, top_idx):
        k = mi - (n - 1)
        if k >= 0:
            nv = n - k
            # h[i]*(1-h[i+k]) for i in [0, nv)
            grad[0:nv] += wi * (1.0 - h[k:n]) * dx
            grad[k:n] += wi * (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * (1.0 - h[0:nv]) * dx
            grad[0:nv] += wi * (-h[kk:n]) * dx
    return grad, c_max

def fista_refine(h, n, deadline, n_iter=50000, lr=0.01, top_k=10):
    """FISTA (Nesterov accelerated) with subgradient"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Gradient at y
        grad_y, _ = subgradient_c5(y, n, top_k)
        
        # Update x
        x_new = project_feasible(y - lr * grad_y, n)
        
        # Update t
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = x_new + ((t_k - 1) / t_new) * (x_new - x)
        y = np.clip(y, 0, 1)
        x = x_new
        t_k = t_new
        
        # Check improvement
        if it % 100 == 0:
            c_now = compute_c5(x, n)
            if c_now < best_c:
                best_c = c_now
                best_h = x.copy()
    
    # Final check
    c_final = compute_c5(x, n)
    if c_final < best_c:
        best_c = c_final
        best_h = x.copy()
    
    return best_h, best_c

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def adjust_sum(h, n):
    """Adjust h to have exact sum n/2"""
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += (diff / tot) * room
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
