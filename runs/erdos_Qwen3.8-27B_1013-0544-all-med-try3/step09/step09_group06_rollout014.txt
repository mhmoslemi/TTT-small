import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
        else:
            idx = np.argmin(np.abs(x - 0.5))
            x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(h, n):
    return project_box_sum(h.copy(), n / 2.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[m:n] += (1.0 - h[0:n-m])
        g[0:n-m] += -h[m:n]
    else:
        g[0:n-m] += (1.0 - h[m:n])
        g[m:n] += -h[0:n-m]
    return g

def smax_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 50:
        return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    with np.errstate(over='ignore'):
        return m + np.log(np.sum(np.exp(s))) / alpha

def smax_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 50:
        w = np.exp(s - sm)
    else:
        with np.errstate(over='ignore'):
            w = np.exp(s)
    s2 = w.sum()
    if s2 < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return w / s2

def grad_full(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smax_w(c, alpha)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            g += w[k] * lag_grad(h, n, k - (n - 1))
    return g * dx

def grad_momentum(h0, n, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [100, 300, 1000, 5000]
    total_time = max(1.0, deadline - time.time())
    for ai, alpha in enumerate(alphas):
        frac = [0.25, 0.25, 0.25, 0.25][ai]
        ph_dl = time.time() + frac * total_time
        v = np.zeros(n)
        lr = 1.5
        for it in range(100000):

            g = grad_full(h, n, alpha)

            h = project_feasible(h - v, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def slsqp_top(h0, n, top_m, alpha, maxiter):
    c = compute_all_lags(h0, n)
    idx = np.argsort(c)[-top_m:]

    def f(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return smax_val(cc[idx], alpha)

    def jf(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        w = smax_w(cc[idx], alpha)
        dx = 2.0 / n
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] > 1e-12:
                g += w[ii] * lag_grad(x, n, k - (n - 1))
        return g * dx

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(f, h0, jac=jf, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def golden_init(n, seed):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    idx = (np.arange(n) * phi * (seed + 1)) % n
    h = np.zeros(n)
    h[np.argsort(idx)[:n//2]] = 1.0
    return h

def modular_init(n, period, seed):
    h = np.zeros(n)
    offset = seed % period
    for i in range(n):
        if ((i + offset) % period) < period // 2:
            h[i] = 1.0
    # Adjust to have exactly n//2 ones
    ones = np.sum(h)
    if ones < n // 2:
        zeros = np.where(h == 0)[0]
        h[zeros[:n//2 - int(ones)]] = 1.0
    elif ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        h[ones_idx[int(ones) - n//2:]] = 0.0
    return h
