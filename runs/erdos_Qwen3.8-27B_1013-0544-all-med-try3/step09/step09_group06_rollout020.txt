def _logsumexp_obj(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _logsumexp_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def _lag_grad(h, n, L):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += (1.0 - h[L:n])
        g[L:] += -h[:n-L]
    else:
        m = -L
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] += -h[m:]
    return g

def grad_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = _logsumexp_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        L = k - (n - 1)
        grad += w[k] * _lag_grad(h, n, L)
    return grad * dx

def grad_descent(h, n, alpha, deadline, lr=3.0, momentum=0.93):
    h_c = h.copy()
    best_h = h_c.copy()
    best_c = compute_c5(h_c, n)
    v = np.zeros(n)
    patience = 0
    for _ in range(1000000):
        if time.time() > deadline:
            break
        g = grad_smooth(h_c, n, alpha)
        if np.abs(g).max() < 1e-14:
            break
        v = momentum * v + lr * g
        h_new = project_feasible(h_c - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_c = h_new
        if patience > 3000:
            lr *= 0.6
            v *= 0.3
            patience = 0
        if lr < 0.01:
            break
    return best_h, best_c

def fast_local_search(h, n, deadline, max_iter=200000):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    lags = np.arange(2 * n - 1) - (n - 1)
    no_improve = 0
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005]

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c_all))
        L_star = lags[k_star]

        if L_star == 0:
            idx = np.argmin(np.abs(h - 0.5))
            if h[idx] > 0.55 or h[idx] < 0.45:
                no_improve += 1
                if no_improve > 500:
                    break
                continue
            delta = 0.02 if h[idx] > 0.5 else -0.02
            h[idx] += delta
            h = project_feasible(h, n)
            c_all = np.correlate(h, 1 - h, mode='full') * dx
            c_new = c_all.max()
            if c_new < best_c - 1e-14:
                best_c = c_new
                no_improve = 0
            else:
                h[idx] -= delta
                h = project_feasible(h, n)
                c_all = np.correlate(h, 1 - h, mode='full') * dx
                no_improve += 1
            continue

        m = abs(L_star)
        moved = False
        for delta_base in step_sizes:
            if time.time() > deadline:
                break
            best_dc_max = 0.0
            best_i, best_j, best_d = -1, -1, 0.0

            if L_star > 0:
                for i in range(0, n - m, 3):
                    j = i + m
                    if h[i] > h[j] + 1e-4:
                        d = min(delta_base, (h[i] - h[j]) * 0.2)
                        # Compute Δc for all lags
                        dc = np.zeros(2 * n - 1)
                        for k2 in range(2 * n - 1):
                            Lk = lags[k2]
                            val = 0.0
                            ii = i
                            jj = j
                            # Position i changes by -d
                            if 0 <= ii + Lk < n:
                                val += -d * (1.0 - h[ii + Lk])
                            if 0 <= ii - Lk < n:
                                val += d * h[ii - Lk]
                            # Position j changes by +d
                            if 0 <= jj + Lk < n:
                                val += d * (1.0 - h[jj + Lk])
                            if 0 <= jj - Lk < n:
                                val += -d * h[jj - Lk]
                            dc[k2] = val
                        new_max = (c_all + dc).max()
                        if new_max < best_dc_max or (best_i == -1 and new_max < c_all.max()):
                            best_dc_max = new_max
                            best_i, best_j, best_d = i, j, -d
                    elif h[j] > h[i] + 1e-4:
                        d = min(delta_base, (h[j] - h[i]) * 0.2)
                        dc = np.zeros(2 * n - 1)
                        for k2 in range(2 * n - 1):
                            Lk = lags[k2]
                            val = 0.0
                            if 0 <= i + Lk < n:
                                val += d * (1.0 - h[i + Lk])
                            if 0 <= i - Lk < n:
                                val += -d * h[i - Lk]
                            if 0 <= j + Lk < n:
                                val += -d * (1.0 - h[j + Lk])
                            if 0 <= j - Lk < n:
                                val += d * h[j - Lk]
                            dc[k2] = val
                        new_max = (c_all + dc).max()
                        if new_max < best_dc_max or (best_i == -1 and new_max < c_all.max()):
                            best_dc_max = new_max
                            best_i, best_j, best_d = j, i, -d
            else:
                for i in range(0, n - m, 3):
                    j = i + m
                    if h[i] > h[j] + 1e-4:
                        d = min(delta_base, (h[i] - h[j]) * 0.2)
                        dc = np.zeros(2 * n - 1)
                        for k2 in range(2 * n - 1):
                            Lk = lags[k2]
                            val = 0.0
                            if 0 <= i - Lk < n:
                                val += -d * (1.0 - h[i - Lk])
                            if 0 <= i + Lk < n:
                                val += d * h[i + Lk]
                            if 0 <= j - Lk < n:
                                val += d * (1.0 - h[j - Lk])
                            if 0 <= j + Lk < n:
                                val += -d * h[j + Lk]
                            dc[k2] = val
                        new_max = (c_all + dc).max()
                        if new_max < best_dc_max or (best_i == -1 and new_max < c_all.max()):
                            best_dc_max = new_max
                            best_i, best_j, best_d = i, j, -d
                    elif h[j] > h[i] + 1e-4:
                        d = min(delta_base, (h[j] - h[i]) * 0.2)
                        dc = np.zeros(2 * n - 1)
                        for k2 in range(2 * n - 1):
                            Lk = lags[k2]
                            val = 0.0
                            if 0 <= i - Lk < n:
                                val += d * (1.0 - h[i - Lk])
                            if 0 <= i + Lk < n:
                                val += -d * h[i + Lk]
                            if 0 <= j - Lk < n:
                                val += -d * (1.0 - h[j - Lk])
                            if 0 <= j + Lk < n:
                                val += d * h[j + Lk]
                            dc[k2] = val
                        new_max = (c_all + dc).max()
                        if new_max < best_dc_max or (best_i == -1 and new_max < c_all.max()):
                            best_dc_max = new_max
                            best_i, best_j, best_d = j, i, -d

            if best_i >= 0:
                if best_d < 0:
                    h[best_i] += abs(best_d)
                    h[best_j] -= abs(best_d)
                else:
                    h[best_i] -= abs(best_d)
                    h[best_j] += abs(best_d)
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
                c_all = np.correlate(h, 1 - h, mode='full') * dx
                c_new = c_all.max()
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    no_improve = 0
                    moved = True
                    break
                else:
                    no_improve += 50
            else:
                no_improve += 1

        if not moved:
            no_improve += 10
        if no_improve > 5000:
            break

    return h, best_c

def targeted_local_search(h, n, deadline, max_iter=100000):
    """Faster local search: only evaluate top few lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    lags = np.arange(2 * n - 1) - (n - 1)
    no_improve = 0

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        # Get top 5 lags
        top_idx = np.argsort(c_all)[-5:]
        L_star = lags[top_idx[-1]]
        m = abs(L_star)
        if m >= n:
            break

        best_improvement = 0.0
        best_move = None
        delta = 0.004

        # Try moves at the worst lag
        if L_star > 0:
            for i in range(0, n - m, 2):
                if time.time() > deadline:
                    break
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 0.01:
                    continue
                d = delta if diff > 0 else -delta
                if d > 0:
                    if h[i] - d < 0 or h[j] + d > 1:
                        continue
                else:
                    if h[i] + d < 0 or h[j] - d > 1:
                        continue
                # Compute change in top lags only
                improved = True
                dc_max = 0.0
                for k2 in top_idx:
                    Lk = lags[k2]
                    val = 0.0
                    if d > 0:
                        # Move from i to j: h[i]-=d, h[j]+=d
                        if 0 <= i + Lk < n:
                            val += -d * (1.0 - h[i + Lk])
                        if 0 <= i - Lk < n:
                            val += d * h[i - Lk]
                        if 0 <= j + Lk < n:
                            val += d * (1.0 - h[j + Lk])
                        if 0 <= j - Lk < n:
                            val += -d * h[j - Lk]
                    else:
                        # Move from j to i: h[j]-=d, h[i]+=d (|d|)
                        ad = abs(d)
                        if 0 <= j + Lk < n:
                            val += -ad * (1.0 - h[j + Lk])
                        if 0 <= j - Lk < n:
                            val += ad * h[j - Lk]
                        if 0 <= i + Lk < n:
                            val += ad * (1.0 - h[i + Lk])
                        if 0 <= i - Lk < n:
                            val += -ad * h[i - Lk]
                    dc_max = max(dc_max, c_all[k2] + val)
                if dc_max < c_all[top_idx[-1]] - 1e-10:
                    imp = c_all[top_idx[-1]] - dc_max
                    if imp > best_improvement:
                        best_improvement = imp
                        best_move = (i, j, d)
        else:
            for i in range(m, n, 2):
                if time.time() > deadline:
                    break
                j = i + m
                diff = h[i] - h[j]
                if abs(diff) < 0.01:
                    continue
                d = delta if diff > 0 else -delta
                if d > 0:
                    if h[i] - d < 0 or h[j] + d > 1:
                        continue
                else:
                    if h[i] + d < 0 or h[j] - d > 1:
                        continue
                dc_max = 0.0
                for k2 in top_idx:
                    Lk = lags[k2]
                    val = 0.0
                    if d > 0:
                        if 0 <= i + Lk < n:
                            val += -d * (1.0 - h[i + Lk])
                        if 0 <= i - Lk < n:
                            val += d * h[i - Lk]
                        if 0 <= j + Lk < n:
                            val += d * (1.0 - h[j + Lk])
                        if 0 <= j - Lk < n:
                            val += -d * h[j - Lk]
                    else:
                        ad = abs(d)
                        if 0 <= j + Lk < n:
                            val += -ad * (1.0 - h[j + Lk])
                        if 0 <= j - Lk < n:
                            val += ad * h[j - Lk]
                        if 0 <= i + Lk < n:
                            val += ad * (1.0 - h[i + Lk])
                        if 0 <= i - Lk < n:
                            val += -ad * h[i - Lk]
                    dc_max = max(dc_max, c_all[k2] + val)
                if dc_max < c_all[top_idx[-1]] - 1e-10:
                    imp = c_all[top_idx[-1]] - dc_max
                    if imp > best_improvement:
                        best_improvement = imp
                        best_move = (i, j, d)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            c_all = np.correlate(h, 1 - h, mode='full') * dx
            c_new = c_all.max()
            if c_new < best_c - 1e-15:
                best_c = c_new
                no_improve = 0
            else:
                no_improve += 10
        else:
            no_improve += 5

        if no_improve > 2000:
            break

    return h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def _obj_full(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return float(_logsumexp_obj(c, alpha))

def _obj_top(x, n, top_idx, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    active = c[top_idx]
    return float(_logsumexp_obj(active, alpha))

def slsqp_polish(h, n, alpha=200000, maxiter=5000, top_m=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(_obj_top, h.copy(), args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        else:
            res = minimize(_obj_full, h.copy(), args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_push(h, n, threshold=0.02):
    """Push values closer to 0 or 1."""
    h_new = h.copy()
    for i in range(n):
        if h_new[i] < threshold:
            h_new[i] = 0.0
        elif h_new[i] > 1.0 - threshold:
            h_new[i] = 1.0
    h_new = project_feasible(h_new, n)
    return h_new

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Gradient descent with increasing alpha
    phases = [(200, 0.20), (500, 0.18), (2000, 0.15), (10000, 0.15), (50000, 0.12)]
    phase_time = deadline - time.time()
    for alpha, frac in phases:
        if time.time() > deadline - 20:
            break
        pd = time.time() + frac * phase_time
        h_new, c_new = grad_descent(h, n, alpha, pd, lr=3.0, momentum=0.93)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new

    # Phase 2: SLSQP with top lags
    if time.time() < deadline - 40:
        for tm in [3, 5, 8]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_polish(h, n, alpha=100000, maxiter=2000, top_m=tm)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: Targeted local search
    if time.time() < deadline - 15:
        ls_dl = min(deadline - 3, time.time() + (deadline - time.time()) * 0.35)
        h_new, c_new = targeted_local_search(best_h, n, ls_dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Full local search (slower but thorough)
    if time.time() < deadline - 10:
        fls_dl = min(deadline - 3, time.time() + (deadline - time.time()) * 0.3)
        h_new, c_new = fast_local_search(best_h, n, fls_dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Final SLSQP
    if time.time() < deadline - 5:
        h_new = slsqp_polish(best_h, n, alpha=500000, maxiter=3000, top_m=5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Main optimization from parent
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_pipeline(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        idx = rng.choice(n_init, n_init // 2, replace=False)
        h_start[idx] = 1.0
        alloc = min(200, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_pipeline(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Try binary push
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        h_bin = binary_push(best_h, n, threshold=0.03)
        c_bin = compute_c5(h_bin, n)
        h_bin2 = binary_push(best_h, n, threshold=0.05)
        c_bin2 = compute_c5(h_bin2, n)
        # Optimize from binary starting points
        for h_try, c_try in [(h_bin, c_bin), (h_bin2, c_bin2)]:
            if time.time() > deadline - 60:
                break
            alloc = min(120, (deadline - time.time()) * 0.2)
            if alloc < 40:
                break
            h_opt, c_opt = optimize_pipeline(h_try, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Try different n values
    n_best = len(best_h)
    n_candidates = [n_best - 20, n_best + 20, n_best - 40, n_best + 40]
    n_candidates = [x for x in n_candidates if x >= 200 and x <= 800 and x != n_best]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.2)
        if alloc < 40:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_pipeline(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_best = n_try

    # Multistart perturbations at best resolution
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_cur)
            hp = project_feasible(hp, n_cur)
            alloc = min(60, (deadline - time.time() - 30) * 0.3)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_pipeline(hp, n_cur, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final aggressive polish
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        # SLSQP with very high alpha
        if time.time() < deadline - 12:
            h_new = slsqp_polish(best_h, n, alpha=1000000, maxiter=5000, top_m=5)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Targeted local search
        if time.time() < deadline - 3:
            best_h, best_c = targeted_local_search(best_h, n, deadline - 1)
        # Final project and evaluate
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
