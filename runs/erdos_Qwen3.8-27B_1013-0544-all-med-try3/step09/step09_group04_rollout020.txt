import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

        val += mu * (x.sum() - n / 2.0) ** 2

def pair_search(h, n, deadline, radius=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if time.time() > deadline:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)

        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def de_run(h_init, n, deadline, pop_size=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    pop = np.zeros((pop_size, n))
    f_vals = np.empty(pop_size)
    pop[0] = best_h.copy()
    f_vals[0] = best_c
    for i in range(1, pop_size):
        pop[i] = project_feasible(best_h + 0.4 * rng.standard_normal(n), n)
        f_vals[i] = compute_c5(pop[i], n)
    order = np.argsort(f_vals)
    pop = pop[order]
    f_vals = f_vals[order]
    gen = 0
    stale = 0
    while time.time() < deadline:
        gen += 1
        F = 0.6 + 0.3 * min(1.0, gen / 300.0)
        CR = 0.8
        for i in range(pop_size):
            if time.time() > deadline:
                break
            if f_vals[i] < best_c:
                best_c = f_vals[i]
                best_h = pop[i].copy()
                stale = 0
            else:
                stale += 1
            tries = 0
            while True:
                r1, r2, r3 = list(rng.choice(pop_size, 3, replace=False))
                tries += 1
                if r1 != i and r2 != i and r3 != i if False else True:
                    pass
                if r1 != i and r2 != i and r3 != i:
                    break
                if tries > 20:
                    r1, r2, r3 = 0, 1, 2
                    if i in (r1, r2, r3):
                        r1, r2, r3 = (r1 + 1) % pop_size, (r2 + 1) % pop_size, (r3 + 1) % pop_size
                    break
            v = pop[r1] + F * (pop[r2] - pop[r3])
            j_rand = rng.randint(n)
            mask = rng.random(n) < CR
            mask[j_rand] = True
            u = np.where(mask, v, pop[i])
            u_p = project_feasible(u, n)
            f_u = compute_c5(u_p, n)
            if f_u <= f_vals[i]:
                pop[i] = u_p
                f_vals[i] = f_u
        order = np.argsort(f_vals)
        pop = pop[order]
        f_vals = f_vals[order]
        if f_vals[0] < best_c:
            best_c = f_vals[0]
            best_h = pop[0].copy()
            stale = 0
        else:
            stale += pop_size
        if stale > pop_size * 10:
            for i in range(pop_size // 3):
                idx = rng.randint(pop_size // 2, pop_size)
                pop[idx] = project_feasible(best_h + 0.3 * rng.standard_normal(n), n)
                f_vals[idx] = compute_c5(pop[idx], n)
            stale = 0
    return best_h, best_c

def gradient_flow(h, n, deadline, n_steps=200000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.005
    alpha = 10000.0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        h
        grad = sm_grad(h, n, alpha, 0.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = h - (eta / max(gnorm, 1e-10)) * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            eta = min(eta * 1.005, 0.02)
        else:
            eta *= 0.95
            if eta < 1e-9:
                break
            h = h_new
    return best_h, best_c

def threshold_restart(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [500, 5000, 50000, 500000, 5000000]
    h_bin = project_feasible(np.where(h > 0.5, 1.0, 0.0), n)
    c_bin = compute_c5(h_bin, n)
    h_opt, c_opt = lbfgs_opt(h_bin, n, deadline, alphas, mu=0.01,
                              maxiters=[500, 800, 1000, 1200, 1500])
    return h_opt, c_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = t0 + 35
    h1, c1 = lbfgs_opt(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 500000],
        mu=0.01, maxiters=[800, 1200, 1500, 2000, 2500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: DE at moderate resolution for global exploration
    de_n = 500
    if time.time() < deadline - 500:
        h_de = upsample(best_h, best_n, de_n) if best_n != de_n else best_h.copy()
        de_end = min(time.time() + 180, deadline - 450)
        de_h, de_c = de_run(h_de, de_n, de_end, pop_size=60, rng=rng)
        if de_c < best_c:
            best_c = de_c
            best_h = de_h
            best_n = de_n

    # Phase 3: L-BFGS at working resolution
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 100, deadline - 300)
        h3, c3 = lbfgs_opt(best_h, best_n, p3_end,
            alphas=[50, 500, 5000, 50000, 200000, 1000000],
            mu=0.005, maxiters=[500, 800, 1200, 1500, 2000, 2500])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to higher resolution
    for target_n in [800, 1200]:
        if time.time() > deadline - 250:
            break
        if target_n <= best_n:
            continue
        h_up = upsample(best_h, best_n, target_n)
        c_up = compute_c5(h_up, target_n)
        p4_end = min(time.time() + 80, deadline - 200)
        h4, c4 = lbfgs_opt(h_up, target_n, p4_end,
            alphas=[20, 200, 2000, 20000, 100000, 500000],
            mu=0.003, maxiters=[500, 700, 1000, 1500, 2000, 2500])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = target_n

    # Phase 5: Active set optimization
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 120, deadline - 150)
        h5, c5 = active_opt(best_h, best_n, p5_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Single constraint dive
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 90)
        h6, c6 = single_dive(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Gradient flow
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 40, deadline - 70)
        h7, c7 = gradient_flow(best_h, best_n, p7_end, n_steps=200000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Binary restart
    if time.time() < deadline - 80:
        p8_end = min(time.time() + 40, deadline - 50)
        h8, c8 = threshold_restart(best_h, best_n, p8_end)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Pair search
    if time.time() < deadline - 80:
        p9_end = min(time.time() + 60, deadline - 40)
        rad = max(20, min(50, best_n // 15))
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=rad)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Subgradient refinement
    if time.time() < deadline - 40:
        p10_end = min(time.time() + 30, deadline - 15)
        h10, c10 = subgrad_refine(best_h, best_n, p10_end, n_iter=50000)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 11: Final active set + single dive
    if time.time() < deadline - 20:
        p11_end = deadline - 3
        h11, c11 = active_opt(best_h, best_n, p11_end, top_k=4,
            alphas=[500000, 5000000, 50000000])
        if c11 < best_c:
            best_c = c11
            best_h = h11
        if time.time() < deadline - 2:
            h11b, c11b = single_dive(best_h, best_n, deadline - 1)
            if c11b < best_c:
                best_c = c11b
                best_h = h11b

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
