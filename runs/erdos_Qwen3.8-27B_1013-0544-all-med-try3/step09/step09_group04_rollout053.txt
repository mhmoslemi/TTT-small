def compute_c5_vec(h, n):

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()

    mx = sh.max()
    w = np.exp(sh - mx)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def de_step(pop, pop_fit, n, F, CR, rng):
    N = len(pop)
    assert N >= 4
    trial_v = pop[0].copy()[:N]  # dummy
    for i in range(N):
        a, b, c_id = rng.choice(N, 3, replace=False)
        x_trial = pop[a] * F + pop[b] * (1 - F) - pop[c_id] * 0  # DE/rand/1
        x_trial = pop[a] + F * (pop[b] - pop[c_id])
        j_rand = rng.integers(0, N)
        mask = rng.random(N) < CR
        mask[j_rand] = True
        x_trial = np.where(mask, x_trial, pop[i])
        x_trial = project_feasible(x_trial, n)
        fit_trial = compute_c5(x_trial, n)
        if fit_trial < pop_fit[i]:
            pop[i] = x_trial
            pop_fit[i] = fit_trial
        elif fit_trial < pop_fit.min():
            pop[rng.integers(0, N)] = x_trial
            pop_fit[rng.integers(0, N)] = fit_trial
    return pop, pop_fit

        pop_fit[i] = compute_c5(pop[i], n)
    
    best_fit = pop_fit.min()
    best_idx = pop_fit.argmin()
    stagnant = 0

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0, 2, n_old + 1)
    x_old = x_old[:-1] + (2.0 / n_old) / 2
    x_new = np.linspace(0, 2, n_new + 1)
    x_new = x_new[:-1] + (2.0 / n_new) / 2
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init(n, seed=0):

    inits = []

    # Block structure
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blk = rng.integers(2, 8)
        h[pos:pos+blk] = 1.0 if (pos // blk) % 2 == 0 else 0.0
        pos += blk
    inits.append(project_feasible(h, n))
    # Alternating with random gaps
    h = np.zeros(n)
    h[0::2] = 1.0
    swaps = rng.randint(0, n, n//4)
    for s in swaps:
        h[s] = 1.0 - h[s]
    inits.append(project_feasible(h, n))
    # Smooth bump
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(4 * np.pi * x + rng.random()) * np.cos(2 * np.pi * x)
    inits.append(project_feasible(np.clip(h, 0, 1), n))
    return inits

    p6_end = min(p6_start + 180, deadline - 80)
    h_high = upsample(best_h, best_n, n_high)
    c_high = compute_c5(h_high, n_high)
    if c_high < best_c + 0.005:
        h6, c6 = run_lbfgs(h_high, n_high, p6_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.003, maxiters=[500, 700, 900, 1100, 1300, 1500, 1800])
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n_high
    
    # Phase 7: Active set at final resolution (60s)
    p7_end = min(time.time() + 60, deadline - 50)
    h7, c7 = run_active_opt(best_h, best_n, p7_end, top_k=4,
        alphas=[200000, 1000000, 5000000, 20000000])
    if c7 < best_c:
        best_c = c7
        best_h = h7
    
    # Phase 8: Subgradient refinement (30s)
    p8_end = min(time.time() + 30, deadline - 25)
    h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=10000)
    if c8 < best_c:
        best_c = c8
        best_h = h8
    
    # Phase 9: Pair search final polish (20s)
    p9_end = deadline - 3
    h9, c9 = pair_search(best_h, best_n, p9_end, radius=25)
    if c9 < best_c:
        best_c = c9
        best_h = h9
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
