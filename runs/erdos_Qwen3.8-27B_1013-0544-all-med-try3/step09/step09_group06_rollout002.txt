from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_exact(h, n):
    """Exact projection onto {h: 0<=h<=1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    h = h.astype(float).copy()
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for tiny numerical errors
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # Distribute to non-bound elements
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def logsumexp_shifted(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return m + (mx + np.log(w.sum())) / alpha, w / w.sum()

def smooth_max(c, alpha):
    return logsumexp_shifted(c, alpha)[0]

def smooth_max_weights(c, alpha):
    return logsumexp_shifted(c, alpha)[1]

            #                -h[j-lag] for j in [lag, n-1]

            grad[m:n] += -wkg * h[0:n-m]
        else:

def nesterov_subgrad(h, n, deadline, alpha_start=50, alpha_end=5000, 
                      lr_start=0.5, lr_end=0.01, top_k=None):
    """Nesterov accelerated subgradient method."""
    h_curr = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    
    n_iter_est = 5000
    for it in range(n_iter_est):
        if time.time() > deadline:
            break
        progress = it / n_iter_est
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        g = grad_smooth_max(y, n, alpha, top_k)
        y_new = project_exact(y - lr * g, n)
        # Nesterov momentum
        t_k = (1 + np.sqrt(1 + 4 * it * it)) / 2
        beta = (t_k - 1) / t_k
        h_curr = project_exact(y_new + beta * (y_new - h_curr), n)
        y = y_new
        
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, top_m, maxiter=2000):

    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max(active, alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        w = smooth_max_weights(cc[top_idx], alpha)

        grad = np.zeros(n)

                grad[m:n] += wg * (1.0 - x[0:n-m])
                grad[0:n-m] += -wg * x[m:n]
        return grad * dx
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)

So 500*500 = 250000 operations per lag, per step size. With 5 step sizes, that's 1.25M operations per round. For 50 rounds, that's 62.5M operations. Each operation involves a correlation which is O(n) = 500. So total is about 31G operations. At ~1G ops/s, that's 31 seconds per round of all 5 step sizes. Hmm, that might be tight.

Actually np.correlate for n=500 is very fast (C implementation), probably < 100μs. So 500 * 100μs = 50ms per lag per step size. 5 step sizes * 50ms = 250ms per round. 500 rounds = 125s. That's feasible.

But I can optimize: instead of recomputing the full correlation, I can update just the affected lag values. When I change h[i] by -d and h[j] by +d (where j = i+m), only lags m and -m are affected (plus lag 0 if we're changing h[i]^2 terms). Actually, for the smooth max, all lags matter, but the change is small.

For a very fine local search, I'll just recompute the full correlation each time since np.correlate is fast enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_exact(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-14:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            proj[free] += diff / nf
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _lse(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    wsum = w.sum()
    return m + (mx + np.log(wsum)) / alpha, w / wsum

def grad_lse(h, n, alpha, top_k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None:
        idx = np.argsort(c)[-top_k:]
        mask = np.zeros(len(c), dtype=bool)
        mask[idx] = True
    else:
        mask = np.ones(len(c), dtype=bool)
    val, w = _lse(c, alpha)
    w = w * mask
    ws = w.sum()
    if ws < 1e-30:
        return val, np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[0:n-m] += wk * (1.0 - h[m:n])
            grad[m:n] += -wk * h[0:n-m]
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] += -wk * h[m:n]
    return val, grad * dx

def nesterov(h, n, deadline, a0=50, a1=500, lr0=1.0, lr1=0.02, tk=None):
    h_y = h.copy()
    h_x = h.copy()
    best = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    while time.time() < deadline:
        it += 1
        p = min(it / 2000.0, 1.0)
        alpha = a0 * (a1 / a0) ** p
        lr = lr0 * (lr1 / lr0) ** p
        _, g = grad_lse(h_y, n, alpha, tk)
        h_y = project_exact(h_y - lr * g, n)
        t = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = (t - 1.0) / t
        h_x = project_exact(h_y + beta * (h_y - h_x), n)
        c = compute_c5(h_x, n)
        if c < best_c:
            best_c = c
            best = h_x.copy()
    return best, best_c

def slsqp(h, n, alpha, tm, mi=2000):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    ti = np.argsort(c)[-tm:]
    
    def f(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        return _lse(cc[ti], alpha)[0]
    
    def fgrad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        _, w = _lse(cc[ti], alpha)
        g = np.zeros(n)
        for ii, k in enumerate(ti):
            if w[ii] < 1e-12:
                continue
            lag = k - (n - 1)
            wk = w[ii]
            if lag == 0:
                g += wk * (1.0 - 2.0 * x)
            elif lag > 0:
                m = lag
                g[0:n-m] += wk * (1.0 - x[m:n])
                g[m:n] += -wk * x[0:n-m]
            else:
                m = -lag
                g[m:n] += wk * (1.0 - x[0:n-m])
                g[0:n-m] += -wk * x[m:n]
        return g * dx
    
    def sc(x):
        return np.sum(x) - n / 2.0
    
    def scj(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(f, h, jac=fgrad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': mi, 'ftol': 1e-18})
        return project_exact(res.x, n)
    except:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    bc = compute_c5(h, n)
    for _ in range(2000):
        if time.time() > deadline:
            break
        ca = compute_all_lags(h, n)
        ks = int(np.argmax(ca))
        lag = ks - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        imp = False
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    dt = min(d, (h[i] - h[j]) * 0.4)
                else:
                    dt = -min(d, (h[j] - h[i]) * 0.4)
                if abs(dt) < 1e-10:
                    continue
                h[i] -= dt
                h[j] += dt
                h = project_exact(h, n)
                cn = compute_c5(h, n)
                if cn < bc - 1e-15:
                    bc = cn
                    imp = True
                    ca = compute_all_lags(h, n)
                else:
                    h[i] += dt
                    h[j] -= dt
                    h = project_exact(h, n)
            if imp:
                break
        if not imp:
            # Try other lags
            imp2 = False
            for k2 in np.argsort(ca)[-5:]:
                if time.time() > deadline:
                    break
                lag2 = k2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for i in range(n - m2):
                    if time.time() > deadline:
                        break
                    j2 = i + m2
                    if h[i] > h[j2]:
                        dt = min(0.005, (h[i] - h[j2]) * 0.3)
                    else:
                        dt = -min(0.005, (h[j2] - h[i]) * 0.3)
                    if abs(dt) < 1e-10:
                        continue
                    h[i] -= dt
                    h[j2] += dt
                    h = project_exact(h, n)
                    cn = compute_c5(h, n)
                    if cn < bc - 1e-15:
                        bc = cn
                        imp2 = True
                    else:
                        h[i] += dt
                        h[j2] -= dt
                        h = project_exact(h, n)
                if imp2:
                    break
            if not imp2:
                break
    return h, bc

def adaptive_local(h, n, deadline, rng):
    """Local search that adapts step size and explores multiple lags."""
    h = h.copy()
    bc = compute_c5(h, n)
    step = 0.02
    for it in range(50000):
        if time.time() > deadline:
            break
        ca = compute_all_lags(h, n)
        top_lags = np.argsort(ca)[-8:]
        k_pick = top_lags[rng.randint(len(top_lags))]
        lag = k_pick - (n - 1)
        if lag == 0:
            lag = 1 if n > 1 else 0
            if lag == 0:
                break
        m = abs(lag)
        i = rng.randint(0, n - m)
        j = i + m
        if h[i] > h[j]:
            dt = -min(step, (h[i] - h[j]) * 0.5)
        elif h[j] > h[i]:
            dt = min(step, (h[j] - h[i]) * 0.5)
        else:
            dt = rng.choice([-step, step])
        h[i] -= dt
        h[j] += dt
        h = project_exact(h, n)
        cn = compute_c5(h, n)
        if cn < bc - 1e-15:
            bc = cn
            step = min(0.05, step * 1.05)
        else:
            h[i] += dt
            h[j] -= dt
            h = project_exact(h, n)
            step = max(0.0001, step * 0.95)
    return h, bc
