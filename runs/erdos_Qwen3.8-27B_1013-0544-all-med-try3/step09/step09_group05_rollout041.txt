def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                h[:] = 0.5
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                h[:] = 0.5
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def full_smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def full_smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_smooth_obj(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def topk_smooth_grad(x, n, top_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def get_active_shifts(c_vec, top_k, tolerance=0.01):
    m = c_vec.max()
    threshold = m - max(1e-8, tolerance * m)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def run_lbfgs_phase(h, n, deadline, alphas, top_k=None, tolerance=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(123)
    
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        h_cur = best_h.copy()
        for restart in range(2):
            if time.time() > deadline - 3:
                break
            if restart > 0:
                h_cur = best_h + rng.normal(0, 0.01, n)
                h_cur = project_feasible(h_cur, n)
            try:
                if top_k is not None:
                    c_vec = compute_c5_vec(h_cur, n)
                    active = get_active_shifts(c_vec, top_k, tolerance)
                    if len(active) == 0:
                        active = np.array([np.argmax(c_vec)])
                    res = minimize(
                        topk_smooth_obj, h_cur, jac=topk_smooth_grad,
                        args=(n, active, alpha), method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 8000, 'ftol': 1e-18,
                                 'gtol': 1e-12, 'maxfun': 80000, 'maxls': 50})
                else:
                    res = minimize(
                        full_smooth_obj, h_cur, jac=full_smooth_grad,
                        args=(n, alpha), method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 8000, 'ftol': 1e-18,
                                 'gtol': 1e-12, 'maxfun': 80000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def iterative_topk_opt(h, n, deadline, max_rounds=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(456)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        tol = max(0.005, 0.02 * (1.0 - m))
        active = get_active_shifts(c_vec, top_k=10, tolerance=tol)
        
        alphas = [50000, 200000, 1000000]
        for alpha in alphas:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(
                    topk_smooth_obj, best_h, jac=topk_smooth_grad,
                    args=(n, active, alpha), method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 5000, 'ftol': 1e-18,
                             'gtol': 1e-12, 'maxfun': 50000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def gradient_local_search(h, n, deadline, max_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(789)
    steps_since_improve = 0
    max_stall = 5000
    base_delta = 0.005
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if steps_since_improve > max_stall:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        tol = max(0.003, 0.01 * m)
        active = np.where(c_vec >= m - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        grad = np.zeros(n)
        c_local = c_vec[active]
        w = c_local - c_local.min()
        w = np.maximum(w, 0.0)
        if w.sum() < 1e-15:
            w = np.ones(len(active))
        else:
            w /= w.sum()
        
        for wi, idx in zip(w, active):
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += wi * (1.0 - best_h[k:n])
                grad[k:n] += wi * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wi * (1.0 - best_h[0:nv])
                grad[0:nv] += wi * (-best_h[kk:n])
        
        top_grad = np.argsort(np.abs(grad))[:200]
        pos_candidates = top_grad[np.grad > 0 if hasattr(np, 'grad') else grad[top_grad] > 0]
        neg_candidates = top_grad[grad[top_grad] < 0]
        
        if len(pos_candidates) == 0 or len(neg_candidates) == 0:
            pos_candidates = top_grad[:100]
            neg_candidates = top_grad[100:min(200, len(top_grad))]
            if len(neg_candidates) == 0:
                steps_since_improve += 1
                continue
        
        found_improve = False
        for _ in range(5):
            i = pos_candidates[rng.randint(len(pos_candidates))]
            j = neg_candidates[rng.randint(len(neg_candidates))]
            if i == j:
                continue
            delta = base_delta * rng.choice([0.25, 0.5, 1.0, 2.0, 4.0])
            if best_h[i] - delta < 0:
                delta = best_h[i] * 0.5
            if best_h[j] + delta > 1:
                delta = (1.0 - best_h[j]) * 0.5
            if delta < 1e-8:
                continue
            best_h[i] -= delta
            best_h[j] += delta
            c_new = compute_c5(best_h, n)
            if c_new < best_c:
                best_c = c_new
                steps_since_improve = 0
                found_improve = True
            else:
                best_h[i] += delta
                best_h[j] -= delta
        if not found_improve:
            steps_since_improve += 1
            if steps_since_improve % 1000 == 0:
                base_delta *= 0.7
    
    return best_h, best_c

def random_restart_opt(h_best, c_best, n, deadline, rng, n_restarts=5):
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        perturbation = rng.choice([0.005, 0.01, 0.02, 0.05])
        h_pert = h_best + rng.normal(0, perturbation, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        for alpha in [50000, 500000]:
            if time.time() > deadline - 5:
                break
            try:
                c_vec = compute_c5_vec(h_pert, n)
                active = get_active_shifts(c_vec, top_k=8, tolerance=0.02)
                res = minimize(
                    topk_smooth_obj, h_pert, jac=topk_smooth_grad,
                    args=(n, active, alpha), method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 4000, 'ftol': 1e-18,
                             'gtol': 1e-12, 'maxfun': 40000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < c_best:
                    h_best = h_new
                    c_best = c_new
            except Exception:
                pass
    return h_best, c_best

def coarse_to_fine(h_coarse, n_coarse, n_fine, deadline):
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_fine = np.interp(x_new, x_old, best_h)
    h_fine = project_feasible(h_fine, n)
    c_fine = compute_c5(h_fine, n)
    if c_fine < best_c:
        best_c = c_fine
        best_h = h_fine
    else:
        best_h = h_fine
        best_c = c_fine
    
    # Phase 1: Full smooth optimization with increasing alpha
    t_phase1_end = min(t0 + 150, deadline - 500)
    if time.time() < t_phase1_end:
        alphas_p1 = [500, 5000, 50000, 200000]
        best_h, best_c = run_lbfgs_phase(best_h, n, t_phase1_end, alphas_p1, top_k=None)
    
    # Phase 2: Top-k focused optimization (iterative updating of active set)
    t_phase2_end = min(t0 + 350, deadline - 350)
    if time.time() < t_phase2_end:
        best_h, best_c = iterative_topk_opt(best_h, n, t_phase2_end, max_rounds=15)
    
    # Phase 3: Aggressive top-k with very high alpha
    t_phase3_end = min(t0 + 500, deadline - 250)
    if time.time() < t_phase3_end:
        alphas_p3 = [500000, 2000000, 5000000]
        best_h, best_c = run_lbfgs_phase(best_h, n, t_phase3_end, alphas_p3, top_k=5, tolerance=0.005)
    
    # Phase 4: Gradient-guided local search
    t_phase4_end = min(t0 + 700, deadline - 150)
    if time.time() < t_phase4_end:
        best_h, best_c = gradient_local_search(best_h, n, t_phase4_end, max_steps=30000)
    
    # Phase 5: Another round of top-k optimization after local search
    t_phase5_end = min(t0 + 800, deadline - 100)
    if time.time() < t_phase5_end:
        alphas_p5 = [1000000, 5000000]
        best_h, best_c = run_lbfgs_phase(best_h, n, t_phase5_end, alphas_p5, top_k=3, tolerance=0.003)
    
    # Phase 6: Random restarts
    t_phase6_end = min(t0 + 950, deadline - 30)
    if time.time() < t_phase6_end:
        best_h, best_c = random_restart_opt(best_h, best_c, n, t_phase6_end, rng, n_restarts=8)
    
    # Phase 7: Final local search polish
    t_phase7_end = deadline - 2
    if time.time() < t_phase7_end:
        best_h, best_c = gradient_local_search(best_h, n, t_phase7_end, max_steps=15000)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
