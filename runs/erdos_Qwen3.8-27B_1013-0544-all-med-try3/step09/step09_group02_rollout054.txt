            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_c5(h, n, active_indices):
    """Compute gradient of c5 (max) as average of gradients of active constraints."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_indices:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad / len(active_indices)

    # DFT(a ⋆ b) = conj(DFT(a)) * DFT(b)

    # r = ifft(fft(a, N) * conj(fft(v, N)))
    # r[k] = sum_n a[(n+k) % N] * conj(v[n])  (circular)
    # With N >= 2n-1, this gives the linear correlation for k=0,...,n-1

Let me also add a key innovation: **adaptive step size with line search**.

At each step:
1. Compute gradient g of the smooth approximation.
2. Try step sizes [1.0, 0.5, 0.25, 0.1, 0.05, 0.01] * ||g||_∞^-1
3. Pick the one that gives the best c5.

This is more robust than fixed step sizes.

Also, **perturbation-re-start**: occasionally add a small random perturbation to escape local minima.

Let me also consider: should I work with the "dual" variable (the distribution p over shifts)? 

The idea: maintain p (a probability distribution over shifts). The objective becomes:
min_h ∑_k p_k * c_k(h)

This is a sum of (not necessarily convex) quadratic functions. The gradient is:
∇_h ∑ p_k c_k = ∑ p_k ∇c_k

And we can update p based on which c_k are large (like in multiplicative weights / exponential weights).

This gives a primal-dual method:
- Primal: min_h ∑ p_k c_k(h) (with feasibility constraints)
- Dual: max_p ∑ p_k * (c_k(h*)) (i.e., concentrate p on the worst constraints)

This is essentially the same as the SQP approach but with a continuous p instead of a discrete active set.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, idx):
    dx = 2.0 / n
    g = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n]) * dx
        g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv]) * dx
        g[0:nv] += (-h[kk:n]) * dx
    return g

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 300:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    # Only iterate over significant weights
    thresh = w.max() * 1e-6
    for mm in range(2 * n - 1):
        if w[mm] < thresh:
            continue
        kk = mm - (n - 1)
        wm = w[mm] * dx
        if kk >= 0:
            nv = n - kk
            g[0:nv] += wm * (1.0 - h[kk:n])
            g[kk:n] += wm * (-h[0:nv])
        else:
            jj = -kk
            nv = n - jj
            g[jj:n] += wm * (1.0 - h[0:nv])
            g[0:nv] += wm * (-h[jj:n])
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def gradient_flow(h, n, deadline, n_active=10):
    """Gradient flow on the active set with adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    step_scale = 0.1
    
    while time.time() < deadline and no_improve < 50:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Get top n_active shifts
        top_idx = np.argsort(-c_vec)[:n_active]
        # Weight by excess
        excess = c_vec[top_idx] - (max_val - 0.01 * max_val)
        excess = np.clip(excess, 0, None)
        if excess.sum() < 1e-12:
            break
        w = excess / excess.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            g += w[i] * grad_single(h, n, int(idx))
        
        # Adaptive step: try multiple sizes
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-14:
            break
        candidates = [step_scale * s for s in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]]
        
        best_step_c = best_c
        best_step_h = h
        for s in candidates:
            h_new = project_feasible(h - s * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_step_c - 1e-15:
                best_step_c = c_new
                best_step_h = h_new
                step_scale = s * 1.5  # try bigger next time
                break
        else:
            # Try even smaller
            h_new = project_feasible(h - 0.001 * g / g_norm, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_step_c = c_new
                best_step_h = h_new
                step_scale *= 0.5
            else:
                no_improve += 1
                step_scale *= 0.5
        
        if best_step_c < best_c - 1e-15:
            best_c = best_step_c
            best_h = best_step_h
            h = best_step_h
            no_improve = 0
    return best_h, best_c

def pair_swap_search(h, n, deadline, radius=30):
    """Local search by swapping mass between pairs of positions."""
    best_c = compute_c5(h, n)
    h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Focus on positions near active constraints
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            active = np.where(c_vec >= max_val - 0.001 * max_val)[0]
            
            # For each active constraint, find pairs to swap
            for idx in active[:5]:
                if time.time() > deadline:
                    return h, best_c
                k = idx - (n - 1)
                if k >= 0:
                    # h[m] and h[m+k]: want to reduce h[m]*(1-h[m+k])
                    # Increase h[m+k] and decrease h[m]
                    nv = n - k
                    for m in range(0, nv, 2):  # stride 2 for speed
                        i = m
                        j = m + k
                        if h[i] > step and h[j] < 1.0 - step:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += step
                                h[j] -= step
                else:
                    kk = -k
                    nv = n - kk
                    for m in range(0, nv, 2):
                        i = m + kk
                        j = m
                        if h[i] > step and h[j] < 1.0 - step:
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] += step
                                h[j] -= step
    return h, best_c

def lbfgs_smooth(h, n, deadline, alpha_configs):
    """L-BFGS with smooth max approximation."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha, mu in alpha_configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def perturbation_restart(h, n, deadline, n_restarts=5):
    """Try small random perturbations and keep the best."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Small random perturbation
        noise = np.random.randn(n) * 0.01
        h_pert = project_feasible(h + noise, n)
        c_pert = compute_c5(h_pert, n)
        
        if c_pert < best_c + 0.001:  # Only continue if not much worse
            h_pert, c_pert = gradient_flow(h_pert, n, min(time.time() + 30, deadline))
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert
    
    return best_h, best_c
