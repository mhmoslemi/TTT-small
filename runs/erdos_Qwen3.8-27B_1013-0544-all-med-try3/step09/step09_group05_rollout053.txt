import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_smooth_obj(n, alpha, active_set):
    def obj(x):
        c = compute_c5_vec(x, n)
        vals = c[active_set]
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        if smx > 500:
            val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = mx + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)
    return obj

def smooth_obj(x, n, alpha, active_set):
    c = compute_c5_vec(x, n)
    vals = c[active_set]
    mx = vals.max()

def smooth_grad(x, n, alpha, active_set):
    c = compute_c5_vec(x, n)
    vals = c[active_set]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    n_half = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, active_set):
        if wi < 1e-15:
            continue
        k = idx - n_half
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    threshold = mx - max(1e-8, 0.005 * mx)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        idx_sorted = active[np.argsort(-c[active])]
        active = idx_sorted[:top_k]
    return active

        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Ultra-fine local search
    if time.time() < deadline - 5:
        h9, c9 = fine_local_search(best_h, n, deadline - 3, max_pairs=30, center_radius=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
