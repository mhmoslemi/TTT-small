import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad_full(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs(h, n, deadline, alphas=None, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 5,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def tabu_search_binary(h_bin, n, deadline, tabu_len=20, max_iter=None, rng=None):
    """Tabu search for binary optimization."""
    if rng is None:
        rng = np.random.RandomState(42)
    if max_iter is None:
        max_iter = 500000
    
    h = h_bin.copy()
    c = compute_c5_vec(h, n)
    best_c = c.max()
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Tabu list: store (index, iteration) for recently moved positions
    tabu = {}  # index -> iteration when it was last moved
    current_iter = 0
    
    # Temperature for occasional diversification
    temp = 0.01
    
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        current_iter = it
        
        # Cheap time check every 1000 iterations
        if it % 1000 == 0 and time.time() > deadline:
            break
        
        # Pick a random 1 and a random 0
        idx = rng.randint(len(ones))
        p = ones[idx]  # position to change from 1 to 0
        idx2 = rng.randint(len(zeros))
        q = zeros[idx2]  # position to change from 0 to 1
        
        # Check tabu
        if it - tabu.get(p, -999) < tabu_len or it - tabu.get(q, -999) < tabu_len:
            continue
        
        # Try the move
        h[p] = 0
        h[q] = 1
        c_new = compute_c5_vec(h, n)
        new_max = c_new.max()
        
        # Aspiration: accept if better than best
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = c_new
            # Update sets
            ones = np.delete(ones, idx)
            zeros = np.append(zeros, p)
            # Remove q from zeros, add to ones
            # This is O(n) each time, which is fine
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            tabu[p] = it
            tabu[q] = it
        elif new_max < best_c:
            # Accept (improvement but not new best)
            c = c_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            tabu[p] = it
            tabu[q] = it
        else:
            # Reject
            h[p] = 1
            h[q] = 0
        
        # Cooling
        if it % 5000 == 0:
            temp *= 0.95
    
    return best_h, float(best_c)

Let me optimize the tabu search to be faster. Instead of recomputing the full correlate, I can compute the delta:

When h[p] goes 1→0 and h[q] goes 0→1:
delta_c = contribution from p going 1→0 + contribution from q going 0→1

For h[p] going 1→0:
- In the first term of c[k] = sum h[i+k]*(1-h[i]): h[p] appears as h[i+k] when i = p-k
  Contribution lost: (1-h[p-k]) for valid (p-k)
- In the second term: h[p] appears as h[i] when i = p
  Contribution gained: h[p+k] * (1-0) - h[p+k] * (1-1) = h[p+k] for valid (p+k)
  Wait, the term is h[i+k]*(1-h[i]). When i=p: h[p+k]*(1-h[p]). Before: h[p+k]*0=0. After: h[p+k]*1=h[p+k]. So delta = +h[p+k].

For h[q] going 0→1:
- h[q] appears as h[i+k] when i = q-k: contribution gained: 1*(1-h[q-k]) = (1-h[q-k]) for valid
- h[q] appears as h[i] when i = q: term h[q+k]*(1-h[q]). Before: h[q+k]*1=h[q+k]. After: h[q+k]*0=0. Delta = -h[q+k].

So:
delta_c[k] = -(1-h[p-k])*[p-k valid for k] + h[p+k]*[p+k valid for k]
           + (1-h[q-k])*[q-k valid for k] - h[q+k]*[q+k valid for k]

Where "valid" means the index is in [0, n-1] and the shift k is valid.

For the correlate with mode='full', the output has 2n-1 elements indexed 0 to 2n-2.
Element m corresponds to shift k = m - (n-1).

For shift k (where -n+1 ≤ k ≤ n-1):
- Valid i range: max(0, -k) ≤ i ≤ min(n-1, n-1-k)
- h[i+k] is valid when 0 ≤ i+k ≤ n-1
- h[i] is valid when 0 ≤ i ≤ n-1

For h[p] appearing as h[i+k] (i.e., i+k=p, so i=p-k):
- Valid if max(0,-k) ≤ p-k ≤ min(n-1, n-1-k) and 0 ≤ p ≤ n-1 (always)
- Simplified: p-k ≥ 0 and p-k ≤ n-1-k (for k≥0) or p-k ≥ -k and p-k ≤ n-1 (for k<0)
- For k≥0: 0 ≤ p-k ≤ n-1-k, i.e., k ≤ p ≤ n-1
- For k<0: -k ≤ p-k ≤ n-1, i.e., 0 ≤ p ≤ n-1+k (i.e., p ≤ n-1+k, which is p+k ≤ n-1)

For h[p] appearing as h[i] (i.e., i=p):
- Valid if max(0,-k) ≤ p ≤ min(n-1, n-1-k)
- For k≥0: 0 ≤ p ≤ n-1-k, i.e., p+k ≤ n-1
- For k<0: -k ≤ p ≤ n-1, i.e., p ≥ -k (i.e., p+k ≥ 0)

    stall_threshold = tabu_len * 10
    
    while time.time() < deadline:
        current_iter += 1
        
        if current_iter % 5000 == 0:
            no_improve_count += 1
            if no_improve_count > 5:
                break
        
        # Diversification: make random non-improving moves
        if div_moves > 0 and current_iter % diversify_interval == 0:
            for _ in range(div_moves):
                if time.time() > deadline:
                    break
                if len(ones_set) == 0 or len(zeros_set) == 0:
                    break
                p = rng.choice(list(ones_set))
                q = rng.choice(list(zeros_set))
                h[p] = 0
                h[q] = 1
                ones_set.remove(p)
                ones_set.add(q)
                zeros_set.remove(q)
                zeros_set.add(p)
                c = compute_c5_vec(h, n)
                tabu_until[p] = current_iter + tabu_len
                tabu_until[q] = current_iter + tabu_len
            continue
        
        # Pick random 1 and random 0 (avoiding tabu)
        ones_list = [p for p in ones_set if tabu_until.get(p, -1) < current_iter]
        zeros_list = [q for q in zeros_set if tabu_until.get(q, -1) < current_iter]
        
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        p = rng.choice(ones_list)
        q = rng.choice(zeros_list)
        
        # Try the move
        h[p] = 0.0
        h[q] = 1.0
        c_new = compute_c5_vec(h, n)
        new_max = float(c_new.max())
        
        # Aspiration: accept if new best
        if new_max < best_c - 1e-14:
            best_c = new_max
            best_h = h.copy()
            no_improve_count = 0
        elif new_max <= best_c + 1e-14:
            # Accept (not worse than best)
            pass
        else:
            # Reject
            h[p] = 1.0
            h[q] = 0.0
            continue
        
        # Commit the move
        ones_set.remove(p)
        ones_set.add(q)
        zeros_set.remove(q)
        zeros_set.add(p)
        c = c_new
        tabu_until[p] = current_iter + tabu_len
        tabu_until[q] = current_iter + tabu_len
    
    return best_h, best_c

def pair_swap_search(h, n, deadline, neighbour_radius=30, max_rounds=200):
    """Local search: try all pairs within a radius."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    ones = np.where(best_h > 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in ones:
            if time.time() > deadline:
                break
            lo = max(0, i - neighbour_radius)
            hi = min(n, i + neighbour_radius + 1)
            for j in zeros:
                if j < lo or j >= hi:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5_vec(best_h, n)
                new_c = float(c.max())
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    zeros.sort()
                    improved = True
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
        if not improved:
            break
    return best_h, best_c

def continuous_to_binary_continuous(h, n):
    """Get a binary solution and also a soft version."""
    h_bin = to_binary(h, n)
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(500, 0.5)
    
    n_orig = len(h0)
    h0 = project_feasible(h0, n_orig)
    
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at n_orig (quick polish)
    p1_end = min(t0 + 80, deadline - 400)
    if p1_end > t0 + 5:
        h1, c1 = phase_lbfgs(best_h, n_orig, p1_end,
                             alphas=[20, 100, 500, 2000, 10000, 100000],
                             mu=0.01)
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = n_orig
    
    # Phase 2: Binary tabu search at n_orig
    p2_end = min(t0 + 250, deadline - 300)
    if p2_end > time.time() + 20:
        h_bin = to_binary(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        h2, c2 = tabu_search(h_bin, best_n, p2_end, tabu_len=max(20, best_n//25),
                             rng=rng, diversify_interval=3000, div_moves=8)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = best_n
    
    # Phase 3: Pair swap at n_orig
    p3_end = min(time.time() + 80, deadline - 250)
    if p3_end > time.time() + 10:
        # Use the binary or continuous best
        h_work = best_h.copy()
        h3, c3 = pair_swap_search(h_work, best_n, p3_end, neighbour_radius=40, max_rounds=100)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n
    
    # Phase 4: L-BFGS polish at n_orig
    p4_end = min(time.time() + 80, deadline - 200)
    if p4_end > time.time() + 10:
        h4, c4 = phase_lbfgs(best_h, best_n, p4_end,
                             alphas=[5000, 50000, 500000, 5000000],
                             mu=0.001)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n
    
    # Phase 5: Multi-resolution - upsample to 2*n and refine
    if time.time() < deadline - 200:
        # Upsample
        h_up = upsample(best_h, factor=2)
        n_up = len(h_up)
        
        # Quick L-BFGS at higher resolution
        p5a_end = min(time.time() + 100, deadline - 150)
        if p5a_end > time.time() + 10:
            h5, c5 = phase_lbfgs(h_up, n_up, p5a_end,
                                 alphas=[100, 1000, 10000, 100000],
                                 mu=0.01)
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_up
        
        # Tabu search at higher resolution
        if time.time() < deadline - 80:
            h_bin_up = to_binary(best_h, best_n)
            p5b_end = min(time.time() + 80, deadline - 60)
            if p5b_end > time.time() + 15:
                h5b, c5b = tabu_search(h_bin_up, best_n, p5b_end,
                                       tabu_len=max(30, best_n//25),
                                       rng=rng, diversify_interval=2000, div_moves=6)
                if c5b < best_c:
                    best_c = c5b
                    best_h = h5b
                    best_n = best_n
    
    # Phase 6: Final L-BFGS polish at best resolution
    if time.time() < deadline - 20:
        p6_end = deadline - 5
        h6, c6 = phase_lbfgs(best_h, best_n, p6_end,
                             alphas=[10000, 100000, 1000000, 10000000],
                             mu=0.001)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n
    
    # Phase 7: Very fine pair search at the end
    if time.time() < deadline - 3:
        h_final = project_feasible(best_h, best_n)
        h7, c7 = pair_swap_search(h_final, best_n, deadline - 2, neighbour_radius=20, max_rounds=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
