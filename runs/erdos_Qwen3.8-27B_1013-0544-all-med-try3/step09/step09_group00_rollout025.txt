    alpha_cool = (T1 / T0) ** (1.0 / max(1, n_iter))
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (alpha_cool ** it)
        ia = rng.randint(len(ones_list))
        ib = rng.randint(len(zeros_list))
        a = ones_list[ia]
        b = zeros_list[ib]
        h[a] = 0.0
        h[b] = 1.0
        c = compute_c5(h, n)
        delta = c - current_c
        if delta < 0 or rng.uniform() < np.exp(-delta / T):
            current_c = c
            ones_list[ia] = b
            zeros_list[ib] = a
            if c < best_c:
                best_c = c
        else:
            h[a] = 1.0
            h[b] = 0.0
    return h, best_c

def fractional_optimize(h, n, deadline):
    """
    Given a near-binary h, identify the fractional values and optimize them.
    Force most to 0 or 1, keep top-K fractional ones free.
    """
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    # Try different numbers of free fractional values
    for n_free in [0, 2, 4, 8, 16, 32]:
        if time.time() > deadline:
            break
        h_work = h.copy()
        if n_free > 0:
            # Find values that are fractional (between 0.01 and 0.99)
            frac_mask = (h_work > 0.01) & (h_work < 0.99)
            if frac_mask.sum() > n_free:
                # Force the most extreme ones to 0 or 1
                frac_vals = h_work[frac_mask]
                # Keep the ones closest to 0.5 (most "fractional")
                dist_from_half = np.abs(frac_vals - 0.5)
                # Sort: keep the smallest dist (closest to 0.5)
                order = np.argsort(dist_from_half)
                keep = order[:n_free]
                force = order[n_free:]
                frac_indices = np.where(frac_mask)[0]
                for i in force:
                    idx = frac_indices[i]
                    h_work[idx] = 1.0 if h_work[idx] > 0.5 else 0.0
                # Re-project
                h_work = project_feasible(h_work, n)
        
        # Now optimize the free values using L-BFGS
        if n_free > 0:
            free_mask = (h_work > 1e-6) & (h_work < 1.0 - 1e-6)
            free_idx = np.where(free_mask)[0]
            if len(free_idx) > 0:
                # Use smooth max objective
                best_c, h = _optimize_free_values(h_work, n, free_idx, deadline)
        
        c = compute_c5(h_work, n)
        if c < best_c:
            best_c = c
            h = h_work.copy()
    
    return h, best_c

def _smooth_max_obj(full_h, free_idx, n, alpha):
    c = compute_c5_vec(full_h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _optimize_free_values(h, n, free_idx, deadline):
    """Optimize values at free_idx to minimize C5 while maintaining sum constraint."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # The sum constraint: sum(free_idx) is determined by the rest
    rest_sum = h.sum() - h[free_idx].sum()
    target_free = n / 2.0 - rest_sum
    
    # If target_free is 0 or len(free_idx), the free values are determined
    if target_free <= 0 or target_free >= len(free_idx):
        return best_c, best_h
    
    # Penalty-based: minimize smooth_max + penalty for sum violation
    mu = 100.0
    alpha = 100000
    
    def obj(x):
        h_full = h.copy()
        h_full[free_idx] = np.clip(x, 0, 1)
        c = compute_c5_vec(h_full, n)
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        if smx > 400:
            val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (h_full[free_idx].sum() - target_free) ** 2
        return float(val)
    
    def grad(x):
        h_full = h.copy()
        h_full[free_idx] = np.clip(x, 0, 1)
        c = compute_c5_vec(h_full, n)
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        dx_val = 2.0 / n
        grad_full = np.zeros(n)
        for mm in range(len(c)):
            if w[mm] < 1e-15:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx_val
            if k >= 0:
                nv = n - k
                grad_full[0:nv] += wm * (1.0 - h_full[k:n])
                grad_full[k:n] += wm * (-h_full[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad_full[kk:n] += wm * (1.0 - h_full[0:nv])
                grad_full[0:nv] += wm * (-h_full[kk:n])
        g = grad_full[free_idx].copy()
        g += 2.0 * mu * (np.clip(x, 0, 1).sum() - target_free)
        return g
    
    x0 = h[free_idx].copy()
    bounds = [(0.0, 1.0)] * len(free_idx)
    
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
        x_opt = np.clip(res.x, 0, 1)
        h_new = h.copy()
        h_new[free_idx] = x_opt
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return c_new, h_new
    except Exception:
        pass
    
    return best_c, best_h

def directed_swap_search(h, n, deadline, max_rounds=50):
    """
    For each 1-position, try swapping with each 0-position and keep the best.
    More directed than random search.
    """
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        for i in ones:
            if time.time() > deadline:
                break
            best_delta = 0
            best_j = -1
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if c < best_c - 1e-15:
                    best_delta = best_c - c
                    best_j = j
            if best_j >= 0:
                h[i] = 0.0
                h[best_j] = 1.0
                best_c = best_c - best_delta
                improved = True
                break
        if not improved:
            break
    return h, best_c

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_smooth_max(h, n, deadline, alpha, mu, maxit):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = compute_c5_vec(x, n)
        v = smooth_max_obj_val(c, alpha)
        v += mu * (x.sum() - n / 2.0) ** 2
        return v

    def grad(x):
        c = compute_c5_vec(x, n)
        g = smooth_max_obj_grad(c, alpha, x, n)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-14,
                               'maxfun': maxit * 8, 'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            return h_new, c_new
    except Exception:
        pass
    return best_h, best_c

def multi_resolution_attempt(h_best, n_best, c_best, deadline, rng):
    for n_try in [n_best + n_best // 2, n_best * 2]:
        if time.time() > deadline - 200:
            break
        if n_try % 2 != 0:
            n_try += 1
        # Upsample
        h_up = np.zeros(n_try)
        ratio = n_try / n_best
        for i in range(n_best):
            lo = int(i * ratio)
            hi = int((i + 1) * ratio)
            if hi > n_try:
                hi = n_try
            if hi > lo:
                h_up[lo:hi] = h_best[i]
        h_up = project_feasible(h_up, n_try)
        c_up = compute_c5(h_up, n_try)
        if c_up >= c_best + 0.002:
            continue
        # Hill climb on upsampled
        dl = min(time.time() + 120, deadline - 100)
        h_up, c_up = hill_climb_random(h_up, n_try, dl, 100000, rng)
        if c_up < c_best:
            h_best = h_up
            c_best = c_up
            n_best = n_try
    return h_best, c_best, n_best

def active_set_lbfgs(h, n, deadline, top_k=8, max_rounds=10):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thr = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        alpha = 500000
        mu = 0.001

        a_vec = c_vec[active]
        a_max = a_vec.max()
        sh = alpha * (a_vec - a_max)
        smx = sh.max()
        if smx > 400:
            w = np.exp(sh - smx)
        else:
            w = np.exp(sh)
        w /= w.sum()

        def obj(x):
            c = compute_c5_vec(x, n)
            av = c[active]
            am = av.max()
            ssh = alpha * (av - am)
            smx2 = ssh.max()
            if smx2 > 400:
                v = am + (smx2 + np.log(np.sum(np.exp(ssh - smx2)))) / alpha
            else:
                v = am + np.log(np.sum(np.exp(ssh))) / alpha
            v += mu * (x.sum() - n / 2.0) ** 2
            return float(v)

        def grad(x):
            c = compute_c5_vec(x, n)
            av = c[active]
            am = av.max()
            ssh = alpha * (av - am)
            smx2 = ssh.max()
            ww = np.exp(ssh - smx2)
            ww /= ww.sum()
            dx_val = 2.0 / n
            g = np.zeros(n)
            for wi, idx in zip(ww, active):
                k = idx - (n - 1)
                wm = wi * dx_val
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g

        maxit = max(200, int((deadline - time.time()) * 1.5))
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = make_beatty(n, (1.0 + np.sqrt(5.0)) / 4.0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    golden2 = (1.0 + np.sqrt(5.0)) / 4.0
    starts = []
    starts.append(make_beatty(n, golden2))
    starts.append(make_beatty(n, (np.sqrt(5.0) - 1.0) / 4.0))
    starts.append(make_beatty(n, np.sqrt(2.0) / 3.0))
    starts.append(make_beatty(n, np.sqrt(3.0) / 4.0))
    starts.append(make_beatty(n, np.pi / 5.0))
    starts.append(make_beatty(n, 0.414))
    starts.append(make_beatty(n, 0.577))
    starts.append(make_random_binary(n, rng))
    starts.append(make_random_binary(n, rng))
    starts.append(make_random_binary(n, rng))
    starts.append(make_staggered(n, 1))
    starts.append(make_staggered(n, 0))

    for s in starts:
        c_s = compute_c5(s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = s.copy()

    # Phase 1: Random hill climb (long)
    p1_end = min(t0 + 200, deadline - 400)
    h1, c1 = hill_climb_random(best_h, n, p1_end, 2000000, rng)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Simulated annealing (escape local min)
    p2_end = min(t0 + 400, deadline - 300)
    h2, c2 = simulated_annealing(best_h, n, p2_end, rng, T0=0.002, T1=0.00001, max_iters=800000)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Directed local search (best improvement)
    p3_end = min(t0 + 600, deadline - 250)
    h3, c3 = directed_local_search(best_h, n, p3_end, max_passes=25)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active-set L-BFGS on fractional values
    p4_end = min(t0 + 750, deadline - 200)
    h4, c4 = active_set_lbfgs(best_h, n, p4_end, top_k=8, max_rounds=8)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: More hill climbing from refined solution
    p5_end = min(t0 + 880, deadline - 150)
    h5, c5 = hill_climb_random(best_h, n, p5_end, 1500000, rng)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Another SA round with different temps
    p6_end = min(t0 + 980, deadline - 100)
    h6, c6 = simulated_annealing(best_h, n, p6_end, rng, T0=0.0005, T1=0.000005, max_iters=400000)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Final directed polish
    p7_end = min(t0 + 1050, deadline - 30)
    h7, c7 = directed_local_search(best_h, n, p7_end, max_passes=15)
    if c7 < best_c:
        best_c = c7
        best_h = h7

    # Phase 8: Final hill climb
    p8_end = min(t0 + budget_s - 8, deadline - 2)
    h8, c8 = hill_climb_random(best_h, n, p8_end, 300000, rng)
    if c8 < best_c:
        best_c = c8
        best_h = h8

    # Multi-resolution (if time remains)
    if time.time() < deadline - 15:
        remaining = deadline - time.time() - 10
        if remaining > 100:
            best_h, best_c, n = multi_resolution_attempt(best_h, n, best_c, t0 + remaining, rng)

    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
