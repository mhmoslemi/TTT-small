```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lp_polish(h0, n, deadline, delta_init=0.1, n_max_iters=50):
    """Successive LP polishing."""
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    h = h0.copy()
    best_c = compute_c5(h, n)
    
    delta = delta_init
    for it in range(n_max_iters):
        if time.time() > deadline:
            break
        
        # Compute current overlaps
        c_full = np.correlate(h, 1 - h, mode='full')  # unscaled by dx
        
        # Compute linearized coefficients
        # For shift s (from -(n-1) to n-1):
        # d(c[idx])/d(h[k]) where idx = center + s
        A_coeff = np.zeros((n2m1, n))
        
        for idx in range(n2m1):
            s = idx - center
            if s >= 0:
                # Valid n: 0 to n-1-s
                # d/dh[k] contribution:
                # (1-h[n]) where k = n+s, n in [0, n-1-s] => k in [s, n-1]
                # -h[n+s] where k = n, n in [0, n-1-s] => k in [0, n-1-s]
                if n - 1 - s >= 0:
                    A_coeff[idx, s:n] += (1 - h[0:n-1-s])
                    A_coeff[idx, 0:n-s] += (-h[s:n])
            else:
                ss = -s
                # Valid n: ss to n-1
                # c[idx] = sum_{n=ss}^{n-1} h[n-ss]*(1-h[n])
                # d/dh[k]:
                # (1-h[n]) where k = n-ss, n in [ss, n-1] => k in [0, n-1-ss]
                # -h[n-ss] where k = n, n in [ss, n-1] => k in [ss, n-1]
                if n - 1 - ss >= 0:
                    A_coeff[idx, 0:n-ss] += (1 - h[ss:n])
                    A_coeff[idx, ss:n] += (-h[0:n-ss])
        
        # Solve LP: min t
        # s.t. A_coeff @ h - t <= A_coeff @ h0 - c_full (for each shift)
        #      h0[i] - delta <= h[i] <= h0[i] + delta
        #      0 <= h[i] <= 1
        #      sum h = n/2
        
        # b_ub for shift constraints:
        # A_coeff[s,:] @ h_new - t <= A_coeff[s,:] @ h - c_full[idx]
        b_ub = A_coeff @ h - c_full  # length n2m1
        
        # Build constraint matrix
        # A_ub has n2m1 rows: [A_coeff, -1]
        A_ub = np.zeros((n2m1, n + 1))
        A_ub[:, :n] = A_coeff
        A_ub[:, n] = -1.0
        
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Bounds
        lb = np.maximum(0, h - delta)
        ub = np.minimum(1, h + delta)
        bounds = list(zip(lb, ub)) + [(None, None)]
        
        # Equality: sum h = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n].copy()
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new.copy()
                    # Decrease delta
                    delta *= 0.7
                else:
                    delta *= 0.5
            else:
                delta *= 0.5
        except Exception:
            delta *= 0.5
        
        if delta < 0.001:
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_A_coeff(h0, n):
    """Compute the linearization matrix A_coeff[idx, k] = d(c[idx])/d(h[k]) at h0."""
    n2m1 = 2 * n - 1
    A_coeff = np.zeros((n2m1, n))
    center = n - 1
    
    for idx in range(n2m1):
        s = idx - center
        if s >= 0:
            # d/dh[k] = (1-h0[k-s]) * 1_{k>=s} + (-h0[k+s]) * 1_{k<=n-1-s}
            if s < n:
                # First part: k from s to n-1
                A_coeff[idx, s:n] += (1.0 - h0[0:n-s])
                # Second part: k from 0 to n-1-s
                A_coeff[idx, 0:n-s] += (-h0[s:n])
        else:
            ss = -s
            if ss < n:
                # d/dh[k] = (1-h0[k+ss]) * 1_{k<=n-1-ss} + (-h0[k-ss]) * 1_{k>=ss}
                # First part: k from 0 to n-1-ss
                A_coeff[idx, 0:n-ss] += (1.0 - h0[ss:n])
                # Second part: k from ss to n-1
                A_coeff[idx, ss:n] += (-h0[0:n-ss])
    
    return A_coeff
```

```python
def lp_polish(h0, n, deadline, delta_init=0.05, max_iters=100):
    h = h0.copy()
    best_c = compute_c5(h, n)
    delta = delta_init
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_full = np.correlate(h, 1 - h, mode='full')  # unscaled
        A_coeff = compute_A_coeff(h, n)
        
        # b_ub: A_coeff @ h - c_full
        b_ub = A_coeff @ h - c_full
        
        # LP: min t
        # s.t. A_coeff @ h_new - t <= b_ub
        #      h[i] - delta <= h_new[i] <= h[i] + delta
        #      0 <= h_new[i] <= 1
        #      sum h_new = n/2
        
        n2m1 = 2 * n - 1
        A_ub = np.column_stack([A_coeff, -np.ones(n2m1)])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        lb = np.maximum(0.0, h - delta)
        ub = np.minimum(1.0, h + delta)
        bounds = list(zip(lb, ub))
        bounds.append((None, None))  # t is unbounded
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'maxiter': 10000})
            if res.success:
                h_new = np.clip(res.x[:n], 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new.copy()
                    delta *= 0.75
                else:
                    delta *= 0.5
            else:
                delta *= 0.5
        except Exception:
            delta *= 0.5
        
        if delta < 0.0005:
            break
    
    return h, best_c
```
