            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(a * m1 + base) <= np.max(a * m2 + base):
                    hi = m2
                else:
                    lo = m1
            t_star = (lo + hi) / 2.0
            g0 = np.max(base)
            g1 = np.max(a + base)
            gt = np.max(a * t_star + base)
            if g0 <= g1 and g0 <= gt:
                t_star = 0.0
            elif g1 <= g0 and g1 <= gt:
                t_star = 1.0
            if abs(t_star - h[i]) > 1e-10:
                h[i] = t_star
                improved += 1
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        if improved == 0:
            break
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def perturb_reoptimize(h, n, deadline, n_perturb=10, scale=0.02):
    """Randomly perturb a few coordinates and re-optimize with short L-BFGS."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(12345)
    bounds = [(0.0, 1.0)] * n
    attempts = 0
    while time.time() < deadline and attempts < 30:
        attempts += 1
        h_pert = h.copy()
        idx = rng.choice(n, size=n_perturb, replace=False)
        h_pert[idx] += rng.normal(0, scale, size=n_perturb)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.005:
            rem = deadline - time.time()
            if rem < 5:
                break
            maxit = max(100, int(rem * 1.5))
            try:
                res = minimize(sm_obj, h_pert, jac=sm_grad,
                               args=(n, 200000, 0.005), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': maxit * 8,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(400, 0.5)
        n_orig = 400

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS polish at original resolution
    p1_end = t0 + 60
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[200, 2000, 20000, 100000, 500000],
        mu=0.005, maxiters=[800, 1000, 1500, 1500, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Coordinate descent (novel method)
    p2_end = min(t0 + 250, deadline - 500)
    h2, c2 = coord_sweep(best_h, best_n, p2_end)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: L-BFGS with very high alpha after coordinate descent
    p3_end = min(t0 + 380, deadline - 400)
    h3, c3 = run_lbfgs(best_h, best_n, p3_end,
        alphas=[500000, 2000000, 5000000, 20000000, 50000000],
        mu=0.003, maxiters=[1500, 2000, 2000, 2000, 2000])
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active set optimization
    p4_end = min(t0 + 500, deadline - 320)
    h4, c4 = run_active_opt(best_h, best_n, p4_end, top_k=5,
        alphas=[200000, 1000000, 5000000, 20000000])
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: More coordinate descent
    p5_end = min(t0 + 620, deadline - 250)
    h5, c5 = coord_sweep(best_h, best_n, p5_end)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Perturbation + re-optimization
    p6_end = min(t0 + 730, deadline - 200)
    h6, c6 = perturb_reoptimize(best_h, best_n, p6_end, n_perturb=8, scale=0.03)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Upsample to n=1000 and optimize
    n_high = 1000
    if time.time() < deadline - 200 and best_n < n_high:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.002:
            p7_end = min(time.time() + 120, deadline - 100)
            h7, c7 = run_lbfgs(h_high, n_high, p7_end,
                alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                mu=0.002, maxiters=[500, 700, 1000, 1200, 1500, 1500, 2000, 2500])
            if c7 < best_c:
                best_c = c7
                best_h = h7
                best_n = n_high
                # Coordinate descent at higher resolution
                rem = deadline - time.time()
                if rem > 40:
                    h7b, c7b = coord_sweep(best_h, best_n, deadline - 30)
                    if c7b < best_c:
                        best_c = c7b
                        best_h = h7b

    # Phase 8: Final coordinate descent + pair search on best solution
    p8_end = deadline - 30
    if time.time() < p8_end:
        h8, c8 = coord_sweep(best_h, best_n, p8_end - 15)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Pair search
    if time.time() < deadline - 5:
        h9, c9 = pair_search(best_h, best_n, deadline - 2, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
