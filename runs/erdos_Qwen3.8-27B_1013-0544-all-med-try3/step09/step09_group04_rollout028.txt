def compute_a_fast(j, h, n, dx):
    """Vectorized computation of a_ℓ for coordinate j."""
    # ℓ ranges from -(n-1) to n-1, stored in index m = ℓ + (n-1)
    # a[m] = [0≤j-ell<n]*(1-h[j-ell])*dx - [0≤j+ell<n]*h[j+ell]*dx (for ell≠0)
    # a[n-1] = (1-2*h[j])*dx (for ell=0)
    
    ell = np.arange(-(n-1), n)  # shape (2n-1,)
    i1 = j - ell  # shape (2n-1,)
    i2 = j + ell  # shape (2n-1,)
    
    valid1 = (i1 >= 0) & (i1 < n)  # shape (2n-1,)
    valid2 = (i2 >= 0) & (i2 < n)  # shape (2n-1,)
    
    # Clip for safe indexing
    i1_clipped = np.clip(i1, 0, n-1)
    i2_clipped = np.clip(i2, 0, n-1)
    
    a = np.zeros(2*n - 1)
    a = np.where(valid1, (1 - h[i1_clipped]) * dx, 0.0)
    a = a - np.where(valid2, h[i2_clipped] * dx, 0.0)
    
    # Fix ℓ=0 (index n-1): both valid1 and valid2 are true, giving (1-h[j]) - h[j] = 1 - 2*h[j]
    # This is already correct! Because a[n-1] = (1-h[j])*dx - h[j]*dx = (1-2*h[j])*dx
    return a
```

This is fully vectorized and O(n) in numpy.

def min_1d_max(D, a, lo=0.0, hi=1.0):
    """Find t in [lo, hi] minimizing max_k (D[k] + a[k]*t)."""
    for _ in range(60):

        f = D + a * t
        idx = np.argmax(f)
        if a[idx] > 1e-14:
            hi = t
        elif a[idx] < -1e-14:
            lo = t
        else:
            break
    return (lo + hi) * 0.5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_a_fast(j, h, n, dx):
    """Compute a_ell for all ell for coordinate j (vectorized)."""
    ell = np.arange(-(n-1), n)
    i1 = j - ell
    i2 = j + ell
    valid1 = (i1 >= 0) & (i1 < n)
    valid2 = (i2 >= 0) & (i2 < n)
    i1c = np.clip(i1, 0, n-1)
    i2c = np.clip(i2, 0, n-1)
    a = np.where(valid1, (1.0 - h[i1c]) * dx, 0.0)
    a = a - np.where(valid2, h[i2c] * dx, 0.0)
    return a

def min_1d_max(D, a, lo=0.0, hi=1.0):
    """Find t in [lo,hi] minimizing max_k(D[k] + a[k]*t)."""
    for _ in range(50):
        t = (lo + hi) * 0.5
        f = D + a * t
        idx = np.argmax(f)
        if a[idx] > 1e-14:
            hi = t
        elif a[idx] < -1e-14:
            lo = t
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, deadline, max_sweeps=100000, verbose=False):
    """Coordinate descent with 1D minimax subproblems."""
    h = h.copy()
    dx = 2.0 / n
    c_vec = compute_c5_vec(h, n)
    best_c = float(c_vec.max())
    best_h = h.copy()
    no_improve = 0
    PATIENCE = 200
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            a = compute_a_fast(j, h, n, dx)
            D = c_vec - h[j] * a
            t_new = min_1d_max(D, a)
            h[j] = t_new
        # End of sweep: project and evaluate
        h = project_feasible(h, n)
        c_vec = compute_c5_vec(h, n)
        c_now = float(c_vec.max())
        if c_now < best_c - 1e-12:
            best_c = c_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= PATIENCE and sweep > 50:
                break
    return best_h, best_c
```
