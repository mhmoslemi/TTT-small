import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs=None):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def simulated_annealing_binary(h, n, t_start, t_end, iterations, deadline, seed=42):
    """Simulated annealing on binary flips."""
    rng = np.random.RandomState(seed)
    h_b = h.copy()
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    cur_c = best_c
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        # Geometric cooling
        frac = it / max(iterations - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        j = rng.randint(n)
        old = h_b[j]
        h_b[j] = 1.0 - old
        # Need to maintain sum constraint: this breaks it!
        # For SA, we should use swaps instead
        h_b[j] = old  # revert
        
        # Use swap: pick a random 1 and random 0, swap them
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = ones[rng.randint(len(ones))]
        m = zeros[rng.randint(len(zeros))]
        if j == m:
            continue
        h_b[j] = 0
        h_b[m] = 1
        new_c = compute_c5(h_b, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 20000.0)]

        if time.time() > deadline - 8:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def to_binary(h, n):
    h_b = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        ones_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):
            h_b[ones_idx[order[i]]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - n_ones):
            h_b[zeros_idx[order[i]]] = 1
    return h_b

def binary_greedy_search(h, n, deadline, max_rounds=2000):
    h_b = h.copy()
    best_c = compute_c5(h_b, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_improvement = 0.0
        best_j = -1
        for j in range(n):
            old = h_b[j]
            h_b[j] = 1.0 - old
            c = compute_c5(h_b, n)
            h_b[j] = old
            imp = best_c - c
            if imp > best_improvement + 1e-15:
                best_improvement = imp
                best_j = j
        if best_j < 0:
            break
        h_b[best_j] = 1.0 - h_b[best_j]
        best_c = compute_c5(h_b, n)
    return h_b, best_c

            if time.time() > deadline:
                return h_b, best_c
            j = int(ones[oi])

        best_h, best_c = lbfgs_optimize(best_h, n_base, t1,
            configs=[(200, 100.0), (1000, 500.0), (5000, 2000.0), (20000, 10000.0)])

    # Phase 2: Binary greedy at base
    h_bin = to_binary(best_h, n_base)
    t2 = min(t0 + 180, deadline - 500)
    if t2 > time.time() + 5:
        h_bin, c_bin = binary_greedy_search(h_bin, n_base, t2)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin

    # Phase 3: Binary pair search at base
    t3 = min(t0 + 240, deadline - 400)
    if t3 > time.time() + 5:
        h_pair, c_pair = binary_pair_search(best_h, n_base, t3, max_rounds=80, window=15)
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair

    # Phase 4: Simulated annealing at base (multiple restarts)
    if time.time() < deadline - 350:
        for sa_seed in range(3):
            t4 = min(time.time() + 40, deadline - 150 - sa_seed * 40)
            if t4 > time.time() + 5:
                h_sa, c_sa = sa_binary(best_h, n_base, 0.05, 1e-5, 50000, t4, seed=seed + sa_seed * 77)
                if c_sa < best_c:
                    best_c = c_sa
                    best_h = h_sa

    # Phase 5: Interpolate to n=1000, optimize + binary search
    n1 = 1000
    if time.time() < deadline - 250:
        h1000 = interpolate_to(best_h, len(best_h), n1)
        t5 = min(time.time() + 60, deadline - 170)
        if t5 > time.time() + 5:
            h1000, c1000 = lbfgs_optimize(h1000, n1, t5,
                configs=[(500, 200.0), (2000, 1000.0), (8000, 4000.0)])
            t5b = min(time.time() + 50, deadline - 100)
            if t5b > time.time() + 5:
                h1000_bin = to_binary(h1000, n1)
                h1000_bin, c1000b = binary_greedy_search(h1000_bin, n1, t5b)
                if c1000b < c1000:
                    c1000 = c1000b
                    h1000 = h1000_bin
            if c1000 < best_c:
                best_c = c1000
                best_h = h1000

    # Phase 6: Interpolate to n=2000, optimize + binary
    n2 = 2000
    if time.time() < deadline - 150:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        t6 = min(time.time() + 50, deadline - 90)
        if t6 > time.time() + 5:
            h2000, c2000 = lbfgs_optimize(h2000, n2, t6,
                configs=[(1000, 500.0), (5000, 2000.0)])
            t6b = min(time.time() + 40, deadline - 40)
            if t6b > time.time() + 5:
                h2000_bin = to_binary(h2000, n2)
                h2000_bin, c2000b = binary_greedy_search(h2000_bin, n2, t6b)
                if c2000b < c2000:
                    c2000 = c2000b
                    h2000 = h2000_bin
            if c2000 < best_c:
                best_c = c2000
                best_h = h2000

    # Phase 7: Multi-start from structured inits
    if time.time() < deadline - 80:
        for ms in range(5):
            if time.time() > deadline - 25:
                break
            h_ms = generate_structured_init(n_base, seed=seed + ms * 200 + 13)
            t7 = min(time.time() + 20, deadline - 25)
            if t7 > time.time() + 4:
                h_ms, c_ms = lbfgs_optimize(h_ms, n_base, t7,
                    configs=[(500, 200.0), (3000, 1500.0)])
                h_ms_bin = to_binary(h_ms, n_base)
                t7b = min(time.time() + 12, deadline - 22)
                if t7b > time.time() + 3:
                    h_ms_bin, c_ms = binary_greedy_search(h_ms_bin, n_base, t7b)
                if c_ms < best_c:
                    best_c = c_ms
                    best_h = h_ms_bin

    # Phase 8: Interpolate best to n=2000 one more time if time
    if time.time() < deadline - 50 and len(best_h) < 2000:
        h_fin = interpolate_to(best_h, len(best_h), 2000)
        t8 = min(time.time() + 30, deadline - 15)
        if t8 > time.time() + 5:
            h_fin, c_fin = lbfgs_optimize(h_fin, 2000, t8,
                configs=[(2000, 1000.0), (8000, 4000.0)])
            h_fin_bin = to_binary(h_fin, 2000)
            t8b = min(time.time() + 15, deadline - 10)
            if t8b > time.time() + 3:
                h_fin_bin, c_fin2 = binary_greedy_search(h_fin_bin, 2000, t8b)
                if c_fin2 < c_fin:
                    c_fin = c_fin2
                    h_fin = h_fin_bin
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin

    # Phase 9: Final targeted refinement + SLSQP
    if time.time() < deadline - 15:
        h_t, c_t = targeted_local_refine(best_h, len(best_h), deadline - 8)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    if time.time() < deadline - 5:
        h_s, c_s = slqp_polish(best_h, len(best_h), deadline - 2, alpha=10000000, maxiter=500)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
