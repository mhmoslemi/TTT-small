def smooth_random_initial(n, rng, amp=0.35, width_frac=0.2):
    kernel_width = max(2, int(n * width_frac * 0.5))

import numpy as np
import time
from scipy.optimize import minimize, linprog

try:
    from scipy.signal import fftconvolve as _fftconvolve
except Exception:
    _fftconvolve = None


def eval_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def compute_c5(h):
    return float(np.max(eval_values(h)))


def add_uniform(h, amount):
    h = np.array(h, dtype=float, copy=True)
    if amount <= 1e-15:
        return np.clip(h, 0.0, 1.0)
    room = 1.0 - h
    total = float(room.sum())
    if total <= amount + 1e-14:
        h = np.where(room > 0.0, 1.0, h)
        return h
    lo = 0.0
    hi = float(room.max())
    if hi <= 0.0:
        hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.minimum(room, mid).sum()) < amount:
            lo = mid
        else:
            hi = mid
    a = 0.5 * (lo + hi)
    return np.minimum(1.0, h + a)


def reduce_uniform(h, amount):
    h = np.array(h, dtype=float, copy=True)
    if amount <= 1e-15:
        return np.clip(h, 0.0, 1.0)
    pos = h
    total = float(pos.sum())
    if total <= amount + 1e-14:
        h = np.where(pos > 0.0, 0.0, h)
        return h
    lo = 0.0
    hi = float(pos.max())
    if hi <= 0.0:
        hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.minimum(pos, mid).sum()) < amount:
            lo = mid
        else:
            hi = mid
    b = 0.5 * (lo + hi)
    return np.maximum(0.0, h - b)


def project_feasible(h):
    n = len(h)
    target = 0.5 * float(n)
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if diff > 1e-13:
        h = add_uniform(h, diff)
    elif diff < -1e-13:
        h = reduce_uniform(h, -diff)

    for _ in range(20):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(1.0 - float(h[idx]), diff)
            if add <= 0.0:
                h = add_uniform(h, diff)
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(float(h[idx]), -diff)
            if sub <= 0.0:
                h = reduce_uniform(h, -diff)
                break
            h[idx] -= sub

    return np.clip(h, 0.0, 1.0)


def fast_convolve(a, b):
    la = len(a)
    lb = len(b)
    if _fftconvolve is not None and la * lb > 1000000:
        try:
            return np.asarray(_fftconvolve(a, b, mode="full"), dtype=float)
        except Exception:
            pass
    return np.asarray(np.convolve(a, b, mode="full"), dtype=float)


def make_weights(f, alpha):
    m = float(np.max(f))
    sh = alpha * (f - m)
    mx = float(np.max(sh))
    w = np.exp(sh - mx)
    s = float(w.sum())
    if s > 0.0:
        w /= s
    else:
        w = np.ones_like(f, dtype=float)
        w /= float(len(f))
    return w


def conv_weight_gradient(h, w):
    n = len(h)
    center = n - 1

    wp = np.zeros(n, dtype=float)
    wp[1:] = w[0:center][::-1]

    wn = np.zeros(n, dtype=float)
    wn[0] = w[center]
    wn[1:] = w[center + 1:]

    g = np.zeros(n, dtype=float)

    if np.any(wp):
        suff = np.cumsum(wp[::-1])[::-1]
        qf = fast_convolve(wp, h[::-1])
        fwd = qf[center::-1]
        cv = fast_convolve(wp, h)
        conv = cv[:n]
        g += suff - fwd - conv

    if np.any(wn):
        cum = np.cumsum(wn)
        qf = fast_convolve(wn, h[::-1])
        fwd = qf[center::-1]
        cv = fast_convolve(wn, h)
        conv = cv[:n]
        g += cum - conv - fwd

    return g


def weighted_lag_gradient(h, w):
    n = len(h)
    active = np.where(w > 1e-12)[0]
    max_active = max(8, min(64, n // 16))

    if active.size <= max_active:
        sum_active = float(w[active].sum()) if active.size else 0.0
        if sum_active > 0.99:
            g = np.zeros(n, dtype=float)
            center = n - 1
            for idx in active:
                z = float(w[idx])
                if z <= 0.0:
                    continue
                k = int(idx) - center
                if k == 0:
                    g += z * (1.0 - 2.0 * h)
                elif k > 0:
                    nv = n - k
                    g[k:n] += z * (1.0 - h[:nv])
                    g[:nv] -= z * h[k:n]
                else:
                    q = -k
                    nv = n - q
                    g[:nv] += z * (1.0 - h[q:n])
                    g[q:n] -= z * h[:nv]
            return g

    return conv_weight_gradient(h, w)


def smooth_objective(x, n, alpha, mu):
    f = eval_values(x)
    m = float(np.max(f))
    sh = alpha * (f - m)
    mx = float(np.max(sh))
    se = sh - mx
    lse = mx + float(np.log(np.sum(np.exp(se))))
    diff = float(np.sum(x) - 0.5 * float(n))
    return float(m + lse / alpha + mu * diff * diff)


def smooth_gradient(x, n, alpha, mu):
    f = eval_values(x)
    w = make_weights(f, alpha)
    g = weighted_lag_gradient(x, w)
    dx = 2.0 / float(n)
    g *= dx
    g += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return g


def lbfgs_refine(h, deadline, alphas, maxiters, mus):
    h = project_feasible(h)
    n = len(h)
    best_c = compute_c5(h)
    if not np.isfinite(best_c):
        best_c = 1.0e100
    bounds = [(0.0, 1.0)] * n

    for j in range(len(alphas)):
        if time.time() > deadline - 2.0:
            break
        alpha = float(alphas[j])
        mi = int(maxiters[j]) if j < len(maxiters) else int(maxiters[-1])
        mu = float(mus[j]) if j < len(mus) else float(mus[-1])
        gtol = 1e-5 if alpha < 500.0 else 1e-6
        maxfun = max(120, mi * 12)

        try:
            res = minimize(
                smooth_objective,
                h,
                jac=smooth_gradient,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": mi,
                    "ftol": 1e-11,
                    "gtol": gtol,
                    "maxfun": maxfun,
                    "maxls": 8,
                },
            )
            if res is not None and hasattr(res, "x") and res.x is not None:
                xnew = np.asarray(res.x, dtype=float)
                if xnew.shape == (n,) and np.all(np.isfinite(xnew)):
                    h_new = project_feasible(xnew)
                else:
                    h_new = h
            else:
                h_new = h
        except Exception:
            h_new = h

        if np.all(np.isfinite(h_new)):
            c_new = compute_c5(h_new)
            if np.isfinite(c_new) and c_new < best_c:
                best_c = c_new
                h = h_new.copy()

    return h, best_c


def slqp_obj(x, n, alpha):
    return smooth_objective(x, n, alpha, 0.0)


def slqp_grad(x, n, alpha):
    return smooth_gradient(x, n, alpha, 0.0)


def slqp_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def slqp_eq_jac(x, n):
    return np.ones(n, dtype=float)


def slqp_polish(h, alpha, maxiter):
    n = len(h)
    h = project_feasible(h)
    best_c = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slqp_eq, "jac": slqp_eq_jac, "args": (n,)}]
    try:
        res = minimize(
            slqp_obj,
            h,
            jac=slqp_grad,
            args=(n, alpha),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "iprint": 0},
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            h_new = project_feasible(res.x)
            c_new = compute_c5(h_new)
            if np.isfinite(c_new) and c_new < best_c:
                return h_new, c_new
    except Exception:
        pass
    return h, best_c


def resample_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return project_feasible(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)


def smooth_noise(n, rng, width_frac=0.18):
    kern_w = max(2, int(n * max(0.02, width_frac) * 0.5))
    k = np.arange(-kern_w, kern_w + 1)
    sig = max(1.0, kern_w / 2.0)
    ker = np.exp(-0.5 * (k / sig) ** 2)
    ker /= float(ker.sum())
    noise = rng.standard_normal(n)
    return np.convolve(noise, ker, mode="same")


def smooth_random_initial(n, rng, amp=0.35, width_frac=0.20):
    h = 0.5 + amp * smooth_noise(n, rng, width_frac=width_frac)
    return project_feasible(h)


def random_perturb(h, sigma, rng, width_frac=0.12):
    if sigma <= 0.0:
        return project_feasible(h)
    return project_feasible(h + float(sigma) * smooth_noise(len(h), rng, width_frac=width_frac))


def binary_push(h):
    hb = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = 0.5 * float(len(hb))
    diff = target - float(hb.sum())

    if diff > 1e-12:
        order = np.argsort(-h)
        for i in order:
            if diff <= 1e-12:
                break
            if hb[i] < 1.0:
                add = min(1.0 - float(hb[i]), diff)
                hb[i] += add
                diff -= add
    elif diff < -1e-12:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            if hb[i] > 0.0:
                sub = min(float(hb[i]), -diff)
                hb[i] -= sub
                diff += sub

    return project_feasible(hb)


def single_lag_gradient(h, idx):
    n = len(h)
    center = n - 1
    k = int(idx) - center
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)

    if k == 0:
        g = (1.0 - 2.0 * h) * dx
    elif k > 0:
        nv = n - k
        g[k:n] += 1.0 - h[:nv]
        g[:nv] -= h[k:n]
        g *= dx
    else:
        q = -k
        nv = n - q
        g[:nv] += 1.0 - h[q:n]
        g[q:n] -= h[:nv]
        g *= dx

    return g


def lp_refine(h, deadline, trust_levels=(0.03, 0.10, 0.30, 0.0), max_rounds=8, topk=32):
    h = project_feasible(h)
    n = len(h)
    best_c = compute_c5(h)
    if not np.isfinite(best_c):
        best_c = 1.0e100

    if topk is None:
        topk = max(16, min(64, n // 32))
    K = int(max(8, min(int(topk), max(8, n // 16))))

    for trust in trust_levels:
        for _round in range(int(max_rounds)):
            if time.time() > deadline:
                break

            f = eval_values(h)
            C = float(np.max(f))
            if C < best_c:
                best_c = C

            m = len(f)
            kk = min(K, m)
            if kk <= 0:
                break
            if kk >= m:
                active = np.arange(m)
            else:
                active = np.argpartition(f, m - kk)[-kk:]

            A_rows = []
            b_list = []

            for idx in active:
                ix = int(idx)
                gi = single_lag_gradient(h, ix)
                f0 = float(f[ix])
                row = np.empty(n + 1, dtype=float)
                row[:n] = gi
                row[-1] = -1.0
                A_rows.append(row)
                b_list.append(f0 - float(np.dot(gi, h)))

            if not A_rows:
                break

            A_ub = np.vstack(A_rows)
            b_ub = np.asarray(b_list, dtype=float)

            A_eq = np.zeros((1, n + 1), dtype=float)
            A_eq[0, :n] = 1.0
            b_eq = np.array([0.5 * float(n)], dtype=float)

            c_obj = np.zeros(n + 1, dtype=float)
            c_obj[-1] = 1.0

            if trust is None or trust <= 0.0:
                bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
            else:
                epsb = 1e-9
                lo = np.maximum(0.0, h - float(trust) - epsb)
                hi = np.minimum(1.0, h + float(trust) + epsb)
                bounds = list(zip(lo, hi)) + [(-1.0, 1.0)]

            try:
                res = linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                    disp=False,
                )
            except Exception:
                try:
                    res = linprog(
                        c_obj,
                        A_ub=A_ub,
                        b_ub=b_ub,
                        A_eq=A_eq,
                        b_eq=b_eq,
                        bounds=bounds,
                        disp=False,
                    )
                except Exception:
                    break

            if res is None or not getattr(res, "success", False):
                break

            xlp = getattr(res, "x", None)
            if xlp is None or xlp.shape[0] < n or not np.all(np.isfinite(xlp[:n])):
                break

            hlp = np.clip(xlp[:n], 0.0, 1.0)
            hlp = project_feasible(hlp)

            improved = False
            for theta in (1.0, 0.5, 0.25, 0.10):
                hc = (1.0 - theta) * h + theta * hlp
                hc = project_feasible(hc)
                cval = compute_c5(hc)
                if np.isfinite(cval) and cval < best_c - 1e-14:
                    best_c = cval
                    h = hc.copy()
                    improved = True
                    break

            if not improved:
                break

        if time.time() > deadline:
            break

    best_c = min(compute_c5(h), best_c)
    return h, best_c


def pair_refine(h, deadline, max_iters=300, seed=20240524):
    h = project_feasible(h)
    n = len(h)
    best_c = compute_c5(h)
    rng = np.random.RandomState(seed)

    for _it in range(int(max_iters)):
        if time.time() > deadline:
            break

        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue

        hi = float(h[i])
        hj = float(h[j])
        d0 = 0.5 * (hj - hi)

        deltas = (d0, 0.25 * d0, 0.75 * d0, 0.01, -0.01, 0.005, -0.005)
        improved = False

        for d in deltas:
            ni = hi + float(d)
            nj = hj - float(d)
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i] = ni
            h[j] = nj
            cval = compute_c5(h)
            if np.isfinite(cval) and cval < best_c - 1e-14:
                best_c = cval
                improved = True
                break
            h[i] = hi
            h[j] = hj

        if not improved:
            continue

    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s) if budget_s is not None else 1100.0
    if B < 60.0:
        B = 60.0
    deadline = t0 + B - 20.0
    if deadline < t0 + 30.0:
        deadline = t0 + max(30.0, B - 5.0)

    rng = np.random.RandomState(int(seed) if seed is not None else 42)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.ndim != 1 or len(h0) == 0 or not np.all(np.isfinite(h0)):
            raise ValueError("bad parent")
    except Exception:
        h0 = smooth_random_initial(512, rng, amp=0.40)

    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(512, 0.5))
        best_c = compute_c5(best_h)

    if B >= 700.0:
        n_mid = 1024
        n_high = 2048
    elif B >= 350.0:
        n_mid = 1024
        n_high = 1024
    else:
        n_mid = max(512, len(best_h))
        n_high = n_mid

    for n_test in (1024, 2048):
        if time.time() > deadline:
            break
        if n_test > len(best_h) and n_test <= n_high:
            cand = resample_to(best_h, n_test)
            cval = compute_c5(cand)
            if np.isfinite(cval) and cval < best_c:
                best_h = cand
                best_c = cval

    work_n = min(n_high, max(n_mid, len(best_h)))
    if work_n != len(best_h):
        work_h = resample_to(best_h, work_n)
    else:
        work_h = best_h.copy()

    work_c = compute_c5(work_h)
    if work_c < best_c:
        best_h = work_h.copy()
        best_c = work_c

    d1 = t0 + 0.11 * B
    d2 = t0 + 0.32 * B
    d3 = t0 + 0.56 * B
    d4 = t0 + 0.78 * B
    d5 = deadline

    if time.time() < d1:
        work_h, work_c = lbfgs_refine(
            work_h,
            d1,
            alphas=(60.0, 200.0, 800.0, 3000.0, 10000.0),
            maxiters=(60, 80, 90, 90, 70),
            mus=(500.0, 5000.0, 30000.0, 150000.0, 400000.0),
        )
        if work_c < best_c:
            best_h = work_h.copy()
            best_c = work_c

    if time.time() < d2 - 30.0:
        work_h, work_c = lp_refine(
            work_h,
            min(d2 - 30.0, time.time() + 50.0),
            trust_levels=(0.05, 0.15),
            max_rounds=6,
            topk=28,
        )
        if work_c < best_c:
            best_h = work_h.copy()
            best_c = work_c

    if time.time() < d2 - 10.0:
        trials = [
            work_h.copy(),
            smooth_random_initial(len(work_h), rng, amp=0.35),
            random_perturb(work_h, 0.05, rng),
            binary_push(work_h),
        ]

        for tri in trials:
            if time.time() > d2 - 12.0:
                break
            htri = project_feasible(np.array(tri, dtype=float, copy=True))
            trial_end = min(d2, time.time() + 55.0)
            if trial_end <= time.time() + 10.0:
                break

            htri, ctri = lbfgs_refine(
                htri,
                trial_end,
                alphas=(300.0, 1500.0, 6000.0),
                maxiters=(40, 60, 60),
                mus=(20000.0, 100000.0, 400000.0),
            )

            if time.time() < trial_end - 5.0:
                htri, ctri = lp_refine(
                    htri,
                    trial_end,
                    trust_levels=(0.05, 0.15),
                    max_rounds=5,
                    topk=26,
                )

            if ctri < best_c:
                best_h = htri.copy()
                best_c = ctri
            if ctri <= work_c + 1e-7:
                work_h = htri
                work_c = ctri

    if n_high > len(work_h) and time.time() < d3 - 30.0:
        hhigh = resample_to(work_h, n_high)
        chigh = compute_c5(hhigh)
        if chigh < best_c:
            best_h = hhigh.copy()
            best_c = chigh

        hhigh, chigh = lbfgs_refine(
            hhigh,
            d3,
            alphas=(200.0, 1000.0, 4000.0, 15000.0),
            maxiters=(70, 90, 110, 80),
            mus=(5000.0, 25000.0, 100000.0, 400000.0),
        )
        if chigh < best_c:
            best_h = hhigh.copy()
            best_c = chigh

        if time.time() < d3:
            hhigh, chigh = lp_refine(
                hhigh,
                d3,
                trust_levels=(0.03, 0.10, 0.30),
                max_rounds=8,
                topk=40,
            )
            if chigh < best_c:
                best_h = hhigh.copy()
                best_c = chigh

        work_h = hhigh
        work_c = chigh
    else:
        hhigh = work_h
        chigh = work_c

    if n_high > len(best_h) and time.time() < d4 - 30.0:
        htrial = random_perturb(hhigh, 0.02, rng)
        t_end = min(d4, time.time() + 90.0)
        htrial, ctrial = lbfgs_refine(
            htrial,
            t_end,
            alphas=(1000.0, 4000.0, 15000.0),
            maxiters=(50, 70, 70),
            mus=(50000.0, 200000.0, 700000.0),
        )
        if time.time() < t_end:
            htrial, ctrial = lp_refine(
                htrial,
                t_end,
                trust_levels=(0.05, 0.20),
                max_rounds=6,
                topk=36,
            )
        if ctrial < best_c:
            best_h = htrial.copy()
            best_c = ctrial
            hhigh = htrial.copy()
        elif ctrial < chigh + 1e-6:
            hhigh = htrial.copy()
            chigh = ctrial

    if n_high > len(best_h) and time.time() < d4 - 30.0:
        htrial = binary_push(hhigh)
        t_end = min(d4, time.time() + 70.0)
        htrial, ctrial = lbfgs_refine(
            htrial,
            t_end,
            alphas=(500.0, 2500.0, 10000.0),
            maxiters=(40, 60, 60),
            mus=(20000.0, 100000.0, 400000.0),
        )
        if time.time() < t_end:
            htrial, ctrial = lp_refine(
                htrial,
                t_end,
                trust_levels=(0.05, 0.20),
                max_rounds=5,
                topk=32,
            )
        if ctrial < best_c:
            best_h = htrial.copy()
            best_c = ctrial
            hhigh = htrial.copy()
        elif ctrial < chigh + 1e-6:
            hhigh = htrial.copy()
            chigh = ctrial

    if len(best_h) != n_high and time.time() < deadline - 90.0:
        hfinal = resample_to(best_h, n_high)
    else:
        hfinal = best_h.copy()

    cf = compute_c5(hfinal)
    if cf < best_c:
        best_h = hfinal.copy()
        best_c = cf

    if time.time() < deadline - 70.0:
        hfinal, cf = lbfgs_refine(
            hfinal,
            deadline - 50.0,
            alphas=(5000.0, 20000.0),
            maxiters=(40, 50),
            mus=(200000.0, 800000.0),
        )
        if cf < best_c:
            best_h = hfinal.copy()
            best_c = cf

    if time.time() < deadline - 40.0:
        hfinal, cf = lp_refine(
            hfinal,
            deadline - 15.0,
            trust_levels=(0.02, 0.08, 0.25, 0.0),
            max_rounds=10,
            topk=48,
        )
        if cf < best_c:
            best_h = hfinal.copy()
            best_c = cf

    if time.time() < deadline - 25.0 and len(best_h) >= 512:
        maxp = 250 if len(best_h) > 1500 else 450
        hp, cp = pair_refine(best_h, deadline - 8.0, max_iters=maxp)
        if cp < best_c:
            best_h = hp
            best_c = cp

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(512, 0.5))
        best_c = compute_c5(best_h)

    return best_h, float(best_c), int(len(best_h))
