```python
def solve_1d_minmax(slopes, intercepts, lb=0.0, ub=1.0):
    """
    Find v in [lb, ub] that minimizes max_k (slopes[k]*v + intercepts[k]).
    
    Uses binary search on the target value t.
    """
    n = len(slopes)
    
    # Check if all slopes are 0
    if np.all(slopes == 0):
        return (lb + ub) / 2
    
    # Binary search on t
    # Lower bound on t: min_k intercepts[k] + slopes[k]*min(0, ...) hmm
    # Just use the value at the midpoint as upper bound
    mid = (lb + ub) / 2
    t_hi = np.max(slopes * mid + intercepts)
    t_lo = np.min(slopes * (lb + ub)/2 + intercepts) - np.max(np.abs(slopes)) * (ub-lb)
    
    for _ in range(60):  # 60 iterations gives ~1e-18 precision
        t_mid = (t_lo + t_hi) / 2
        if feasible(t_mid, slopes, intercepts, lb, ub):
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    # Now find v in [lb, ub] such that slopes[k]*v + intercepts[k] ≤ t_hi for all k
    # v ≤ (t_hi - intercepts[k]) / slopes[k] for slopes[k] > 0
    # v ≥ (t_hi - intercepts[k]) / slopes[k] for slopes[k] < 0
    pos = slopes > 1e-15
    neg = slopes < -1e-15
    zero = ~(pos | neg)
    
    v_lo = lb
    v_hi = ub
    if np.any(pos):
        v_hi = min(v_hi, np.min((t_hi - intercepts[pos]) / slopes[pos]))
    if np.any(neg):
        v_lo = max(v_lo, np.max((t_hi - intercepts[neg]) / slopes[neg]))
    
    if v_lo > v_hi:
        # Numerical issue, return midpoint
        return (lb + ub) / 2
    return (v_lo + v_hi) / 2

def feasible(t, slopes, intercepts, lb, ub):
    pos = slopes > 1e-15
    neg = slopes < -1e-15
    zero = ~(pos | neg)
    
    if np.any(zero) and np.any(intercepts[zero] > t):
        return False
    
    v_lo = lb
    v_hi = ub
    if np.any(pos):
        v_hi = min(v_hi, np.min((t - intercepts[pos]) / slopes[pos]))
    if np.any(neg):
        v_lo = max(v_lo, np.max((t - intercepts[neg]) / slopes[neg]))
    
    return v_lo <= v_hi + 1e-12
```
