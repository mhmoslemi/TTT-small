    h = np.zeros(n)
    h[:n//2] = h_free
    h[n//2:] = h_free[::-1]

    return h[:n//2].copy()

    """Objective for symmetric optimization."""
    h = expand_symmetric(x, n)
    c = compute_c5_vec(h, n)
    m = c.max()
    ac = alpha * (c - m)
    am = ac.max()
    se = np.exp(ac - am).sum()
    val = m + (am + np.log(se)) / alpha
    val += mu * (2.0 * x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(h, n)

                    grad[0:nv] += wm * (-best_h[kk:n])
        
        # Projected gradient step with backtracking
        lr = 0.01

            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 1e-15:
                best_h = h_new
                best_c = c_new
                break
            lr *= 0.5
        if lr < 1e-10:
            break
    
    return best_h, best_c

    best_h = h.copy()

    n_changes = 0

                    break
                else:

            if made_change:
                break
        if not made_change:
            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def expand_symmetric(x, n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = x
    h[half:] = x[::-1]
    return h

def project_feasible_sym(h, n):
    h = 0.5 * (h + h[::-1].copy())
    return project_feasible(h, n)

def _smax_val(c, alpha):
    m = c.max()
    ac = alpha * (c - m)
    am = ac.max()
    se = np.exp(ac - am).sum()
    return m + (am + np.log(se)) / alpha, w_from_ac(ac, am)

def w_from_ac(ac, am):
    w = np.exp(ac - am)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(ac)] = 1.0
    else:
        w /= s
    return w

def _add_shift_grad(grad, h, n, idx, wm):
    center = n - 1
    k = idx - center
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def sym_smooth_obj(x, n, alpha, mu):
    h = expand_symmetric(x, n)
    c = compute_c5_vec(h, n)
    val = _smax_val(c, alpha)[0]
    val += mu * (2.0 * x.sum() - n / 2.0) ** 2
    return float(val)

def sym_smooth_grad(x, n, alpha, mu):
    h = expand_symmetric(x, n)
    c = compute_c5_vec(h, n)
    m = c.max()
    ac = alpha * (c - m)
    am = ac.max()
    w = w_from_ac(ac, am)
    dx = 2.0 / n
    half = n // 2
    grad_full = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        _add_shift_grad(grad_full, h, n, idx, w[idx] * dx)
    grad_free = grad_full[:half].copy()
    grad_free += grad_full[half:][::-1]
    grad_free += 4.0 * mu * (2.0 * x.sum() - n / 2.0)
    return grad_free

def full_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = _smax_val(c, alpha)[0]
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)
