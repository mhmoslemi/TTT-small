def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)

def de_step(pop, fitness, n, F, CR, deadline):
    """One generation of differential evolution"""
    NP = len(pop)
    new_pop = pop.copy()
    for i in range(NP):
        # Select 3 distinct random indices
        idxs = list(range(NP))
        idxs.remove(i)
        a, b, c = np.random.choice(idxs, 3, replace=False)
        # Mutation
        v = pop[a] + F * (pop[b] - pop[c])
        # Crossover
        r = np.random.random(n)
        jrand = np.random.randint(n)
        trial = np.where(r < CR, v, pop[i])
        trial[jrand] = v[jrand]
        trial = project_feasible(trial, n)
        f_trial = compute_c5(trial, n)
        if f_trial <= fitness[i]:
            new_pop[i] = trial
            # Update fitness externally
            return new_pop, i, f_trial
    return new_pop, -1, None

def coordinate_descent(h, n, deadline, n_rounds=5):
    """Cyclic coordinate descent with backtracking"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                    break
            # Optimize h[i] while keeping others fixed
            # c_k(h) is quadratic in h[i] for each k
            # We just do a 1D search
            old_val = best_h[i]
            if old_val <= 1e-8 or old_val >= 1.0 - 1e-8:
                continue
            # Try a few values
            for trial in [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          1.0 - old_val * 0.5, 1.0 - old_val * 0.75,
                          old_val + 0.1, old_val - 0.1, old_val + 0.05, old_val - 0.05]:
                if trial < 0 or trial > 1:
                    continue
                best_h[i] = trial
                # Re-project to maintain sum constraint (approximate)
                h_test = project_feasible(best_h, n)
                c_test = compute_c5(h_test, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h_test
                    break
            best_h[i] = old_val
            # Re-project
            best_h = project_feasible(best_h, n)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    return grad

def nesterov_subgrad(h, n, deadline, n_iter=20000):
    """Nesterov-accelerated subgradient on the max of c_k"""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    y = best_h.copy()
    t_nest = 1.0
    dx_val = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        alpha_nest = (t_nest - 1.0) / t_new
        t_nest = t_new
        y_new = y + alpha_nest * (y - best_h)
        y_new = np.clip(y_new, 0.0, 1.0)
        # Subgradient of max at y_new
        c_vec = compute_c5_vec(y_new, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - y_new[k:n]) * dx_val
                grad[k:n] += (-y_new[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - y_new[0:nv]) * dx_val
                grad[0:nv] += (-y_new[kk:n]) * dx_val
        grad /= len(active)
        # Step size (diminishing)
        eta = 0.05 / (1.0 + 0.1 * np.sqrt(it + 1.0))
        x_new = y_new - eta * grad
        x_new = project_feasible(x_new, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        y = y_new
    return best_h, best_c

            bounds = [(0.0, 1.0)] * best_n
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(best_n, active, 500000000, 0.0001),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-22,
                                        'gtol': 1e-16, 'maxfun': 500,
                                        'maxls': 25, 'maxcor': 25})
                h_f = project_feasible(res.x, best_n)
                c_f = compute_c5(h_f, best_n)
                if c_f < best_c:
                    best_c = c_f
                    best_h = h_f
            except Exception:
                pass

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
