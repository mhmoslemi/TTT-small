def run_active_pass(h, n, top_k, alpha, deadline, maxiter=5000):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-9, 0.01 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c_vec[active])[:top_k]]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(active_obj, h, jac=active_grad,
                       args=(n, active, alpha), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 8, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def subgradient_exact(h, n, deadline, base_step=0.005, patience_limit=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = base_step
    patience = 0
    num_active = 1
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - 1e-12
        active = np.where(c_vec >= threshold)[0]
        if len(active) > num_active:
            active = active[np.argsort(-c_vec[active])[:num_active]]
        
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad(h, n, idx)
        grad /= len(active)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.03)
            patience = 0
        else:
            patience += 1
            step *= 0.85
            if patience > patience_limit:
                num_active = min(num_active + 2, 30)
                step = base_step * 0.5
                patience = 0
            if step < 1e-9:
                break
    return best_h, best_c

def pair_search_fine(h, n, deadline, max_radius=50):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def sharpen_and_reopt(h, n, deadline):
    """Push h toward binary, then re-optimize."""
    # Sharpen: values close to 0 become 0, close to 1 become 1
    h_sharp = h.copy()
    h_sharp[h_sharp < 0.15] = 0.0
    h_sharp[h_sharp > 0.85] = 1.0
    h_sharp = project_feasible(h_sharp, n)
    c_sharp = compute_c5(h_sharp, n)
    
    # Moderate sharpen
    h_mod = h.copy()
    h_mod[h_mod < 0.3] = 0.0
    h_mod[h_mod > 0.7] = 1.0
    h_mod = project_feasible(h_mod, n)
    c_mod = compute_c5(h_mod, n)
    
    best = h.copy()
    best_c = compute_c5(h, n)
    if c_sharp < best_c:
        best = h_sharp
        best_c = c_sharp
    if c_mod < best_c:
        best = h_mod
        best_c = c_mod
    
    # Re-optimize the best sharpened version
    if time.time() < deadline - 30:
        alpha_pass = 100000
        h2 = run_lbfgs_pass(best, n, alpha_pass, maxiter=3000, deadline=deadline)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best = h2
            best_c = c2
    return best, best_c

def gradient_direction_search(h, n, deadline, steps=500):
    """Follow gradient of active max shifts with adaptive step."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.003
    
    for _ in range(steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 15:
            active = active[np.argsort(-c_vec[active])[:15]]
        
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad(h, n, idx)
        grad /= len(active)
        
        # Find best global step via line search
        best_local_c = best_c
        best_local_step = 0
        for mult in [1.0, 0.7, 0.5, 0.3, 0.1, -0.3, -0.5, -0.7]:
            h_try = h + step * mult * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_local_c:
                best_local_c = c_try
                best_local_step = mult
        
        if best_local_step != 0:
            h = project_feasible(h + step * best_local_step * grad, n)
            if best_local_c < best_c:
                best_c = best_local_c
                best_h = h.copy()
            step = min(step * 1.1, 0.02)
        else:
            step *= 0.8
            if step < 1e-7:
                break
    return best_h, best_c

def slqp_refine(h, n, deadline, alpha=500000):
    def sq(x, nn):
        return float(x.sum() - nn / 2.0)
    
    def sq_jac(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_orig = 400
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # ===== Phase 1: Quick L-BFGS at original resolution =====
    if time.time() < deadline - 600:
        h1 = run_lbfgs_pass(best_h, n_orig, 5000, maxiter=3000)
        h1 = project_feasible(h1, n_orig)
        c1 = compute_c5(h1, n_orig)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # ===== Phase 2: Upsample to n=1000 =====
    n = 1000
    h_work = upsample(best_h, len(best_h), n)
    c_work = compute_c5(h_work, n)
    if c_work < best_c:
        best_c = c_work
        best_h = h_work
    else:
        best_h = h_work
        best_c = c_work
    
    # ===== Phase 3: L-BFGS with increasing alpha at n=1000 =====
    alphas = [1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 200:
            break
        remaining = deadline - time.time() - 150
        if remaining < 10:
            break
        maxiter = min(6000, int(remaining * 200))
        if maxiter < 500:
            maxiter = 500
        h_work = run_lbfgs_pass(h_work, n, alpha, maxiter=maxiter)
        h_work = project_feasible(h_work, n)
        c_work = compute_c5(h_work, n)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # ===== Phase 4: Active-set focused optimization =====
    if time.time() < deadline - 150:
        for top_k in [10, 20, 40]:
            if time.time() > deadline - 100:
                break
            remaining = deadline - time.time() - 80
            if remaining < 20:
                break
            h_work = run_active_pass(h_work, n, top_k, 500000,
                                     deadline - 80, maxiter=4000)
            h_work = project_feasible(h_work, n)
            c_work = compute_c5(h_work, n)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
    
    # ===== Phase 5: Subgradient with exact max =====
    if time.time() < deadline - 200:
        sg_deadline = time.time() + min(120, deadline - time.time() - 180)
        h_work, c_work = subgradient_exact(h_work, n, sg_deadline,
                                           base_step=0.003, patience_limit=60)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # ===== Phase 6: Gradient direction search =====
    if time.time() < deadline - 120:
        gd_deadline = time.time() + min(80, deadline - time.time() - 100)
        h_work, c_work = gradient_direction_search(h_work, n, gd_deadline, steps=800)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # ===== Phase 7: Sharpen + re-optimize =====
    if time.time() < deadline - 120:
        sh_deadline = time.time() + min(80, deadline - time.time() - 100)
        h_sharp, c_sharp = sharpen_and_reopt(h_work, n, sh_deadline)
        if c_sharp < best_c:
            best_c = c_sharp
            best_h = h_sharp.copy()
    
    # ===== Phase 8: Another round of L-BFGS at very high alpha =====
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 50
        if remaining > 20:
            h_work = run_lbfgs_pass(h_work, n, 5000000, maxiter=min(4000, int(remaining * 100)))
            h_work = project_feasible(h_work, n)
            c_work = compute_c5(h_work, n)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
    
    # ===== Phase 9: Fine pair search =====
    if time.time() < deadline - 30:
        ps_deadline = time.time() + min(25, deadline - time.time() - 20)
        h_work, c_work = pair_search_fine(h_work, n, ps_deadline, max_radius=30)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()
    
    # ===== Phase 10: SLSQP final polish =====
    if time.time() < deadline - 15:
        h_final = slqp_refine(h_work, n, deadline - 15, alpha=2000000)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
    
    # ===== Final validation =====
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Also check if original resolution solution is better
    if n_orig != n:
        h_orig_final = project_feasible(best_h_init_orig, n_orig) if 'best_h_init_orig' in dir() else project_feasible(best_h, n)
    
    return best_h, best_c, len(best_h)
