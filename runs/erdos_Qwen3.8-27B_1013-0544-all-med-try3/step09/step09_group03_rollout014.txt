def compute_swap_delta(h, n, k, ones, zeros):
    """Compute best swap delta for shift k."""
    if k >= 0:
        # A[p] = -(1-h[p+k])*I(0<=p<=n-1-k) + h[p-k]*I(k<=p<=n-1)
        A = np.zeros(n)
        A[:n-k] -= (1 - h[k:n])  # p in [0, n-1-k], term: -(1-h[p+k])
        A[k:] += h[:n-k]         # p in [k, n-1], term: +h[p-k]
        
        B = np.zeros(n)
        B[:n-k] += (1 - h[k:n])  # q in [0, n-1-k], term: +(1-h[q+k])
        B[k:] -= h[:n-k]         # q in [k, n-1], term: -h[q-k]
    else:
        kk = -k
        A = np.zeros(n)
        A[kk:] -= (1 - h[:n-kk])  # p in [kk, n-1], term: -(1-h[p-kk])
        A[:n-kk] += h[kk:]        # p in [0, n-1-kk], term: +h[p+kk]
        
        B = np.zeros(n)
        B[kk:] += (1 - h[:n-kk])  # q in [kk, n-1], term: +(1-h[q-kk])
        B[:n-kk] -= h[kk:]        # q in [0, n-1-kk], term: -h[q+kk]
    
    # Find min A among ones, min B among zeros
    A_ones = A[ones]
    B_zeros = B[zeros]
    min_A = np.min(A_ones)
    min_B = np.min(B_zeros)
    p_best = ones[np.argmin(A_ones)]
    q_best = zeros[np.argmin(B_zeros)]
    
    return min_A + min_B, p_best, q_best

```python
def project_simple(h, n):
    """Project onto {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        # Add to elements with room
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += (diff / room.sum()) * room
    elif diff < 0:
        # Remove from elements
        if h.sum() > 1e-14:
            h += (diff / h.sum()) * h
    return np.clip(h, 0.0, 1.0)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_simple(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += (diff / tot) * room
    elif diff < 0:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return project_feasible(h, n)

def even_indices(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    return project_feasible(h, n)

def subgradient_step(h, n, k, dx):
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:n]) * dx
        grad[k:] -= h[:n-k] * dx
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk]) * dx
        grad[:n-kk] -= h[kk:] * dx
    return grad

def subgradient_method(h_init, n, deadline, max_iter=200000):
    h = h_init.copy()
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    h_avg = h.copy()
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = int(np.argmax(corr))
        k = worst_idx - (n - 1)
        current_c = corr[worst_idx] * dx
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        
        grad = subgradient_step(h, n, k, dx)
        eta = 1.0 / np.sqrt(t) * 2.0
        
        h = h - eta * grad
        h = project_simple(h, n)
        h_avg = h_avg + (h - h_avg) / t

        best_c = c_curr

def binary_local_search(h_bin, n, deadline, max_rounds=50000):
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    
    stuck_count = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        corr = np.correlate(best_h, 1 - best_h, mode='full')

        zeros = np.where(best_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:

                B[:n-k] += (1.0 - best_h[k:n])

                B[kk:] += (1.0 - best_h[:n-kk])
                B[:n-kk] -= best_h[kk:]

        B_zeros = B[zeros]
        
        found_improvement = False
        # Try top 5 candidates
        try_count = min(5, len(ones))
        A_sorted = np.argsort(A_ones)
        B_sorted = np.argsort(B_zeros)
        
        for ai in range(min(try_count, len(A_ones))):
            if time.time() > deadline:
                return best_h, best_c

                q = zeros[B_sorted[bi]]
                if p == q:
                    continue

                best_h[q] = 1.0
                c_new = compute_c5(best_h, n)

                    best_c = c_new
                    found_improvement = True
                    break
                else:
                    best_h[p] = 1.0
                    best_h[q] = 0.0
            if found_improvement:
                break

            stuck_count += 1
            if stuck_count > 100:
                # Random restart within binary space
                if np.random.random() < 0.3:
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    best_h[p] = 0.0
                    best_h[q] = 1.0
                stuck_count = 0
            elif stuck_count > 500:
                break
        else:
            stuck_count = 0

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):

                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)

        except Exception:

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def coordinate_refine(h, n, deadline, max_rounds=200):

    best_h = h.copy()

            for i in range_idx:
