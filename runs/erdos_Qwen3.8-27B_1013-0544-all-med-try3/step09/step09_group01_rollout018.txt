import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad_part(h, n, k):
    """Compute the gradient of overlap at shift k w.r.t. h. Returns array of length n."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[k:] += (1.0 - h[:nv]) * dx
        g[:nv] += (-h[k:]) * dx
    else:
        kk = -k
        nv = n - kk
        g[:nv] += (1.0 - h[kk:]) * dx
        g[kk:] += (-h[:nv]) * dx
    return g

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    val = np.power(S, 1.0 / p)
    if mu > 0:

def power_grad(x, n, p, mu):

    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiters is not None and i < len(maxiters):

        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-18, 'maxfun': maxit * 10,

                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(power_obj, best_h, jac=power_grad,
                          args=(n, p, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=6, alphas=None, rounds=10):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000]
    
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_lse_obj, best_h, jac=act_lse_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def admm_refine(h, n, deadline, p=64, rho=10.0, n_iter=3000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    z = best_h.copy()
    u = np.zeros(n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def admm_h_obj(x, n, p, rho, z, u):
        r = z - u
        c = compute_c5_vec(x, n)
        c_safe = np.maximum(c, 1e-300)
        S = np.sum(np.power(c_safe, p))
        val = np.power(S, 1.0/p) + (rho/2.0) * np.sum((x - r)**2)
        return float(val)
    
    def admm_h_grad(x, n, p, rho, z, u):
        r = z - u
        c = compute_c5_vec(x, n)
        c_safe = np.maximum(c, 1e-300)
        S = np.sum(np.power(c_safe, p))
        if S < 1e-300:
            w = np.ones(2*n-1)/(2*n-1)
        else:
            factor = np.power(S, 1.0/p - 1.0)
            w = np.power(c_safe, p-1) * factor
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in range(2*n-1):
            if w[mm] < 1e-25:
                continue
            k = mm - (n-1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[:nv] += wm * (1.0 - x[kk:kk+nv])
                grad[kk:kk+nv] += wm * (-x[:nv])
        grad += rho * (x - r)
        return grad
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        remaining_frac = (deadline - time.time()) / max(1, deadline - t0_global)
        cur_rho = rho * (1.0 + 0.5 * it / max(1, n_iter//3))
        
        maxit = max(20, int(50 * remaining_frac) + 20)
        try:
            res = minimize(admm_h_obj, best_h, jac=admm_h_grad,
                          args=(n, p, cur_rho, z, u), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit*8})
            h_new = res.x
        except Exception:
            h_new = best_h.copy()
        
        z_new = project_feasible(h_new + u, n)
        u = u + h_new - z_new
        best_h = project_feasible(h_new, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_sweeps=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = best_h[i]
            for trial in [old_val - 0.01, old_val - 0.001, old_val + 0.001, old_val + 0.01]:
                if trial < 0 or trial > 1:
                    continue
                adj = old_val - trial
                if target - best_h.sum() + adj < -1e-12 or target - best_h.sum() + adj > 1e-12:
                    continue
                best_h[i] = trial
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improved = True
                    old_val = trial
                    break
                else:
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def pair_swap_refine(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if (i - j) % 2 == 0:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def make_periodic(n, pattern):
    h = np.zeros(n)
    p = len(pattern)
    for i in range(n):
        h[i] = pattern[i % p]
    return project_feasible(h, n)

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = int(round(n / 2.0))
    if n % 2 == 0:
        n_ones = n // 2
    else:
        n_ones = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def slsqp_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=200000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_fun, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def subgrad_refine(h, n, deadline, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_exp = 0.0008
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_exp * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:k+nv]) * dx
                grad[k:k+nv] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[:nv] += (1.0 - h[kk:kk+nv]) * dx
                grad[kk:kk+nv] += (-h[:nv]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

t0_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    t0 = t0_global
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(400, 0.5)
        n_orig = 400
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick multi-start at n_orig (120s)
    p1_end = t0 + 120
    starts = [h0.copy()]
    h_inv = 1.0 - h0
    h_inv = project_feasible(h_inv, n_orig)
    starts.append(h_inv)
    
    for period in [3, 4, 5, 7, 8, 12]:
        if n_orig >= period:
            patt = [(1.0 if i < period//2 + period//4 else 0.0) for i in range(period)]
            s = make_periodic(n_orig, patt)
            starts.append(s)
    
    h_rand = rng.random(n_orig)
    h_rand = project_feasible(h_rand, n_orig)
    starts.append(h_rand)
    
    for s in starts:
        if time.time() > p1_end - 5:
            break
        s = project_feasible(s.copy(), n_orig)
        s, c_s = run_power_lbfgs(s, n_orig, min(time.time() + 15, p1_end - 3),
                                  p_list=[8, 32, 128], mu=0.005, maxiters=[300, 500, 500])
        c_s = compute_c5(s, n_orig)
        if c_s < best_c:
            best_c = c_s
            best_h = s
            best_n = n_orig
    
    # Phase 2: Detailed LSE optimization at best_n (280s)
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 280, deadline - 400)
        h2, c2 = run_lse_lbfgs(best_h, best_n, p2_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.008, maxiters=[2000, 2000, 3000, 3000, 3000, 4000, 4000, 5000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Power-mean refinement (150s)
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 150, deadline - 250)
        h3, c3 = run_power_lbfgs(best_h, best_n, p3_end,
            p_list=[16, 32, 64, 128, 256], mu=0.003, maxiters=[2000, 2500, 3000, 3000, 4000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Active-set refinement (120s)
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = run_active_set(best_h, best_n, p4_end, top_k=5,
            alphas=[20000, 100000, 500000, 2000000], rounds=8)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try other resolutions (200s)
    if time.time() < deadline - 250:
        for n_try in [600, 800, 1000]:
            if time.time() > deadline - 220:
                break
            x_fine = np.linspace(0, 2, n_try)
            x_coarse = np.linspace(0, 2, best_n)
            h_fine = np.interp(x_fine, x_coarse, best_h)
            h_fine = project_feasible(h_fine, n_try)
            c_fine = compute_c5(h_fine, n_try)
            
            p_sub_end = min(time.time() + 60, deadline - 200)
            h_f, c_f = run_lse_lbfgs(h_fine, n_try, p_sub_end,
                alphas=[200, 2000, 20000, 100000], mu=0.005,
                maxiters=[1500, 2000, 2500, 3000])
            c_f = compute_c5(h_f, n_try)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
                best_n = n_try
    
    # Phase 6: ADMM (120s)
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 120, deadline - 150)
        h6, c6 = admm_refine(best_h, best_n, p6_end, p=64, rho=20.0, n_iter=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Active-set again (80s)
    if time.time() < deadline - 120:
        p7_end = min(time.time() + 80, deadline - 120)
        h7, c7 = run_active_set(best_h, best_n, p7_end, top_k=4,
            alphas=[50000, 200000, 1000000], rounds=5)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Subgradient (60s)
    if time.time() < deadline - 80:
        p8_end = min(time.time() + 60, deadline - 80)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=2000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Pair swap (80s)
    if time.time() < deadline - 60:
        p9_end = min(time.time() + 80, deadline - 60)
        h9, c9 = pair_swap_refine(best_h, best_n, p9_end, radius=40)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: SLSQP polish (50s)
    if time.time() < deadline - 40:
        for alpha in [100000, 500000]:
            if time.time() > deadline - 35:
                break
            p10_end = min(time.time() + 25, deadline - 30)
            h10, c10 = run_slqp(best_h, best_n, p10_end, alpha=alpha, maxiter=800)
            if c10 < best_c:
                best_c = c10
                best_h = h10
    
    # Phase 11: Final LSE + active (remaining)
    if time.time() < deadline - 15:
        p11_end = deadline - 3
        h11, c11 = run_lse_lbfgs(best_h, best_n, p11_end,
            alphas=[500000, 2000000, 10000000], mu=0.0,
            maxiters=[1000, 1500, 2000])
        if c11 < best_c:
            best_c = c11
            best_h = h11
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
