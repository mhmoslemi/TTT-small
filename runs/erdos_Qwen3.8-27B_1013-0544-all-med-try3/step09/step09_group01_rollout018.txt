def fast_grad(x, n, w):
    # w is the weight vector (length 2n-1)
    dx = 2.0/n
    grad = np.zeros(n)
    # For positive shifts k=1..n-1 (indices n..2n-2 in correlation):
    #   c_{n+k-1} = sum_{j=0}^{n-k-1} x[j]*(1-x[j+k]) * dx
    #   d(c)/dx[i] = (1-x[i+k])*dx for i < n-k, and -x[i-k]*dx for i >= k

    for k in range(1, n):

        wk = w[idx] * dx
        if wk < 1e-14: continue
        nv = n - k
        grad[:nv] += wk * (1.0 - x[k:])
        grad[k:] += wk * (-x[:nv])
    # For negative shifts k=-(n-1)..-1 (indices 0..n-2):
    for k in range(1, n):

        wk = w[idx] * dx
        if wk < 1e-14: continue
        nv = n - k
        grad[k:] += wk * (1.0 - x[:nv])
        grad[:nv] += wk * (-x[k:])
    return grad

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

        else:

            if tot < 1e-14:

def act_grad(x, n, active, alpha, mu):

def _corr_grad(w, x, n, mu):
    """Compute gradient of weighted sum of correlations.
    w has length 2*n-1, indexed by correlation output index.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for k in range(1, n):
        # Positive shift k: correlation index = center + k
        wk = w[center + k] * dx
        if wk > 1e-14:
            nv = n - k
            grad[:nv] += wk * (1.0 - x[k:])
            grad[k:] += wk * (-x[:nv])
        # Negative shift k: correlation index = center - k
        wk = w[center - k] * dx
        if wk > 1e-14:
            nv = n - k
            grad[k:] += wk * (1.0 - x[:nv])
            grad[:nv] += wk * (-x[k:])
    # k=0
    wk0 = w[center] * dx
    if wk0 > 1e-14:
        grad += wk0 * (1.0 - 2.0 * x)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Use progressively smaller steps
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving from j to i
                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    # Try moving from i to j
                    if best_h[j] + step <= 1.0 and best_h[i] - step >= 0.0:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.002, 0.01)
        else:
            eta *= 0.95
        h = h_new
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    def obj(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a
        return m + np.log(np.sum(np.exp(sh))) / a
    def jac(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        return _corr_grad(w, x, n, 0.0)
    
    try:
        res = minimize(obj, h, jac=jac, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def make_periodic_binary(n, period, ones_per_period, rng):

                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.012)
        else:
            eta *= 0.94
        h = h_new
    return best_h, best_c


def pair_search(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(1, n - 1, 3):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    if best_h[j] + step <= 1.0 and best_h[i] - step >= 0.0:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
    return best_h, best_c


def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, jac=slqp_jac, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)


def make_binary_start(h, n, rng):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: L-BFGS at original resolution (~180s)
    p1_deadline = min(t0 + 180, deadline - 400)
    h_p1, c_p1 = lbfgs_optimize(
        best_h, best_n, p1_deadline,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01,
        maxiter_list=[1000, 1500, 2000, 2500, 3000, 3500, 4000, 5000])
    if c_p1 < best_c:
        best_c = c_p1
        best_h = h_p1

    # Phase 2: Active-set at original resolution (~100s)
    p2_deadline = min(time.time() + 100, deadline - 350)
    if time.time() < p2_deadline:
        h_p2, c_p2 = active_optimize(
            best_h, best_n, p2_deadline, top_k=6,
            alphas=[5000, 20000, 100000, 500000, 2000000], mu=0.005)
        if c_p2 < best_c:
            best_c = c_p2
            best_h = h_p2

    # Phase 3: Upsample to n=1000 and optimize (~300s)
    n_hi = 1000
    p3_deadline = min(time.time() + 320, deadline - 200)
    if time.time() < p3_deadline:
        x_fine = np.linspace(0, 2, n_hi, endpoint=False)
        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_hi)
        c_fine = compute_c5(h_fine, n_hi)

        h_p3, c_p3 = lbfgs_optimize(
            h_fine, n_hi, p3_deadline,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
            mu=0.005,
            maxiter_list=[2000, 2500, 3000, 3000, 3500, 4000, 4000])
        best_h = h_p3
        best_c = min(best_c, c_p3)
        best_n = n_hi

    # Phase 4: Active-set at n=1000 (~200s)
    p4_deadline = min(time.time() + 200, deadline - 120)
    if time.time() < p4_deadline:
        h_p4, c_p4 = active_optimize(
            best_h, best_n, p4_deadline, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000], mu=0.003)
        if c_p4 < best_c:
            best_c = c_p4
            best_h = h_p4

    # Phase 5: Subgradient refinement (~80s)
    p5_deadline = min(time.time() + 80, deadline - 60)
    if time.time() < p5_deadline:
        h_p5, c_p5 = subgrad_refine(best_h, best_n, p5_deadline, n_iter=50000)
        if c_p5 < best_c:
            best_c = c_p5
            best_h = h_p5

    # Phase 6: Pair search (~60s)
    p6_deadline = min(time.time() + 60, deadline - 25)
    if time.time() < p6_deadline:
        h_p6, c_p6 = pair_search(best_h, best_n, p6_deadline, radius=35)
        if c_p6 < best_c:
            best_c = c_p6
            best_h = h_p6

    # Phase 7: SLSQP polish (~30s)
    if time.time() < deadline - 20:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 15:
                break
            p7_deadline = min(time.time() + 25, deadline - 10)
            h_p7, c_p7 = slqp_polish(best_h, best_n, p7_deadline,
                                     alpha=alpha, maxiter=1500)
            if c_p7 < best_c:
                best_c = c_p7
                best_h = h_p7

    # Phase 8: Final tiny adjustments (~10s)
    if time.time() < deadline - 2:
        h_p8, c_p8 = pair_search(best_h, best_n, deadline - 1, radius=20)
        if c_p8 < best_c:
            best_c = c_p8
            best_h = h_p8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
