            grad = np.zeros(n)
            for idx in active[:30]:
                lag = idx - (n - 1)
                if lag >= 0:
                    nv = n - lag
                    grad[:nv] += (1.0 - cur_h[lag:])
                    grad[lag:] += -cur_h[:nv]
                else:
                    kk = -lag
                    nv = n - kk
                    grad[kk:] += (1.0 - cur_h[:nv])
                    grad[:nv] += -cur_h[kk:]
            
            pos_mask = grad > 0.03
            neg_mask = grad < -0.03
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            if len(pos_idx) > 0 and len(neg_idx) > 0:
                pos_w = grad[pos_idx] ** 2
                pos_w /= pos_w.sum()
                neg_w = (-grad[neg_idx]) ** 2
                neg_w /= neg_w.sum()
                i = pos_idx[rng.choice(len(pos_idx), p=pos_w)]
                j = neg_idx[rng.choice(len(neg_idx), p=neg_w)]
            else:
                i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = rng.choice(n, size=2, replace=False)
        
        progress = min(1.0, it / 80000)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta < 0.0002:
            continue
        base_delta = 0.015 * (1.0 - progress) + 0.001 * progress
        delta = min(rng.uniform(0.4 * base_delta, 1.6 * base_delta), max_delta * 0.7)
        if delta < 0.0003:
            delta = min(0.0003, max_delta * 0.5)
        
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c = compute_c5(cur_h, n)
        
        if new_c <= cur_c:
            cur_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = cur_h.copy()
                last_best_time = time.time()
        else:
            dc = new_c - cur_c
            if dc < T * 5.0:
                if rng.random() < np.exp(-dc / T):
                    cur_c = new_c
                else:
                    cur_h[i] += delta
                    cur_h[j] -= delta
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
        
        if it % 2000 == 0:
            T *= 0.94
        if T < 0.0003:
            T = T_init * 0.6
            if T < 0.0003:
                break
    
    return best_h, best_c

def fine_sa(h, n, deadline, T_init=0.0008, seed=123):
    rng = np.random.RandomState(seed)
    T = T_init
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c = best_c
    it = 0
    last_best_time = time.time()
    
    while time.time() < deadline:
        it += 1
        elapsed_since_best = time.time() - last_best_time
        if elapsed_since_best > 40:
            cur_h = best_h.copy()
            cur_c = best_c
            T = max(0.0001, T_init * 0.4)
            last_best_time = time.time()
        
        c_vec = compute_c5_vec(cur_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) > 0 and rng.random() < 0.7:
            grad = np.zeros(n)
            for idx in active[:20]:
                lag = idx - (n - 1)
                if lag >= 0:
                    nv = n - lag
                    grad[:nv] += (1.0 - cur_h[lag:])
                    grad[lag:] += -cur_h[:nv]
                else:
                    kk = -lag
                    nv = n - kk
                    grad[kk:] += (1.0 - cur_h[:nv])
                    grad[:nv] += -cur_h[kk:]
            pos_idx = np.where(grad > 0.01)[0]
            neg_idx = np.where(grad < -0.01)[0]
            if len(pos_idx) > 0 and len(neg_idx) > 0:
                i = pos_idx[rng.randint(len(pos_idx))]
                j = neg_idx[rng.randint(len(neg_idx))]
            else:
                i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = rng.choice(n, size=2, replace=False)
        
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta < 0.0001:
            continue
        delta = min(rng.uniform(0.0003, 0.002), max_delta * 0.6)
        
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c = compute_c5(cur_h, n)
        
        if new_c <= cur_c:
            cur_c = new_c
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = cur_h.copy()
                last_best_time = time.time()
        else:
            dc = new_c - cur_c
            if dc < T * 8.0:
                if rng.random() < np.exp(-dc / T):
                    cur_c = new_c
                else:
                    cur_h[i] += delta
                    cur_h[j] -= delta
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
        
        if it % 3000 == 0:
            T *= 0.93
        if T < 0.00005:
            T = T_init * 0.5
            if T < 0.00005:
                break
    
    return best_h, best_c

def smooth_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def gradient_polish(h, n, deadline, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pairwise_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    radius = 40
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            order = np.argsort(-np.abs(compute_c5_vec(cur_h, n) - compute_c5_vec(cur_h, n).max()))[:40]
            for i in order:
                if time.time() > deadline - 5:
                    return cur_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if cur_h[i] >= step and cur_h[j] + step <= 1.0:
                        cur_h[i] -= step
                        cur_h[j] += step
                        c = compute_c5(cur_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            cur_h[i] += step
                            cur_h[j] -= step
                    if cur_h[i] - step >= 0.0 and cur_h[j] + step <= 1.0:
                        cur_h[i] -= step
                        cur_h[j] += step
                        c = compute_c5(cur_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            cur_h[i] += step
                            cur_h[j] -= step
    return cur_h, best_c

def multires_explore(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for n_coarse in [n // 2, n // 4, n // 5]:
        if n_coarse < 40:
            continue
        if time.time() > deadline - 10:
            break
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        t_alloc = min(30, (deadline - time.time()) * 0.2)
        if t_alloc > 10:
            h_c, c_c = directed_sa(h_c, n_coarse, time.time() + t_alloc,
                                   T_init=0.01, seed=77)
        
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    n = n_orig
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Directed SA exploration (main phase)
    t1 = min(t0 + 350, deadline - 400)
    if time.time() < t1 - 5:
        h, c = directed_sa(h, n, t1, T_init=0.005, seed=seed)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 2: Multi-resolution exploration
    if time.time() < deadline - 300:
        t2 = min(time.time() + 80, deadline - 300)
        h_mr, c_mr = multires_explore(h, n, t2)
        if c_mr < best_c:
            best_c = c_mr
            best_h = h_mr.copy()
            h = h_mr.copy()
        else:
            h = project_feasible(h, n)
    
    # Phase 3: Second SA pass with different parameters
    if time.time() < deadline - 250:
        t3 = min(time.time() + 200, deadline - 250)
        h, c = directed_sa(h, n, t3, T_init=0.003, seed=seed + 100)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 4: Fine SA
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        h, c = fine_sa(h, n, t4, T_init=0.0006, seed=seed + 200)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 5: Gradient polish
    if time.time() < deadline - 80:
        t5 = min(time.time() + 60, deadline - 80)
        h, c = gradient_polish(h, n, t5, alphas=[2000, 20000, 200000, 1000000])
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 6: Very fine SA after gradient
    if time.time() < deadline - 50:
        t6 = min(time.time() + 35, deadline - 50)
        h, c = fine_sa(h, n, t6, T_init=0.0003, seed=seed + 300)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 7: Pairwise refinement
    if time.time() < deadline - 20:
        t7 = min(time.time() + 18, deadline - 20)
        h, c = pairwise_refine(h, n, t7)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = h.copy()
    
    # Phase 8: Final micro-SA + gradient
    if time.time() < deadline - 5:
        h, c = fine_sa(h, n, deadline - 2, T_init=0.0002, seed=seed + 400)
        if c < best_c:
            best_c = c
            best_h = h
            h = h.copy()
    
    # Final: also try grammar polish at the very end
    if time.time() < deadline - 1:
        h, c = gradient_polish(h, n, deadline - 0.5, alphas=[500000, 2000000])
        if c < best_c:
            best_c = c
            best_h = h
            h = h.copy()
    
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
