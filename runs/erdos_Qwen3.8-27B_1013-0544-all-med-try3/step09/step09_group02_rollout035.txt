            start = rng.integers(0, n - n // 4)
            noise[start:start + n // 4] = rng.choice([-0.05, 0.05], size=n // 4)
        
        h_pert = project_feasible(h_best + noise, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS with moderate alpha
        t_phase = min(time.time() + 10, deadline - 5)
        try:
            h_new, c_new = lbfgs_run(h_pert, n, 5000, 1000, t_phase, maxiter=2000, memory=30)
            if c_new < best_c:
                best_c = c_new
                h_best = h_new.copy()
                found = True
        except Exception:
            pass
        
        # Also try top-k focused
        if time.time() < deadline - 10:
            try:
                h_new2, c_new2 = topk_lbfgs_run(h_best, n, 3, 500, min(time.time() + 5, deadline - 5), maxiter=1000)
                if c_new2 < best_c:
                    best_c = c_new2
                    h_best = h_new2.copy()
            except Exception:
                pass
    
    return h_best, best_c

def local_search_focused(h, n, deadline, best_c):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    thresh = max_val - 0.005 * max_val
    active = np.where(c_vec >= thresh)[0]
    
    grad_sum = np.zeros(n)
    for idx in active:
        grad_sum += shift_grad_single(h, n, idx)
    if len(active) > 0:
        grad_sum /= len(active)
    
    # Find indices with largest gradient magnitude
    top_indices = np.argsort(-np.abs(grad_sum))[:min(100, n)]
    
    steps = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i_val in top_indices:
                if time.time() > deadline:
                    break
                i = int(i_val)
                # Try swapping with neighbors
                for offset in range(-15, 16):
                    if offset == 0:
                        continue
                    j = i + offset
                    if j < 0 or j >= n:
                        continue
                    # Move mass i -> j
                    if h[i] >= step and h[j] <= 1.0 - step:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    # Move mass j -> i
                    if h[j] >= step and h[i] <= 1.0 - step:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
    return h, best_c

def upsample_and_optimize(h, n, deadline):
    n_high = max(n * 2, 1000)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_high)
    h_high = np.interp(x_new, x_old, h)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)
    
    # Quick L-BFGS at high res
    if time.time() < deadline - 20:
        t1 = min(time.time() + 25, deadline - 20)
        h_high, c_high = lbfgs_run(h_high, n_high, 5000, 1000, t1, maxiter=3000, memory=40)
    
    if time.time() < deadline - 10:
        t2 = min(time.time() + 20, deadline - 10)
        h_high, c_high = lbfgs_run(h_high, n_high, 50000, 10000, t2, maxiter=3000, memory=40)
    
    if time.time() < deadline - 5:
        t3 = min(time.time() + 10, deadline - 5)
        h_high, c_high = topk_lbfgs_run(h_high, n_high, 3, 5000, t3, maxiter=1000)
    
    return h_high, c_high

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Initial L-BFGS polish with moderate alpha (50s)
    if time.time() < deadline - 500:
        t1 = min(t0 + 50, deadline - 480)
        h1, c1 = lbfgs_run(best_h, n, 2000, 500, t1, maxiter=8000, memory=50)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: High-alpha L-BFGS (80s)
    if time.time() < deadline - 400:
        t2 = min(time.time() + 40, deadline - 380)
        h2, c2 = lbfgs_run(best_h, n, 20000, 5000, t2, maxiter=10000, memory=50)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        
        if time.time() < deadline - 360:
            t2b = min(time.time() + 35, deadline - 350)
            h2b, c2b = lbfgs_run(best_h, n, 100000, 20000, t2b, maxiter=10000, memory=50)
            if c2b < best_c:
                best_c = c2b
                best_h = h2b
    
    # Phase 3: Top-k focused optimization (80s)
    if time.time() < deadline - 300:
        for k_val in [1, 3, 5]:
            if time.time() > deadline - 280:
                break
            t3 = min(time.time() + 25, deadline - 270)
            h3, c3 = topk_lbfgs_run(best_h, n, k_val, 2000, t3, maxiter=3000)
            if c3 < best_c:
                best_c = c3
                best_h = h3
    
    # Phase 4: Worst-shift gradient descent (60s)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 60, deadline - 240)
        h4, c4 = worst_shift_gradient_descent(best_h, n, t4, initial_step=0.005)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Multi-start exploration (250s)
    if time.time() < deadline - 120:
        t5 = min(time.time() + 250, deadline - 100)
        best_h, best_c = multi_start(best_h, best_c, n, t5, n_starts=30)
    
    # Phase 6: Very high alpha refinement (100s)
    if time.time() < deadline - 80:
        t6a = min(time.time() + 50, deadline - 50)
        h6, c6 = lbfgs_run(best_h, n, 200000, 50000, t6a, maxiter=15000, memory=60)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        
        if time.time() < deadline - 30:
            t6b = min(time.time() + 30, deadline - 25)
            h6b, c6b = lbfgs_run(best_h, n, 500000, 100000, t6b, maxiter=10000, memory=60)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Top-k + worst-shift after multi-start (40s)
    if time.time() < deadline - 50:
        t7 = min(time.time() + 20, deadline - 30)
        h7, c7 = topk_lbfgs_run(best_h, n, 1, 5000, t7, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        
        if time.time() < deadline - 20:
            t7b = min(time.time() + 15, deadline - 15)
            h7b, c7b = worst_shift_gradient_descent(best_h, n, t7b, initial_step=0.002)
            if c7b < best_c:
                best_c = c7b
                best_h = h7b
    
    # Phase 8: Local search (60s)
    if time.time() < deadline - 40:
        t8 = min(time.time() + 60, deadline - 30)
        best_h, best_c = local_search_focused(best_h, n, t8, best_c)
    
    # Phase 9: High-resolution refinement
    if time.time() < deadline - 60:
        t9 = deadline - 10
        h_high, c_high = upsample_and_optimize(best_h, n, t9)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n = len(best_h)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
