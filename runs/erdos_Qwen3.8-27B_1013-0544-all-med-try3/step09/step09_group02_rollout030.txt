```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

# Smoothed objective for BFGS
def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

# SQP with LP
def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    grads = []
    c_vals = []
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        grads.append(g)
        c_vals.append(c_vec[idx])
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        A_ub[row, 0:n] = grads[j]
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vals[j]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05, max_active=30):

        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])

        delta, t_opt = sqp_lp_step(h, n, active, rho)

            rho *= 0.5

        if np.max(np.abs(delta)) < 1e-12:

        h_new = project_feasible(h + delta, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0

            rho *= 0.7

def binary_local_search(h, n, deadline, max_rounds=50):

def run_lbfgs(h, n, deadline, configs=None):

            h_new = project_feasible(res.x, n)

    for step in steps:

            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()

            grad = np.zeros(n)

            grad /= len(active)
            top_pos = np.argsort(-grad)[:min(center_radius * 3, n)]
            for i_val in top_pos:

                i = int(i_val)

                hi = min(n, i + center_radius + 1)
```

```python
def swap_delta(O, h, i, j, n):
    """Compute change in overlap vector when swapping positions i and j (h[i]:1->0, h[j]:0->1)"""
    dx = 2.0 / n
    # For shift mm, k = mm - (n-1)
    mm = np.arange(2*n - 1)
    k = mm - (n - 1)
    
    delta = np.zeros(2*n - 1)
    
    # -(1-h[i+k]): valid when 0 <= i+k < n
    idx_ik = i + k
    valid_ik = (idx_ik >= 0) & (idx_ik < n)
    delta[valid_ik] -= (1 - h[idx_ik[valid_ik]])
    
    # +h[i-k]: valid when 0 <= i-k < n
    idx_ik2 = i - k
    valid_ik2 = (idx_ik2 >= 0) & (idx_ik2 < n)
    delta[valid_ik2] += h[idx_ik2[valid_ik2]]
    
    # +(1-h[j+k]): valid when 0 <= j+k < n
    idx_jk = j + k
    valid_jk = (idx_jk >= 0) & (idx_jk < n)
    delta[valid_jk] += (1 - h[idx_jk[valid_jk]])
    
    # -h[j-k]: valid when 0 <= j-k < n
    idx_jk2 = j - k
    valid_jk2 = (idx_jk2 >= 0) & (idx_jk2 < n)
    delta[valid_jk2] -= h[idx_jk2[valid_jk2]]
    
    return delta * dx
```

```python
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05, max_active=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        if np.max(np.abs(delta)) < 1e-12:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            rho = min(rho * 1.2, 0.12)
        else:
            no_improve += 1
            rho *= 0.6
            if rho < 1e-10 or no_improve > 40:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 100.0), (2000, 1000.0), (5000, 5000.0), (10000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def swap_delta_vec(h, i, j, n):
    dx = 2.0 / n
    mm = np.arange(2 * n - 1)
    k = mm - (n - 1)
    delta = np.zeros(2 * n - 1)
    idx_ik = i + k
    valid_ik = (idx_ik >= 0) & (idx_ik < n)
    delta[valid_ik] -= (1.0 - h[idx_ik[valid_ik]])
    idx_ik2 = i - k
    valid_ik2 = (idx_ik2 >= 0) & (idx_ik2 < n)
    delta[valid_ik2] += h[idx_ik2[valid_ik2]]
    idx_jk = j + k
    valid_jk = (idx_jk >= 0) & (idx_jk < n)
    delta[valid_jk] += (1.0 - h[idx_jk[valid_jk]])
    idx_jk2 = j - k
    valid_jk2 = (idx_jk2 >= 0) & (idx_jk2 < n)
    delta[valid_jk2] -= h[idx_jk2[valid_jk2]]
    return delta * dx

def simulated_annealing(h, O, n, deadline, T_init=0.004, T_end=0.0001, restart_patience=3000):
    best_h = h.copy()
    best_c = float(O.max())
    T = T_init
    n_iter = 0
    no_improve = 0
    rng = np.random.RandomState(42)
    
    ones_mask = h == 1.0
    zeros_mask = h == 0.0
    ones_count = int(ones_mask.sum())
    
    while time.time() < deadline:
        if rng.random() < 0.01:
            idx = rng.randint(n)
            if ones_mask[idx]:
                j_idx = rng.randint(n)
                if not ones_mask[j_idx]:
                    i, j = idx, j_idx
                else:
                    continue
            else:
                i_idx = rng.randint(n)
                if ones_mask[i_idx]:
                    i, j = i_idx, idx
                else:
                    continue
        else:
            i = int(rng.choice(np.where(ones_mask)[0]))
            j = int(rng.choice(np.where(zeros_mask)[0]))
        
        dO = swap_delta_vec(h, i, j, n)
        O_new_max = max(float((O + dO).max()), 0.0)
        dC = O_new_max - float(O.max())
        
        if dC < 0 or rng.random() < np.exp(-dC / T):
            O = O + dO
            if O.max() < best_c:
                best_c = float(O.max())
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            h[i] = 0.0
            h[j] = 1.0
            ones_mask[i] = False
            ones_mask[j] = True
        else:
            no_improve += 1
        
        n_iter += 1
        if n_iter % 500 == 0:
            progress = min(1.0, n_iter / 100000.0)
            T = T_init * (T_end / T_init) ** progress
            if no_improve > restart_patience:
                # Perturb: random few swaps
                for _ in range(5):
                    ii = int(rng.choice(np.where(ones_mask)[0]))
                    jj = int(rng.choice(np.where(~ones_mask)[0]))
                    dO2 = swap_delta_vec(h, ii, jj, n)
                    O = O + dO2
                    h[ii] = 0.0
                    h[jj] = 1.0
                    ones_mask[ii] = False
                    ones_mask[jj] = True
                no_improve = 0
                T = T_init * 0.5
    
    return best_h, best_c

def directed_binary_search(h_bin, n, deadline, max_rounds=200):
    best_c = float(compute_c5_vec(h_bin, n).max())
    best_h = h_bin.copy()
    O = compute_c5_vec(best_h, n)
    rng = np.random.RandomState(123)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones_mask = best_h == 1.0
        zeros_mask = best_h == 0.0
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        
        # Try swaps biased toward positions that affect the max shifts
        max_shifts = np.where(O >= best_c - 0.002)[0]
        
        n_tries = min(3000, len(ones_idx) * len(zeros_idx) // 2)
        for _ in range(n_tries):
            if time.time() > deadline:
                return best_h, best_c
            i = int(ones_idx[rng.randint(len(ones_idx))])
            j = int(zeros_idx[rng.randint(len(zeros_idx))])
            dO = swap_delta_vec(best_h, i, j, n)
            new_max = float((O + dO).max())
            if new_max < best_c - 1e-15:
                O = O + dO
                best_c = new_max
                best_h[i] = 0.0
                best_h[j] = 1.0
                ones_mask[i] = False
                ones_mask[j] = True
                improved = True
                if _ % 100 == 0:
                    ones_idx = np.where(ones_mask)[0]
                    zeros_idx = np.where(zeros_mask)[0]
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick SQP (40s)
    t1 = min(t0 + 40, deadline - 700)
    if time.time() < t1:
        best_h, best_c = sqp_optimize(best_h, n, t1, max_iter=200, rho_init=0.06, max_active=50)
    
    # Phase 2: BFGS warmup (50s)
    t2 = min(t0 + 90, deadline - 600)
    if time.time() < t2:
        best_h, best_c = run_lbfgs(best_h, n, t2,
            configs=[(500, 100.0), (2000, 1000.0), (5000, 5000.0)])
    
    # Phase 3: Binary conversion + directed search (80s)
    t3 = min(t0 + 170, deadline - 500)
    if time.time() < t3:
        h_bin = make_exactly_binary(best_h, n)
        c_bin = compute_c5(h_bin, n)
        O_bin = compute_c5_vec(h_bin, n)
        h_bin, c_bin = directed_binary_search(h_bin, n, t3, max_rounds=300)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 4: SQP polish on best (40s)
    t4 = min(t0 + 210, deadline - 450)
    if time.time() < t4:
        best_h, best_c = sqp_optimize(best_h, n, t4, max_iter=150, rho_init=0.04, max_active=60)
    
    # Phase 5: Simulated annealing (120s)
    t5 = min(t0 + 330, deadline - 350)
    if time.time() < t5:
        h_bin = make_exactly_binary(best_h, n)
        O_bin = compute_c5_vec(h_bin, n)
        h_bin, c_bin = simulated_annealing(h_bin, O_bin, n, t5, T_init=0.003, T_end=0.00005)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 6: SQP after SA (40s)
    t6 = min(t0 + 370, deadline - 300)
    if time.time() < t6:
        best_h, best_c = sqp_optimize(best_h, n, t6, max_iter=150, rho_init=0.03, max_active=60)
    
    # Phase 7: Directed search again (60s)
    t7 = min(t0 + 430, deadline - 240)
    if time.time() < t7:
        h_bin = make_exactly_binary(best_h, n)
        h_bin, c_bin = directed_binary_search(h_bin, n, t7, max_rounds=200)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 8: BFGS with high alpha (60s)
    t8 = min(t0 + 490, deadline - 180)
    if time.time() < t8:
        best_h, best_c = run_lbfgs(best_h, n, t8,
            configs=[(5000, 5000.0), (10000, 10000.0), (20000, 20000.0)])
    
    # Phase 9: Try n=1000 (200s)
    n2 = 1000
    if time.time() < deadline - 210:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        
        t9 = min(time.time() + 50, deadline - 150)
        if time.time() < t9:
            h2, c2 = sqp_optimize(h2, n2, t9, max_iter=150, rho_init=0.05, max_active=40)
        
        t9b = min(time.time() + 50, deadline - 90)
        if time.time() < t9b:
            h2, c2 = run_lbfgs(h2, n2, t9b,
                configs=[(2000, 1000.0), (5000, 5000.0), (10000, 10000.0)])
        
        t9c = min(time.time() + 40, deadline - 40)
        if time.time() < t9c:
            h2_bin = make_exactly_binary(h2, n2)
            O2 = compute_c5_vec(h2_bin, n2)
            h2_bin, c2b = directed_binary_search(h2_bin, n2, t9c, max_rounds=100)
            if c2b < c2:
                h2, c2 = h2_bin, c2b
        
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 10: Final SQP polish
    if time.time() < deadline - 5:
        best_h, best_c = sqp_optimize(best_h, len(best_h), deadline - 2,
            max_iter=80, rho_init=0.02, max_active=60)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
