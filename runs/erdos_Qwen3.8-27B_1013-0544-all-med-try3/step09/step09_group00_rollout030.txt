import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_beatty(n, alpha, offset=0):
    h = np.zeros(n)
    target = n // 2
    seen = np.zeros(n, dtype=bool)
    count = 0
    for k in range(1, 50 * n):
        pos = (int(k * alpha) + offset) % n
        if not seen[pos]:
            seen[pos] = True
            count += 1
            if count == target:
                break
    if count < target:
        zeros = np.where(~seen)[0]
        for i in range(min(target - count, len(zeros))):
            seen[zeros[i]] = True
    h[seen] = 1.0
    return h

def upsample(h, factor):
    return np.repeat(h, factor)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_smooth_max_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_smooth_max_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:

    return grad

def lbfgs_full_obj(x, n, alpha, mu):

    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2

def phase_topk_optimize(h, n, deadline, top_k, alphas, mu, max_rounds):

    best_c = compute_c5(best_h, n)

def binary_pair_optimize(h, n, deadline, max_rounds=50000):
    """Combinatorial binary optimization via greedy pair swaps targeting worst lags."""
    h = project_feasible(h, n)
    # Bin
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    while h_bin.sum() > n / 2:
        ones = np.where(h_bin == 1)[0]
        h_bin[ones[0]] = 0
    while h_bin.sum() < n / 2:
        zeros = np.where(h_bin == 0)[0]
        h_bin[zeros[0]] = 1
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Active lags: those within 2% of max
        tol = max(1e-10, 0.02 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Compute gradient from active lags
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - best_h[k:n])
                g[k:n] -= best_h[0:nv]
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - best_h[0:nv])
                g[0:nv] -= best_h[kk:n]
        
        # Find best swap: decrease h[i] (where g[i] > 0 and h[i]=1), increase h[j] (where g[j] < 0 and h[j]=0)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Rank by gradient
        ones_sorted = ones[np.argsort(-g[ones])]
        zeros_sorted = zeros[np.argsort(g[zeros])]
        
        found = False
        for i in ones_sorted[:50]:
            if g[i] <= 0:
                break
            for j in zeros_sorted[:50]:
                if g[j] >= 0:
                    break
                # Try swap
                best_h[i] = 0
                best_h[j] = 1
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    found = True
                    break
                else:
                    best_h[i] = 1
                    best_h[j] = 0
            if found:
                break
        
        if not found:
            # Try less strict: accept if max doesn't increase
            for i in ones_sorted[:30]:
                for j in zeros_sorted[:30]:
                    best_h[i] = 0
                    best_h[j] = 1
                    c_new = compute_c5(best_h, n)
                    if c_new <= best_c + 1e-15:
                        best_h = best_h.copy()
                        best_c = min(best_c, c_new)
                        found = True
                        break
                    else:
                        best_h[i] = 1
                        best_h[j] = 0
                if found:
                    break
            if not found:
                break
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    rem = max(10, int(deadline - time.time()))
    maxit = min(maxiter, max(100, rem * 5))
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
