        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        if mx > 400:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    
    def grad(x):
        c_vec = np.correlate(x, 1 - x, mode='full')
        vals = c_vec * dx
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-15:
                continue
            k = mm - n_center
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    return obj, grad

def bundle_optimize(h, n, deadline, max_rounds=30):
    """Bundle method: iteratively optimize against growing set of active shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Track bundle of shifts that have been near-max
    bundle = set()
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    threshold = max_val - 0.01 * max_val
    for idx in np.where(c_vec >= threshold)[0]:
        bundle.add(int(idx))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Determine active set: bundle plus current near-max
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-5, 0.005 * max_val)
        current_active = set(int(i) for i in np.where(c_vec >= threshold)[0])
        active_set = bundle | current_active
        
        # Limit active set size for computational tractability
        if len(active_set) > 40:
            c_arr = np.array([c_vec[i] for i in active_set])
            order = np.argsort(-c_arr)
            top_half = set(list(active_set)[:len(active_set)])
            # Just use top 40 by correlation value
            all_indices = list(active_set)
            c_vals = [c_vec[i] for i in all_indices]
            top_order = np.argsort(-c_vals)[:40]
            active_set = set(all_indices[i] for i in top_order)
        
        active_idx = np.array(sorted(active_set), dtype=int)
        
        # Choose alpha based on bundle size
        if len(active_idx) <= 5:
            alpha = 500000
            mu = 500.0
        elif len(active_idx) <= 15:
            alpha = 100000
            mu = 100.0
        elif len(active_idx) <= 30:
            alpha = 20000
            mu = 20.0
        else:
            alpha = 5000
            mu = 5.0
        
        obj_f, grad_f = make_obj_grad(n, active_idx, alpha, mu)
        
        try:
            res = minimize(obj_f, best_h, jac=grad_f, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                # Update bundle with new near-max shifts
                c_vec = compute_c5_vec(best_h, n)
                max_val = c_vec.max()
                thr = max_val - max(1e-5, 0.005 * max_val)
                for idx in np.where(c_vec >= thr)[0]:
                    bundle.add(int(idx))
            else:
                # If no improvement, update bundle anyway
                c_vec = compute_c5_vec(best_h, n)
                max_val = c_vec.max()
                thr = max_val - max(1e-5, 0.003 * max_val)
                for idx in np.where(c_vec >= thr)[0]:
                    bundle.add(int(idx))
        except Exception:
            pass
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, base_temp=0.004, cool_rate=0.9985, seed=42):
    """Simulated annealing with 2-point moves."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    T = base_temp
    no_improve = 0
    
    while time.time() < deadline:
        if T < 1e-12:
            break
        
        # 2-point perturbation
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        magnitude = 0.01 * (0.5 + rng.random())
        delta = (1.0 if rng.random() < 0.5 else -1.0) * magnitude
        
        if cur_h[i] + delta > 1.0 + 1e-12 or cur_h[i] + delta < -1e-12:
            continue
        if cur_h[j] - delta > 1.0 + 1e-12 or cur_h[j] - delta < -1e-12:
            continue
        
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c = compute_c5(cur_h, n)
        
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
            no_improve = 0
        else:
            dc = new_c - cur_c
            if dc / T < 40:
                if rng.random() < np.exp(-dc / T):
                    cur_c = new_c
                else:
                    cur_h[i] -= delta
                    cur_h[j] += delta
            else:
                cur_h[i] -= delta
                cur_h[j] += delta
            no_improve += 1
        
        if no_improve > 300:
            T *= cool_rate
            no_improve = 0
    
    return best_h, best_c

def fine_pair_polish(h, n, deadline):
    """Very fine 2-point moves to polish."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(99)
    
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved_this_round = True
        while improved_this_round:
            improved_this_round = False
            if time.time() > deadline:
                return best_h, best_c
            # Try random pairs
            for _ in range(500):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for sign in [1, -1]:
                    if best_h[i] + sign * d > 1.0 + 1e-12 or best_h[i] + sign * d < -1e-12:
                        continue
                    if best_h[j] - sign * d > 1.0 + 1e-12 or best_h[j] - sign * d < -1e-12:
                        continue
                    best_h[i] += sign * d
                    best_h[j] -= sign * d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved_this_round = True
                    else:
                        best_h[i] -= sign * d
                        best_h[j] += sign * d
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick improvement at original resolution (60s)
    if time.time() < deadline - 500:
        phase1_deadline = min(t0 + 60, deadline - 500)
        # Quick L-BFGS with moderate alpha
        c_vec = compute_c5_vec(best_h, n_orig)
        max_val = c_vec.max()
        threshold = max_val - 0.01 * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 20:
            order = np.argsort(-c_vec[active])
            active = active[order[:20]]
        
        bounds = [(0.0, 1.0)] * n_orig
        obj_f, grad_f = make_obj_grad(n_orig, active, 10000, 50.0)
        try:
            res = minimize(obj_f, best_h, jac=grad_f, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-18, 'gtol': 1e-13})
            h_new = project_feasible(res.x, n_orig)
            c_new = compute_c5(h_new, n_orig)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_n = np.interp(x_new, x_old, best_h)
    h_n = project_feasible(h_n, n)
    c_n = compute_c5(h_n, n)
    
    if c_n <= best_c:
        best_h = h_n
        best_c = c_n
    else:
        best_h = h_n
        # Keep best_c from coarser grid as reference but work at n=1000
    
    n = len(best_h)
    
    # Phase 3: Bundle optimization at n=1000 (400s)
    if time.time() < deadline - 350:
        phase3_deadline = min(t0 + 460, deadline - 350)
        best_h, c_bundle = bundle_optimize(best_h, n, phase3_deadline, max_rounds=25)
        if c_bundle < best_c:
            best_c = c_bundle
    
    # Phase 4: Simulated annealing (150s)
    if time.time() < deadline - 200:
        sa_deadline = min(time.time() + 150, deadline - 200)
        h_sa, c_sa = simulated_annealing(best_h, n, sa_deadline, 
                                          base_temp=0.003, cool_rate=0.9988, seed=seed)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 5: Bundle optimization again after SA (150s)
    if time.time() < deadline - 150:
        phase5_deadline = min(time.time() + 150, deadline - 150)
        best_h, c_b2 = bundle_optimize(best_h, n, phase5_deadline, max_rounds=15)
        if c_b2 < best_c:
            best_c = c_b2
    
    # Phase 6: Very fine SA (60s)
    if time.time() < deadline - 80:
        sa2_deadline = min(time.time() + 60, deadline - 80)
        h_sa2, c_sa2 = simulated_annealing(best_h, n, sa2_deadline,
                                            base_temp=0.001, cool_rate=0.9995, seed=seed+1)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2
    
    # Phase 7: Final bundle polish (60s)
    if time.time() < deadline - 40:
        phase7_deadline = min(time.time() + 60, deadline - 40)
        best_h, c_f = bundle_optimize(best_h, n, phase7_deadline, max_rounds=8)
        if c_f < best_c:
            best_c = c_f
    
    # Phase 8: Fine pair polish (30s)
    if time.time() < deadline - 20:
        polish_deadline = min(time.time() + 30, deadline - 20)
        best_h, c_p = fine_pair_polish(best_h, n, polish_deadline)
        if c_p < best_c:
            best_c = c_p
    
    # Final: one more quick L-BFGS on top-5 active shifts
    if time.time() < deadline - 5:
        c_vec = compute_c5_vec(best_h, n)
        top5 = np.argsort(-c_vec)[:5]
        bounds = [(0.0, 1.0)] * n
        obj_f, grad_f = make_obj_grad(n, top5, 2000000, 10000.0)
        try:
            res = minimize(obj_f, best_h, jac=grad_f, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Final validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
