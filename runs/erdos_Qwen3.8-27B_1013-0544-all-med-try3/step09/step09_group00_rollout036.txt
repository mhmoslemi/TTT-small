- Constraints: n-1 + (n-1) + 1 = 2n-1 + 1 = 2n linear inequality constraints (for each k)

  + 1 equality constraint (sum h = n/2)

But will the LP give a good solution? The linearized problem is a relaxation, so the LP solution will have f_k(h_LP) ≤ f_k approximation, but f_k(h_LP) might be > f_k(h_LP approx). We need to check feasibility.

The standard approach (similar to the "quadratic programming by successive linear programming" or "Eaves' method"):
- Solve the LP
- Check if the LP solution satisfies the original (nonlinear) constraints
- If yes, we're done (or we can continue to improve)
- If no, the solution is infeasible for the original problem
- We can use the LP solution as a new starting point and re-linearize

This is essentially a "fixed-point" or "successive linearization" method. It might not converge, but it often works well in practice.

def lp_linearize_step(h, n):
    """One step of successive linearization."""
    from scipy.optimize import linprog
    
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    
    # Linearize each f_k around h
    # f_k approx = f_k(h) + grad_k · (h_new - h)
    # We want: f_k approx ≤ t for all k
    # i.e., f_k(h) + grad_k · h_new - grad_k · h ≤ t
    # i.e., grad_k · h_new - t ≤ grad_k · h - f_k(h)
    
    # But we have 2n-1 constraints. Let's only use the "active" ones (near the max)
    # to keep the LP small.
    
    active = np.where(c_vec >= max_val * 0.9)[0]
    
    # Build LP:
    # Variables: [h_new[0], ..., h_new[n-1], t]
    # Minimize: -t (maximize t... wait, we want to minimize t)
    # Actually: minimize t
    # Subject to: for each active k: grad_k · h_new - t ≤ RHS_k
    #             0 ≤ h_new[i] ≤ 1
    #             sum h_new = n/2
    
    m = len(active)
    # A_ub: m + 0 rows (just the active constraints)
    # c: [0, 0, ..., 0, 1] (minimize t)
    
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    dx = 2.0 / n
    for l, idx in enumerate(active):
        k = idx - (n - 1)
        # Compute grad_k
        gk = np.zeros(n)
        if k >= 0:
            nv = n - k
            gk[0:nv] += (1.0 - h[k:n])
            gk[k:n] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            gk[kk:n] += (1.0 - h[0:nv])
            gk[0:nv] += (-h[kk:n])
        gk *= dx
        
        f_k_val = c_vec[idx]
        grad_dot_h = gk @ h  # should equal f_k_val (by Euler's theorem for homogeneous... well, approximately)
        # Actually f_k is not homogeneous, so grad_dot_h ≠ f_k
        # f_k(h) = sum h[i](1-h[i+k])
        # grad_k = d/dh of f_k
        # grad_dot_h = sum_j grad_k[j] * h[j]
        # This is NOT equal to f_k(h) in general
        
        # The linearization: f_k(h_new) ≈ f_k(h) + grad_k · (h_new - h)
        # We want: f_k(h) + grad_k · (h_new - h) ≤ t
        # i.e., grad_k · h_new - t ≤ grad_k · h - f_k(h)
        
        A_ub[l, :n] = gk
        A_ub[l, n] = -1.0
        b_ub[l] = gk @ h - f_k_val
    
    # Bounds: 0 ≤ h[i] ≤ 1, t is free
    bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return h, max_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optim(h, n, deadline, alphas, mu=0.01, maxit_per_alpha=5000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = min(maxit_per_alpha, max(300, int(remaining * 2.5)))
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def local_search(h, n, deadline, n_pairs_per_round=2000, max_rounds=200, amt=0.005):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(12345)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Compute gradient once per round
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= max(1, len(active))
        
        # Greedy: find best (i,j) pair to transfer from i to j
        for _ in range(n_pairs_per_round):
            if time.time() > deadline:
                break
            # Pick i from top positive gradient, j from top negative gradient
            pos_candidates = np.where((grad > 0) & (best_h > amt))[0]
            neg_candidates = np.where((grad < -1e-10) & (best_h < 1 - amt))[0]
            if len(pos_candidates) == 0 or len(neg_candidates) == 0:
                break
            # Sample a few
            i = pos_candidates[rng.randint(len(pos_candidates))]
            j = neg_candidates[rng.randint(len(neg_candidates))]
            a = min(amt, best_h[i], 1.0 - best_h[j])
            if a < 1e-10:
                continue
            best_h[i] -= a
            best_h[j] += a
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
            else:
                best_h[i] += a
                best_h[j] -= a
                best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

        elif strategy == 2:

                i, j = rng.choice(n, 2, replace=False)
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        else:
            # Circular shift by small amount
            shift = rng.randint(1, 5)
            h_pert = np.roll(h_pert, shift)

    best_h = h_parent.copy()
    best_c = c_parent
    best_n = n_parent
    
    # === Strategy 1: High resolution optimization at n=1000 ===
    n_hi = 1000
    h_hi = upsample(h_parent, n_parent, n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    
    # Phase 1a: L-BFGS broad search at n=1000
    p1_end = min(t0 + 200, deadline - 500)
    h_hi, c_hi = lbfgs_optim(h_hi, n_hi, p1_end,
        alphas=[50, 200, 1000, 5000, 20000],
        mu=0.01, maxit_per_alpha=3000)
    if c_hi < best_c:
        best_c = c_hi
        best_h = h_hi.copy()
        best_n = n_hi
    
    # Phase 1b: L-BFGS fine search at n=1000
    if time.time() < deadline - 350:
        p1b_end = min(time.time() + 180, deadline - 300)
        h_hi2, c_hi2 = lbfgs_optim(h_hi, n_hi, p1b_end,
            alphas=[50000, 200000, 1000000, 5000000],
            mu=0.005, maxit_per_alpha=5000)
        if c_hi2 < best_c:
            best_c = c_hi2
            best_h = h_hi2.copy()
            best_n = n_hi
    
    # Phase 1c: Active set at n=1000
    if time.time() < deadline - 250:
        p1c_end = min(time.time() + 180, deadline - 200)
        h_hi3, c_hi3 = active_set_opt(h_hi, n_hi, p1c_end,
            top_k=10, n_rounds=35, alphas=[200000, 1000000, 5000000, 20000000])
        if c_hi3 < best_c:
            best_c = c_hi3
            best_h = h_hi3.copy()
            best_n = n_hi
    
    # === Strategy 2: Try different initialization at n=1000 ===
    if time.time() < deadline - 200:
        # Beatty sequence at n=1000
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        h_beatty = make_beatty_binary(n_hi, golden / 2.0)
        c_beatty = compute_c5(h_beatty, n_hi)
        
        # Smooth bump
        h_bump = make_smooth_bump(n_hi)
        c_bump = compute_c5(h_bump, n_hi)
        
        # Two bumps
        h_2bump = make_two_bumps(n_hi)
        c_2bump = compute_c5(h_2bump, n_hi)
        
        # Random smooth
        x = np.linspace(0, 2, n_hi, endpoint=False)
        h_rand = 0.5 + 0.2 * np.sin(rng.uniform(1, 5) * np.pi * x) + 0.1 * np.sin(rng.uniform(2, 8) * np.pi * x)
        h_rand = project_feasible(np.clip(h_rand, 0.01, 0.99), n_hi)
        c_rand = compute_c5(h_rand, n_hi)
        
        candidates = [(c_beatty, h_beatty), (c_bump, h_bump), 
                      (c_2bump, h_2bump), (c_rand, h_rand)]
        candidates.sort(key=lambda x: x[0])
        
        for c_init, h_init in candidates[:2]:
            if time.time() > deadline - 100:
                break
            p2_end = min(time.time() + 80, deadline - 80)
            h_opt, c_opt = lbfgs_optim(h_init, n_hi, p2_end,
                alphas=[1000, 10000, 100000, 1000000],
                mu=0.01, maxit_per_alpha=3000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n_hi
    
    # === Strategy 3: Local search refinement ===
    if time.time() < deadline - 120:
        p3_end = min(time.time() + 80, deadline - 80)
        h_ls, c_ls = local_search(best_h, best_n, p3_end,
            n_pairs_per_round=3000, max_rounds=150, amt=0.003)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()
    
    # === Strategy 4: Active set final polish ===
    if time.time() < deadline - 80:
        p4_end = min(time.time() + 50, deadline - 40)
        h_as, c_as = active_set_opt(best_h, best_n, p4_end,
            top_k=6, n_rounds=20, alphas=[5000000, 20000000, 100000000])
        if c_as < best_c:
            best_c = c_as
            best_h = h_as.copy()
    
    # === Strategy 5: Perturbation and re-optimization ===
    if time.time() < deadline - 60:
        p5_end = deadline - 30
        h_pert, c_pert = perturb_and_reopt(best_h, best_n, p5_end, rng, n_perturbations=3)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
    
    # === Strategy 6: Final L-BFGS polish with very high alpha ===
    if time.time() < deadline - 15:
        p6_end = deadline - 5
        h_final, c_final = lbfgs_optim(best_h, best_n, p6_end,
            alphas=[10000000, 50000000, 200000000],
            mu=0.001, maxit_per_alpha=10000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
