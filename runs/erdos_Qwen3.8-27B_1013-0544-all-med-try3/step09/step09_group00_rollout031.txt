                ones_set.append(best_i)
            if best_h[best_j] > 0.5:
                zeros_set.remove(best_j)
                ones_set.append(best_j)
            else:
                ones_set.remove(best_j)
                zeros_set.append(best_j)
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                break
    
    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 300:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [100, 1000, 10000, 100000, 1000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 8, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k=8, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000]
    
    c_vals = compute_c5_vec(best_h, n)
    max_c = c_vals.max()
    near_thresh = max_c - max(1e-8, 0.1 * max_c)
    near_active = set(np.where(c_vals >= near_thresh)[0].tolist())
    
    for rnd in range(rounds):
        if time.time() > deadline - 15:
            break
        c_vals = compute_c5_vec(best_h, n)
        max_c = c_vals.max()
        thresh = max_c - max(1e-9, 0.02 * max_c)
        active = np.where(c_vals >= thresh)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vals[active])[:top_k])
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            
            def topK_obj_fn(x, n=n, act=active, a=alpha, m=0.001):
                cv = compute_c5_vec(x, n)
                vals = cv[act]
                mx = vals.max()
                sh = a * (vals - mx)
                smx = sh.max()
                if smx > 300:
                    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / a
                else:
                    val = mx + np.log(np.sum(np.exp(sh))) / a
                val += m * (x.sum() - n / 2.0) ** 2
                return float(val)
            
            def topK_grad_fn(x, n=n, act=active, a=alpha, m=0.001):
                cv = compute_c5_vec(x, n)
                vals = cv[act]
                mx = vals.max()
                sh = a * (vals - mx)
                smx = sh.max()
                w = np.exp(sh - smx)
                w /= w.sum()
                dx_val = 2.0 / n
                grad = np.zeros(n)
                for wi, idx in zip(w, act):
                    if wi < 1e-12:
                        continue
                    k = idx - (n - 1)
                    wm = wi * dx_val
                    if k >= 0:
                        nv = n - k
                        grad[0:nv] += wm * (1.0 - x[k:n])
                        grad[k:n] += wm * (-x[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        grad[kk:n] += wm * (1.0 - x[0:nv])
                        grad[0:nv] += wm * (-x[kk:n])
                grad += 2.0 * m * (x.sum() - n / 2.0)
                return grad
            
            try:
                res = minimize(topK_obj_fn, best_h, jac=topK_grad_fn,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgradient_method(h, n, deadline, alpha=1000, lr_start=0.01, momentum=0.9):
    """亚梯度法 with momentum on the active set."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h_curr = best_h.copy()
    velocity = np.zeros(n)
    lr = lr_start
    dx = 2.0 / n
    
    while time.time() < deadline:
        c_vals = compute_c5_vec(h_curr, n)
        max_c = c_vals.max()
        thresh = max_c - max(1e-9, 0.01 * max_c)
        active = np.where(c_vals >= thresh)[0]
        if len(active) == 0:
            break
        
        # Compute gradient of smooth max
        sh = alpha * c_vals[active]
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            if wi < 1e-10:
                continue
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h_curr[k:n])
                grad[k:n] += wm * (-h_curr[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h_curr[0:nv])
                grad[0:nv] += wm * (-h_curr[kk:n])
        
        # Update with momentum and projection
        velocity = momentum * velocity - lr * grad
        h_new = h_curr + velocity
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        h_curr = h_new
        lr *= 0.995
        if lr < 1e-6:
            break
    
    return best_h, best_c

def make_beatty_start(n, frac):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * frac) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def interpolate_to_resolution(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new resolution."""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = make_beatty_start(500, (1.0 + np.sqrt(5.0)) / 4.0)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS warmup
    if time.time() < deadline - 400:
        p1_end = min(t0 + 40, deadline - 380)
        h1, c1 = lbfgs_refine(best_h, best_n, p1_end, alphas=[50, 500, 5000, 50000], mu=0.01)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Binary search on full vector (works for both binary and near-binary)
    if time.time() < deadline - 300:
        # Round to binary for the swap search
        h_bin = (best_h > 0.5).astype(float)
        # Adjust to have exactly n/2 ones
        n_ones = int(h_bin.sum())
        target_ones = best_n // 2
        if n_ones > target_ones:
            # Remove excess ones (pick ones with highest "pressure")
            c_vals = compute_c5_vec(h_bin, best_n)
            ones_idx = np.where(h_bin == 1)[0]
            # Sort ones by their local contribution
            scores = np.array([np.sum(c_vals) for _ in ones_idx])
            # Just remove random ones
            excess = n_ones - target_ones
            to_remove = rng.choice(ones_idx, size=excess, replace=False)
            h_bin[to_remove] = 0.0
        elif n_ones < target_ones:
            deficit = target_ones - n_ones
            zeros_idx = np.where(h_bin == 0)[0]
            to_add = rng.choice(zeros_idx, size=deficit, replace=False)
            h_bin[to_add] = 1.0
        
        h_bin = project_feasible(h_bin, best_n)
        c_bin = compute_c5(h_bin, best_n)
        
        p2_end = min(time.time() + 200, deadline - 250)
        h2, c2 = greedy_binary_search(h_bin, best_n, p2_end, max_rounds=30000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = best_n
    
    # Phase 3: Active set optimization (fractional)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 150, deadline - 180)
        h3, c3 = active_set_optimize(best_h, best_n, p3_end, top_k=8, 
                                      alphas=[100000, 500000, 2000000], rounds=15)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Subgradient polishing
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 100, deadline - 130)
        h4, c4 = subgradient_method(best_h, best_n, p4_end, alpha=5000, lr_start=0.005)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: More binary search (fine)
    if time.time() < deadline - 100:
        h_bin2 = (best_h > 0.5).astype(float)
        n_ones2 = int(h_bin2.sum())
        target_ones2 = best_n // 2
        if n_ones2 > target_ones2:
            excess = n_ones2 - target_ones2
            ones_idx = np.where(h_bin2 == 1)[0]
            to_remove = rng.choice(ones_idx, size=excess, replace=False)
            h_bin2[to_remove] = 0.0
        elif n_ones2 < target_ones2:
            deficit = target_ones2 - n_ones2
            zeros_idx = np.where(h_bin2 == 0)[0]
            to_add = rng.choice(zeros_idx, size=deficit, replace=False)
            h_bin2[to_add] = 1.0
        h_bin2 = project_feasible(h_bin2, best_n)
        
        p5_end = min(time.time() + 80, deadline - 80)
        h5, c5 = greedy_binary_search(h_bin2, best_n, p5_end, max_rounds=20000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: L-BFGS final polish
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 35, deadline - 30)

        if c_hi < best_c + 0.01:  # Only if promising

            if c7 < best_c:

        n_ones3 = int(h_bin3.sum())
        target_ones3 = best_n // 2
        if n_ones3 > target_ones3:
            excess = n_ones3 - target_ones3
            ones_idx = np.where(h_bin3 == 1)[0]
            to_remove = rng.choice(ones_idx, size=excess, replace=False)
