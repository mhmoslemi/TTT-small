def greedy_lag_swap(h, n, rag_idx, step_size, rng):
    """Try to reduce overlap at a specific lag by swapping values."""
    lag = rag_idx - (n - 1)
    if lag == 0:
        return h, False
    m = abs(lag)
    if m >= n:
        return h, False
    
    h = h.copy()
    improved = False
    
    if lag > 0:
        # overlap = sum h[i+m]*(1-h[i]) for i=0..n-1-m
        # To reduce: decrease h[i+m] where h[i] is small, or increase h[i] where h[i+m] is large
        # Or: move value from position i+m (where h is high and h[i] is low) to position i
        for _ in range(200):
            # Score each position for potential improvement
            # If h[i+m] is high and h[i] is low: moving from i+m to i helps
            score = h[m:] * (1 - h[:-m])  # current overlap contribution per position
            # We want to reduce high-scoring positions
            # Move: decrease h[i+m], increase h[i]
            top_positions = np.argsort(-score)[:5]
            for p in top_positions:
                i = p
                j = p + m
                if h[j] > h[i] + 1e-6:
                    d = min(step_size, (h[j] - h[i]) * 0.3)
                    if d > 1e-9:
                        h[j] -= d
                        h[i] += d
                        h = project_feasible(h, n)
                        improved = True
                        break
    else:
        m2 = m
        # overlap = sum h[i]*(1-h[i+m2]) for i=0..n-1-m2
        score = h[:-m2] * (1 - h[m2:])
        top_positions = np.argsort(-score)[:5]
        for p in top_positions:
            i = p
            j = p + m2
            if h[i] > h[j] + 1e-6:

                if d > 1e-9:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    improved = True
                    break
    
    return h, improved

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth_val(c, alpha, k):
    idx = np.argpartition(c, -k)[-k:]
    top = c[idx]
    m = top.max()
    shifted = alpha * (top - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val), idx, m

def obj_grad_topk(x, n, alpha, k, penalty_coeff):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    obj_val, idx, m = topk_smooth_val(c, alpha, k)
    diff = x.sum() - n / 2.0
    obj_val += penalty_coeff * diff * diff
    top = c[idx]
    shifted = alpha * (top - m)
    mx = shifted.max()
    if mx > 500:
        w_top = np.exp(shifted - mx)
    else:
        w_top = np.exp(shifted)
    w_top = w_top / w_top.sum()
    grad = np.zeros(n)
    for ii in range(len(idx)):
        w = w_top[ii]
        if w < 1e-15:
            continue
        lag = idx[ii] - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += w * (1.0 - x[:-lag])
            grad[:-lag] += w * (-x[lag:])
        else:
            m2 = -lag
            grad[:-m2] += w * (1.0 - x[m2:])
            grad[m2:] += w * (-x[:-m2])
    grad *= dx
    grad += 2.0 * penalty_coeff * diff
    return obj_val, grad

def lbfgs_phase(h, n, alpha, k, deadline, maxiter, penalty_coeff=1e6):
    def func(x):
        return obj_grad_topk(x, n, alpha, k, penalty_coeff)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h, method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10,
                                'maxls': 40, 'iprint': -1})
        h_out = project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        h_out = project_feasible(h.copy(), n)
    return h_out

def binary_swap_delta(h, n, p, q, top_lags):
    """Compute change in top lag overlaps when h[p]:1->0, h[q]:0->1."""
    deltas = np.zeros(len(top_lags))
    for li, lag_idx in enumerate(top_lags):
        m = lag_idx - (n - 1)
        if m == 0:
            continue
        am = abs(m)
        if am >= n:
            continue
        # h[p] going 1->0
        dp = 0.0
        if p >= am:
            dp -= (1.0 - h[p - am])
        if p < n - am:
            dp += h[p + am]
        # h[q] going 0->1
        dq = 0.0
        if q >= am:
            dq += (1.0 - h[q - am])
        if q < n - am:
            dq -= h[q + am]
        deltas[li] = dp + dq
    return deltas

def binary_local_search(h_init, n, deadline, rng, K=5, max_iter=50000):
    """Efficient binary local search with multi-lag awareness."""
    h = (h_init > 0.5).astype(float)
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        one_idx = np.where(h == 1)[0]
        for _ in range(num_ones - target):
            order = rng.choice(len(one_idx), min(10, len(one_idx)), replace=False)
            best_p = one_idx[order[0]]
            best_c = compute_c5(h, n)
            h[best_p] = 0
            for p2 in one_idx[order]:
                h[p2] = 0
                c_test = compute_c5(h, n)
                h[p2] = 1
                if c_test < best_c:
                    best_c = c_test
                    best_p = p2
            h[best_p] = 0
            one_idx = np.where(h == 1)[0]
    elif num_ones < target:
        zero_idx = np.where(h == 0)[0]
        for _ in range(target - num_ones):
            order = rng.choice(len(zero_idx), min(10, len(zero_idx)), replace=False)
            for q in zero_idx[order]:
                h[q] = 1
                c_test = compute_c5(h, n)
                h[q] = 0
            h[zero_idx[order[0]]] = 1
            zero_idx = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    one_set = np.where(h == 1)[0]
    zero_set = np.where(h == 0)[0]
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_k = np.argpartition(c, -K)[-K:]
        
        # Try random swaps
        found = False
        num_trials = min(200, max(20, len(one_set) * len(zero_set) // 100))
        for _ in range(num_trials):
            p = one_set[rng.randint(len(one_set))]
            q = zero_set[rng.randint(len(zero_set))]
            deltas = binary_swap_delta(h, n, p, q, top_k)
            # Check if this swap reduces the estimated max
            current_top_vals = c[top_k]
            new_top_vals = current_top_vals + deltas
            est_max = max(np.max(new_top_vals), c.max() * 0.99)
            if est_max < best_c * n / 2.0 - 0.001:  # rough threshold
                # Apply swap
                h[p] = 0
                h[q] = 1
                c_actual = compute_c5(h, n)
                if c_actual < best_c - 1e-15:
                    best_c = c_actual
                    best_h = h.copy()
                    one_set = np.where(h == 1)[0]
                    zero_set = np.where(h == 0)[0]
                    no_improve = 0
                    found = True
                    break
                else:
                    h[p] = 1
                    h[q] = 0
        if not found:
            no_improve += 1
            if no_improve > 500:
                # Perturb and restart
                flip_count = rng.randint(5, 20)
                for _ in range(flip_count):
                    p = one_set[rng.randint(len(one_set))]
                    q = zero_set[rng.randint(len(zero_set))]
                    h[p] = 0
                    h[q] = 1
                best_c = compute_c5(h, n)
                best_h = h.copy()
                one_set = np.where(h == 1)[0]
                zero_set = np.where(h == 0)[0]
                no_improve = 0
    
    return best_h, best_c

    h = h.copy()

    no_improve = 0

            break

                        if abs(d) < 1e-8:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_t = compute_c5(h, n)
                        if c_t < best_c - 1e-15:
                            best_c = c_t
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
                    if improved:
                        break
                continue
            
            m = abs(lag)
            if m >= n:
                continue
            # Targeted swap at this lag
            if lag > 0:
                score = h[m:] * (1.0 - h[:-m])
            else:
                score = h[:-m] * (1.0 - h[m:])
            top_positions = np.argsort(-score)[:8]
            for pos in top_positions:
                i = pos
                j = pos + m
                if h[j] > h[i] + 1e-5:
                    d = min(0.04, (h[j] - h[i]) * 0.3, 0.5)
                elif h[i] > h[j] + 1e-5:
                    d = -min(0.04, (h[i] - h[j]) * 0.3, 0.5)
                else:
                    continue
                if abs(d) < 1e-9:
                    continue
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_t = compute_c5(h, n)
                if c_t < best_c - 1e-15:
                    best_c = c_t
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
            if improved:
                break
        
        # Random small swaps
        if not improved:
            for _ in range(20):
                lag_r = top_k[rng.randint(K)]
                lag_val = lag_r - (n - 1)
                if lag_val == 0:
                    lag_val = rng.randint(1, n)
                m_r = abs(lag_val)
                if m_r >= n:

                i = rng.randint(0, n - m_r)
                j = i + m_r
                if h[i] > h[j] + 1e-5:
                    d = min(0.015, (h[i] - h[j]) * 0.25)
                    h[i] -= d
                    h[j] += d
                elif h[j] > h[i] + 1e-5:

                    h[j] -= d
                else:
                    continue
                h = project_feasible(h, n)
                c_t = compute_c5(h, n)
                if c_t < best_c - 1e-15:
                    best_c = c_t
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    if h[i] - h[j] > -d and h[i] + h[j] < 1:
                        pass
                    # Revert
                    if h[i] < 0.5 and h[j] > 0.5:
                        pass
                    # Just re-project
                    h = project_feasible(best_h, n)
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if no_improve > 200:
            # Restart from best with perturbation
            h = project_feasible(best_h + rng.normal(0, 0.02, n), n)

    return best_h, best_c
