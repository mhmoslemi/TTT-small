Function active_gradients(c_indices, h, n):
M=2n-1, off=n-1.
Initialize grad zeros n.
For each idx in c_indices:
   d=idx-off
   if d>=0:
      nv=n-d
      # as first: i=0..nv-1, d=i+d? derivative + (1 - h[i+d])
      grad[0:nv] += (1.0 - h[d:n])
      # as second: i=d..n-1, j=i-d? derivative - h[i-d]
      grad[d:n] += -h[0:nv]

      grad[kk:n] += (1.0 - h[0:nv])

import time
import numpy as np
from scipy.optimize import minimize, linprog


def compute_c_counts(h):
    arr = np.asarray(h)
    if np.issubdtype(arr.dtype, np.integer):
        arr = arr.astype(np.int32, copy=False)
        return np.correlate(arr, 1 - arr, mode="full").astype(np.int32, copy=False)
    arr = arr.astype(np.float64, copy=False)
    return np.correlate(arr, 1.0 - arr, mode="full")


def c5_from_c(c, n):
    return float(np.max(c)) * (2.0 / float(n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    if len(h) != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    diff = target - s

    if abs(diff) <= 1e-12 * max(1.0, float(n)):
        hp = h
    else:
        lo = -1.0
        hi = 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            v = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
            if v < target:
                lo = mid
            else:
                hi = mid
        hp = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(hp.sum())
    if abs(diff) > 1e-14:
        room = np.minimum(hp, 1.0 - hp)
        idx = int(np.argmax(room))
        if room[idx] > 1e-12:
            hp[idx] += diff
        else:
            if diff > 0:
                room = 1.0 - hp
                tot = float(room.sum())
                if tot > 1e-15:
                    hp += diff * room / tot
            else:
                tot = float(hp.sum())
                if tot > 1e-15:
                    hp += diff * hp / tot
    return np.clip(hp, 0.0, 1.0)


def upsample_piecewise(h0, n0, n):
    if n == n0:
        return np.asarray(h0, dtype=np.float64).copy()
    idx = (np.arange(n) * n0) // n
    return np.asarray(h0, dtype=np.float64)[idx].copy()


def topk_binary(h, n):
    h = project_feasible(h, n)
    target = int(n // 2)
    x = np.zeros(n, dtype=np.int8)
    if target <= 0:
        return x
    if target >= n:
        x.fill(1)
        return x
    part = np.argpartition(-h, target - 1)[:target]
    x[part] = 1
    return x


def make_random_binary(n, rng):
    x = np.zeros(n, dtype=np.int8)
    target = int(n // 2)
    if target > 0:
        idx = rng.choice(n, size=target, replace=False)
        x[idx] = 1
    return x


def make_mechanical(n, alpha, beta):
    t = (np.arange(n, dtype=np.float64) * alpha + beta) % 1.0
    x = t < 0.5
    target = int(n // 2)
    cnt = int(np.sum(x))
    if cnt > target:
        idx = np.where(x)[0]
        flip = idx[np.argsort(-t[idx])[:cnt - target]]
        x[flip] = False
    elif cnt < target:
        idx = np.where(~x)[0]
        flip = idx[np.argsort(t[idx])[:target - cnt]]
        x[flip] = True
    return x.astype(np.int8)


def active_gradient(indices, h, n):
    h = np.asarray(h, dtype=np.float64)
    grad = np.zeros(n, dtype=np.float64)
    off = n - 1
    for idx in indices:
        d = int(idx) - off
        if d <= 0:
            kk = -d
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:n] += 1.0 - h[0:nv]
            grad[0:nv] += -h[kk:n]
        else:
            nv = n - d
            if nv <= 0:
                continue
            grad[0:nv] += 1.0 - h[d:n]
            grad[d:n] += -h[0:nv]
    return grad


def select_active(c, limit):
    cmax = float(np.max(c))
    tol = max(1e-10, 0.002 * cmax)
    idx = np.where(c >= cmax - tol)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    if len(idx) > limit:
        idx = idx[np.argsort(-c[idx])[:limit]]
    return idx


def smooth_max_standard(vals, alpha):
    vals = np.asarray(vals, dtype=np.float64)
    mx = float(vals.max())
    s = alpha * (vals - mx)
    smx = float(s.max())
    if smx > 400:
        return mx + (smx + float(np.log(np.sum(np.exp(s - smx))))) / alpha
    return mx + float(np.log(np.sum(np.exp(s)))) / alpha


def smooth_weights(vals, alpha):
    vals = np.asarray(vals, dtype=np.float64)
    mx = float(vals.max())
    s = alpha * (vals - mx)
    smx = float(s.max())
    w = np.exp(s - smx)
    w = w / float(w.sum())
    return w


def topk_obj(x, n, active, alpha, mu):
    c = compute_c_counts(x)
    vals = c[active]
    val = smooth_max_standard(vals, alpha)
    if mu > 0.0:
        diff = float(np.sum(x)) - 0.5 * float(n)
        val += mu * diff * diff
    return float(val)


def topk_grad(x, n, active, alpha, mu):
    c = compute_c_counts(x)
    vals = c[active]
    w = smooth_weights(vals, alpha)
    active = list(active)
    g = np.zeros(n, dtype=np.float64)
    for wi, idx in zip(w, active):
        if wi <= 1e-15:
            continue
        gg = active_gradient(np.array([idx], dtype=int), x, n)
        g += float(wi) * gg
    if mu > 0.0:
        g += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return g


def lbfgs_active_polish(h, n, deadline, active_limit=28):
    h = project_feasible(h, n)
    c = compute_c_counts(h)
    best_h = h.copy()
    best_c = c5_from_c(c, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 2000, 20000, 150000]

    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 25:
            break
        active = select_active(c, active_limit)
        if len(active) == 0:
            break
        maxit = min(180, max(40, int(rem * 1.2)))
        try:
            res = minimize(
                topk_obj,
                h,
                jac=topk_grad,
                args=(n, active, alpha, 1.0),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxit, "ftol": 1e-22, "gtol": 1e-15,
                         "maxfun": maxit * 8, "maxls": 30, "maxcor": 45},
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c_counts(h_new)
            val_new = c5_from_c(c_new, n)
            if val_new < best_c - 1e-14:
                best_c = val_new
                best_h = h_new.copy()
                h = h_new.copy()
                c = c_new
            elif val_new <= best_c + 1e-14:
                h = h_new.copy()
                c = c_new

            pass
    return best_h, best_c

        b_ub = -np.asarray(f, dtype=np.float64)

def binary_eval_swap_max(c, x_rev, x_minus_1, one_minus_x, i, j, n):
    off = n - 1

    c[off - i: off - i + n] += x_minus_1
    c[off + i - n + 1: off + i + 1] += x_rev
    c[off + j - n + 1: off + j + 1] -= x_rev
    c[off - j: off - j + n] += one_minus_x
    c[off + j - i] += 1
    c[off + i - j] += 1
    c[off] = 0

    new_max = int(c.max())

    c[off] = 0
    c[off - i: off - i + n] -= x_minus_1
    c[off + i - n + 1: off + i + 1] -= x_rev
    c[off + j - n + 1: off + j + 1] += x_rev
    c[off - j: off - j + n] -= one_minus_x
    c[off + j - i] -= 1
    c[off + i - j] -= 1

    return new_max


def binary_commit_swap(c, x, x_rev, x_minus_1, one_minus_x, i, j, n):
    off = n - 1
    c[off - i: off - i + n] += x_minus_1
    c[off + i - n + 1: off + i + 1] += x_rev
    c[off + j - n + 1: off + j + 1] -= x_rev
    c[off - j: off - j + n] += one_minus_x
    c[off + j - i] += 1
    c[off + i - j] += 1
    c[off] = 0
    x[i] = 0
    x[j] = 1


def binary_local_search(x0, deadline, rng, top_pairs=45, random_pairs=55):

    T = 2.5
    no_improve = 0

        rnd += 1
        if cur_max > best_max + 6:
            x = best_x.copy()
            c = compute_c_counts(x).astype(np.int32)
            cur_max = best_max
            T = 2.2

        if len(active) == 0:

            break

        top_ones = ones[np.argsort(-grad[ones])[:top_pairs]]
        top_zeros = zeros[np.argsort(grad[zeros])[:top_pairs]]

        x_rev = x[::-1]
        x_minus_1 = x - 1
        one_minus_x = 1 - x

        impr_val = cur_max
        impr_i = -1
        impr_j = -1
        alt_val = cur_max
        alt_i = -1
        alt_j = -1

        for i in top_ones:
            for j in top_zeros:
                v = binary_eval_swap_max(c, x_rev, x_minus_1, one_minus_x, int(i), int(j), n)
                if v < impr_val:
                    impr_val = v
                    impr_i = int(i)
                    impr_j = int(j)
                if v < alt_val:
                    alt_val = v
                    alt_i = int(i)
                    alt_j = int(j)

        for _ in range(random_pairs):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            v = binary_eval_swap_max(c, x_rev, x_minus_1, one_minus_x, i, j, n)
            if v < impr_val:
                impr_val = v
                impr_i = i
                impr_j = j
            if v < alt_val:
                alt_val = v
                alt_i = i
                alt_j = j

        if impr_i >= 0:
            cand_i, cand_j, cand_val = impr_i, impr_j, impr_val
        elif alt_i >= 0:
            cand_i, cand_j, cand_val = alt_i, alt_j, alt_val
        else:
            break

        if cand_val < cur_max:
            binary_commit_swap(c, x, x_rev, x_minus_1, one_minus_x, cand_i, cand_j, n)
            cur_max = int(c.max())
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_x = x.copy()
            T = max(0.45, T * 0.995)
        elif cand_val == cur_max:
            if no_improve > 18 and rng.random() < 0.18:
                binary_commit_swap(c, x, x_rev, x_minus_1, one_minus_x, cand_i, cand_j, n)
                cur_max = int(c.max())
                no_improve = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_x = x.copy()
                T = max(0.45, T * 0.99)
            else:
                no_improve += 1
                T = max(0.0, T * 0.84)
        else:
            accept = False
            if T > 0.55 and rng.random() < float(np.exp(-float(cand_val - cur_max) / T)):
                accept = True
            if accept:
                binary_commit_swap(c, x, x_rev, x_minus_1, one_minus_x, cand_i, cand_j, n)
                cur_max = int(c.max())
                no_improve = max(0, no_improve // 2)
                if cur_max < best_max:
                    best_max = cur_max
                    best_x = x.copy()
                T = max(0.45, T * 0.99)
            else:
                no_improve += 1
                T = max(0.0, T * 0.84)

        if no_improve > 150:
            x = best_x.copy()
            c = compute_c_counts(x).astype(np.int32)
            cur_max = best_max
            no_improve = 0
            T = 2.0
            if rnd > 800:
                break

    return best_x, (2.0 * float(best_max) / float(n))


def binary_multi_start(n, deadline, rng, parent_h=None):
    best_x = None
    best_c = float("inf")
    if n <= 0 or n % 2 != 0:
        return best_x, best_c

    starts = []

    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(topk_binary(parent_h, n))
        else:
            p = project_feasible(parent_h, len(parent_h))
            starts.append(topk_binary(upsample_piecewise(p, len(p), n), n))

    starts.append(make_mechanical(n, 0.6180339887, 0.17))
    starts.append(make_mechanical(n, 0.4142135624, 0.31))
    starts.append(make_random_binary(n, rng))

    # De-duplicate trivially by byte content.
    uniq = []
    seen = set()
    for arr in starts:
        arr = np.asarray(arr, dtype=np.int8)
        key = arr.tobytes()
        if key not in seen:
            seen.add(key)
            uniq.append(arr)
    starts = uniq

    now = time.time()
    total_available = max(0.0, deadline - now - 20)
    if total_available < 20:
        return best_x, best_c

    per = max(15.0, int(total_available / float(len(starts))))
    for arr in starts:
        t_end = min(deadline - 5, time.time() + per)
        if time.time() >= t_end:
            break
        x, cval = binary_local_search(arr, t_end, rng, top_pairs=38, random_pairs=45)
        if cval < best_c:
            best_c = cval
            best_x = x

    if best_x is None:
        x = make_random_binary(n, rng)
        best_c = c5_from_c(compute_c_counts(x), n)
        best_x = x
    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s) - 18.0)
    rng = np.random.RandomState(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    best_h = None
    best_c = float("inf")
    best_n = 0

    def _update(h, c, n):
        nonlocal best_h, best_c, best_n
        if h is None or n <= 0:
            return
        h = project_feasible(h, n)
        c = c5_from_c(compute_c_counts(h), n)
        if c < best_c:
            best_h = h.copy()
            best_c = c
            best_n = int(n)

    if parent is not None and len(parent) > 0:
        n0 = int(len(parent))
        h0 = project_feasible(parent, n0)
        _update(h0, c5_from_c(compute_c_counts(h0), n0), n0)
    else:
        n0 = 500
        h0 = make_mechanical(n0, 0.6180339887, 0.17).astype(np.float64)
        _update(h0, c5_from_c(compute_c_counts(h0), n0), n0)

    # Phase 1: polish the supplied parent at its native resolution.
    phase1_end = min(deadline - 420, t0 + 260)
    if phase1_end - time.time() > 40:
        h1, c1 = refine_candidate(h0, n0, phase1_end, binary_seconds=70, rng=rng)
        _update(h1, c1, n0)

    # Phase 2: double resolution from the improved parent.
    if time.time() < deadline - 320:
        n2 = int(max(512, 2 * n0))
        if n2 > 2500:
            n2 = 1000
        if n2 % 2:
            n2 += 1
        src = best_h if best_n == n0 else h0
        h2 = project_feasible(upsample_piecewise(src, best_n if best_n == n0 else n0, n2), n2)
        phase2_end = min(deadline - 240, time.time() + 360)
        if phase2_end - time.time() > 60:
            h2b, c2b = refine_candidate(h2, n2, phase2_end, binary_seconds=170, rng=rng)
            _update(h2b, c2b, n2)

    # Phase 3: dedicated binary search at a good even resolution.
    if time.time() < deadline - 180:
        bin_n = best_n if (best_n % 2 == 0 and 0 < best_n <= 2200) else 1000
        if bin_n % 2:
            bin_n -= 1
        src = best_h if best_n == bin_n else (best_h if best_h is not None else h0)
        hbin_parent = project_feasible(upsample_piecewise(src, len(src), bin_n), bin_n)
        phase3_end = min(deadline - 120, time.time() + 180)
        if phase3_end - time.time() > 60:
            xbin, cbin = binary_multi_start(bin_n, phase3_end, rng, parent_h=hbin_parent)
            if xbin is not None:
                _update(xbin.astype(np.float64), cbin, bin_n)
                if cbin < best_c:
                    # Polish the binary result continuously.
                    if time.time() < deadline - 40:
                        hf, cf = lp_polish(best_h, best_n, deadline - 20,
                                           max_iters=6,
                                           active_count=16 if best_n > 1000 else 26)
                        _update(hf, cf, best_n)

    # Phase 4: quadruple resolution if the parent grid is small enough.
    if time.time() < deadline - 150:
        n4 = int(max(1024, 4 * n0))
        if n4 > 2400:
            n4 = 2000
        if n4 % 2:
            n4 -= 1
        src = best_h if best_n == n4 else h0
        src_len = best_n if best_n == n4 else n0
        h4 = project_feasible(upsample_piecewise(src, src_len, n4), n4)
        phase4_end = min(deadline - 90, time.time() + 160)
        if phase4_end - time.time() > 50:
            h4b, c4b = refine_candidate(h4, n4, phase4_end, binary_seconds=70, rng=rng)
            _update(h4b, c4b, n4)

    # Phase 5: final short polish of the global champion.
    if best_h is not None and time.time() < deadline - 8:
        hf, cf = lp_polish(best_h, best_n, deadline - 4,
                           max_iters=6,
                           active_count=14 if best_n > 1000 else 24)
        _update(hf, cf, best_n)

    if best_h is None:
        best_h = project_feasible(h0, n0)
        best_c = c5_from_c(compute_c_counts(best_h), n0)
        best_n = n0

    best_h = project_feasible(best_h, best_n)
    best_c = c5_from_c(compute_c_counts(best_h), best_n)
    return best_h, float(best_c), int(best_n)
