import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def gradient_for_lag(h, m, n):
    """Compute gradient of f(m) = sum h[j]*(1-h[j+m]) for positive lag m."""
    grad = np.zeros(n)
    for j in range(n - m):
        hj = h[j]
        hjm = h[j + m]
        # d/dh[j]: (1-hjm)
        # d/dh[j+m]: -hj
        grad[j] += (1 - hjm)
        grad[j + m] += -hj
    return grad * (2.0 / n)

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e10):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            obj_val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        else:
            obj_val = m + np.log(np.sum(np.exp(shifted))) / alpha
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        
        # Gradient of smooth max
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[k] * (-x[0:n - lag])
                grad[0:n - lag] += w[k] * (1.0 - x[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w[k] * (1.0 - x[idx:n])
                grad[idx:n] += w[k] * (-x[0:n - idx])
        grad *= dx
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

                j = worst + m

                i = worst + m
                j = worst

**New idea: Alternating Lag Reduction (ALR)**
- Identify the top 3 lags
- For each, compute the "optimal redistribution" along that lag
- Alternate between lags
- This is similar to coordinate descent but on the "lag" variable rather than the "position" variable

For a fixed positive lag m, the contribution is:
f_m = Σ_{j=0}^{n-1-m} h[j]*(1-h[j+m])*dx

If I only vary h along the "chains" defined by lag m (i.e., positions j, j+m, j+2m, ...), I can optimize f_m while keeping the sum constraint (approximately).

But this gets complicated because changing h[j] also affects other lags.

Simpler approach: **Gradient-free "hill climbing" on the worst lag only**
- Find the worst lag m
- For many random pairs (i, i+m), try small transfers
- Accept only if the global max decreases
- Repeat until stuck
- Then move to the next-worst lag

This is essentially what the previous "intensive local search" does, but I'll make it more focused and with smaller steps.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_raw_lags(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (-h[0:n - lag])
            grad[0:n - lag] += w[k] * (1.0 - h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e10):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            obj_val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        else:
            obj_val = m + np.log(np.sum(np.exp(shifted))) / alpha
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff

        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()

        grad = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[k] * (-x[0:n - lag])
                grad[0:n - lag] += w[k] * (1.0 - x[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w[k] * (1.0 - x[idx:n])
                grad[idx:n] += w[k] * (-x[0:n - idx])
        grad *= dx
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def top_lag_surgical(h, n, deadline, top_k=5, small_step=0.0005, med_step=0.002, large_step=0.008):
    """Focused surgical reduction of the top few lags with adaptive steps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    stall = 0

    while time.time() < deadline:
        c_raw = compute_raw_lags(h)
        c_all = c_raw * dx
        max_c = c_all.max()

        if max_c < best_c - 1e-15:
            best_c = float(max_c)
            best_h = h.copy()
            stall = 0
        elif stall > 0:
            h = best_h.copy()

        sorted_idx = np.argsort(c_raw)[::-1]
        top_lags = sorted_idx[:top_k]

        improved = False
        for k_idx in top_lags:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            if lag > 0:
                terms = h[:n - m] * (1.0 - h[m:n])
            else:
                terms = h[m:n] * (1.0 - h[:n - m])

            # Try top candidates
            candidates = np.argsort(terms)[::-1][:20]
            for pos in candidates:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i = pos
                    j = pos + m
                else:
                    i = pos + m
                    j = pos

                for step in [small_step, med_step, large_step]:
                    delta = step * min(max(h[i], 0.01), max(1.0 - h[j], 0.01))
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < max_c - 1e-16:
                        max_c = c_new
                        improved = True
                        break
                    else:
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h, n)
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb slightly
                perturb = np.random.RandomState(int(time.time() * 1000) % 100000)
                hp = best_h + perturb.normal(0, 0.0005, n)
                hp = project_feasible(hp, n)
                if compute_c5(hp, n) <= best_c + 0.0001:
                    h = hp
                stall = 0
                if stall > 50:
                    break

    return best_h, best_c

def multi_resolution_search(h_init, n_init, deadline, rng):
    """Try multiple resolutions and return the best."""
    best_h = h_init.copy()

    # Try n=1000
    for n_try in [1000, 800]:

def cooldown_local_search(h, n, deadline, rng):
    """Adaptive-step local search focused on reducing the single worst lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]
    current_size_idx = 2
    no_improve = 0

    while time.time() < deadline and no_improve < 2000:
        c_raw = compute_raw_lags(h)
        max_idx = np.argmax(c_raw)
        max_val = c_raw[max_idx]
        max_c = max_val * (2.0 / n)

        if max_c < best_c - 1e-15:
            best_c = float(max_c)
            best_h = h.copy()
            no_improve = 0

        lag = max_idx - (n - 1)
        if lag == 0:
            lag2_idx = np.argsort(c_raw)[-5:]
            lag2_found = False
            for ki in lag2_idx:
                lag = ki - (n - 1)
                if lag != 0:
                    lag2_found = True
                    break
            if not lag2_found:
                break

        m = abs(lag)
        if m >= n:
            break

        if lag > 0:
            terms = h[:n - m] * (1.0 - h[m:n])
        else:
            terms = h[m:n] * (1.0 - h[:n - m])

        step = step_sizes[current_size_idx]
        pos = np.argmax(terms)

        if lag > 0:
            i = pos
            j = pos + m
        else:
            i = pos + m
            j = pos

        delta = step * min(max(h[i], 0.005), max(1.0 - h[j], 0.005))
        if delta < 1e-8:
            no_improve += 1
            current_size_idx = max(0, current_size_idx - 1)
            continue

        h[i] -= delta
        h[j] += delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)

        if c_new < max_c - 1e-16:

            if current_size_idx < len(step_sizes) - 1 and no_improve < 50:
                pass  # Keep current step
        else:
            h[i] += delta
            h[j] -= delta
            h = project_feasible(h, n)
            no_improve += 1
            if no_improve > 200:
                current_size_idx = max(0, current_size_idx - 1)
                no_improve = 0
                if current_size_idx == 0:
                    # Try random restarts
                    h = best_h + rng.normal(0, 0.001, n)

                    if compute_c5(h, n) > best_c + 0.001:

        # Occasionally try second-worst lag
        if int(time.time() * 10) % 7 == 0:
            c_raw2 = compute_raw_lags(h)
            mask = c_raw.copy()
            mask[np.argmax(mask)] = -1
            second_idx = np.argmax(mask)
            second_lag = second_idx - (n - 1)
            if second_lag != 0:
                sm = abs(second_lag)
                if sm < n:
                    if second_lag > 0:
                        st = h[:n - sm] * (1.0 - h[sm:n])
                    else:
                        st = h[sm:n] * (1.0 - h[:n - sm])
                    sp = np.argmax(st)
                    if second_lag > 0:
                        si, sj = sp, sp + sm
                    else:
                        si, sj = sp + sm, sp
                    sd = 0.001 * min(max(h[si], 0.005), max(1.0 - h[sj], 0.005))
                    if sd > 1e-8:
                        h[si] -= sd
                        h[sj] += sd
                        h = project_feasible(h, n)
                        if compute_c5(h, n) > best_c * 1.0001:
                            h[si] += sd
                            h[sj] -= sd
                            h = project_feasible(h, n)

    return best_h, best_c
