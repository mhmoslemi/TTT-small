    return grad

def sqp_lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=500, rho_init=0.05, max_active=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active_mask = c_vec >= thresh
        active = list(np.where(active_mask)[0])
        if not active:
            break
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        delta, t_opt = sqp_lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            consecutive_no_improve = 0
            if rho < 0.1:
                rho = min(rho * 1.5, 0.12)
        else:
            consecutive_no_improve += 1
            rho *= 0.5
            if rho < 1e-10:
                break
            if consecutive_no_improve > 30:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 10.0), (2000, 100.0), (10000, 1000.0), (50000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_local_search(h, n, deadline, center_radius=60):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:20]:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= min(len(active), 20)
            top_count = min(center_radius * 3, n // 2)
            top_pos = np.argsort(-grad)[:top_count]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a)
        return float(m + np.log(np.sum(np.exp(sh))) / a)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def perturb_h(h, n, magnitude, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    h_p += rng.uniform(-magnitude, magnitude, size=n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Quick LBFGS on initial resolution (25s)
    t1 = min(t0 + 25, deadline - 800)
    if t1 > time.time() + 3:
        best_h, best_c = run_lbfgs(best_h, best_n, t1,
            configs=[(200, 50.0), (1000, 200.0), (5000, 2000.0)])

    # Phase 2: SQP on initial resolution (20s)
    t2 = min(t0 + 45, deadline - 750)
    if t2 > time.time() + 3:
        best_h, best_c = sqp_optimize(best_h, best_n, t2, max_iter=200, rho_init=0.05)

    # Phase 3: Upsample to n=1000 and optimize (120s)
    n2 = 1000
    t3 = min(t0 + 165, deadline - 500)
    if t3 > time.time() + 10:
        h2 = upsample(best_h, best_n, n2)
        c2 = compute_c5(h2, n2)
        h2, c2 = run_lbfgs(h2, n2, min(t3 - 40, t3),
            configs=[(500, 50.0), (3000, 500.0), (10000, 2000.0), (50000, 10000.0)])
        t3b = min(time.time() + 40, t3)
        if t3b > time.time() + 5:
            h2, c2 = sqp_optimize(h2, n2, t3b, max_iter=300, rho_init=0.05, max_active=40)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2

    # Phase 4: Upsample to n=2000 and optimize (250s)
    n3 = 2000
    t4 = min(t0 + 415, deadline - 250)
    if t4 > time.time() + 20:
        h3 = upsample(best_h, best_n, n3)
        c3 = compute_c5(h3, n3)
        t4a = min(time.time() + 120, t4 - 60)
        h3, c3 = run_lbfgs(h3, n3, t4a,
            configs=[(500, 50.0), (3000, 300.0), (10000, 2000.0), (50000, 10000.0)])
        t4b = min(time.time() + 50, t4 - 20)
        if t4b > time.time() + 3:
            h3, c3 = sqp_optimize(h3, n3, t4b, max_iter=200, rho_init=0.04, max_active=50)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n3

    # Phase 5: Try perturbation on best solution and re-optimize (80s)
    t5 = min(t0 + 495, deadline - 100)
    if t5 > time.time() + 10:
        for pert_mag in [0.01, 0.02, 0.005]:
            if time.time() > t5 - 5:
                break
            h_p = perturb_h(best_h, best_n, pert_mag, seed + 100)
            c_p = compute_c5(h_p, best_n)
            if c_p < best_c:
                h_p, c_p = run_lbfgs(h_p, best_n, min(time.time() + 15, t5 - 3),
                    configs=[(5000, 2000.0), (20000, 5000.0)])
                h_p, c_p = sqp_optimize(h_p, best_n, min(time.time() + 10, t5 - 2),
                    max_iter=100, rho_init=0.03)
                if c_p < best_c:
                    best_c = c_p
                    best_h = h_p

    # Phase 6: Fine local search (60s)
    t6 = min(t0 + 555, deadline - 50)
    if t6 > time.time() + 5:
        best_h, best_c = fine_local_search(best_h, best_n, t6, center_radius=40)

    # Phase 7: Final SQP polish (40s)
    t7 = min(t0 + 595, deadline - 20)
    if t7 > time.time() + 3:
        best_h, best_c = sqp_optimize(best_h, best_n, t7, max_iter=200, rho_init=0.02, max_active=50)

    # Phase 8: SLSQP final polish
    if time.time() < deadline - 5:
        h_f, c_f = slqp_polish(best_h, best_n, deadline - 2, alpha=5000000, maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
