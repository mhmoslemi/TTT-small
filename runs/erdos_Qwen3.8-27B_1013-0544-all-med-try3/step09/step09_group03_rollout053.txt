```python
def compute_swap_delta(h, a, b, n):
    """Compute change in full correlation when h[a] goes 0→1 and h[b] goes 1→0."""
    s = np.arange(-(n-1), n)  # All shifts
    delta = np.zeros(2*n-1)
    
    # h[a]: 0→1
    idx = a - s  # position in h for the first role
    valid1 = (idx >= 0) & (idx < n)
    delta[valid1] += (1 - h[idx[valid1]])
    
    idx2 = a + s  # position in h for the second role
    valid2 = (idx2 >= 0) & (idx2 < n)
    delta[valid2] -= h[idx2[valid2]]
    
    # h[b]: 1→0
    idx3 = b - s
    valid3 = (idx3 >= 0) & (idx3 < n)
    delta[valid3] += (h[idx3[valid3]] - 1)
    
    idx4 = b + s
    valid4 = (idx4 >= 0) & (idx4 < n)
    delta[valid4] += h[idx4[valid4]]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 20000.0), (200000, 80000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def binarize(h, n):
    h_b = np.where(h >= 0.5, 1.0, 0.0).copy()
    target = n / 2.0
    diff = target - h_b.sum()
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros])[:int(np.ceil(abs(diff)))]
        h_b[zeros[order]] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])[:int(np.ceil(abs(diff)))]
        h_b[ones[order]] = 0.0
    # Final fix
    if h_b.sum() != target:
        # Adjust one element
        if h_b.sum() < target:
            idx = np.where(h_b == 0)[0][0]
            h_b[idx] = 1.0
        else:
            idx = np.where(h_b == 1)[0][0]
            h_b[idx] = 0.0
    return h_b

def swap_delta(h, a, b, n):
    """Compute ΔC vector when h[a]:0→1 and h[b]:1→0."""
    s = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    idx1 = a - s
    v1 = (idx1 >= 0) & (idx1 < n)
    delta[v1] += (1.0 - h[idx1[v1]])
    
    idx2 = a + s
    v2 = (idx2 >= 0) & (idx2 < n)
    delta[v2] -= h[idx2[v2]]
    
    idx3 = b - s
    v3 = (idx3 >= 0) & (idx3 < n)
    delta[v3] += (h[idx3[v3]] - 1.0)
    
    idx4 = b + s
    v4 = (idx4 >= 0) & (idx4 < n)
    delta[v4] += h[idx4[v4]]
    
    return delta

def binary_local_search(h, n, deadline, temperature=0.0):
    """Binary local search with optional simulated annealing."""
    h_b = binarize(h, n)
    C = compute_c5_vec(h_b, n)
    best_max = C.max()
    best_h = h_b.copy()
    best_C = C.copy()
    rng = np.random.RandomState(777)
    
    zeros = np.where(h_b == 0)[0]
    ones = np.where(h_b == 1)[0]
    
    stall_count = 0
    T = temperature
    
    while time.time() < deadline - 5:
        improved = False
        # Shuffle for diversity
        zero_list = rng.permutation(zeros)
        one_list = rng.permutation(ones)
        
        # Limit candidates per pass for speed
        n_candidates = min(5000, len(zero_list) * len(one_list))
        
        for ci in range(n_candidates):
            if time.time() > deadline - 5:
                break
            a = zero_list[ci % len(zero_list)]
            b = one_list[ci % len(one_list)]
            
            dc = swap_delta(h_b, a, b, n)
            new_max = np.max(C + dc)
            
            if new_max < best_max - 1e-15:
                # Accept
                h_b[a] = 1.0
                h_b[b] = 0.0
                C += dc
                best_max = max(C.max(), new_max)  # C.max() might be at a different shift
                best_max = C.max()
                best_h = h_b.copy()
                best_C = C.copy()
                improved = True
                
                # Update zero/one lists
                zeros = np.where(h_b == 0)[0]
                ones = np.where(h_b == 1)[0]
                zero_list = rng.permutation(zeros)
                one_list = rng.permutation(ones)
                
                if T > 0:
                    T *= 0.999
            elif T > 0 and new_max - best_max < 0.01:
                # Simulated annealing: accept worse with prob exp(-Δ/T)
                if rng.rand() < np.exp(-(new_max - best_max) / T):
                    h_b[a] = 1.0
                    h_b[b] = 0.0
                    C += dc
                    improved = True
                    zeros = np.where(h_b == 0)[0]
                    ones = np.where(h_b == 1)[0]
                    zero_list = rng.permutation(zeros)
                    one_list = rng.permutation(ones)
                    T *= 0.999
        
        if not improved:
            stall_count += 1
            if stall_count > 5:
                # Let's try a bigger kick
                T = max(T, 0.005)
                stall_count = 0
            if T == 0 and stall_count > 20:
                break
        else:
            stall_count = 0
    
    # Restore best
    if best_max < C.max():
        h_b = best_h.copy()
    
    return h_b, best_max

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    
    def sq_jac_fn(x, na):
        return np.ones(na)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
```
