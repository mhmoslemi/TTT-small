```
def bound_push(h, n, eps):
    y = np.asarray(h, dtype=float).copy()
    m0 = y < eps
    m1 = y > 1.0 - eps
    if not (m0.any() or m1.any()):
        return y
    y[m0] = 0.0
    y[m1] = 1.0
    target = 0.5 * float(n)
    free = ~(m0 | m1)
    for _ in range(100):
        diff = target - y.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - y
            mask = free & (room > 1e-14)
            if not np.any(mask): break
            y[mask] += (diff / room[mask].sum()) * room[mask]
        else:
            mask = free & (y > 1e-14)
            if not np.any(mask): break
            y[mask] += (diff / y[mask].sum()) * y[mask]
        y = np.clip(y, 0.0, 1.0)
    return y
```

import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    n = int(n)
    target = 0.5 * float(n)
    x = np.clip(x, 0.0, 1.0)

    if abs(x.sum() - target) < 1e-13:
        return x

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        sy = float(y.sum())
        if sy > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(50):
        diff = target - float(y.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - y
            mask = room > 1e-15
            if not np.any(mask):
                break
            y[mask] += (diff / float(room[mask].sum())) * room[mask]
        else:
            mask = y > 1e-15
            if not np.any(mask):
                break
            y[mask] += (diff / float(y[mask].sum())) * y[mask]
        y = np.clip(y, 0.0, 1.0)

    return y


def c5_with_values(h, n):
    n = int(n)
    dx = 2.0 / float(n)
    counts = np.correlate(h, 1.0 - h, mode="full")
    vals = counts * dx
    return float(np.max(vals)), vals


def default_mu(n):
    n = int(n)
    if n <= 700:
        return 1.0
    if n <= 1400:
        return 0.3
    return 0.1


def make_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))]


def make_sp_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))] + [(0.05, 0.7)]


def weighted_obj(x, n, Wdx, mu):
    n = int(n)
    counts = np.correlate(x, 1.0 - x, mode="full")
    val = float(np.dot(Wdx, counts))
    diff = float(x.sum()) - 0.5 * float(n)
    val += float(mu) * diff * diff
    return val


def weighted_grad(x, n, Wdx, mu):
    n = int(n)
    g = np.zeros(n, dtype=float)
    c = n - 1

    wc = float(Wdx[c])
    if wc != 0.0:
        g += wc * (1.0 - 2.0 * x)

    if n > 1:
        A = Wdx[:c][::-1]
        Bw = Wdx[c + 1:]
        C = A + Bw

        if C.size and np.any(C != 0.0):
            prefA = np.cumsum(A)
            suffix = np.zeros(n, dtype=float)
            suffix[:c] = prefA[::-1]

            prefix = np.zeros(n, dtype=float)
            prefix[1:n] = np.cumsum(Bw)

            g += suffix + prefix

            kernel = np.zeros(2 * n - 1, dtype=float)
            kernel[:c] = C[::-1]
            kernel[c + 1:] = C

            conv = np.convolve(x, kernel, mode="full")
            g -= conv[c:c + n]

    if float(mu) != 0.0:
        g += 2.0 * float(mu) * (float(x.sum()) - 0.5 * float(n))

    return g


def run_weighted_stage(x, n, Wdx, mu, maxiter, deadline):
    n = int(n)
    x = project_feasible(x, n)
    if time.time() > deadline - 10.0:
        return x
    maxiter = max(50, int(maxiter))

    try:
        res = minimize(
            weighted_obj,
            x,
            jac=weighted_grad,
            args=(n, Wdx, float(mu)),
            method="L-BFGS-B",
            bounds=make_bounds(n),
            options={
                "maxiter": maxiter,
                "ftol": 1e-18,
                "gtol": 1e-14,
                "maxcor": 12,
                "maxls": 50,
            },
        )
        if res is not None and getattr(res, "x", None) is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return x


def smooth_log_obj(x, n, alpha, mu):
    n = int(n)
    dx = 2.0 / float(n)
    counts = np.correlate(x, 1.0 - x, mode="full")
    B = counts * dx
    m = float(np.max(B))
    sh = float(alpha) * (B - m)
    mx = float(np.max(sh))
    ex = np.exp(sh - mx)
    s = float(ex.sum())
    if s <= 0.0:
        s = 1.0
    val = m + (mx + float(np.log(s))) / float(alpha)
    diff = float(x.sum()) - 0.5 * float(n)
    val += float(mu) * diff * diff
    return float(val)


def smooth_log_grad(x, n, alpha, mu):
    n = int(n)
    dx = 2.0 / float(n)
    counts = np.correlate(x, 1.0 - x, mode="full")
    B = counts * dx
    m = float(np.max(B))
    sh = float(alpha) * (B - m)
    mx = float(np.max(sh))
    ex = np.exp(sh - mx)
    s = float(ex.sum())
    if s <= 0.0:
        p = np.zeros_like(B)
        p[int(np.argmax(B))] = 1.0
    else:
        p = ex / s
    Wdx = dx * p
    return weighted_grad(x, n, Wdx, float(mu))

def softplus_values(z):
    z = np.asarray(z, dtype=float)
    out = np.empty_like(z)
    pos = z > 0.0
    out[pos] = z[pos] + np.log1p(np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = np.log1p(ez)
    return out


def sp_obj(y, n, beta, mu):
    n = int(n)
    dx = 2.0 / float(n)
    h = y[:n]
    c = float(y[n])
    counts = np.correlate(h, 1.0 - h, mode="full")
    B = counts * dx
    z = float(beta) * (B - c)
    sp = softplus_values(z)
    val = c + float(sp.sum()) / float(beta)
    diff = float(h.sum()) - 0.5 * float(n)
    val += float(mu) * diff * diff
    return float(val)


def sp_grad(y, n, beta, mu):
    n = int(n)
    dx = 2.0 / float(n)
    h = y[:n]
    c = float(y[n])
    counts = np.correlate(h, 1.0 - h, mode="full")
    B = counts * dx
    z = float(beta) * (B - c)

    sig = np.empty_like(z)
    pos = z > 0.0
    e = np.exp(-z[pos])
    sig[pos] = 1.0 / (1.0 + e)
    ez = np.exp(z[~pos])
    sig[~pos] = ez / (1.0 + ez)

    Wdx = dx * sig
    gh = weighted_grad(h, n, Wdx, float(mu))
    gc = 1.0 - float(sig.sum())

    g = np.empty(n + 1, dtype=float)
    g[:n] = gh
    g[n] = gc
    return g

    try:
        res = minimize(
            sp_obj,

    return x


def bound_push(h, n, eps):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * float(n)

    m0 = h < float(eps)
    m1 = h > 1.0 - float(eps)
    if not (np.any(m0) or np.any(m1)):
        return h

    h[m0] = 0.0
    h[m1] = 1.0
    free = ~(m0 | m1)

    for _ in range(100):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            mask = free & (room > 1e-14)
            if not np.any(mask):
                break
            h[mask] += (diff / float(room[mask].sum())) * room[mask]
        else:
            mask = free & (h > 1e-14)
            if not np.any(mask):
                break
            h[mask] += (diff / float(h[mask].sum())) * h[mask]
        h = np.clip(h, 0.0, 1.0)

    return h


def active_weighted_polish(h, n, deadline):
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c, vals = c5_with_values(h, n)
    mu = default_mu(n)

    base_maxit = 900 if n <= 800 else (650 if n <= 1400 else 450)
    tol_list = [5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5]
    stalls = 0

    for tol in tol_list:
        if time.time() > deadline - 10.0:
            break
        if stalls >= 6:
            break

        improved = False
        for _ in range(2):
            if time.time() > deadline - 10.0:
                break

            c, vals = c5_with_values(best_h, n)
            cmax = float(np.max(vals))
            low = cmax - float(tol)
            mask = vals > low
            excess = vals - low

            p = np.zeros(vals.shape, dtype=float)
            p[mask] = excess[mask]
            s = float(p.sum())

            if s <= 1e-18:
                idx = int(np.argmax(vals))
                p[idx] = 1.0
                s = 1.0
            else:
                top = np.argsort(vals)[::-1][:5]
                p[top] += 1e-6 / max(1.0, float(len(top)))
                s = float(p.sum())

            p = p / s
            Wdx = (2.0 / float(n)) * p

            maxit = int(base_maxit * (0.70 if tol < 1e-3 else 1.0))
            h_new = run_weighted_stage(best_h, n, Wdx, mu, maxit, deadline - 5.0)
            h_new = project_feasible(h_new, n)
            c_new, _ = c5_with_values(h_new, n)

            if c_new < c - 1e-12:
                best_h = h_new
                best_c = float(c_new)
                improved = True
            else:
                break

        if not improved:
            stalls += 1
        else:
            stalls = 0

    for eps in [1e-3, 1e-4, 1e-5]:
        if time.time() > deadline - 5.0:
            break
        hp = bound_push(best_h, n, eps)
        cp, _ = c5_with_values(hp, n)
        if cp < best_c:
            best_h = hp
            best_c = float(cp)

    return best_h, best_c


def sym_version(h):
    return 0.5 * (h + h[::-1])


def random_perturb(h, n, seed, mag=0.02):
    n = int(n)
    rng = np.random.RandomState(seed)
    y = h + float(mag) * (rng.rand(n) - 0.5)
    return project_feasible(y, n)


def upsample_linear(h, n_new):
    n_old = int(len(h))
    n_new = int(n_new)
    if n_new == n_old:
        return project_feasible(h, n_new)

    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(y, n_new)


def optimize_pipeline(h, n, deadline, lite=False):
    n = int(n)
    h = project_feasible(h, n)

    c_traj, _ = c5_with_values(h, n)
    h_traj = h.copy()

    best_h = h.copy()
    best_c = float(c_traj)

    cfgs = logsum_configs(n)
    if lite:
        cfgs = cfgs[-3:] if len(cfgs) > 3 else cfgs

    for alpha, mu, maxit in cfgs:
        if time.time() > deadline - 20.0:
            break
        h_new = run_logsum_stage(h_traj, n, alpha, mu, maxit, deadline)
        h_new = project_feasible(h_new, n)
        c_new, _ = c5_with_values(h_new, n)

        if c_new < c_traj - 1e-12:
            h_traj = h_new
            c_traj = float(c_new)
        elif c_new < c_traj + 1e-5:
            h_traj = h_new
            c_traj = float(c_new)

        if c_new < best_c:
            best_h = h_new
            best_c = float(c_new)

    cfgs = sp_configs(n)
    if lite:
        cfgs = cfgs[-2:] if len(cfgs) > 2 else cfgs

    for beta, mu, maxit in cfgs:
        if time.time() > deadline - 25.0:
            break
        h_new = run_sp_stage(h_traj, n, beta, mu, maxit, deadline)
        h_new = project_feasible(h_new, n)

        if c_new < c_traj - 1e-12:

            h_traj = h_new

        if c_new < best_c:

            best_c = float(c_new)

    if time.time() < deadline - 45.0:
