import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    one_minus_x = 1.0 - x
    for k in range(n):
        wm = w[k + n - 1] * dx
        if wm < 1e-14:
            continue
        grad[:n - k] += wm * one_minus_x[k:]
        grad[k:] -= wm * x[:n - k]
    for k in range(1, n):
        wm = w[n - 1 - k] * dx
        if wm < 1e-14:
            continue
        grad[k:] += wm * one_minus_x[:n - k]
        grad[:n - k] -= wm * x[k:]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    one_minus_x = 1.0 - x
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            grad[:n - k] += wm * one_minus_x[k:]
            grad[k:] -= wm * x[:n - k]
        else:
            kk = -k
            grad[kk:] += wm * one_minus_x[:n - kk]
            grad[:n - kk] -= wm * x[kk:]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs_sequence(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        est_iter_time = 0.004 * (n / 500.0) ** 2
        maxit = max(200, int(remaining * 0.75 / est_iter_time))
        maxit = min(maxit, 20000)
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-22, 'maxfun': maxit * 15,
                                    'maxls': 50, 'maxcor': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)

def run_subgradient(h, n, deadline, n_iter=20000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h_curr = best_h.copy()
    eta = 0.005
    best_eta = eta

    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 50 == 0 and it > 0:
            eta = max(0.0001, eta * 0.995)

        c_vec = compute_c5_vec(h_curr, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        dx_val = 2.0 / n
        grad = np.zeros(n)
        one_minus_x = 1.0 - h_curr
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                grad[:n - k] += one_minus_x[k:] * dx_val
                grad[k:] -= h_curr[:n - k] * dx_val
            else:
                kk = -k
                grad[kk:] += one_minus_x[:n - kk] * dx_val
                grad[:n - kk] -= h_curr[kk:] * dx_val
        grad /= len(active)

        h_new = h_curr - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            best_eta = eta
        else:
            eta *= 0.95
        h_curr = h_new
    return best_h, best_c


def run_slqp_polish(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    def slqp_obj(x, nn, alpha):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)

    def slqp_eq_fun(x, nn):
        return x.sum() - nn / 2.0

    def slqp_eq_jac(x, nn):
        return np.ones(nn)

    for alpha in [100000, 1000000, 10000000]:
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(50, int(remaining * 0.8 / 0.1))
        maxit = min(maxit, 5000)
        cons = [{'type': 'eq', 'fun': slqp_eq_fun, 'jac': slqp_eq_jac, 'args': (n,)}]
        try:
            res = minimize(slqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-22})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def coord_descent(h, n, deadline):
    best_h = project_feasible(h, n).copy()
    best_c = compute_c5(best_h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

    for d_idx, d in enumerate(deltas):
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                for j in range(max(0, i - 20), min(n, i + 21)):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return project_feasible(best_h, n), best_c


def binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] = remaining
    return project_feasible(h_bin, n)


def random_feasible(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.rand(n)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 0: Quick exploration at low resolution to find diverse starts
    if time.time() < deadline - 100:
        n_explore = 200
        candidates = [h0[:n_explore].copy() if n_orig >= n_explore else upsample(h0, n_orig, n_explore)]
        for s in [1, 7, 42, 123, 999, 2024]:
            candidates.append(random_feasible(n_explore, s))
        # Also try binary and smoothed versions
        candidates.append(binary_start(h0, n_orig)[:n_explore].copy() if n_orig >= n_explore else binary_start(upsample(h0, n_orig, n_explore), n_explore))

        for cand in candidates:
            cand = project_feasible(cand, n_explore)
            c_cand = compute_c5(cand, n_explore)
            # Upsample to n_orig and evaluate
            cand_up = upsample(cand, n_explore, n_orig)
            c_up = compute_c5(cand_up, n_orig)
            if c_up < best_c:
                best_c = c_up
                best_h = cand_up

    # Phase 1: L-BFGS-B at original resolution
    p1_end = min(t0 + 50, deadline - 200)
    h1, c1 = run_lbfgs_sequence(best_h, best_n, p1_end,
                                 alphas=[100, 500, 2000, 10000, 50000, 200000, 1000000],
                                 mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to working resolution
    n_work = 1000
    if time.time() < deadline - 300:
        h_work = upsample(best_h, best_n, n_work)
        c_work = compute_c5(h_work, n_work)
        if c_work < best_c + 0.005:
            best_h = h_work
            best_c = min(best_c, c_work)
            best_n = n_work
        else:
            # Keep working at original resolution if upsampled is much worse
            pass

    # Phase 3: Extended L-BFGS-B at working resolution
    n_main = best_n
    h_main = best_h.copy()
    p3_end = min(t0 + 400, deadline - 250)
    h_main, c_main = run_lbfgs_sequence(h_main, n_main, p3_end,
                                         alphas=[50, 200, 1000, 5000, 20000, 100000,
                                                  500000, 2000000, 10000000, 50000000],
                                         mu=0.005)
    if c_main < best_c:
        best_c = c_main
        best_h = h_main
        best_n = n_main

    # Phase 4: Active set push at main resolution
    p4_end = min(t0 + 650, deadline - 150)
    h_main, c_main = run_active_push(h_main, n_main, p4_end)
    if c_main < best_c:
        best_c = c_main
        best_h = h_main
        best_n = n_main

    # Phase 5: Try higher resolution
    if time.time() < deadline - 250 and best_n < 1500:
        n_high = 1500
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.005:
            p5_end = min(t0 + 780, deadline - 200)
            h_high, c_high = run_lbfgs_sequence(h_high, n_high, p5_end,
                                                 alphas=[100, 1000, 10000, 100000,
                                                         1000000, 10000000, 50000000],
                                                 mu=0.003)
            if c_high < best_c:
                best_c = c_high
                best_h = h_high
                best_n = n_high

    # Phase 6: Active set at highest resolution found
    p6_end = min(t0 + 880, deadline - 100)
    h_a, c_a = run_active_push(best_h, best_n, p6_end)
    if c_a < best_c:
        best_c = c_a
        best_h = h_a
        best_n = best_n

    # Phase 7: SLSQP polish
    if time.time() < deadline - 60:
        p7_end = min(t0 + 950, deadline - 40)
        h_s, c_s = run_slqp_polish(best_h, best_n, p7_end)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
            best_n = best_n

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 40:
        p8_end = min(t0 + 1020, deadline - 20)
        h_sub, c_sub = run_subgradient(best_h, best_n, p8_end, n_iter=100000)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub
            best_n = best_n

    # Phase 9: Final active set push
    if time.time() < deadline - 10:
        p9_end = deadline - 5
        h_f, c_f = run_active_push(best_h, best_n, p9_end)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
            best_n = best_n

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
