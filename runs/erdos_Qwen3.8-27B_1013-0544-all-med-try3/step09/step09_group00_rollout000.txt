def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp(c, alpha):

    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:

    return m + np.log(np.sum(np.exp(sh))) / alpha

            res = minimize(smooth_max_obj, best_h,
                          jac=lambda x: smooth_max_grad(x, n, alpha, mu, active),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def simulate_annealing(h, n, deadline, iters_per_temp=100, temps=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if temps is None:
        temps = [0.05, 0.03, 0.02, 0.01, 0.005]
    
    for temp in temps:
        h_cur = best_h.copy()
        c_cur = best_c
        for it in range(iters_per_temp):
            if time.time() > deadline - 5:
                break
            # Pick two random positions to swap
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            # Try moving h[i] to h[j] (or vice versa)
            amt = min(h_cur[i], 0.1)
            if amt < 1e-10:
                continue
            h_try = h_cur.copy()
            h_try[i] -= amt
            h_try[j] += amt
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            delta = c_try - c_cur
            if delta < 0 or np.random.random() < np.exp(-delta / (temp * 0.001)):
                h_cur = h_try
                c_cur = c_try
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
            # Also try the reverse
            if h_cur[j] > 1e-10:
                h_try2 = h_cur.copy()
                amt2 = min(h_cur[j], 0.1)
                h_try2[j] -= amt2
                h_try2[i] += amt2
                h_try2 = project_feasible(h_try2, n)
                c_try2 = compute_c5(h_try2, n)
                delta2 = c_try2 - c_cur
                if delta2 < 0 or np.random.random() < np.exp(-delta2 / (temp * 0.001)):
                    h_cur = h_try2
                    c_cur = c_try2
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
    return best_h, best_c
