        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def gradient_at_index(x, n, idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - x[k:n]) * dx
        grad[k:n] += (-x[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - x[0:nv]) * dx
        grad[0:nv] += (-x[kk:n]) * dx
    return grad

def fw_direction(c_vec, h, n, top_m=10):
    """Frank-Wolfe direction: find direction that most reduces the max."""
    m = c_vec.max()
    tol = max(m * 0.005, 1e-8)
    active = np.where(c_vec >= m - tol)[0]
    if len(active) > top_m:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_m]]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    
    # Compute gradients for active constraints
    grads = np.array([gradient_at_index(h, n, idx) for idx in active])
    # Average gradient direction (simple case)
    avg_grad = grads.mean(axis=0)
    
    # Project to feasible direction: subtract mean to maintain sum constraint
    avg_grad -= avg_grad.mean()
    
    # Check against bounds
    # Scale to unit step
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-15:
        return None, 0.0
    direction = -avg_grad / norm
    return direction, norm

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(150, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 12,
                                    'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=6, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000, 50000000]
    for rnd in range(80):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(80, int(rem2 * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 12,
                                        'maxls': 50, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def fw_refine(h, n, deadline, n_iter=5000):
    """Frank-Wolfe style subgradient with adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > 15:
            order = np.argsort(-c_vec[active])
            active = active[order[:15]]
        
        # Compute average gradient of active constraints
        grad = np.zeros(n)
        for idx in active:
            grad += gradient_at_index(best_h, n, idx)
        grad /= len(active)
        # Remove mean to maintain sum constraint approximately
        grad -= grad.mean()
        
        # Line search with a few trial steps
        best_step_c = best_c
        best_step_h = best_h
        for factor in [1.0, 0.5, 0.25, 0.1]:
            step = eta * factor
            h_trial = best_h - step * grad
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_step_c - 1e-16:
                best_step_c = c_trial
                best_step_h = h_trial
                break
        
        if best_step_c < best_c - 1e-16:
            best_c = best_step_c
            best_h = best_step_h
            eta = min(eta * 1.02, 0.015)
        else:
            eta *= 0.95
            if eta < 1e-6:
                break
    return best_h, best_c

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def multi_restart_opt(h, n, deadline, n_restarts=5):
    """Try perturbations and re-optimize from each."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    per_restart = max(10, (deadline - time.time()) / n_restarts - 5)
    
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        # Create perturbation
        noise = rng.randn(n) * 0.01
        noise -= noise.mean()  # maintain sum constraint approximately
        h_pert = project_feasible(best_h + noise, n)
        c_pert = compute_c5(h_pert, n)
        
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
        
        # Quick L-BFGS from perturbed point
        rt_end = min(time.time() + per_restart, deadline - 10)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(sm_obj, h_pert, jac=sm_grad,
                           args=(n, 500000, 0.005), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': int(per_restart * 3), 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': int(per_restart * 4),
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def final_squeeze(h, n, deadline):
    """Very aggressive final optimization with tiny steps."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Try progressively higher alpha
    for alpha in [1000000, 10000000, 100000000]:
        rem = deadline - time.time()
        if rem < 5:
            break
        maxit = max(50, int(rem * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, 0.0001), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-24,
                                    'gtol': 1e-20, 'maxfun': maxit * 5,
                                    'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Subgradient with very small steps
    h, c = fw_refine(best_h, n, deadline, n_iter=3000)
    if c < best_c:
        best_c = c
        best_h = h
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)

    # Upsample to high resolution
    n_target = 2000
    h = upsample(h0, n_orig, n_target)
    n = n_target
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS with progressive alpha (300s)
    p1_end = min(t0 + 280, deadline - 400)
    if p1_end > time.time() + 20:
        h1, c1 = run_lbfgs(best_h, n, p1_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.005, maxiters=[400, 600, 800, 1000, 1200, 1500, 2000, 2500, 3000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization (150s)
    p2_end = min(time.time() + 150, deadline - 250)
    if p2_end > time.time() + 15:
        h2, c2 = run_active_opt(best_h, n, p2_end, top_k=5,
            alphas=[50000, 200000, 1000000, 5000000, 20000000, 100000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Frank-Wolfe refinement (100s)
    p3_end = min(time.time() + 100, deadline - 150)
    if p3_end > time.time() + 10:
        h3, c3 = fw_refine(best_h, n, p3_end, n_iter=5000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Multi-restart exploration (100s)
    p4_end = min(time.time() + 100, deadline - 100)
    if p4_end > time.time() + 15:
        h4, c4 = multi_restart_opt(best_h, n, p4_end, n_restarts=4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Another active set pass (80s)
    p5_end = min(time.time() + 80, deadline - 50)
    if p5_end > time.time() + 10:
        h5, c5 = run_active_opt(best_h, n, p5_end, top_k=4,
            alphas=[200000, 2000000, 20000000, 200000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Final squeeze (remaining time)
    h6, c6 = final_squeeze(best_h, n, deadline - 2)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Final pair search with tiny steps
    if time.time() < deadline - 3:
        h7, c7 = pair_search(best_h, n, deadline - 1, radius=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
