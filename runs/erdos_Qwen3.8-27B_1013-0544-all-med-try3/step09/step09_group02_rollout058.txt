import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_shift(h, n, idx):
    """Gradient of C_idx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
        if k > 0 and nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
        if kk > 0 and nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
            if k > 0 and nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
            if kk > 0 and nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_step(h, n, active_idx, c_vec, rho):
    """One SQP step: solve LP to find direction and step."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    n_act = len(active_idx)
    # Constraints: grad_k^T * delta - t <= -C_k for active k
    #              -h[i] <= delta[i] <= 1-h[i] for all i
    #              -rho <= delta[i] <= rho for all i
    # Equality: sum delta = 0
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    
    dx = 2.0 / n
    for j in range(n_act):
        idx = active_idx[j]
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:

            if nv > 0:
                g[kk:n] = (1.0 - h[0:nv]) * dx

                g[0:nv] += -h[kk:n] * dx

        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0

def coord_descent(h, n, active_idx, max_sweeps=50, deadline=None):
    """Coordinate descent: optimize one variable at a time."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Process variables in order of potential impact
        c_vec = compute_c5_vec(h, n)
        grad_mag = np.zeros(n)
        for idx in active_idx:
            grad_mag += np.abs(compute_grad_shift(h, n, idx))
        grad_mag /= len(active_idx)
        order = np.argsort(-grad_mag)
        
        for ii in range(n):
            if deadline is not None and time.time() > deadline:
                break
            i = order[ii]
            # C_k as function of h[i]: C_k = B_k + slope_k * h[i]
            slopes = np.zeros(len(active_idx))
            intercepts = np.zeros(len(active_idx))
            for j, idx in enumerate(active_idx):
                g = compute_grad_shift(h, n, idx)
                slope_k = g[i]
                # C_k(h) = C_k(current) - slope_k * h[i] + slope_k * h_new[i]
                # = (C_k(current) - slope_k * h[i]) + slope_k * h_new[i]
                intercepts[j] = c_vec[idx] - slope_k * h[i]
                slopes[j] = slope_k
            
            # Minimize max_j (intercepts[j] + slopes[j] * x) over x in [0,1]
            # This is: find x* such that the max is minimized
            # If all slopes are 0, no change
            # Otherwise, the optimal x is where the two "tightest" constraints cross
            # or at a boundary
            
            # Try x = 0, x = 1, and crossings
            candidates = [0.0, 1.0]
            # Find crossings: slopes[j]*x + intercepts[j] = slopes[l]*x + intercepts[l]
            for j in range(len(active_idx)):
                for l in range(j+1, len(active_idx)):
                    denom = slopes[l] - slopes[j]
                    if abs(denom) > 1e-15:
                        x_cross = (intercepts[j] - intercepts[l]) / denom
                        if 0.0 <= x_cross <= 1.0:
                            candidates.append(x_cross)
            
            best_x = h[i]
            best_val = max(intercepts + slopes * best_x)
            for x_c in candidates:
                val = max(intercepts + slopes * x_c)
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x_c
            
            if abs(best_x - h[i]) > 1e-12:
                h[i] = best_x
                improved = True
        
        if not improved:
            break
        
        # Update active set
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.005 * max_val)
        new_active = list(np.where(c_vec >= thresh)[0])
        if len(new_active) > 30:
            top_order = np.argsort(-c_vec[new_active])
            new_active = [new_active[i] for i in top_order[:30]]
        if len(new_active) > 0:
            active_idx = new_active
        
        c_new = c_vec.max()
        if c_new < best_c:
            best_c = c_new
    
    return h, best_c

C_k(x) = intercept_k + slope_k * x

To minimize max_k C_k(x) over x ∈ [0,1]:
- The maximum of linear functions is convex
- The minimum is at the "kink" where two (or more) functions achieve the max simultaneously, or at a boundary

The subgradient of max_k at x is the set of all convex combinations of slopes of active functions. The optimal x* satisfies 0 ∈ subgradient, i.e., there exist weights w_k ≥ 0, Σw_k = 1, such that:
Σ w_k * slope_k = 0 (for interior x*)

And C_{k1}(x*) = C_{k2}(x*) for the active k's.

For 2 active functions: x* = (intercept_k2 - intercept_k1) / (slope_k1 - slope_k2)

For more functions, it's more complex, but the pairwise approach finds the right answer (the optimum is at a pairwise crossing or boundary).

                i = int(ii)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if time.time() > deadline:
                        return h, best_c
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_flag = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_flag = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def slqp_polish(h, n, deadline, alpha=3000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    def eq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def eq_jac(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fn, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'eps': 1e-12})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb(h, n, eps, rng):
    h_new = h + rng.uniform(-eps, eps, size=n)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    # Upsample to high resolution
    n_hi = 2000
    h = upsample(h_init, n_orig, n_hi)
    best_c = compute_c5(h, n_hi)
    best_h = h.copy()

    # Phase 1: L-BFGS coarse (120s)
    d1 = min(t0 + 120, deadline - 700)
    if d1 > time.time() + 5:
        h, best_c = run_lbfgs(h, n_hi, d1,
            configs=[(200, 30.0), (800, 300.0), (2500, 2000.0)])
        best_h = h.copy()

    # Phase 2: SQP main (250s)
    d2 = min(t0 + 370, deadline - 450)
    if d2 > time.time() + 10:
        h, bc2 = sqp_optimize(h, n_hi, d2, max_iter=500, rho_init=0.1, max_active=40)
        if bc2 < best_c:
            best_c = bc2
            best_h = h.copy()

    # Phase 3: Coordinate descent (150s)
    d3 = min(t0 + 520, deadline - 300)
    if d3 > time.time() + 5:
        h, bc3 = coord_descent_optimize(best_h, n_hi, d3, max_sweeps=100)
        if bc3 < best_c:
            best_c = bc3
            best_h = h.copy()

    # Phase 4: SQP refinement (200s)
    d4 = min(t0 + 720, deadline - 200)
    if d4 > time.time() + 10:
        h, bc4 = sqp_optimize(best_h, n_hi, d4, max_iter=400, rho_init=0.05, max_active=50)
        if bc4 < best_c:
            best_c = bc4
            best_h = h.copy()

    # Phase 5: Coordinate descent fine (100s)
    d5 = min(t0 + 820, deadline - 130)
    if d5 > time.time() + 5:
        h, bc5 = coord_descent_optimize(best_h, n_hi, d5, max_sweeps=200)
        if bc5 < best_c:
            best_c = bc5
            best_h = h.copy()

    # Phase 6: Local search (80s)
    d6 = min(t0 + 900, deadline - 60)
    if d6 > time.time() + 5:
        h, bc6 = pairwise_local_search(best_h, n_hi, d6, center_radius=80)
        if bc6 < best_c:
            best_c = bc6
            best_h = h.copy()

    # Phase 7: Random restarts + short SQP (60s)
    d7 = min(t0 + 960, deadline - 25)
    if d7 > time.time() + 20:
        for trial in range(3):
            if time.time() > d7 - 10:
                break
            h_try = perturb(best_h, n_hi, 0.02, rng)
            h_try, c_try = sqp_optimize(h_try, n_hi, min(time.time() + 15, d7 - 5),
                                        max_iter=60, rho_init=0.05, max_active=30)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()

    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 10:
        h, bf = slqp_polish(best_h, n_hi, deadline - 3, alpha=5000000, maxiter=2000)
        if bf < best_c:
            best_c = bf
            best_h = h.copy()

    # Also try at original resolution for comparison
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    # Check if upsampled original is better
    try:
        c_orig = compute_c5(project_feasible(h_init, n_orig), n_orig)
    except Exception:
        c_orig = best_c
    if c_orig < best_c:
        best_c = c_orig
        best_h = project_feasible(h_init, n_orig)
        n_final = n_orig

    return best_h, best_c, n_final
