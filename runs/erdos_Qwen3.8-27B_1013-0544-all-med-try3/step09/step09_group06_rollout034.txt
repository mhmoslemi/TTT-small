    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    
    while time.time() < deadline:
        t += 1
        g = compute_gradient_smooth_max(h_curr, n, alpha)
        
        m_t = betas[0] * m_t + (1 - betas[0]) * g
        v_t = betas[1] * v_t + (1 - betas[1]) * g**2
        m_hat = m_t / (1 - betas[0]**t)
        v_hat = v_t / (1 - betas[1]**t)

        if t % 500 == 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
        return w
    return w / s

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            grad[:] += w[k] * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[:n - m] += w[k] * (1.0 - h[m:]) * dx
            grad[m:] += w[k] * (-h[:n - m]) * dx
        else:
            grad[m:] += w[k] * (1.0 - h[:n - m]) * dx
            grad[:n - m] += w[k] * (-h[m:]) * dx
    return grad

def adam_optimize(h, n, deadline, alpha=500, lr=0.05):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        g = grad_smooth_max(h_curr, n, alpha)
        m_t = b1 * m_t + (1 - b1) * g
        v_t = b2 * v_t + (1 - b2) * g ** 2
        m_hat = m_t / (1 - b1 ** t)
        v_hat = v_t / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)

        h_curr = h_new
        if t % 2000 == 0:

                    c_all = compute_all_lags(h, n)
                else:
                    h[i] = 0.0
                    h[j] = 1.0
        
        if not improved:
            c_all = compute_all_lags(h, n)
            top_lags = np.argsort(c_all)[-10:]
            any_imp = False
            for k2 in top_lags:
                if time.time() > deadline:
                    break
                lag2 = k2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                if m2 >= n:
                    continue
                for i in range(n - m2):
                    if time.time() > deadline:
                        break
                    j = i + m2
                    if h[i] == 1.0 and h[j] == 0.0:
                        h[i] = 0.0
                        h[j] = 1.0
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            any_imp = True
                        else:
                            h[i] = 1.0
                            h[j] = 0.0
                    elif h[i] == 0.0 and h[j] == 1.0:
                        h[i] = 1.0
                        h[j] = 0.0
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            any_imp = True
                        else:
                            h[i] = 0.0
                            h[j] = 1.0
                if any_imp:
                    break
            if not any_imp:
                break
    
    return best_h, compute_c5(best_h, n)

def binarize(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Adam gradient descent
    if time.time() < deadline - 30:
        adam_deadline = min(time.time() + 40, deadline - 60)
        h_new, c_new = adam_optimize(h, n, adam_deadline, alpha=500, lr=0.05)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 2: More Adam with higher alpha
    if time.time() < deadline - 20:
        adam_deadline = min(time.time() + 30, deadline - 40)
        h_new, c_new = adam_optimize(h, n, adam_deadline, alpha=2000, lr=0.02)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase 3: SLSQP on active set
    if time.time() < deadline - 15:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_optimize(h, n, alpha=500000, maxiter=5000, top_m=top_m)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 4: Full SLSQP
    if time.time() < deadline - 10:
        h_new = slsqp_optimize(h, n, alpha=1000000, maxiter=3000, top_m=None)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Aggressive local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        best_h, best_c = local_search(best_h, n, ls_deadline)
    
    # Phase 6: Binary refinement
    if time.time() < deadline - 10:
        h_bin = binarize(best_h, n)
        bin_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.3)
        h_bin_best, c_bin = binary_local_search(h_bin, n, bin_deadline)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin_best
    
    # Phase 7: If binary was better, do fractional local search from it
    if time.time() < deadline - 5:
        frac_deadline = min(deadline - 2, time.time() + 20)
        best_h, best_c = local_search(best_h, n, frac_deadline)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Main optimization at initial resolution
    if h_init is not None:
        alloc = min(500, (deadline - t0) * 0.55)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc = min(200, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_resolution(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Try higher resolutions
    if time.time() < deadline - 80:
        n_hi = n_init + 100
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_hi)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_hi)
        alloc = min(200, (deadline - time.time()) * 0.25)
        if alloc > 50:
            h_opt, c_opt = optimize_at_resolution(h_interp, n_hi, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 40:
        n_restarts = max(1, int((deadline - time.time() - 20) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 20) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive local search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
