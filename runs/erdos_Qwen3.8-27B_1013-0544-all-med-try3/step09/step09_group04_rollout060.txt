def thue_morse_binary(n):
    # Thue-Morse sequence: t(0)=0, t(2n)=t(n), t(2n+1)=1-t(n)
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        # t(i) = popcount(i) mod 2
        tm[i] = bin(i).count('1') % 2
    # Make it have exactly n/2 ones
    n_ones = n // 2
    if tm.sum() == n_ones:
        return tm.astype(float)
    # If not, adjust: swap some bits
    h = tm.astype(float)
    # If too many ones, flip some 1s to 0s (preferring those with low "cost")
    # Simple approach: just set to 0 and 1
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if h.sum() > n_ones:
        # Flip excess ones to zero
        excess = int(h.sum()) - n_ones
        h[ones_idx[:excess]] = 0
    elif h.sum() < n_ones:
        # Flip some zeros to one
        deficit = n_ones - int(h.sum())
        h[zeros_idx[:deficit]] = 1
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def slse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii, idx in enumerate(active):
        wi = w[ii]
        if wi < 1e-15:

        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk

    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(42)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    while time.time() < deadline:
        step = steps[rng.integers(len(steps))]
        npert = int(rng.integers(1, 4))
        idxs = rng.choice(n, size=npert, replace=False)
        h_trial = best_h.copy()
        valid = True
        for idx in idxs:
            delta = float(rng.choice([-1.0, 1.0])) * step
            if h_trial[idx] + delta < 0.0 or h_trial[idx] + delta > 1.0:
                valid = False
                break
            h_trial[idx] += delta
        if not valid:
            continue
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:
            best_c = c_trial
            best_h = h_trial
    return best_h, best_c

def single_peak_dive(h, n, deadline, alpha=100000000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for it in range(50):
        rem = deadline - time.time()
        if rem < 8:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(n, active, alpha, 0.001), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': 8000,
                                    'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            else:
                break
        except Exception:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick polish at original resolution
    p1_end = min(t0 + 60, deadline - 500)
    h1, c1 = lbfgs_opt(best_h, best_n, p1_end,
                       alphas=[50, 500, 5000, 50000, 200000],
                       mu=0.01, maxiters_list=[400, 600, 800, 1000, 1500])
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Upsample to n=1000 and main optimization
    n_main = 1000
    p2_end = min(t0 + 450, deadline - 300)
    h_main = upsample(best_h, best_n, n_main)
    c_main = compute_c5(h_main, n_main)
    h2, c2 = lbfgs_opt(h_main, n_main, p2_end,
                       alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000],
                       mu=0.005,
                       maxiters_list=[400, 600, 800, 1000, 1200, 1500, 2000, 2500, 3000, 4000])
    if c2 < best_c:
        best_c, best_h, best_n = c2, h2, n_main

    # Phase 3: Active set optimization (aggressive)
    p3_end = min(time.time() + 180, deadline - 150)
    h3, c3 = active_opt(best_h, best_n, p3_end, top_k=5,
                        alphas=[100000, 500000, 2000000, 10000000, 50000000, 200000000])
    if c3 < best_c:
        best_c, best_h = c3, h3

    # Phase 4: Single peak dive
    p4_end = min(time.time() + 80, deadline - 100)
    h4, c4 = single_peak_dive(best_h, best_n, p4_end, alpha=100000000)
    if c4 < best_c:
        best_c, best_h = c4, h4

    # Phase 5: Try n=800 as alternative
    if time.time() < deadline - 200 and best_n != 800:
        h800 = upsample(best_h, best_n, 800)
        c800 = compute_c5(h800, 800)
        if c800 < best_c + 0.002:
            p5_end = min(time.time() + 120, deadline - 80)
            h5, c5 = lbfgs_opt(h800, 800, p5_end,
                               alphas=[200, 2000, 20000, 200000, 2000000, 20000000],
                               mu=0.005,
                               maxiters_list=[600, 1000, 1500, 2000, 2500, 3000])
            if c5 < best_c:
                best_c, best_h, best_n = c5, h5, 800

    # Phase 6: Gradient-based refinement
    p6_end = min(time.time() + 60, deadline - 30)
    h6, c6 = gradient_refine(best_h, best_n, p6_end)
    if c6 < best_c:
        best_c, best_h = c6, h6

    # Phase 7: Stochastic local search
    p7_end = min(time.time() + 80, deadline - 15)
    h7, c7 = stochastic_refine(best_h, best_n, p7_end)
    if c7 < best_c:
        best_c, best_h = c7, h7

    # Phase 8: Final peak dive
    if time.time() < deadline - 8:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 500000000, 0.001),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-25,
                                    'gtol': 1e-22, 'maxfun': 10000,
                                    'maxls': 50, 'maxcor': 50})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c, best_h = c_final, h_final
        except Exception:
            pass

    # Phase 9: One more gradient step
    if time.time() < deadline - 3:
        h9, c9 = gradient_refine(best_h, best_n, deadline - 1)
        if c9 < best_c:
            best_c, best_h = c9, h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
