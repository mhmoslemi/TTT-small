from scipy.optimize import minimize as scipy_minimize

def steepest_descent_direction(h, n, active_set):
    """Find the steepest descent direction in the subdifferential"""
    m = len(active_set)
    # Compute gradients
    G = np.column_stack([grad_component(h, n, idx) for idx in active_set])  # n x m
    
    # Minimize ||G @ λ||² subject to λ >= 0, sum(λ) = 1
    # This is a QP: min λ' G'G λ s.t. sum(λ)=1, λ>=0
    
    # Use scipy
    from scipy.optimize import minimize
    
    def q_obj(lam):
        g = G @ lam
        return np.dot(g, g)
    
    def q_grad(lam):
        g = G @ lam
        return 2 * G.T @ g
    
    bounds = [(0, 1)] * m
    cons = [{'type': 'eq', 'fun': lambda lam: lam.sum() - 1, 'jac': lambda: np.ones(m)}]
    
    lam0 = np.ones(m) / m
    res = scipy_minimize(q_obj, lam0, jac=q_grad, bounds=bounds, constraints=[cons[0]], method='SLSQP')
    
    if res.success:
        g_star = G @ res.x
    else:
        g_star = G @ (np.ones(m) / m)
    
    return -g_star  # descent direction

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def grad_component(h, n, idx):
    dx = 2.0 / n
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def active_gradient_avg(h, n, active_set):
    grad = np.zeros(n)
    for idx in active_set:
        grad += grad_component(h, n, idx)
    return grad / len(active_set)

def steepest_desc_dir(h, n, active_set):
    m = len(active_set)
    if m == 1:
        return -grad_component(h, n, active_set[0])
    G = np.concatenate([grad_component(h, n, idx) for idx in active_set]).reshape(n, m)
    H = G.T @ G
    def q_obj(lam):
        g = G @ lam
        return float(np.dot(g, g))
    def q_grad(lam):
        g = G @ lam
        return 2.0 * G.T @ g
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda: np.ones(m)}]
    lam0 = np.ones(m) / m
    try:
        res = minimize(q_obj, lam0, jac=q_grad, bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': 100, 'ftol': 1e-16})
        if res.success:
            return -(G @ res.x)
    except Exception:
        pass
    return -(G @ lam0)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def backtracking_ls(h, n, direction, c_current, max_lines=40):
    best_alpha = 0.0
    best_c = c_current
    best_h = h.copy()
    alpha = 1.0
    for _ in range(max_lines):
        h_trial = h + alpha * direction
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c:
            best_c = c_trial
            best_alpha = alpha
            best_h = h_trial
        alpha *= 0.6
        if alpha < 1e-12:
            break
    return best_h, best_c, best_alpha

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 0.001 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 0.002 * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, maxiters_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters_list and i < len(maxiters_list):
            maxit = maxiters_list[i]
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 8,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_pursuit(h, n, deadline, max_rounds=500, top_k_range=(3, 5)):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h = best_h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > top_k_range[1]:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k_range[1]]]
        if len(active) < top_k_range[0]:
            extra = np.argsort(-c_vec)[:(top_k_range[0] - len(active))]
            active = np.unique(np.concatenate([active, extra]))
        
        direction = steepest_desc_dir(h, n, active)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        
        h_new, c_new, _ = backtracking_ls(h, n, direction, c_max)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def single_constraint_dive(h, n, deadline, max_iter=300):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h = best_h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst = int(np.argmax(c_vec))
        c_max = c_vec[worst]
        direction = -grad_component(h, n, worst)
        h_new, c_new, _ = backtracking_ls(h, n, direction, c_max)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            n_steps = max(1, n // 200)
            for i in range(0, n, n_steps):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, n_steps):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=20000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h = best_h.copy()
    eta = 0.005
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) > 4:
            order = np.argsort(-c_vec[active])
            active = active[order[:4]]
        g = active_gradient_avg(h, n, active)
        h_new = project_feasible(h - eta * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.005, 0.02)
        else:
            eta *= 0.99
        h = h_new
    return best_h, best_c

def random_perturb(h, n, sigma=0.02):
    rng = np.random.default_rng(42)
    h_pert = h + sigma * rng.standard_normal(n)
    return project_feasible(h_pert, n)
