def compute_all_lags(h, n):

    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff/tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff/tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

    c = compute_all_lags(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)

        c_new = compute_c5(h_new, n)

            best_h = h_new.copy()
        h_curr = h_new

    c = compute_all_lags(x, n)

    return smooth_max_value(c, alpha)

    if top_m:
        top_idx = np.argsort(c)[-top_m:]
    else:
        top_idx = None

        return slsqp_obj(x, n, alpha, top_idx)
    def obj_grad(x):
        return slsqp_obj_grad(x, n, alpha, top_idx)
    def cons_fun(x):
        return np.sum(x) - n/2.0
    def cons_jac(x):
        return np.ones(n)

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h.copy()

        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        if t % 3000 == 0:
            lr *= 0.85
            if lr < 0.02:
                break
    return best_h, best_c

def _slsqp_obj(x, n, alpha, top_idx):
    c = compute_all_lags(x, n)
    if top_idx is not None:
        c = c[top_idx]
    return smooth_max_value(c, alpha)

def _slsqp_obj_grad(x, n, alpha, top_idx):
    dx = 2.0 / n
    c_all = compute_all_lags(x, n)
    if top_idx is not None:
        c = c_all[top_idx]
        w_sub = np.exp(alpha * (c - c.max()))
        w_sub /= w_sub.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_sub
    else:
        w_sub = np.exp(alpha * (c_all - c_all.max()))
        w_sub /= w_sub.sum()
        weights = w_sub
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += _lag_grad(x, n, lag, weights[k])
    return grad * dx

def _sum_eq(x, n):
    return np.sum(x) - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, top_m, maxiter=3000):
    c = compute_all_lags(h, n)
    if top_m is not None:
        top_idx = np.argsort(c)[-top_m:]
    else:
        top_idx = None
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h,
                      args=(n, alpha, top_idx),
                      jac=lambda x: _slsqp_obj_grad(x, n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimized_local_search(h, n, deadline, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        max_c = c_all.max()
        # Find all lags close to maximum
        threshold = max_c * 0.9995
        active_indices = np.where(c_all > threshold)[0]
        active_lags = []
        for ki in active_indices:
            lag = ki - (n - 1)
            if lag != 0:
                active_lags.append((lag, c_all[ki]))
        
        if not active_lags:
            break
        
        improved = False
        # Sort by correlation value descending
        active_lags.sort(key=lambda x: -x[1])
        
        for lag, _ in active_lags[:5]:
            if time.time() > deadline:
                break
            m = abs(lag)
            if m >= n:
                continue
            
            for step in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                if time.time() > deadline:
                    break
                # Try all positions
                if lag > 0:
                    for i in range(n - m):
                        if time.time() > deadline:
                            break
                        j = i + m
                        if h[i] > h[j] + 1e-9:
                            delta = min(step, (h[i] - h[j]) * 0.4, h[i] * 0.4, (1 - h[j]) * 0.4)
                            if delta < 1e-9:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                else:
                    for i in range(m, n):
                        if time.time() > deadline:
                            break
                        j = i - m
                        if h[j] > h[i] + 1e-9:
                            delta = min(step, (h[j] - h[i]) * 0.4, h[j] * 0.4, (1 - h[i]) * 0.4)
                            if delta < 1e-9:
                                continue
                            h[j] -= delta
                            h[i] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h.copy()
                                improved = True
                            else:
                                h[j] += delta
                                h[i] -= delta
                if improved:
                    h = best_h
        if not improved:
            break
    return best_h, best_c

def cross_lag_search(h, n, deadline, max_iter=20000):
    """Try moves that reduce the max considering multiple lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if m >= n:
            break
        
        # Find the pair with maximum contribution to this lag
        if lag > 0:
            contrib = h[0:n-m] * (1 - h[m:n])
        else:
            contrib = h[m:n] * (1 - h[0:n-m])
        
        # Try reducing the max contribution pair
        i_max = int(np.argmax(contrib))
        if lag > 0:
            i, j = i_max, i_max + m
        else:
            i, j = i_max + m, i_max
        
        # Move mass from higher to lower
        if h[i] > h[j]:
            for delta in [0.002, 0.001, 0.0005]:
                if delta >= h[i] or delta >= 1 - h[j]:
                    continue
                h[i] -= delta
                h[j] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
        elif h[j] > h[i]:
            for delta in [0.002, 0.001, 0.0005]:
                if delta >= h[j] or delta >= 1 - h[i]:
                    continue
                h[j] -= delta
                h[i] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    break
                else:
                    h[j] += delta
                    h[i] -= delta
        else:
            # Try random pair
            i2 = int(np.random.randint(0, n - m))
            j2 = i2 + m
            if h[i2] > h[j2]:
                delta = 0.001
                h[i2] -= delta
                h[j2] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i2] += delta
                    h[j2] -= delta
    
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase A: Momentum with moderate alpha
    dl_a = time.time() + (deadline - time.time()) * 0.25
    if time.time() < dl_a:
        h_new, c_new = momentum_loop(h, n, 5000, dl_a, lr=1.0, mu=0.9)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase B: Adam with higher alpha
    dl_b = time.time() + (deadline - time.time()) * 0.2
    if time.time() < dl_b:
        h_new, c_new = adam_loop(h, n, 20000, dl_b, lr=0.3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new
    
    # Phase C: SLSQP with top_m=1
    if time.time() < deadline - 20:
        for top_m in [1, 2, 3]:
            if time.time() > deadline - 15:
                break
            for alpha in [50000, 200000]:
                if time.time() > deadline - 10:
                    break
                h_new = slsqp_polish(h, n, alpha, top_m, maxiter=2000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
    
    # Phase D: Local search
    if time.time() < deadline - 10:
        ls_deadline = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = optimized_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase E: Cross-lag search
    if time.time() < deadline - 5:
        h_new, c_new = cross_lag_search(best_h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    best_h = None
    best_c = float('inf')
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    
    # Phase 1: Optimize at n_init (500)
    alloc1 = min(180, (deadline - t0) * 0.25)
    h1, c1 = optimize_at_resolution(h_init, n_init, t0 + alloc1)
    best_h, best_c = h1, c1
    
    # Phase 2: Try n=600
    n600 = 600
    x_old = np.linspace(0, 2, len(best_h))
    x_new = np.linspace(0, 2, n600)
    h_600 = np.interp(x_new, x_old, best_h)
    h_600 = project_feasible(h_600, n600)
    
    remaining = deadline - time.time() - 30
    if remaining > 120:
        alloc2 = min(250, remaining * 0.4)
        h2, c2 = optimize_at_resolution(h_600, n600, time.time() + alloc2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Try n=700 if time permits
    n700 = 700
    remaining = deadline - time.time() - 30
    if remaining > 150:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n700)
        h_700 = np.interp(x_new, x_old, best_h)
        h_700 = project_feasible(h_700, n700)
        alloc3 = min(200, remaining * 0.4)
        h3, c3 = optimize_at_resolution(h_700, n700, time.time() + alloc3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: If best is at n=500, try to improve with more local search
    n_best = len(best_h)
    remaining = deadline - time.time() - 10
    if remaining > 60:
        # More aggressive local search
        ls_dl = time.time() + min(remaining * 0.5, 200)
        h_ls, c_ls = optimized_local_search(best_h, n_best, ls_dl)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    # Phase 5: Final SLSQP polish
    remaining = deadline - time.time() - 8
    if remaining > 30:
        for top_m in [1, 2]:
            if time.time() > deadline - 8:
                break
            h_p = slsqp_polish(best_h, n_best, 100000, top_m, maxiter=3000)
            c_p = compute_c5(h_p, n_best)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p
    
    # Phase 6: Final micro local search
    if time.time() < deadline - 3:
        h_final, c_final = optimized_local_search(best_h, n_best, deadline - 1)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
