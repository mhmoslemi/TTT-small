import time

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def smax_grad(c, alpha, h, n):
    """Gradient of soft-max of c with respect to h"""
    w = soft_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_smax(h, n, alpha, mu=0.0):
    c = compute_c5_vec(h, n)
    val = soft_max_val(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def jac_smax(h, n, alpha, mu=0.0):
    c = compute_c5_vec(h, n)
    g = smax_grad(c, alpha, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def soft_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.exp(sh)
        s = w.sum()
    w /= s
    return w

def compute_smax_grad(c, alpha, h, n):
    w = soft_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_smax(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val = soft_max_value(c, alpha)
    if mu > 0:
        diff = h.sum() - n / 2.0
        val += mu * diff * diff
    return val

def jac_smax(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    g = compute_smax_grad(c, alpha, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def topk_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_smax, h, jac=jac_smax, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-24,
                                'gtol': 1e-18, 'maxfun': maxit * 12,
                                'maxls': 50, 'maxcor': 60})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def topk_optimize(h, n, deadline, active, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(topk_obj, h, jac=topk_grad, args=(n, active, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-24,
                                'gtol': 1e-18, 'maxfun': maxit * 10,
                                'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def get_active_lags(c_vec, max_val, frac, top_k):
    threshold = max_val - max(1e-10, frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    return active

def pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for d in deltas:
        improved = True
        iters = 0
        while improved and time.time() < deadline and iters < 500000:
            improved = False
            iters += 1
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                if lo == 0 and hi == n:
                    candidates = list(range(n))
                else:
                    candidates = []
                    for j in range(lo, hi):
                        if j != i:
                            candidates.append(j)
                if len(candidates) == 0:
                    continue
                for j in candidates:
                    if i == j:
                        continue
                    h_trial = best_h.copy()
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        h_trial[i] += d
                        h_trial[j] -= d
                        h_trial = project_feasible(h_trial, n)
                        c = compute_c5(h_trial, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_trial
                            improved = True
                            break
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        h_trial = best_h.copy()
                        h_trial[i] -= d
                        h_trial[j] += d
                        h_trial = project_feasible(h_trial, n)
                        c = compute_c5(h_trial, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_trial
                            improved = True
                            break
    return best_h, best_c

def grad_pair_search(h, n, deadline, step_sizes, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:

                    grad[k:n] += (-best_h[0:nv]) * dx_val

                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                gi = grad[i]
                for j in neg_idx[:50]:
                    g = gi + abs(grad[j])
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i] * 0.9, (1.0 - best_h[best_j]) * 0.9)
            if amt < 1e-12:

            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)

            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)

    return best_h, best_c

def binary_flip_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    ones = np.where(best_h > 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    attempts = 0
    while time.time() < deadline and attempts < 50000:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_trial = best_h.copy()
        h_trial[i] = 1.0 - h_trial[i]
        h_trial[j] = 1.0 - h_trial[j]
        h_trial = project_feasible(h_trial, n)
        c = compute_c5(h_trial, n)
        attempts += 1
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_trial
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        if attempts % 5000 == 0:
            break
    return best_h, best_c

def random_perturb_refit(h, n, rng, sigma, deadline):

    h_pert = project_feasible(h_pert, n)
    c = compute_c5(h_pert, n)
    alphas = [50000, 200000, 1000000]

        if c_new < c:

            h_pert = h_new
    return h_pert, c

def make_interpolated(h_parent, n_parent, n_new):
    x_parent = np.linspace(0, 2, n_parent, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_parent, h_parent, period=2.0)
    h_new = project_feasible(h_new, n_new)
    return h_new

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)
