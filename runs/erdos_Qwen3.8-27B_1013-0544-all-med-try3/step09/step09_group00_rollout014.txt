def project_binary(h, n):

    h = np.round(h)

    # Adjust to get exactly n/2 ones
    while len(ones) > n // 2:

        idx = ones[np.argmin(h[ones])]
        h[idx] = 0
        ones = np.delete(ones, np.where(ones == idx)[0][0])
        zeros = np.append(zeros, idx)

        # Add the zero with highest h value
        idx = zeros[np.argmax(h[zeros])]
        h[idx] = 1

def make_binary(h, n):
    """Project to binary with exactly n//2 ones."""
    h_bin = (h > 0.5).astype(float)
    # Fix count
    count = h_bin.sum()
    target = n // 2
    if count > target:
        # Set some 1s to 0 (those closest to 0.5)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])  # lowest original value first
        for idx in order[:int(count - target)]:
            h_bin[ones[idx]] = 0
    elif count < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])  # highest original value first
        for idx in order[:int(target - count)]:
            h_bin[zeros[idx]] = 1
    return h_bin
