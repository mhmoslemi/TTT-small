    pen = mu * (x.sum() - n / 2.0) ** 2
    return base + pen

def cvar_grad_fn(x, n, beta, mu):
    c = compute_c5_vec(x, n)
    K = len(c)
    k = max(1, int(beta * K))
    threshold = np.partition(c, K - k)[-k]
    top_idx = np.where(c >= threshold)[0]
    grad = np.zeros(n)
    for idx in top_idx:
        grad += shift_gradient(x, n, idx)
    grad /= len(top_idx)
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def run_cvar_lbfgs(h, n, deadline, betas=None, mu=1000.0):
    if betas is None:
        betas = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for beta in betas:
        if time.time() > deadline - 10:
            break
        k_eff = max(1, int(beta * (2 * n - 1)))
        maxiter = max(500, min(5000, 50000 // k_eff))
        try:
            res = minimize(cvar_obj_fn, best_h, args=(n, beta, mu),
                         jac=cvar_grad_fn, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_ = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad
