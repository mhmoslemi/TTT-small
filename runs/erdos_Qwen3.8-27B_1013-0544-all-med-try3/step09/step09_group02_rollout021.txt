```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    v = lse_value(c, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    nm1 = n - 1
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm] * dx
        if wm < 1e-15:
            continue
        k = mm - nm1
        if k >= 0:
            nv2 = n - k
            grad[:nv2] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv2])
        else:
            kk = -k
            nv2 = n - kk
            grad[kk:] += wm * (1.0 - x[:nv2])
            grad[:nv2] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lp_step(h, n, c_vec, active_idx, rho, t_cap):
    nm1 = n - 1
    dx = 2.0 / n
    ng = len(active_idx)
    nvar = n + 1
    
    f = np.zeros(nvar)
    f[-1] = 1.0
    
    if ng == 0:
        return np.zeros(n), float(c_vec.max())
    
    A = np.zeros((ng, nvar))
    b = np.zeros(ng)
    for j, idx in enumerate(active_idx):
        k = idx - nm1
        g = np.zeros(n)
        if k >= 0:
            nv2 = n - k
            g[:nv2] = (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv2]) * dx
        else:
            kk = -k
            nv2 = n - kk
            g[kk:] = (1.0 - h[:nv2]) * dx
            g[:nv2] += (-h[kk:]) * dx
        A[j, :n] = g
        A[j, n] = -1.0
        b[j] = -c_vec[idx]
    
    Aeq = np.array([np.concatenate([np.ones(n), [0.0]])])
    beq = np.array([0.0])
    
    lo = np.maximum(-rho, -h)
    hi = np.minimum(rho, 1.0 - h)
    bounds = list(zip(lo, hi)) + [(-np.inf, t_cap)]
    
    try:
        res = linprog(f, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

            h_new = project_feasible(res.x, n)

            if c_new < best_c:

                h = h_new
        except Exception:

def coordinate_descent(h, n, deadline, steps=None):
    best_c = compute_c5(h, n)
    nvar = len(h)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            
            # Compute subgradient of max C5
            # Use top-5 active shifts
            top5 = np.argsort(-c_vec)[:5]
            dx = 2.0 / n
            nm1 = n - 1
            grad = np.zeros(n)
            for idx in top5:
                k = int(idx) - nm1
                if k >= 0:
                    nv2 = n - k
                    grad[:nv2] += (1.0 - h[k:]) * dx
                    grad[k:] += (-h[:nv2]) * dx
                else:
                    kk = -k
                    nv2 = n - kk
                    grad[kk:] += (1.0 - h[:nv2]) * dx
                    grad[:nv2] += (-h[kk:]) * dx
            grad /= len(top5)
            
            # Find best coordinate to update
            # Decreasing h[i] helps if grad[i] > 0 (reduces the weighted sum)
            # We want to move in the -grad direction
            # But we need to maintain sum(h) = n/2
            # So: increase h[j] where grad[j] < 0, decrease h[i] where grad[i] > 0
            
            # Simple approach: for each i, try h[i] -= step, h[j] += step for best j
            # This is O(n^2) which is too slow. Instead:
            # Pick i with largest grad[i] (most positive), j with smallest grad[j] (most negative)
            
            i_best = int(np.argmax(grad))
            j_best = int(np.argmin(grad))
            
            # Try several pairs
            top_pos = np.argsort(-grad)[:5]
            top_neg = np.argsort(grad)[:5]
            
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                for j_val in top_neg:
                    j = int(j_val)
                    if i == j:
                        continue
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:

                            improved = True
                        else:
                            h[i] = old_i

                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        old_i, old_j = h[i], h[j]
                        h[i] += step
                        h[j] -= step
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-14:
