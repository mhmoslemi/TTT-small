def project_feasible(h, n, max_iter=5000):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        logsum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    val = m + logsum / alpha
    if mu > 0:

The Erdős problem asks: what is the minimum possible value of max_k |∫ h(x)h(x+k)dx - 1|... no wait, it's max_k ∫h(x)(1-h(x+k))dx.

For h ∈ {0,1} with ∫h = 1 (so h = 1_A where |A| = 1):
∫h(x)(1-h(x+k))dx = |A \ (A+k)| = |A| - |A ∩ (A+k)| = 1 - |A ∩ (A+k)|

So we're minimizing max_k (1 - |A ∩ (A+k)|) = 1 - min_k |A ∩ (A+k)|.

Equivalently, maximizing min_k |A ∩ (A+k)|.

This is related to the problem of finding a set A of measure 1 in [0,2] such that A intersects A+k in large measure for all k. This is a "difference set" or "additive combinatorics" type problem.

For the continuous case on the torus Z/2Z, this is related to the "Erdős difference sets" problem. The optimal value is related to the structure of A.

One known approach: take A to be a union of intervals. For example, A = [0, 1/2] ∪ [1, 3/2] (two intervals of length 1/2 each). Then |A| = 1.
|A ∩ (A+k)|: 
- For k=0: 1
- For small k: the two intervals overlap with their shifted versions
- For k=1: A ∩ (A+1) = [0,1/2] ∩ [2,2.5] ∪ [1,1.5] ∩ [1,1.5]...

One more thought: the problem has a natural "duality" in the sense that if we define g = 1-h, then ∫g = 1 also, and:
∫g(x)(1-g(x+k))dx = ∫g(x)·h(x+k)dx

This is the "cross-correlation" between g and h. And since g = 1-h, we have:
∫g·h(·+k) = ∫(1-h)·h(·+k) = ∫h(·+k) - ∫h·h(·+k) = 1 - ∫h·h(·+k)

Which is the same as ∫h(1-h(·-k)) (by change of variables). So the problem is symmetric.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n, max_iter=5000):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_grad(x, n, alpha, mu, w_thresh=1e-10):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for mm in range(n_full):
        if w[mm] < w_thresh:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, wi in zip(active, w):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def interp_upsample(h, n_from, n_to):
    x_fine = np.linspace(0, 2, n_to)
    x_coarse = np.linspace(0, 2, n_from)
    h_fine = np.interp(x_fine, x_coarse, h)
    return project_feasible(h_fine, n_to)

def lbfgs_optimize(h, n, deadline, alphas, mu, maxiters, w_thresh=1e-10):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = maxiters[i] if i < len(maxiters) else maxiters[-1]
        maxit = min(maxit, max(50, int(remaining * 2)))
        try:
            res = minimize(soft_obj, best_h, jac=soft_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_refine(h, n, deadline, top_k, alphas):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    n_rounds = 15
    for rnd in range(n_rounds):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.002), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx_val
                    grad[k:n] += (-h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                    grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] = min(h[i] + d, 1.0)
                        h[j] = max(h[j] - d, 0.0)
                        h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            pass
            h = best_h.copy()
    return best_h, best_c

def make_binary(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def smooth_start(n, rng):
    coeffs = rng.randn(n // 2 + 1) * 0.3
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for k, c in enumerate(coeffs):
        h += c * np.cos(np.pi * k * x)
    h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    h = project_feasible(h, n)
    return h
