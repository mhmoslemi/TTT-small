Δc_k = -(1-h[a+k]) * [a ≤ n-1-k] + h[a-k] * [a ≥ k] + (1-h[b+k]) * [b ≤ n-1-k] - h[b-k] * [b ≥ k]

All h values on the RHS are the ORIGINAL values (before swap).

Similarly for k < 0 (let l = -k > 0, correlate index m = n-1-l):
c_{-l} = sum_{i=l}^{n-1} h[i]*(1-h[i-l])

a contributes:
- "First" position (i=a): if a ≥ l, term was 1*(1-h[a-l]) = (1-h[a-l])
  After: 0. Change: -(1-h[a-l])
- "Second" position (i-l=a, i=a+l): if a+l ≤ n-1
  Term was h[a+l]*(1-1) = 0
  After: h[a+l]*(1-0) = h[a+l]. Change: +h[a+l]

b contributes:
- "First" position (i=b): if b ≥ l, term was 0. After: (1-h[b-l]). Change: +(1-h[b-l])
- "Second" position (i=b+l): if b+l ≤ n-1
  Term was h[b+l]*(1-0) = h[b+l]. After: h[b+l]*0 = 0. Change: -h[b+l]

So Δc_{-l} = -(1-h[a-l])*[a≥l] + h[a+l]*[a+l≤n-1] + (1-h[b-l])*[b≥l] - h[b+l]*[b+l≤n-1]

def compute_swap_delta(h, n, a, b):
    """Compute change in all c_k for swapping h[a] (1->0) and h[b] (0->1)."""
    dx = 2.0 / n
    delta = np.zeros(2*n-1)
    
    # Positive shifts k=1..n-1, stored at indices n-1+k
    ks = np.arange(1, n)
    
    # a as first element: affects k where k <= n-1-a
    mask_a1 = ks <= (n - 1 - a)
    if mask_a1.any():
        delta[n-1+ks[mask_a1]] -= (1 - h[a + ks[mask_a1]]) * dx
    
    # a as second element: affects k where k <= a
    mask_a2 = ks <= a
    if mask_a2.any():
        delta[n-1+ks[mask_a2]] += h[a - ks[mask_a2]] * dx
    
    # b as first element: affects k where k <= n-1-b
    mask_b1 = ks <= (n - 1 - b)
    if mask_b1.any():
        delta[n-1+ks[mask_b1]] += (1 - h[b + ks[mask_b1]]) * dx
    
    # b as second element: affects k where k <= b
    mask_b2 = ks <= b
    if mask_b2.any():
        delta[n-1+ks[mask_b2]] -= h[b - ks[mask_b2]] * dx
    
    # Negative shifts: k=-l, stored at indices n-1-l
    ls = np.arange(1, n)
    
    # a as first element (i=a >= l): affects l where l <= a
    mask_an1 = ls <= a
    if mask_an1.any():
        delta[n-1-ls[mask_an1]] -= (1 - h[a - ls[mask_an1]]) * dx
    
    # a as second element (i=a+l <= n-1): affects l where l <= n-1-a
    mask_an2 = ls <= (n - 1 - a)
    if mask_an2.any():
        delta[n-1-ls[mask_an2]] += h[a + ls[mask_an2]] * dx
    
    # b as first element (i=b >= l): affects l where l <= b
    mask_bn1 = ls <= b
    if mask_bn1.any():
        delta[n-1-ls[mask_bn1]] += (1 - h[b - ls[mask_bn1]]) * dx
    
    # b as second element (i=b+l <= n-1): affects l where l <= n-1-b
    mask_bn2 = ls <= (n - 1 - b)
    if mask_bn2.any():
        delta[n-1-ls[mask_bn2]] -= h[b + ls[mask_bn2]] * dx
    
    return delta

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.minimum(1.0, h + frac * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.maximum(0.0, h * (1.0 + frac))
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, alpha, mu=0.01, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 6,
                               'maxls': 40, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except:
        return h, compute_c5(h, n)

def to_binary(h, n):
    """Convert to binary pattern with exactly n/2 ones."""
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def swap_delta(h, n, a, b):
    """Compute change in all c_k for swapping h[a] (1->0) and h[b] (0->1)."""
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    ks = np.arange(1, n)
    
    # Positive shift k (index n-1+k):
    # c_k += -(1-h[a+k])*[k<=n-1-a] + h[a-k]*[k<=a] + (1-h[b+k])*[k<=n-1-b] - h[b-k]*[k<=b]
    m1 = ks <= (n - 1 - a)
    if m1.any():
        delta[n-1+ks[m1]] -= (1.0 - h[a + ks[m1]]) * dx
    m2 = ks <= a
    if m2.any():
        delta[n-1+ks[m2]] += h[a - ks[m2]] * dx
    m3 = ks <= (n - 1 - b)
    if m3.any():
        delta[n-1+ks[m3]] += (1.0 - h[b + ks[m
