```python
    # Cross-correlation r[k] = sum_i h[i+k] * b[i]
    # = ifft(fa * conj(fb)) shifted appropriately
    # ifft(FFT(a)*conj(FFT(b))) gives circular correlation: sum a[n] b[n-k]
    # We want sum h[n+k] b[n] = sum h[m] b[m-k] (sub m=n+k)
    # = circular correlation of h and b at lag k
    # = ifft(FFT(h) * conj(FFT(b))) at index k
    cross = np.fft.irfft(fa * np.conj(fb), n=fsize)
    # For linear correlation (full mode):
    # k ranges from -(n-1) to (n-1)
    # In the circular correlation array (length fsize >= 2n-1):
    #   k = 0: cross[0]
    #   k > 0: cross[k]
    #   k < 0: cross[fsize + k] = cross[fsize - |k|]
    result = np.zeros(2 * n - 1)
    for k in range(n):  # k = 0, 1, ..., n-1
        result[k + n - 1] = cross[k]
    for k in range(1, n):  # k = -1, -2, ..., -(n-1)
        result[n - 1 - k] = cross[fsize - k]
    return result
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_corr_fft(h):
    n = len(h)
    b = 1.0 - h
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(h, n=fsize)
    fb = np.fft.rfft(b, n=fsize)
    cross = np.fft.irfft(fa * np.conj(fb), n=fsize)
    result = np.zeros(size)
    for k in range(n):
        result[k + n - 1] = cross[k]
    for k in range(1, n):
        result[n - 1 - k] = cross[fsize - k]
    return result

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_val(x, n, alpha, use_fft):

    else:
        c = compute_corr(x)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, n, alpha, use_fft):
    dx = 2.0 / n
    if use_fft:
        c = compute_corr_fft(x)
    else:
        c = compute_corr(x)
    m = c.max()
    shifted = alpha * (c - m)

    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    
    grad = np.zeros(n)

            nv = n - kk

    return float(val), grad
```
