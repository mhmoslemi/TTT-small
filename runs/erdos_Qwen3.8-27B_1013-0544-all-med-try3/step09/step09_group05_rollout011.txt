import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    if active is not None:
        c_sel = c[active]
    else:
        c_sel = c
    m = float(c_sel.max())
    sh = alpha * (c_sel - m)
    mx = float(sh.max())
    if mx > 500:
        val = m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    else:
        val = m + float(np.log(np.sum(np.exp(sh)))) / alpha
    return val

def lse_gradient(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    if active is not None:
        c_sel = c[active]
    else:
        c_sel = c
    m = float(c_sel.max())
    sh = alpha * (c_sel - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    if active is not None:
        idx_list = active
    else:
        idx_list = range(2 * n - 1)
    for wi, mm in zip(w, idx_list):
        if wi < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs_stage(h, n, deadline, alpha, active=None, maxiter=10000):
    h0 = project_feasible(h, n)
    c0 = compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h0, jac=lse_gradient,
                      args=(n, alpha, active), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxiter * 20,
                               'maxls': 100})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < c0:
            return h_new, c_new, True
        return h0, c0, False
    except Exception:
        return h0, c0, False

def get_active_shifts(h, n, threshold_frac=0.005, max_count=50):
    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()
    thresh = m - threshold_frac
    active = np.where(c_vec >= thresh)[0]
    if len(active) > max_count:
        active = active[np.argsort(-c_vec[active])][:max_count]
    return active

    return h, best_c
