def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[0:n-m] -= h[m:n]
        grad[m:n] += (1.0 - h[0:n-m])
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n-m]
    return grad


def interpolate_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)

def slsqp_polish(h, n, alpha, top_m, maxiter):
    bounds = [(0.0, 1.0)] * n
    sum_target = n / 2.0
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        active = cc[top_m]
        m_val = active.max()
        shifted = alpha * (active - m_val)
        mx = shifted.max()
        return float(m_val + np.log(np.sum(np.exp(shifted - mx))) / alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        active = cc[top_m]
        m_val = active.max()
        shifted = alpha * (active - m_val)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w = w / w.sum()
        g = np.zeros(n)
        for i in range(len(top_m)):
            if w[i] < 1e-10:
                continue
            lag = top_m[i] - (n - 1)
            g += w[i] * lag_gradient(x, n, lag)
        return g * (2.0 / n)
    
    def constr_eq(x):
        return float(np.sum(x) - sum_target)
    
    def constr_eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_eq_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < float('inf'):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def make_binary_start(n, rng, period=None):

    return h


def make_fourier_start(n, rng, n_modes=3):
