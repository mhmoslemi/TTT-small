import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def power_mean_val(c, p):
    return float((np.sum(np.power(c, p)) ** (1.0 / p)))

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            grad[0:nv] += wm * (-x[kk:n])

                            best_h[a], best_h[b], best_h[c1], best_h[d] = old
                    if any_22:
                        break
                if any_22:
                    break
            if not improved:
                break
    return best_h, best_c

def coord_descent_fine(h, n, deadline, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    if j - i > n // 2:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def grad_descent_steps(h, n, deadline, steps_per_size=5000, step_sizes=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in step_sizes:
        count = 0
        while time.time() < deadline and count < steps_per_size:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-10)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i, best_j = -1, -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            count += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    n_orig = len(h0)
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    candidates = []
    candidates.append(('init', h0.copy()))
    candidates.append(('beatty_g', make_beatty_start(n_orig, (golden - 1) / 2.0)))
    candidates.append(('beatty_s2', make_beatty_start(n_orig, 1.0 / np.sqrt(2.0))))
    candidates.append(('beatty_s3', make_beatty_start(n_orig, 1.0 / np.sqrt(3.0))))
    candidates.append(('rand1', make_random_binary(n_orig, seed)))
    candidates.append(('rand2', make_random_binary(n_orig, seed + 100)))
    candidates.append(('rand3', make_random_binary(n_orig, seed + 200)))

    # Quick L-BFGS on all candidates
    for name, hc in candidates:
        if time.time() > deadline - 300:
            break
        hc = project_feasible(hc, n_orig)
        c_init = compute_c5(hc, n_orig)
        # Very quick warmup
        bounds = [(0.0, 1.0)] * n_orig
        try:
            res = minimize(lm_obj, hc, jac=lm_grad,
                          args=(n_orig, 50, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 3000,
                                   'maxls': 30, 'maxcor': 30})
            hc2 = project_feasible(res.x, n_orig)
            c2 = compute_c5(hc2, n_orig)
            if c2 < best_c:
                best_c = c2
                best_h = hc2.copy()
        except Exception:
            pass
        if c_init < best_c:
            best_c = c_init
            best_h = hc.copy()

    # Phase 1: Main L-BFGS optimization on best candidate
    p1_end = min(t0 + 180, deadline - 500)
    h1, c1 = lbfags_phase(best_h, best_n, p1_end,
        alphas=[20, 50, 200, 1000, 5000, 20000, 100000],
        mu=0.005,
        maxits=[800, 1000, 1500, 2000, 2500, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 150, deadline - 300)
        h2, c2 = active_set_phase(best_h, best_n, p2_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000], rounds=20)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Binary local search (key differentiator)
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 150, deadline - 250)
        h3, c3 = binary_local_search(best_h, best_n, p3_end, max_rounds=40)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: L-BFGS refinement after binary search
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = lbfags_phase(best_h, best_n, p4_end,
            alphas=[50000, 200000, 1000000, 5000000],
            mu=0.001,
            maxits=[2000, 3000, 4000, 5000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Another round of binary local search
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 100, deadline - 150)
        h5, c5 = binary_local_search(best_h, best_n, p5_end, max_rounds=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Gradient-descent fine steps
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 80, deadline - 100)
        h6, c6 = grad_descent_steps(best_h, best_n, p6_end,
            steps_per_size=8000,
            step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Active set + grad descent again
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 60, deadline - 60)
        h7, c7 = active_set_phase(best_h, best_n, p7_end, top_k=4,
            alphas=[5000000, 20000000, 50000000], rounds=10)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        if time.time() < deadline - 30:
            p7b_end = min(time.time() + 30, deadline - 20)
            h7b, c7b = grad_descent_steps(best_h, best_n, p7b_end,
                steps_per_size=5000,
                step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Phase 8: Final micro adjustments
    if time.time() < deadline - 5:
        h8, c8 = grad_descent_steps(best_h, best_n, deadline - 2,
            steps_per_size=3000,
            step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
