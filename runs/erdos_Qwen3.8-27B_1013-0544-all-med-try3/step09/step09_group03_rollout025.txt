import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    active = np.where(w > 1e-12)[0]
    for idx in active:

        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def compute_effect(h, j, n):

    n2m1 = 2 * n - 1
    center = n - 1
    effect = np.zeros(n2m1)
    # k >= 0 part (si >= center)
    # First term: (1-h[j+k]) for k=0..min(j+k<n)

    max_k2 = min(j, n - 1)

    # First term: (1-h[j-kk]) for kk=1..min(j, n-1)
    max_kk1 = min(j, n - 1)

    max_kk2 = n - 1 - j
    for kk in range(1, max_kk2 + 1):
        effect[center - kk] -= h[j + kk]
    return effect

    """Generate diverse structured initial candidates."""
    rng = np.random.RandomState(seed)
    starts = []
    x = np.linspace(0, 2, n)
    
    # Sine-based
    for f in [1, 2, 3, 4, 5]:
        for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
            h = 0.5 + 0.5 * np.sin(f * np.pi * x + phase)
            starts.append(project_feasible(np.clip(h, 0, 1), n))
    
    # Square wave
    for n_periods in [1, 2, 3, 4]:
        for duty in [0.3, 0.5, 0.7]:
            h = np.zeros(n)
            period = n // n_periods
            for p in range(n_periods):
                start = p * period
                end = start + int(period * duty)
                if end < n:
                    h[start:end] = 1.0
            starts.append(project_feasible(h, n))
    
    # Random binary
    for _ in range(5):
        idx = rng.choice(n, size=n//2, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        starts.append(h)
    
    # Smooth random
    for _ in range(3):
        h = 0.3 + 0.4 * rng.rand(n)
        starts.append(project_feasible(h, n))
    
    return starts

def targeted_refine(h, n, deadline):
    """Coordinate descent trying small perturbations on worst-affected points."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        found = False
        # Try adjusting points near the worst shift
        candidates = list(range(max(0, k_worst), min(n, n - abs(k_worst))))
        rng = np.random.RandomState(rnd + 100)
        if len(candidates) > 50:
            candidates = list(rng.choice(candidates, 50, replace=False))
        
        for i in candidates:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.5, -0.5]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = min(max(nv, 0), 1)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = best_h[i]
            if found:
                h = best_h.copy()
                break
        if not found and rnd > 50:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    h_init = project_feasible(h_init, n_base)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick optimization at base resolution (80s)
    t1 = min(t0 + 80, deadline - 800)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(100, 20.0), (500, 100.0), (2000, 500.0),
                     (5000, 2000.0), (10000, 5000.0), (30000, 20000.0)])
    
    # Phase 2: Try structured starts at base (40s)
    t2 = min(t0 + 120, deadline - 750)
    if t2 > time.time() + 15:
        starts = generate_structured_starts(n_base, seed)
        for s in starts:
            if time.time() > t2 - 5:
                break
            c_s = compute_c5(s, n_base)
            if c_s < best_c + 0.05:
                h_try, c_try = run_lbfgs(s, n_base, min(time.time() + 20, t2 - 5),
                    configs=[(500, 100.0), (2000, 500.0), (5000, 2000.0)])
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
    
    # Phase 3: Binary CD at base resolution (40s)
    t3 = min(t0 + 160, deadline - 700)
    if t3 > time.time() + 10:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n_base)
        c_vec_bin = compute_c5_vec(h_bin, n_base)
        h_bin, c_vec_bin, c_bin = binary_swap_cd(h_bin, n_base, c_vec_bin,
                                                  max_passes=50, deadline=t3)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        # Also try: optimize the binary solution with continuous method
        if time.time() < t3 - 10:
            h_bo, c_bo = run_lbfgs(h_bin, n_base, t3,
                configs=[(1000, 200.0), (3000, 500.0), (8000, 2000.0)])
            if c_bo < best_c:
                best_c = c_bo
                best_h = h_bo.copy()
    
    # Phase 4: SLSQP polish at base (30s)
    t4 = min(t0 + 190, deadline - 650)
    if t4 > time.time() + 10:
        h4, c4 = slqp_polish(best_h, n_base, t4, alpha=2000000, maxiter=5000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Upsample to n=1000, optimize (200s)
    n1 = 1000
    t5_end = min(t0 + 390, deadline - 400)
    if time.time() < t5_end - 30:
        h1000 = interpolate_to(best_h, n_base, n1)
        t5_mid = min(time.time() + 120, t5_end - 60)
        if t5_mid > time.time() + 10:
            h1000, c1000 = run_lbfgs(h1000, n1, t5_mid,
                configs=[(200, 50.0), (500, 200.0), (2000, 500.0),
                         (5000, 2000.0), (10000, 5000.0), (30000, 20000.0),
                         (50000, 50000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        # Binary CD at n=1000
        if time.time() < t5_end - 30:
            h_b1000 = np.where(best_h > 0.5, 1.0, 0.0)
            h_b1000 = project_feasible(h_b1000, n1)
            c_v1000 = compute_c5_vec(h_b1000, n1)
            h_b1000, c_v1000, c_b1000 = binary_swap_cd(h_b1000, n1, c_v1000,
                                                        max_passes=30, deadline=t5_end - 15)
            if c_b1000 < best_c:
                best_c = c_b1000
                best_h = h_b1000.copy()
            # Continuous optimization from binary
            if time.time() < t5_end - 15:
                h1000b, c1000b = run_lbfgs(h_b1000, n1, t5_end - 5,
                    configs=[(2000, 500.0), (5000, 2000.0), (15000, 10000.0)])
                if c1000b < best_c:
                    best_c = c1000b
                    best_h = h1000b.copy()
        # SLSQP at n=1000
        if time.time() < t5_end - 10:
            h5, c5 = slqp_polish(best_h, n1, t5_end, alpha=5000000, maxiter=5000)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: Upsample to n=2000, optimize (350s)
    n2 = 2000
    t6_end = min(t0 + 740, deadline - 150)
    if time.time() < t6_end - 60:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        # L-BFGS at n=2000
        t6_mid = min(time.time() + 200, t6_end - 100)
        if t6_mid > time.time() + 15:
            h2000, c2000 = run_lbfgs(h2000, n2, t6_mid,
                configs=[(500, 100.0), (2000, 500.0), (5000, 2000.0),
                         (10000, 5000.0), (30000, 10000.0), (50000, 30000.0),
                         (100000, 50000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        # SLSQP at n=2000
        if time.time() < t6_end - 30:
            h6, c6 = slqp_polish(best_h, n2, t6_end - 15, alpha=10000000, maxiter=5000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
        # More L-BFGS with higher alpha
        if time.time() < t6_end - 15:
            h6b, c6b = run_lbfgs(best_h, n2, t6_end,
                configs=[(50000, 30000.0), (100000, 50000.0), (200000, 100000.0)])
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Perturbation search at final resolution (80s)
    t7_end = min(t0 + 820, deadline - 50)
    if time.time() < t7_end - 20:
        n_final = len(best_h)
        rng = np.random.RandomState(seed + 777)
        for trial in range(8):
            if time.time() > t7_end - 20:
                break
            for mag in [0.01, 0.03, 0.05]:
                if time.time() > t7_end - 20:
                    break
                h_p = best_h + mag * (rng.rand(n_final) - 0.5)
                h_p = project_feasible(h_p, n_final)
                try:
                    res = minimize(smooth_obj, h_p, jac=smooth_grad,
                                  args=(n_final, 10000, 10000.0),
                                  method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * n_final,
                                  options={'maxiter': 5000, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
                    h_new = project_feasible(res.x, n_final)
                    c_new = compute_c5(h_new, n_final)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                except Exception:
                    pass
    
    # Phase 8: Final targeted refinement (60s)
    t8_end = min(t0 + 880, deadline - 10)
    if time.time() < t8_end - 10:
        best_h, best_c = targeted_refine(best_h, len(best_h), t8_end - 5)
    
    # Phase 9: Very final SLSQP + L-BFGS push
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        h9, c9 = slqp_polish(best_h, n_fin, deadline - 2, alpha=50000000, maxiter=2000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
