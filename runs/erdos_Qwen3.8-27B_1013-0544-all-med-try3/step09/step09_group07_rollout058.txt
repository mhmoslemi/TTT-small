def interpolate_to_resolution(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    h_dst = np.interp(x_dst, x_src, h_src, left=h_src[0], right=h_src[-1])
    return project_feasible(h_dst, n_dst)

def multi_lag_balance(h, n, deadline, rng, k_bal=3):
    """Try to equalize the top-k lags by moving from high to low lag regions."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        top_idx, c_all = active_lags(h, n, k_bal)
        lags = [int(ki - (n-1)) for ki in top_idx]
        vals = c_all[top_idx]
        spread = vals.max() - vals.min()
        if spread < 1e-8:
            break
        # Try to reduce the spread
        improved = False
        for _ in range(2000):
            if time.time() > deadline:
                break
            # Pick two active lags
            la = lags[rng.randint(k_bal)]
            lb = lags[rng.randint(k_bal)]
            if la == lb:
                continue
            ma, mb = abs(la), abs(lb)
            if ma >= n or mb >= n:
                continue
            # Try a move that reduces lag ma and increases lag mb (to equalize)
            i = rng.randint(0, max(1, n - ma - 1))
            j = i + ma
            d = 0.002 * rng.uniform(0.5, 1.5)
            if h[i] > h[j]:
                h[i] -= d
                h[j] += d
            else:
                h[i] += d
                h[j] -= d
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            # Accept if max didn't increase much and spread decreased
            new_top, new_c = active_lags(h, n, k_bal)
            new_spread = new_c[new_top].max() - new_c[new_top].min()
            if c_new < best_c * (1 + 1e-8) and new_spread < spread - 1e-10:
                improved = True
            elif c_new < best_c * (1 - 1e-6):
                improved = True
            else:
                h = best_h.copy()
            
            if c_new < best_c:
                best_c = float(c_new)
                best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    h_cur = project_feasible(h_init, n_init)
    best_c = compute_c5(h_cur, n_init)
    best_h = h_cur.copy()
    
    # Phase 1: L-BFGS-B at initial resolution with increasing alpha
    for alpha, maxit in [(5000, 500), (50000, 800), (200000, 1000), (1000000, 1500), (5000000, 2000)]:
        if time.time() > deadline - 500:
            break
        rem = deadline - 500 - time.time()
        if rem < 10:
            break
        h_try = lbfgsb_active(best_h, n_init, alpha, time.time() + min(rem, 45),
                              top_k=10, maxiter=maxit)
        c_try = compute_c5(h_try, n_init)
        if c_try < best_c:
            best_c, best_h = c_try, h_try
    
    # Phase 2: Try n=800
    n800 = 800
    if time.time() < deadline - 350:
        h800 = interpolate_to_resolution(best_h, n_init, n800)
        for alpha, maxit in [(100000, 1000), (500000, 1500), (2000000, 2000)]:
            if time.time() > deadline - 250:
                break
            rem = deadline - 250 - time.time()
            if rem < 10:
                break
            h_try = lbfgsb_active(h800, n800, alpha, time.time() + min(rem, 40),
                                  top_k=10, maxiter=maxit)
            c_try = compute_c5(h_try, n800)
            if c_try < best_c:
                best_c, best_h = c_try, h_try
                h800 = h_try.copy()
            else:
                h800 = h_try.copy()  # still use as starting point
    
    # Phase 3: Momentum optimization at best resolution
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        h_mom, c_mom = momentum_optimize(best_h, n_cur, min(deadline - 150, time.time() + 120),
                                         top_k=10, alpha=200000, lr_init=5.0)
        if c_mom < best_c:
            best_c, best_h = c_mom, h_mom
    
    # Phase 4: Targeted local search
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        ls_dead = min(deadline - 40, time.time() + 120)
        h_ls, c_ls = targeted_local_search(best_h, n_cur, ls_dead, rng, top_k_lags=5)
        if c_ls < best_c:
            best_c, best_h = c_ls, h_ls
    
    # Phase 5: Try n=1000
    n1000 = 1000
    if time.time() < deadline - 150:
        h1000 = interpolate_to_resolution(best_h, len(best_h), n1000)
        for alpha, maxit in [(200000, 1000), (1000000, 1500), (5000000, 2000)]:
            if time.time() > deadline - 80:
                break
            rem = deadline - 80 - time.time()
            if rem < 10:
                break
            h_try = lbfgsb_active(h1000, n1000, alpha, time.time() + min(rem, 40),
                                  top_k=8, maxiter=maxit)
            c_try = compute_c5(h_try, n1000)
            if c_try < best_c:
                best_c, best_h = c_try, h_try
                h1000 = h_try.copy()
            else:
                h1000 = h_try.copy()
        # Local search at n=1000
        if time.time() < deadline - 30:
            h_ls, c_ls = targeted_local_search(best_h, len(best_h), deadline - 10, rng, top_k_lags=5)
            if c_ls < best_c:
                best_c, best_h = c_ls, h_ls
    
    # Phase 6: Second round of momentum + local search at best resolution
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        # Small perturbation to escape local optimum
        h_pert = best_h + rng.normal(0, 0.005, n_cur)
        h_pert = project_feasible(h_pert, n_cur)
        h_mom, c_mom = momentum_optimize(h_pert, n_cur, min(deadline - 30, time.time() + 50),
                                         top_k=8, alpha=500000, lr_init=3.0)
        if c_mom < best_c:
            best_c, best_h = c_mom, h_mom
    
    # Phase 7: Final L-BFGS-B polish
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h_final = lbfgsb_active(best_h, n_cur, 10000000, deadline - 3, top_k=5, maxiter=3000)
        c_final = compute_c5(h_final, n_cur)
        if c_final < best_c:
            best_c, best_h = c_final, h_final
    
    # Phase 8: Ultra-fine local search
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        h_fin, c_fin = targeted_local_search(best_h, n_cur, deadline - 1, rng, top_k_lags=3)
        if c_fin < best_c:
            best_c, best_h = c_fin, h_fin
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
