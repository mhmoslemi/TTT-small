def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                # (1-x[j-k]) for j in [k, n-1] → add to grad[k:].
                #   (1-x[0:n-k]) added to grad[k:]
                # (-x[j+k]) for j in [0, n-k-1] → add to grad[:n-k]
                #   -x[k:n] added to grad[:n-k]
                grad[k:k+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[k:n])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                # (1-x[j+kk]) for j in [0, n-kk-1] → add to grad[:n-kk]
                #   (1-x[kk:n]) added to grad[:n-kk]
                # (-x[j-kk]) for j in [kk, n-1] → add to grad[kk:]
                #   -x[0:n-kk] added to grad[kk:]
                grad[0:nv] += wm * (1.0 - x[kk:n])
                grad[kk:n] += wm * (-x[0:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

Actually, for the binary case (h ∈ {0,1} with exactly n/2 ones):
C5 = max_k (number of i where h[i]=1 and h[i+k]=0) * dx = max_k count / (n/2) * ... 

Wait: c_k = sum_i h[i]*(1-h[i+k]) * dx. For binary h, h[i]*(1-h[i+k]) = 1 iff h[i]=1 and h[i+k]=0.
So c_k = (count of such pairs) * dx = (count) * 2/n.

We want to minimize max_k count. The minimum possible max count is related to the "discrepancy" of the binary sequence.

For a random binary sequence with n/2 ones, the expected count at shift k is approximately (n-k)/4 (for k not too large). The max over k would be around (n/2)/4 = n/8 (roughly), giving C5 ≈ (n/8) * 2/n = 1/4 = 0.25. But that's way too optimistic because the max concentrates.

Actually for a truly random sequence, C5 would be close to the mean plus a few standard deviations. The mean of c_k for k≈0 is about (n/2) * 1/2 * 2/n = 1/2 (approximately). The std is much smaller. So C5 for random is about 0.5.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sobj(x, n, alpha, mu):
    """Smooth objective (log-sum-exp approximation of max)."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sgrad(x, n, alpha, mu):
    """Gradient of smooth objective."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx

            nv = n - k
            if nv > 0:
                grad[k:k+nv] += wm * (1.0 - x[0:nv])

                grad[0:nv] += wm * (1.0 - x[kk:n])
                grad[kk:n] += wm * (-x[0:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    bounds = [(0.0, 1.0)] * n

        alphas = [50, 200, 500, 2000, 10000, 50000]

            break

                          bounds=bounds,

                                   'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_optimize(h, n, deadline, alpha=100000, maxiter=2000):

    def obj_fn(x, na, al):
        return sobj(x, na, al, 0.0)

        return x.sum() - na / 2.0
    def con_jac(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
