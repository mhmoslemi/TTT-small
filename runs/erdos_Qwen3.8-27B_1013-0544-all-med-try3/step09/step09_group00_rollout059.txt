def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()

            tot = h.sum()
            if tot < 1e-14:

def get_c5_load_vector(c, alpha, top_k=None):
    """Get softmax weights over correlation values."""
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -100, 100)
    if top_k is not None and len(c) > top_k:
        # Only keep top_k
        idx = np.argsort(-sh)[:top_k]
        sh2 = np.zeros_like(sh)
        sh2[idx] = sh[idx]
        sh = sh2
    e = np.exp(sh - sh.max())
    w = e / e.sum()
    return w

                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -50, 50)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / s

def weighted_gradient(c, h, n, weights):
    dx = 2.0 / n
    nn = len(c)
    grad = np.zeros(n)
    for mm in range(nn):
        if weights[mm] < 1e-16:
            continue
        k = mm - (n - 1)
        wm = weights[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def topk_gradient(c, h, n, active_indices):
    dx = 2.0 / n
    grad = np.zeros(n)
    w_each = 1.0 / len(active_indices)
    for mm in active_indices:
        k = mm - (n - 1)
        wm = w_each * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_smooth(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def grad_smooth(x, n, alpha):
    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha)
    return weighted_gradient(c, x, n, w)

def obj_topk(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    sh = np.clip(sh, -50, 50)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def grad_topk(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)
    w_local = e / e.sum()
    w_full = np.zeros(len(c))
    w_full[active] = w_local
    return weighted_gradient(c, x, n, w_full)

def lsq_fun(x, n):
    return x.sum() - n / 2.0

def lsq_jac(x, n):
    return np.ones(n)

def do_lbfgs(h, n, deadline, alpha, maxiter, use_sum_constraint=False):
    bounds = [(0.0, 1.0)] * n
    try:
        if use_sum_constraint:
            cons = [{'type': 'eq', 'fun': lsq_fun, 'jac': lsq_jac, 'args': (n,)}]
            res = minimize(obj_smooth, h, jac=grad_smooth, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': min(maxiter, 300), 'ftol': 1e-16})
        else:
            res = minimize(obj_smooth, h, jac=grad_smooth, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxiter * 8,
                                   'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def do_lbfgs_topk(h, n, deadline, active, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_topk, h, jac=grad_topk,
                      args=(n, active, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': maxiter * 6,
                               'maxls': 30, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def subgrad_crawl(h, n, deadline, alpha_start=2000, alpha_end=500000,
                  lr_start=0.015, noise_start=0.025, momentum=0.4,
                  max_iter=200000, seed=7):
    rng = np.random.RandomState(seed)
    x = project_feasible(h.copy(), n)
    best_x = x.copy()
    best_c = compute_c5(x, n)
    v = np.zeros(n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start / (1.0 + 2.0 * frac)
        noise = noise_start * (1.0 - frac) ** 2
        
        c = compute_c5_vec(x, n)
        w = smooth_max_weights(c, alpha)
        grad = weighted_gradient(c, x, n, w)
        
        if noise > 1e-5:
            grad = grad + noise * rng.randn(n)
        
        v = momentum * v - lr * grad
        x_new = x + v
        x_new = project_feasible(x_new, n)
        x = x_new
        
        c_new = compute_c5(x, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_x = x.copy()
    
    return best_x, best_c

def topk_focus_optimize(h, n, deadline, top_k_list, alpha_list, maxiter_list, seed=99):
    rng = np.random.RandomState(seed)
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for tk, alpha, maxit in zip(top_k_list, alpha_list, maxiter_list):
        for _ in range(3):
            if time.time() > deadline - 20:
                break
            c = compute_c5_vec(best_h, n)
            mv = c.max()
            thr = mv - max(1e-10, 0.0005 * mv)
            active = np.where(c >= thr)[0]
            if len(active) > tk:
                active = np.array(np.argsort(-c[active])[:tk])
            if len(active) == 0:
                break
            
            h_new, c_new = do_lbfgs_topk(best_h, n, deadline - 10, active, alpha, maxit)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            else:
                # Try perturbation
                pert = best_h + 0.01 * rng.randn(n)
                pert = project_feasible(pert, n)
                h_new2, c_new2 = do_lbfgs_topk(pert, n, deadline - 10, active, alpha, maxit // 2)
                if c_new2 < best_c - 1e-15:
                    best_c = c_new2
                    best_h = h_new2
    
    return best_h, best_c

def pair_descent(h, n, deadline, rng, n_trials_per_step=500):
    x = project_feasible(h.copy(), n)
    best_c = compute_c5(x, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            order = np.argsort(x)
            lows = order[:n // 2]
            highs = order[n // 2:]
            
            for _ in range(n_trials_per_step):
                if time.time() > deadline:
                    break
                i = lows[rng.randint(len(lows))]
                j = highs[rng.randint(len(highs))]
                
                amt = min(step, x[j] - x[i], 1.0 - x[j], x[i])
                if amt < 1e-10:
                    continue
                x[j] -= amt
                x[i] += amt
                x = project_feasible(x, n)
                c_new = compute_c5(x, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    x[j] += amt
                    x[i] -= amt
    
    x = project_feasible(x, n)
    return x, best_c

def targeted_direction(h, n, deadline, rng, max_iters=50000):
    """Directly target the worst correlation with a steepest-descent pair move."""
    x = project_feasible(h.copy(), n)
    best_c = compute_c5(x, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(x, n)
        max_idx = c.argmax()
        max_val = c[max_idx]
        k = max_idx - (n - 1)
        
        # Gradient of THIS correlation
        dx_val = 2.0 / n
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] = (1.0 - x[k:]) * dx_val
            g[k:] = (-x[:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            g[kk:] = (1.0 - x[:nv]) * dx_val
            g[:nv] = (-x[kk:]) * dx_val
        
        # Find best pair: decrease x where g>0, increase x where g<0
        pos_idx = np.where(g > 1e-8)[0]
        neg_idx = np.where(g < -1e-8)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            # Switch to second worst
            c2 = c.copy()
            c2[max_idx] = -np.inf
            max_idx = c2.argmax()
            if c2[max_idx] < 0:
                break
            k = max_idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[:nv] = (1.0 - x[k:]) * dx_val
                g[k:] = (-x[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                g[kk:] = (1.0 - x[:nv]) * dx_val
                g[:nv] = (-x[kk:]) * dx_val
            pos_idx = np.where(g > 1e-8)[0]
            neg_idx = np.where(g < -1e-8)[0]
            if len(pos_idx) == 0 or len(neg_idx) == 0:
                break
        
        # Pick best pair by gradient difference
        best_gain = 0
        best_i = -1
        best_j = -1
        # Sample for efficiency
        ps = pos_idx[rng.randint(0, len(pos_idx), size=min(30, len(pos_idx)))]
        ns = neg_idx[rng.randint(0, len(neg_idx), size=min(30, len(neg_idx)))]
        for ii in ps:
            for jj in ns:
                gain = g[ii] - g[jj]
                if gain > best_gain and x[ii] > 1e-6 and x[jj] < 1 - 1e-6:
                    best_gain = gain
                    best_i = ii
                    best_j = jj
        
        if best_i < 0:
            # Add noise to escape
            noise_pos = pos_idx[rng.randint(len(pos_idx))]
            noise_neg = neg_idx[rng.randint(len(neg_idx))]
            best_i = noise_pos
            best_j = noise_neg
            best_gain = 1e-6
        
        # Adaptive step size
        step = min(0.01, 0.1 * best_gain / max(1e-8, np.abs(g).max()))
        amt = min(step, x[best_i] - 1e-6, 1.0 - x[best_j] - 1e-6)
        if amt < 1e-8:
            break
        
        x[best_i] -= amt
        x[best_j] += amt
        x = project_feasible(x, n)
        c_new = compute_c5(x, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            # Reject and add small random kick
            x[best_i] += amt
            x[best_j] -= amt
            kick = 0.005 * rng.randn(n)
            x = x + kick
            x = project_feasible(x, n)
    
    x = project_feasible(x, n)
    return x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Phase 1: L-BFGS warmup with increasing alpha (70s)
    p1_end = t0 + 70
    for a in [50, 200, 1000, 5000, 20000, 100000]:
        if time.time() > p1_end:
            break
        h_new, c_new = do_lbfgs(best_h, n, p1_end, a, maxiter=250)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Top-k focused optimization (250s)
    p2_end = min(time.time() + 250, deadline - 500)
    h2, c2 = topk_focus_optimize(best_h, n, p2_end,
                                  top_k_list=[3, 5, 8, 12, 20],
                                  alpha_list=[50000, 200000, 500000, 2000000, 5000000],
                                  maxiter_list=[400, 400, 350, 300, 250],
                                  seed=seed)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    
    # Phase 3: Subgradient crawl with annealing (250s)
    p3_end = min(time.time() + 250, deadline - 350)
    h3, c3 = subgrad_crawl(best_h, n, p3_end,
                            alpha_start=5000, alpha_end=500000,
                            lr_start=0.012, noise_start=0.02,
                            momentum=0.35, max_iter=150000, seed=seed + 1)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Targeted direction search (200s)
    p4_end = min(time.time() + 200, deadline - 250)
    h4, c4 = targeted_direction(best_h, n, p4_end, rng, max_iters=80000)
    if c4 < best_c:
        best_c = c4
        best_h = h4
    
    # Phase 5: L-BFGS with high alpha + topk (150s)
    p5_end = min(time.time() + 150, deadline - 150)
    for a in [500000, 2000000, 10000000]:
        if time.time() > p5_end:
            break
        h_new, c_new = do_lbfgs(best_h, n, p5_end, a, maxiter=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 120:
        h2b, c2b = topk_focus_optimize(best_h, n, deadline - 120,
                                        top_k_list=[2, 4, 6],
                                        alpha_list=[5000000, 20000000, 80000000],
                                        maxiter_list=[200, 200, 150],
                                        seed=seed + 2)
        if c2b < best_c:
            best_c = c2b
            best_h = h2b
    
    # Phase 6: Pair descent (120s)
    p6_end = min(time.time() + 120, deadline - 60)
    rng2 = np.random.RandomState(seed + 100)
    h6, c6 = pair_descent(best_h, n, p6_end, rng2, n_trials_per_step=400)
    if c6 < best_c:
        best_c = c6
        best_h = h6
    
    # Phase 7: Final targeted + subgrad polish (50s)
    p7_end = min(time.time() + 50, deadline - 5)
    h7, c7 = targeted_direction(best_h, n, p7_end, rng2, max_iters=30000)
    if c7 < best_c:
        best_c = c7
        best_h = h7
    if time.time() < deadline - 10:
        h7b, c7b = subgrad_crawl(best_h, n, deadline - 3,
                                   alpha_start=1000000, alpha_end=5000000,
                                   lr_start=0.001, noise_start=0.001,
                                   momentum=0.2, max_iter=50000, seed=seed + 3)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
