But the gradient of the log-sum-exp is well-defined and reliable. Let me stick with it.

Actually, let me think about another approach: **direct minimax via bisection**.

The problem is: min_h max_k c_k(h) subject to 0≤h≤1, sum(h)*dx=1.

This can be reformulated as:
Find the smallest t such that there exists h with c_k(h) ≤ t for all k.

This is a feasibility problem for a given t. We can solve it with:
- LP relaxation (if we relax the bilinear terms)
- Or trust-region SQP

The bilinear terms make it non-convex, so LP won't work directly. But we can use the fact that c_k(h) = Σ_i h_i(1-h_{i+k}) is bilinear, and use alternating minimization or semi-definite relaxation.

For alternating: fix "1-h" part, solve for h (linear in h for each k), then fix h and solve for "1-h" part. This is similar to PMD (Projecting Subgradient) methods.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            if frac > 1.0:
                frac = 1.0
            h += frac * room
            h = np.clip(h, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            scale = 1.0 + diff / tot
            if scale < 0:
                scale = 0.0
            h *= scale
            h = np.clip(h, 0, 1)
    return h

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_lse(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def smooth_lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-14:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_acg(x, n, alpha, beta, mu):
    val = smooth_lse(x, n, alpha)
    val += beta * np.sum(x * (1.0 - x))
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_acg(x, n, alpha, beta, mu):
    g = smooth_lse_grad(x, n, alpha)
    g += beta * (1.0 - 2.0 * x)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def nesterov_optimize(h, n, deadline, alpha=500, beta=10.0, mu=100.0, step_init=0.01):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m = min(50, max(10, n // 10))
    prev_ys = [None] * m
    prev_grads = [None] * m
    yk = (h.sum() - n / 2.0) ** 2
    yr = yk
    delta = 0.0
    step = step_init
    for it in range(5000):
        if time.time() > deadline:
            break
        grad = grad_acg(h, n, alpha, beta, mu)
        step_norm = np.linalg.norm(grad)
        if step_norm < 1e-12:
            break
        step = step * (yr / max(yk, 1e-30)) ** 0.5
        step = np.clip(step, 1e-6, 0.5)
        yk_new = (h.sum() - n / 2.0) ** 2
        delta = 0.25 * (1.0 + (delta + 1.0) ** 0.5 - delta)
        h_new = h + step * (grad * delta + (1.0 - delta) * (h - best_h))
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            yr = yk_new
        h = h_new
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 5.0, 50.0), (500, 20.0, 500.0), (2000, 50.0, 2000.0),
                   (5000, 100.0, 5000.0), (15000, 200.0, 10000.0)]
    for alpha, beta, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(obj_acg, h, jac=grad_acg,
                          args=(n, alpha, beta, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.04):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        if len(active) > 15:
            order = np.argsort(-c_vec[active])
            active = active[order[:15]]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        n_act = len(active)
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for idx in active:
            g = shift_grad_single(h, n, idx)
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vec[idx]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        bounds = [(None, None)] * n + [(0.0, rho * 2)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                if np.max(np.abs(delta)) < 1e-13:
                    break
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    stall = 0
                    if rho < 0.06:
                        rho = min(rho * 1.4, 0.08)
                else:
                    stall += 1
                    rho *= 0.6
                    if rho < 1e-10 or stall > 30:
                        break
            else:
                rho *= 0.5
                if rho < 1e-10:
                    break
        except Exception:
            rho *= 0.5
            if rho < 1e-10:
                break
    return best_h, best_c

def local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for st in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active[:10]:
                grad += np.abs(shift_grad_single(h, n, idx))
            grad /= max(len(active), 1)
            top = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + st <= 1.0 + 1e-12 and h[j] - st >= -1e-12:
                        h[i] += st
                        h[j] -= st
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= st
                            h[j] += st
                    if h[i] - st >= -1e-12 and h[j] + st <= 1.0 + 1e-12:
                        h[i] -= st
                        h[j] += st
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += st
                            h[j] -= st
    return h, best_c

def slqp_polish(h, n, deadline, alpha=100000, maxiter=1500):
    def obj_fn(x):
        return smooth_lse(x, n, alpha)
    def obj_jac(x):
        return smooth_lse_grad(x, n, alpha)
    def sq_fn(x):
        return x.sum() - n / 2.0
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': min(maxiter, 300), 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def perturb_reopt(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for trial in range(4):
        if time.time() > deadline - 20:
            break
        noise = rng.randn(n) * 0.03
        h_pert = project_feasible(h + noise, n)
        t_sub = deadline - 30
        if t_sub < time.time() + 5:
            break
        h_opt, c_opt = lbfgs_optimize(h_pert, n, t_sub,
            configs=[(500, 20.0, 500.0), (3000, 100.0, 2000.0)])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def multi_resolution_optimize(h_best, c_best, n_best, deadline, seed):
    for n_try in [350, 600, 800]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        c_try = compute_c5(h_try, n_try)
        t_end = min(time.time() + 80, deadline - 40)
        if t_end < time.time() + 10:
            continue
        h_try, c_try = lbfgs_optimize(h_try, n_try, t_end,
            configs=[(300, 10.0, 200.0), (1500, 50.0, 1000.0), (5000, 200.0, 3000.0)])
        if c_try < c_best:
            c_best = c_try
            h_best = h_try
            n_best = n_try
    return h_best, c_best, n_best
