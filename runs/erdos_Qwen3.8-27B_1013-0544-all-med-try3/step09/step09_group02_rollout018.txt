            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sym_to_full(free, n):
    h = np.zeros(n)
    h[:n // 2] = free
    h[n // 2:] = 1.0 - free[::-1]
    return h

def make_symmetric(h, n):
    m = n // 2
    sym = np.zeros(n)
    for i in range(m):
        avg = 0.5 * (h[i] + 1.0 - h[n - 1 - i])
        avg = np.clip(avg, 0.0, 1.0)
        sym[i] = avg
        sym[n - 1 - i] = 1.0 - avg
    return sym

def free_from_h(h, n):
    m = n // 2
    h_sym = make_symmetric(h, n)
    return h_sym[:m].copy()

def smooth_obj_sym(x, n, alpha):
    h = sym_to_full(x, n)
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    if s_max > 500:
        val = m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    return float(val)

def smooth_grad_sym(x, n, alpha):
    h = sym_to_full(x, n)
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    w = np.exp(s - s_max)
    w /= w.sum()
    dx_val = 2.0 / n
    grad_full = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad_full[0:nv] += wm * (1.0 - h[k:n])
                grad_full[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad_full[kk:n] += wm * (1.0 - h[0:nv])
                grad_full[0:nv] += wm * (-h[kk:n])
    m_half = n // 2
    grad = np.zeros(m_half)
    for i in range(m_half):
        grad[i] = grad_full[i] - grad_full[n - 1 - i]
    return grad

def smooth_obj_asym(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    if s_max > 500:
        val = m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_asym(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    w = np.exp(s - s_max)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_sym_step(free, n, active_idx, rho):
    m = n // 2
    n_var = m + 1
    h = sym_to_full(free, n)
    c_vec = compute_c5_vec(h, n)
    dx_val = 2.0 / n
    n_act = len(active_idx)
    total_ineq = n_act + 4 * m
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        k = idx - (n - 1)
        grad_full = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad_full[0:nv] = (1.0 - h[k:n]) * dx_val
                grad_full[k:n] = (-h[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad_full[kk:n] = (1.0 - h[0:nv]) * dx_val
                grad_full[0:nv] = (-h[kk:n]) * dx_val
        grad_free = np.zeros(m)
        for i in range(m):
            grad_free[i] = grad_full[i] - grad_full[n - 1 - i]
        A_ub[row, 0:m] = grad_free
        A_ub[row, m] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(m):
        A_ub[row, i] = -1.0
        b_ub[row] = free[i]
        row += 1
    for i in range(m):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - free[i]
        row += 1
    for i in range(m):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(m):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    c_obj = np.zeros(n_var)
    c_obj[m] = 1.0
    bounds = [(None, None)] * m + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=None, b_eq=None,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:m]
            return delta, res.x[m]
    except Exception:
        pass
    return None, None

def sqp_sym_optimize(free, n, deadline, max_iter=200, rho_init=0.05):
    best_free = free.copy()
    h = sym_to_full(free, n)
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h = sym_to_full(free, n)
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]
        delta, _ = sqp_sym_step(free, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        free_new = np.clip(free + delta, 0.0, 1.0)
        h_new = sym_to_full(free_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_free = free_new.copy()
            free = free_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.2, 0.1)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 30:
                break
    return best_free, best_c

def lbfgs_sym(free, n, deadline, configs=None):
    best_free = free.copy()
    h = sym_to_full(free, n)
    best_c = compute_c5(h, n)
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    if configs is None:
        configs = [(100, None), (500, None), (2000, None), (5000, None), (10000, None)]
    for alpha in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj_sym, free, jac=smooth_grad_sym,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 50})
            free_new = np.clip(res.x, 0.0, 1.0)
            h_new = sym_to_full(free_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_free = free_new.copy()
                free = free_new
        except Exception:
            pass
    return best_free, best_c

def local_search_sym(free, n, deadline, steps_list=None, top_k=80):
    if steps_list is None:
        steps_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    m = n // 2
    best_free = free.copy()
    h = sym_to_full(free, n)
    best_c = compute_c5(h, n)
    for step in steps_list:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            h = sym_to_full(free, n)
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad_full = np.zeros(n)
            if len(active) > 0:
                for idx in active:
                    k = idx - (n - 1)
                    if k >= 0:
                        nv = n - k
                        if nv > 0:
                            grad_full[0:nv] += (1.0 - h[k:n])
                            grad_full[k:n] += (-h[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        if nv > 0:
                            grad_full[kk:n] += (1.0 - h[0:nv])
                            grad_full[0:nv] += (-h[kk:n])
                grad_full /= len(active)
            grad_free = np.zeros(m)
            for i in range(m):
                grad_free[i] = grad_full[i] - grad_full[n - 1 - i]
            top_pos = np.argsort(-np.abs(grad_free))[:min(top_k, m)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return best_free, best_c
                i = int(i_val)
                if grad_free[i] < 0:
                    delta = step
                else:
                    delta = -step
                if free[i] + delta >= 0.0 - 1e-12 and free[i] + delta <= 1.0 + 1e-12:
                    free[i] += delta
                    h_new = sym_to_full(free, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_free = free.copy()
                        improved = True
                    else:
                        free[i] -= delta
            # Paired moves among top indices
            for ai in range(len(top_pos)):
                for bi in range(ai + 1, len(top_pos)):
                    if time.time() > deadline:
                        return best_free, best_c
                    ia = int(top_pos[ai])
                    ib = int(top_pos[bi])
                    da = step if grad_free[ia] < 0 else -step
                    db = -da
                    if (free[ia] + da >= -1e-12 and free[ia] + da <= 1.0 + 1e-12 and
                        free[ib] + db >= -1e-12 and free[ib] + db <= 1.0 + 1e-12):
                        free[ia] += da
                        free[ib] += db
                        h_new = sym_to_full(free, n)
                        c_new = compute_c5(h_new, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_free = free.copy()
                            improved = True
                        else:
                            free[ia] -= da
                            free[ib] -= db
    return best_free, best_c

def upsample_sym(free_old, n_old, n_new):
    m_old = n_old // 2
    m_new = n_new // 2
    x_old = np.linspace(0, 2, n_old)
    h_old = sym_to_full(free_old, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = make_symmetric(h_new, n_new)
    free_new = h_new[:m_new].copy()
    return np.clip(free_new, 0.0, 1.0)

def slqp_sym_polish(free, n, deadline, alpha=1000000, maxiter=3000):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(smooth_obj_sym, free, jac=smooth_grad_sym,
                      args=(n, alpha), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': 100000, 'maxls': 50})
        free_new = np.clip(res.x, 0.0, 1.0)
        h_new = sym_to_full(free_new, n)
        c_new = compute_c5(h_new, n)
        return free_new, c_new
    except Exception:
        return free.copy(), compute_c5(sym_to_full(free, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init_raw = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init_raw = 500

    if n_init_raw % 2 != 0:
        n_init_raw = n_init_raw - 1
    h_init = h_init[:n_init_raw]
    h_init = project_feasible(h_init, n_init_raw)
    h_init = make_symmetric(h_init, n_init_raw)

    best_c = compute_c5(h_init, n_init_raw)
    best_h = h_init.copy()
    n_best = n_init_raw

    # Phase 1: L-BFGS at initial resolution
    n1 = n_init_raw
    m1 = n1 // 2
    free1 = h_init[:m1].copy()
    t1_end = min(t0 + 35, deadline - 700)
    if t1_end > time.time() + 3:
        free1, c1 = lbfgs_sym(free1, n1, t1_end,
                              configs=[(200, None), (1000, None), (3000, None), (8000, None)])
        if c1 < best_c:
            best_c = c1
            best_h = sym_to_full(free1, n1)
            n_best = n1

    # Phase 2: SQP at initial resolution
    t2_end = min(t0 + 70, deadline - 600)
    if t2_end > time.time() + 5:
        free1, c2 = sqp_sym_optimize(free1, n1, t2_end, max_iter=300, rho_init=0.05)
        if c2 < best_c:
            best_c = c2
            best_h = sym_to_full(free1, n1)
            n_best = n1

    # Phase 3: Upsample to n=1000 and optimize
    n2 = 1000
    free2 = upsample_sym(free1, n1, n2)
    t3_end = min(time.time() + 120, deadline - 400)
    if t3_end > time.time() + 10:
        free2, c3 = lbfgs_sym(free2, n2, t3_end,
                              configs=[(500, None), (2000, None), (5000, None), (10000, None)])
        if c3 < best_c:
            best_c = c3
            best_h = sym_to_full(free2, n2)
            n_best = n2

    t3b_end = min(time.time() + 60, deadline - 300)
    if t3b_end > time.time() + 5:
        free2, c3b = sqp_sym_optimize(free2, n2, t3b_end, max_iter=200, rho_init=0.04)
        if c3b < best_c:
            best_c = c3b
            best_h = sym_to_full(free2, n2)
            n_best = n2

    # Phase 4: Upsample to n=2000 and optimize
    n3 = 2000
    free3 = upsample_sym(free2, n2, n3)
    t4_end = min(time.time() + 150, deadline - 250)
    if t4_end > time.time() + 10:
        free3, c4 = lbfgs_sym(free3, n3, t4_end,
                              configs=[(1000, None), (3000, None), (8000, None), (15000, None)])
        if c4 < best_c:
            best_c = c4
            best_h = sym_to_full(free3, n3)
            n_best = n3

    t4b_end = min(time.time() + 60, deadline - 150)
    if t4b_end > time.time() + 5:
        free3, c4b = sqp_sym_optimize(free3, n3, t4b_end, max_iter=150, rho_init=0.03)
        if c4b < best_c:
            best_c = c4b
            best_h = sym_to_full(free3, n3)
            n_best = n3

    # Phase 5: Fine local search at n=2000
    t5_end = min(time.time() + 200, deadline - 80)
    if t5_end > time.time() + 10:
        free3, c5 = local_search_sym(free3, n3, t5_end, top_k=60)
        if c5 < best_c:
            best_c = c5
            best_h = sym_to_full(free3, n3)
            n_best = n3

    # Phase 6: Another SQP after local search
    t6_end = min(time.time() + 40, deadline - 50)
    if t6_end > time.time() + 5:
        free3, c6 = sqp_sym_optimize(free3, n3, t6_end, max_iter=100, rho_init=0.02)
        if c6 < best_c:
            best_c = c6
            best_h = sym_to_full(free3, n3)
            n_best = n3

    # Phase 7: Very fine local search
    t7_end = min(time.time() + 100, deadline - 30)
    if t7_end > time.time() + 5:
        free3, c7 = local_search_sym(free3, n3, t7_end,
                                     steps_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                     top_k=40)
        if c7 < best_c:
            best_c = c7
            best_h = sym_to_full(free3, n3)
            n_best = n3

    # Phase 8: Final L-BFGS polish
    t8_end = deadline - 3
    if t8_end > time.time() + 5:
        free3, c8 = slqp_sym_polish(free3, n3, t8_end, alpha=5000000, maxiter=5000)
        if c8 < best_c:
            best_c = c8
            best_h = sym_to_full(free3, n3)
            n_best = n3

    # Final validation
    best_h = sym_to_full(np.clip(free3, 0.0, 1.0), n_best)
    best_c = compute_c5(best_h, n_best)

    # Also check if a slightly different n gives better result
    # (the discrete optimum might depend on n)
    if time.time() < deadline - 20:
        for n_try in [1500, 2500, 3000]:
            if time.time() > deadline - 20:
                break
            if n_try % 2 != 0:
                continue
            free_try = upsample_sym(np.clip(free3, 0, 1), n_best, n_try)
            h_try = sym_to_full(free_try, n_try)
            c_try = compute_c5(h_try, n_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
                n_best = n_try

    return best_h, best_c, n_best
