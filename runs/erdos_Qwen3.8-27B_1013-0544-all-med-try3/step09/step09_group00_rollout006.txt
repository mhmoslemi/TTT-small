    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n_new)

def gradient_local_search(h, n, deadline, top_k=8, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = -1
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                gi = grad[i] * best_h[i]
                for j in neg_idx[:40]:
                    g = gi - grad[j] * (1.0 - best_h[j])
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                if steps_done % 100 == 0:
                    break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        iters = 0
        while improved and time.time() < deadline:
            improved = False
            iters += 1
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            j = hi
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            j = hi
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def multimove_local_search(h, n, deadline, base_step=0.01, max_rounds=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 12:
            active = np.array(np.argsort(-c_vec[active])[:12])
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        improve_total = 0.0
        mask = (np.abs(grad) > 1e-9)
        idxs = np.where(mask)[0]
        rng_r = np.random.RandomState(rnd)
        rng_r.shuffle(idxs)
        for i in idxs[:200]:
            if time.time() > deadline - 5:
                break
            direction = -1.0 if grad[i] > 0 else 1.0
            delta = direction * base_step
            new_val = best_h[i] + delta
            if new_val < 0 or new_val > 1:
                continue
            comp = best_h.sum() - new_val
            mask2 = (best_h > 0) & (best_h < 1.0) & (np.arange(n) != i)
            if not mask2.any():
                continue
            candidates = np.where(mask2)[0]
            if len(candidates) > 50:
                g_sub = grad[candidates]
                top_cands = candidates[np.argsort(-np.abs(g_sub))[:50]]
            else:
                top_cands = candidates
            for j in top_cands[:10]:
                corr = delta * (1.0 - (best_h.sum() - best_h[i]) / comp) if comp > 1e-15 else 0
                adj = -delta * best_h[j] / comp if comp > 1e-15 else 0
                new_j = best_h[j] + adj
                if new_j < 0 or new_j > 1:
                    continue
                old_i = best_h[i]
                old_j = best_h[j]
                best_h[i] = new_val
                best_h[j] = new_j
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improve_total += (max_val - c_new)
                    break
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
        if improve_total < 1e-14:
            base_step *= 0.5
            if base_step < 1e-8:
                break
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxit_base=2000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = min(maxit_base + i * 1000, max(500, int(remaining * 3)))
        try:
            res = minimize(lm_obj, best_h, jac=lm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, configs=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [
            (4, 50000, 4, 0.001),
            (6, 200000, 4, 0.001),
            (8, 1000000, 3, 0.0005),
            (12, 5000000, 3, 0.0005),
            (16, 20000000, 2, 0.0001),
            (24, 100000000, 2, 0.0001),
        ]
    for top_k, alpha, rounds, mu in configs:
        if time.time() > deadline - 10:
            break
        for rnd in range(rounds):
            remaining = deadline - time.time()
            if remaining < 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            maxit = max(200, int(remaining * 2))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def surge_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        max_idx = np.argmax(c_vec)
        k = max_idx - (n - 1)
        dx_val = 2.0 / n
        if k >= 0:
            nv = n - k
            contrib_a = best_h[k:n]
            contrib_b = 1.0 - best_h[0:nv]
            others_a = np.zeros(n)
            others_a[k:n] = 1.0 - best_h[0:nv]
            others_b = np.zeros(n)
            others_b[0:nv] = -best_h[k:n]
        else:
            kk = -k
            nv = n - kk
            contrib_a = best_h[kk:n]
            contrib_b = 1.0 - best_h[0:nv]
            others_a = np.zeros(n)
            others_a[kk:n] = 1.0 - best_h[0:nv]
            others_b = np.zeros(n)
            others_b[0:nv] = -best_h[kk:n]
        score_decrease = (others_a * best_h + others_b * (1.0 - best_h)) * dx_val
        score_increase = (others_a * (1.0 - best_h) + others_b * best_h) * dx_val
        cand_dec = np.where(score_decrease > 0.001)[0]
        cand_inc = np.where(score_increase < -0.001)[0]
        if len(cand_dec) == 0 or len(cand_inc) == 0:
            break
        found = False
        for i in cand_dec[:30][np.random.choice(len(cand_dec[:30]), min(10, len(cand_dec[:30])), replace=False)]:
            if time.time() > deadline:
                break
            for j in cand_inc[:30][np.random.choice(len(cand_inc[:30]), min(10, len(cand_inc[:30])), replace=False)]:
                amt = min(0.01, best_h[i], 1.0 - best_h[j])
                if amt < 1e-10:
                    continue
                old_i, old_j = best_h[i], best_h[j]
                best_h[i] -= amt
                best_h[j] += amt
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    found = True
                    break
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
            if found:
                break
        if not found:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = make_beatty_start(500, 0.6180339887)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate diverse starting points at original resolution
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alphas_list = [golden / 2.0, np.sqrt(2.0) / 2.0, np.sqrt(3.0) / 2.0,
                   1.0 / np.e, (1.0 + np.sqrt(5.0)) / 4.0, 0.5 + 0.1 * np.pi]
    for ap in alphas_list:
        h_alt = make_beatty_start(n_orig, ap)
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    for _ in range(4):
        h_rnd = make_random_binary(n_orig, rng)
        c_rnd = compute_c5(h_rnd, n_orig)
        if c_rnd < best_c:
            best_c = c_rnd
            best_h = h_rnd.copy()

    # Phase 1: Optimize at original resolution (n_orig)
    # L-BFGS with increasing smoothness
    p1_end = min(t0 + 120, deadline - 500)
    h1, c1 = lbfgs_optimize(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000], mu=0.01, maxit_base=1500)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Active set optimization
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 120, deadline - 350)
        h2, c2 = active_set_optimize(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Gradient local search
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 80, deadline - 250)
        h3, c3 = gradient_local_search(best_h, best_n, p3_end,
            top_k=10, step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=80000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Surge search (focus on the single worst shift)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 50, deadline - 200)
        h4, c4 = surge_search(best_h, best_n, p4_end, max_rounds=100)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Fine pair search at original resolution
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 60, deadline - 150)
        h5, c5 = fine_pair_search(best_h, best_n, p5_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 2: Try higher resolution (n=1000)
    n_hi = max(1000, best_n * 2)
    if time.time() < deadline - 200:
        # Upsample
        h_up = upsample(best_h, best_n, n_hi)
        c_up = compute_c5(h_up, n_hi)
        h_hi = h_up.copy()
        c_hi = c_up

        # Also try fresh Beatty at high res
        for ap in [golden / 2.0, np.sqrt(2.0) / 2.0, np.sqrt(3.0) / 2.0]:
            h_b = make_beatty_start(n_hi, ap)
            c_b = compute_c5(h_b, n_hi)
            if c_b < c_hi:
                c_hi = c_b
                h_hi = h_b

        # Optimize at high resolution
        p6_end = min(time.time() + 150, deadline - 100)
        h6, c6 = lbfgs_optimize(h_hi, n_hi, p6_end,
            alphas=[20, 100, 500, 2000, 10000], mu=0.01, maxit_base=1000)
        if c6 < c_hi:
            c_hi = c6
            h_hi = h6

        # Active set at high res
        if time.time() < deadline - 60:
            p7_end = min(time.time() + 60, deadline - 40)
            h7, c7 = active_set_optimize(h_hi, n_hi, p7_end,
                configs=[(4, 50000, 3, 0.001), (8, 500000, 2, 0.0005), (12, 5000000, 2, 0.0001)])
            if c7 < c_hi:
                c_hi = c7
                h_hi = h7

        # Quick local search at high res
        if time.time() < deadline - 30:
            p8_end = min(time.time() + 30, deadline - 15)
            h8, c8 = gradient_local_search(h_hi, n_hi, p8_end,
                top_k=8, step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                max_steps=30000)
            if c8 < c_hi:
                c_hi = c8
                h_hi = h8

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 3: Try even higher resolution if we made progress
    if best_n < 2000 and time.time() < deadline - 100:
        n_hi2 = max(2000, best_n * 2)
        h_up2 = upsample(best_h, best_n, n_hi2)
        c_up2 = compute_c5(h_up2, n_hi2)
        h_hi2 = h_up2.copy()
        c_hi2 = c_up2

        p9_end = min(time.time() + 80, deadline - 40)
        h9, c9 = lbfgs_optimize(h_hi2, n_hi2, p9_end,
            alphas=[10, 50, 200, 1000], mu=0.01, maxit_base=800)
        if c9 < c_hi2:
            c_hi2 = c9
            h_hi2 = h9

        if time.time() < deadline - 15:
            p10_end = min(time.time() + 20, deadline - 10)
            h10, c10 = gradient_local_search(h_hi2, n_hi2, p10_end,
                top_k=6, step_sizes=[0.001, 0.0005, 0.0002, 0.0001],
                max_steps=20000)
            if c10 < c_hi2:
                c_hi2 = c10
                h_hi2 = h10

        if c_hi2 < best_c:
            best_c = c_hi2
            best_h = h_hi2.copy()
            best_n = n_hi2

    # Phase 4: Final polishing at best resolution
    if time.time() < deadline - 20:
        pfin_end = min(time.time() + 40, deadline - 10)
        hfin, cfin = fine_pair_search(best_h, best_n, pfin_end, radius=40,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if cfin < best_c:
            best_c = cfin
            best_h = hfin

    # Final: very small nudges
    if time.time() < deadline - 5:
        pfin2_end = min(time.time() + 15, deadline - 2)
        hfin2, cfin2 = gradient_local_search(best_h, best_n, pfin2_end,
            top_k=12, step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            max_steps=20000)
        if cfin2 < best_c:
            best_c = cfin2
            best_h = hfin2

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
