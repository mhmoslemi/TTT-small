def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.clip(h + frac * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = diff / tot
            h = np.clip(h + frac * h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            zeros = np.where(h < 1.0)[0]

        else:

            if len(ones) > 0:

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    arr = np.array(list(positions))
    if len(arr) >= n_ones:
        h[arr[:n_ones]] = 1.0
    else:
        h[arr] = 1.0
    remaining = n / 2.0 - h.sum()
    if remaining > 0:
        zeros = np.where(h == 0)[0]
        for _ in range(min(int(remaining), len(zeros))):
            h[zeros[_]] = 1.0
        rem2 = n / 2.0 - h.sum()
        if rem2 > 0:
            h[zeros[int(remaining)]] += rem2
    return project_feasible(h, n)

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = min(maxit_list[i], int(remaining * 5))
        else:
            maxit = max(200, int(remaining * 3))
        maxit = max(100, maxit)
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active_set(h, n, deadline, configs=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [
            (20, 50000, 3, 0.002),
            (10, 200000, 4, 0.001),
            (6, 500000, 4, 0.001),
            (4, 2000000, 5, 0.0005),
            (3, 5000000, 5, 0.0005),
            (2, 10000000, 6, 0.0003),
        ]
    for top_k, alpha, rounds, mu in configs:
        for rnd in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0003 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            remaining = deadline - time.time()
            maxit = max(150, int(remaining * 2))
            maxit = min(maxit, 3000)
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 35, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-14:
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def grad_coord_descent(h, n, deadline, max_steps=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    step_schedules = [
        (0.005, 0.001, 20000),
        (0.002, 0.0005, 30000),
        (0.001, 0.0002, 40000),
        (0.0005, 0.0001, 40000),
        (0.0002, 0.00005, 30000),
        (0.0001, 0.00002, 30000),
        (0.00005, 0.00001, 20000),
    ]
    total_steps = 0
    for step_hi, step_lo, max_s in step_schedules:
        for _ in range(max_s):
            if time.time() > deadline or total_steps >= max_steps:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            grad = np.zeros(n)
            for idx in active[:30]:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            if len(active) > 0:
                grad /= min(len(active), 30)
            pos_mask = (grad > 5e-8) & (best_h > 5e-8)
            neg_mask = (grad < -5e-8) & (best_h < 1.0 - 5e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            step = step_hi * (step_lo / step_hi) ** (min(1.0, total_steps / 10000.0))
            # Try multiple candidate pairs
            candidates = []
            for i in pos_idx[:15]:
                for j in neg_idx[:15]:
                    g = grad[i] - grad[j]
                    candidates.append((g, i, j))
            candidates.sort(reverse=True)
            improved = False
            for g, i, j in candidates[:5]:
                if g < 0:
                    break
                amt = min(step, best_h[i] * 0.9, (1.0 - best_h[j]) * 0.9)
                if amt < 1e-12:
                    continue
                best_h[i] -= amt
                best_h[j] += amt
                c_new = compute_c5(best_h, n)
                total_steps += 1
                if c_new < best_c - 1e-17:
                    best_c = c_new
                    improved = True
                    break
                else:
                    best_h[i] += amt
                    best_h[j] -= amt
            if not improved:
                break
    return best_h, best_c

def fine_pair_scan(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n - 1, i + radius)
                for j in range(lo, hi + 1):
                    if i == j:
                        continue
                    # Move mass from i to j
                    if best_h[i] >= d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Move mass from j to i
                    if best_h[j] >= d and best_h[i] + d <= 1.0:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

def block_coord_descent(h, n, deadline, block_size=3, max_rounds=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    target = n / 2.0
    rng = np.random.RandomState(12345)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Pick a random block to decrease and a random block to increase
        i_start = rng.randint(0, n - block_size)
        j_start = rng.randint(0, n - block_size)
        if abs(i_start - j_start) < block_size * 2:
            continue
        i_block = np.arange(i_start, i_start + block_size)
        j_block = np.arange(j_start, j_start + block_size)
        total_i = best_h[i_block].sum()
        total_j = best_h[j_block].sum()
        if total_i < 1e-6 or total_j > n / 2.0 - 1e-6:
            continue
        # Try transferring small amount
        for frac in [0.01, 0.005, 0.002, 0.001]:
            amt = total_i * frac
            if amt < 1e-8:
                break
            best_h[i_block] -= amt / block_size
            best_h[j_block] += amt / block_size
            best_h = np.clip(best_h, 0, 1)
            c = compute_c5(best_h, n)
            if c < best_c - 1e-17:
                best_c = c
                break
            else:
                best_h[i_block] += amt / block_size
                best_h[j_block] -= amt / block_size
                best_h = np.clip(best_h, 0, 1)
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    def _obj(x):
        return smooth_max_val(compute_c5_vec(x, n), alpha)
    try:
        rem = max(5, deadline - time.time())
        maxit = min(maxiter, int(rem * 8))
        maxit = max(50, maxit)
        res = minimize(_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(400, 0.5)
        n_orig = 400

    h0 = project_feasible(h0, n_orig)

    # Choose target resolution
    n_target = 1000

    # Generate candidates at target resolution
    candidates = []

    # Upsample from initial
    h_up = upsample(h0, n_orig, n_target)
    candidates.append(('up', h_up, compute_c5(h_up, n_target)))

    # Beatty sequences
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    beatty_starts = [
        ('b_g', make_beatty_start(n_target, golden / 2.0)),
        ('b_s2', make_beatty_start(n_target, np.sqrt(2.0) / 2.0)),
        ('b_s3', make_beatty_start(n_target, np.sqrt(3.0) / 2.0)),
    ]
    for name, hb in beatty_starts:
        c_b = compute_c5(hb, n_target)
        candidates.append((name, hb, c_b))

    # Random binary
    for ri in range(3):
        hr = make_random_binary(n_target, rng)
        c_r = compute_c5(hr, n_target)
        candidates.append(('r%d' % ri, hr, c_r))

    # Sinusoidal
    for freq in [0.75, 1.0, 1.5]:
        hs = make_sinusoidal_start(n_target, freq)
        c_s = compute_c5(hs, n_target)
        candidates.append(('s%.1f' % freq, hs, c_s))

    # Also keep the original resolution solution as backup
    c_orig = compute_c5(h0, n_orig)
    best_h = h0.copy()
    best_c = c_orig
    best_n = n_orig

    # Pick best candidate
    candidates.sort(key=lambda x: x[2])
    for name, hc, cc in candidates:
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
            best_n = n_target

    # Phase 1: L-BFGS warmup (first ~300s)
    p1_end = min(t0 + 280, deadline - 400)
    if best_n == n_target:
        h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
            alphas=[5, 20, 100, 500, 2000, 10000, 50000, 200000, 500000],
            mu=0.005)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization
    if time.time() < deadline - 250:
        p2_end = min(time.time() + 200, deadline - 200)
        h2, c2 = phase_active_set(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient coordinate descent
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 180, deadline - 150)
        h3, c3 = grad_coord_descent(best_h, best_n, p3_end, max_steps=300000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Active set again (with refined active set)
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 120, deadline - 120)
        h4, c4 = phase_active_set(best_h, best_n, p4_end,
            configs=[
                (8, 500000, 3, 0.001),
                (4, 2000000, 4, 0.0005),
                (3, 5000000, 5, 0.0003),
                (2, 10000000, 6, 0.0002),
            ])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Fine pair scan
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 100, deadline - 70)
        h5, c5 = fine_pair_scan(best_h, best_n, p5_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Block coordinate descent
    if time.time() < deadline - 70:
        p6_end = min(time.time() + 60, deadline - 50)
        h6, c6 = block_coord_descent(best_h, best_n, p6_end, block_size=4, max_rounds=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish
    if time.time() < deadline - 40:
        for alpha_val in [5000000, 20000000, 50000000]:
            if time.time() > deadline - 25:
                break
            h7, c7 = slqp_polish(best_h, best_n, deadline - 10, alpha=alpha_val, maxiter=500)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Very fine adjustments
    if time.time() < deadline - 10:
        h8, c8 = fine_pair_scan(best_h, best_n, deadline - 3, radius=25,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: also try optimizing at original resolution if that's better
    if best_n != n_orig:
        # Compare with optimized original resolution
        h_orig_opt, c_orig_opt = phase_lbfgs(h0, n_orig, deadline - 2,
            alphas=[100000, 500000, 2000000], mu=0.005)
        if c_orig_opt < best_c:
            best_c = c_orig_opt
            best_h = h_orig_opt
            best_n = n_orig

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
