Actually, for the Fourier approach, I can compute the correlation analytically! If h(x) = 0.5 + f(x), then:

Actually for the discrete case, it's easier to just compute numerically. But the gradient should be computed efficiently.

Let me think about this differently. Instead of computing the gradient analytically in the coefficient space, I can use the chain rule:

∂C_k/∂θ_j = Σ_i (∂C_k/∂h_m) * (∂h_m/∂θ_j)

where ∂C_k/∂h_m is the gradient with respect to h values (which I already have from the previous program), and ∂h_m/∂θ_j is just the basis function evaluated at x_m.

So the gradient in coefficient space is:
∂C_k/∂θ_j = Σ_m (∂C_k/∂h_m) * φ_j(x_m)

This is a matrix product: grad_coeffs = B^T @ grad_h where B is the basis matrix.

This is much more efficient! Let me implement it this way.

Let me use a soft constraint: instead of clipping, use a sigmoid-like mapping or just a large penalty.

Actually, for the Fourier approach, the key advantage is that we're in a low-dimensional space (2p parameters instead of n). So even if the landscape is not perfectly smooth, L-BFGS-B should handle it well.

Let me also think about what p to use. If p is too small, we can't represent binary-like functions. If p is too large, we lose the dimensionality advantage.

For a binary function with sharp transitions, we need high-frequency components. The transition width in the discrete case is ~1/n, so we need frequencies up to ~n/2. But that would make p ~ n/2, defeating the purpose.

A compromise: use p = 20-50. This captures the global structure (which lags are bad, where the mass is concentrated) but not the fine details. Then refine with local search.

```python
def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    shifted = alpha * (c - m)
    mx = shifted.max()

        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        w = smooth_max_weights(c, alpha)
        grad = grad_wrt_h(x, n, w)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_wrt_h(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        lag = k - (n - 1)
        wk = w[k] * dx
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - h[:-lag])
            grad[:-lag] += wk * (-h[lag:])
        elif -n < lag < 0:
            idx = -lag
            grad[:-idx] += wk * (1.0 - h[idx:])
            grad[idx:] += wk * (-h[:-idx])
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        w = smooth_max_weights(c, alpha)
        grad = grad_wrt_h(x, n, w)
        grad += 2.0e6 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def fourier_optimize(h_start, n, p, alpha, deadline, maxiter=1000):
    x = np.linspace(0, 2, n, endpoint=False)
    B = np.zeros((n, 2 * p))
    for k in range(1, p + 1):
        B[:, 2 * (k - 1)] = np.cos(np.pi * k * x)
        B[:, 2 * (k - 1) + 1] = np.sin(np.pi * k * x)
    target = h_start - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(B, target, rcond=None)
    pen_coeff = 5e7

    def obj_and_grad(theta):
        h_unc = 0.5 + B @ theta
        pos_v = np.maximum(0, h_unc - 1)
        neg_v = np.maximum(0, -h_unc)
        penalty = pen_coeff * (np.sum(pos_v ** 2) + np.sum(neg_v ** 2))
        h = np.clip(h_unc, 0, 1)
        c = compute_all_lags(h, n)
        obj = smooth_max_value(c, alpha) + penalty
        w = smooth_max_weights(c, alpha)
        grad_h = grad_wrt_h(h, n, w)
        grad_theta = B.T @ grad_h
        grad_theta += B.T @ (2 * pen_coeff * (pos_v - neg_v))
        return obj, grad_theta

    try:
        res = minimize(obj_and_grad, coeffs, method='L-BFGS-B',
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': -1})
        h_final = np.clip(0.5 + B @ res.x, 0, 1)
        return project_feasible(h_final, n)
    except Exception:
        h_final = np.clip(0.5 + B @ coeffs, 0, 1)
        return project_feasible(h_final, n)

def binary_sa(h, n, deadline, rng):
    h_cur = (h > 0.5).astype(float)
    num_ones = int(h_cur.sum())
    target = n // 2
    if num_ones > target:
        idx = np.where(h_cur == 1)[0]
        order = np.argsort(h[idx])
        for i in order[:num_ones - target]:
            h_cur[idx[i]] = 0
    elif num_ones < target:
        idx = np.where(h_cur == 0)[0]
        order = np.argsort(-h[idx])
        for i in order[:target - num_ones]:
            h_cur[idx[i]] = 1

    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    cur_c = best_c

    T = 0.12
    t_min = 0.00003
    cool = 0.99992
    stall = 0
    max_stall = 8000

    while time.time() < deadline and T > t_min:
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        if r < 0.45:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            c_all = compute_all_lags(h_cur, n)
            top = np.argsort(c_all)[-4:]
            k_star = top[rng.randint(len(top))]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h_cur[i] == h_cur[j]:
                T *= cool
                continue
        else:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j or h_cur[i] == h_cur[j]:
                T *= cool
                continue

        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        new_c = compute_c5(h_cur, n)
        de = new_c - cur_c

        if de < 0 or (T > t_min and rng.random() < np.exp(-de / max(T, 1e-12))):
            cur_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h_cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            stall += 1

        T *= cool
        if stall > max_stall:
            h_cur = best_h.copy()
            for _ in range(20):
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if h_cur[i] != h_cur[j]:
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            cur_c = compute_c5(h_cur, n)
            stall = 0
            T = max(T, 0.015)

    return best_h, best_c

def local_search(h, n, deadline, rng):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    K = 5

    for outer in range(2000):
        if time.time() > deadline:
            break
        top_lags = np.argsort(c_all)[-K:]
        improved = False

        for k_sel in top_lags:
            if time.time() > deadline:
                break
            lag = k_sel - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            best_move_c = best_c
            best_move = None
            n_pairs = n - m
            step = max(1, n_pairs // 150)
            for i in range(0, n_pairs, step):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    d = min(0.008, (h[i] - h[j]) * 0.25)
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_move_c:
                        best_move_c = c_test
                        best_move = (i, j, d)
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    d = min(0.008, (h[j] - h[i]) * 0.25)
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_move_c:
                        best_move_c = c_test
                        best_move = (i, j, -d)
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)

            if best_move is not None:
                i, j, d = best_move
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                if best_move_c < best_c - 1e-14:
                    best_c = best_move_c
                    best_h = h.copy()
                    improved = True
                c_all = compute_all_lags(h, n)
                top_lags = np.argsort(c_all)[-K:]

        if not improved:
            # Try smaller steps
            improved2 = False
            c_all = compute_all_lags(h, n)
            for k_sel2 in np.argsort(c_all)[-8:]:
                if time.time() > deadline:
                    break
                lag2 = k_sel2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                if m2 >= n:
                    continue
                n_pairs2 = n - m2
                step2 = max(1, n_pairs2 // 100)
                for i2 in range(0, n_pairs2, step2):
                    if time.time() > deadline:
                        break
                    j2 = i2 + m2
                    if h[i2] > h[j2] + 1e-9:
                        d2 = min(0.001, (h[i2] - h[j2]) * 0.15)
                        if d2 < 1e-12:
                            continue
                        h[i2] -= d2
                        h[j2] += d2
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-14:
                            best_c = c_test
                            best_h = h.copy()
                            improved2 = True
                            c_all = compute_all_lags(h, n)
                        else:
                            h[i2] += d2
                            h[j2] -= d2
                            h = project_feasible(h, n)
                    elif h[j2] > h[i2] + 1e-9:
                        d2 = min(0.001, (h[j2] - h[i2]) * 0.15)
                        if d2 < 1e-12:
                            continue
                        h[i2] += d2
                        h[j2] -= d2
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-14:
                            best_c = c_test
                            best_h = h.copy()
                            improved2 = True
                            c_all = compute_all_lags(h, n)
                        else:
                            h[i2] -= d2
                            h[j2] += d2
                            h = project_feasible(h, n)
            if not improved2:
                break

    return best_h, best_c

def continuous_sa(h, n, deadline, rng, alpha_sa=50000, t_init=0.005, t_min=0.00002, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stall = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.6:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.08, 0.003 + T * 6)
            if h[i] > h[j]:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.4, (1.0 - h[j]) * 0.4)
            elif h[j] > h[i]:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.4, -(1.0 - h[i]) * 0.4)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-3:]
            k_star = top_k[rng.randint(3)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-7:
                d = rng.uniform(0.002, 0.03)
                d = min(d, (h[i] - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-7:
                d = -rng.uniform(0.002, 0.03)
                d = max(d, -(h[j] - h[i]) * 0.3)
            else:
                T *= cool
                continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = float(c_new.max())

        h[i] += d
        h[j] -= d
        h = project_feasible(h, n)
        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 40:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            if max_new < best_c - 1e-14:
                best_c = max_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        T *= cool
        if stall > 50000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            stall = 0
            T = max(T, 0.002)

    return best_h, best_c

def interpolate(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B at parent resolution with increasing alpha
    for alpha, maxit in [(500, 500), (5000, 600), (50000, 800), (500000, 1000)]:
        if time.time() > deadline - 450:
            break
        t_phase = min(25, deadline - 450 - time.time())
        if t_phase < 3:
            break
        h_opt = lbfgsb_optimize(best_h, n_init, alpha, time.time() + t_phase, maxiter=maxit)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Fourier coefficient optimization
    if time.time() < deadline - 350:
        for p in [15, 30, 50]:
            if time.time() > deadline - 300:
                break
            t_phase = min(35, deadline - 300 - time.time())
            if t_phase < 5:
                break
            for alpha_f in [10000, 100000]:
                if time.time() > deadline - 280:
                    break
                h_four = fourier_optimize(best_h, n_init, p, alpha_f, time.time() + min(20, t_phase), maxiter=800)
                c_four = compute_c5(h_four, n_init)
                if c_four < best_c:
                    best_c = c_four
                    best_h = h_four
                    # Polish
                    h_pol = lbfgsb_optimize(best_h, n_init, 100000, time.time() + 15, maxiter=1500)
                    c_pol = compute_c5(h_pol, n_init)
                    if c_pol < best_c:
                        best_c = c_pol
                        best_h = h_pol

    # Phase 3: Binary SA
    if time.time() < deadline - 250:
        sa_dead = min(deadline - 180, time.time() + 220)
        h_bin, c_bin = binary_sa(best_h, n_init, sa_dead, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Polish binary with continuous optimization
        if time.time() < deadline - 150:
            h_pol = lbfgsb_optimize(best_h, n_init, 200000, time.time() + 40, maxiter=3000)
            c_pol = compute_c5(h_pol, n_init)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol

    # Phase 4: Continuous SA
    if time.time() < deadline - 150:
        sa_dead2 = min(deadline - 80, time.time() + 100)
        h_sa, c_sa = continuous_sa(best_h, n_init, sa_dead2, rng, alpha_sa=50000, t_init=0.004, t_min=0.00002, cool=0.99992)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 5: Higher resolution (n=1000)
    if time.time() < deadline - 100:
        n_high = 1000
        h_high = interpolate(best_h, len(best_h), n_high)
        h_opt = lbfgsb_optimize(h_high, n_high, 20000, time.time() + 30, maxiter=2000)
        c_opt = compute_c5(h_opt, n_high)
        if time.time() < deadline - 60:
            h_opt2 = lbfgsb_optimize(h_opt, n_high, 500000, time.time() + 30, maxiter=3000)
            c_opt2 = compute_c5(h_opt2, n_high)
            if c_opt2 < c_opt:
                h_opt, c_opt = h_opt2, c_opt2
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Intensive local search at best resolution
    if time.time() < deadline - 40:
        n_final = len(best_h)
        ls_dead = min(deadline - 15, time.time() + 120)
        h_ls, c_ls = local_search(best_h, n_final, ls_dead, rng)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 7: Final L-BFGS-B polish
    if time.time() < deadline - 8:
        n_final = len(best_h)
        t_rem = deadline - 6 - time.time()
        if t_rem > 3:
            h_opt = lbfgsb_optimize(best_h, n_final, 5000000, time.time() + min(t_rem, 30), maxiter=5000)
            c_opt = compute_c5(h_opt, n_final)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 8: Tiny final local search
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_ls, c_ls = local_search(best_h, n_final, deadline - 2, rng)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Final validation
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    best_c = compute_c5(best_h, n_out)

    return best_h, best_c, n_out
```
