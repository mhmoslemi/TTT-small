```python
import time
import numpy as np

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

_OPT = {'n': 0, 'alpha': 1.0, 'pen': 1.0e5, 'target': 0.0}


def compute_c5_exact(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def make_c_unified(h, n):
    return np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float, copy=False)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - float(h.sum())
        if abs(diff) < 1.0e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1.0e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot < 1.0e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def resample_h(h_src, n_src, n_dst):
    if n_dst == n_src:
        return project_feasible(h_src, n_dst)
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def search_params(n):
    if n <= 600:
        return 18, 1, 18000, 0.02
    if n <= 1200:
        return 22, 1, 12000, 0.02
    return 16, 0, 8000, 0.02


def smooth_value_c(c, alpha):
    m = float(c.max())
    z = float(alpha) * (c - m)
    zm = float(z.max())
    if zm > 0.0:
        z = z - zm
    z = np.maximum(z, -745.0)
    logsum = float(np.log(np.sum(np.exp(z))))
    return m + (zm + logsum) / float(alpha)


def smooth_weights_c(c, alpha):
    m = float(c.max())
    z = float(alpha) * (c - m)
    zm = float(z.max())
    if zm > 0.0:
        z = z - zm
    z = np.maximum(z, -745.0)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0:
        return np.full_like(w, 1.0 / float(w.size))
    return w / s


def smooth_max_gradient_np(h, c, w, n):
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    L = 2 * n - 1
    for p in range(L):
        wp = float(w[p])
        if wp <= 1.0e-15:
            continue
        lag = p - (n - 1)
        if lag == 0:
            grad += wp * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wp * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wp * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wp * (1.0 - h[idx:n])
            grad[idx:n] += wp * (-h[0:n - idx])
    return grad * dx


def opt_fg(x):
    n = int(_OPT['n'])
    alpha = float(_OPT['alpha'])
    pen = float(_OPT['pen'])
    target = float(_OPT['target'])
    h = x
    if h.min() < -1.0e-12 or h.max() > 1.0 + 1.0e-12:
        h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_value_c(c, alpha)
    diff = float(h.sum()) - target
    val += pen * diff * diff
    w = smooth_weights_c(c, alpha)
    grad = smooth_max_gradient_np(h, c, w, n)
    grad += 2.0 * pen * diff
    return val, grad


def lbfgsb_smooth(h_in, n, alpha, maxiter, deadline):
    h = project_feasible(h_in, n)
    if minimize is None:
        return h
    rem = float(deadline) - time.time()
    if rem < 15.0 or maxiter <= 0:
        return h
    if n <= 550:
        rate = 28.0
    elif n <= 900:
        rate = 11.0
    else:
        rate = 5.0
    maxit = int(min(maxiter, max(15, int(rem * 0.55 * rate))))
    _OPT['n'] = n
    _OPT['alpha'] = float(alpha)
    _OPT['pen'] = 1.0e5
    _OPT['target'] = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * n
    options = {
        'maxiter': maxit,
        'ftol': 1.0e-15,
        'gtol': 1.0e-8,
        'maxls': 30,
        'iprint': -1,
    }
    x0 = h
    try:
        res = minimize(opt_fg, x0, method='L-BFGS-B', bounds=bounds, jac=True, options=options)
        if res is not None and res.x is not None:
            x = np.asarray(res.x, dtype=float)
        else:
            x = x0
    except Exception:
        x = x0
    return project_feasible(np.clip(x, 0.0, 1.0), n)


def smooth_polish(h_in, n, deadline):
    h = np.asarray(h_in, dtype=float).copy()
    h = project_feasible(h, n)
    c5 = compute_c5_exact(h, n)
    alphas = (1200.0, 12000.0, 120000.0, 1200000.0)
    max_iters = (180, 220, 260, 320)
    for i in range(len(alphas)):
        rem = float(deadline) - time.time()
        if rem < 35.0:
            break
        h2 = lbfgsb_smooth(h, n, alphas[i], max_iters[i], deadline)
        c2 = compute_c5_exact(h2, n)
        if c2 < c5 - 1.0e-14:
            h = h2
            c5 = c2
    return h, float(c5)


def fill_move_A(om, hrev, A, i, j, n):
    A[:] = 0.0
    A[n - 1 - i:2 * n - 1 - i] -= om
    A[i:i + n] += hrev
    A[n - 1 - j:2 * n - 1 - j] += om
    A[j:j + n] -= hrev


def apply_move_aux(h, om, hrev, c, i, j, delta, n):
    if i == j or delta == 0.0:
        return
    d = float(delta)
    c[n - 1 - i:2 * n - 1 - i] -= d * om
    c[i:i + n] += d * hrev
    c[n - 1 - j:2 * n - 1 - j] += d * om
    c[j:j + n] -= d * hrev
    d2 = d * d
    c[n - 1] -= 2.0 * d2
    sep = i - j
    if sep < 0:
        sep = -sep
    if sep != 0:
        c[n - 1 + sep] += d2
        c[n - 1 - sep] += d2
    h[i] -= d
    h[j] += d
    hi = h[i]
    hj = h[j]
    if hi < 0.0:
        if hi > -1.0e-12:
            h[i] = 0.0
    elif hi > 1.0:
        if hi < 1.0 + 1.0e-12:
            h[i] = 1.0
    if hj < 0.0:
        if hj > -1.0e-12:
            h[j] = 0.0
    elif hj > 1.0:
        if hj < 1.0 + 1.0e-12:
            h[j] = 1.0
    om[i] = 1.0 - h[i]
    om[j] = 1.0 - h[j]


def line_search_pair(h, om, hrev, c, A, tmp, i, j, n, base, current_max, detail):
    hi = h[i]
    hj = h[j]
    maxd = hi
    one_minus_hj = 1.0 - hj
    if one_minus_hj < maxd:
        maxd = one_minus_hj
    if maxd <= 1.0e-12:
        return 0.0, current_max
    maxd *= (1.0 - 1.0e-12)

    cands = []
    base_d = base
    if base_d > maxd:
        base_d = maxd
    primary = (base_d, maxd, 0.5 * maxd, max(0.002, 0.05 * maxd), 0.001)
    extra = ()
    if detail >= 2:
        extra = (0.0005, 0.0002, 0.2 * maxd)
    allvals = primary + extra
    for dv in allvals:
        d = dv
        if d > maxd:
            d = maxd
        if d <= 1.0e-11:
            continue
        dup = False
        for y in cands:
            tol = 1.0e-7 + 1.0e-4 * max(1.0, abs(y))
            if abs(d - y) <= tol:
                dup = True
                break
        if not dup:
            cands.append(d)
    if not cands:
        return 0.0, current_max

    fill_move_A(om, hrev, A, i, j, n)
    sep = i - j
    if sep < 0:
        sep = -sep
    idxp = n - 1 + sep
    idxm = n - 1 - sep

    best_d = 0.0
    best_val = float(current_max)
    for d in cands:
        np.multiply(A, d, out=tmp)
        tmp += c
        d2 = d * d
        tmp[n - 1] -= 2.0 * d2
        if sep != 0:
            tmp[idxp] += d2
            tmp[idxm] += d2
        val = float(tmp.max())
        if val < best_val - 1.0e-12:
            best_val = val
            best_d = d
    return best_d, best_val


def hill_search(h_in, n, deadline, rng, base0=0.02, K0=18, detail=1, max_pairs=18000):
    h = np.asarray(h_in, dtype=float).copy()
    c = make_c_unified(h, n)
    current_max = float(c.max())
    om = 1.0 - h
    hrev = h[::-1]
    L = 2 * n - 1
    A = np.zeros(L, dtype=float)
    tmp = np.zeros(L, dtype=float)

    best_h = h.copy()
    best_max = current_max
    base = float(base0)
    no_improve = 0
    passes = 0

    if K0 >= L:
        K0 = max(1, L // 2)

    while time.time() < float(deadline):
        if current_max > best_max + 0.1:
            h = best_h.copy()
            c = make_c_unified(h, n)
            current_max = float(c.max())
            om = 1.0 - h
            hrev = h[::-1]
            base = float(base0)
            no_improve = 0

        top = np.argpartition(c, K0)[-K0:]
        top = top[np.argsort(c[top])[::-1]]

        improved = False
        pair_count = 0

        for p in top:
            if time.time() > float(deadline) - 1.0:
                break
            if pair_count >= max_pairs:
                break
            lag = int(p) - (n - 1)

            if lag == 0:
                a = int(np.argmin(h))
                b = int(np.argmax(h))
                if h[a] < h[b] - 1.0e-12:
                    bd, nv = line_search_pair(h, om, hrev, c, A, tmp, a, b, n, base, current_max, detail)
                    pair_count += 1
                    if bd > 1.0e-12 and nv < current_max - 1.0e-12:
                        apply_move_aux(h, om, hrev, c, a, b, bd, n)
                        current_max = float(nv)
                        improved = True
                        if current_max < best_max - 1.0e-10:
                            best_max = current_max
                            best_h = h.copy()
                for _ in range(20):
                    if pair_count >= max_pairs or time.time() > float(deadline) - 1.0:
                        break
                    a = int(rng.randint(n))
                    b = int(rng.randint(n))
                    if a == b:
                        continue
                    if h[a] < h[b] - 1.0e-12:
                        ai = a
                        bj = b
                    elif h[b] < h[a] - 1.0e-12:
                        ai = b
                        bj = a
                    else:
                        continue
                    bd, nv = line_search_pair(h, om, hrev, c, A, tmp, ai, bj, n, base, current_max, detail)
                    pair_count += 1
                    if bd > 1.0e-12 and nv < current_max - 1.0e-12:
                        apply_move_aux(h, om, hrev, c, ai, bj, bd, n)
                        current_max = float(nv)
                        improved = True
                        if current_max < best_max - 1.0e-10:
                            best_max = current_max
                            best_h = h.copy()
                continue

            m = abs(lag)
            num = n - m
            if num <= 1:
                continue
            stride = (num * K0 + max_pairs - 1) // max_pairs
            if stride < 1:
                stride = 1

            for i in range(0, num, stride):
                if pair_count >= max_pairs or time.time() > float(deadline) - 1.0:
                    break
                if lag > 0:
                    a = i
                    b = i + m
                else:
                    a = i + m
                    b = i
                if h[a] <= 1.0e-12 or h[b] >= 1.0 - 1.0e-12:
                    continue
                bd, nv = line_search_pair(h, om, hrev, c, A, tmp, a, b, n, base, current_max, detail)
                pair_count += 1
                if bd > 1.0e-12 and nv < current_max - 1.0e-12:
                    apply_move_aux(h, om, hrev, c, a, b, bd, n)
                    current_max = float(nv)
                    improved = True
                    if current_max < best_max - 1.0e-10:
                        best_max = current_max
                        best_h = h.copy()

        passes += 1
        if improved:
            no_improve = 0
            if passes % 20 == 0:
                c_new = make_c_unified(h, n)
                c[:] = c_new
                current_max = float(c.max())
                om[:] = 1.0 - h
                if current_max < best_max - 1.0e-11:
                    best_max = current_max
                    best_h = h.copy()
                if current_max > best_max + 1.0e-6:
                    h = best_h.copy()
                    c = make_c_unified(h, n)
                    current_max = float(c.max())
                    om = 1.0 - h
                    hrev = h[::-1]
        else:
            no_improve += 1
            base *= 0.85
            if base < 1.0e-4:
                if detail < 2:
                    detail = 2
                    base = 1.0e-4
            if no_improve >= 3 and K0 < 48:
                K0 = min(48, K0 * 2)
            if no_improve >= 6:
                if time.time() < float(deadline) - 25.0:
                    sigma = 0.03
                    if no_improve > 9:
                        sigma = 0.08
                    h = best_h.copy()
                    h = h + rng.normal(0.0, sigma, n)
                    h = project_feasible(h, n)
                    c = make_c_unified(h, n)
                    om = 1.0 - h
                    hrev = h[::-1]
                    current_max = float(c.max())
                    if current_max < best_max:
                        best_max = current_max
                        best_h = h.copy()
                    if current_max > best_max + 0.005:
                        h = best_h.copy()
                        c = make_c_unified(h, n)
                        om = 1.0 - h
                        hrev = h[::-1]
                        current_max = best_max
                    base = max(0.01, base)
                    detail = 1
                    no_improve = 0
                else:
                    break

    if current_max < best_max - 1.0e-13:
        best_max = current_max
        best_h = h.copy()

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_exact(best_h, n)


def simulated_annealing_exact(h_in, n, deadline, rng, t_init=0.003, t_min=1.0e-6,
                              cool=0.9998, K=12, base_step=0.02):
    h = np.asarray(h_in, dtype=float).copy()
    c = make_c_unified(h, n)
    current_max = float(c.max())
    best_h = h.copy()
    best_max = current_max
    om = 1.0 - h
    hrev = h[::-1]
    dx = 2.0 / float(n)
    T = float(t_init)
    if K >= 2 * n - 1:
        K = max(1, (2 * n - 1) // 2)
    top = np.argpartition(c, K)[-K:]
    move_count = 0

    while time.time() < float(deadline):
        if T < float(t_min):
            break

        if (move_count % 2000) == 0:
            top = np.argpartition(c, K)[-K:]
            if time.time() > float(deadline) - 1.0:
                break
            c_new = make_c_unified(h, n)
            c[:] = c_new
            current_max = float(c.max())
            om[:] = 1.0 - h
            if current_max < best_max - 1.0e-11:
                best_max = current_max
                best_h = h.copy()
            if current_max > best_max + 0.2:
                h = best_h.copy()
                c = make_c_unified(h, n)
                current_max = float(c.max())
                om = 1.0 - h
                hrev = h[::-1]
                top = np.argpartition(c, K)[-K:]
                T = max(T, 0.4 * float(t_init))

        r = float(rng.random())
        ai = -1
        aj = -1

        if r < 0.72:
            p = int(top[int(rng.randint(K))])
            lag = p - (n - 1)
            m = abs(lag)
            if m == 0:
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if h[a] < h[b] - 1.0e-12:
                    ai = a
                    aj = b
                elif h[b] < h[a] - 1.0e-12:
                    ai = b
                    aj = a
            else:
                num = n - m
                i = int(rng.randint(num))
                if lag > 0:
                    ai = i
                    aj = i + m
                else:
                    ai = i + m
                    aj = i
        elif r < 0.87:
            a = int(rng.randint(n))
            b = int(rng.randint(n))
            if a == b:
                T *= cool
                move_count += 1
                continue
            if float(rng.random()) < 0.5:
                ai = a
                aj = b
            else:
                ai = b
                aj = a
        else:
            a = int(np.argmin(h))
            b = int(np.argmax(h))
            if h[a] < h[b] - 1.0e-12:
                ai = a
                aj = b
            elif h[b] < h[a] - 1.0e-12:
                ai = b
                aj = a

        if ai < 0 or aj < 0 or ai == aj:
            T *= cool
            move_count += 1
            continue

        maxd = min(h[ai], 1.0 - h[aj])
        if maxd <= 1.0e-10:
            T *= cool
            move_count += 1
            continue
        maxd *= (1.0 - 1.0e-12)

        if float(rng.random()) < 0.22:
            delta = maxd
        else:
            lo = max(1.0e-4, 0.08 * float(base_step))
            hi = min(float(base_step), maxd)
            if hi <= lo:
                delta = maxd
            else:
                delta = float(rng.uniform(lo, hi))

        apply_move_aux(h, om, hrev, c, ai, aj, delta, n)
        new_max = float(c.max())
        de = dx * (new_max - current_max)
        accept = False
        if de <= 1.0e-15:
            accept = True
        elif de < T:
            ratio = -de / T
            if ratio < 30.0 and float(rng.random()) < float(np.exp(-ratio)):
                accept = True

        if accept:
            current_max = new_max
            if new_max < best_max - 1.0e-12:
                best_max = new_max
                best_h = h.copy()
        else:
            apply_move_aux(h, om, hrev, c, ai, aj, -delta, n)

        T *= cool
        move_count += 1

    if current_max < best_max - 1.0e-13:
        best_max = current_max
        best_h = h.copy()

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_exact(best_h, n)


def binaryize_balanced(h, n):
    target = n // 2
    b = (h > 0.5).astype(float)
    ones = int(b.sum())
    if ones > target:
        one_idx = np.where(b > 0.5)[0]
        order = np.argsort(h[one_idx])
        excess = ones - target
        for idx in order[:excess]:
            b[one_idx[int(idx)]] = 0.0
    elif ones < target:
        zero_idx = np.where(b < 0.5)[0]
        order = np.argsort(-h[zero_idx])
        missing = target - ones
        for idx in order[:missing]:
            b[zero_idx[int(idx)]] = 1.0
    return b


def make_fallback_h(seed):
    rng = np.random.RandomState(seed)
    n = 500
    h = np.zeros(n, dtype=float)
    pick = np.argsort(rng.rand(n))[:n // 2]
    h[pick] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + max(10.0, total) - 3.0
    rng = np.random.RandomState(seed)

    h_parent = None
    try:
        h_parent = np.asarray(initial_h_values, dtype=float)
        if h_parent.ndim != 1 or len(h_parent) < 10:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        n0 = int(len(h_parent))
        h0 = np.asarray(h_parent, dtype=float).copy()
    else:
        n0 = 500
        h0 = make_fallback_h(seed)

    if n0 % 2 != 0:
        n0 = n0 + 1
        h0 = resample_h(h0, len(h0), n0)
    else:
        h0 = project_feasible(h0, n0)

    if n0 > 1500:
        ndown = 1000
        h0 = resample_h(h0, n0, ndown)
        n0 = ndown

    best_h = h0.copy()
    best_n = n0
    best_c5 = compute_c5_exact(best_h, best_n)

    # Phase 1: strong exact hill search at the original / parent resolution.
    Kp, det, mp, bs = search_params(best_n)
    chunk = min(330.0, max(30.0, 0.30 * total))
    d1 = t0 + chunk
    reserve1 = max(180.0, 0.25 * total)
    if d1 > deadline - reserve1:
        d1 = deadline - reserve1
    if d1 > time.time() + 12.0:
        h_try, c_try = hill_search(best_h, best_n, d1, rng, bs, Kp, det, mp)
        if c_try < best_c5 - 1e-14:
            best_h = h_try
            best_n = len(h_try)
            best_c5 = c_try

    # Phase 2: smooth max projection polish.
    chunk = min(95.0, max(20.0, 0.08 * total))
    d2 = time.time() + chunk
    reserve2 = max(150.0, 0.22 * total)
    if d2 > deadline - reserve2:
        d2 = deadline - reserve2
    if d2 > time.time() + 15.0:
        h_try, c_try = smooth_polish(best_h, best_n, d2)
        if c_try < best_c5 - 1e-14:
            best_h = h_try
            best_n = len(h_try)
            best_c5 = c_try

    # Phase 3: another exact hill-search pass after smoothing.
    Kp, det, mp, bs = search_params(best_n)
    chunk = min(300.0, max(40.0, 0.26 * total))
    d3 = time.time() + chunk
    reserve3 = max(130.0, 0.20 * total)
    if d3 > deadline - reserve3:
        d3 = deadline - reserve3
    if d3 > time.time() + 15.0:
        h_try, c_try = hill_search(best_h, best_n, d3, rng, bs, Kp, det, mp)
        if c_try < best_c5 - 1e-14:
            best_h = h_try
            best_n = len(h_try)
            best_c5 = c_try

    # Phase 4: short exact simulated-annealing search for basin escape.
    rem = deadline - time.time()
    if rem > 120.0 and best_n <= 1500:
        chunk = min(120.0, max(30.0, 0.10 * total))
        d4 = time.time() + chunk
        reserve4 = max(70.0, 0.12 * total)
        if d4 > deadline - reserve4:
            d4 = deadline - reserve4
        if d4 > time.time() + 15.0:
            h_try, c_try = simulated_annealing_exact(best_h, best_n, d4, rng)
            if c_try < best_c5 - 1e-14:
                best_h = h_try
                best_n = len(h_try)
                best_c5 = c_try

    # Phase 5: higher-resolution continuation if useful and time remains.
    rem = deadline - time.time()
    if rem > 260.0 and best_n <= 700:
        n2 = 2 * best_n
        if n2 <= 2000 and n2 % 2 == 0:
            h2 = resample_h(best_h, best_n, n2)
            c2 = compute_c5_exact(h2, n2)
            h2b = np.repeat(best_h, 2).astype(float)
            h2b = project_feasible(h2b, n2)
            c2b = compute_c5_exact(h2b, n2)
            if c2b < c2:
                h2 = h2b
                c2 = c2b

            Kp, det, mp, bs = search_params(n2)
            chunk = min(280.0, max(50.0, 0.22 * total))
            d5 = time.time() + chunk
            reserve5 = max(80.0, 0.10 * total)
            if d5 > deadline - reserve5:
                d5 = deadline - reserve5
            if d5 > time.time() + 20.0:
                h_try, c_try = hill_search(h2, n2, d5, rng, bs, Kp, det, mp)
                if c_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_n = n2
                    best_c5 = c_try

                # If higher resolution improved the bound, give it an extra polish pass.
                if best_n == n2:
                    rem = deadline - time.time()
                    if rem > 120.0:
                        chunk = min(160.0, max(30.0, 0.12 * total))
                        d6 = time.time() + chunk
                        reserve6 = max(40.0, 0.05 * total)
                        if d6 > deadline - reserve6:
                            d6 = deadline - reserve6
                        if d6 > time.time() + 15.0:
                            h_try, c_try = hill_search(best_h, best_n, d6, rng, bs, Kp, det, mp)
                            if c_try < best_c5 - 1e-14:
                                best_h = h_try
                                best_n = n2
                                best_c5 = c_try
                    rem = deadline - time.time()
                    if rem > 70.0 and best_n <= 800:
                        h_try, c_try = smooth_polish(best_h, best_n, deadline - 10.0)
                        if c_try < best_c5 - 1e-14:
                            best_h = h_try
                            best_n = len(h_try)
                            best_c5 = c_try

    # Phase 6: balanced binary relaxation followed by exact hill search.
    rem = deadline - time.time()
    if rem > 130.0:
        hb = binaryize_balanced(best_h, best_n)
        cb = compute_c5_exact(hb, best_n)
        if cb < best_c5 + 0.005:
            Kp, det, mp, bs = search_params(best_n)
            chunk = min(110.0, max(30.0, 0.08 * total))
            d7 = time.time() + chunk
            reserve7 = max(50.0, 0.06 * total)
            if d7 > deadline - reserve7:
                d7 = deadline - reserve7
            if d7 > time.time() + 15.0:
                h_try, c_try = hill_search(hb, best_n, d7, rng, bs, max(16, Kp), 1, max(12000, mp))
                if c_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_n = len(h_try)
                    best_c5 = c_try

    # Final phase: use remaining time for a final exact squeeze on the best representation.
    rem = deadline - time.time()
    if rem > 35.0:
        Kp, det, mp, bs = search_params(best_n)
        dfinal = deadline - 2.5
        if dfinal > time.time() + 10.0:
            h_try, c_try = hill_search(best_h, best_n, dfinal, rng, max(0.005, bs / 2.0), max(20, Kp), det, max(12000, mp))
            if c_try < best_c5 - 1e-14:
                best_h = h_try
                best_n = len(h_try)
                best_c5 = c_try

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5_exact(best_h, best_n)
    return best_h, float(best_c5), int(best_n)
```
