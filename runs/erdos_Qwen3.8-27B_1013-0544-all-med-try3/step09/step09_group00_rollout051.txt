from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_beatty(n, slope):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    k = 1
    while len(positions) < n_ones and k < 20 * n:
        pos = int(k * slope) % n
        positions.add(pos)
        k += 1
    for pos in positions:
        h[pos] = 1.0
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 350:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:

def swap_delta(h, n, i, j):
    """Compute delta in c5_vec (before *dx) for swap h[i]:1->0, h[j]:0->1"""
    m = np.arange(2 * n - 1)
    a = i + m - (n - 1)
    b = j + m - (n - 1)
    c_arr = i - m + (n - 1)
    d = j - m + (n - 1)
    delta = np.zeros(2 * n - 1)
    mask_a = (a >= 0) & (a < n)
    delta[mask_a] -= (1.0 - h[a[mask_a]])
    mask_c = (c_arr >= 0) & (c_arr < n)
    delta[mask_c] += h[c_arr[mask_c]]
    mask_b = (b >= 0) & (b < n)
    delta[mask_b] += (1.0 - h[b[mask_b]])
    mask_d = (d >= 0) & (d < n)
    delta[mask_d] -= h[d[mask_d]]
    return delta

            best_c = c7

    # Phase 8: Final L-BFGS polish (30s)
    if time.time() < deadline - 10:
        p8_dead = min(time.time() + 20, deadline - 5)
        h8, c8 = run_lbfgs(best_h, best_n, p8_dead,
                          alphas=[100000, 500000], mu=0.001)
        if c8 < best_c - 1e-16:
            best_c = c8
            best_h = h8

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
