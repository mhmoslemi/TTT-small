    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    return float(val)

def lse_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-15:
            continue
        k = j - (n - 1)
        wm = w[j] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def local_pair_search(h, n, deadline, radius=20):

            break

            improved = False

                    return h, best_c

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):

                        continue

                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                            improved = True

                            h[j] -= step

def topk_obj(h, n, top_idx, alpha):
    c = compute_c5_vec(h, n)
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh - mx))) / alpha
    return float(val)
