import numpy as np
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, active, alpha):
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, h, active, alpha, n):
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_fn(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, active, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_fn(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, x, active, alpha, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, np.arange(len(c)), alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, x, np.arange(len(c)), alpha, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sim_anneal(h, n, deadline, t0_temp=None, cooldown=0.97, max_iter=50000):

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_quadratic_residue_start(n, p):

    n_ones = n // 2
    positions = set()
    for k in range(p):
        pos = (k * k) % p
        if pos < n:

    if h.sum() < n_ones:
        zeros = np.where(h == 0)[0]
        for z in zeros[:n_ones - int(h.sum())]:

            step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_rounds=15)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try different resolutions - 100s
    p5_end = min(t0 + 850, deadline - 80)
    if time.time() < p5_end:
        for n_try in [550, 450, 600]:
            if time.time() > p5_end - 20:
                break
            # Interpolate or create new start
            if n_try == n:
                continue
            h_try = np.zeros(n_try)
            h_try[:n_try//2] = 1.0
            h_try = project_feasible(h_try, n_try)
            # Also try Beatty
            h_beatty = make_beatty_start(n_try, (1.0 + math.sqrt(5.0)) / 4.0)
            h_try = min(h_try, h_beatty, key=lambda x: compute_c5(x, n_try))
            c_try0 = compute_c5(h_try, n_try)
            
            # Quick L-BFGS
            h_opt, c_opt = run_lbfgs(h_try, n_try, min(time.time() + 60, p5_end - 10),
                alphas=[100, 1000, 10000, 100000],
                mu=0.01, top_k=8)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: Final active set + SLSQP polish
    p6_end = min(t0 + 950, deadline - 30)
    if time.time() < p6_end:
        h6, c6 = active_set_phase(best_h, len(best_h), min(time.time() + 100, p6_end),
            base_config=[(6, 5000000, 3), (12, 20000000, 3), (20, 50000000, 2)])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 10:
        for alpha in [5000000, 20000000]:
            if time.time() > deadline - 8:
                break
            h7, c7 = run_slqp(best_h, len(best_h), deadline - 3, alpha=alpha, maxiter=300)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Final fine coordinate descent
    if time.time() < deadline - 3:
        h8, c8 = coord_descent(best_h, len(best_h), deadline - 1,
            step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
            max_rounds=10)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    # Verify constraints
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c, final_n
