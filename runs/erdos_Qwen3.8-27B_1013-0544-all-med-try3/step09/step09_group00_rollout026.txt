            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(200, int((deadline - time.time()) * 0.4))
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def exhaustive_local_search(h, n, deadline, R=80, max_passes=10):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline - 10:
            break
        improved = True
        passes = 0
        while improved and passes < max_passes and time.time() < deadline - 5:
            improved = False
            passes += 1
            order = np.random.permutation(n)
            for i in order:
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - R)
                hi = min(n, i + R + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    delta = min(d, 1.0 - best_h[i], best_h[j])
                    if delta > 1e-10:
                        old_i, old_j = best_h[i], best_h[j]
                        best_h[i] += delta
                        best_h[j] -= delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
                    delta = min(d, best_h[i], 1.0 - best_h[j])
                    if delta > 1e-10:
                        old_i, old_j = best_h[i], best_h[j]
                        best_h[i] -= delta
                        best_h[j] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
    return best_h, best_c

def binary_swap_search(h, n, deadline, R=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    improved = True
    passes = 0
    while improved and passes < 5 and time.time() < deadline - 10:
        improved = False
        passes += 1
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        np.random.shuffle(ones)
        for i in ones:
            if time.time() > deadline - 10:
                break
            lo = max(0, i - R)
            hi = min(n, i + R + 1)
            for j in range(lo, hi):
                if best_h[j] >= 0.5:
                    continue
                old_i, old_j = best_h[i], best_h[j]
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    best_h = project_feasible(h.copy(), n)
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
    return best_h, best_c

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            pos_mask = (grad > 1e-10) & (best_h > 1e-6)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-6)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                gi = grad[i]
                for j in neg_idx[:40]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(5, deadline - time.time())
        maxit = min(maxiter, max(100, int(rem * 5)))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_c = compute_c5(h0, n0)
    best_h = h0.copy()
    best_n = n0

    # Phase 1: Diverse initialization (30s budget)
    p1_end = t0 + 30
    candidates = [h0.copy()]
    alphas_beatty = [
        (1.0 + np.sqrt(5.0)) / 4.0,
        np.sqrt(2.0) / 2.0,
        np.sqrt(3.0) / 2.0,
        (np.sqrt(5.0) - 1.0) / 4.0,
        1.0 / 3.0,
        2.0 / 5.0,
        (1.0 + np.sqrt(2.0)) / 4.0,
        np.sqrt(5.0) / 4.0,
    ]
    for a in alphas_beatty:
        if time.time() > p1_end:
            break
        candidates.append(make_beatty(n0, a))
    candidates.append(make_thue_morse(n0))
    for _ in range(8):
        if time.time() > p1_end:
            break
        candidates.append(make_random_binary(n0, rng))

    for h_c in candidates:
        if time.time() > p1_end:
            break
        c_c = compute_c5(h_c, n0)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c.copy()

    # Phase 2: L-BFGS warmup (120s budget)
    p2_end = min(t0 + 150, deadline - 380)
    h2, c2 = phase_lbfgs(best_h, best_n, p2_end,
        alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01, maxit_list=[500, 800, 1000, 1200, 1500, 2000, 2500, 3000, 3500])
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Active set optimization (150s budget)
    p3_end = min(time.time() + 150, deadline - 250)
    h3, c3 = phase_active_set(best_h, best_n, p3_end, top_k=10, rounds=25)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Gradient-guided search (100s budget)
    p4_end = min(time.time() + 100, deadline - 180)
    h4, c4 = grad_guided_search(best_h, best_n, p4_end,
        step_sizes=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
        max_steps=150000)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Exhaustive local search (150s budget)
    p5_end = min(time.time() + 150, deadline - 120)
    h5, c5 = exhaustive_local_search(best_h, best_n, p5_end, R=60, max_passes=5)
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Active set again on improved solution (80s)
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 80, deadline - 80)
        h6, c6 = phase_active_set(best_h, best_n, p6_end, top_k=6, rounds=15)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try higher resolution n=1000 (200s budget)
    if time.time() < deadline - 60 and best_n < 800:
        n_fine = 1000
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        p7_end = min(time.time() + 150, deadline - 40)
        if c_fine > best_c - 0.01:
            h7, c7 = phase_lbfgs(h_fine, n_fine, p7_end,
                alphas=[50, 500, 5000, 50000, 500000],
                mu=0.01, maxit_list=[500, 800, 1000, 1500, 2000])
            if time.time() < deadline - 20 and c7 < c_fine + 0.005:
                h7b, c7b = phase_active_set(h7, n_fine, deadline - 15, top_k=8, rounds=10)
                if c7b < c7:
                    h7, c7 = h7b, c7b
            if c7 < best_c:
                best_c = c7
                best_h = h7
                best_n = n_fine

    # Phase 8: Fine exhaustive search (60s)
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 50, deadline - 10)
        h8, c8 = exhaustive_local_search(best_h, best_n, p8_end, R=40, max_passes=3)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: SLSQP polish (15s)
    if time.time() < deadline - 5:
        for alpha in [2000000, 10000000]:
            if time.time() > deadline - 3:
                break
            h9, c9 = run_slqp(best_h, best_n, deadline - 2, alpha=alpha, maxiter=600)
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Final: very fine local adjustments
    if time.time() < deadline - 2:
        h10, c10 = exhaustive_local_search(best_h, best_n, deadline - 1, R=20, max_passes=2)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
