import numpy as np
import time
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            h = np.clip(h, 0.0, 1.0)
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 15,
                                    'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def grad_single_shift(h, n, shift_idx):
    k = shift_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[0:nv] += (1.0 - h[k:n])
            g[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:n] += (1.0 - h[0:nv])
            g[0:nv] += (-h[kk:n])
    return dx * g

def line_search_worst(h, n, t0=0.08, max_iter=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    k0 = int(np.argmax(c_vec))
    g = grad_single_shift(h, n, k0)
    p = -g
    t = t0
    for _ in range(max_iter):
        if time.time() > deadline_global - 5:
            break
        h_trial = h + t * p
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-14:
            best_h = h_trial.copy()
            best_c = c_trial
            h = h_trial
            t = min(t * 1.25, 0.25)
            k0 = int(np.argmax(compute_c5_vec(h, n)))
            g = grad_single_shift(h, n, k0)
            p = -g
        else:
            t = t * 0.75
            if t < 1e-10:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global deadline_global
    t0 = time.time()
    deadline_global = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    p1_end = t0 + 40
    if time.time() < p1_end:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[500, 2000, 10000, 50000, 200000],
            mu=0.008, maxiters=[200, 400, 600, 800, 1000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    n1 = 1000
    if time.time() < deadline_global - 500:
        h_up = upsample(best_h, best_n, n1)
        c_up = compute_c5(h_up, n1)
        if c_up < best_c + 5e-4:
            alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
            maxiters = [300, 500, 800, 1200, 2000, 3000, 5000, 8000]
            deadline_a = deadline_global - 250
            h_a, c_a = run_lbfgs(h_up, n1, deadline_a, alphas=alphas, mu=0.004, maxiters=maxiters)
            if c_a < best_c:
                best_c = c_a
                best_h = h_a
                best_n = n1
            
            if time.time() < deadline_global - 400:
                h_pert = h_up.copy()
                interior = (h_pert > 0.02) & (h_pert < 0.98)
                if interior.sum() > 10:
                    noise = rng.normal(0, 0.03, interior.sum())
                    h_pert[interior] += noise
                    h_pert = project_feasible(h_pert, n1)
                    deadline_b = deadline_global - 200
                    h_b, c_b = run_lbfgs(h_pert, n1, deadline_b, alphas=alphas[:6], mu=0.004, maxiters=maxiters[:6])
                    if c_b < best_c:
                        best_c = c_b
                        best_h = h_b
                        best_n = n1
            
            if time.time() < deadline_global - 100:
                deadline_ls = deadline_global - 50
                h_ls, c_ls = line_search_worst(best_h, best_n, t0=0.06, max_iter=80)
                if c_ls < best_c:
                    best_c = c_ls
                    best_h = h_ls
                if time.time() < deadline_ls - 20:
                    h_ls2, c_ls2 = line_search_worst(best_h, best_n, t0=0.03, max_iter=60)
                    if c_ls2 < best_c:
                        best_c = c_ls2
                        best_h = h_ls2
    
    n2 = 2000
    if time.time() < deadline_global - 350 and best_c < 0.38090:
        h_up2 = upsample(best_h, best_n, n2)
        c_up2 = compute_c5(h_up2, n2)
        if c_up2 < best_c + 3e-4:
            alphas2 = [500, 2000, 10000, 50000, 200000, 1000000]
            maxiters2 = [400, 800, 1500, 2500, 4000, 6000]
            deadline_c = deadline_global - 150
            h_c, c_c = run_lbfgs(h_up2, n2, deadline_c, alphas=alphas2, mu=0.003, maxiters=maxiters2)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c
                best_n = n2
            if time.time() < deadline_global - 50:
                deadline_ls2 = deadline_global - 20
                h_ls3, c_ls3 = line_search_worst(best_h, best_n, t0=0.04, max_iter=50)
                if c_ls3 < best_c:
                    best_c = c_ls3
                    best_h = h_ls3
    
    if time.time() < deadline_global - 20:
        final_alphas = [100000, 500000, 2000000]
        final_maxit = [500, 1000, 2000]
        h_fin, c_fin = run_lbfgs(best_h, best_n, deadline_global - 10,
                                  alphas=final_alphas, mu=0.002, maxiters=final_maxit)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    return best_h, best_c, final_n
