import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def de_search(h_init, n, deadline, pop_size=60, F=0.8, CR=0.8, max_gens=None):
    rng = np.random.default_rng(42)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Initialize population
    pop = []
    c_vals = []
    # First member is the best we have
    pop.append(best_h.copy())
    c_vals.append(best_c)
    
    # Add diverse initializations
    strategies = [
        lambda: best_h + 0.1 * rng.standard_normal(n),
        lambda: best_h + 0.2 * rng.standard_normal(n),
        lambda: best_h + 0.05 * rng.standard_normal(n),
        lambda: np.maximum(0, np.minimum(1, best_h + 0.3 * np.sin(np.linspace(0, 4*np.pi, n)) * rng.random())),
        lambda: np.maximum(0, np.minimum(1, best_h + 0.2 * rng.random(n))),
    ]
    
    while len(pop) < pop_size:
        tried = False
        for strat in strategies:
            h_try = project_feasible(strat(), n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c + 0.05:  # reasonably close
                pop.append(h_try)
                c_vals.append(c_try)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                break
            tried = True
        if tried or len(pop) >= pop_size:
            if len(pop) < pop_size:
                h_try = project_feasible(best_h + 0.15 * rng.standard_normal(n), n)
                c_try = compute_c5(h_try, n)
                pop.append(h_try)
                c_vals.append(c_try)
    
    # DE loop
    gen = 0
    if max_gens is None:
        max_gens = 500
    
    while gen < max_gens and time.time() < deadline:
        gen += 1
        # Adaptive F and CR
        F_adapt = F * (1.0 - 0.3 * gen / max_gens)
        CR_adapt = min(0.95, CR + 0.1 * gen / max_gens)
        
        kept = 0
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            # Select three different random individuals
            r1, r2, r3 = rng.choice(pop_size, size=3, replace=False)
            
            # Mutation
            v = pop[r1] + F_adapt * (pop[r2] - pop[r3])
            v = project_feasible(v, n)
            
            # Crossover
            trial = pop[i].copy()
            j_rand = rng.integers(0, n)
            for j in range(n):
                if rng.random() < CR_adapt or j == j_rand:
                    trial[j] = v[j]
            trial = project_feasible(trial, n)
            
            # Evaluate
            c_trial = compute_c5(trial, n)
            
            # Selection
            if c_trial < c_vals[i]:
                pop[i] = trial
                c_vals[i] = c_trial
                kept += 1
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = trial.copy()
        
        # Adapt population: replace worst with perturbation of best
        if gen % 10 == 0:
            worst_idx = np.argmax(c_vals)
            pop[worst_idx] = project_feasible(best_h + 0.05 * rng.standard_normal(n), n)
            c_vals[worst_idx] = compute_c5(pop[worst_idx], n)
    
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 5,
                                    'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2)  # periodic interp
    # Actually, the domain is [0,2] and we're not wrapping, so regular interp
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def coordinate_descent(h, n, deadline, n_sweeps=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 5,
                                    'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def de_vectorized(h_init, n, deadline, pop_size=40, F=0.7, CR=0.8):
    """Vectorized differential evolution"""
    rng = np.random.default_rng(12345)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    c_vals = np.zeros(pop_size)
    pop[0] = best_h
    c_vals[0] = best_c
    
    for i in range(1, pop_size):
        # Diverse initializations
        mode = i % 4
        if mode == 0:
            h_try = best_h + 0.1 * rng.standard_normal(n)
        elif mode == 1:
            h_try = best_h + 0.3 * rng.standard_normal(n)
        elif mode == 2:
            h_try = best_h + 0.15 * np.sin(np.linspace(0, (i+1)*np.pi, n))
        else:
            h_try = best_h + 0.2 * rng.random(n) - 0.1
        pop[i] = project_feasible(h_try, n)
        c_vals[i] = compute_c5(pop[i], n)
        if c_vals[i] < best_c:
            best_c = c_vals[i]
            best_h = pop[i].copy()
    
    # DE loop
    no_improve = 0
    while time.time() < deadline:
        gen_F = F * max(0.3, 1.0 - no_improve / 50.0)
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            # Pick 3 distinct random indices
            mask3 = rng.choice(pop_size, size=3, replace=False)
            r1, r2, r3 = mask3
            
            # Mutation
            v = pop[r1] + gen_F * (pop[r2] - pop[r3])
            v = project_feasible(v, n)
            
            # Crossover (vectorized)
            cr_mask = rng.random(n) < CR
            j_rand = rng.integers(0, n)
            cr_mask[j_rand] = True
            trial = np.where(cr_mask, v, pop[i])
            trial = project_feasible(trial, n)
            
            # Evaluate
            c_trial = compute_c5(trial, n)
            
            # Selection
            if c_trial < c_vals[i]:
                pop[i] = trial
                c_vals[i] = c_trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = trial.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        # Replace worst with perturbed best every 20 iterations
        if no_improve > 20:
            worst_idx = np.argmax(c_vals)
            pop[worst_idx] = project_feasible(best_h + 0.08 * rng.standard_normal(n), n)
            c_vals[worst_idx] = compute_c5(pop[worst_idx], n)
            no_improve = 0
        if no_improve > 80:
            # More aggressive replacement
            q = int(pop_size * 0.3)
            worst_q = np.argsort(-c_vals)[:q]
            for wi in worst_q:
                pop[wi] = project_feasible(best_h + 0.15 * rng.standard_normal(n), n)
                c_vals[wi] = compute_c5(pop[wi], n)
            no_improve = 0
    
    return best_h, best_c

def multi_start_polish(h, n, deadline, n_starts=5, perturb_scale=0.1):
    """Try multiple perturbed starts and polish each"""
    rng = np.random.default_rng(777)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    start = time.time()
    per_start = (deadline - start) / n_starts
    
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        # Generate perturbation
        mode = s % 3
        if mode == 0:
            h_try = h + perturb_scale * rng.standard_normal(n)
        elif mode == 1:
            # Low-frequency perturbation
            k = rng.integers(2, 10)
            h_try = h + perturb_scale * np.sin(np.linspace(0, k * 2 * np.pi, n) + rng.random() * 2 * np.pi)
        else:
            # Block perturbation
            h_try = h.copy()
            block_size = n // 10
            for _ in range(3):
                b = rng.integers(0, n - block_size)

        c_try = compute_c5(h_try, n)

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.003 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)

            k = idx - (n - 1)

                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx

        c_new = compute_c5(best_h, n)

        else:
