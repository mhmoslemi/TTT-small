import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    npts = len(c)
    center = n - 1
    for ii in range(npts):
        if w[ii] < 1e-15:
            continue
        k = ii - center
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
