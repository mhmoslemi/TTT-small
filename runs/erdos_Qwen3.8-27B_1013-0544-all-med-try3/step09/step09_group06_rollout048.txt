import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_max_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _lag_grad(h, n, lag, grad):
    m = abs(lag)
    if m >= n:
        return
    if lag == 0:
        grad[:] += 1.0 - 2.0 * h
        return
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n-m]

def grad_smooth_max(h, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_idx is None:
        top_idx = np.arange(2 * n - 1)
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for idx, wk in zip(top_idx, w):
        if wk < 1e-10:
            continue
        lag = idx - (n - 1)
        _lag_grad(h, n, lag, grad)
    return grad * dx

def grad_descent(h, n, deadline, alpha, lr, momentum):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    top_idx = None
    for it in range(50000):
        if time.time() > deadline:
            break
        if top_idx is None or it % 200 == 0:
            c_all = compute_all_lags(h_curr, n)
            top_idx = np.argsort(c_all)[-8:]
        g = grad_smooth_max(h_curr, n, alpha, top_idx)
        v = momentum * v + lr * g
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        if it % 3000 == 2999:
            lr *= 0.8
    return best_h, best_c

def thorough_local_search(h, n, deadline, rng, top_k=5):

def slsqp_polish(h, n, alpha, maxiter, top_m=None, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    c_all = compute_all_lags(h, n)
    if top_m is not None:
        top_idx = np.argsort(c_all)[-top_m:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    def obj(x):
        return _smooth_max_obj(x, n, alpha, top_idx)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_snip_and_restore(h, n, deadline, rng):
    """Try binary relaxation: snap to binary, do a few binary swaps, restore continuity"""
    h_work = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_work, n)
    
    # Snap to binary
    h_bin = np.zeros(n)
    idx = np.argsort(h_work)[::-1][:n//2]
    h_bin[idx] = 1.0
    c_bin = compute_c5(h_bin, n)
    
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Try a few binary swaps
    for _ in range(2000):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i], h_bin[j] = 0.0, 1.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
    
    # Restore: blend binary back with original
    for blend in [1.0, 0.9, 0.8, 0.7]:
        h_mixed = blend * best_h + (1 - blend) * h
        h_mixed = project_feasible(h_mixed, n)
        c_mixed = compute_c5(h_mixed, n)
        if c_mixed < best_c:
            best_c = c_mixed
            best_h = h_mixed.copy()
    
    return best_h, best_c
