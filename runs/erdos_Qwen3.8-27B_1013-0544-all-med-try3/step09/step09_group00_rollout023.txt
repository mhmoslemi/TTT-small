def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(30000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()

                break
            h += (diff /_tot) * room

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def nesterov_subgrad(h, n, deadline, alpha=100000, step_init=0.01, momentum=0.9, max_iter=200000):
    """Nesterov-accelerated projected subgradient on smooth max"""
    x = h.copy()
    x_prev = h.copy()
    best_x = h.copy()
    best_c = compute_c5(h, n)
    bounds_lo = np.zeros(n)
    bounds_hi = np.ones(n)
    target = n / 2.0
    momentum_val = momentum
    step = step_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov extrapolation
        y = x + momentum_val * (x - x_prev)
        y = np.clip(y, bounds_lo, bounds_hi)
        # Compute subgradient at y
        c = compute_c5_vec(y, n)
        w = smooth_max_weights(c, alpha)
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for mm in range(len(c)):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx_val
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - y[k:n])
                grad[k:n] += wm * (-y[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - y[0:nv])
                grad[0:nv] += wm * (-y[kk:n])
        # Gradient step
        x_new = y - step * grad
        # Project to [0,1]^n with sum constraint
        x_new = np.clip(x_new, 0.0, 1.0)
        diff = target - x_new.sum()
        if diff > 0:
            room = 1.0 - x_new
            tot = room.sum()
            if tot > 1e-14:
                x_new += (diff / tot) * room
        else:
            tot = x_new.sum()
            if tot > 1e-14:
                x_new += (diff / tot) * x_new
        x_new = np.clip(x_new, 0.0, 1.0)
        # Update
        x_prev = x
        x = x_new
        it_c = compute_c5(x, n)
        if it_c < best_c - 1e-15:
            best_c = it_c
            best_x = x.copy()
        # Decay step
        step *= 0.9999
        momentum_val = 0.5 * (1.0 + momentum_val)
    return best_x, best_c

def binary_2swap(h, n, deadline, max_iter=100000, top_k=30, anneal=True):
    """Binary 2-swap local search with gradient guidance and optional SA"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    cur_h = h.copy()
    cur_c = best_c
    ones = np.where(cur_h > 0.5)[0]
    zeros = np.where(cur_h < 0.5)[0]
    rng = np.random.RandomState(12345)
    temp = 0.002
    temp_min = 1e-7
    cool = 0.9995
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if no_improve > 10000 and not anneal:
            break
        # Compute active shifts
        c_vec = compute_c5_vec(cur_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        # Compute subgradient
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - cur_h[k:n]) * dx_val
                grad[k:n] += (-cur_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - cur_h[0:nv]) * dx_val
                grad[0:nv] += (-cur_h[kk:n]) * dx_val
        grad /= len(active)
        # Rank candidates
        pos_rank = np.argsort(-grad[ones])[:top_k]
        neg_rank = np.argsort(grad[zeros])[:top_k]
        pos_cand = ones[pos_rank]
        neg_cand = zeros[neg_rank]
        # Try swaps
        best_gain = -np.inf
        best_i = -1
        best_j = -1
        for i in pos_cand:
            for j in neg_cand:
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                c_new = compute_c5(cur_h, n)
                gain = cur_c - c_new
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
                cur_h[i] = 1.0
                cur_h[j] = 0.0
        if best_i < 0:
            break
        # Apply best swap
        cur_h[best_i] = 0.0
        cur_h[best_j] = 1.0
        c_new = compute_c5(cur_h, n)
        delta = c_new - cur_c
        accept = delta < -1e-14
        if not accept and anneal and temp > temp_min:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_c = c_new
            # Update sets
            oi = np.where(ones == best_i)[0]
            if len(oi) > 0:
                ones = np.delete(ones, oi[0])
            oz = np.where(zeros == best_j)[0]
            if len(oz) > 0:
                zeros = np.delete(zeros, oz[0])
            ones = np.append(ones, best_j)
            zeros = np.append(zeros, best_i)
            no_improve = 0
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # Revert
            cur_h[best_i] = 1.0
            cur_h[best_j] = 0.0
            no_improve += 1
        temp *= cool
    return best_h, best_c

def pair_search_fine(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try increase i, decrease j
                    if best_h[i] + d <= 1.0 + 1e-10 and best_h[j] - d >= -1e-10:
                        best_h[i] = min(1.0, best_h[i] + d)
                        best_h[j] = max(0.0, best_h[j] - d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = max(0.0, best_h[i] - d)
                            best_h[j] = min(1.0, best_h[j] + d)
                            best_h = project_feasible(best_h, n)
                    # Try decrease i, increase j
                    if best_h[i] - d >= -1e-10 and best_h[j] + d <= 1.0 + 1e-10:
                        best_h[i] = max(0.0, best_h[i] - d)
                        best_h[j] = min(1.0, best_h[j] + d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = min(1.0, best_h[i] + d)
                            best_h[j] = max(0.0, best_h[j] - d)
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = make_beatty(500, (1.0 + np.sqrt(5.0)) / 2.0 * 0.5)
        n_orig = 500
    
    n = n_orig
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Generate alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        make_beatty(n, golden / 2.0),
        make_beatty(n, np.sqrt(2.0) / 2.0),
        make_beatty(n, np.sqrt(3.0) / 3.0),
        make_random_binary(n, rng),
        make_random_binary(n, rng),
    ]
    for s in starts:
        c_s = compute_c5(s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = s.copy()
    
    # Phase 1: L-BFGS warmup (60s)
    p1_end = t0 + 60
    if time.time() < p1_end:
        bounds = [(0.0, 1.0)] * n
        for alpha in [50, 500, 5000, 50000, 200000]:
            if time.time() > p1_end - 5:
                break
            try:
                maxit = min(500, max(100, int((p1_end - time.time()) * 3)))
                res = minimize(obj_full, best_h, jac=grad_full,
                            args=(n, alpha, 0.01), method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxiter': maxit, 'ftol': 1e-22,
                                     'gtol': 1e-16, 'maxfun': maxit*8,
                                     'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: Binary 2-swap local search (250s)
    p2_end = min(t0 + 310, deadline - 350)
    if time.time() < p2_end:
        # First, round to binary
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c + 0.01:
            # Binary is competitive, do swap search on it
            h_sw, c_sw = binary_2swap(h_bin, n, p2_end, max_iter=80000, top_k=25, anneal=True)
            if c_sw < best_c:
                best_c = c_sw
                best_h = h_sw
                h_bin = h_sw.copy()
        # Also do swap search from current best
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, n)
        if time.time() < p2_end:
            h_sw2, c_sw2 = binary_2swap(h_bin2, n, p2_end, max_iter=40000, top_k=20, anneal=False)
            if c_sw2 < best_c:
                best_c = c_sw2
                best_h = h_sw2
    
    # Phase 3: Fractional fine-tuning via Nesterov (200s)
    p3_end = min(t0 + 510, deadline - 250)
    if time.time() < p3_end:
        h_nest, c_nest = nesterov_subgrad(best_h, n, p3_end, alpha=50000, step_init=0.005, momentum=0.85, max_iter=150000)
        if c_nest < best_c:
            best_c = c_nest
            best_h = h_nest
    
    # Phase 4: Multi-k active-set L-BFGS (150s)
    p4_end = min(t0 + 660, deadline - 180)
    if time.time() < p4_end:
        h_akt, c_akt = multi_k_active_lbfgs(best_h, n, p4_end)
        if c_akt < best_c:
            best_c = c_akt
            best_h = h_akt
    
    # Phase 5: Fractional fine-tuning (coordinate descent) (150s)
    p5_end = min(t0 + 810, deadline - 100)
    if time.time() < p5_end:
        h_ft, c_ft = fractional_fine_tune(best_h, n, p5_end, max_iter=300000)
        if c_ft < best_c:
            best_c = c_ft
            best_h = h_ft
    
    # Phase 6: Another Nesterov pass with different params (80s)
    p6_end = min(t0 + 890, deadline - 50)
    if time.time() < p6_end:
        h_nest2, c_nest2 = nesterov_subgrad(best_h, n, p6_end, alpha=200000, step_init=0.002, momentum=0.92, max_iter=80000)
        if c_nest2 < best_c:
            best_c = c_nest2
            best_h = h_nest2
    
    # Phase 7: Final active-set L-BFGS polish (40s)
    p7_end = min(t0 + 930, deadline - 15)
    if time.time() < p7_end:
        configs_final = [(4, 2000000, 2), (6, 10000000, 2), (10, 50000000, 1)]
        h_fin, c_fin = multi_k_active_lbfgs(best_h, n, p7_end, configs=configs_final)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    
    # Phase 8: Very fine pair search on remaining time
    p8_end = deadline - 2
    if time.time() < p8_end:
        h_pf, c_pf = pair_search_fine(best_h, n, p8_end, radius=30,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c_pf < best_c:
            best_c = c_pf
            best_h = h_pf
    
    # Final check and projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
