import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5_at_shift(h, n, k_idx):
    """Gradient of the k-th correlation value with respect to h."""
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def softmax_obj(x, n, alpha, active_indices, mu):
    c = compute_c5_vec(x, n)
    if len(active_indices) > 0:
        vals = c[active_indices]
        m = vals.max()
        sh = alpha * (vals - m)
        val = m + np.log(np.sum(np.exp(sh - max(0, sh.max())))) / alpha
    else:
        val = c.max()
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def softmax_grad(x, n, alpha, active_indices, mu):
    c = compute_c5_vec(x, n)
    if len(active_indices) > 0:
        vals = c[active_indices]
        m = vals.max()
        sh = alpha * (vals - m)
        smx = max(0.0, float(sh.max()))
        w = np.exp(sh - smx)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, idx in zip(w, active_indices):
            grad += wi * grad_c5_at_shift(x, n, idx)
    else:
        max_idx = np.argmax(c)
        grad = grad_c5_at_shift(x, n, max_idx)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient_step(h, n, c_vec, active, step_size):

    dx = 2.0 / n
    grad = np.zeros(n)

def upscale(h, n_old, n_new):
    """Upscale a solution from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def downscale(h, n_old, n_new):
    """Downscale by averaging blocks."""
    ratio = n_old // n_new
    h_new = h[:n_new * ratio].reshape(n_new, ratio).mean(axis=1)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    """Create a Beatty sequence binary pattern."""
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)
