from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()

    mx = sh.max()
    w = np.exp(sh - mx)

    dx = 2.0 / n

            # dc_k/dh[i] = (1-h[i+k])*dx (from j=i term, if i+k<n) 
            #             + (-h[i-k])*dx (from j=i-k term, if i-k>=0)
            #             (for the appropriate k ranges)
            
            # For the max active constraint, we want to decrease it.
            # Let's find the best value for h[i] by trying a few values.
            old_val = best_h[i]
            # Try a few candidate values
            candidates = np.linspace(0, 1, 21)
            best_local = old_val
            best_local_c = best_c
            
            # Actually this is too slow for n=1000. Let's use gradient.
            # Compute which correlations involve h[i] and their current values.
            c_vec = compute_c5_vec(best_h, n)
            # h[i] appears in c_k for various k
            # c_{k} = sum_j h[j](1-h[j+k])*dx
            # h[i] appears in c_k when:
            #   j = i (first factor): contributes (1-h[i+k])*dx to c_k, valid if i+k < n and k>=0, or i+k>=0 and k<=0... 
            #   Actually in the full correlation: c[mm] where mm = i + (i+k) ... 
            # Let me just compute the gradient for all active constraints at i.
            
            # Simpler: just try moving h[i] a small amount
            for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                best_h[i] = new_val
                # Adjust another variable to maintain sum
                # For now, just check if c decreases (sum constraint handled by projection later)
                c_new = compute_c5(best_h, n)
                if c_new < best_local_c - 1e-16:
                    best_local_c = c_new
                    best_local = new_val
                best_h[i] = old_val
            
            if best_local != old_val:
                best_h[i] = best_local
                best_h = project_feasible(best_h, n)
                best_c = compute_c5(best_h, n)
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c

def pair_descent(h, n, deadline, radius=50):
    """Pair coordinate descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):  # stride 3 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):  # stride 2
                    if i == j:
                        continue
                    # Move d from j to i
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Move d from i to j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgradient_optimize(h, n, deadline, n_iter=10000):
    """Subgradient method with adaptive step size."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    eta = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        # Active set: top few
        top_k = min(10, 2*n-1)
        active = np.argsort(-c_vec)[:top_k]
        
        grad = grad_for_subset(best_h, n, active)
        
        new_h = best_h - eta * grad
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = new_h.copy()
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.97
            if eta < 1e-6:
                eta = 0.005
    
    return best_h, best_c

def make_binary_start(n, rng):
    """Create a random binary-like start with sum = n/2."""
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    # Add small perturbation
    h += 0.01 * rng.randn(n)
    return project_feasible(h, n)

def structured_start(n, pattern='sine'):
    """Create structured starting points."""
    x = np.linspace(0, 2, n, endpoint=False)
    if pattern == 'sine':
        h = 0.5 + 0.3 * np.sin(2*np.pi*x)
    elif pattern == 'twosine':
        h = 0.5 + 0.2 * np.sin(2*np.pi*x) + 0.2 * np.sin(4*np.pi*x + 0.5)
    elif pattern == 'fourier':
        h = 0.5
        for freq, amp, phase in [(1, 0.3, 0), (2, 0.15, 0.3), (3, 0.1, 0.7)]:
            h += amp * np.sin(2*np.pi*freq*x + phase)
    elif pattern == 'square':
        h = (np.sin(2*np.pi*x) > 0).astype(float)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
