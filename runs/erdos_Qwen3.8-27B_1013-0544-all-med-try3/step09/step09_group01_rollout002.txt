For s < 0, s' = -s: c[n-1-s'] = sum_{i=0}^{n-1-s'} h[i]*(1-h[i+s'])

Gradient of c[n-1+s] w.r.t. h[j]:

Case s = 0: c = sum_i h[i] - h[i]²

Case s > 0: c = sum_{i=0}^{n-1-s} h[i+s] - h[i+s]*h[i]
dc/dh[j] = [1 if s ≤ j ≤ n-1] - [h[j-s] if s ≤ j ≤ n-1] - [h[j+s] if 0 ≤ j ≤ n-1-s]

Case s < 0 (s' = -s): c = sum_{i=0}^{n-1-s'} h[i] - h[i]*h[i+s']
dc/dh[j] = [1 if 0 ≤ j ≤ n-1-s'] - [h[j+s'] if 0 ≤ j ≤ n-1-s'] - [h[j-s'] if s' ≤ j ≤ n-1]

def compute_c5_fft(h, n):
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    b = (1.0 - h)[::-1]
    B = np.fft.rfft(b, N)
    corr = np.fft.irfft(H * B, N)
    return corr[:2*n - 1]

            if tot < 1e-14:

def logsumexp_obj(h, n, alpha):
    c = compute_c5_fft(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    sh = alpha * (c - mx)

        wm = w[idx]
        k = idx - (n - 1)

            grad += wm * (1.0 - 2.0 * h)

            grad[0:nv] += wm * 1.0
            grad[0:nv] -= wm * h[kk:n]
            grad[kk:n] -= wm * h[0:nv]
    return grad
