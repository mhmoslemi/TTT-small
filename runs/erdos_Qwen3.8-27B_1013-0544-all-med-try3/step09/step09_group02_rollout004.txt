    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(active_idx)):
        idx = active_idx[ii]
        wi = w[ii]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_ls_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_ls_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_round(h, n, deadline, alpha, mu, maxiter, top_k=40, frac=0.005):
    h = project_feasible(h, n)
    c_vec = compute_c5_vec(h, n)
    active = get_active_set(c_vec, top_k, frac)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(ls_obj, h, jac=ls_grad, args=(n, alpha, mu, active),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-16, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def sqp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for j in range(n_act):
        idx = active_idx[j]
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def sqp_optimize(h, n, deadline, max_iter=400, rho_init=0.05, top_k=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        active = get_active_set(c_vec, top_k, 0.008)
        if not active:
            break
        delta, t_opt = sqp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-10:
                break
            continue
        if np.max(np.abs(delta)) < 1e-13:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.06:
                rho = min(rho * 1.25, 0.08)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-11:
                break
            if no_improve > 30:
                break
    return best_h, best_c

def proj_grad_descent(h, n, deadline, lr_init=0.01, top_k=30, max_iter=50000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        active = get_active_set(c_vec, top_k, 0.005)
        if not active:
            break
        alpha = 1000.0
        mu = 0.0
        g = ls_grad(h, n, alpha, mu, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_normed = g / g_norm
        best_h_try = h
        best_c_try = best_c
        for ls in range(5):
            lr_try = lr * (0.7 ** ls)
            h_try = h - lr_try * g_normed
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_try - 1e-16:
                best_h_try = h_try
                best_c_try = c_try
                break
        if best_c_try < best_c - 1e-16:
            best_c = best_c_try
            h = best_h_try
            if lr < 0.02:
                lr = min(lr * 1.05, 0.03)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return h, best_c

def local_search_2move(h, n, deadline, radius=60, n_steps=8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, 5e-05, 2e-05]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            active = get_active_set(c_vec, 25, 0.003)
            grad_sum = np.zeros(n)
            for idx in active:
                grad_sum += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad_sum /= len(active)
            top = np.argsort(-grad_sum)[:min(radius * 2, n)]
            for i_val in top:
                if time.time() > deadline + 2:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius)
                for j in range(lo, hi, max(1, n // n_steps)):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def simulated_annealing(h, n, deadline, n_iter=20000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T = 0.002
    T_min = 1e-7
    cooling = (T_min / T) ** (1.0 / max(1, n_iter))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooling
        idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.05, 0.05)
        h[idx] += delta
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        c_best_c = compute_c5(best_h, n)
        if c < c_best_c or np.random.rand() < np.exp(-(c - c_best_c) / max(T, 1e-10)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h = best_h.copy()
            h[idx] += delta
            h = project_feasible(h, n)
    return best_h, best_c

def slqsq_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution (25s)
    t1 = t0 + 25
    if t1 < deadline - 60:
        for alpha in [500, 2000, 8000]:
            if time.time() > t1 - 2:
                break
            h_new = lbfgs_round(best_h, n_orig, t1 - 2, alpha, 50.0, 3000, top_k=30, frac=0.01)
            c_new = compute_c5(h_new, n_orig)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 2: SQP at original resolution (30s)
    t2 = t0 + 55
    if t2 < deadline - 50:
        best_h, best_c = sqp_optimize(best_h, n_orig, t2, max_iter=200, rho_init=0.05, top_k=25)

    # Phase 3: Upsample to n_target and optimize (major phase)
    n_target = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n_target)
    h_work = np.interp(x_new, x_old, best_h)
    h_work = project_feasible(h_work, n_target)
    c_work = compute_c5(h_work, n_target)
    if c_work < best_c:
        best_c = c_work
        best_h = h_work
    else:
        h_work = best_h.copy() if len(best_h) == n_target else np.interp(x_new, x_old, best_h)
        h_work = project_feasible(h_work, n_target)

    # Phase 4: L-BFGS at n_target (300s)
    t4 = t0 + 330
    if time.time() < t4 - 10:
        alphas_ls = [(200, 20.0, 5000), (500, 100.0, 5000), (1000, 500.0, 5000),
                     (3000, 2000.0, 5000), (8000, 5000.0, 5000), (20000, 20000.0, 3000)]
        for alpha, mu, iters in alphas_ls:
            if time.time() > t4 - 5:
                break
            t_sub = min(time.time() + 45, t4 - 3)
            h_work = lbfgs_round(h_work, n_target, t_sub, alpha, mu, iters, top_k=40, frac=0.008)
            c_work = compute_c5(h_work, n_target)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

    # Phase 5: SQP at n_target (150s)
    t5 = t0 + 480
    if time.time() < t5 - 10:
        h_work, c_work = sqp_optimize(h_work, n_target, t5, max_iter=300, rho_init=0.04, top_k=30)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 6: Projected gradient at n_target (80s)
    t6 = t0 + 560
    if time.time() < t6 - 10:
        h_work, c_work = proj_grad_descent(h_work, n_target, t6, lr_init=0.008, top_k=30, max_iter=30000)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 7: Local search at n_target (120s)
    t7 = t0 + 680
    if time.time() < t7 - 10:
        h_work, c_work = local_search_2move(best_h, n_target, t7, radius=80, n_steps=6)
        if c_work < best_c:
            best_c = c_work

    # Phase 8: Another SQP round (60s)
    t8 = t0 + 740
    if time.time() < t8 - 10:
        h_work, c_work = sqp_optimize(best_h, n_target, t8, max_iter=250, rho_init=0.03, top_k=25)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 9: Simulated annealing / random restarts (100s)
    t9 = t0 + 840
    if time.time() < t9 - 10:
        h_work, c_work = simulated_annealing(best_h, n_target, t9, n_iter=10000)
        if c_work < best_c:
            best_c = c_work
            best_h = h_work.copy()

    # Phase 10: Final L-BFGS polish (60s)
    t10 = t0 + 900
    if time.time() < t10 - 10:
        for alpha, mu in [(5000, 5000.0), (15000, 20000.0), (40000, 80000.0)]:
            if time.time() > t10 - 5:
                break
            h_new = lbfgs_round(best_h, n_target, t10 - 3, alpha, mu, 3000, top_k=50, frac=0.01)
            c_new = compute_c5(h_new, n_target)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 11: Try higher resolution if time permits (n=1500)
    if time.time() < deadline - 120:
        n_high = 1500
        x_hi = np.linspace(0, 2, n_high)
        x_mid = np.linspace(0, 2, n_target)
        h_hi = np.interp(x_hi, x_mid, best_h)
        h_hi = project_feasible(h_hi, n_high)
        c_hi = compute_c5(h_hi, n_high)
        if c_hi < best_c + 0.001:
            t_hi = min(time.time() + 80, deadline - 60)
            h_hi = lbfgs_round(h_hi, n_high, t_hi - 20, 3000, 2000.0, 4000, top_k=40, frac=0.01)
            c_hi = compute_c5(h_hi, n_high)
            if c_hi < best_c + 0.0005:
                h_hi, c_hi = sqp_optimize(h_hi, n_high, min(time.time() + 40, deadline - 30),
                                          max_iter=150, rho_init=0.03, top_k=30)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi
                n_target = n_high

    # Phase 12: Final SLSQP polish
    if time.time() < deadline - 20:
        h_fin, c_fin = slqsq_polish(best_h, len(best_h), deadline - 5, alpha=2000000, maxiter=2000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
