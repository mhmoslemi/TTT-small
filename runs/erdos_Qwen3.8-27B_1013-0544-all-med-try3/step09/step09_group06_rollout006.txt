def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, -5.0, 5.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(h - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        free = (z > 1e-14) & (z < 1.0 - 1e-14)
        if free.sum() > 0:
            z[free] += diff / free.sum()
        else:
            z = np.clip(z, 0, 1)
            if z.sum() < target:
                z[z < 1.0] += (target - z.sum()) / np.sum(z < 1.0)
            else:
                z[z > 0.0] += (target - z.sum()) / np.sum(z > 0.0)
    return np.clip(z, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h

        grad[m:n] += (-h[0:n-m])
    else:
        grad[0:n-m] += (-h[m:n])
        grad[m:n] += (1.0 - h[0:n-m])
    return grad

def slsqp_polish(h, n, deadline, alpha=100000, top_m=5, maxiter=2000):
    if time.time() > deadline:
        return h
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    dx = 2.0 / n

    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def grad_obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 400:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w /= w.sum()
        g = np.zeros(n)
        for idx, k in enumerate(top_idx):
            if w[idx] < 1e-10:
                continue
            lag = k - (n - 1)
            g += w[idx] * lag_gradient(x, n, lag)
        return g * dx

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h):
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new, n) < compute_c5(h, n):
                return h_new
    except Exception:
        pass
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    # Also handle boundary
    x_old_ext = np.concatenate([[0.0], x_old, [2.0]])
    h_old_ext = np.concatenate([[h_old[0]], h_old, [h_old[-1]]])
    h_new = np.interp(x_new, x_old_ext, h_old_ext)
    return project_feasible(h_new, n_new)

def random_restart(h_best, n, rng, scale):
    h = h_best + rng.normal(0, scale, n)
    return project_feasible(h, n)

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase A: Cyclic prox-gradient
    grad_deadline = min(time.time() + (deadline - time.time()) * 0.4, deadline - 10)
    h_new, c_new = cyclic_prox_gradient(h, n, grad_deadline, n_top=5,
                                         lr_init=0.03, momentum_init=0.88)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new
    h = h_new

    # Phase B: SLSQP polish
    if time.time() < deadline - 15:
        slq_deadline = min(time.time() + 30, deadline - 10)
        h_slsqp = slsqp_polish(h, n, slq_deadline, alpha=50000, top_m=5, maxiter=1500)
        c_slsqp = compute_c5(h_slsqp, n)
        if c_slsqp < best_c:
            best_c = c_slsqp
            best_h = h_slsqp
        h = h_slsqp

    # Phase C: Greedy local search
    ls_deadline = min(time.time() + (deadline - time.time()) * 0.5, deadline - 5)
    h_new, c_new = greedy_local_search(h, n, ls_deadline, top_k=3)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new

    # Phase D: Fine polish
    if time.time() < deadline - 3:
        fp_deadline = min(time.time() + (deadline - time.time()) * 0.4, deadline - 1)
        h_new, c_new = fine_polish(best_h, n, fp_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Step 1: Optimize at initial resolution
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        best_c = compute_c5(h_init, n_init)
        best_h = h_init.copy()
        alloc = min(300, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc = min(200, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_resolution(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Step 2: Try higher resolutions
    n_candidates = [600, 750, 1000, 1200]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 30
        if remaining < 80:
            break
        alloc = min(120, remaining * 0.35)
        if alloc < 60:
            break
        h_start = interpolate_h(best_h, len(best_h), n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Step 3: Try lower resolution for comparison
    if time.time() < deadline - 80:
        for n_try in [400, 450]:
            if time.time() > deadline - 50:
                break
            h_start = interpolate_h(best_h, len(best_h), n_try)
            alloc = min(60, (deadline - time.time() - 20) * 0.4)
            if alloc < 40:
                break
            h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Step 4: Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        restarts = [
            (0.01, "small_noise"),
            (0.03, "med_noise"),
            (0.05, "large_noise"),
        ]
        for scale, _ in restarts:
            if time.time() > deadline - 40:
                break
            h_pert = random_restart(best_h, n_best, rng, scale)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(h_pert, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Step 5: Final intensive local search
    n_final = len(best_h)
    if time.time() < deadline - 10:
        c_all = compute_all_lags(best_h, n_final)
        top_k = 5
        ls_deadline = deadline - 3
        h_new, c_new = greedy_local_search(best_h, n_final, ls_deadline, top_k=top_k)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Step 6: Ultra-fine final polish
    if time.time() < deadline - 2:
        fp_deadline = deadline - 0.5
        h_new, c_new = fine_polish(best_h, n_final, fp_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final SLSQP if time allows
    if time.time() < deadline - 5:
        h_slsqp = slsqp_polish(best_h, n_final, deadline - 1, alpha=200000, top_m=3, maxiter=1000)
        c_slsqp = compute_c5(h_slsqp, n_final)
        if c_slsqp < best_c:
            best_c = c_slsqp
            best_h = h_slsqp

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
