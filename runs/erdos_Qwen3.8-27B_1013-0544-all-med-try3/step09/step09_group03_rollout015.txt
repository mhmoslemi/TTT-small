Let me use the FFT-based gradient for higher resolutions:

For the gradient, I need:
g[j] = sum over all shifts k of w_k * ∂c_k/∂h_j

This can be decomposed into convolutions that I can compute with FFT.

Let me think about this more carefully.

For shift k ≥ 0:
∂c_k/∂h[j] = (1-h[j+k]) for j = 0,...,n-1-k
             -h[j-k] for j = k,...,n-1
             (and 0 elsewhere, but the above covers all j)

For shift -k (k > 0):
∂c_{-k}/∂h[j] = (1-h[j-k]) for j = k,...,n-1
                 -h[j+k] for j = 0,...,n-1-k

So the total gradient is:
g[j] = w_0*(1-2h[j]) 
     + sum_{k=1}^{n-1} w_k * [(1-h[j+k])*[j≤n-1-k] - h[j-k]*[j≥k]]
     - sum_{k=1}^{n-1} w_k * [(1-h[j-k])*[j≥k] - h[j+k)*[j≤n-1-k]]

Wait, I need to be careful with the indexing of w. Let me define:
- w_pos[k] = w[center + k] * dx for k = 0, 1, ..., n-1 (these are weights for positive shifts)
- w_neg[k] = w[center - k] * dx for k = 0, 1, ..., n-1 (these are weights for negative shifts)

Note: w_pos[0] = w_neg[0] = w[center] * dx (shift 0)

g[j] = w_pos[0]*(1-2*h[j])
     + sum_{k=1}^{min(j, n-1)} w_neg[k]*(1-h[j-k])  [from negative shift -k, first term]
     - sum_{k=1}^{min(n-1-j, n-1)} w_neg[k]*h[j+k]  [from negative shift -k, second term]
     + sum_{k=1}^{min(n-1-j, n-1)} w_pos[k]*(1-h[j+k])  [from positive shift k, first term]
     - sum_{k=1}^{min(j, n-1)} w_pos[k]*h[j-k]  [from positive shift k, second term]

Let me define:
A[j] = sum_{k=1}^{min(n-1-j, n-1)} w_pos[k]*(1-h[j+k])  [forward correlation of w_pos with (1-h)]
B[j] = sum_{k=1}^{min(j, n-1)} w_pos[k]*h[j-k]  [backward correlation of w_pos with h]
C[j] = sum_{k=1}^{min(j, n-1)} w_neg[k]*(1-h[j-k])  [backward correlation of w_neg with (1-h)]
D[j] = sum_{k=1}^{min(n-1-j, n-1)} w_neg[k]*h[j+k]  [forward correlation of w_neg with h]

g[j] = w_pos[0]*(1-2*h[j]) + A[j] - B[j] + C[j] - D[j]

Now, A[j] is the cross-correlation of w_pos[1:] with (1-h), evaluated at lag j.
Specifically: A[j] = sum_{k=1}^{n-1-j} w_pos[k] * (1-h[j+k])

If I define a = w_pos[1:n] (length n-1) and b = 1-h (length n), then:
A[j] = sum_{k=0}^{n-2-j} a[k] * b[j+1+k]

This is a "convolution" (actually cross-correlation) that can be computed via FFT.

Similarly for B, C, D.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    one_minus_x = 1.0 - x
    grad = np.zeros(n)
    center = n - 1
    # Sort by weight for early termination
    w_idx = np.arange(2*n-1)
    for idx in w_idx:
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - center
        wm = wi * dx
        if k == 0:
            grad += wm * (1 - 2*x)
        elif k > 0:
            nv = n - k
            grad[0:nv] += wm * one_minus_x[k:n]
            grad[k:n] -= wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * one_minus_x[0:nv]
            grad[0:nv] -= wm * x[kk:n]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (10000, 5000.0), (30000, 15000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=200000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def gradient_descent_refine(h, n, deadline, n_steps=500):
    """Subgradient-like refinement with line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    
    for step_num in range(n_steps):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(best_h, n)
        m = c.max()
        
        # Subgradient at the max shift
        active_threshold = m - 0.00005
        active = c >= active_threshold
        if not active.any():
            active_idx = int(np.argmax(c))
            active = np.zeros(2*n-1)
            active[active_idx] = True
        
        sg = np.zeros(n)
        one_minus_x = 1.0 - best_h
        for idx in np.where(active)[0]:
            k = idx - center
            if k == 0:
                sg += (1 - 2*best_h)
            elif k > 0:
                nv = n - k
                sg[0:nv] += one_minus_x[k:n]
                sg[k:n] -= best_h[0:nv]
            else:
                kk = -k
                nv = n - kk
                sg[kk:n] += one_minus_x[0:nv]
                sg[0:nv] -= best_h[kk:n]
        
        sg *= dx
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        
        # Line search
        best_step = 0
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.050, 0.025, 0.015]:
            h_try = project_feasible(best_h - step * sg / sg_norm, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try
                best_step = step
                break
        
        if best_step == 0:
            # Try coordinate adjustments on top gradient components
            sg_sorted = np.argsort(-np.abs(sg))[::-1]
            improved = False
            for i in sg_sorted[:30]:
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.003, -0.003, 0.001, -0.001, 0.005, -0.005]:
                    nv = best_h[i] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    best_h[i] = nv
                    c_try = compute_c5(best_h, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                        improved = True
                        break
                    else:
                        best_h[i] -= delta
            if not improved:
                # Try pairs
                for i in sg_sorted[:10]:
                    for j in sg_sorted[:10]:
                        if i == j:
                            continue
                        for di, dj in [(0.002, -0.002), (-0.002, 0.002), (0.001, -0.001), (-0.001, 0.001)]:
                            ni = best_h[i] + di
                            nj = best_h[j] + dj
                            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                                continue
                            best_h[i] = ni
                            best_h[j] = nj
                            c_try = compute_c5(best_h, n)
                            if c_try < best_c - 1e-15:
                                best_c = c_try
                                improved = True
                                break
                            else:
                                best_h[i] -= di
                                best_h[j] -= dj
                        if improved:
                            break
                    if improved:
                        break
                if not improved:
                    break
    
    return best_h, best_c

def multi_start_search(h, n, deadline, n_starts=5, seed=42):
    """Try random perturbations and optimize each."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    
    for t in range(n_starts):
        if time.time() > deadline - 60:
            break
        # Structured perturbation: sine wave at random frequency
        x = np.linspace(0, 2, n)
        freq = 3 + 7 * rng.rand()  # 3-10 periods
        phase = 2 * np.pi * rng.rand()
        amp = 0.05 + 0.1 * rng.rand()
        h_p = h + amp * np.sin(freq * np.pi * x + phase)
        h_p = project_feasible(h_p, n)
        
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def fine_local_search(h, n, deadline):
    """Very fine single-point and pair adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for round_num in range(100):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        # Identify affected points
        if k_worst >= 0:
            range1 = range(0, min(n - k_worst, n))
            range2 = range(k_worst, n)
        else:
            range1 = range(-k_worst, n)
            range2 = range(0, n + k_worst)
        
        affected = set(list(range1) + list(range2))
        affected_list = list(affected)
        
        # Sort by value (extremes first)
        affected_list.sort(key=lambda i: -abs(best_h[i] - 0.5))
        
        found = False
        for i in affected_list[:20]:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002, 0.0002, -0.0002]:
                nv = best_h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                best_h[i] = nv
                c_try = compute_c5(best_h, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    found = True
                    break
                else:
                    best_h[i] -= delta
            if found:
                break
        
        # Try pairs among affected
        if not found and len(affected_list) >= 2:
            for i in affected_list[:8]:
                for j in affected_list[:8]:
                    if i == j:
                        continue
                    for di, dj in [(0.001,-0.001), (-0.001,0.001), (0.0005,-0.0005), (-0.0005,0.0005)]:
                        ni = best_h[i] + di
                        nj = best_h[j] + dj
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        best_h[i] = ni
                        best_h[j] = nj
                        c_try = compute_c5(best_h, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            found = True
                            break
                        else:
                            best_h[i] -= di
                            best_h[j] -= dj
                    if found:
                        break
                if found:
                    break
        
        if not found and round_num > 30:
            break
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    one_minus_x = 1.0 - x
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - center
        wm = wi * dx
        if k == 0:
            grad += wm * (1 - 2 * x)
        elif k > 0:
            nv = n - k
            grad[0:nv] += wm * one_minus_x[k:n]
            grad[k:n] -= wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * one_minus_x[0:nv]
            grad[0:nv] -= wm * x[kk:n]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (1000, 200.0),
                   (3000, 800.0), (8000, 3000.0), (20000, 8000.0),
                   (50000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': 150000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_obj_fn(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slqp_eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def slqp_eq_jac(x, n):
    return np.ones(n)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_eq_fun, 'jac': slqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def gradient_descent_refine(h, n, deadline, n_steps=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1

    for step_num in range(n_steps):
        if time.time() > deadline:
            break

        c = compute_c5_vec(best_h, n)
        m = c.max()
        active_threshold = m - 0.00003
        active_mask = c >= active_threshold
        if not active_mask.any():
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[int(np.argmax(c))] = True

        sg = np.zeros(n)
        one_minus_x = 1.0 - best_h
        active_indices = np.where(active_mask)[0]
        for idx in active_indices:
            k = idx - center
            if k == 0:
                sg += (1 - 2 * best_h)
            elif k > 0:
                nv = n - k
                sg[0:nv] += one_minus_x[k:n]
                sg[k:n] -= best_h[0:nv]
            else:
                kk = -k
                nv = n - kk
                sg[kk:n] += one_minus_x[0:nv]
                sg[0:nv] -= best_h[kk:n]

        sg *= dx
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-14:
            break

        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.03, 0.015, 0.008, 0.004]:
            h_try = project_feasible(best_h - step * sg / sg_norm, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try
                improved = True
                break

        if not improved:
            sg_abs = np.abs(sg)
            top_idx = np.argsort(-sg_abs)[:25]
            for i in top_idx:
                if time.time() > deadline:
                    return best_h, best_c
                direction = -1.0 if sg[i] > 0 else 1.0
                for mag in [0.003, 0.001, 0.005, 0.0005, 0.002]:
                    nv = best_h[i] + direction * mag
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    best_h[i] = nv
                    c_try = compute_c5(best_h, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                        improved = True
                        break
                    else:
                        best_h[i] -= direction * mag
                if improved:
                    break

            if not improved:
                pair_done = False
                for i in top_idx[:10]:
                    if pair_done:
                        break
                    for j in top_idx[:10]:
                        if i == j or pair_done:
                            continue
                        ni_dir = -1 if sg[i] > 0 else 1
                        nj_dir = -1 if sg[j] > 0 else 1
                        for mag_p in [0.002, 0.001, 0.003, 0.0005]:
                            ni = best_h[i] + ni_dir * mag_p
                            nj = best_h[j] + nj_dir * mag_p
                            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                                continue
                            best_h[i] = ni
                            best_h[j] = nj
                            c_try = compute_c5(best_h, n)
                            if c_try < best_c - 1e-15:
                                best_c = c_try
                                pair_done = True
                                break
                            else:
                                best_h[i] = ni - ni_dir * mag_p
                                best_h[j] = nj - nj_dir * mag_p

                if not pair_done:
                    break

    return best_h, best_c

def multi_start_search(h, n, deadline, n_starts=4, seed=99):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)

    for t in range(n_starts):
        if time.time() > deadline - 50:
            break
        freq = 2.0 + 8.0 * rng.rand()
        phase = 2.0 * np.pi * rng.rand()
        amp = 0.03 + 0.08 * rng.rand()
        h_p = h + amp * np.sin(freq * np.pi * x + phase)
        h_p = project_feasible(h_p, n)

        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c

def fine_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    center = n - 1

    for rnd in range(80):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(best_h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - center

        if k_worst >= 0:
            aff1 = list(range(0, n - k_worst))
            aff2 = list(range(k_worst, n))
        else:
            aff1 = list(range(-k_worst, n))
            aff2 = list(range(0, n + k_worst))

        affected = sorted(set(aff1 + aff2), key=lambda i: -abs(best_h[i] - 0.5))

        found = False
        for i in affected[:25]:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.0005, -0.0005, 0.001, -0.001, 0.0002, -0.0002, 0.002, -0.002]:
                nv = best_h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:

                best_h[i] = nv
                c_try = compute_c5(best_h, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    found = True
                    break
                else:
                    best_h[i] -= delta
            if found:
                break

        if not found and len(affected) >= 2:
            for i in affected[:10]:
                if found:

                for j in affected[:10]:
                    if i == j or found:

                    for di, dj in [(0.0005, -0.0005), (-0.0005, 0.0005),
