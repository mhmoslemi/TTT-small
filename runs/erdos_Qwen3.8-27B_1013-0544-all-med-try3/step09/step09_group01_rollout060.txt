import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smo_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def smo_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process significant weights
    significant = w > 1e-12
    nsig = significant.sum()
    if nsig == 0:
        idx_sig = [0]
        w_sig = [0.0]
    else:
        idx_sig = np.where(significant)[0]
        w_sig = w[idx_sig]
    for ii in range(len(idx_sig)):
        mm = idx_sig[ii]
        wi = w_sig[ii]
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

For i in [k, n-k-1] (if k < n-k): both terms apply: (1-h[i+k]) - h[i-k]

So v_k[i] = (1-h[i+k]) * [i < n-k] - h[i-k] * [i ≥ k]

This can be written as:
v_k = [(1-h[k:n])] · [1,0,...,0] (padded to length n) + [0,...,0,1,1,...,1] · [-h[0:n-k]]

In numpy:
v_k = np.zeros(n)
v_k[:n-k] += (1 - h[k:])
v_k[k:] += (-h[:n-k])

And grad += W[k-（n-1)] * v_k (where the index into W is the shift)

This is what the previous code does in the loop. The loop over k has n-1 iterations for positive k and n-1 for negative k, total 2n-2 iterations. Each iteration is O(n) numpy operations. Total: O(n²) but with a small constant.

For n=2000: 3998 iterations × (2+2 numpy ops on ~1000-2000 elements) ≈ 3998 × 4 × 0.5μs ≈ 8ms. That's fine.

grad[i] = dx * sum_k w_shift[k] * ((1-h[i+k])·[0≤i+k<n] - h[i-k]·[0≤i-k<n])

= dx * (sum_k w_shift[k]·(1-h[i+k])·[0≤i+k<n] - sum_k w_shift[k]·h[i-k]·[0≤i-k<n])

The first sum is a cross-correlation of w_shift with (1-h), and the second is a cross-correlation of w_shift with h. Both can be computed using FFT!

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
