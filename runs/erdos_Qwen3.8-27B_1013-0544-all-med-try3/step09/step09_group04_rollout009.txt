import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_opt(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-20,
                                    'gtol': 1e-16, 'maxfun': maxit * 5,
                                    'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def block_slp(h, n, deadline, block_size=20, top_k=1000):

    best_c = compute_c5(best_h, n)

    block_starts = [i * block_size for i in range(n_blocks)]
    block_ends = [min((i+1)*block_size, n) for i in range(n_blocks)]
    
    # Precompute which solver to use
    solver = cp.GLOP
    
    for sweep in range(200):
        if time.time() > deadline - 2:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Top k active constraints
        top_idx = np.argsort(-c_vec)[:top_k]
        
        for bi in range(n_blocks):
            if time.time() > deadline - 1:
                break
            start = block_starts[bi]
            end = block_ends[bi]
            b = end - start
            
            # Target for this block
            target_block = n/2.0 - (best_h[:start].sum() + best_h[end:].sum())
            target_block = max(0, min(b, target_block))
            
            # Build LP
            try:
                x_var = cp.Variable(b)
                T = cp.Variable()
                obj = cp.Minimize(T)
                cons = [x_var >= 0, x_var <= 1,
                        cp.sum(x_var) == target_block]
                
                for k_idx in top_idx:
                    s = k_idx - (n - 1)
                    # Compute g_{k,j} for j in block
                    g = np.zeros(b)
                    for jj in range(b):
                        j = start + jj
                        # g = dx * ((1-h[j+s]) - h[j-s])
                        c1 = 0.0
                        if 0 <= j+s < n:
                            c1 = 1.0 - best_h[j+s]
                        c2 = 0.0
                        if 0 <= j-s < n:
                            c2 = best_h[j-s]
                        g[jj] = (2.0/n) * (c1 - c2)
                    
                    # C_k = -f_k(h*) + sum g_{k,j} * h*[j]
                    f_k = c_vec[k_idx]
                    h_block = best_h[start:end]
                    C_k = -f_k + np.dot(g, h_block)
                    
                    cons.append(cp.sum(g * x_var) - T <= C_k)
                
                prob = cp.Problem(obj, cons)
                prob.solve(solver=solver)
                
                if x_var.value is not None:

                    h_new[start:end] = x_var.value
                    h_new = project_feasible(h_new, n)

                    if c_new < best_c:

                        best_h = h_new
            except Exception:
