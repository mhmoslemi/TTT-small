    for idx in active:
        k = idx - (n - 1)
        def make_cf(idx_val):
            def cf(x_):
                h_ = x_[:n_]
                t_ = x_[n_]
                c_ = np.correlate(h_, 1.0 - h_, mode='full') * (2.0 / n_)
                return c_[idx_val] - t_
            def cf_jac(x_):
                n_ = x_.shape[0] - 1
                h_ = x_[:n_]
                dx_ = 2.0 / n_
                grad_h = np.zeros(n_)
                idx_v = idx_val
                kk = idx_v - (n_ - 1)
                if kk >= 0:
                    nv_ = n_ - kk
                    grad_h[0:nv_] += dx_ * (1.0 - h_[kk:n_])
                    grad_h[kk:n_] += dx_ * (-h_[0:nv_])
                else:
                    kk2 = -kk
                    nv_ = n_ - kk2
                    grad_h[kk2:n_] += dx_ * (1.0 - h_[0:nv_])
                    grad_h[0:nv_] += dx_ * (-h_[kk2:n_])
                return np.concatenate([grad_h, [-1.0]])
            return cf, cf_jac
        cf, cfj = make_cf(idx)
        cons.append({'type': 'ineq', 'fun': lambda x_, cf=cf: -cf(x_), 'jac': lambda x_, cfj=cfj: -cfj(x_)})
    return cons

def run_lbfgs(h_init, n, deadline, alphas, mu=0.005, maxiter_list=None):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiter_list is not None and i < len(maxiter_list):
            maxit = maxiter_list[i]
        else:
            maxit = max(500, int(remaining * 6))
        try:
            res = minimize(lse_obj, best_h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_sls_minmax(h_init, n, deadline, top_k=8):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    for round_idx in range(6):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = np.argsort(-c_vec)[:top_k]
        
        x0 = np.concatenate([best_h, [best_c + 0.01]])
        bounds_mm = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        h_sub = best_h.copy()
        n_sub = n
        def obj(x_):
            h_v = x_[:n_sub]
            t_v = x_[n_sub]
            c_v = np.correlate(h_v, 1.0 - h_v, mode='full') * (2.0 / n_sub)
            pen = 0.0
            for idx in active:
                if c_v[idx] > t_v:
                    pen += (c_v[idx] - t_v) ** 2
            return t_v + 500.0 * pen
        
        def obj_jac(x_):
            h_v = x_[:n_sub]
            t_v = x_[n_sub]
            dx_v = 2.0 / n_sub
            c_v = np.correlate(h_v, 1.0 - h_v, mode='full') * dx_v
            grad_h = np.zeros(n_sub)
            for idx in active:
                kk = idx - (n_sub - 1)
                if c_v[idx] > t_v:
                    scale = 1.0 + 1000.0 * max(0.0, c_v[idx] - t_v)
                else:
                    scale = 1.0
                if kk >= 0:
                    nv_ = n_sub - kk
                    grad_h[0:nv_] += scale * dx_v * (1.0 - h_v[kk:n_sub])
                    grad_h[kk:n_sub] += scale * dx_v * (-h_v[0:nv_])
                else:
                    kk2 = -kk
                    nv_ = n_sub - kk2
                    grad_h[kk2:n_sub] += scale * dx_v * (1.0 - h_v[0:nv_])
                    grad_h[0:nv_] += scale * dx_v * (-h_v[kk2:n_sub])
            grad_t = 1.0
            for idx in active:
                if c_v[idx] > t_v:
                    grad_t -= 1000.0 * (c_v[idx] - t_v)
            return np.concatenate([grad_h, [grad_t]])
        
        cons_mm = [{'type': 'eq',
                    'fun': lambda x_: x_[:n_sub].sum() - n_sub / 2.0,
                    'jac': lambda x_: np.concatenate([np.ones(n_sub), [0.0]])}]
        
        maxit_mm = max(200, int(remaining * 2))
        try:
            res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                          bounds=bounds_mm, constraints=cons_mm,
                          options={'maxiter': maxit_mm, 'ftol': 1e-18})
            if res.success or res.fun < obj(x0):
                h_new = project_feasible(res.x[:n], n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def active_set_lbfgs(h_init, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000]
    
    for rnd in range(8):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 4))
            
            def act_obj(x_):
                c_v = compute_c5_vec(x_, n)
                vals = c_v[active]
                mx_ = vals.max()
                sh_ = alpha * (vals - mx_)
                smx_ = sh_.max()
                if smx_ > 500:
                    v = mx_ + (smx_ + np.log(np.sum(np.exp(sh_ - smx_)))) / alpha
                else:
                    v = mx_ + np.log(np.sum(np.exp(sh_))) / alpha
                v += 0.01 * (x_.sum() - n / 2.0) ** 2
                return float(v)
            
            def act_grad(x_):
                c_v = compute_c5_vec(x_, n)
                vals = c_v[active]
                mx_ = vals.max()
                sh_ = alpha * (vals - mx_)
                smx_ = sh_.max()
                w_ = np.exp(sh_ - smx_)
                w_ /= w_.sum()
                dx_ = 2.0 / n
                g = np.zeros(n)
                for wi, idx in zip(w_, active):
                    kk = idx - (n - 1)
                    wm = wi * dx_
                    if kk >= 0:
                        nv_ = n - kk
                        g[0:nv_] += wm * (1.0 - x_[kk:n])
                        g[kk:n] += wm * (-x_[0:nv_])
                    else:
                        kk2 = -kk
                        nv_ = n - kk2
                        g[kk2:n] += wm * (1.0 - x_[0:nv_])
                        g[0:nv_] += wm * (-x_[kk2:n])
                g += 0.02 * (x_.sum() - n / 2.0) * np.ones(n)
                return g
            
            try:
                res = minimize(act_obj, best_h, jac=act_grad, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def binary_sa(h_init, n, deadline, n_iter=50000):
    h_bin = np.zeros(n)
    n_ones = int(n / 2.0)
    sorted_idx = np.argsort(-h_init)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        for i in range(n_ones, n):
            add = min(1.0 - h_bin[i], remaining)
            h_bin[i] += add
            remaining -= add
            if abs(remaining) < 1e-12:
                break
    h_cur = project_feasible(h_bin, n)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    
    T = 0.001
    T_min = 1e-7
    cooling = 0.999
    
    ones_pos = np.where(h_cur >= 0.5)[0]
    zeros_pos = np.where(h_cur < 0.5)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooling
        i = np.random.randint(len(ones_pos)) if len(ones_pos) > 0 else 0
        j = np.random.randint(len(zeros_pos)) if len(zeros_pos) > 0 else 0
        pos_i = ones_pos[i]
        pos_j = zeros_pos[j]
        
        h_cur[pos_i] = 0.0
        h_cur[pos_j] = 1.0
        c_new = compute_c5(h_cur, n)
        c_old = best_c
        
        if c_new < c_old:
            best_h = h_cur.copy()
            best_c = c_new
        else:
            delta = c_new - c_old
            if delta > 0 and np.random.rand() < np.exp(-delta / max(T, 1e-15)):
                best_h = h_cur.copy()
                best_c = c_new
            else:
                h_cur[pos_i] = 1.0
                h_cur[pos_j] = 0.0
    
    return best_h, best_c

def coordinate_descent(h_init, n, deadline, max_rounds=100):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        for rnd in range(max_rounds):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for d in [step, -step]:
                    if best_h[i] + d >= 0.0 and best_h[i] + d <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] += d
                        h_try = project_feasible(h_try, n)
                        c_try = compute_c5(h_try, n)
                        if c_try < best_c - 1e-16:
                            best_c = c_try
                            best_h = h_try
                            improved = True
                            break
            if not improved:
                break
    return best_h, best_c

def try_m resolution(n_target, h_coarse, coarse_n, deadline):
    if coarse_n == n_target:
        return project_feasible(h_coarse, n_target), compute_c5(h_coarse, n_target)
    x_fine = np.linspace(0, 2, n_target, endpoint=False)
    x_coarse = np.linspace(0, 2, coarse_n, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_feasible(h_fine, n_target)
    return h_fine, compute_c5(h_fine, n_target)

def free_param_nm(h_init, n, deadline, nfree=8, maxiter=500):
    free_mask = (h_init > 0.01) & (h_init < 0.99)
    if free_mask.sum() == 0:
        free_mask = (h_init > 0.001) & (h_init < 0.999)
    if free_mask.sum() == 0:
        return h_init, compute_c5(h_init, n)
    
    free_idx = np.where(free_mask)[0]
    if len(free_idx) > nfree:
        free_vals = h_init[free_idx]
        mid = 0.5
        dists = np.abs(free_vals - mid)
        order = np.argsort(dists)
        free_idx = free_idx[order[:nfree]]
    
    free_counts = len(free_idx)
    h_base = h_init.copy()
    
    def nm_obj(params):
        h_test = h_base.copy()
        h_test[free_idx] = np.clip(params, 0.001, 0.999)
        h_test = project_feasible(h_test, n)
        return compute_c5(h_test, n)
    
    x0 = h_init[free_idx].copy()
    try:
        res = fmin(nm_obj, x0, disp=False, maxiter=maxiter, maxfun=maxiter * 3)
        h_new = h_base.copy()
        h_new[free_idx] = np.clip(res, 0.001, 0.999)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        c_orig = compute_c5(h_init, n)
        if c_new < c_orig:
            return h_new, c_new
        else:
            return h_init, c_orig
    except Exception:
        return h_init, compute_c5(h_init, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: L-BFGS at original resolution
    p1_end = t0 + 180
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 500, 5000, 50000, 500000],
        mu=0.005, maxiter_list=[1500, 2000, 2500, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set L-BFGS
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 120, deadline - 280)
        h2, c2 = active_set_lbfgs(best_h, best_n, p2_end, top_k=5,
            alphas=[50000, 500000, 2000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: SLSQP min-max
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 100, deadline - 240)
        h3, c3 = run_sls_minmax(best_h, best_n, p3_end, top_k=6)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try n=1000
    n_high = 1000
    if time.time() < deadline - 250:
        x_fine = np.linspace(0, 2, n_high, endpoint=False)
        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_high)
        c_fine = compute_c5(h_fine, n_high)
        
        p4_end = min(time.time() + 200, deadline - 200)
        h4, c4 = run_lbfgs(h_fine, n_high, p4_end,
            alphas=[20, 200, 2000, 20000, 200000],
            mu=0.005, maxiter_list=[1000, 1500, 2000, 2500, 3000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high
    
    # Phase 5: Active set at best resolution
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 80, deadline - 170)
        h5, c5 = active_set_lbfgs(best_h, best_n, p5_end, top_k=4,
            alphas=[100000, 1000000, 5000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: SLSQP at best resolution
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 60, deadline - 110)
        h6, c6 = run_sls_minmax(best_h, best_n, p6_end, top_k=5)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Multi-start with perturbations
    if time.time() < deadline - 100:
        for trial in range(4):
            remaining = deadline - time.time()
            if remaining < 40:
                break
            pert = rng.randn(best_n) * 0.02
            h_pert = project_feasible(best_h + pert, best_n)
            c_pert = compute_c5(h_pert, best_n)
            
            p_t_end = min(time.time() + 30, deadline - 80)
            h_t, c_t = run_lbfgs(h_pert, best_n, p_t_end,
                alphas=[5000, 50000, 500000],
                mu=0.005, maxiter_list=[800, 1000, 1500])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
    
    # Phase 8: Free parameter Nelder-Mead
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 25, deadline - 30)
        h8, c8 = free_param_nm(best_h, best_n, p8_end, nfree=10, maxiter=300)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Coordinate descent
    if time.time() < deadline - 10:
        p9_end = deadline - 3
        h9, c9 = coordinate_descent(best_h, best_n, p9_end, max_rounds=50)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final active set push
    if time.time() < deadline - 2:
        h10, c10 = active_set_lbfgs(best_h, best_n, deadline - 1, top_k=3,
            alphas=[5000000, 20000000])
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
