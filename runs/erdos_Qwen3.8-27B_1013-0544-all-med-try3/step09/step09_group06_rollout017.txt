    """Approximate change in max C5 when h[idx] changes by delta."""
    dx = 2.0 / n
    c_change = np.zeros(len(top_lags))
    for li, lag in enumerate(top_lags):
        m = abs(lag)
        if m >= n:
            continue

            g = 0.0
            if idx <= n - m - 1:
                g += (1 - h[idx + m])
            if idx >= m and idx - m <= n - m - 1:
                g -= h[idx - m]
        elif lag < 0:
            g = 0.0
            if idx >= m:
                g += (1 - h[idx - m])
            if idx + m <= n - 1:
                g -= h[idx + m]
        else:
            g = 1 - 2 * h[idx]
        c_change[li] = delta * g * dx
    
    new_c_top = c_current + c_change
    return new_c_top.max()

                        h = project_feasible(h, n)
                if improved:
                    break
