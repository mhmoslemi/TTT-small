def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def _active_smooth_obj(x, n, top_idx, alpha):

    cc = np.correlate(x, 1.0 - x, mode='full') * dx

    shifted = alpha * (active - m)
    mx = float(np.max(shifted))

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h, n, top_k=10, alpha=100000, maxiter=5000):

    top_idx = np.argsort(c)[-top_k:]

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

        h_new = symmetrize(h_new)

def block_perturb(h, n, rng, block_len=5):

    h = h_new
    
    phase1b_deadline = time.time() + (deadline - time.time()) * 0.15
    h_new, c_new = adam_symmetric(h, n, phase1b_deadline, alpha=2000, top_k=20, lr=0.5)
    if c_new < best_c:

    h = h_new
    
    if time.time() < deadline - 60:

def optimize_general(h_start, n, deadline, rng):

def _adam_general(h, n, deadline, alpha=300, top_k=50, lr=1.0):
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    patience = 0
    while time.time() < deadline:
        t += 1
        g, c = grad_smooth_max(h_curr, n, alpha, top_k)
        m_v = 0.9 * m_v + 0.1 * g
        v_v = 0.999 * v_v + 0.001 * (g * g)
        m_hat = m_v / (1.0 - 0.9**t)
        v_hat = v_v / (1.0 - 0.999**t)
        step = lr / t**0.5 * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h_curr - step
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h_curr = h_new
        if patience > 3000:
            lr *= 0.7
            m_v *= 0.5
            v_v *= 0.5
            patience = 0
    return best_h, best_c

def try_resolution(h_good, n_good, n_try, deadline, rng):
    x_good = np.linspace(0, 2, n_good)
    x_try = np.linspace(0, 2, n_try)
    h_start = np.interp(x_try, x_good, h_good)
    h_start = symmetrize(h_start)
    h_start = project_feasible(h_start, n_try)
    alloc = min(150, (deadline - time.time()) * 0.2)
    if alloc < 40:
        return h_good, compute_c5(h_good, n_good), n_good
    h_opt, c_opt = optimize_symmetric(h_start, n_try, time.time() + alloc, rng)
    c_good = compute_c5(h_good, n_good)
    if c_opt < c_good:
        return h_opt, c_opt, n_try
    return h_good, c_good, n_good

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    best_h_global = None
    best_c_global = float('inf')
    best_n_global = 0
    
    try:
        h_init_raw = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init_raw)
    except Exception:
        h_init_raw = None
        n_init = 500
    
    # Phase 1: Optimize initial construction with symmetry
    if h_init_raw is not None:
        alloc1 = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_symmetric(h_init_raw, n_init, t0 + alloc1, rng)
        best_h_global = h_opt
        best_c_global = c_opt
        best_n_global = n_init
    else:
        h_start = np.full(n_init, 0.5)
        alloc1 = min(200, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_symmetric(h_start, n_init, t0 + alloc1, rng)
        best_h_global = h_opt
        best_c_global = c_opt
        best_n_global = n_init
    
    # Phase 2: Try to improve with different starting points at same resolution
    n_cur = best_n_global
    if time.time() < deadline - 100:
        # Symmetric perturbation
        hp = symmetrize(best_h_global + rng.normal(0, 0.01, n_cur))
        hp = project_feasible(hp, n_cur)
        alloc2 = min(120, (deadline - time.time()) * 0.15)
        if alloc2 > 30:
            h_opt, c_opt = optimize_symmetric(hp, n_cur, time.time() + alloc2, rng)
            if c_opt < best_c_global:
                best_c_global = c_opt
                best_h_global = h_opt
    
    # Phase 3: Try other resolutions
    for n_try in [600, 700, 400]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        x_good = np.linspace(0, 2, len(best_h_global))
        x_try = np.linspace(0, 2, n_try)
        h_start = np.interp(x_try, x_good, best_h_global)
        h_start = symmetrize(h_start)
        h_start = project_feasible(h_start, n_try)
        alloc3 = min(130, remaining * 0.22)
        if alloc3 < 40:
            break
        h_opt, c_opt = optimize_symmetric(h_start, n_try, time.time() + alloc3, rng)
        if c_opt < best_c_global:
            best_c_global = c_opt
            best_h_global = h_opt
            best_n_global = n_try
    
    # Phase 4: Multistart at best resolution
    n_best = len(best_h_global)
    if time.time() < deadline - 60:
        n_restarts = 0
        while time.time() < deadline - 40 and n_restarts < 5:
            scale = 0.015 * (1.0 + 0.5 * n_restarts)
            hp = best_h_global + rng.normal(0, scale, n_best)
            hp = symmetrize(hp)
            hp = project_feasible(hp, n_best)
            alloc4 = min(50, (deadline - time.time() - 30) * 0.25)
            if alloc4 < 25:
                break
            h_opt, c_opt = optimize_symmetric(hp, n_best, time.time() + alloc4, rng)
            if c_opt < best_c_global:
                best_c_global = c_opt
                best_h_global = h_opt
            n_restarts += 1
    
    # Phase 5: Final polish - SLSQP then local search
    if best_h_global is not None and time.time() < deadline - 10:
        n_f = len(best_h_global)
        # SLSQP polish
        if time.time() < deadline - 20:
            h_new = slsqp_polish(best_h_global, n_f, top_k=10, alpha=200000, maxiter=5000)
            c_new = compute_c5(h_new, n_f)
            if c_new < best_c_global:
                best_c_global = c_new
                best_h_global = h_new
        
        # Local search
        if time.time() < deadline - 5:
            h_new, c_new = local_search_sym(best_h_global, n_f, deadline - 1, max_rounds=300)
            if c_new < best_c_global:
                best_c_global = c_new
                best_h_global = h_new
    
    # Final validation
    if best_h_global is None:
        best_h_global = project_feasible(np.full(n_init, 0.5), n_init)
        best_c_global = compute_c5(best_h_global, n_init)
    
    best_h_global = symmetrize(best_h_global)
    best_h_global = project_feasible(best_h_global, len(best_h_global))
    best_c_global = compute_c5(best_h_global, len(best_h_global))
    
    return best_h_global, best_c_global, len(best_h_global)
