    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)

                break
    return best_h, best_c

def cyclic_subgrad(h, n, deadline, n_cycles=500):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eps = 0.02
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        top_k = np.argsort(-c_vec)[:5]
        for idx in top_k:
            g = shift_grad_single(h, n, idx)
            d = -g + np.mean(g)
            h_try = h + eps * d
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
            h = h_try
        if cyc % 50 == 0 and cyc > 0:
            eps *= 0.9
            if eps < 0.0005:
                eps = 0.0005
    return best_h, best_c

def local_search_pairs(h, n, deadline, radius=20, steps_seq=None):
    best_c = compute_c5(h, n)
    if steps_seq is None:
        steps_seq = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps_seq:
        if time.time() > deadline:
            break
        for sweep in range(5):
            if time.time() > deadline:
                break
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.005 * max_val)
            active_shifts = np.where(c_vec >= thresh)[0]
            if len(active_shifts) == 0:
                break
            grad = np.zeros(n)
            for idx in active_shifts[:10]:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active_shifts) > 0:
                grad /= min(len(active_shifts), 10)
            top_idx = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top_idx:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
            if not improved:
                break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1500):
    def slqp_obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh)))) / alpha

    def slqp_eq(x, n):
        return float(x.sum() - n / 2.0)

    def slqp_eq_jac(x, n):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_eq, 'jac': slqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n0 = 500

    h = project_feasible(h_init, n0)
    best_c = compute_c5(h, n0)
    best_h = h.copy()
    best_n = n0

    # Phase 1: L-BFGS on n0 (80s)
    t_phase = min(t0 + 80, deadline - 800)
    if time.time() < t_phase - 5:
        for alpha, mu in [(200, 50.0), (1000, 200.0), (5000, 1000.0), (20000, 5000.0)]:
            if time.time() > t_phase - 5:
                break
            h, c = lbfgs_optimize(h, n0, alpha, mu, t_phase, maxiter=2000)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 2: SQP on n0 (50s)
    t_phase = min(t0 + 130, deadline - 750)
    if time.time() < t_phase - 5:
        h, c = sqp_optimize(best_h, n0, t_phase, max_iter=100, rho_init=0.05, max_active=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Cyclic subgradient on n0 (30s)
    t_phase = min(t0 + 160, deadline - 720)
    if time.time() < t_phase - 3:
        h, c = cyclic_subgrad(best_h, n0, t_phase, n_cycles=200)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Upsample to n=1000 and optimize (main phase, 500s)
    n1 = 1000
    t_phase_start = time.time()
    t_phase_end = min(t0 + 660, deadline - 200)
    if t_phase_end > t_phase_start + 20:
        h1 = upsample(best_h, best_n, n1)
        h1 = project_feasible(h1, n1)
        c1 = compute_c5(h1, n1)

        # L-BFGS rounds on n=1000 with increasing alpha
        lbfgs_configs = [
            (500, 100.0, 2000),
            (2000, 500.0, 2000),
            (8000, 2000.0, 1500),
            (30000, 5000.0, 1500),
            (80000, 10000.0, 1000),
        ]
        for alpha, mu, mit in lbfgs_configs:
            if time.time() > t_phase_end - 60:
                break
            h1, c1 = lbfgs_optimize(h1, n1, alpha, mu, t_phase_end - 60, maxiter=mit)
            if c1 < best_c:
                best_c = c1
                best_h = h1.copy()
                best_n = n1

        # SQP on n=1000
        if time.time() < t_phase_end - 60:
            h1, c1 = sqp_optimize(h1, n1, t_phase_end - 30, max_iter=100, rho_init=0.04, max_active=20)
            if c1 < best_c:
                best_c = c1
                best_h = h1.copy()
                best_n = n1

        # Cyclic subgradient on n=1000
        if time.time() < t_phase_end - 10:
            h1, c1 = cyclic_subgrad(h1, n1, t_phase_end - 5, n_cycles=400)
            if c1 < best_c:
                best_c = c1
                best_h = h1.copy()
                best_n = n1

        # Local search on n=1000
        if time.time() < t_phase_end - 5:
            h1, c1 = local_search_pairs(h1, n1, t_phase_end - 2, radius=15,
                                        steps_seq=[0.004, 0.002, 0.001, 0.0005, 0.0002])
            if c1 < best_c:
                best_c = c1
                best_h = h1.copy()
                best_n = n1

    # Phase 5: Try n=2000 if time permits (200s)
    n2 = 2000
    t_phase_end2 = min(t0 + 900, deadline - 60)
    if time.time() < t_phase_end2 - 30:
        h2 = upsample(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)

        for alpha, mu, mit in [(1000, 200.0, 1000), (5000, 1000.0, 1000), (20000, 5000.0, 800)]:
            if time.time() > t_phase_end2 - 20:
                break
            h2, c2 = lbfgs_optimize(h2, n2, alpha, mu, t_phase_end2 - 20, maxiter=mit)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n2

        if time.time() < t_phase_end2 - 15:
            h2, c2 = sqp_optimize(h2, n2, t_phase_end2 - 10, max_iter=50, rho_init=0.03, max_active=15)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n2

        if time.time() < t_phase_end2 - 5:
            h2, c2 = cyclic_subgrad(h2, n2, t_phase_end2 - 3, n_cycles=200)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n2

    # Phase 6: Final SQP + polish on best solution
    if time.time() < deadline - 20:
        hf = project_feasible(best_h, best_n)
        cf = compute_c5(hf, best_n)
        hf, cf = sqp_optimize(hf, best_n, deadline - 10, max_iter=80, rho_init=0.03, max_active=20)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    if time.time() < deadline - 10:
        hf, cf = slqp_polish(best_h, best_n, deadline - 3, alpha=1000000, maxiter=1000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
