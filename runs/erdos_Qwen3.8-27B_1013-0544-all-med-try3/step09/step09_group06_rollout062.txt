import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def grad_lag(h, n, lag):
    """Exact gradient of c[lag] w.r.t. h, where lag is the integer shift."""
    n_pts = n
    dx = 2.0 / n_pts
    grad = np.zeros(n_pts)
    m = abs(lag)
    if m >= n_pts:
        return grad
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[m:] += (1.0 - h[:-m]) * dx
        grad[:n_pts-m] += -h[m:] * dx
    else:
        grad[:n_pts-m] += (1.0 - h[m:]) * dx
        grad[m:] += -h[:-m] * dx
    return grad

def subgradient_descent(h, n, deadline, max_iter=50000, initial_lr=0.5):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = initial_lr
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        g = grad_lag(h_curr, n, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Backtracking line search
        alpha = lr
        found = False
        for _ in range(30):
            h_new = h_curr - alpha * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15 or c_new < compute_c5(h_curr, n) - 0.5 * alpha * g_norm**2:
                found = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not found:
            h_new = project_feasible(h_curr - alpha * g, n)
            c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
            lr = min(lr * 1.1, 5.0)
        else:
            patience += 1
            if patience % 50 == 0:
                lr *= 0.8
        
        h_curr = h_new
        if patience > 500:
            break
    
    return best_h, best_c

            # Mask: valid moves

            d = min(0.01, h2[best_i] * 0.3, (1-h1[best_i]) * 0.3)
            
            # Make the move: transfer from h2[best_i] to h1[best_i]
            if lag > 0:
                h[m + best_i] -= d
                h[best_i] += d
            else:
                h[best_i] -= d
                h[m + best_i] += d
            
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)

                best_h = h.copy()

            h = best_h + 0.01 * np.random.randn(n)
            h = project_feasible(h, n)
            stagnant = 0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

def grad_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[m:] += (1.0 - h[:n-m]) * dx
        grad[:n-m] += -h[m:] * dx
    else:
        grad[:n-m] += (1.0 - h[m:]) * dx
        grad[m:] += -h[:n-m] * dx
    return grad

def nesterov_subgrad(h, n, deadline, max_iter=200000, base_lr=0.8):
    h_curr = h.copy()
    h_y = h_curr.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = base_lr
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_y, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        g = grad_lag(h_y, n, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        beta = (1.0 + np.sqrt(1 + 4 * it * it)) / 2.0
        h_prev = h_y.copy()
        h_y = h_curr + (it / (it + beta)) * (h_curr - h_y) if it > 0 else h_curr.copy()
        h_new = h_y - lr / (1.0 + 0.01 * it) * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stagnant = 0
            lr = min(lr * 1.02, 4.0)
        else:
            stagnant += 1
            if stagnant % 40 == 0:
                lr *= 0.88
        h_curr = h_new
        if stagnant > 3000:
            h_curr = best_h.copy()
            h_y = best_h.copy()
            stagnant = 0
            if it > 50000:
                break
    return best_h, best_c

def smooth_grad_ascent(h, n, deadline, alpha=5000, max_iter=50000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.3
    v = np.zeros(n)
    stagnant = 0

        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        m = c_all.max()
        shifted = alpha * (c_all - m)

        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            grad += w[k] * grad_lag(h_curr, n, lag)
        v = 0.9 * v + lr * grad
        h_new = project_feasible(h_curr - v, n)
        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
            stagnant = 0
        else:

                v *= 0.5

            h_curr = best_h.copy()
            v = np.zeros(n)
            stagnant = 0
            if it > 30000:

    return best_h, best_c

def fine_local_search(h, n, deadline, max_iter=500000):

def slsqp_polish(h, n, deadline, top_m=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]

    def obj(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        m = active.max()
        shifted = 200000.0 * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / 200000.0)
        return float(m + np.log(np.sum(np.exp(shifted))) / 200000.0)

    def obj_grad(x):
        cc = compute_all_lags(x, n)
        active = cc[top_idx]
        m = active.max()
        shifted = 200000.0 * (active - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for j, idx in enumerate(top_idx):
            if w[j] < 1e-8:
                continue
            lag = int(idx) - (n - 1)
            grad += w[j] * grad_lag(x, n, lag)
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        x = project_feasible(res.x, n)
        return x
    except Exception:
        return h.copy()

def try_resolution(h_best, n_best, n_try, deadline, rng):
    x_best = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
    x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
    h_start = np.interp(x_new, x_best, h_best)
    h_start = project_feasible(h_start, n_try)
    h_opt, c_opt = nesterov_subgrad(h_start, n_try, deadline, max_iter=50000, base_lr=0.5)
    return h_opt, c_opt
