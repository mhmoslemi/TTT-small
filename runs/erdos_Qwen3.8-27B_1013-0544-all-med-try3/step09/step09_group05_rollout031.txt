import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # Find lambda such that sum(clamp(h - lambda, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - result
            tot = room.sum()
            if tot > 1e-15:
                result += (diff / tot) * room
        else:
            tot = result.sum()
            if tot > 1e-15:
                result += (diff / tot) * result
    return np.clip(result, 0.0, 1.0)

def compute_shift_grad(h, n, shift_idx):
    """Compute gradient of c_k at h, where shift_idx is the index in the full correlation (0 to 2n-2)."""
    dx = 2.0 / n
    k = shift_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        
        c_vec = compute_c5_vec(h_curr, n)
        max_val = c_vec.max()
        
        if max_val < best_c - 1e-12:
            best_c = max_val
            best_h = h_curr.copy()
        
        # Identify active set (shifts within tol of max)
        threshold = max_val - max(tol, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        #             sum(h) = n/2
        #             0 <= h <= 1
        #             t >= 0
        
        # For simplicity, just minimize t (no L1 penalty)
        n_vars = n + 1 + 2 * n  # h, t, s_plus, s_minus (for L1)
        # Actually, let's skip the L1 penalty for now and just do min t
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        
        for idx, s_idx in enumerate(active):

            v = c_vec[s_idx]
            A_ub[idx, :n] = g
            A_ub[idx, n] = -1.0
            b_ub[idx] = -v + np.dot(g, h_curr)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_lp = res.x[:n]
        t_lp = res.x[n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_try = alpha * h_lp + (1.0 - alpha) * h_curr
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)

                continue
            else:
                break

                improved = True
                break
        
        if not improved:
            # Try moving toward LP solution partially even if not better than best
            # (to escape local plateau)
            if stagnant > 20:
                ht = 0.1 * h_lp + 0.9 * hc
                ht = project_feasible(ht, n)
                ct = compute_c5(ht, n)
                if ct <= best_c + 1e-10:
                    hc = ht.copy()
                else:
                    break
            else:
                break
    
    return best_h, best_c

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        g += w[mm] * shift_grad(h, n, mm)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def top_k_smooth_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def top_k_smooth_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n)
    for wi, si in zip(w, active):
        if wi < 1e-12:
            continue
        g += wi * shift_grad(h, n, si)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_smooth(h, n, deadline, alphas, mu=100.0, maxiter=8000):
    hc = h.copy()
    best_c = compute_c5(hc, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, hc, jac=smooth_max_grad,
                         args=(n, a, mu), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-18,
                                  'maxfun': maxiter * 3, 'maxls': 40})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                hc = hn
        except Exception:
            pass
    return hc, best_c

def lbfgs_topk(h, n, deadline, alphas, mu=100.0, top_k=5):
    hc = h.copy()
    best_c = compute_c5(hc, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(hc, n)
        mv = c_vec.max()
        thr = mv - max(1e-6, 0.001 * mv)
        active = np.where(c_vec >= thr)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        try:
            res = minimize(top_k_smooth_obj, hc, jac=top_k_smooth_grad,
                         args=(n, active, a, mu), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-18,
                                  'maxfun': 15000, 'maxls': 40})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                hc = hn
        except Exception:
            pass
    return hc, best_c

def pair_search(h, n, deadline, rng_seed=42):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    hc = h.copy()
    rng = np.random.RandomState(rng_seed)
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    budget_per_step = max(0.5, (deadline - time.time()) / len(steps))
    
    for step in steps:
        if time.time() > deadline - 0.5:
            break
        step_deadline = min(deadline - 0.5, time.time() + budget_per_step * 2)
        
        c_vec = compute_c5_vec(hc, n)
        mv = c_vec.max()
        thr = mv - max(1e-6, 0.002 * mv)
        act = np.where(c_vec >= thr)[0]
        grads = np.array([shift_grad(hc, n, s) for s in act]) if len(act) > 0 else np.zeros((1, n))
        
        while time.time() < step_deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            for direction in [1, -1]:
                di = direction * step
                dj = -direction * step
                if hc[i] + di < -1e-12 or hc[i] + di > 1 + 1e-12:
                    continue
                if hc[j] + dj < -1e-12 or hc[j] + dj > 1 + 1e-12:
                    continue
                
                # Quick gradient-based check
                if len(act) > 0 and len(grads) > 0:
                    delta_max = step * np.max(grads[:, i] - grads[:, j])
                    if direction == -1:
                        delta_max = step * np.max(grads[:, j] - grads[:, i])
                    if mv + delta_max > mv + 1e-9:
                        continue
                
                hc[i] = np.clip(hc[i] + di, 0, 1)
                hc[j] = np.clip(hc[j] + dj, 0, 1)
                c = compute_c5(hc, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = hc.copy()
                    c_vec = compute_c5_vec(hc, n)
                    mv = c_vec.max()
                    thr = mv - max(1e-6, 0.002 * mv)
                    act = np.where(c_vec >= thr)[0]
                    grads = np.array([shift_grad(hc, n, s) for s in act]) if len(act) > 0 else np.zeros((1, n))
                else:
                    hc[i] -= di
                    hc[j] -= dj
    
    return best_h, best_c

def coordinated_pair_search(h, n, deadline):
    """Sweep all pairs systematically near the current solution."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    hc = h.copy()
    
    c_vec = compute_c5_vec(hc, n)
    mv = c_vec.max()
    
    # Focus on pairs where gradient difference is large
    steps = [0.002, 0.001, 0.0005, 0.0002]
    budget_per = max(1.0, (deadline - time.time()) / (len(steps) * 2))
    
    for step in steps:
        if time.time() > deadline - 1:
            break
        sub_deadline = min(deadline - 1.0, time.time() + budget_per)
        
        c_vec = compute_c5_vec(hc, n)
        mv = c_vec.max()
        thr = mv - max(1e-6, 0.001 * mv)
        act = np.where(c_vec >= thr)[0]
        
        if len(act) == 0:
            break
        
        # Compute gradient for combined active set
        combined_grad = np.zeros(n)
        for si in act:
            combined_grad += shift_grad(hc, n, si)
        combined_grad /= len(act)
        
        # Try moving mass from high-grad to low-grad positions
        indices = np.argsort(combined_grad)
        
        n_try = 0
        for i_pos in range(0, n, max(1, n // 200)):
            if time.time() > sub_deadline:
                break
            for j_pos in range(0, n, max(1, n // 200)):
                if i_pos == j_pos:
                    continue
                n_try += 1
                i = indices[i_pos]
                j = indices[j_pos]
                
                # Move from high grad to low grad (decrease high, increase low)
                di = -step
                dj = step
                if hc[i] + di >= 0 and hc[j] + dj <= 1:
                    hc[i] += di
                    hc[j] += dj
                    c = compute_c5(hc, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = hc.copy()
                    else:
                        hc[i] -= di
                        hc[j] -= dj
    
    return best_h, best_c

def multi_restart(h, n, deadline, n_restarts=3, perturbation_scale=0.01):
    """Try small random perturbations and optimize each."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    budget_per = (deadline - time.time()) / (n_restarts + 1)
    
    for r in range(n_restarts):
        if time.time() > deadline - budget_per:
            break
        # Perturb
        noise = rng.randn(n) * perturbation_scale
        hp = project_feasible(best_h + noise, n)
        cp = compute_c5(hp, n)
        
        # Quick level method on perturbed solution
        sub_deadline = min(deadline - 2, time.time() + budget_per * 0.8)
        hr, cr = level_method(hp, n, sub_deadline, max_iter=500)
        if cr < best_c:
            best_c = cr
            best_h = hr
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: Level method (main workhorse) - 120s
    t1 = min(t0 + 120, deadline - 600)
    best_h, best_c = level_method(best_h, n0, t1, max_iter=10000)
    
    # Phase 2: L-BFGS top-k polish - 80s
    if time.time() < deadline - 500:
        t2 = min(time.time() + 80, deadline - 500)
        best_h, best_c = lbfgs_topk(best_h, n0, t2,
                                     alphas=[5000, 50000, 500000, 2000000], mu=500.0, top_k=5)
    
    # Phase 3: Level method again - 100s
    if time.time() < deadline - 400:
        t3 = min(time.time() + 100, deadline - 400)
        best_h, best_c = level_method(best_h, n0, t3, max_iter=10000)
    
    # Phase 4: L-BFGS full smooth - 60s
    if time.time() < deadline - 320:
        t4 = min(time.time() + 60, deadline - 320)
        best_h, best_c = lbfgs_smooth(best_h, n0, t4,
                                       alphas=[100000, 500000, 2000000, 10000000], mu=5000.0)
    
    # Phase 5: Level method - 80s
    if time.time() < deadline - 250:
        t5 = min(time.time() + 80, deadline - 250)
        best_h, best_c = level_method(best_h, n0, t5, max_iter=10000)
    
    # Phase 6: Coordinated pair search - 60s
    if time.time() < deadline - 180:
        t6 = min(time.time() + 60, deadline - 180)
        best_h, best_c = coordinated_pair_search(best_h, n0, t6)
    
    # Phase 7: Multi-restart with perturbation - 60s
    if time.time() < deadline - 120:
        t7 = min(time.time() + 60, deadline - 120)
        best_h, best_c = multi_restart(best_h, n0, t7, n_restarts=3, perturbation_scale=0.005)
    
    # Phase 8: Level method on best - 60s
    if time.time() < deadline - 60:
        t8 = min(time.time() + 60, deadline - 60)
        best_h, best_c = level_method(best_h, n0, t8, max_iter=10000)
    
    # Phase 9: Random pair search - 40s
    if time.time() < deadline - 25:
        t9 = min(time.time() + 40, deadline - 25)
        best_h, best_c = pair_search(best_h, n0, t9, rng_seed=seed)
    
    # Phase 10: Final level method - remaining time
    if time.time() < deadline - 5:
        t10 = min(time.time() + 20, deadline - 3)
        best_h, best_c = level_method(best_h, n0, t10, max_iter=5000)
    
    # Final: quick L-BFGS if time allows
    if time.time() < deadline - 2:
        best_h, best_c = lbfgs_topk(best_h, n0, deadline - 1,
                                     alphas=[10000000, 50000000], mu=10000.0, top_k=3)
    
    best_h = project_feasible(best_h, n0)
    best_c = compute_c5(best_h, n0)
    
    return best_h, best_c, n0
