```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (300, 200.0), (1000, 1000.0),
                   (3000, 5000.0), (10000, 20000.0), (30000, 100000.0)]
    for alpha, mu in configs:

            break
        try:

                          args=(n, alpha, mu), method='L-BFGS-B',

                                   'gtol': 1e-14, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=100000, maxiter=2000):

    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
```

Actually, I just realized: for the 1D problem, I can use the fact that the optimal x* is where:
- If μ = 0: x* is at a "knee" of the max-of-linear function
- If μ > 0: x* is slightly shifted toward n/2 - S_other

A very fast approximation:
1. Compute the "unpenalized optimum" x0: the x that minimizes max_k (b_k + a_k*x)
   - This is at the smallest x where the two "critical" lines have equal value
   - Can be found by: sort by slope a_k, then scan
   - Or: just evaluate at x=0 and x=1, and bisect

2. Apply penalty correction: x* ≈ x0 - (x0 - (n/2 - S_other)) / (1 + 2μ * range_of_x)

```python
def solve_1d(a, b, S_other, mu, n):
    """Solve min_{x in [0,1]} max_k (b[k] + a[k]*x) + mu*(S_other + x - n/2)^2"""
    target = n / 2.0
    
    def f(x):
        return np.max(b + a * x) + mu * (S_other + x - target)**2
    
    # Bisection on the subgradient
    # f'(x) = a[k*] + 2*mu*(S_other + x - target) where k* = argmax
    # At optimum: a[k*] + 2*mu*(S_other + x* - target) = 0 (if interior)
    # x* = target - S_other - a[k*]/(2*mu)
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        k_star = np.argmax(b + a * mid)
        grad = a[k_star] + 2 * mu * (S_other + mid - target)
        # Subgradient: any convex combination of a[k] + 2mu*(...) for k in argmax
        # For bisection, just use the one at mid
        if grad > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (300, 200.0), (1000, 1000.0),
                   (3000, 5000.0), (10000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 80000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_1d_solve(a, b, S_other, mu, target):
    """Solve min_{x in [0,1]} max_k (b[k]+a[k]*x) + mu*(S_other+x-target)^2 via bisection."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = b + a * mid
        k_star = np.argmax(vals)
        grad = a[k_star] + 2.0 * mu * (S_other + mid - target)
        if grad > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coordinate_descent(h, n, deadline, max_passes=500, mu_start=100.0, mu_end=50000.0):
    """Coordinate descent exploiting the 1D convex structure of each update."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    target = n / 2.0
    n2m1 = 2 * n - 1
    rng = np.random.RandomState(777)
    
    for p in range(max_passes):
        if time.time() > deadline - 30:
            break
        # Adaptive penalty
        frac = p / max(1, max_passes)
        mu = mu_start * (mu_end / mu_start) ** frac
        
        # Compute current correlation vector (for fast b_k computation)
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Random coordinate order
        order = rng.permutation(n)
        
        for i in order:
            if time.time() > deadline - 30:
                break
            # Compute a_k for all shifts k (index 0..n2m1-1 corresponds to shift -(n-1)..n-1)
            # Shift for index idx is k = idx - (n-1)
            # a_k = (1-h[i+k])*I(0<=i+k<n) - h[i-k]*I(0<=i-k<n)
            a = np.zeros(n2m1)
            # Term 1: (1-h[i+k]) for 0 <= i+k < n => k in [-i, n-i-1] => idx in [n-1-i, n-1+n-i-1]
            lo1 = max(0, n - 1 - i)
            hi1 = min(n2m1, n - 1 + n - i)
            if hi1 > lo1:
                # k ranges from -(i) to n-1-i
                # i+k ranges from 0 to n-1
                idxs = np.arange(lo1, hi1)
                ik_vals = i + (idxs - (n - 1))  # = i+k
                a[lo1:hi1] += (1.0 - h[ik_vals.astype(int)])
            
            # Term 2: -h[i-k] for 0 <= i-k < n => k in [i-n+1, i] => idx in [n-1+i-n+1, n-1+i]
            lo2 = max(0, n - 1 + i - n + 1)  # = max(0, i)
            hi2 = min(n2m1, n - 1 + i + 1)  # = min(2n-1, n+i)
            if hi2 > lo2:
                # k ranges from i-n+1 to i
                # i-k ranges from n-1 down to 0... let me redo
                # idx from lo2 to hi2-1, k = idx - (n-1)
                idxs2 = np.arange(lo2, hi2)
                ik2_vals = i - (idxs2 - (n - 1))  # = i-k
                a[lo2:hi2] -= h[ik2_vals.astype(int)]
            
            # b_k = corr[idx] - a[idx] * h[i] (since corr = sum of all terms including h[i])
            xi = h[i]
            S_other = h.sum() - xi
            b = corr - a * xi
            
            # Solve 1D subproblem
            x_new = coord_1d_solve(a, b, S_other, mu, target)
            h[i] = x_new
        
        # Periodic projection
        if p % 20 == 19:
            h = project_feasible(h, n)
            c_cur = compute_c5(h, n)
            if c_cur < best_c:
                best_c = c_cur
    
    h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
    return h, best_c

def binary_local_search(h, n, deadline, max_rounds=5000):
    """Binary local search: swap 1s and 0s to reduce max overlap."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    # Ensure exact count
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    while h_b.sum() > target:
        idx = ones[rng.randint(len(ones))] if 'rng' in dir() else ones[-1]
        h_b[idx] = 0.0
        ones = np.delete(ones, np.where(ones == idx)[0])
    while h_b.sum() < target:
        idx = zeros[-1]
        h_b[idx] = 1.0
        zeros = np.delete(zeros, np.where(zeros == idx)[0])
    
    best_c = compute_c5(h_b, n)
    best_h = h_b.copy()
    rng = np.random.RandomState(999)
    
    for rd in range(max_rounds):
        if time.time() > deadline - 30:
            break
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        n_trials = min(300, len(ones) * len(zeros))
        for _ in range(n_trials):
            if time.time() > deadline - 30:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_b[a] = 0.0
            h_b[b] = 1.0
            c = compute_c5(h_b, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h_b.copy()
                improved = True
            else:
                h_b[b] = 0.0
                h_b[a] = 1.0
        if not improved and rd > 200:
            break
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
```
