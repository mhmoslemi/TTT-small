            best_h = h_new
            eta = min(eta * 1.02, 0.05)
            stale = 0
        else:
            stale += 1
            if stale > 50:
                eta *= 0.95
                stale = 0
            if eta < 1e-6:
                break
    return best_h, best_c

def coordinate_refine(h, n, deadline, n_passes=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for sp in step_sizes:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if sp > 0:
                    if best_h[i] < 1.0 - sp:
                        h_t = best_h.copy()
                        h_t[i] += sp
                        h_t[i - 1 if i > 0 else i + 1] -= sp
                        h_t = project_feasible(h_t, n)
                        c = compute_c5(h_t, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h_t
                            progress = True
                    if best_h[i] > sp:
                        h_t = best_h.copy()
                        h_t[i] -= sp
                        h_t[i - 1 if i > 0 else i + 1] += sp
                        h_t = project_feasible(h_t, n)
                        c = compute_c5(h_t, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h_t
                            progress = True
    return best_h, best_c

def run_erdos(h, n):
    return compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution
    p1_end = min(t0 + 40, deadline - 500)
    if p1_end > t0 + 5:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[100, 1000, 10000, 100000],
            mu=0.01, maxiters=[500, 800, 1000, 1500])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to n=2000 and optimize aggressively
    n2 = 2000
    if time.time() < deadline - 400:
        h2 = upsample(best_h, best_n, n2)
        c2_chk = compute_c5(h2, n2)
        if c2_chk < best_c + 0.005:
            p2_end = min(time.time() + 180, deadline - 250)
            h2, c2 = run_lbfgs(h2, n2, p2_end,
                alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
                mu=0.005, maxiters=[400, 600, 800, 1000, 1200, 1500, 2000, 2500, 3000])
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n2

    # Phase 3: Active set at n=2000
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 120, deadline - 150)
        h3, c3 = run_active_opt(best_h, best_n, p3_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Subgradient polish
    if time.time() < deadline - 100:
        p4_end = min(time.time() + 80, deadline - 60)
        h4, c4 = subgrad_polish(best_h, best_n, p4_end, n_iter=100000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try higher resolution n=3000
    n3 = 3000
    if time.time() < deadline - 150 and best_n < n3:
        h3b = upsample(best_h, best_n, n3)
        c3b_chk = compute_c5(h3b, n3)
        if c3b_chk < best_c + 0.003:
            p5_end = min(time.time() + 100, deadline - 80)
            h3b, c3b = run_lbfgs(h3b, n3, p5_end,
                alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
                mu=0.003, maxiters=[300, 500, 700, 900, 1000, 1200, 1500, 2000])
            if c3b < best_c:
                best_c = c3b
                best_h = h3b
                best_n = n3

    # Phase 6: Active set at final resolution
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 60, deadline - 40)
        h6, c6 = run_active_opt(best_h, best_n, p6_end, top_k=4,
            alphas=[500000, 2000000, 10000000, 50000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: More subgradient
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 30, deadline - 20)
        h7, c7 = subgrad_polish(best_h, best_n, p7_end, n_iter=100000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final active set push
    if time.time() < deadline - 15:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        bounds = [(0.0, 1.0)] * best_n
        for a in [10000000, 100000000]:
            if time.time() > deadline - 10:
                break
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(best_n, active, a, 0.0005),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-22,
                                        'gtol': 1e-16, 'maxfun': 2000})
                h_final = project_feasible(res.x, best_n)
                c_final = compute_c5(h_final, best_n)
                if c_final < best_c:
                    best_c = c_final
                    best_h = h_final
            except Exception:
                pass

    # Final: one more subgradient micro-polish
    if time.time() < deadline - 2:
        h_final, c_final = subgrad_polish(best_h, best_n, deadline - 1, n_iter=100000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
