```python
    v = np.zeros(n)

    best_c = compute_c5(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx2 = -lag
            grad[0:n - idx2] += w[k] * (1.0 - h[idx2:n])
            grad[idx2:n] += w[k] * (-h[0:n - idx2])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0
    penalty_coeff = 1e8

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def topk_gradient(h, n, k=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_k = np.argsort(c)[-k:]
    c_top = c[top_k]
    shifted = c_top - c_top.max()
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for idx in range(k):
        ki = top_k[idx]
        lag = ki - (n - 1)
        wdx = w[idx] * dx
        if lag == 0:
            grad += (1.0 - 2.0 * h) * wdx
        elif lag > 0:
            grad[0:n-lag] += (1.0 - h[lag:n]) * wdx
            grad[lag:n] += (-h[0:n-lag]) * wdx
        else:
            m = -lag
            grad[0:n-m] += (-h[m:n]) * wdx
            grad[m:n] += (1.0 - h[0:n-m]) * wdx
    return grad

def nesterov_descent(h, n, deadline, k=5, lr_init=0.003, momentum=0.91):
    h = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    t_count = 0
    no_improve = 0

    while time.time() < deadline:
        t_count += 1
        t_mom = (1.0 + t_count * 0.001) / (1.0 + t_count * 0.001 + momentum)
        y = h + t_mom * (h - y) if t_count > 1 else h.copy()
        grad = topk_gradient(y, n, k)
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.8
                if lr < 1e-7:
                    break
                h = best_h + np.random.RandomState(t_count).normal(0, 0.005, n)
                h = project_feasible(h, n)
                y = h.copy()
                no_improve = 0
        h = h_new
    return best_h, best_c

                stagnant = 0
            else:
                h = best_h.copy()
                stagnant += 1
                if stagnant > 2000:
                    h = best_h + rng.normal(0, 0.015, n)
                    h = project_feasible(h, n)
                    stagnant = 0

    return best_h, best_c

def multi_lag_local_search(h, n, deadline, rng):
    """Try to reduce multiple top lags simultaneously using coordinate descent."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-8:]
        improved = False

        for k_idx in top_k:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            if lag > 0:
                misalign = h[0:n-m] * (1.0 - h[m:n])
            else:
                misalign = h[m:n] * (1.0 - h[0:n-m])

            # Try top 10 most misaligned pairs
            top_pairs = np.argsort(misalign)[-10:]
            for pi in top_pairs:
                if time.time() > deadline:
                    break
                i = pi
                j = pi + m
                if lag > 0:
                    d = min(0.02, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                    if d < 1e-9:
                        continue
                    h[i] -= d
                    h[j] += d
                else:
                    d = min(0.02, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                    if d < 1e-9:
                        continue
                    h[j] -= d
                    h[i] += d

                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()

        if not improved:
            stagnant += 1
            if stagnant > 300:
                h = best_h + rng.normal(0, 0.01, n)
                h = project_feasible(h, n)
                stagnant = 0
            elif stagnant > 2000:
                break
        else:
            stagnant = 0

    return best_h, best_c

def interpolate_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: L-BFGS-B with progressive alpha at initial resolution
    alpha_schedule = [(200, 400), (2000, 500), (20000, 500), (200000, 600)]
    for alpha, maxit in alpha_schedule:
        if time.time() > deadline - 400:
            break
        alloc = min(45, (deadline - 400 - t0) * 0.08)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t0 + time.time() - t0 + alloc, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Nesterov gradient descent with top-k gradient
    if time.time() < deadline - 300:
        alloc = min(50, (deadline - 300 - t0) * 0.1)
        h_opt, c_opt = nesterov_descent(best_h, n, t0 + time.time() - t0 + alloc, k=5, lr_init=0.003, momentum=0.92)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Targeted local search at initial resolution
    if time.time() < deadline - 250:
        alloc = min(80, (deadline - 250 - t0) * 0.12)
        h_opt, c_opt = targeted_local_search(best_h, n, t0 + time.time() - t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish
    if time.time() < deadline - 200:
        alloc = min(30, (deadline - 200 - t0) * 0.08)
        h_opt = lbfgsb_optimize(best_h, n, 1000000, t0 + time.time() - t0 + alloc, maxiter=1000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Upscale to higher resolution and optimize
    resolutions_to_try = [750, 1000, 1250]
    for n_try in resolutions_to_try:
        if time.time() > deadline - 120:
            break
        remaining = deadline - 60 - time.time()
        if remaining < 50:
            break

        h_interp = interpolate_resolution(best_h, n, n_try)
        # L-BFGS-B at new resolution
        alloc1 = min(25, remaining * 0.3)
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, t0 + time.time() - t0 + alloc1, maxiter=400)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try

        # Targeted local search at new resolution
        if time.time() < deadline - 60:
            alloc2 = min(40, (deadline - 60 - t0) * 0.15)
            h_opt2, c_opt2 = targeted_local_search(h_opt, n_try, t0 + time.time() - t0 + alloc2, rng)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2
                n = n_try

        # Quick L-BFGS-B polish at new resolution
        if time.time() < deadline - 40:
            alloc3 = min(15, (deadline - 40 - t0) * 0.1)
            h_opt3 = lbfgsb_optimize(best_h, n, 500000, t0 + time.time() - t0 + alloc3, maxiter=500)
            c_opt3 = compute_c5(h_opt3, n)
            if c_opt3 < best_c:
                best_c = c_opt3
                best_h = h_opt3

    # Phase 6: Multi-lag local search
    if time.time() < deadline - 50:
        alloc = min(60, (deadline - 50 - t0) * 0.2)
        h_opt, c_opt = multi_lag_local_search(best_h, n, t0 + time.time() - t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Another round of targeted local search with more exploration
    if time.time() < deadline - 30:
        alloc = min(40, (deadline - 30 - t0) * 0.2)
        h_opt, c_opt = targeted_local_search(best_h, n, t0 + time.time() - t0 + alloc, rng,
                                              top_lag_count=8, exploration_prob=0.4)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        h_opt = lbfgsb_optimize(best_h, n, 5000000, deadline - 3, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final local search
    if time.time() < deadline - 3:
        h_opt, c_opt = targeted_local_search(best_h, n, deadline - 1, rng,
                                              top_lag_count=5, exploration_prob=0.15)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
