from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_val_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    ac = alpha * c
    am = ac.max()
    exp_terms = np.exp(ac - am)
    log_sum = am + np.log(exp_terms.sum())
    val = log_sum / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    
    w = exp_terms / exp_terms.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
            if k > 0:
                grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
            if kk > 0:
                grad[:nv] += wm * (-h[kk:])
    
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    
    return float(val), grad

def sqp_step(h, n, active_idx, rho):
    """Solve SQP subproblem as LP."""
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    
    dx = 2.0 / n
    for idx in active_idx:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] = (1.0 - h[k:]) * dx
            if k > 0:
                g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] = (1.0 - h[:nv]) * dx
            if kk > 0:
                g[:nv] += (-h[kk:]) * dx
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    # 0 <= h + delta
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = -h[i]
        row += 1
    # h + delta <= 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    # |delta| <= rho
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 3.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def lbfgs_optimize(h, n, deadline, alphas, mu=0.001, max_iter_each=3000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def obj(x):
            v, _ = smooth_val_grad(x, n, alpha, mu)
            return v
        def grad(x):
            _, g = smooth_val_grad(x, n, alpha, mu)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter_each, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iter_each * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception as e:
            pass
    
    return h, best_c

def upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

np.correlate(h, 1-h, mode='full') has length 2n-1.
Element m (0-indexed) = sum over j of h[j + m - (n-1)] * (1-h[j])

where the sum is over all j where both indices are valid.

The box constraints and step constraints are just bounds on the delta variables, which HiGHS handles efficiently via bounds. The sum constraint is one equation.      |delta_i| ≤ rho for all i (bounds, tighter)

With HiGHS, bounds are handled very efficiently. The only general constraint matrix is n_active × (n+1). For n_active=30 and n=2000: 30×2001 = 60K elements. That's tiny!

```python
def sqp_step(h, n, active_idx, rho):
    n_var = n + 1  # delta[0:n] + t
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    
    # Inequality: g_k^T delta - t <= -c_k for each active k
    A_ub = np.zeros((n_act, n_var))
    b_ub = np.zeros(n_act)
    
    dx = 2.0 / n
    for row, idx in enumerate(active_idx):
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            A_ub[row, :nv] += (1.0 - h[k:]) * dx
            if k > 0:
                A_ub[row, k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            A_ub[row, kk:] += (1.0 - h[:nv]) * dx
            if kk > 0:
                A_ub[row, :nv] += (-h[kk:]) * dx
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: max(-h[i], -rho) <= delta[i] <= min(1-h[i], rho)
    lo = np.maximum(-h, -rho)
    hi = np.minimum(1.0 - h, rho)
    bounds = list(zip(lo, hi)) + [(0.0, 3.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_val_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    ac = alpha * c
    am = ac.max()
    exp_terms = np.exp(ac - am)
    log_sum = am + np.log(exp_terms.sum())
    val = log_sum / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    w = exp_terms / exp_terms.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
            if k > 0:
                grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
            if kk > 0:
                grad[:nv] += wm * (-h[kk:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return float(val), grad

def _obj_wrap(x, n, alpha, mu):
    v, _ = smooth_val_grad(x, n, alpha, mu)
    return v

def _grad_wrap(x, n, alpha, mu):
    _, g = smooth_val_grad(x, n, alpha, mu)
    return g

def shift_gradient(h, n, idx):
