            if rho < 1e-11:
                break
            if consecutive_no_improve > 30:
                break
    return best_h, best_c

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 1000.0), (5000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 20000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgradient_step(h, n, lr):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    idx = int(np.argmax(c_vec))
    g = shift_grad_single(h, n, idx)
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_optimize(h, n, deadline, lr_init=0.01, decay=0.999):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    patience = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        h_new = subgradient_step(h, n, lr)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience > 200:
                lr *= 0.9
                patience = 0
                if lr < 1e-8:
                    break
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=30, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            top_pos = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    else:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def make_sinusoidal(n, freq=1.0, phase=0.0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def make_block_pattern(n, period=4):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 0.8 if (i % period) < period // 2 else 0.2
    return project_feasible(h, n)

def make_random_smooth(n, seed_val, smoothness=3):
    rng = np.random.RandomState(seed_val)
    h = rng.rand(n)
    for _ in range(smoothness):
        h = np.convolve(h, np.ones(5)/5.0, mode='same')
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def intensive_optimize(h, n, deadline):
    """Run multiple optimization phases on a single candidate."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)

    # Phase A: L-BFGS with smooth max
    t_a = min(time.time() + 30, deadline - 60)
    if t_a > time.time() + 3:
        h_a, c_a = run_lbfgs(best_h, n, t_a,
            configs=[(200, 50.0), (1000, 200.0), (3000, 1000.0), (8000, 5000.0)])
        if c_a < best_c:
            best_c = c_a
            best_h = h_a

    # Phase B: SQP
    t_b = min(time.time() + 30, deadline - 30)
    if t_b > time.time() + 3:
        h_b, c_b = sqp_optimize(best_h, n, t_b, max_iter=200, rho_init=0.04, top_k=12)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b

    # Phase C: Subgradient (for further refinement)
    t_c = min(time.time() + 15, deadline - 15)
    if t_c > time.time() + 2:
        h_c, c_c = subgradient_optimize(best_h, n, t_c, lr_init=0.005, decay=0.9995)
        if c_c < best_c:
            best_c = c_c
            best_h = h_c

    # Phase D: Fine local search
    t_d = min(time.time() + 20, deadline - 5)
    if t_d > time.time() + 2:
        h_d, c_d = fine_local_search(best_h, n, t_d, radius=25)
        if c_d < best_c:
            best_c = c_d
            best_h = h_d

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        h_parent = project_feasible(h_parent, n_parent)
        c_parent = compute_c5(h_parent, n_parent)
    except Exception:
        h_parent = np.full(200, 0.5)
        n_parent = 200
        c_parent = compute_c5(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c = c_parent

    # Phase 1: Quick multi-start optimization (200s)
    candidates = []  # list of (c, h, n)
    
    # Add parent as candidate
    candidates.append((c_parent, h_parent.copy(), n_parent))
    
    # Try a range of n values with different initializations
    n_values = [100, 150, 200, 250, 300, 400, 500, 600]
    t_phase1_end = t0 + 180
    
    for n_try in n_values:
        if time.time() > t_phase1_end - 10:
            break
        if n_try == n_parent:
            continue  # already have parent
        # Create several inits
        inits = []
        # Sinusoidal
        inits.append(make_sinusoidal(n_try, freq=1.0, phase=0.0))
        inits.append(make_sinusoidal(n_try, freq=2.0, phase=0.5))
        inits.append(make_sinusoidal(n_try, freq=3.0, phase=0.0))
        # Block
        inits.append(make_block_pattern(n_try, period=4))
        inits.append(make_block_pattern(n_try, period=6))
        # Random smooth
        inits.append(make_random_smooth(n_try, seed_val=seed + n_try))
        
        best_for_n = None
        t_quick_end = min(time.time() + 18, t_phase1_end - 5)
        for init_h in inits:
            if time.time() > t_quick_end - 2:
                break
            # Quick L-BFGS
            try:
                h_opt, c_opt = run_lbfgs(init_h, n_try, t_quick_end,
                    configs=[(200, 50.0), (1000, 200.0)])
            except Exception:
                continue
            if best_for_n is None or c_opt < best_for_n[0]:
                best_for_n = (c_opt, h_opt, n_try)
        
        if best_for_n is not None:
            candidates.append(best_for_n)
            if best_for_n[0] < best_c:
                best_c = best_for_n[0]
                best_h = best_for_n[1].copy()

    # Sort candidates by objective value
    candidates.sort(key=lambda x: x[0])

    # Phase 2: Intensive optimization of top candidates (500s)
    t_phase2_end = t0 + 700
    top_count = min(4, len(candidates))
    time_per_candidate = 100
    
    for ci in range(top_count):
        if time.time() > t_phase2_end - 30:
            break
        c_cand, h_cand, n_cand = candidates[ci]
        if c_cand > best_c + 0.01:
            break  # not worth optimizing
        t_cand_end = min(time.time() + time_per_candidate, t_phase2_end - 10)
        if t_cand_end < time.time() + 5:
            break
        h_opt, c_opt = intensive_optimize(h_cand, n_cand, t_cand_end)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        candidates[ci] = (c_opt, h_opt, n_cand)

    # Phase 3: Try upsample/downsample of best
    t_phase3_end = t0 + 900
    if time.time() < t_phase3_end:
        n_best = len(best_h)
        # Try 2x upsample
        n_up = n_best * 2
        h_up = upsample(best_h, n_best, n_up)
        c_up = compute_c5(h_up, n_up)
        t_up_end = min(time.time() + 60, t_phase3_end - 10)
        if t_up_end > time.time() + 5:
            h_up, c_up = intensive_optimize(h_up, n_up, t_up_end)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up

        # Try integer resolutions
        for n_try in [200, 300, 400, 500, 700, 800]:
            if time.time() > t_phase3_end - 20:
                break
            if n_try == n_best or n_try == n_up:
                continue
            h_try = upsample(best_h, len(best_h), n_try)
            c_try = compute_c5(h_try, n_try)
            t_try_end = min(time.time() + 25, t_phase3_end - 10)
            if t_try_end > time.time() + 3:
                h_try, c_try = intensive_optimize(h_try, n_try, t_try_end)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try

    # Phase 4: Final polish (remaining time)
    if time.time() < deadline - 10:
        n_final = len(best_h)
        # SLSQP polish
        t_slqp = min(time.time() + 30, deadline - 15)
        if t_slqp > time.time() + 3:
            h_sp, c_sp = slqp_polish(best_h, n_final, t_slqp, alpha=5000000, maxiter=3000)
            if c_sp < best_c:
                best_c = c_sp
                best_h = h_sp

        # Very fine local search
        t_fine = min(time.time() + 30, deadline - 5)
        if t_fine > time.time() + 2:
            h_fs, c_fs = fine_local_search(best_h, len(best_h), t_fine,
                                           radius=20,
                                           steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001])
            if c_fs < best_c:
                best_c = c_fs
                best_h = h_fs

        # Final SQP
        t_sqp = deadline - 3
        if t_sqp > time.time() + 3:
            h_sq, c_sq = sqp_optimize(best_h, len(best_h), t_sqp, max_iter=100, rho_init=0.01, top_k=8)
            if c_sq < best_c:
                best_c = c_sq
                best_h = h_sq

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
