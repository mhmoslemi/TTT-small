import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        if nv > 0:
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        if nv > 0:
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def lp_step(h, n, active_idx, rho):
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    # Active constraints + box constraints + step bound + equality
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_idx:
        g = shift_grad_single(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, max(0.0, 2*rho))]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            t_opt = res.x[n]
            return delta, t_opt
    except Exception:
        pass
    return None, None

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pairwise_refine(h, n, deadline, top_k=20):

    best_c = compute_c5(h, n)

    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.003 * max_val)
            active = np.where(c_vec >= thresh)[0]
            if len(active) == 0:
                break
            # For each active shift, the gradient tells us which pairs to adjust

                if time.time() > deadline:
                    return h, best_c
                g = shift_grad_single(h, n, idx)
                # g[i] > 0 means increasing h[i] increases c_idx, so we should decrease h[i]
                # g[i] < 0 means increasing h[i] decreases c_idx, so we should increase h[i]
                # Best move: increase where g is most negative, decrease where g is most positive
                pos_i = np.argmax(g)  # most to decrease
                neg_i = np.argmin(g)  # most to increase
                for _ in range(200):  # try several pairs
                    if time.time() > deadline:
                        break
                    # Random pair biased toward useful directions
                    candidates_inc = np.argsort(g)[:10]  # most negative g
                    candidates_dec = np.argsort(-g)[:10]  # most positive g
                    for ci in candidates_inc:
                        for cj in candidates_dec:
                            if ci == cj:
                                continue

                                    improved = True
                                else:
                                    h[ci] -= step
                                    h[cj] += step
    return h, best_c

    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0 * rho)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[0:n], res.x[n]
    except Exception:
        pass
    return None, None

def active_set_loop(h, n, deadline, max_iter=500, rho_init=0.04, max_active=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if len(active) > max_active:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:max_active]]
        if not active:
            break
        delta, t_opt = lp_step(h, n, active, rho)
        if delta is None:
            rho *= 0.5
            if rho < 1e-12:
                break
            continue
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if rho < 0.08:
                rho = min(rho * 1.3, 0.12)
        else:
            no_improve += 1
            rho *= 0.5
            if rho < 1e-12:
                break
            if no_improve > 50:
                break
    return best_h, best_c

def obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = vectorized_grad(x, n, w, max_nz=400)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_run(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 50.0), (800, 300.0), (3000, 2000.0), (10000, 8000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(obj_pen, h, jac=grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_trials=200000):
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(777)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    per_step = max(1000, max_trials // len(steps))
    for step in steps:
        if time.time() > deadline:
            break
        for _ in range(per_step):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
            if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                h[i] += step
                h[j] -= step
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                else:
                    h[i] -= step
                    h[j] += step
            elif h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                h[i] -= step
                h[j] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                else:
                    h[i] += step
                    h[j] -= step
    return h, best_c

def gradient_descent_proj(h, n, deadline, lr=0.001, alpha=5000):
    """Simple projected gradient with exact subgradient of max."""
    best_c = compute_c5(h, n)
    n_iter = 0
    while time.time() < deadline and n_iter < 50000:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Subgradient: average of gradients at active points
        thresh = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            break
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = vectorized_grad(h, n, w, max_nz=200)
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            h = h_new
            lr = min(lr * 1.1, 0.01)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
        n_iter += 1
    return h, best_c

def upsample_func(h, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=3000):
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x, n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)

    # Initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: L-BFGS at original res (15s)
    t_end = min(t0 + 15, deadline - 950)
    if t_end > time.time() + 3:
        best_h, best_c = lbfgs_run(best_h, n0, t_end,
            configs=[(300, 100.0), (1000, 500.0)])

    # Phase 2: Active set at original res (35s)
    t_end = min(t0 + 50, deadline - 900)
    if t_end > time.time() + 5:
        best_h, best_c = active_set_loop(best_h, n0, t_end,
            max_iter=400, rho_init=0.05, max_active=50)

    # Phase 3: Upsample to n=2000
    n_main = 2000
    h_main = upsample_func(best_h, n0, n_main)
    h_main = project_feasible(h_main, n_main)
    c_main = compute_c5(h_main, n_main)

    # Phase 4: L-BFGS at n=2000 (70s)
    t_end = min(t0 + 120, deadline - 700)
    if t_end > time.time() + 10:
        h_main, c_main = lbfgs_run(h_main, n_main, t_end,
            configs=[(200, 50.0), (800, 300.0), (3000, 2000.0)])

    # Phase 5: Active set at n=2000 (150s) - main optimization
    t_end = min(t0 + 270, deadline - 550)
    if t_end > time.time() + 10:
        h_main, c_main = active_set_loop(h_main, n_main, t_end,
            max_iter=500, rho_init=0.03, max_active=100)

    if c_main < best_c:
        best_c = c_main
        best_h = h_main.copy()

    # Phase 6: Gradient descent with subgradient (50s)
    t_end = min(t0 + 320, deadline - 450)
    if t_end > time.time() + 5:
        h_tmp, c_tmp = gradient_descent_proj(h_main.copy(), n_main, t_end,
            lr=0.002, alpha=5000)
        if c_tmp < best_c:
            best_c = c_tmp
            best_h = h_tmp
            h_main = h_tmp

    # Phase 7: Active set round 2 (100s)
    t_end = min(t0 + 420, deadline - 400)
    if t_end > time.time() + 10:
        h_main, c_main = active_set_loop(h_main, n_main, t_end,
            max_iter=400, rho_init=0.02, max_active=80)
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 8: Coordinate descent (60s)
    t_end = min(t0 + 480, deadline - 350)
    if t_end > time.time() + 5:
        h_main, c_main = coord_descent(h_main, n_main, t_end, max_trials=100000)
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 9: Active set round 3 (80s)
    t_end = min(t0 + 560, deadline - 300)
    if t_end > time.time() + 10:
        h_main, c_main = active_set_loop(h_main, n_main, t_end,
            max_iter=300, rho_init=0.015, max_active=60)
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 10: Upsample to n=4000
    n_hi = 4000
    if time.time() < deadline - 250:
        h_hi = upsample_func(best_h, n_main, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)

        # L-BFGS at n=4000 (60s)
        t_end = min(time.time() + 60, deadline - 120)
        if t_end > time.time() + 10:
            h_hi, c_hi = lbfgs_run(h_hi, n_hi, t_end,
                configs=[(500, 100.0), (2000, 1000.0)])

        # Active set at n=4000 (80s)
        t_end = min(time.time() + 80, deadline - 40)
        if t_end > time.time() + 10:
            h_hi, c_hi = active_set_loop(h_hi, n_hi, t_end,
                max_iter=150, rho_init=0.02, max_active=60)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi

    # Phase 11: Final SLSQP polish (up to 40s)
    n_final = len(best_h)
    if time.time() < deadline - 20 and n_final <= 2000:
        h_fin, c_fin = slqp_polish(best_h, n_final, deadline - 4,
            alpha=5000000, maxiter=2000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
