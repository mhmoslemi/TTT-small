        candidates = free[np.argsort(-score[free])][:20]
        for i in candidates:
            if time.time() > deadline:
                return h, best_c
            for target in [0.0, 1.0]:
                delta = target - h[i]
                j_candidates = np.where(((h > delta) & (delta > 1 - h) & (delta != 0)) |
                                        ((h < -delta) & (-delta > 1 - h) & (delta < 0)))[0]
                if len(j_candidates) == 0:
                    continue
                if delta > 0:
                    j_pool = np.where((h < delta) & (h > 0))[0]
                else:
                    j_pool = np.where((h > -delta) & (h < 1))[0]
                if len(j_pool) == 0:
                    continue
                j_pool_sorted = j_pool[np.argsort(h[j_pool] if delta > 0 else -h[j_pool])]
                for j in j_pool_sorted[:5]:
                    if i == j:
                        continue
                    h_try = h.copy()
                    h_try[i] = target
                    h_try[j] = h[j] - delta
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-12:
                        best_c = c_try
                        h = h_try
                        improved = True
                        flips_done += 1
                        break
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    def obj_grad(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        thresh = 1e-14
        for mm in range(2 * n - 1):
            if w[mm] < thresh:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def target_search(h, n, deadline, radius=30, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-7, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            for idx in active:
                grad += np.abs(shift_grad_single(h, n, idx))
            if len(active) > 0:
                grad /= len(active)
            top = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_perturb(h, n, magnitude, seed):
    rng = np.random.RandomState(seed)
    h_pert = h.copy()
    mask = (h_pert > 0.05) & (h_pert < 0.95)
    h_pert[mask] += rng.uniform(-magnitude, magnitude, mask.sum())
    h_pert = project_feasible(h_pert, n)
    return h_pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS at original resolution (40s)
    t1 = min(t0 + 40, deadline - 950)
    if t1 > time.time() + 3:
        h1, c1 = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 100.0), (2000, 1000.0), (8000, 5000.0)])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: SQP at original resolution (70s)
    t2 = min(t0 + 110, deadline - 880)
    if t2 > time.time() + 5:
        h2, c2 = sqp_optimize(best_h, n_orig, t2, max_iter=400, rho_init=0.04, max_active=40)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Combinatorial refinement (40s)
    t3 = min(t0 + 150, deadline - 840)
    if t3 > time.time() + 3:
        h3, c3 = combinatorial_refine(best_h, n_orig, t3, free_thresh=0.35, max_flips=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Targeted local search (40s)
    t4 = min(t0 + 190, deadline - 800)
    if t4 > time.time() + 3:
        h4, c4 = target_search(best_h, n_orig, t4, radius=40,
                                steps=[0.003, 0.001, 0.0005, 0.0002])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Upsample to n=1000 and optimize (150s)
    n2 = 1000
    if time.time() < deadline - 600:
        h_hi = upsample(best_h, best_n, n2)
        c_hi = compute_c5(h_hi, n2)

        t5 = min(time.time() + 60, deadline - 500)
        if t5 > time.time() + 5:
            h_hi, c_hi = run_lbfgs(h_hi, n2, t5,
                configs=[(1000, 200.0), (5000, 2000.0), (20000, 20000.0)])

        t6 = min(time.time() + 70, deadline - 380)
        if t6 > time.time() + 5:
            h_hi, c_hi = sqp_optimize(h_hi, n2, t6, max_iter=300, rho_init=0.04, max_active=50)

        t6b = min(time.time() + 30, deadline - 320)
        if t6b > time.time() + 3:
            h_hi, c_hi = combinatorial_refine(h_hi, n2, t6b, free_thresh=0.3, max_flips=200)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
            best_n = n2

    # Phase 6: Upsample to n=2000 and optimize (200s)
    n3 = 2000
    if time.time() < deadline - 350:
        h_hh = upsample(best_h, best_n, n3)
        c_hh = compute_c5(h_hh, n3)

        t7 = min(time.time() + 60, deadline - 260)
        if t7 > time.time() + 5:
            h_hh, c_hh = run_lbfgs(h_hh, n3, t7,
                configs=[(2000, 500.0), (10000, 5000.0)])

        t8 = min(time.time() + 80, deadline - 150)
        if t8 > time.time() + 5:
            h_hh, c_hh = sqp_optimize(h_hh, n3, t8, max_iter=200, rho_init=0.03, max_active=40)

        t8b = min(time.time() + 30, deadline - 100)
        if t8b > time.time() + 3:
            h_hh, c_hh = target_search(h_hh, n3, t8b, radius=25,
                                        steps=[0.002, 0.001, 0.0005])

        if c_hh < best_c:
            best_c = c_hh
            best_h = h_hh
            best_n = n3

    # Phase 7: Random restarts (use ~100s)
    if time.time() < deadline - 80:
        for restart in range(5):
            if time.time() > deadline - 60:
                break
            h_r = random_perturb(best_h, best_n, 0.05, seed=restart * 7 + 1)
            c_r = compute_c5(h_r, best_n)
            t_r = min(time.time() + 15, deadline - 60)
            if t_r > time.time() + 2:
                h_r, c_r = sqp_optimize(h_r, best_n, t_r, max_iter=50, rho_init=0.03, max_active=20)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 15:
        h_f, c_f = slqp_polish(best_h, best_n, deadline - 5, alpha=2000000, maxiter=1500)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 9: Final targeted search
    if time.time() < deadline - 10:
        h_f2, c_f2 = target_search(best_h, best_n, deadline - 5, radius=20,
                                     steps=[0.001, 0.0005, 0.0002, 0.0001])
        if c_f2 < best_c:
            best_c = c_f2
            best_h = h_f2

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
