        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 60,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_descent(h, n, deadline, n_iters=20000):
    """Direct subgradient descent on the max correlation."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    initial_step = 0.05
    
    for t in range(n_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = np.argmax(c_all)
        lag = k_star - (n - 1)
        grad = compute_lag_gradient(h, n, lag)
        
        # Adaptive step size: start larger, shrink over time
        step = initial_step / (1.0 + 0.01 * t)
        
        # Backtracking line search
        for trial in range(10):
            h_new = project_feasible(h - step * grad, n)
            c_new = c_all.copy()
            # Only need to recompute affected lags, but for simplicity recompute
            c_new = compute_all_lags(h_new, n)
            max_new = c_new.max()
            max_old = c_all.max()
            if max_new < max_old - 1e-15 or trial == 9:
                if max_new < max_old:
                    h = h_new
                    if max_new < best_c - 1e-15:
                        best_c = float(max_new)
                        best_h = h.copy()
                break
            step *= 0.5
        else:
            continue
    
    return best_h, best_c

def targeted_local_search(h, n, deadline, rng, n_top=5, max_rounds=50000):
    """Targeted local search focusing on top lags with hot-spot identification."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    round_count = 0
    stuck = 0
    
    while round_count < max_rounds and time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-n_top:]
        
        improved_this_round = False
        for _ in range(2000):
            if time.time() > deadline or round_count >= max_rounds:
                break
            round_count += 1
            
            k_star = top_idx[rng.randint(n_top)]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Identify hot spots where overlap is worst
            if lag > 0:
                overlap = h[0:n-m] * (1.0 - h[lag:n])
                # We want to reduce h[i]*(1-h[i+lag]) where it's high
                # i.e., where h[i] is high and h[i+lag] is low
                # Move mass from h[i] to h[i+lag]
                hot = np.where(overlap > 0.15)[0]
                if len(hot) == 0:
                    continue
                i = hot[rng.randint(len(hot))]
                j = i + lag
                d = rng.uniform(0.01, 0.06)
                d = min(d, (h[i] - 0.1) * 0.4, (1.0 - h[j]) * 0.4)
                if d < 1e-8:
                    continue
                h[i] -= d
                h[j] += d
            else:
                idx = -lag
                overlap = h[idx:n] * (1.0 - h[0:n-idx])
                hot = np.where(overlap > 0.15)[0]
                if len(hot) == 0:
                    continue
                i = hot[rng.randint(len(hot))] + idx
                j = i + lag  # j = i - idx
                d = rng.uniform(0.01, 0.06)
                d = min(d, (h[i] - 0.1) * 0.4, (1.0 - h[j]) * 0.4)
                if d < 1e-8:
                    continue
                h[i] -= d
                h[j] += d
            
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-15:
                best_c = c_test
                best_h = h.copy()
                improved_this_round = True
                c_all = compute_all_lags(h, n)
                top_idx = np.argsort(c_all)[-n_top:]
            else:
                # Revert
                h = best_h.copy()
        
        if not improved_this_round:
            stuck += 1
            if stuck > 3:
                # Perturb and restart
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                stuck = 0
            else:
                # Try different lags
                n_top = min(n_top + 2, 15)
        else:
            stuck = 0
            n_top = max(3, n_top - 1)
    
    return best_h, best_c

def quick_simulated_annealing(h, n, deadline, rng, alpha_sa=100000, t_init=0.005,
                               t_min=0.00002, cool=0.9999, max_iters=100000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0
    iters = 0
    
    while time.time() < deadline and T > t_min and iters < max_iters:
        iters += 1
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-4:]
        k_star = top_k[rng.randint(4)]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue
        
        if lag > 0:
            i = rng.randint(0, n - m)
            j = i + m
        else:
            i = rng.randint(0, n - m) + m
            j = i + lag
        
        if h[i] > h[j] + 1e-6:
            d = rng.uniform(0.005, 0.04 + T * 5)
            d = min(d, (h[i] - h[j]) * 0.3)
        elif h[j] > h[i] + 1e-6:
            d = -rng.uniform(0.005, 0.04 + T * 5)
            d = max(d, -(h[j] - h[i]) * 0.3)
        else:
            T *= cool
            continue
        
        if abs(d) < 1e-9:
            T *= cool
            continue
        
        c_old = smooth_max_value(c_all, alpha_sa)
        
        h_old_i, h_old_j = h[i], h[j]
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        
        de = e_new - c_old
        accept = de < 0 or (T > t_min and rng.random() < np.exp(-min(de / T, 30)))
        
        if accept:
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] = h_old_i
            h[j] = h_old_j
            h = project_feasible(h, n)
            it_since_improve += 1
        
        T *= cool
        
        if it_since_improve > 20000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.003)
    
    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    n = len(best_h)
    
    # Phase 1: L-BFGS-B with fine alpha schedule
    alpha_list = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000,
                  50000, 100000, 300000, 1000000, 3000000, 10000000, 50000000]
    maxiter_list = [200, 200, 300, 300, 400, 400, 500, 500, 500,
                    600, 600, 700, 700, 800, 800, 1000]
    
    for alpha, maxit in zip(alpha_list, maxiter_list):
        if time.time() > deadline - 400:
            break
        remaining = deadline - 400 - time.time()
        if remaining < 10:
            break
        alloc = min(remaining, 45)
        h_opt = lbfgsb_optimize(best_h, n, alpha, t0 + time.time() - t0 + alloc,
                                 maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Subgradient descent
    if time.time() < deadline - 350:
        sg_deadline = min(deadline - 250, time.time() + 120)
        h_opt, c_opt = subgradient_descent(best_h, n, sg_deadline, n_iters=15000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: More L-BFGS-B polish at very high alpha
    if time.time() < deadline - 300:
        for alpha in [20000000, 100000000, 500000000]:
            if time.time() > deadline - 200:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 25, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Quick SA rounds (3 short restarts)
    if time.time() < deadline - 200:
        for sa_round in range(3):
            if time.time() > deadline - 150:
                break
            sa_dead = min(deadline - 100, time.time() + 50)
            h_pert = best_h + rng.normal(0, 0.003 * (sa_round + 1), n)
            h_pert = project_feasible(h_pert, n)
            h_opt, c_opt = quick_simulated_annealing(h_pert, n, sa_dead, rng,
                                                      alpha_sa=100000 + sa_round * 50000,
                                                      t_init=0.004,
                                                      t_min=0.00002,
                                                      cool=0.99992)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Targeted local search
    if time.time() < deadline - 100:
        ls_dead = min(deadline - 50, time.time() + 150)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_dead, rng, n_top=5,
                                              max_rounds=40000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Final L-BFGS-B + subgradient
    if time.time() < deadline - 60:
        h_opt = lbfgsb_optimize(best_h, n, 200000000, time.time() + 20, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        
        if time.time() < deadline - 30:
            h_opt, c_opt = subgradient_descent(best_h, n, time.time() + 20, n_iters=10000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 7: Try a different resolution if time allows
    if time.time() < deadline - 80:
        for n_try in [400, 600]:
            if time.time() > deadline - 50:
                break
            remaining = deadline - 40 - time.time()
            if remaining < 30:
                break
            h_interp = expand_to_resolution(best_h, len(best_h), n_try)
            h_opt = lbfgsb_optimize(h_interp, n_try, 500000, time.time() + min(20, remaining*0.4),
                                     maxiter=2000)
            h_opt = lbfgsb_optimize(h_opt, n_try, 50000000, time.time() + min(15, remaining*0.3),
                                     maxiter=2000)
            c_opt = compute_c5(h_opt, n_try)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 8: Ultra-fine local search at the very end
    if time.time() < deadline - 15:
        n_final = len(best_h)
        ls_dead2 = deadline - 3
        h_opt, c_opt = targeted_local_search(best_h, n_final, ls_dead2, rng,
                                              n_top=3, max_rounds=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 9: Very final polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 500000000, deadline - 2, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
