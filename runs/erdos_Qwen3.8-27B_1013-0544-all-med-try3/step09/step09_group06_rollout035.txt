```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                h = h / h.sum() * target_sum
                h = np.clip(h, 0, 1)
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def adam_grad_opt(h, n, alpha, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """ADAM-based gradient optimization with smooth max."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_v = np.zeros(n)  # First moment
    v_v = np.zeros(n)  # Second moment
    t = 0
    patience = 0
    lr_curr = lr
    alpha_curr = alpha
    
    for t in range(100000):
        if time.time() > deadline:
            break
        t += 1
        
        g = grad_smooth_max(h_curr, n, alpha_curr)
        
        # ADAM update
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        step = lr_curr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > 200:
            lr_curr *= 0.8
            patience = 0
        if patience > 1000:
            alpha_curr = max(100, alpha_curr * 0.5)
            patience = 0
        if t % 5000 == 0:
            lr_curr = max(0.001, lr_curr * 0.9)
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):

    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        return smooth_max_value(active, alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        w = smooth_max_weights(cc[top_idx], alpha)
        grad = np.zeros(n)
        for idx, ti in zip(w, top_idx):
            if idx < 1e-10:
                continue
            lag = ti - (n - 1)
            grad += idx * _lag_gradient(x, n, lag)
        return grad * dx
    
    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except:
        pass
    return h.copy()
```

```python
                        c_all = compute_all_lags(h_curr, n)
                        improved = True
                        break
                    else:
                        h_curr[i] += d
                        h_curr[j] -= d
                        h_curr = project_feasible(h_curr, n)

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 300:
                h_curr = h_curr + rng.normal(0, 0.015, n)
                h_curr = project_feasible(h_curr, n)
                stall = 0
            if stall > 2000:
                break

    return best_h, best_c

def _slsqp_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    return smooth_max_value(active, alpha)

def _slsqp_grad(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    w = smooth_max_weights(cc[top_idx], alpha)
    grad = np.zeros(n)
    for wi, ti in zip(w, top_idx):
        if wi < 1e-10:
            continue
        lag = int(ti) - (n - 1)
        grad += wi * _lag_gradient(x, n, lag)
    return grad * dx

def _slsqp_sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_sum_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_sum_con, 'jac': _slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _slsqp_obj(h, n, top_idx, alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_full(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_sum_con, 'jac': _slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(n, np.arange(2*n-1), alpha),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _slsqp_obj(h, n, np.arange(2*n-1), alpha):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def fourier_init(n, k, A, phase=0.0):
    i = np.arange(n)
    h = 0.5 + A * np.cos(2.0 * np.pi * k * i / n + phase)
    return project_feasible(h, n)

def binary_proj(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    remaining = max(1, deadline - time.time())

    # Phase A: Momentum gradient
    if time.time() < deadline - remaining * 0.2:
        phase_dl = time.time() + remaining * 0.3
        alphas = [300, 1500, 8000]
        for alpha in alphas:
            if time.time() > phase_dl:
                break
            h_new, c_new = momentum_grad_opt(h, n, alpha, phase_dl, lr=0.7, momentum=0.9)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            h = h_new

    # Phase B: ADAM gradient
    if time.time() < deadline - remaining * 0.15:
        phase_dl = time.time() + remaining * 0.2
        h_new, c_new = adam_grad_opt(h, n, 5000, phase_dl, lr=0.03)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase C: Targeted lag descent
    if time.time() < deadline - remaining * 0.12:
        phase_dl = time.time() + remaining * 0.22
        h_new, c_new = targeted_lag_descent(h, n, phase_dl)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase D: Pair swap local search
    if time.time() < deadline - remaining * 0.08:
        phase_dl = time.time() + remaining * 0.25
        h_new, c_new = pair_swap_local_search(best_h, n, phase_dl, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase E: SLSQP polish
    if time.time() < deadline - 8:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 4:
                break
            h_new = slsqp_polish(best_h, n, 100000, maxiter=1200, top_m=top_m)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    # Phase 1: Optimize at parent resolution
    if h_init is not None:
        n = n_init
        alloc1 = min(280, (deadline - t0) * 0.33)
        h_opt, c_opt = optimize_at_resolution(h_init, n, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
    else:
        n = 500
        h_start = np.full(n, 0.5)
        alloc1 = min(180, (deadline - t0) * 0.22)
        h_opt, c_opt = optimize_at_resolution(h_start, n, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try higher resolutions
    n_candidates = [700, 1000]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 20
        if remaining < 70:
            break
        alloc = min(remaining * 0.35, 180)
        if alloc < 50:
            break

        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)

        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(1, int((deadline - time.time() - 30) / 45))
        n_restarts = min(n_restarts, 6)
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                hp = best_h + rng.normal(0, 0.012, n_best)
            elif r == 1:
                hp = best_h.copy()
                seg = n_best // 4
                hp[seg:2 * seg] += rng.normal(0, 0.025, seg)
            elif r == 2:
                hp = best_h.copy()
                k_pert = rng.randint(2, 8)
                amp = rng.uniform(0.01, 0.035)
                i_arr = np.arange(n_best)
                hp += amp * np.cos(2.0 * np.pi * k_pert * i_arr / n_best)
            elif r == 3:
                hp = binary_proj(best_h, n_best)
                hp = 0.3 * hp + 0.7 * best_h
            elif r == 4:
                hp = best_h + rng.normal(0, 0.006, n_best)
            else:
                hp = fourier_init(n_best, rng.randint(1, 5), 0.3)

            hp = project_feasible(hp, n_best)
            alloc = min(40, max(25, (deadline - time.time() - 25) * 0.3))
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final intensive refinement
    if time.time() < deadline - 10:
        n = len(best_h)

        if time.time() < deadline - 15:
            h_new, c_new = targeted_lag_descent(best_h, n, deadline - 12)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        if time.time() < deadline - 8:
            h_new, c_new = pair_swap_local_search(best_h, n, deadline - 5, rng)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        if time.time() < deadline - 5:
            h_new = slsqp_polish(best_h, n, 200000, maxiter=1500, top_m=5)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        # Ultra-fine local search
        if time.time() < deadline - 2:
            h_new, c_new = pair_swap_local_search(best_h, n, deadline - 1, rng)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if best_h is None:
        best_h = project_feasible(np.full(500, 0.5), 500)
        best_c = compute_c5(best_h, 500)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
