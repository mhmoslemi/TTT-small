from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def top_m_logsumexp_obj(h, n, top_idx, alpha, var_penalty):
    """Objective: smoothed max over top-m shifts + variance penalty"""
    c_vec = compute_c5_vec(h, n)
    vals = c_vec[top_idx]
    v_max = vals.max()
    sh = alpha * (vals - v_max)
    smx = sh.max()
    if smx > 500:
        val = v_max + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = v_max + np.log(np.sum(np.exp(sh))) / alpha
    # Variance penalty: penalize spread of active shifts
    val += var_penalty * np.var(vals)
    return float(val)

def top_m_logsumexp_grad(h, n, top_idx, alpha, var_penalty):
    c_vec = compute_c5_vec(h, n)
    vals = c_vec[top_idx]
    v_max = vals.max()
    sh = alpha * (vals - v_max)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def find_active_shifts(c_vec, n, rel_tol=0.01, abs_tol=1e-5, max_count=20):

    threshold = max_val * (1.0 - rel_tol) - abs_tol

                active = find_active_shifts(c_vec, n3, rel_tol=0.02, abs_tol=1e-4, max_count=20)
                h_new = lbfgs_smooth_max(h_up3, n3, active, alpha, vp, bounds3, maxiter=2000, maxls=20)
                c_new = compute_c5(h_new, n3)
                if c_new < c_up3:
                    c_up3 = c_new
                    h_up3 = h_new
            if c_up3 < best_c:
                best_c = c_up3
                best_h = h_up3.copy()
    
    # Phase 9: Final local search on best solution
    if time.time() < deadline - 5:
        n_final = len(best_h)
        t9 = min(time.time() + 20, deadline - 3)
        best_h, best_c = local_pair_search(best_h, n_final, t9, center_radius=20, base_delta=0.0005)
    
    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
