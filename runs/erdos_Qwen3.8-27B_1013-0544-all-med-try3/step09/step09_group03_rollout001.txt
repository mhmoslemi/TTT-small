from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def logsumexp_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):

            continue

        wm = w[idx] * dx

def fw_step(x, n, alpha, mu):
    """Frank-Wolfe linear minimization step: find best binary vertex."""
    # Compute gradient of smooth objective at x
    g = logsumexp_grad(x, n, alpha, mu)
    # The FW step: minimize g^T y over y in P (binary, sum = n/2)
    # Just pick n/2 smallest g values to be 1
    idx = np.argsort(g)
    y = np.zeros(n)
    y[idx[:n//2]] = 1.0
    return y, g

def fw_iter(x, n, alpha, mu, gamma=None):
    """One Frank-Wolfe iteration."""
    y, g = fw_step(x, n, alpha, mu)
    if gamma is None:

        gamma = 0.1
    x_new = (1 - gamma) * x + gamma * y
    return project_feasible(x_new, n), y
