    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

            tot = h.sum()
            if tot < 1e-14:

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

            break

                if deadline is not None and time.time() > deadline:
                    return h, best_c
            else:
                # Revert
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_bin, n, deadline, n_random=500):
    """Improve binary h via random pair flips."""
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].astype(int)
    zeros = np.where(h < 0.5)[0].astype(int)
    rng = np.random.RandomState(777)
    
    iter_count = 0
    while time.time() < deadline:
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Try n_random random pair flips
        i_ones = rng.randint(0, n_ones, size=n_random)
        j_zeros = rng.randint(0, n_zeros, size=n_random)
        
        for t in range(n_random):
            i = ones[i_ones[t]]
            j = zeros[j_zeros[t]]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                # Update sets
                ones = np.delete(ones, i_ones[t])
                ones = np.append(ones, j)
                zeros = np.delete(zeros, j_zeros[t])
                zeros = np.append(zeros, i)
                n_ones = len(ones)
                n_zeros = len(zeros)
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        iter_count += 1
        if not improved and iter_count > 20:
            break
    
    return h, best_c

def subgradient_step(h, n, lr, top_k=20):
    """One step of projected subgradient descent."""
    c = compute_c5_vec(h, n)
    idx = np.argsort(-c)[:top_k]
    grad = np.zeros(n)
    center = n - 1
    w_each = 1.0 / top_k
    dx = 2.0 / n
    
    for idx_k in idx:
        k = idx_k - center
        w = w_each * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[k:] += w * (1.0 - h[:nv])
