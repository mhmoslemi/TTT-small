                               args=(n, active, a, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-22,
                                        'gtol': 1e-16, 'maxfun': 2000,
                                        'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Phase 0: Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS polish at native resolution (60s)
    p1_end = t0 + 60
    if time.time() < p1_end:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[500, 2000, 10000, 50000, 200000],
            mu=0.01, maxiters=[500, 800, 1500, 2000, 2500])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: DE at n=200 for global exploration (150s)
    n_de = 200
    p2_end = min(t0 + 220, deadline - 500)
    if time.time() < p2_end:
        h_de_down = upsample(best_h, best_n, n_de)
        c_de_down = compute_c5(h_de_down, n_de)
        inits = [h_de_down]
        # Add a few diverse initializations
        x = np.linspace(0, 2, n_de)
        for f in [1.5, 2.5, 3.0, 4.5]:
            h_s = (np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28)) > 0).astype(float)
            inits.append(project_feasible(h_s, n_de))
        h_de, c_de = de_run(n_de, 35, p2_end, rng, init_list=inits)
        # Upsample to native resolution
        h_de_up = upsample(h_de, n_de, best_n)
        c_de_up = compute_c5(h_de_up, best_n)
        if c_de_up < best_c:
            best_c = c_de_up
            best_h = h_de_up

    # Phase 3: DE at n=400 (120s)
    n_de2 = 400
    p3_end = min(t0 + 360, deadline - 400)
    if time.time() < p3_end:
        h_de3_down = upsample(best_h, best_n, n_de2)
        h_de3, c_de3 = de_run(n_de2, 30, p3_end, rng, init_list=[h_de3_down])
        h_de3_up = upsample(h_de3, n_de2, best_n)
        c_de3_up = compute_c5(h_de3_up, best_n)
        if c_de3_up < best_c:
            best_c = c_de3_up
            best_h = h_de3_up

    # Phase 4: L-BFGS refinement at native resolution (120s)
    p4_end = min(t0 + 500, deadline - 300)
    if time.time() < p4_end:
        h4, c4 = run_lbfgs(best_h, best_n, p4_end,
            alphas=[5000, 20000, 100000, 500000, 2000000],
            mu=0.005, maxiters=[1000, 1500, 2000, 2500, 3000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active set optimization (150s)
    p5_end = min(t0 + 680, deadline - 200)
    if time.time() < p5_end:
        h5, c5 = run_active_opt(best_h, best_n, p5_end, top_k=5,
            alphas=[200000, 1000000, 5000000, 20000000, 100000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Upsample to n=1000 and optimize (150s)
    n_high = 1000
    p6_end = min(t0 + 850, deadline - 150)
    if time.time() < p6_end - 60:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.005:
            h6, c6 = run_lbfgs(h_high, n_high, min(time.time() + 80, p6_end),
                alphas=[1000, 5000, 20000, 100000, 500000],
                mu=0.003, maxiters=[500, 800, 1200, 1500, 2000])
            if c6 < best_c:
                best_c = c6
                best_h = h6
                best_n = n_high

    # Phase 7: Basin hopping (100s)
    p7_end = min(t0 + 970, deadline - 80)
    if time.time() < p7_end:
        h7, c7 = basin_hopping(best_h, best_n, p7_end, rng, n_iter=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Active set at final resolution (60s)
    p8_end = min(t0 + 1050, deadline - 30)
    if time.time() < p8_end:
        h8, c8 = run_active_opt(best_h, best_n, p8_end, top_k=4,
            alphas=[500000, 2000000, 10000000, 50000000])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Single constraint dive (30s)
    if time.time() < deadline - 15:
        h9, c9 = single_dive(best_h, best_n, min(time.time() + 25, deadline - 10), alpha=20000000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Pair search (remaining)
    if time.time() < deadline - 3:
        h10, c10 = pair_search(best_h, best_n, deadline - 2, radius=50)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
