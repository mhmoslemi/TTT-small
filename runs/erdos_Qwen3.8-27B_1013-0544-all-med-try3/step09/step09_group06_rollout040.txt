```python
def solve_il_lp(h_old, n, active_lags, active_gradients, active_values):
    """Solve the LP for iterated linearization.
    
    active_lags: list of lag indices (in the 0..2n-2 range of the full correlation)
    active_gradients: array of shape (len(active_lags), n)
    active_values: array of shape (len(active_lags),) - O_s(h_old) for each active lag
    """
    M = len(active_lags)
    
    # Variables: h[0..n-1], t → total n+1 variables
    # Minimize: t → c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag s: a_s^T h - t <= -b_s
    # where b_s = O_s(h_old) - a_s^T h_old
    A_ub = np.zeros((M + 2*n, n + 1))
    b_ub = np.zeros(M + 2*n)
    
    for i, (a_s, val_s) in enumerate(zip(active_gradients, active_values)):
        b_s = val_s - np.dot(a_s, h_old)
        A_ub[i, :n] = a_s
        A_ub[i, n] = -1.0
        b_ub[i] = -b_s
    
    # h[j] <= 1: handled by bounds
    # h[j] >= 0: handled by bounds
    
    # Equality: sum h[j] = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

```python
    all_lags = compute_all_lags(h, n)
```
