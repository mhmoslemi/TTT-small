import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, L):
    dx = 2.0 / n
    m = abs(L)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        for k in range(n - L):
            grad[k + L] += (1.0 - h[k])
            grad[k] -= h[k + L]
    else:
        for k in range(n - m):
            grad[k] += (1.0 - h[k + m])
            grad[k + m] -= h[k]
    return grad * dx

def top_lags_gradient(h, n, c_all, top_k):
    """Gradient of sum of top_k lags."""
    idx = np.argsort(c_all)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        grad += lag_gradient(h, n, L)
    return grad / top_k

def gradient_optimize(h, n, deadline, top_k=3, max_iter=100000, lr_init=1.0):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = top_lags_gradient(h, n, c_all, top_k)
        
        # Try a few step sizes
        improved = False
        for scale in [1.0, 0.5, 0.2, 0.1]:
            h_new = project_feasible(h - lr * scale * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_h = h_new.copy()
                best_c = c_new
                c_all = compute_all_lags(h, n)
                improved = True
                break
        if not improved:
            lr *= 0.7
            if lr < 0.001:
                break
        
        # Periodically increase top_k
        if it % 1000 == 999:
            top_k = min(top_k + 1, 8)
            lr = max(lr, 0.5)
    
    return best_h, best_c

def slsqp_polish(h, n, top_m, alpha, maxiter=3000):
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    
    def objective(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(shifted)))) / alpha
    
    def constraint_fun(x):
        return float(np.sum(x) - n / 2.0)
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, L):
    dx = 2.0 / n
    m = abs(L)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        for k in range(n - L):
            grad[k + L] += (1.0 - h[k])
            grad[k] -= h[k + L]
    else:
        for k in range(n - m):
            grad[k] += (1.0 - h[k + m])
            grad[k + m] -= h[k]
    return grad * dx

def top_lags_gradient(h, n, c_all, top_k):
    idx = np.argsort(c_all)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        grad += lag_gradient(h, n, L)
    return grad / float(top_k)

def gradient_optimize(h, n, deadline, top_k=3, lr_init=1.0, max_iter=200000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    lr = lr_init
    stale = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = top_lags_gradient(h, n, c_all, top_k)
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_feasible(h - lr * scale * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_h = h_new.copy()
                best_c = c_new
                c_all = compute_all_lags(h, n)
                improved = True
                lr = min(lr * 1.1, 5.0)
                stale = 0
                break
        if not improved:
            stale += 1
            lr *= 0.8
            if lr < 0.005:
                break
            if stale > 200:
                top_k = min(top_k + 1, 10)
                lr = max(lr, 0.3)
                stale = 0

    return best_h, best_c

def _slsqp_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def _slsqp_eq_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_eq_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, top_m, alpha, maxiter=3000):
    dx = 2.0 / n
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, max_rounds=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnation = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-8:]
        improved = False

        for trial in range(300):
            if time.time() > deadline:
                break
            k = top_idx[rng.randint(len(top_idx))]
            L = k - (n - 1)
            if L == 0:
                L = (rng.randint(1, n) if rng.random() < 0.5 else -rng.randint(1, n))
                k = L + (n - 1)
            m = abs(L)
            if m >= n:
                continue
            i = rng.randint(n - m)
            j = i + m
            diff = h[i] - h[j]
            if abs(diff) < 1e-7:
                continue
            base_d = 0.001 + 0.004 * rng.random()
            for d_frac in [base_d, base_d * 0.5, base_d * 0.2]:
                if diff > 0:
                    delta = min(d_frac, diff * 0.4, h[i] * 0.9, (1.0 - h[j]) * 0.9)
                else:
                    delta = -min(d_frac, (-diff) * 0.4, h[j] * 0.9, (1.0 - h[i]) * 0.9)
                if abs(delta) < 1e-9:
                    continue
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0, 1)

        if not improved:
            stagnation += 1
            if stagnation > 15:
                # Try structured perturbation to escape
                if time.time() > deadline - 2:
                    break
                scale = 0.005 + 0.01 * (stagnation - 15) * 0.3
                h = project_feasible(h + rng.normal(0, scale, n), n)
                stagnation = 0
        else:
            stagnation = 0

    return best_h, best_c

def deterministic_local_search(h, n, deadline, max_rounds=100):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnation = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-5:]
        improved = False

        for k2 in top_idx:
            if time.time() > deadline:
                break
            L = k2 - (n - 1)
            if L == 0:
                continue
            m = abs(L)
            for d in [0.002, 0.001, 0.0005]:
                if time.time() > deadline:
                    break
                if L > 0:
                    for i in range(0, n - m, max(1, (n // 500))):
                        j = i + m
                        if h[j] <= h[i]:
                            continue
                        delta = min(d, (h[j] - h[i]) * 0.3, h[j] * 0.9, (1.0 - h[i]) * 0.9)
                        if delta < 1e-9:
                            continue
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
                else:
                    for i in range(0, n - m, max(1, (n // 500))):
                        j = i + m
                        if h[i] <= h[j]:
                            continue
                        delta = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.9, (1.0 - h[j]) * 0.9)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta

        if not improved:
            stagnation += 1
            if stagnation > 20:
                break
        else:
            stagnation = 0

    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Gradient
    if time.time() < deadline - 30:
        gd_deadline = min(deadline - 30, time.time() + (deadline - time.time()) * 0.3)
        h_new, c_new = gradient_optimize(h, n, gd_deadline, top_k=3, lr_init=1.5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new

    # Phase 2: SLSQP
    if time.time() < deadline - 25:
        for tm in [3, 5, 8]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_polish(h, n, tm, 500000, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: Deterministic local search
    if time.time() < deadline - 10:
        ls_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.35)
        h_new, c_new = deterministic_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Stochastic local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 2, time.time() + (deadline - time.time()) * 0.4)
        h_new, c_new = local_search(best_h, n, ls_deadline, rng=rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Final SLSQP
    if time.time() < deadline - 3:
        h_new = slsqp_polish(best_h, n, 5, 1000000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    # Start from initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    # Phase A: Optimize at initial resolution
    alloc_a = min(300, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_pipeline(h_init, n_init, t0 + alloc_a, rng)
    best_h, best_c = h_opt, c_opt

    # Phase B: Try higher resolution
    n_hi = min(n_init + 250, 800)
    if time.time() < deadline - 120:
        remaining_b = min(250, (deadline - time.time()) * 0.25)
        if remaining_b > 80:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_hi)
            h_start_hi = np.interp(x_new, x_old, best_h)
            h_start_hi = project_feasible(h_start_hi, n_hi)
            h_opt_hi, c_opt_hi = optimize_pipeline(h_start_hi, n_hi, time.time() + remaining_b, rng)
            if c_opt_hi < best_c:
                best_c = c_opt_hi
                best_h = h_opt_hi

    # Phase C: Try slightly lower resolution (can sometimes help)
    n_lo = max(n_init - 100, 300)
    if time.time() < deadline - 100 and n_lo != n_init:
        remaining_c = min(150, (deadline - time.time()) * 0.15)
        if remaining_c > 60:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_lo)
            h_start_lo = np.interp(x_new, x_old, best_h)
            h_start_lo = project_feasible(h_start_lo, n_lo)
            h_opt_lo, c_opt_lo = optimize_pipeline(h_start_lo, n_lo, time.time() + remaining_c, rng)
            if c_opt_lo < best_c:
                best_c = c_opt_lo
                best_h = h_opt_lo

    # Phase D: Intensive local search on best
    n_best = len(best_h)
    if time.time() < deadline - 30:
        remaining_d = deadline - time.time() - 20
        alloc_d = min(remaining_d * 0.5, 350)
        if alloc_d > 30:
            h_new, c_new = local_search(best_h, n_best, time.time() + alloc_d, rng=rng)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase E: Deterministic local search + SLSQP polish
    if time.time() < deadline - 10:
        remaining_e = deadline - time.time() - 5
        alloc_e1 = min(remaining_e * 0.4, 80)
        if alloc_e1 > 10:
            h_new, c_new = deterministic_local_search(best_h, n_best, time.time() + alloc_e1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase F: Final SLSQP polish
    if time.time() < deadline - 5:
        for tm in [3, 5, 8, 12]:
            if time.time() > deadline - 3:
                break
            h_new = slsqp_polish(best_h, n_best, tm, 2000000, maxiter=5000)
            c_new = compute_c5(h_new, n_best)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase G: Final micro local search
    if time.time() < deadline - 2:
        h_new, c_new = local_search(best_h, n_best, deadline - 1, rng=rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final projection and verification
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
        n_best = n_init

    return best_h, best_c, n_best
