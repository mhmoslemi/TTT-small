def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample_block(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        idx = int(i * n_fine / n_coarse)
        if idx < n_fine:
            h_fine[idx] = h_coarse[i]
            if idx + 1 < n_fine:
                h_fine[idx + 1] = h_coarse[i]
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def upsample_interp(h_coarse, n_fine):

    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def logsumexp_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def logsumexp_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()

def balance_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    excess = np.maximum(c - m, 0)
    val = m + alpha * np.sum(excess ** 2)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

            except Exception:
                pass
        else:
            try:

def active_set_refine(h, n, deadline, top_k=8, alphas=None, rounds=20):

def pair_swap_search(h, n, deadline, radius=60, deltas=None):

    best_c = compute_c5(h, n)

                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:

                            improved = True
                        else:

def gradient_pair_search(h, n, deadline, step_sizes=None, max_steps=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    if nv > 0:
                        grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                        grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    if nv > 0:
                        grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                        grad[0:nv] += (-best_h[kk:n]) * dx_val
            if len(active) > 0:
                grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                gi = grad[i]
                for j in neg_idx[:40]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty(n, alpha_param, rng=None):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_golden_blocks(n):
    h = np.zeros(n)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * golden) - int(i * golden)
        h[i] = 1.0 if frac < 0.5 else 0.0
    h = project_feasible(h, n)
    return h

def run_constrained(h, n, deadline, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):

        c_new = compute_c5(h_new, n)
