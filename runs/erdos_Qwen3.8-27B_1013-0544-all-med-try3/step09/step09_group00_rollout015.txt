def make_sturmian(n, alpha_param):
    """Generate a Sturmian-like binary sequence."""
    h = np.zeros(n)
    n_ones = n // 2
    # Use Beatty sequence
    seen = set()
    k = 1
    while len(seen) < n_ones and k < 10 * n:
        pos = int(k * alpha_param) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
        k += 1
    # Fill remaining if needed
    if h.sum() < n_ones:
        zeros = np.where(h == 0)[0]
        for z in zeros[:int(n_ones - h.sum())]:
            h[z] = 1.0
    # Fix sum
    while h.sum() > n_ones:
        ones = np.where(h == 1)[0]
        h[ones[-1]] = 0
    return h

def make_periodic(n, period_pattern):
    """Create a periodic pattern and tile to n."""
    h = np.zeros(n)
    p = len(period_pattern)
    for i in range(n):
        h[i] = period_pattern[i % p]
    # Adjust to have exactly n/2 ones
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[-int(-diff):]] = 0.0
    return h

def binary_swap_search(h, n, deadline, n_outer=50, n_inner_trials=200, temperature=1.0):
    """Stochastic binary swap local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(42)
    
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        improved = False
        for trial in range(n_inner_trials):
            if time.time() > deadline:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            if i == j:
                continue
            # Try swap
            best_h[i] = 0.0
            best_h[j] = 1.0
            c_new = compute_c5(best_h, n)

                ones = np.where(best_h > 0.5)[0]

            else:

                if rng.random() < np.exp(-((c_new - best_c) / temperature)):

                    best_h[j] = 0.0
        # Cool down
        temperature *= 0.9
        if not improved and temperature < 0.01:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not mask.any():
                break
            room = (1.0 - h) * mask
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-10
            if not mask.any():
                break
            tot = (h * mask).sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * (h * mask)
    return np.clip(h, 0.0, 1.0)

def make_sturmian(n, alpha_param):

    seen = set()

    while len(seen) < n_ones and k < 50 * n:

            seen.add(pos)
            h[pos] = 1.0
        k += 1

                        h_tmp = project_feasible(best_h, n)
                        c_new = compute_c5(h_tmp, n)
                        if c_new < best_c - 1e-15:
                            best_h = h_tmp
                            best_c = c_new
                            improved = True
                            break
                        best_h[i] = 1.0
                        best_h[j] = 0.0
                    if improved:
                        break
        if not improved:
            break
    return best_h, best_c

def fine_nonbinary_adjust(h, n, deadline, step=0.001, max_rounds=2000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_idx = np.argmax(c_vec)
    k = max_idx - (n - 1)
    rng = np.random.RandomState(123)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            kk = idx - (n - 1)
            if kk >= 0:
                nv = n - kk
                grad[0:nv] += (1.0 - best_h[kk:n]) * dx_val
                grad[kk:n] += (-best_h[0:nv]) * dx_val
            else:
                kk2 = -kk
                nv = n - kk2
                grad[kk2:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk2:n]) * dx_val
        grad /= len(active)
        pos_mask = (grad > 1e-8) & (best_h > 1e-8)
        neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        i = pos_idx[rng.randint(len(pos_idx))]
        j = neg_idx[rng.randint(len(neg_idx))]
        amt = min(step, best_h[i], 1.0 - best_h[j])
        if amt < 1e-10:
            break
        best_h[i] -= amt
        best_h[j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new >= best_c:
            best_h[i] += amt
            best_h[j] -= amt
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def upsample(h_low, n_low, n_high):
    h = np.zeros(n_high)
    for i in range(n_high):
        src = int(i * n_low / n_high)
        if src >= n_low:
            src = n_low - 1
        h[i] = h_low[src]
    return project_feasible(h, n_high)

def slqp_polish(h, n, deadline, alpha=1000000):
    bounds = [(0.0, 1.0)] * n
    cons_list = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn / 2.0,
                  'jac': lambda x, nn=n: np.ones(nn)}]
    maxit = max(200, int((deadline - time.time()) * 8))
    try:
        res = minimize(full_smooth_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons_list,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Generate diverse initializations
    candidates = [h0.copy()]
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    slopes = [golden / 2.0, np.sqrt(2.0) / 2.0, np.sqrt(3.0) / 2.0,
              (1.0 + np.sqrt(2.0)) / 3.0, np.e / 4.0,
              (np.sqrt(5.0) - 1.0) / 4.0, 0.381966, 0.414214]
    for s in slopes:
        candidates.append(make_sturmian(n, s))
    for _ in range(4):
        candidates.append(make_random_binary(n, rng))
    for off in [0, 1, 2]:
        candidates.append(make_interleaved(n, off))

    # Evaluate and select top candidates
    scored = []
    for c in candidates:
        c = project_feasible(c, n)
        cv = compute_c5(c, n)
        scored.append((cv, c))
    scored.sort(key=lambda x: x[0])

    # Take top 4 diverse candidates
    selected = [scored[0][1]]
    for sc, hc in scored[1:]:
        if len(selected) >= 4:
            break
        selected.append(hc)

    # Optimize each candidate
    for si, start_h in enumerate(selected):
        remaining = deadline - time.time()
        if remaining < 100:
            break
        cur_h = start_h.copy()
        cur_c = compute_c5(cur_h, n)
        if cur_c >= best_c:
            continue

        # Phase A: Binary swap local search
        phase_end = min(time.time() + 80, deadline - 100)
        cur_h, cur_c = binary_swap_local(cur_h, n, phase_end,
                                         n_trials=30000, seed=seed + si * 100)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()

        # Phase B: Non-binary L-BFGS (full smooth max, increasing alpha)
        if time.time() < deadline - 100:
            phase_end = min(time.time() + 120, deadline - 80)
            cur_h, cur_c = full_smooth_lbfgs(cur_h, n, phase_end,
                                             alphas=[50, 200, 1000, 5000, 20000],
                                             mu=0.005)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()

        # Phase C: Active set L-BFGS
        if time.time() < deadline - 60:
            phase_end = min(time.time() + 100, deadline - 40)
            cur_h, cur_c = active_set_lbfgs(cur_h, n, phase_end,
                                            alphas=[50000, 200000, 1000000, 5000000],
                                            mu=0.001, max_active=8, rounds_per_alpha=3)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()

        # Phase D: Binary targeted search on worst lag
        if time.time() < deadline - 40:
            phase_end = min(time.time() + 50, deadline - 20)
            cur_h, cur_c = targeted_binary_search(cur_h, n, phase_end, max_passes=15)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()

        # Phase E: Fine non-binary adjustment
        if time.time() < deadline - 20:
            phase_end = min(time.time() + 30, deadline - 10)
            cur_h, cur_c = fine_nonbinary_adjust(cur_h, n, phase_end,
                                                 step=0.002, max_rounds=5000)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()

    # Final polish on the global best
    if time.time() < deadline - 30:
        # More aggressive active set
        phase_end = min(time.time() + 60, deadline - 20)
        h_pol, c_pol = active_set_lbfgs(best_h, best_n if 'best_n' in dir() else n,
                                         phase_end,
                                         alphas=[100000, 500000, 2000000, 10000000],
                                         mu=0.0005, max_active=12, rounds_per_alpha=4)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # Binary micro-adjustments on worst lag
    if time.time() < deadline - 15:
        phase_end = min(time.time() + 20, deadline - 5)
        h_pol, c_pol = targeted_binary_search(best_h, n, phase_end, max_passes=20)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # Very fine non-binary
    if time.time() < deadline - 5:
        h_pol, c_pol = fine_nonbinary_adjust(best_h, n, min(time.time() + 10, deadline - 2),
                                             step=0.0005, max_rounds=5000)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # SLSQP final polish
    if time.time() < deadline - 3:
        h_pol, c_pol = slqp_polish(best_h, n, deadline - 1, alpha=5000000)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
