```python
        # Find the most critical lag
        k_star = int(np.argmax(c_all))
        lag = k_star - (n-1)
        if lag == 0:
            # Try other lags
            c_sorted = np.sort(c_all)[::-1]
            if c_sorted[0] - c_sorted[1] < 1e-10:
                break
            # Use second best lag
            c_sorted_ind = np.argsort(c_all)[::-1]
            for idx in c_sorted_ind:
                lag = idx - (n-1)
                if lag != 0:
                    break
            else:
                break
        
        m = abs(lag)
        improved = False
        
        # Try small adjustments at the critical lag
        for i in range(0, n - m):
            if time.time() > deadline:
                return h, best_c
            j = i + m
            # Try moving mass from h[i] to h[j] or vice versa
            hi, hj = h[i], h[j]
            if hi <= hj:
                continue
            # Move small amount from i to j
            for frac in [0.001, 0.002, 0.005, 0.01]:
                delta = frac * (hi - hj)
                if delta < 1e-10 or delta >= hi - 1e-10:
                    continue
                h[i] -= delta
                h[j] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    c_all = compute_all_lags(h, n)
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
        if not improved:
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] -= h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] -= h[0:n-m]
    return grad

def grad_direct_max(h, n):
    c_all = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    k_star = int(np.argmax(c_all))
    lag = k_star - (n-1)
    grad = _lag_gradient(h, n, lag)
    return grad * (2.0/n), c_all, k_star

def grad_smooth_max(h, n, alpha, top_k=20):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[::-1][:top_k]
    c_top = c[idx]
    m = c_top.max()
    shifted = alpha * (c_top - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n), c
    w = w / ws
    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-12:
            continue
        lag = idx[i] - (n-1)
        grad += w[i] * _lag_gradient(h, n, lag)
    return grad * dx, c

def gradient_descent(h, n, deadline, alpha_seq=None, lr=0.3, momentum=0.8):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    while time.time() < deadline:
        if alpha_seq is not None:
            frac = min(float(it) / len(alpha_seq), 1.0)
            idx_a = int(frac * (len(alpha_seq)-1))
            alpha = alpha_seq[idx_a]
        else:
            alpha = 500.0
        g, c_all = grad_smooth_max(h_curr, n, alpha, top_k=15)
        g_norm = np.linalg.norm(g)
        if g_norm > 3.0:
            g = g / g_norm * 3.0
        v = momentum * v - lr * g
        step_scale = 0.15
        h_new = project_feasible(h_curr + v * step_scale, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.6
                v *= 0.4
                step_scale *= 0.7
                no_improve = 0
                if lr < 1e-3:
                    break
        h_curr = h_new
        it += 1
    return best_h, best_c

def direct_max_descent(h, n, deadline, lr=0.02):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    it = 0
    no_improve = 0
    while time.time() < deadline:
        g, c_all, k_star = grad_direct_max(h_curr, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm * min(lr, g_norm)
        h_new = project_feasible(h_curr - g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            h_new = project_feasible(h_curr - g * 0.5, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
                h_curr = h_new
            else:
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.7
                    no_improve = 0
                    if lr < 1e-6:
                        break
        h_curr = best_h if c_new < best_c else h_curr
        it += 1
    return best_h, best_c

def slsqp_objective(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_sum_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, top_m=5, alpha=100000, maxiter=2000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum_eq, 'jac': slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_objective, h.copy(), args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, n)
        return h_out
    except Exception:
        return h.copy()

def fine_local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    while time.time() < deadline:
        improved = False
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[::-1][:8]
        for step_s in steps:
            if time.time() > deadline:
                return h, best_c
            for k2 in top_lags:
                if time.time() > deadline:
                    return h, best_c
                lag = k2 - (n-1)
                if lag == 0:
                    continue
                m = abs(lag)
                for i in range(0, n - m):
                    if time.time() > deadline:
                        return h, best_c
                    j = i + m
                    hi, hj = h[i], h[j]
                    if hi <= hj + 1e-10:
                        continue
                    delta = min(step_s, (hi - hj) * 0.3)
                    if delta < 1e-12:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
        if not improved:
            step_s_current = steps[-1]
            if step_s_current < 1e-5:
                break
            steps = steps[1:]
            if not steps:
                break
    return h, best_c

def multi_lag_local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n-1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        for d in [0.003, 0.001, 0.0003, 0.0001]:
            if time.time() > deadline:
                return h, best_c
            for i in range(0, n - m):
                if time.time() > deadline:
                    return h, best_c
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                else:
                    delta = -min(d, (h[j] - h[i]) * 0.4)
                if abs(delta) < 1e-12:
                    continue
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[i] += delta
                    h[j] -= delta
        if not improved:
            break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pipeline_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    remaining = deadline - time.time()
    if remaining < 20:
        return best_h, best_c

    # Phase A: Gradient descent with smooth max
    t_alpha = min(30, remaining * 0.3)
    alpha_seq = [2000, 800, 300, 100, 50]
    h_new, c_new = gradient_descent(h, n, time.time() + t_alpha, alpha_seq=alpha_seq, lr=0.4, momentum=0.85)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new
    h = h_new

    # Phase B: SLSQP
    remaining = deadline - time.time()
    if remaining > 30:
        for tm in [5, 10]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_polish(h, n, top_m=tm, alpha=200000, maxiter=1500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase C: Direct max descent
    remaining = deadline - time.time()
    if remaining > 20:
        t_dir = min(20, remaining * 0.3)
        h_new, c_new = direct_max_descent(h, n, time.time() + t_dir, lr=0.01)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = h_new

    # Phase D: Fine local search
    remaining = deadline - time.time()
    if remaining > 15:
        t_ls = min(remaining - 5, 40)
        h_new, c_new = fine_local_search(h, n, time.time() + t_ls)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase E: Multi-lag search
    remaining = deadline - time.time()
    if remaining > 10:

        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
```
