import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift_s(h, n, s):
    """Gradient of C_s (for shift s>=0) w.r.t. h. C_s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])"""
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    nv = len(c)
    for mm in range(nv):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm]
        if k >= 0:
            grad[k:] += wm * (1.0 - x[:n - k])
            grad[:n - k] += wm * (-x[k:])
        else:
            ss = -k
            grad[ss:] += wm * (1.0 - x[:n - ss])
            grad[:n - ss] += wm * (-x[ss:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mus):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for idx, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_set_grad(h, n, K=8):
    """Compute average gradient of top-K shifts at the max."""
    c = compute_c5_all(h, n)
    m = c.max()
    thresh = m - max(1e-12, 0.0005 * m)
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.argmax(c).reshape(1)
    if len(active) > K:
        top_c = c[active]
        order = np.argsort(-top_c)
        active = active[order[:K]]
    grad = np.zeros(n)
    nv = len(c)
    for mm in active:
        k = mm - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] += (-h[k:])
        else:
            ss = -k
            grad[ss:] += (1.0 - h[:n - ss])
            grad[:n - ss] += (-h[ss:])
    grad /= len(active)
    return grad, c

def projected_mirror_step(h, n, grad, step_size):
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def active_set_optimize(h, n, deadline, max_iter=5000, K=8, lr_init=0.001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        grad, c_vec = active_set_grad(h, n, K)
        m = c_vec.max()
        momentum = beta * momentum + (1 - beta) * grad
        h_trial = h - lr * momentum
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial
            stall_count = 0
            if lr < lr_init:
                lr = min(lr * 1.05, lr_init)
        else:
            hill_trial = h + 0.5 * lr * momentum
            hill_trial = project_feasible(hill_trial, n)
            c_hill = compute_c5(hill_trial, n)
            if c_hill < best_c - 1e-15:
                best_c = c_hill
                best_h = hill_trial.copy()
                h = hill_trial
                lr *= 0.8
                stall_count = 0
            else:
                lr *= 0.7
                stall_count += 1
                if stall_count > 50:
                    lr = lr_init * 0.1
                    momentum *= 0.5
                    stall_count = 0
                if lr < 1e-10:

    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    def obj_fn(x):
        c = compute_c5_all(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_fn(x):
        return float(x.sum() - n / 2.0)
    def sq_jac_fn(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, args=(), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def local_search(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_all(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            grad = np.zeros(n)
            if len(active) > 0:
                for mm in active:
                    k = mm - (n - 1)
                    if k >= 0:
                        grad[k:] += np.abs(1.0 - h[:n - k])
                        grad[:n - k] += np.abs(-h[k:])
                    else:
                        ss = -k
                        grad[ss:] += np.abs(1.0 - h[:n - ss])
                        grad[:n - ss] += np.abs(-h[ss:])
                grad /= len(active)
            top_idx = np.argsort(-grad)[:min(radius * 2, n)]
            for i_val in top_idx:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def sqp_step(h, n, active_indices, rho):
    n_act = len(active_indices)
    if n_act == 0:
        return h, compute_c5(h, n)
    c_vec = compute_c5_all(h, n)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    for idx in active_indices:
        k = idx - (n - 1)
        cv = c_vec[idx]
        if k >= 0:
            A_ub[row, k:n] += (1.0 - h[:n - k])
            A_ub[row, :n - k] += (-h[k:])
        else:
            ss = -k
            A_ub[row, ss:n] += (1.0 - h[:n - ss])
            A_ub[row, :n - ss] += (-h[ss:])
        A_ub[row, n] = -1.0
        b_ub[row] = -cv
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    A_eq = np.zeros((1, n_var))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[0:n]
            h_new = project_feasible(h + delta, n)
            return h_new, compute_c5(h_new, n)
    except Exception:
        pass
    return None, None

            rho *= 0.5
            if rho < 1e-10 or no_improve > 30:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    # Interpolate to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, h_init)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS warmup at n=1000 (100s)
    t1 = min(t0 + 100, deadline - 700)
    if t1 > time.time() + 5:
        h, best_c = lbfgs_phase(h, n, t1,
            alphas=[500, 2000, 8000],
            mus=[200.0, 2000.0, 20000.0])
        best_h = h.copy()
    
    # Phase 2: Active-set projected gradient (120s)
    t2 = min(t0 + 220, deadline - 500)
    if t2 > time.time() + 5:
        h, c2 = active_set_optimize(best_h, n, t2, max_iter=10000, K=6, lr_init=0.002)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()
    
    # Phase 3: SQP refinement (80s)
    t3 = min(t0 + 300, deadline - 400)
    if t3 > time.time() + 5:
        h3, c3 = sqp_optimize(best_h, n, t3, max_iter=150, rho_init=0.04)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: SLSQP polish (60s)
    t4 = min(t0 + 360, deadline - 350)
    if t4 > time.time() + 5:
        h4, c4 = slqp_polish(best_h, n, t4, alpha=5000000, maxiter=5000)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 5: L-BFGS with very high alpha (80s)
    t5 = min(t0 + 440, deadline - 300)
    if t5 > time.time() + 5:
        h5, c5v = lbfgs_phase(best_h, n, t5,
            alphas=[20000, 100000],
            mus=[100000.0, 500000.0])
        if c5v < best_c:
            best_c = c5v
            best_h = h5.copy()
    
    # Phase 6: Active-set again (80s)
    t6 = min(t0 + 520, deadline - 250)
    if t6 > time.time() + 5:
        h6, c6 = active_set_optimize(best_h, n, t6, max_iter=10000, K=4, lr_init=0.0005)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    
    # Phase 7: Fine local search (120s)
    t7 = min(t0 + 640, deadline - 200)
    if t7 > time.time() + 5:
        h7, c7 = local_search(best_h, n, t7, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    
    # Phase 8: Try n=2000 if time permits (150s)
    n_hi = 2000
    t8_start = time.time()
    t8 = min(t8_start + 150, deadline - 50)
    if t8 > t8_start + 20:
        x_old2 = np.linspace(0, 2, n)
        x_hi = np.linspace(0, 2, n_hi)
        h_hi = np.interp(x_hi, x_old2, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        
        t8a = min(t8_start + 60, t8 - 60)
        if t8a > t8_start + 5:
            h_hi, c_hi = lbfgs_phase(h_hi, n_hi, t8a,
                alphas=[2000, 10000], mus=[5000.0, 50000.0])
        
        t8b = min(t8_start + 120, t8 - 20)
        if t8b > t8_start + 60:
            h_hi, c_hi = active_set_optimize(h_hi, n_hi, t8b, max_iter=5000, K=6, lr_init=0.001)
        
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            n = n_hi
    
    # Phase 9: Final SLSQP polish
    if time.time() < deadline - 10:
        h_fin, c_fin = slqp_polish(best_h, len(best_h), deadline - 3, alpha=10000000, maxiter=3000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
