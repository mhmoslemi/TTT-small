    positions = set()
    for k in range(1, 6 * n):
        pos = int(k * alpha_frac) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample_nearest(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    indices = (np.arange(target_n) * n / target_n).astype(int)
    indices = np.clip(indices, 0, n - 1)
    h_new = h[indices].astype(float)
    return project_feasible(h_new, target_n)

def run_lbfgs_active(h, n, deadline, alphas, top_k=8, mu=0.001, maxit_per=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 20:
            break
        for sub in range(3):
            if time.time() > deadline - 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            active = get_active_lags(c_vec, n, top_k=top_k, tol_frac=0.002)
            if len(active) == 0:
                break
            maxit = min(maxit_per, max(100, int((deadline - time.time()) * 2)))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 40, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-14:
                    best_h = h_new
            except Exception:
                break
    return best_h, best_c

def run_lbfgs_full(h, n, deadline, alphas, mu=0.01, maxit_per=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 30:
            break
        maxit = min(maxit_per, max(50, int((deadline - time.time()) // 3)))
        try:
            res = minimize(full_obj, best_h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(h, n, deadline, max_rounds=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = list(np.where(best_h > 0.5)[0])
    zeros = list(np.where(best_h < 0.5)[0])
    np.random.seed(int(time.time() % 10000))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros[:max(10, len(zeros)//10)]:
                if best_h[i] < 0.1 or best_h[j] > 0.9:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_h = project_feasible(best_h, n)
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def block_move_search(h, n, deadline, block_sizes=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if block_sizes is None:
        block_sizes = [2, 3, 4, 5, 7, 10, 15]
    
    rng = np.random.RandomState(12345)
    
    for bsz in block_sizes:
        if time.time() > deadline:
            break
        n_trials = 2000
        for trial in range(n_trials):
            if time.time() > deadline:
                break
            a = rng.randint(0, n - bsz)
            c = rng.randint(0, n - bsz)
            if abs(a - c) < bsz:
                continue
            src = best_h[a:a+bsz].copy()
            dst = best_h[c:c+bsz].copy()
            if np.all(src == dst):
                continue
            best_h[a:a+bsz] = dst
            best_h[c:c+bsz] = src
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                best_h[a:a+bsz] = src
                best_h[c:c+bsz] = dst
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def pair_swap_search(h, n, deadline, radius=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = list(np.where(best_h > 0.5)[0])
    zeros = list(np.where(best_h < 0.5)[0])
    
    for sweep in range(20):
        if time.time() > deadline:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        ones_sub = ones[:max(20, len(ones)//5)]
        zeros_sub = zeros[:max(20, len(zeros)//5)]
        
        for i in ones_sub:
            if time.time() > deadline:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in zeros_sub:
                if j < lo or j >= hi:
                    continue
                if best_h[i] < 0.1 or best_h[j] > 0.9:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_h = project_feasible(best_h, n)
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def simulated_annealing(h, n, deadline, t_start=0.1, t_end=0.001, n_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    ones = list(np.where(cur_h > 0.5)[0])
    zeros = list(np.where(cur_h < 0.5)[0])
    rng = np.random.RandomState(42)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        progress = it / max(1, n_iter)
        temperature = t_start * (t_end / t_start) ** progress
        
        if not ones or not zeros:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i == j:
            continue
        
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cur_h = project_feasible(cur_h, n)
        c_new = compute_c5(cur_h, n)
        
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-10)):
            cur_c = c_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            cur_h = project_feasible(cur_h, n)
    
    return best_h, best_c

def gradient_descent_step(h, n, deadline, lr=0.001, n_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= max(1, len(active))
        
        pos_mask = (grad > 1e-8) & (best_h > 1e-8)
        neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
        
        if not pos_mask.any() or not neg_mask.any():
            break
        
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos_idx[:15]:
            gi = grad[i]
            for j in neg_idx[:15]:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        amt = min(lr, best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-10:
            break
        
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
    
    return best_h, best_c

def fine_tuning(h, n, deadline, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            ones = np.where(best_h > 0.01)[0]
            zeros = np.where(best_h < 0.99)[0]
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            for i in ones[:100]:
                if time.time() > deadline:
                    break
                for j in zeros[:100]:
                    if i == j:
                        continue
                    if best_h[i] - d >= 0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
                        if improved:
                            break
                if improved:
                    break
    return best_h, best_c

def multi_start_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas_coarse = [5, 20, 100, 500, 2000]
    alphas_fine = [10000, 50000, 200000, 1000000, 5000000]
    
    # Coarse L-BFGS
    p1_end = min(time.time() + 60, deadline - 200)
    h1, c1 = run_lbfgs_full(best_h, n, p1_end, alphas_coarse, mu=0.001, maxit_per=200)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Active set L-BFGS
    p2_end = min(time.time() + 120, deadline - 150)
    h2, c2 = run_lbfgs_active(best_h, n, p2_end, alphas_fine, top_k=8, mu=0.0005, maxit_per=150)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    
    best_overall_h = h0.copy()
    best_overall_c = compute_c5(h0, n_orig)
    best_overall_n = n_orig
    
    # Generate candidate starting points at n_orig
    candidates = []
    candidates.append(('parent', h0.copy()))
    
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    candidates.append(('beatty_gr', make_beatty(n_orig, golden / 2.0)))
    candidates.append(('beatty_gr2', make_beatty(n_orig, (golden - 1.0) / 2.0)))
    candidates.append(('beatty_sqrt2', make_beatty(n_orig, np.sqrt(2.0) / 2.0)))
    candidates.append(('beatty_sqrt3', make_beatty(n_orig, np.sqrt(3.0) / 2.0)))
    candidates.append(('beatty_pi', make_beatty(n_orig, np.pi / 4.0)))
    candidates.append(('rand1', make_random_binary(n_orig, rng)))
    candidates.append(('rand2', make_random_binary(n_orig, rng)))
    
    # Find best starting point
    best_start = h0.copy()
    best_start_c = compute_c5(h0, n_orig)
    for name, hc in candidates:
        c = compute_c5(hc, n_orig)
        if c < best_start_c:
            best_start_c = c
            best_start = hc.copy()
    
    # Phase 1: Optimize at original resolution
    p1_end = min(t0 + 200, deadline - 500)
    h1, c1 = multi_start_optimize(best_start, n_orig, p1_end)
    if c1 < best_overall_c:
        best_overall_c = c1
        best_overall_h = h1
        best_overall_n = n_orig
    
    # Phase 2: Local search at n_orig
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 100, deadline - 380)
        h2, c2 = pair_swap_search(best_overall_h, best_overall_n, p2_end, radius=60)
        if c2 < best_overall_c:
            best_overall_c = c2
            best_overall_h = h2
    
    # Phase 3: Block moves at n_orig
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 80, deadline - 280)
        h3, c3 = block_move_search(best_overall_h, best_overall_n, p3_end)
        if c3 < best_overall_c:
            best_overall_c = c3
            best_overall_h = h3
    
    # Phase 4: Simulated annealing at n_orig
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 80, deadline - 230)
        h4, c4 = simulated_annealing(best_overall_h, best_overall_n, p4_end,
                                      t_start=0.05, t_end=0.001, n_iter=30000)
        if c4 < best_overall_c:
            best_overall_c = c4
            best_overall_h = h4
    
    # Phase 5: Try higher resolution
    n_high = 1000
    if time.time() < deadline - 250:
        h_high = upsample_nearest(best_overall_h, n_high)
        h_high_c = compute_c5(h_high, n_high)
        
        if h_high_c <= best_overall_c + 1e-10:
            # Optimize at high resolution
            p5_end = min(time.time() + 120, deadline - 150)
            h5, c5 = multi_start_optimize(h_high, n_high, p5_end)
            if c5 < best_overall_c:
                best_overall_c = c5
                best_overall_h = h5
                best_overall_n = n_high
            
            # Local search at high resolution
            if time.time() < deadline - 120:
                p5b_end = min(time.time() + 60, deadline - 100)
                h5b, c5b = pair_swap_search(best_overall_h, best_overall_n, p5b_end, radius=50)
                if c5b < best_overall_c:
                    best_overall_c = c5b
                    best_overall_h = h5b
            
            if time.time() < deadline - 80:
                p5c_end = min(time.time() + 40, deadline - 60)
                h5c, c5c = block_move_search(best_overall_h, best_overall_n, p5c_end,
                                              block_sizes=[2, 3, 4, 5])
                if c5c < best_overall_c:
                    best_overall_c = c5c
                    best_overall_h = h5c
    
    # Phase 6: Gradient descent refinement
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 40, deadline - 40)
        h6, c6 = gradient_descent_step(best_overall_h, best_overall_n, p6_end,
                                         lr=0.001, n_steps=30000)
        if c6 < best_overall_c:
            best_overall_c = c6
            best_overall_h = h6
    
    # Phase 7: Final fine tuning
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 25, deadline - 5)
        h7, c7 = fine_tuning(best_overall_h, best_overall_n, p7_end,
                              deltas=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c7 < best_overall_c:
            best_overall_c = c7
            best_overall_h = h7
    
    # Final active set polish
    if time.time() < deadline - 10:
        p8_end = min(time.time() + 15, deadline - 2)
        alphas_final = [5000000, 20000000, 100000000]
        h8, c8 = run_lbfgs_active(best_overall_h, best_overall_n, p8_end,
                                    alphas_final, top_k=5, mu=0.0001, maxit_per=100)
        if c8 < best_overall_c:
            best_overall_c = c8
            best_overall_h = h8
    
    # Final projection and verification
    final_n = len(best_overall_h)
    best_overall_h = project_feasible(best_overall_h, final_n)
    best_overall_c = compute_c5(best_overall_h, final_n)
    
    assert np.all(best_overall_h >= 0.0) and np.all(best_overall_h <= 1.0)
    assert abs(best_overall_h.sum() - final_n / 2.0) < 1e-8
    
    return best_overall_h, best_overall_c, final_n
