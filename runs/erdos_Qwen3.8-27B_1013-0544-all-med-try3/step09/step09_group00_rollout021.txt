def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

        np.random.shuffle(ones)
        for i in ones[:n_ones - target]:
            h_bin[i] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target - n_ones]:
            h_bin[i] = 1
    h_bin = prox_project(h_bin, n)
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline: break
        improved = False
        # Find active lags
        c_vec = compute_c5_vec(h_bin, n)
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - 0.001*max_val)[0]
        
        # Try pair swaps
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        for i in ones[:50]:
            if time.time() > deadline: break
            for j in zeros[:50]:
                h_bin[i] = 0
                h_bin[j] = 1
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h_bin.copy()
                    improved = True
                    # Update sets
                    break
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        
        if not improved:
            break
    
    return best_h, best_c

# === L-BFGS ===

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smax_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smax_grad(c, alpha, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def run_lbfgs(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 1000, 10000, 100000, 1000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10: break
        maxit = max(200, int((deadline - time.time()) * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit*5,
                                   'maxls': 40, 'maxcor': 50})
            h_new = prox_project(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
        except:
            pass
    return best_h, best_c

# === Start generation ===

def make_beatty(n, alpha):
    h = np.zeros(n)
    positions = set()
    for k in range(1, 3*n):
        pos = int(k * alpha) % n
        positions.add(pos)
        if len(positions) >= n//2:
            break
    for pos in positions:
        h[pos] = 1.0
    return prox_project(h, n)

def make_thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        # t[i] = popcount(i) mod 2
        j = i
        cnt = 0
        while j:
            cnt += j & 1
            j >>= 1
        t[i] = cnt % 2
    # Make it have exactly n/2 ones
    # If not, adjust
    h = t.copy()
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in ones[:n_ones - target]:
            h[i] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target - n_ones]:
            h[i] = 1
    return prox_project(h, n)

def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

# === Coordinate descent ===

def coord_descent(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005, 0.00001]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in steps:
        if time.time() > deadline: break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Gradient direction
            active = np.where(c_vec >= max_val - 0.002*max_val)[0]
            dx = 2.0/n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n-1)
                if k >= 0:
                    nv = n-k
                    if nv > 0:
                        grad[:nv] += (1.0 - best_h[k:])*dx
                        grad[k:] += (-best_h[:nv])*dx
                else:
                    kk = -k
                    nv = n-kk
                    if nv > 0:
                        grad[kk:] += (1.0 - best_h[:nv])*dx
                        grad[:nv] += (-best_h[kk:])*dx
            grad /= len(active)
            
            pos = np.where((grad > 1e-8) & (best_h > 1e-6))[0]
            neg = np.where((grad < -1e-8) & (best_h < 1.0-1e-6))[0]
            if len(pos) == 0 or len(neg) == 0: break
            
            best_gain = 0
            best_i, best_j = -1, -1
            for i in pos[:20]:
                for j in neg[:20]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i, best_j = i, j
            if best_i < 0: break
            
            amt = min(step, best_h[best_i], 1.0-best_h[best_j])
            if amt < 1e-10: break
            
            best_h[best_i] -= amt
            best_h[best_j] += amt
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
    
    return best_h, best_c
