                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_mask(c_vec, frac=0.002, top_k=10):
    mx = c_vec.max()
    threshold = mx - frac * mx
    active = c_vec >= threshold
    if active.sum() >= min(top_k, len(c_vec)):
        return active
    k = min(top_k, len(c_vec))
    top = np.argsort(-c_vec)[:k]
    active = np.zeros(len(c_vec), dtype=bool)
    active[top] = True
    return active

def soft_obj(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

def soft_grad(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    act_idx = np.where(active)[0]
    for wi, idx in zip(w, act_idx):
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        if s >= 0:
            nv = n - s
            grad[:nv] += wi * dx * (1.0 - x[s:])
            grad[s:] += wi * dx * (-x[:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:] += wi * dx * (1.0 - x[:nv])
            grad[:nv] += wi * dx * (-x[kk:])
    return grad

def try_lbfgs(h, n, active, alpha, bounds, maxiter, deadline):
    if time.time() > deadline - 5:
        return h, compute_c5(h, n)
    try:
        res = minimize(soft_obj, h, jac=soft_grad, args=(n, alpha, active),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-18,
                                'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def pair_descent(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for eps in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline - 5:
            break
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + eps <= 1.0 and best_h[j] - eps >= 0.0:
                        best_h[i] += eps
                        best_h[j] -= eps
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= eps
                            best_h[j] += eps
                    if best_h[i] - eps >= 0.0 and best_h[j] + eps <= 1.0:
                        best_h[i] -= eps
                        best_h[j] += eps
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += eps
                            best_h[j] -= eps
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(seed)

    # Phase 1: Direct L-BFGS optimization at parent, increasing alpha
    c_vec = compute_c5_vec(best_h, n)
    active = get_active_mask(c_vec, frac=0.003, top_k=12)
    p1_alphas = [500, 2000, 10000, 50000, 200000, 500000]
    for alpha in p1_alphas:
        if time.time() > deadline - 350:
            break
        maxit = 4000
        h_new, c_new = try_lbfgs(best_h, n, active, alpha, bounds, maxit, deadline)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            c_vec = compute_c5_vec(best_h, n)
            active = get_active_mask(c_vec, frac=0.003, top_k=12)

    # Phase 2: Multi-restart with random perturbations
    n_restarts = 18
    for r in range(n_restarts):
        if time.time() > deadline - 250:
            break
        scale = 0.04 * (1.0 - 0.7 * r / float(n_restarts))
        pert = rng.randn(n) * scale
        h_pert = project_feasible(best_h + pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
            h_pert = h_pert

        c_vec = compute_c5_vec(h_pert, n)
        active = get_active_mask(c_vec, frac=0.004, top_k=12)
        for alpha in [5000, 30000, 100000]:
            if time.time() > deadline - 200:
                break
            maxit = max(1500, int((deadline - time.time()) / 25))
            h_new, c_new = try_lbfgs(h_pert, n, active, alpha, bounds, maxit, deadline)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
                h_pert = h_new
                c_vec = compute_c5_vec(h_pert, n)
                active = get_active_mask(c_vec, frac=0.004, top_k=12)

    # Phase 3: Structured perturbations
    # Reflection
    h_ref = project_feasible(best_h[::-1].copy(), n)
    c_ref = compute_c5(h_ref, n)
    if c_ref < best_c:
        best_c = c_ref
        best_h = h_ref
    # Complement
    h_comp = project_feasible(1.0 - best_h, n)
    c_comp = compute_c5(h_comp, n)
    if c_comp < best_c:
        best_c = c_comp
        best_h = h_comp
    # Small shifts
    for s in [1, 2, 3, -1, -2, -3]:
        h_sh = project_feasible(np.roll(best_h, s), n)
        c_sh = compute_c5(h_sh, n)
        if c_sh < best_c:
            best_c = c_sh
            best_h = h_sh
            break

    # Phase 4: Upsample to 2n and optimize
    if time.time() < deadline - 180:
        n2 = 2 * n
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2)
        c_up = compute_c5(h_up, n2)
        if c_up < best_c + 0.002:
            bounds2 = [(0.0, 1.0)] * n2
            c_vec2 = compute_c5_vec(h_up, n2)
            active2 = get_active_mask(c_vec2, frac=0.004, top_k=15)
            for alpha in [2000, 10000, 50000, 200000]:
                if time.time() > deadline - 100:
                    break
                maxit = 2500
                h_new, c_new = try_lbfgs(h_up, n2, active2, alpha, bounds2, maxit, deadline)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
                    n = n2
                    bounds = bounds2
                    c_vec = compute_c5_vec(best_h, n)
                    active = get_active_mask(c_vec, frac=0.003, top_k=12)

    # Phase 5: Additional restarts from best (if still improving)
    if time.time() < deadline - 80:
        for r in range(8):
            if time.time() > deadline - 60:
                break
            scale = 0.015 * (1.0 - r / 8.0)
            pert = rng.randn(n) * scale
            h_pert = project_feasible(best_h + pert, n)
            c_vec = compute_c5_vec(h_pert, n)
            active = get_active_mask(c_vec, frac=0.002, top_k=8)
            for alpha in [50000, 200000]:
                if time.time() > deadline - 50:
                    break
                maxit = 2000
                h_new, c_new = try_lbfgs(h_pert, n, active, alpha, bounds, maxit, deadline)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new

    # Phase 6: Pair descent
    if time.time() < deadline - 30:
        h_pd, c_pd = pair_descent(best_h, n, deadline - 10, radius=35)
        if c_pd < best_c:
            best_c = c_pd
            best_h = h_pd

    # Phase 7: Final high-alpha polish
    if time.time() < deadline - 8:
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_mask(c_vec, frac=0.0005, top_k=5)
        for alpha in [500000, 2000000, 5000000]:
            if time.time() > deadline - 4:
                break
            maxit = 1500
            h_new, c_new = try_lbfgs(best_h, n, active, alpha, bounds, maxit, deadline)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
