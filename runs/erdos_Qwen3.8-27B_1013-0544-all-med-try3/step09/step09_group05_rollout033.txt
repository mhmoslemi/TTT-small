def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_active_shifts(c_vec, top_k=12, tol_frac=0.005):
    """Find shifts that are near the maximum."""
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, tol_frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def topk_smooth_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 300:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_smooth_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_off = n - 1
    for wi, idx in zip(w, active_idx):
        k = idx - n_off
        wm = wi * dx

                grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def paired_coordinate_descent(h, n, deadline, max_rounds=2000):
    """Paired coordinate descent: move mass between pairs of coordinates."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for _ in range(max_rounds):
        if time.time() > deadline - 3:
            break
        
        c_vec = compute_c5_vec(h, n)
        active = find_active_shifts(c_vec, top_k=8, tol_frac=0.003)
        
        if len(active) < 2:
            break
        
        # Compute aggregate gradient for active shifts
        dx = 2.0 / n
        n_off = n - 1
        grad = np.zeros(n)
        for idx in active:
            k = idx - n_off
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Find top candidates by gradient magnitude
        top_indices = np.argsort(-np.abs(grad))[:80]
        
        improved = False
        for d in steps[:3]:  # Try only a few step sizes per round
            for ii in range(len(top_indices)):
                if time.time() > deadline - 3:
                    break
                i = top_indices[ii]
                gi = grad[i]
                # Determine direction: if gi > 0, decreasing h[i] helps
                # if gi < 0, increasing h[i] helps
                if gi > 0:
                    di = -1  # decrease h[i]
                else:
                    di = 1   # increase h[i]
                
                # Find a good partner j
                for jj in range(len(top_indices)):
                    if ii == jj:
                        continue
                    j = top_indices[jj]
                    gj = grad[j]
                    # We want to move h[i] in direction di and h[j] in opposite direction
                    # to preserve the sum
                    dj = -di  # opposite direction
                    
                    if di == 1:  # h[i] increases
                        if h[i] + d > 1.0 or h[j] - d < 0.0:
                            continue
                    else:  # h[i] decreases
                        if h[i] - d < 0.0 or h[j] + d > 1.0:
                            continue
                    
                    h[i] += di * d
                    h[j] += dj * d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= di * d
                        h[j] -= dj * d
            
            if improved:
                break
    
    return h, best_c

def subgradient_method(h, n, deadline, max_iter=5000):
    """Projected subgradient method with adaptive step size."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_init = best_c
    lower_bound = 0.0  # Trivial lower bound
    t = 0
    
    for t in range(max_iter):
        if time.time() > deadline - 5:
            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        dx = 2.0 / n
        n_off = n - 1
        
        # Find active set
        threshold = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            k = idx - n_off
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Polyak step size
        gap = best_c - lower_bound
        if gap < 1e-10:
            gap = 1e-10
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = gap / (grad_norm ** 2)
        step = min(step, 0.1)  # Cap step size
        step = max(step, 0.001)  # Floor step size
        
        # Take step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def multi_start_optimize(h_best, n, deadline, n_restarts=5):
    """Try perturbations of the best solution and re-optimize."""
    rng = np.random.default_rng(12345)
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()
    
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        
        # Create perturbation
        scale = 0.02 * (1 - r / n_restarts)
        h_pert = h_best + rng.standard_normal(n) * scale
        h_pert = project_feasible(h_pert, n)
        
        # Quick optimization
        t_sub = min(60, (deadline - time.time()) / max(1, n_restarts - r))
        h_opt, c_opt = iterative_topk_lbfgs(h_pert, n, time.time() + t_sub, max_rounds=50)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def binary_proj(h, n):
    """Project towards binary: threshold at 0.5 with smoothing."""
    # Values far from 0.5 get pushed towards 0 or 1
    h_bin = h.copy()
    mask_high = h_bin > 0.7
    mask_low = h_bin < 0.3
    h_bin[mask_high] = np.minimum(1.0, h_bin[mask_high] + 0.1)
    h_bin[mask_low] = np.maximum(0.0, h_bin[mask_low] - 0.1)
    return project_feasible(h_bin, n)
