        m_range = np.arange(m_start, m_end + 1)
        offset = m_range - i

        slopes = 1.0 - 2.0 * h_new[offset]
        intercepts = c[m_range].copy()

        idx_2i = 2 * i
        has_2i = (m_start <= idx_2i <= m_end)
        if has_2i:
            pos_2i = idx_2i - m_start
            lin_mask = np.ones(len(m_range), dtype=bool)
            lin_mask[pos_2i] = False
        else:
            lin_mask = None

        mask_aff = np.zeros(n_corr, dtype=bool)
        mask_aff[m_range] = True
        if np.any(~mask_aff):
            max_unaff = float(c[~mask_aff].max())
        else:
            max_unaff = -1e300

        if lin_mask is not None and lin_mask.any():
            vals = intercepts[lin_mask] + v_grid[:, None] * slopes[lin_mask]
            max_lin = vals.max(axis=1)
        elif lin_mask is not None:
            max_lin = np.full(len(v_grid), -1e300)
        else:
            vals = intercepts + v_grid[:, None] * slopes
            max_lin = vals.max(axis=1)

        if has_2i:
            quad = c[idx_2i] - old * (1.0 - old) + v_grid - v_grid ** 2
            f_vals = np.maximum(np.maximum(max_lin, max_unaff), quad)
        else:
            f_vals = np.maximum(max_lin, max_unaff)

        best_idx = int(np.argmin(f_vals))
        new_v = v_grid[best_idx]

        if 0 < best_idx < n_vpoints - 1:
            lo_v = v_grid[best_idx - 1]
            hi_v = v_grid[best_idx + 1]
            v_ref = np.linspace(lo_v, hi_v, 300)
            if lin_mask is not None and lin_mask.any():
                vals_r = intercepts[lin_mask] + v_ref[:, None] * slopes[lin_mask]
                max_lin_r = vals_r.max(axis=1)
            else:
                vals_r = intercepts + v_ref[:, None] * slopes
                max_lin_r = vals_r.max(axis=1)
            if has_2i:
                quad_r = c[idx_2i] - old * (1.0 - old) + v_ref - v_ref ** 2
                f_r = np.maximum(np.maximum(max_lin_r, max_unaff), quad_r)
            else:
                f_r = np.maximum(max_lin_r, max_unaff)
            new_v = float(v_ref[np.argmin(f_r)])

        new_v = float(np.clip(new_v, 0.0, 1.0))

        if abs(new_v - old) > 1e-15:
            h_new[i] = new_v
            dv = new_v - old
            if has_2i:
                if lin_mask.any():
                    c[m_range[lin_mask]] += dv * (1.0 - 2.0 * h_new[m_range[lin_mask] - i])
                c[idx_2i] += new_v * (1.0 - new_v) - old * (1.0 - old)
            else:
                c[m_range] += dv * (1.0 - 2.0 * h_new[m_range - i])

    return h_new, c


def soft_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def soft_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            if nv > 0:
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            if nv > 0:
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 100.0), (2000, 500.0), (10000, 5000.0), (50000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(soft_max_obj, h, jac=soft_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    def sq_fn(x):
        return float(x.sum() - n / 2.0)

    def sq_jac_fn(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def targeted_local_search(h, n, deadline, center_radius=40):
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = int(idx) - (n - 1)
                if k >= 0:
                    nv = n - k
                    if nv > 0:
                        grad[0:nv] += (1.0 - h[k:n]) * dx_val
                        grad[k:n] += (-h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    if nv > 0:
                        grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                        grad[0:nv] += (-h[kk:n]) * dx_val
            if len(active) > 0:
                grad /= len(active)
            top_pos = np.argsort(-np.abs(grad))[:min(center_radius * 2, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Coordinate descent on n=500 (60s)
    t1 = min(t0 + 60, deadline - 800)
    if t1 > time.time() + 5:
        c_vec = compute_c5_vec(best_h, n_orig)
        for sweep in range(200):
            if time.time() > t1:
                break
            best_h, c_vec = coord_desc_sweep(best_h, n_orig, c_vec, n_vpoints=101)
            best_h = project_feasible(best_h, n_orig)
            c_new = compute_c5(best_h, n_orig)
            if c_new < best_c:
                best_c = c_new
            c_vec = compute_c5_vec(best_h, n_orig)

    # Phase 2: L-BFGS soft-max refinement (80s)
    t2 = min(t0 + 140, deadline - 700)
    if t2 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_orig, t2,
            configs=[(500, 100.0), (5000, 1000.0), (20000, 5000.0), (80000, 20000.0)])

    # Phase 3: More coordinate descent (60s)
    t3 = min(t0 + 200, deadline - 600)
    if t3 > time.time() + 5:
        c_vec = compute_c5_vec(best_h, n_orig)
        for sweep in range(150):
            if time.time() > t3:
                break
            best_h, c_vec = coord_desc_sweep(best_h, n_orig, c_vec, n_vpoints=201)
            best_h = project_feasible(best_h, n_orig)
            c_new = compute_c5(best_h, n_orig)
            if c_new < best_c:
                best_c = c_new
            c_vec = compute_c5_vec(best_h, n_orig)

    # Phase 4: Upsample to n=1000
    n_high = 1000
    if time.time() < deadline - 400:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high > best_c:
            c_high = best_c  # track best across resolutions

        # L-BFGS on high res (80s)
        t4 = min(time.time() + 80, deadline - 250)
        if t4 > time.time() + 5:
            h_high, c_high = run_lbfgs(h_high, n_high, t4,
                configs=[(500, 100.0), (5000, 1000.0), (20000, 5000.0)])

        # Coordinate descent on high res (60s)
        t5 = min(time.time() + 60, deadline - 150)
        if t5 > time.time() + 5:
            c_vec_h = compute_c5_vec(h_high, n_high)
            for sweep in range(80):
                if time.time() > t5:
                    break
                h_high, c_vec_h = coord_desc_sweep(h_high, n_high, c_vec_h, n_vpoints=101)
                h_high = project_feasible(h_high, n_high)
                c_new_h = compute_c5(h_high, n_high)
                if c_new_h < c_high:
                    c_high = c_new_h
                c_vec_h = compute_c5_vec(h_high, n_high)

        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            n_orig = n_high

    # Phase 5: SLSQP polish (50s)
    t6 = min(time.time() + 50, deadline - 60)
    if t6 > time.time() + 5:
        h_s, c_s = slqp_polish(best_h, len(best_h), t6, alpha=2000000, maxiter=3000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 6: Final coordinate descent (40s)
    t7 = min(time.time() + 40, deadline - 20)
    if t7 > time.time() + 3:
        c_vec_f = compute_c5_vec(best_h, len(best_h))
        for sweep in range(100):
            if time.time() > t7:
                break
            best_h, c_vec_f = coord_desc_sweep(best_h, len(best_h), c_vec_f, n_vpoints=201)
            best_h = project_feasible(best_h, len(best_h))
            c_new_f = compute_c5(best_h, len(best_h))
            if c_new_f < best_c:
                best_c = c_new_f
            c_vec_f = compute_c5_vec(best_h, len(best_h))

    # Phase 7: Local search polish (remaining time)
    if time.time() < deadline - 5:
        best_h, best_c = targeted_local_search(best_h, len(best_h), deadline - 2, center_radius=30)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
