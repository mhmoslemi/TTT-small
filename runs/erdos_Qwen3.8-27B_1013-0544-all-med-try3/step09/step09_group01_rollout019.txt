    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu, maxiters):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-17, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_phase(h, n, deadline, top_k_list=(4, 8, 15, 30)):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50000, 200000, 1000000]
    
    for top_k in top_k_list:
        for _round in range(4):
            remaining = deadline - time.time()
            if remaining < 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.argsort(-c_vec[active])[:top_k]
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            
            for alpha in alphas:
                if time.time() > deadline - 10:
                    break
                maxit = max(300, int((deadline - time.time()) * 2))
                try:
                    res = minimize(active_obj, best_h, jac=active_grad,
                                  args=(n, active, alpha, 0.01), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-20,
                                           'gtol': 1e-17, 'maxfun': maxit * 8,
                                           'maxls': 40, 'maxcor': 50})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    return best_h, best_c

def coord_descent(h, n, deadline, mu=0.5, max_sweeps=10000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    K = 2 * n - 1
    
    t_start = time.time()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        one_minus_h = 1.0 - best_h
        
        for j in range(n):
            # Compute coefficient vector a_j (length K)
            a = np.zeros(K)
            # Part 1: (1-h[j-k])*dx for 0 <= j-k < n
            k_lo = max(0, j - n + 1)
            k_hi = min(j, K - 1)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi + 1)
                a[k_lo:k_hi+1] += one_minus_h[j - ks] * dx
            # Part 2: -h[j+k]*dx for 0 <= j+k < n
            k_lo2 = max(0, -j)
            k_hi2 = min(n - 1 - j, K - 1)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2 + 1)
                a[k_lo2:k_hi2+1] -= best_h[j + ks2] * dx
            
            # b_j = c_vec - a_j * h[j]
            b = c_vec - a * best_h[j]
            
            # Solve min_{0<=t<=1} max_k(a[k]*t + b[k]) + 2*mu*(sum(h)-n/2)*t
            lam = 2.0 * mu * (best_h.sum() - n / 2.0)
            A = a + lam
            
            # Ternary search for 1D convex minimization
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A * m1 + b)
                f2 = np.max(A * m2 + b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_star = (lo + hi) / 2.0
            best_h[j] = t_star
        
        # Project to ensure feasibility
        best_h = project_feasible(best_h, n)
        c_now = compute_c5(best_h, n)
        if c_now < best_c:
            best_c = c_now
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1500):
    def slq_obj(x, nn, al):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        return float(m + np.log(np.sum(np.exp(sh))) / al)
    
    def sq_f(x, nn):
        return x.sum() - nn / 2.0
    
    def sq_j(x, nn):
        return np.ones(nn)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_f, 'jac': sq_j, 'args': (n,)}]
    try:
        res = minimize(slq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        best_h = h_new
    
    return best_h, best_c

def make_binary_from(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def smooth_binary(h, n, sigma=3):
    """Smooth a binary-like pattern with a Gaussian kernel."""
    kernel = np.exp(-0.5 * (np.arange(-10, 11) / sigma) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    return project_feasible(h_smooth, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()

    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS warmup at parent resolution
    p1_end = min(t0 + 120, deadline - 600)
    h1, c1 = lbfgs_phase(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 50000],
        mu=0.01, maxiters=[2000, 2000, 3000, 3000, 4000])
    if c1 < best_c:
        best_c, best_h = c1, h1
    
    # Phase 2: High-alpha L-BFGS for precision
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 250, deadline - 400)
        h2, c2 = lbfgs_phase(best_h, best_n, p2_end,
            alphas=[100000, 500000, 2000000, 5000000],
            mu=0.005, maxiters=[4000, 5000, 5000, 5000])
        if c2 < best_c:
            best_c, best_h = c2, h2
    
    # Phase 3: Active set optimization
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 150, deadline - 300)
        h3, c3 = active_set_phase(best_h, best_n, p3_end,
            top_k_list=(4, 8, 15, 30))
        if c3 < best_c:
            best_c, best_h = c3, h3
    
    # Phase 4: Coordinate descent (key new method)
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 200, deadline - 180)
        h4, c4 = coord_descent(best_h, best_n, p4_end, mu=0.3, max_sweeps=5000)
        if c4 < best_c:
            best_c, best_h = c4, h4
    
    # Phase 5: Subgradient refinement
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5v = subgrad_refine(best_h, best_n, p5_end, n_iter=3000)
        if c5v < best_c:
            best_c, best_h = c5v, h5
    
    # Phase 6: Fine pair search
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 60, deadline - 50)
        h6, c6 = pair_search(best_h, best_n, p6_end, radius=40)
        if c6 < best_c:
            best_c, best_h = c6, h6
    
    # Phase 7: Try n=200 (lower resolution may give better discrete C5)
    n_low = 200
    if time.time() < deadline - 200:
        x_coarse = np.linspace(0, 2, n_low)
        x_fine = np.linspace(0, 2, best_n)
        h_low = np.interp(x_coarse, x_fine, best_h)
        h_low = project_feasible(h_low, n_low)
        c_low = compute_c5(h_low, n_low)
        
        p7_end = min(time.time() + 120, deadline - 150)
        h7, c7 = lbfgs_phase(h_low, n_low, p7_end,
            alphas=[100, 1000, 10000, 100000, 500000],
            mu=0.01, maxiters=[2000, 3000, 3000, 4000, 4000])
        if c7 < best_c:
            best_c, best_h, best_n = c7, h7, n_low
        
        if time.time() < deadline - 50:
            p7b_end = min(time.time() + 50, deadline - 40)
            h7b, c7b = coord_descent(best_h, best_n, p7b_end, mu=0.3, max_sweeps=2000)
            if c7b < best_c:
                best_c, best_h = c7b, h7b
    
    # Phase 8: Try n=1000 (higher resolution)
    n_high = 1000
    if time.time() < deadline - 150 and best_n != n_high:
        x_fine2 = np.linspace(0, 2, n_high)
        x_cur = np.linspace(0, 2, best_n)
        h_high = np.interp(x_fine2, x_cur, best_h)
        h_high = project_feasible(h_high, n_high)
        
        p8_end = min(time.time() + 100, deadline - 100)
        h8, c8 = lbfgs_phase(h_high, n_high, p8_end,
            alphas=[50, 500, 5000, 50000],
            mu=0.005, maxiters=[2000, 3000, 3000, 4000])
        if c8 < best_c:
            best_c, best_h, best_n = c8, h8, n_high
    
    # Phase 9: Final coordinate descent on best
    if time.time() < deadline - 30:
        p9_end = min(time.time() + 30, deadline - 20)
        h9, c9 = coord_descent(best_h, best_n, p9_end, mu=0.5, max_sweeps=2000)
        if c9 < best_c:
            best_c, best_h = c9, h9
    
    # Phase 10: SLSQP final polish
    if time.time() < deadline - 10:
        p10_end = deadline - 3
        h10, c10 = slqp_polish(best_h, best_n, p10_end, alpha=2000000, maxiter=800)
        if c10 < best_c:
            best_c, best_h = c10, h10
    
    # Phase 11: Final tiny pair search
    if time.time() < deadline - 2:
        h11, c11 = pair_search(best_h, best_n, deadline - 1, radius=20)
        if c11 < best_c:
            best_c, best_h = c11, h11
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
