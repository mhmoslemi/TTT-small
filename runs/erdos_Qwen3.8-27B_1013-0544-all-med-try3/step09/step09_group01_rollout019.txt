def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_exact(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2} using bisection."""
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    
    def sum_at_shift(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    
    lo = h.min() - 0.01
    hi = h.max() + 0.01
    
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if sum_at_shift(mid) > target:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) * 0.5
    result = np.clip(h - t, 0.0, 1.0)
    return result

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    val = m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wm * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

import time

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_exact(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-15:
            continue
        k = mm - (n - 1)
        wm *= dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    vals = c[active]
    mx = vals.max()

        h_new = project_exact(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def slqp_solve(h, n, deadline, alpha, maxiter):
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    def sq_jac(x, nn):
        return np.ones(nn)
    def obj(x, nn, a):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_exact(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_exact(h.copy(), n), compute_c5(h, n)

def coord_descent(h, n, deadline, radius=60):
    best_h = project_exact(h, n)
    best_c = compute_c5(best_h, n)
    h = best_h.copy()
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        old_i, old_j = h[i], h[j]
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        h_p = project_exact(h, n)
                        c = compute_c5(h_p, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_p
                            h = h_p.copy()
                            improved = True
                        else:
                            h[i], h[j] = old_i, old_j
    return best_h, best_c

def binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += max(0.0, min(1.0, remaining))
    return project_exact(h_bin, n)

def restart_optimize(h, n, deadline, sigma):
    best_h = project_exact(h, n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    num_restarts = 3
    per_budget = (deadline - time.time()) / num_restarts
    for rs in range(num_restarts):
        if time.time() > deadline:
            break
        perturbed = best_h + rng.randn(n) * sigma
        perturbed = project_exact(perturbed, n)
        c_pert = compute_c5(perturbed, n)
        end = min(time.time() + per_budget * 0.6, deadline)
        if time.time() < end - 5:
            h_r, c_r = lbfgs_phase(perturbed, n, end,
                alphas=[1000, 10000, 100000, 500000],
                mu=0.005, maxit_list=[500, 800, 800, 1000])
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
        end2 = min(time.time() + per_budget * 0.3, deadline)
        if time.time() < end2 - 5:
            h_r2, c_r2 = active_phase(best_h, n, end2, top_k=10,
                alphas=[50000, 200000], rounds=2)
            if c_r2 < best_c:
                best_c = c_r2
                best_h = h_r2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(500, 0.5)
    n = len(h0)

    h0 = project_exact(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Main L-BFGS soft-max annealing (300s)
    p1_end = t0 + 300
    h1, c1 = lbfgs_phase(best_h, n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.005, maxit_list=[2000, 2500, 3000, 3000, 3500, 4000, 4000, 5000])
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Active-set focused optimization (180s)
    if time.time() < deadline - 320:
        p2_end = min(time.time() + 180, deadline - 280)
        h2, c2 = active_phase(best_h, n, p2_end, top_k=14,
            alphas=[20000, 100000, 500000, 2000000], rounds=6)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Subgradient refinement (60s)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 60, deadline - 180)
        h3, c3 = subgrad_refine(best_h, n, p3_end, n_iter=5000)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Active-set again with different top_k (100s)
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 100, deadline - 120)
        h4, c4 = active_phase(best_h, n, p4_end, top_k=8,
            alphas=[100000, 500000, 2000000], rounds=4)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: SLSQP polish (80s)
    if time.time() < deadline - 100:
        for alpha in [200000, 1000000, 5000000]:
            if time.time() > deadline - 90:
                break
            p5_end = min(time.time() + 25, deadline - 80)
            h5, c5 = slqp_solve(best_h, n, p5_end, alpha=alpha, maxiter=1200)
            if c5 < best_c:
                best_c, best_h = c5, h5

    # Phase 6: Restart with small perturbations (80s)
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 80, deadline - 60)
        h6, c6 = restart_optimize(best_h, n, p6_end, sigma=0.02)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: Coordinate descent (60s)
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 60, deadline - 30)
        h7, c7 = coord_descent(best_h, n, p7_end, radius=50)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Final active-set + SLSQP (30s)
    if time.time() < deadline - 10:
        p8_end = min(time.time() + 15, deadline - 5)
        h8, c8 = active_phase(best_h, n, p8_end, top_k=6,
            alphas=[500000, 5000000], rounds=2)
        if c8 < best_c:
            best_c, best_h = c8, h8
        if time.time() < deadline - 3:
            h9, c9 = slqp_solve(best_h, n, deadline - 2, alpha=5000000, maxiter=500)
            if c9 < best_c:
                best_c, best_h = c9, h9

    # Final projection
    best_h = project_exact(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
