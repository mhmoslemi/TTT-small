        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += -wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += -wm * x[kk:n]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += -wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += -wm * x[kk:n]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=10.0, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        iters = min(maxiter, int(remaining * 50))
        if iters < 50:
            break
        try:
            res = minimize(softmax_obj, h, jac=softmax_grad,
                          args=(n, a, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': iters * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_topk(h, n, deadline, top_k=10, alphas=None):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [1000, 5000, 20000, 100000]
    
    for round_i in range(5):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order][:top_k]
        
        if len(active) == 0:
            break
            
        for a in alphas:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(topk_obj, h, jac=topk_grad,
                              args=(n, active, a, 1.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 1000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 10000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def local_pair_search(h, n, deadline, max_radius=40):
    """Local search with pair mass transfers."""
    h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            # Focus on shifts near the maximum
            threshold = max_val - 0.01 * max_val
            active_shifts = np.where(c_vec >= threshold)[0]
            
            # Find indices to modify: those involved in active shifts
            candidates = set()
            for idx in active_shifts:
                k = idx - (n - 1)
                if k >= 0:
                    candidates.update(range(max(0, k - max_radius), min(n, n - k + max_radius)))
                else:
                    kk = -k
                    candidates.update(range(max(0, kk - max_radius), min(n, n - kk + max_radius)))
            
            # Limit candidates
            cand_list = list(candidates)[:200]
            
            for i in cand_list:
                if time.time() > deadline:
                    return h, best_c
                for j in range(max(0, i-10), min(n, i+11)):
                    if i == j:
                        continue
                    # Try h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step; h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= step; h[j] += step
                    # Try h[i] -= step, h[j] += step
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step; h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += step; h[j] -= step
    return h, best_c

def run_slqp(h, n, deadline, alpha=100000, maxiter=2000):
    def sq(x, n):
        return x.sum() - n / 2.0
    def sq_jac(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, jac=softmax_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
