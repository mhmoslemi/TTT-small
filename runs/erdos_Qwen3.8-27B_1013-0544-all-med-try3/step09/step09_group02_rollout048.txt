So yes, the original code has a bug in the k >= 0 case! The gradient is wrong.

But wait - the code still achieves 0.380861. How? 

Maybe the bug causes it to effectively compute the gradient of out[n-1-k] instead of out[n-1+k], which is the overlap for the opposite shift. Since we're taking the max over ALL shifts, and the function is symmetric (the set of all overlaps for +k and -k are both in the array), using the wrong sign of the shift might still give a valid descent direction (just for a different constraint).

Actually, I think the issue is more subtle. The "active set" in the SQP includes indices where c_vec is near the max. If the gradient for index m is actually the gradient of c_vec[n-2-m] (the symmetric counterpart), then we're still reducing a constraint that's in the active set (assuming the active set is symmetric, which it approximately is).

def shift_grad(h, n, m_idx):
    """Correct gradient of c_vec[m_idx] w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = m_idx - (n - 1)  # effective shift
    if k >= 0:
        nv = n - k
        # j in [k, n-1]: (1-h[j-k])*dx
        grad[k:n] += (1.0 - h[0:nv]) * dx
        # j in [0, n-1-k]: -h[j+k]*dx
        grad[0:nv] += (-h[k:n]) * dx
    else:
        mk = -k
        nv = n - mk
        # j in [0, n-1-mk]: (1-h[j+mk])*dx
        grad[0:nv] += (1.0 - h[mk:n]) * dx
        # j in [mk, n-1]: -h[j-mk]*dx
        grad[mk:n] += (-h[0:nv]) * dx
    return grad

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_correct(h, n, m_idx):
    """Gradient of c_vec[m_idx] w.r.t. h (CORRECTED)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = m_idx - (n - 1)

    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros(n)

            continue
        grad += w[mm] * shift_grad_correct(h, n, mm)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def sqp_step(h, n, active_idx, rho):

    c_vec = compute_c5_vec(h, n)
    n_act = len(active_idx)
    if n_act == 0:
        return None, None
    
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    total_ineq = n_act + 4 * n
    A_ub = np.zeros((total_ineq, n_var))
    b_ub = np.zeros(total_ineq)
    row = 0
    
    for idx in active_idx:
        g = shift_grad_correct(h, n, idx)
        A_ub[row, 0:n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c_vec[idx]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = rho
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = rho
        row += 1
    
    A_eq = np.zeros((1, n_var))
