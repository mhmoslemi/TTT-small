from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-10:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-10:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_value(c, alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * x)
        elif lag > 0:
            g = lag
            grad[0:n-g] += w[k] * (-x[g:n])
            grad[g:n] += w[k] * (1.0 - x[0:n-g])
        else:
            g = -lag
            grad[0:n-g] += w[k] * (1.0 - x[g:n])
            grad[g:n] += w[k] * (-x[0:n-g])
    grad *= dx
    return grad

def _obj_func(x, n, alpha):
    return smooth_max_obj(x, n, alpha)

def _grad_func(x, n, alpha):
    return smooth_max_grad(x, n, alpha)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, n, alpha, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_func, h, args=(n, alpha), jac=_grad_func,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def compute_lag_gradient_contribution(h, n, k_pos, m):
    """Compute ∂c/∂h at position k_pos for lag m (signed)."""
    if m > 0:
        # c[m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        contrib = 0.0
        if k_pos >= m:
            contrib += (1.0 - h[k_pos - m])
        if k_pos <= n - 1 - m:
            contrib += (-h[k_pos + m])
        return contrib
    elif m < 0:
        a = -m
        # c[-a] = sum_{i=0}^{n-1-a} h[i]*(1-h[i+a])
        contrib = 0.0
        if k_pos <= n - 1 - a:
            contrib += (1.0 - h[k_pos + a])
        if k_pos >= a:
            contrib += (-h[k_pos - a])
        return contrib
    else:
        return 1.0 - 2.0 * h[k_pos]

def transfer_change(h, n, src, dst, delta, m):
    """Exact change in correlation at lag m when h[src] -= delta, h[dst] += delta."""
    dc_src = compute_lag_gradient_contribution(h, n, src, m)
    dc_dst = compute_lag_gradient_contribution(h, n, dst, m)
    return delta * (dc_dst - dc_src)

def fast_local_search(h, n, deadline, rng, max_steps=200000):
    h = h.copy()
    c_all = np.correlate(h, 1 - h, mode='full')  # without dx factor for speed
    dx = 2.0 / n
    best_c_val = float(np.max(c_all))  # in units of dx
    best_h = h.copy()
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        # Find worst lag
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            # Try to make h more binary (reduce h(1-h))
            # Find position with h closest to 0.5
            k = int(np.argmax(np.abs(h - 0.5)))
            if h[k] > 0.5:
                # Decrease h[k], increase some other position
                j = int(np.argmin(h))
                delta = min(0.01, (h[k] - h[j]) * 0.3)
            else:
                j = int(np.argmax(h))
                delta = min(0.01, (h[j] - h[k]) * 0.3)
            if delta < 1e-10:
                break
        else:
            m = abs(lag)
            # Find a good transfer for this lag
            # For lag m > 0: c = sum h[i+m]*(1-h[i]), want to reduce
            # Best: decrease h[i+m] where h[i] is large, or increase h[i] where h[i+m] is large
            # Transfer from high h to low h at positions (i, i+m)
            
            # Pick random i
            i = rng.randint(0, n - m)
            
            if lag > 0:
                # h[i+m] is the "source" to decrease, h[i] is the "dest" to increase
                src = i + m
                dst = i
            else:
                # h[i] is the "source" to decrease, h[i+m] is the "dest" to increase
                src = i
                dst = i + m
            
            if h[src] <= h[dst]:
                continue
            
            delta = min(0.02, (h[src] - h[dst]) * 0.2)
            if delta < 1e-10:
                continue
        
        # Compute new correlation for all lags
        # Only lags affected by src and dst change
        new_c = c_all.copy()
        for k in range(2 * n - 1):
            lag_k = k - (n - 1)
            change = transfer_change(h, n, src, dst, delta, lag_k)
            if abs(change) > 1e-15:
                new_c[k] += change
        
        new_max = float(np.max(new_c))
        
        if new_max < best_c_val - 1e-14:
            # Accept
            h[src] -= delta
            h[dst] += delta
            c_all = new_c
            best_c_val = new_max
            best_h = h.copy()
        else:
            # Try smaller delta
            delta *= 0.3
            if delta > 1e-10:
                new_c2 = c_all.copy()
                for k in range(2 * n - 1):
                    lag_k = k - (n - 1)
                    change = transfer_change(h, n, src, dst, delta, lag_k)
                    if abs(change) > 1e-15:
                        new_c2[k] += change
                new_max2 = float(np.max(new_c2))
                if new_max2 < best_c_val - 1e-14:
                    h[src] -= delta
                    h[dst] += delta
                    c_all = new_c2
                    best_c_val = new_max2
                    best_h = h.copy()
    
    return best_h, float(best_c_val * dx)
