    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    return w

    c = compute_all_lags(h, n)

            h_high_opt = lbfgsb_optimize(h_high, n_high, 100000, time.time() + remaining_high * 0.3,
                                         maxiter=2000, penalty_coeff=1e10)
            h_high_opt = lbfgsb_optimize(h_high_opt, n_high, 10000000, time.time() + remaining_high * 0.2,
                                         maxiter=1500, penalty_coeff=1e12)
            c_high = compute_c5(h_high_opt, n_high)
            if c_high < best_c:
                best_c = c_high
                best_h = h_high_opt
                # Local search at high res
                if time.time() < deadline - 30:
                    h_opt, c_opt = targeted_local_search(best_h, n_high,
                                                         min(deadline - 10, time.time() + 40),
                                                         rng, K_lags=6)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt
    
    # Phase 7: Another local search pass at the best resolution
    n_final = len(best_h)
    if time.time() < deadline - 30:
        ls_deadline2 = min(deadline - 10, time.time() + 60)
        h_opt, c_opt = targeted_local_search(best_h, n_final, ls_deadline2, rng, K_lags=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 15:
        h_opt = lbfgsb_optimize(best_h, n_final, 100000000, deadline - 5,
                                maxiter=2000, penalty_coeff=1e12)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 9: Final micro local search
    if time.time() < deadline - 3:
        h_opt, c_opt = exhaustive_pair_search(best_h, n_final, deadline - 1, K_lags=3, max_passes=50)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation and projection
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    best_c = compute_c5(best_h, n_out)
    
    # Verify constraints
    assert abs(best_h.sum() - n_out / 2.0) < 1e-8, f"Sum constraint violated: {best_h.sum()} vs {n_out/2}"
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_out)
    best_c = compute_c5(best_h, n_out)
    
    return best_h, best_c, n_out
