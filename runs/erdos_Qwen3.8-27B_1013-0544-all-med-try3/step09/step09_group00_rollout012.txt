        h_opt, c_opt = phase_active(h_pert, n, sub_end,
                                     top_ks=[3, 5], alphas=[500000, 5000000])
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS warmup at original resolution
    p1_end = min(t0 + 100, deadline - 700)
    h1, c1 = phase_full_lbfgs(best_h, best_n, p1_end,
                              alphas=[50, 200, 1000, 5000, 20000, 100000, 500000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    # Save best at original n
    save_h_orig = best_h.copy()
    save_c_orig = best_c

    # Phase 2: Active set optimization at original n
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 130, deadline - 450)
        h2, c2 = phase_active(best_h, best_n, p2_end,
                              top_ks=[3, 5, 8, 15],
                              alphas=[100000, 500000, 2000000, 10000000, 50000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    save_h_orig = best_h.copy()
    save_c_orig = best_c

    # Phase 3: Coordinate descent at original n
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 120, deadline - 300)
        h3, c3 = phase_coord_descent(best_h, best_n, p3_end, maxiter=200000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    save_h_orig = best_h.copy()
    save_c_orig = best_c

    # Phase 4: Upsample to 2x and optimize
    if time.time() < deadline - 250:
        n_up = best_n * 2
        h_up = upsample(save_h_orig, best_n, n_up)
        p4a_end = min(time.time() + 80, deadline - 200)
        h4a, c4a = phase_full_lbfgs(h_up, n_up, p4a_end,
                                    alphas=[100, 1000, 10000, 100000, 500000])
        if time.time() < deadline - 120:
            p4b_end = min(time.time() + 80, deadline - 100)
            h4b, c4b = phase_active(h4a, n_up, p4b_end,
                                    top_ks=[3, 5, 10],
                                    alphas=[200000, 2000000, 10000000, 50000000])
        else:
            h4b, c4b = h4a, c4a
        if c4b < best_c:
            best_c = c4b
            best_h = h4b
            best_n = n_up

    # Phase 5: Fine local search on current best
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 60, deadline - 50)
        h5, c5 = phase_fine_local(best_h, best_n, p5_end, radius=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: More active set refinement
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 30, deadline - 25)
        h6, c6 = phase_active(best_h, best_n, p6_end,
                              top_ks=[3, 5],
                              alphas=[5000000, 20000000, 100000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final coord descent
    if time.time() < deadline - 10:
        p7_end = deadline - 5
        h7, c7 = phase_coord_descent(best_h, best_n, p7_end, maxiter=100000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    if best_h.any(np.isnan):
        best_h = np.full(final_n, 0.5)
        best_h = project_feasible(best_h, final_n)
        best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
