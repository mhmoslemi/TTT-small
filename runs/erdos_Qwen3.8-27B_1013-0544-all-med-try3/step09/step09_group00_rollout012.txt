import time

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_grad(c, x, n, alpha):

    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()

    grad = np.zeros(n)
    for idx in np.where(w > 1e-10)[0]:
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def subgradient_descent(h, n, deadline, n_iter=5000, alpha_init=0.5, alpha_min=0.01):
    """Subgradient method for the max of correlations."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    dx_val = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h_cur, n)
        max_idx = np.argmax(c)
        
        # Subgradient of the max: gradient of the active component
        k = max_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad = np.zeros(n)
            grad[:nv] += (1.0 - h_cur[k:]) * dx_val
            grad[k:] += (-h_cur[:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad = np.zeros(n)
            grad[kk:] += (1.0 - h_cur[:nv]) * dx_val
            grad[:nv] += (-h_cur[kk:]) * dx_val
        
        # Step size (Polyak-like)
        alpha_t = alpha_init / (1.0 + 0.01 * it)
        
        # Random direction in null space of sum constraint
        random_dir = np.random.randn(n) - np.random.randn(n).mean()
        step = -alpha_t * grad + 0.1 * alpha_t * random_dir / np.std(random_dir)
        
        h_new = h_cur + step
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_cur = h_new
    
    return best_h, best_c

def anneal_pair_swap(h, n, deadline, T_init=1.0, T_min=0.01, n_iter=50000):
    """Simulated annealing with pair swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    c_cur = best_c
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_min / T_init) ** (it / n_iter)
        
        # Pick random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Try swap: move delta from i to j
        delta = 0.01
        if h_cur[i] + delta > 1.0:
            delta = 1.0 - h_cur[i]
        if h_cur[j] - delta < 0.0:
            delta = h_cur[j]
        if delta < 1e-10:
            continue
        
        h_cur[i] -= delta
        h_cur[j] += delta
        h_cur = project_feasible(h_cur, n)
        c_new = compute_c5(h_cur, n)
        
        delta_c = c_new - c_cur
        if delta_c < 0 or np.random.random() < np.exp(-delta_c / max(T, 1e-10)):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_cur.copy()
        else:
            h_cur[i] += delta
            h_cur[j] -= delta
            h_cur = project_feasible(h_cur, n)
    
    return best_h, best_c

def block_swap_search(h, n, deadline):
    """Try swapping blocks of consecutive points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    block_sizes = [3, 5, 8, 12, 20]
    
    for bs in block_sizes:
        if time.time() > deadline:
            break
        for start_i in range(0, n - bs, bs // 2):
            if time.time() > deadline:
                break
            for start_j in range(start_i + bs, n - bs + 1, bs // 2):
                if abs(start_j - start_i) < bs:
                    continue
                # Swap block [start_i, start_i+bs) with [start_j, start_j+bs)
                h_cur = best_h.copy()
                h_cur[start_i:start_i+bs], h_cur[start_j:start_j+bs] = \
                    h_cur[start_j:start_j+bs].copy(), h_cur[start_i:start_i+bs].copy()
                h_cur = project_feasible(h_cur, n)
                c = compute_c5(h_cur, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h_cur.copy()
    
    return best_h, best_c

def binary_round(h, n):
    """Round to binary, maintaining the sum constraint."""
    h_bin = (h > 0.5).astype(float)
    # Adjust to maintain sum = n/2
    target = n // 2
    curr_sum = h_bin.sum()
    if curr_sum > target:

            h_bin[idx] = 0.0

        order = np.argsort(h)
        for idx in order:
            if h_bin.sum() >= target:
                break
            h_bin[idx] = 1.0
    return h_bin

    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, x, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        wm = w[idx] * dx_val
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_obj_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad(c, x, n, alpha)
    if mu > 0:
        g = g + 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(lbfgs_obj_func, best_h, jac=lbfgs_grad_func,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=8, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        use_alpha = alphas[min(rnd % len(alphas), len(alphas) - 1)]
        # Local objective using only active set
        def act_obj(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = use_alpha * (vals - mx)
            smx = sh.max()
            if smx > 400:
                val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / use_alpha
            else:
                val = mx + np.log(np.sum(np.exp(sh))) / use_alpha
            return val + 0.0005 * (x.sum() - n / 2.0) ** 2

        def act_grad(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = use_alpha * (vals - mx)
            smx = sh.max()
            w = np.exp(sh - smx)
            w /= w.sum()
            dx_val = 2.0 / n
            g = np.zeros(n)
            for wi, idx in zip(w, active):
                if wi < 1e-12:
                    continue
                k = idx - (n - 1)
                wm = wi * dx_val
                if k >= 0:
                    nv = n - k
                    g[:nv] += wm * (1.0 - x[k:])
                    g[k:] += wm * (-x[:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:] += wm * (1.0 - x[:nv])
                    g[:nv] += wm * (-x[kk:])
            g = g + 2.0 * 0.0005 * (x.sum() - n / 2.0)
            return g

        maxit = max(200, int((deadline - time.time()) * 1.5))
        try:
            res = minimize(act_obj, best_h, jac=act_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 6,
                                   'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def grad_pair_search(h, n, deadline, max_steps=200000, pool_size=80):
    """Gradient-guided pair exchange with larger pool."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    steps_done = 0
    stuck_count = 0
    
    while time.time() < deadline and steps_done < max_steps:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx_val
                grad[k:] += (-best_h[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx_val
                grad[:nv] += (-best_h[kk:]) * dx_val
        grad /= max(1, len(active))
        
        pos_mask = (grad > 1e-12) & (best_h > 1e-8)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            stuck_count += 1
            if stuck_count > 10:
                break
            continue
        
        # Shuffle and take pool
        rng_i = np.random.RandomState(steps_done)
        if len(pos_idx) > pool_size:
            pos_idx = pos_idx[rng_i.choice(len(pos_idx), pool_size, replace=False)]
        if len(neg_idx) > pool_size:
            neg_idx = neg_idx[rng_i.choice(len(neg_idx), pool_size, replace=False)]
        
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos_idx:
            gi = grad[i]
            for j in neg_idx:
                gain = gi - grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            stuck_count += 1
            if stuck_count > 10:
                break
            continue
        
        amt = min(0.005, best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-10:
            break
        
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        steps_done += 1
        
        if c_new < best_c - 1e-16:
            best_c = c_new
            stuck_count = 0
        else:
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
            stuck_count += 1
            if stuck_count > 30:
                break
    
    return best_h, best_c

def anneal_restart(h, n, deadline, n_restarts=12, frac_flip=0.05):
    """Simulated annealing with grasp restarts on the binary structure."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for rst in range(n_restarts):
        remaining = deadline - time.time()
        if remaining < 60:
            break
        phase_end = time.time() + min(remaining - 10, 40)
        
        # Create perturbed binary start
        h_bin = (best_h > 0.5).astype(float)
        n_flip = max(3, int(frac_flip * n * (1.0 + 0.5 * rng.random())))
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) < n_flip or len(zeros) < n_flip:
            continue
        flip_from = rng.choice(ones, min(n_flip, len(ones)), replace=False)
        flip_to = rng.choice(zeros, min(n_flip, len(zeros)), replace=False)
        nf = min(len(flip_from), len(flip_to))
        h_bin[flip_from[:nf]] = 0.0
        h_bin[flip_to[:nf]] = 1.0
        h_cur = project_feasible(h_bin, n)
        
        # Quick L-BFGS polish
        maxit = max(100, int((phase_end - time.time()) * 3))
        if maxit > 50:
            try:
                bounds = [(0.0, 1.0)] * n
                res = minimize(lbfgs_obj_func, h_cur, jac=lbfgs_grad_func,
                              args=(n, 500, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 40})
                h_cur = project_feasible(res.x, n)
            except Exception:
                pass
        
        c_cur = compute_c5(h_cur, n)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
    
    return best_h, best_c

def fine_swap_search(h, n, deadline, max_pairs_per_round=5000, deltas=None):
    """Local pair swap with larger neighborhood."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    rng = np.random.RandomState(777)
    
    for d in deltas:
        improved = True
        rounds = 0
        while improved and time.time() < deadline:
            improved = False
            rounds += 1
            for _ in range(max_pairs_per_round):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                # Move mass from i to j
                if best_h[i] >= d and best_h[j] <= 1.0 - d:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                # Move mass from j to i
                if best_h[j] >= d and best_h[i] <= 1.0 - d:
                    best_h[j] -= d
                    best_h[i] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        best_h[j] += d
                        best_h[i] -= d
                if not improved:
                    pass
    return best_h, best_c

def two_point_coords(h, n, deadline):
    """Optimize pairs of coordinates simultaneously via small 2D search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    dx_val = 2.0 / n
    
    # Find active set
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    active = np.where(c_vec >= max_val - 0.01 * max_val)[0]
    
    # For each active shift, find the pair of points that most contributes to overlap
    for _ in range(200):
        if time.time() > deadline:
            break
        idx = active[rng.randint(len(active))]
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            # Overlap at positions 0..nv-1 where h[i]=1, h[i+k]=0
            # and positions k..n-1 where h[i-k]=1, h[i]=0
            contrib_pos = np.zeros(n)
            contrib_pos[:nv] = best_h[:nv] * (1.0 - best_h[k:])
            contrib_pos[k:] += best_h[k:] * (1.0 - best_h[:nv])
        else:
            kk = -k
            nv = n - kk
            contrib_pos = np.zeros(n)
            contrib_pos[kk:] = best_h[kk:] * (1.0 - best_h[:nv])
            contrib_pos[:nv] += best_h[:nv] * (1.0 - best_h[kk:])
        
        # Find top contributors
        order = np.argsort(-contrib_pos)
        top = order[:5]
        for i in top:
            for j in top:
                if i == j:
                    continue
                if best_h[i] > 0.01 and best_h[j] < 0.99:
                    # Try moving 0.01 from i to j
                    amt = 0.01
                    best_h[i] -= amt
                    best_h[j] += amt
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                    else:
                        best_h[i] += amt
                        best_h[j] -= amt
                        best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Periodic extension
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_new = np.interp(x_new, x_ext, h_ext)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = make_beatty_start(500, (1.0 + np.sqrt(5.0)) / 4.0)
        n = len(h0)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Try alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alternatives = [
        make_beatty_start(n, golden / 2.0),
        make_beatty_start(n, golden / 2.0 + 0.01),
        make_beatty_start(n, np.sqrt(2.0) / 2.0 - 0.02),
    ]
    for _ in range(2):
        h_rand = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        alternatives.append(project_feasible(h_rand, n))
    
    for h_alt in alternatives:
        c_alt = compute_c5(h_alt, n)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup (0-150s)
    p1_end = min(t0 + 150, deadline - 300)
    h1, c1 = run_lbfgs_phase(best_h, n, p1_end,
        alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01, maxit_list=[800, 1000, 1200, 1500, 1500, 1500, 1500, 1200, 1000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization (150-300s)
    if time.time() < deadline - 250:
        p2_end = min(time.time() + 150, deadline - 250)
        h2, c2 = run_active_set(best_h, n, p2_end, top_k=8,
                               alphas=[50000, 200000, 1000000, 5000000],
                               rounds=15)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Annealing restarts to escape local minima (300-500s)
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 200, deadline - 200)
        h3, c3 = anneal_restart(best_h, n, p3_end, n_restarts=15, frac_flip=0.04)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Second L-BFGS + active set round (500-700s)
    if time.time() < deadline - 180:
        p4_end = min(time.time() + 100, deadline - 180)
        h4, c4 = run_lbfgs_phase(best_h, n, p4_end,
            alphas=[1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxit_list=[1500, 1500, 1500, 1500, 1000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
        if time.time() < deadline - 80:
            p4b_end = min(time.time() + 80, deadline - 80)
            h4b, c4b = run_active_set(best_h, n, p4b_end, top_k=6,
                alphas=[100000, 500000, 2000000], rounds=8)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b

    # Phase 5: Gradient pair search (700-850s)
    if time.time() < deadline - 130:
        p5_end = min(time.time() + 150, deadline - 130)
        h5, c5 = grad_pair_search(best_h, n, p5_end, max_steps=150000, pool_size=100)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Two-point coordinate optimization (850-950s)
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 100, deadline - 80)
        h6, c6 = two_point_coords(best_h, n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine swap search (950-1040s)
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 90, deadline - 40)
        h7, c7 = fine_swap_search(best_h, n, p7_end, max_pairs_per_round=3000,
                                  deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final L-BFGS polish with very high alpha (last 40s)
    if time.time() < deadline - 10:
        p8_end = deadline - 5
        h8, c8 = run_lbfgs_phase(best_h, n, p8_end,
            alphas=[1000000, 5000000, 20000000],
            mu=0.001, maxit_list=[800, 600, 400])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final gradient pair polish
    if time.time() < deadline - 3:
        h9, c9 = grad_pair_search(best_h, n, deadline - 2, max_steps=20000, pool_size=30)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
