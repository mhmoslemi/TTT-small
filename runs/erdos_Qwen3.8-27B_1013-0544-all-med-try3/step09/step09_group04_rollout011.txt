def subgrad_step(h, n):
    """One subgradient step with minimum norm subgradient."""
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    tol = max(1e-10, 0.001 * c_max)
    active = np.where(c_vec >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    dx = 2.0 / n
    grads = []
    for idx in active:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] += (1.0 - h[:nv]) * dx
            g[:nv] += (-h[kk:]) * dx
        grads.append(g)
    # Minimum norm convex combination
    grads = np.array(grads)  # shape (m, n)
    m = grads.shape[0]
    if m == 1:
        avg_g = grads[0]
    else:
        # Solve for weights that minimize ||sum w_i g_i||^2
        # s.t. sum w_i = 1, w_i >= 0
        # Unconstrained: w = G^T (GG^T)^{-1} 1
        G = grads  # (m, n)
        A = G @ G.T  # (m, m)
        b = np.ones(m)
        try:
            w = np.linalg.solve(A + 1e-10 * np.eye(m), b)
            w = np.maximum(w, 0)
            w /= w.sum()
        except:
            w = np.ones(m) / m
        avg_g = w @ grads
    return avg_g, active

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_alpha(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    m2 = sh.max()
    if m2 > 500:
        val = m + (m2 + np.log(np.sum(np.exp(sh - m2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_alpha(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    m2 = sh.max()
    w = np.exp(sh - m2)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_obj_alpha(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    m2 = sh.max()
    if m2 > 500:
        val = m + (m2 + np.log(np.sum(np.exp(sh - m2)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_grad_alpha(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    m2 = sh.max()
    w = np.exp(sh - m2)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for w_i, idx in zip(w, active):
        if w_i < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w_i * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def run_lbfgs_stage(h, n, deadline, alphas, mu=0.005, maxiters_list=None):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 8:
            break
        if maxiters_list is not None and i < len(maxiters_list):
            maxit = max(50, maxiters_list[i])
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(smooth_obj_alpha, best_h, jac=smooth_grad_alpha,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-25,
                                    'gtol': 1e-20, 'maxfun': maxit * 12,
                                    'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
        if time.time() >= deadline:
            break
    return best_h, best_c

def run_active_stage(h, n, deadline, top_k=4, alphas=None):

        alphas = [10000, 100000, 1000000, 10000000, 100000000]

            break

        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 6:
                break
            maxit = max(50, int(rem2))
            try:
                res = minimize(active_obj_alpha, best_h, jac=active_grad_alpha,
                               args=(n, active, alpha, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-25,
                                        'gtol': 1e-18, 'maxfun': maxit * 12,
                                        'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, max_iter=5000):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    eta = 0.003
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grads = []
        for idx in active:
            k = idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[:nv] += (1.0 - best_h[k:]) * dx
                g[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:] += (1.0 - best_h[:nv]) * dx
                g[:nv] += (-best_h[kk:]) * dx
            grads.append(g)
        grads_arr = np.array(grads)
        m_active = grads_arr.shape[0]
        if m_active == 1:
            avg_g = grads_arr[0]
        else:
            A = grads_arr @ grads_arr.T
            b = np.ones(m_active)
            try:
                w = np.linalg.solve(A + 1e-10 * np.eye(m_active), b)
                w = np.maximum(w, 0.0)
                s = w.sum()
                if s < 1e-15:
                    w = np.ones(m_active) / m_active
                else:
                    w /= s
            except Exception:
                w = np.ones(m_active) / m_active
            avg_g = w @ grads_arr
        h_new = best_h - eta * avg_g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.03, 0.015)
        else:
            eta *= 0.93
            if eta < 1e-9:
                break
    return best_h, best_c

def pair_local_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 1):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                h = h.copy()
    return best_h, best_c
