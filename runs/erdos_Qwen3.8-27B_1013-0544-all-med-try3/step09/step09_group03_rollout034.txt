**Key improvement idea: "Shift relaxation"**

At the optimum, suppose shifts k*1, k*2, ..., k*M are active (all achieve C₅). Then we can:

3. At the optimum, all s_i ≤ 0 for all active k*
4. Try to reduce the maximum by adjusting h in directions that decrease the worst s_i

Start with low alpha (smooth approximation) and gradually increase to very high alpha (sharp max). At each level:
- Run L-BFGS to convergence
- This gives a sequence of solutions that converges to the true optimum

        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return float(obj), grad

def run_lbfgs(h, n, deadline, alphas=None, max_iter=3000, use_exact_grad=False):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        func = obj_grad if use_exact_grad else obj_grad_simple
        try:
            res = minimize(func, h, args=(n, alpha), jac=True,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iter * 40, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def structured_init(n, seed):
    """Create a structured initialization based on a low-discrepancy-like sequence."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Use a Thue-Morse-like pattern mixed with random perturbation
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        tm[i] = 1.0 if c % 2 == 1 else 0.0
    # Add random perturbation
    h = 0.7 * tm + 0.3 * rng.rand(n)
    return project_feasible(h, n)

def targeted_refine(h, n, deadline, max_rounds=200):
    """Local refinement: adjust individual points to reduce worst overlap."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    rng = np.random.RandomState(999)
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        if k_worst >= 0:
            affected = list(range(0, n - k_worst)) + list(range(k_worst, n))
        else:
            kk = -k_worst
            affected = list(range(kk, n)) + list(range(0, n - kk))
        # Deduplicate
        affected = list(set(affected))
        rng.shuffle(affected)
        improved = False
        deltas = [0.003, -0.003, 0.001, -0.001, 0.005, -0.005, 0.002, -0.002]
        for i in affected[:300]:
            if time.time() > deadline:
                return best_h, best_c
            for delta in deltas:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = best_h[i]
            if improved:
                h = best_h.copy()
        if not improved and round_num > 30:
            break
    return best_h, best_c

def shift_cycling(h, n, deadline, n_cycles=50):
    """Alternate between minimizing different worst shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    center = n - 1
    
    for cycle in range(n_cycles):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        # Get top 3 worst shifts
        top3 = np.argsort(c_vec)[-3:]
        # Pick one randomly from top 3
        k_idx = top3[np.random.randint(3)]
        k = k_idx - center
        
        # Define objective: just this shift's value
        # c_k = sum_i h[i]*(1-h[i+k])*dx
        # We want to minimize this subject to sum(h)=n/2, 0<=h<=1
        
        def single_shift_obj(x):
            if k >= 0:
                nv = n - k
                return float(np.sum(x[0:nv] * (1.0 - x[k:n])) * dx)
            else:
                kk = -k
                nv = n - kk
                return float(np.sum(x[kk:n] * (1.0 - x[0:nv])) * dx)
        
        def single_shift_grad(x):
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[0:nv] += dx * (1.0 - x[k:n])
                g[k:n] += dx * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += dx * (1.0 - x[0:nv])
                g[0:nv] += dx * (-x[kk:n])
            return g
        
        def constraint_val(x):
            return float(x.sum() - n / 2.0)
        
        def constraint_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': constraint_val, 'jac': constraint_jac}]
        
        try:
            res = minimize(single_shift_obj, h, jac=single_shift_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            if res is not None and hasattr(res, 'x'):
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = best_h.copy()
        except Exception:
            pass
    
    return best_h, best_c
