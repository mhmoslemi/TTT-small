            if time.time() > phase_deadline:
                break
            g = grad_smooth_max(h_curr, n, alpha)
            v = momentum * v - lr * g
            h_new = project_feasible(h_curr + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                if patience > 100:
                    h_curr = h_new
                    if patience > 500:
                        lr *= 0.7
                        v *= 0.5
                        patience = 0
                        if lr < 0.005:
                            break
                    continue
            h_curr = h_new
        # Reset momentum between alpha phases
        h_curr = best_h.copy()
    
    return best_h, best_c

def slsqp_polish(h, n, alpha, top_m, maxiter, deadline):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_lags_obj, h, args=(n, top_idx, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      jac=lambda x: active_lags_grad(x, n, top_idx, alpha),
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Find all near-max lags
        threshold = best_c * (1 - 1e-4)
        active_ks = np.where(c_all >= threshold)[0]
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = 0
        
        # Try small transfers at the critical lag
        for d in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                j = i + m
                if h[i] > h[j] + 1e-6:
                    delta = min(d, (h[i] - h[j]) * 0.3)
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved += 1
                        c_all = compute_all_lags(h, n)
                        if improved > 100:
                            break
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-6:
                    delta = min(d, (h[j] - h[i]) * 0.3)
                    if delta < 1e-8:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved += 1
                        c_all = compute_all_lags(h, n)
                        if improved > 100:
                            break
                    else:
                        h[i] -= delta
                        h[j] += delta
            if improved > 50:
                break
        
        # Also try second-worst lag
        if improved < 5 and len(active_ks) > 1:
            sorted_ks = np.argsort(c_all)[::-1]
            for k2 in sorted_ks[1:4]:
                lag2 = k2 - (n - 1)
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                for d2 in [0.003, 0.001]:
                    if time.time() > deadline:
                        break
                    for i in range(0, n - m2):
                        if time.time() > deadline:
                            break
                        j2 = i + m2
                        if h[i] > h[j2] + 1e-6:
                            delta2 = min(d2, (h[i] - h[j2]) * 0.2)
                            if delta2 < 1e-8:
                                continue
                            h[i] -= delta2
                            h[j2] += delta2
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved += 1
                                c_all = compute_all_lags(h, n)
                            else:
                                h[i] += delta2
                                h[j2] -= delta2
                        elif h[j2] > h[i] + 1e-6:
                            delta2 = min(d2, (h[j2] - h[i]) * 0.2)
                            if delta2 < 1e-8:
                                continue
                            h[i] += delta2
                            h[j2] -= delta2
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved += 1
                                c_all = compute_all_lags(h, n)
                            else:
                                h[i] -= delta2
                                h[j2] += delta2
        
        if improved == 0:
            # Try random pairs across multiple lags
            rng_l = np.random.RandomState(rnd * 7 + 3)
            found = False
            for _ in range(200):
                if time.time() > deadline:
                    break
                k_rand = rng_l.randint(0, 2 * n - 1)
                lag_r = k_rand - (n - 1)
                if lag_r == 0:
                    continue
                m_r = abs(lag_r)
                i_r = rng_l.randint(0, n - m_r)
                j_r = i_r + m_r
                if h[i_r] > h[j_r] + 1e-6:
                    delta_r = 0.001 * rng_l.random()
                    delta_r = min(delta_r, (h[i_r] - h[j_r]) * 0.2)
                    if delta_r < 1e-8:
                        continue
                    h[i_r] -= delta_r
                    h[j_r] += delta_r
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        found = True
                        c_all = compute_all_lags(h, n)
                    else:
                        h[i_r] += delta_r
                        h[j_r] -= delta_r
                elif h[j_r] > h[i_r] + 1e-6:
                    delta_r = 0.001 * rng_l.random()
                    delta_r = min(delta_r, (h[j_r] - h[i_r]) * 0.2)
                    if delta_r < 1e-8:
                        continue
                    h[i_r] += delta_r
                    h[j_r] -= delta_r
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        found = True
                        c_all = compute_all_lags(h, n)
                    else:
                        h[i_r] -= delta_r
                        h[j_r] += delta_r
            if not found:
                break
    
    return h, best_c

def fine_polish(h, n, deadline):
    """Very fine polishing using multiple SLSQP passes and pair search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Multiple SLSQP passes with decreasing top_m
    for top_m in [3, 5, 7, 3, 5]:
        if time.time() > deadline - 5:
            break
        alpha = 50000
        h_new = slsqp_polish(h, n, alpha, top_m, maxiter=2000, deadline=deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    
    # Pair local search
    if time.time() < deadline - 3:
        h_new, c_new = pair_local_search(h, n, deadline - 1, max_rounds=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # One more SLSQP after local search
    if time.time() < deadline - 3:
        h_new = slsqp_polish(best_h, n, 100000, 3, maxiter=1000, deadline=deadline - 1)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def try_resolution(h_best, c_best, n_target, deadline, rng):
    """Interpolate to new resolution and optimize briefly."""
    x_old = np.linspace(0, 2, len(h_best))
    x_new = np.linspace(0, 2, n_target)
    h_start = np.interp(x_new, x_old, h_best)
    h_start = project_feasible(h_start, n_target)
    
    # Quick gradient optimization
    h_opt, c_opt = projected_gradient(h_start, n_target, deadline,
                                       alphas=[100, 500, 2000, 5000])
    return h_opt, c_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        alloc1 = min(250, deadline - t0 - 300)
        h_opt, c_opt = projected_gradient(h_init, n_init, t0 + alloc1,
                                           alphas=[50, 100, 200, 500, 1000, 3000, 8000, 20000])
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        alloc1 = min(200, deadline - t0 - 300)
        h_opt, c_opt = projected_gradient(h_start, n_init, t0 + alloc1,
                                           alphas=[50, 100, 200, 500, 1000, 3000])
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: SLSQP polish at initial resolution
    if time.time() < deadline - 400:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 390:
                break
            h_new = slsqp_polish(best_h, n_init, 20000, top_m, maxiter=3000,
                                 deadline=deadline - 385)
            c_new = compute_c5(h_new, n_init)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Try higher resolutions
    for n_try in [550, 600, 650, 700, 800]:
        if time.time() > deadline - 350:
            break
        remaining_to_350 = deadline - 350 - time.time()
        if remaining_to_350 < 60:
            break
        alloc = min(80, remaining_to_350 * 0.3)
        if alloc < 40:
            break
        h_opt_r, c_opt_r = try_resolution(best_h, best_c, n_try, time.time() + alloc, rng)
        if c_opt_r < best_c:
            best_c = c_opt_r
            best_h = h_opt_r
            n_init = n_try
    
    # Phase 4: Fine polish at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 200:
        fine_polish_deadline = min(deadline - 100, time.time() + 180)
        best_h, best_c = fine_polish(best_h, n_best, fine_polish_deadline)
    
    # Phase 5: Try multistart at best resolution
    if time.time() < deadline - 150:
        n_restarts = 3
        for r in range(n_restarts):
            if time.time() > deadline - 130:
                break
            scale = 0.01 * (1 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(45, (deadline - 120 - time.time()) * 0.4)
            if alloc_r < 20:
                break
            h_opt_r, c_opt_r = projected_gradient(hp, n_best, time.time() + alloc_r,
                                                   alphas=[500, 2000, 8000])
            if c_opt_r < best_c:
                best_c = c_opt_r
                best_h = h_opt_r
    
    # Phase 6: Final SLSQP + local search
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        # SLSQP passes
        for top_m in [3, 5, 4]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_polish(best_h, n, 50000, top_m, maxiter=2000,
                                 deadline=deadline - 20)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final pair local search
        if time.time() < deadline - 5:
            best_h, best_c = pair_local_search(best_h, n, deadline - 2, max_rounds=500)
        
        # One final SLSQP
        if time.time() < deadline - 3:
            h_new = slsqp_polish(best_h, n, 100000, 3, maxiter=1500, deadline=deadline - 1)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
