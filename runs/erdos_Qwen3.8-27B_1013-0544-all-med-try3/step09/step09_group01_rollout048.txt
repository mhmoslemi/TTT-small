import numpy as np
from scipy.optimize import minimize
import time

# Core computation
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            frac = min(1.0, diff / tot) if tot > 0 else 0
            h += frac * room
        else:
            mask = h > 1e-15
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            frac = min(1.0, -diff / tot) if tot > 0 else 0
            h -= frac * h
        h = np.clip(h, 0.0, 1.0)
    return h

# Smooth max objective and gradient
def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 100:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    off = n - 1
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - off
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()

                                   'maxls': 50, 'maxcor': 50})

def pair_search(h, n, deadline, radius=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n - 1):
                if time.time() > deadline - 2:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n - 1, i + radius)
                for j in range(lo, hi + 1):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try h[i] -= d, h[j] += d

                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

**Key idea: Use a penalty-free approach with exact constraints**

The previous code uses L-BFGS-B with bounds [0,1] and a penalty term for the sum constraint. This means the sum constraint is only approximately satisfied during optimization, and the projection at the end might move the solution away from the optimal direction.

Instead, I could:
1. Use SLSQP with the exact equality constraint and bounds
2. Or use a barrier method with the exact constraint
