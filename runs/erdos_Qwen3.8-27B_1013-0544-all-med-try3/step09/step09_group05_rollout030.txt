import numpy as np
from scipy.optimize import minimize
import time

ALPHA_STEPS = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000, 3000000]
TOL_STEPS = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
PAIR_FRACS = (0.0, 0.075, 0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0)


def eval_exact(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n or not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h2 = np.zeros(n, dtype=float)
        m = min(n, h.size)
        if m > 0:
            h2[:m] = h[:m]
        h = h2

    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    s = float(h.sum())

    if abs(s - target) > 1e-14:
        if s < target:
            denom = float(n) - s
            if denom > 1e-15:
                r = (target - s) / denom
                h *= (1.0 - r)
                h += r
            else:
                h[:] = target / float(n)
        else:
            if s > 1e-15:
                r = (target - s) / s
                h *= (1.0 + r)
            else:
                h[:] = target / float(n)

    np.clip(h, 0.0, 1.0, out=h)
    res = target - float(h.sum())

    if abs(res) > 1e-15:
        cand = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if cand.size > 0:
            idx = cand[0]
            adj = res
            if adj > 1.0 - h[idx]:
                adj = 1.0 - h[idx]
            elif adj < -h[idx]:
                adj = -h[idx]
            h[idx] += adj
            res = target - float(h.sum())

    if abs(res) > 1e-12:
        if res > 0.0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                take = min(res, float(room[idx]))
                if take <= 0.0:
                    break
                h[idx] += take
                res -= take
                if abs(res) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                take = min(-res, float(h[idx]))
                if take <= 0.0:
                    break
                h[idx] -= take
                res += take
                if abs(res) < 1e-15:
                    break

    return h


def soft_max_value(values, alpha):
    if len(values) == 0:
        return 0.0
    m = float(np.max(values))
    z = alpha * (values - m)
    smx = float(np.max(z))
    e = np.exp(z - smx)
    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return float(m + (smx + np.log(s)) / alpha)


def soft_weights(values, alpha):
    if len(values) == 0:
        return np.array([], dtype=float)
    m = float(np.max(values))
    z = alpha * (values - m)
    smx = float(np.max(z))
    w = np.exp(z - smx)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.zeros(len(values), dtype=float)
        w[int(np.argmax(values))] = 1.0
        return w
    return w / s


def full_select(h, n, tol_rel, max_count):
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    thr = m - max(tol_rel * m, 1e-10)
    idx = np.where(corr >= thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(corr))])
    if idx.size > max_count:
        order = np.argsort(-corr[idx])[:max_count]
        idx = idx[order]
    ks = (n - 1 - idx).astype(int)
    return m, ks


def active_c(h, ks, n):
    dx = 2.0 / float(n)
    oh = 1.0 - h
    out = np.empty(len(ks), dtype=float)
    for r, k in enumerate(ks):
        k = int(k)
        if k == 0:
            out[r] = dx * float(np.dot(h, oh))
        elif k > 0:
            m = n - k
            out[r] = dx * float(np.dot(h[:m], oh[k:]))
        else:
            q = -k
            m = n - q
            out[r] = dx * float(np.dot(h[q:], oh[:m]))
    return out


def active_grad(h, ks, n, weights=None):
    dx = 2.0 / float(n)
    oh = 1.0 - h
    g = np.zeros(n, dtype=float)
    if weights is None:
        weights = np.ones(len(ks), dtype=float)

    for r, k in enumerate(ks):
        w = float(weights[r])
        if abs(w) < 1e-300:
            continue
        k = int(k)
        if k == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            wk = w * dx
            g[:m] += wk * oh[k:]
            g[k:] -= wk * h[:m]
        else:
            q = -k
            m = n - q
            wk = w * dx
            g[q:] += wk * oh[:m]
            g[:m] -= wk * h[q:]
    return g


def active_smooth_value(h, n, ks, alpha):
    vals = active_c(h, ks, n)
    return soft_max_value(vals, alpha)


def transfer_smooth_value(h, n, ks, i, j, t, alpha):
    y = h.copy()
    y[i] -= t
    y[j] += t
    vals = active_c(y, ks, n)
    return soft_max_value(vals, alpha)


def smooth_obj(x, n, ks, alpha, mu):
    val = active_smooth_value(x, n, ks, alpha)
    if mu != 0.0:
        s = float(x.sum()) - 0.5 * float(n)
        val += mu * s * s
    return float(val)


def smooth_grad(x, n, ks, alpha, mu):
    vals = active_c(x, ks, n)
    w = soft_weights(vals, alpha)
    g = active_grad(x, ks, n, w)
    if mu != 0.0:
        g += 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return g


def sum_eq(x, n):
    return float(x.sum()) - 0.5 * float(n)


def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def pair_line_search(h, n, deadline):
    h = make_feasible(h.copy(), n)
    best = h.copy()
    best_c = eval_exact(h, n)
    cur = h.copy()
    cur_c = best_c
    no_round = 0

    while time.time() < deadline - 1.0 and no_round < 12:
        m, ks = full_select(cur, n, 1e-5, 48)
        vals = active_c(cur, ks, n)
        w = soft_weights(vals, 20000.0)
        g = active_grad(cur, ks, n, w)

        src = np.argsort(g)[-48:][::-1]
        dst = np.argsort(g)[:48]

        cur_sv = soft_max_value(vals, 50000.0)
        improved = False
        tested = 0

        for idx_s in src:
            if tested >= 80:
                break
            i = int(idx_s)
            for idx_d in dst:
                if tested >= 80:
                    break
                j = int(idx_d)
                if i == j:
                    continue
                if float(cur[i]) <= 1e-9 or float(cur[j]) >= 1.0 - 1e-9:
                    continue
                if g[i] - g[j] < 1e-12:
                    continue

                margin = min(float(cur[i]), 1.0 - float(cur[j]))
                if margin <= 1e-9:
                    continue

                best_t = 0.0
                best_sv = cur_sv
                for frac in PAIR_FRACS:
                    t = frac * margin
                    if t <= 1e-12:
                        sv = cur_sv
                    else:
                        sv = transfer_smooth_value(cur, n, ks, i, j, t, 50000.0)
                    if sv < best_sv - 1e-14:
                        best_sv = sv
                        best_t = t

                if best_t > 1e-12:
                    old = cur.copy()
                    cur[i] -= best_t
                    cur[j] += best_t
                    cur = make_feasible(cur, n)
                    cfull = eval_exact(cur, n)

                    if cfull < best_c - 1e-15:
                        best = cur.copy()
                        best_c = cfull
                        cur_c = cfull
                        improved = True
                    else:
                        cur = old

                tested += 1

            if improved or tested >= 80:
                break

        if improved:
            no_round = 0
            continue

        g0 = g - np.mean(g)
        norm = float(np.linalg.norm(g0))
        if norm > 1e-14:
            d = g0 / norm
            for step in (0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
                y = cur - step * d
                y = make_feasible(y, n)
                c = eval_exact(y, n)
                if c < best_c - 1e-15:
                    best = y.copy()
                    best_c = c
                    cur = y
                    cur_c = c
                    improved = True
                    break

        if improved:
            no_round = 0
        else:
            no_round += 1
            break

    return best, best_c


def slqp_polish(h, n, deadline):
    h = make_feasible(h.copy(), n)
    best = h.copy()
    best_c = eval_exact(h, n)
    bounds = [(0.0, 1.0)] * n

    if n <= 600:
        maxiter = 100
        alphas = [50000, 200000, 1000000]
    else:
        maxiter = 40
        alphas = [20000, 100000, 500000]

    for a in alphas:
        if time.time() > deadline - 1.0:
            break
        m, ks = full_select(best, n, 1e-6, 80)
        cons = [{"type": "eq", "fun": sum_eq, "jac": sum_jac, "args": (n,)}]
        try:
            res = minimize(
                smooth_obj,
                best,
                jac=smooth_grad,
                args=(n, ks, float(a), 0.0),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxiter), "ftol": 1e-12},
            )
        except Exception:
            continue

        if getattr(res, "x", None) is not None:
            h2 = make_feasible(np.asarray(res.x, dtype=float), n)
            c2 = eval_exact(h2, n)
            if c2 < best_c - 1e-15:
                best = h2
                best_c = c2

    return best, best_c


def upsample_step(h0, n0, n1):
    if n1 <= 0 or n0 <= 0:
        return make_feasible(np.zeros(n1, dtype=float), n1)

    if n1 == n0:
        return make_feasible(h0.copy(), n1)

    if n1 % n0 == 0:
        m = n1 // n0
        return make_feasible(np.repeat(h0, m), n1)

    x = (np.arange(n1) + 0.5) * (2.0 / float(n1))
    idx = np.floor(x * (float(n0) / 2.0)).astype(int)
    idx[idx >= n0] = n0 - 1
    idx[idx < 0] = 0
    return make_feasible(h0[idx], n1)


def round_binary(h, n):
    h = make_feasible(h, n)
    b = np.zeros(n, dtype=float)
    target = int(n // 2)
    if target > 0:
        idx = np.argsort(-h, kind="stable")[:target]
        b[idx] = 1.0
    return make_feasible(b, n)


def noise_start(h, n, scale, seed=0):
    rng = np.random.RandomState(int(seed))
    y = h.copy()
    mask = (y > 1e-6) & (y < 1.0 - 1e-6)
    cnt = int(np.count_nonzero(mask))
    if cnt > 0:
        y[mask] += rng.normal(0.0, scale, cnt)
    else:
        ones = np.flatnonzero(y > 0.5)
        zeros = np.flatnonzero(y <= 0.5)
        k = min(len(ones), len(zeros), max(4, n // 50))
        if k > 0:
            ii = rng.choice(ones.size, k, replace=False)
            jj = rng.choice(zeros.size, k, replace=False)
            for a, b in zip(ii, jj):
                a = int(a)
                b = int(b)
                d = 0.25 * min(float(y[a]), 1.0 - float(y[b]))
                y[a] -= d
                y[b] += d
    return make_feasible(np.clip(y, 0.0, 1.0), n)


def thue_start(n, seed=0):
    arr = np.array([int(bin(i).count("1")) & 1 for i in range(n)], dtype=float)
    target = int(n // 2)
    ones = np.flatnonzero(arr > 0.5)
    zeros = np.flatnonzero(arr <= 0.5)

    if ones.size > target:
        for idx in ones[target:]:
            arr[idx] = 0.0
    elif ones.size < target:
        for idx in zeros[: target - ones.size]:
            arr[idx] = 1.0

    return make_feasible(arr, n)
