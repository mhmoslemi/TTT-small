def active_max_optimize(h, n, deadline, max_iters=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.01
    lr_decay = 0.999
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Find active set (top shifts)
        threshold = max_val - 1e-6
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Compute combined gradient
        grad = np.zeros(n)
        n_active = len(active)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n])
                grad[k:n] += (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv])
                grad[0:nv] += (-best_h[kk:n])
        grad /= n_active
        grad *= dx
        
        # Projected gradient step
        step = project_feasible(best_h - lr * grad, n)
        c_new = compute_c5(step, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = step
            # Maybe increase lr slightly
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= lr_decay
        
        if lr < 1e-8:
            break
    
    return best_h, best_c

            if time.time() > deadline:

def upsample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()

    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n0)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS at original resolution
    p1_end = min(t0 + 180, deadline - 400)
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
                          alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                          mu=0.005)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set at original resolution
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 150, deadline - 250)
        h2, c2 = phase_active_set(best_h, best_n, p2_end, top_k=8, rounds=20)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient at original resolution
    if time.time() < deadline - 200:
        p3_end = min(time.time() + 100, deadline - 150)
        h3, c3 = subgrad_optimize(best_h, best_n, p3_end, max_iters=100000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Coordinate descent at original resolution
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 80, deadline - 100)
        h4, c4 = coord_pair_search(best_h, best_n, p4_end, max_rounds=30)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Multi-resolution optimization (upscale to 2*n0)
    if time.time() < deadline - 200:
        p5_end = deadline - 80
        results = multi_res_optimize(best_h, best_n, p5_end)
        for c_r, n_r, h_r in results:
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
                best_n = n_r

    # Phase 6: Refine best solution with active set + L-BFGS
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 60, deadline - 40)
        h6, c6 = phase_lbfgs(best_h, best_n, p6_end,
                              alphas=[100000, 500000, 2000000], mu=0.001)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        if time.time() < deadline - 30:
            h6b, c6b = phase_active_set(best_h, best_n, deadline - 20, top_k=6, rounds=10)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b

    # Phase 7: Subgradient polish
    if time.time() < deadline - 40:
        h7, c7 = subgrad_optimize(best_h, best_n, min(time.time() + 30, deadline - 20),
                                   max_iters=50000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine coordinate search
    if time.time() < deadline - 25:
        h8, c8 = coord_pair_search(best_h, best_n, min(time.time() + 20, deadline - 10),
                                    max_rounds=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Very fine sweep
    if time.time() < deadline - 10:
        h9, c9 = fine_sweep(best_h, best_n, deadline - 3, radius=30)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
