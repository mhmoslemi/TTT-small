    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_all(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = corr_w_grad(w, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = corr_w_grad(w, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(obj_all, best_h, jac=grad_all,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k_list, alphas_list, rounds_list):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for top_k, alpha, rounds in zip(top_k_list, alphas_list, rounds_list):
        for rnd in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            maxit = max(200, int((deadline - time.time()) * 1.2))
            try:
                res = minimize(obj_active, best_h, jac=grad_active,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def multiswap_search(h, n, deadline, max_stag=500):
    """Multi-pair swap local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    stag = 0
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in step_sizes:
        while time.time() < deadline:
            if stag > max_stag:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            # Compute weighted gradient
            w_active = c_vec[active] - (max_val - tol)
            w_active = np.maximum(w_active, 1e-10)
            w_active /= w_active.sum()
            grad = np.zeros(n)
            for wi, idx in zip(w_active, active):
                k = idx - (n - 1)
                wm_dx = wi * dx_val
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += wm_dx * (1.0 - best_h[k:n])
                    grad[k:n] += wm_dx * (-best_h[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += wm_dx * (1.0 - best_h[0:nv])
                    grad[0:nv] += wm_dx * (-best_h[kk:n])
            # Gradient-guided swap
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            if len(pos_idx) == 0 or len(neg_idx) == 0:
                stag += 1
                continue
            # Try best single swap
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                stag += 1
                continue
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                stag += 1
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                stag = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stag += 1
            # Try 2-pair swap
            if len(pos_idx) >= 2 and len(neg_idx) >= 2 and stag < 100:
                best_gain2 = 0
                bi2 = []
                bj2 = []
                for i1 in pos_idx[:10]:
                    for i2 in pos_idx[:10]:
                        if i1 == i2:
                            continue
                        for j1 in neg_idx[:10]:
                            for j2 in neg_idx[:10]:
                                if j1 == j2:
                                    continue
                                g = (grad[i1] + grad[i2]) - (grad[j1] + grad[j2])
                                if g > best_gain2 and i1 != j1 and i2 != j2:
                                    best_gain2 = g
                                    bi2 = [i1, i2]
                                    bj2 = [j1, j2]
                if len(bi2) == 2:
                    amt2 = min(step * 0.5, min(best_h[bi2]))
                    amt2 = min(amt2, 1.0 - max(best_h[bj2]))
                    if amt2 > 1e-12:
                        for i in bi2:
                            best_h[i] -= amt2
                        for j in bj2:
                            best_h[j] += amt2
                        best_h = project_feasible(best_h, n)
                        c2 = compute_c5(best_h, n)
                        if c2 < best_c - 1e-16:
                            best_c = c2
                            stag = 0
                        else:
                            for i in bi2:
                                best_h[i] += amt2
                            for j in bj2:
                                best_h[j] -= amt2
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def random_perturb(h, n, rng, strength):
    """Add random perturbation and re-project."""
    h_new = h.copy()
    mask = (h_new > 1e-4) & (h_new < 1.0 - 1e-4)
    if mask.sum() > 0:
        h_new[mask] += rng.randn(mask.sum()) * strength
    h_new = project_feasible(h_new, n)
    return h_new

def try_binary_round(h, n):
    """Round to binary and optimize the binary structure."""
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        # Remove excess ones (from smallest non-.5 values)
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by original h value (keep the largest)
        order = np.argsort(-h[ones_idx])
        for idx in order[n_target:]:
            h_bin[idx] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:n_target - n_ones]:
            h_bin[idx] = 1
    return project_feasible(h_bin, n)

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new by linear interpolation."""
    x_old = np.linspace(0, 2, n_old + 1)
    y_old = np.concatenate([h, [h[0]]])
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old[:n_new * 2 // n_old if n_new > n_old else n_old], 
                      y_old[:n_new * 2 // n_old + 1 if n_new > n_old else n_old + 1])
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    # Primary resolution: n=1000 (upsample from parent)
    n_high = 1000
    h_high = np.repeat(h_parent, n_high // n_parent)
    if len(h_high) > n_high:
        h_high = h_high[:n_high]
    h_high = project_feasible(h_high, n_high)

    # Also upsample via linear interpolation for smoother start
    x_old = np.linspace(0, 2, n_parent, endpoint=False)
    x_new = np.linspace(0, 2, n_high, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_parent)
    h_interp = project_feasible(h_interp, n_high)

    # Parent at original resolution
    h_low = project_feasible(h_parent, n_parent)

    # Evaluate all starting points
    best_h = h_low.copy()
    best_c = compute_c5(h_low, n_parent)
    best_n = n_parent

    c_rep = compute_c5(h_high, n_high)
    c_interp = compute_c5(h_interp, n_high)
    
    if c_rep < best_c:
        best_c = c_rep
        best_h = h_high.copy()
        best_n = n_high
    if c_interp < best_c:
        best_c = c_interp
        best_h = h_interp.copy()
        best_n = n_high

    # Try Beatty starts at n_high
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    beatty_starts = [
        make_beatty_start(n_high, golden / 2.0),
        make_beatty_start(n_high, np.sqrt(2.0) / 2.0),
        make_beatty_start(n_high, golden / 3.0),
        make_beatty_start(n_high, (np.sqrt(5.0) - 1.0) / 4.0),
    ]
    for bs in beatty_starts:
        c_bs = compute_c5(bs, n_high)
        if c_bs < best_c:
            best_c = c_bs
            best_h = bs.copy()
            best_n = n_high

    # Phase 1: L-BFGS warmup on best_n (use ~120s)
    p1_end = min(t0 + 120, deadline - 500)
    alphas_p1 = [10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h1, c1 = run_lbfgs(best_h, best_n, p1_end, alphas_p1, mu=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization (~200s)
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 200, deadline - 350)
        top_ks = [4, 8, 16, 32, 64, 128, 256, 512, 1999]
        alphas_2 = [100000, 500000, 2000000, 5000000, 10000000] * 2
        rounds_2 = [8, 6, 6, 5, 5, 4, 4, 3, 3]
        h2, c2 = run_active_set(best_h, best_n, p2_end, top_ks, alphas_2, rounds_2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Multi-swap search (~150s)
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 150, deadline - 200)
        h3, c3 = multiswap_search(best_h, best_n, p3_end, max_stag=800)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Stochastic perturb + re-optimize (try to escape local min, ~100s)
    if time.time() < deadline - 200:
        for pert_strength in [0.05, 0.02, 0.01, 0.005]:
            if time.time() > deadline - 100:
                break
            h_pert = random_perturb(best_h, best_n, rng, pert_strength)
            c_pert = compute_c5(h_pert, best_n)
            if c_pert < best_c + 0.005:  # accept if not too much worse
                p4_end = min(time.time() + 40, deadline - 80)
                h4, c4 = run_lbfgs(h_pert, best_n, p4_end, [10000, 100000, 1000000], mu=0.01)
                if c4 < best_c:
                    best_c = c4
                    best_h = h4

    # Phase 5: Active set refinement (~100s)
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 100, deadline - 60)
        h5, c5 = run_active_set(best_h, best_n, p5_end,
                                [8, 32, 128, 512],
                                [1000000, 5000000, 20000000, 50000000],
                                [5, 5, 5, 4])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Multi-swap refinement (~60s)
    if time.time() < deadline - 50:
        p6_end = min(time.time() + 55, deadline - 30)
        h6, c6 = multiswap_search(best_h, best_n, p6_end, max_stag=300)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine pair search (~30s)
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 25, deadline - 10)
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=30,
                                   deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very fine adjustments
    if time.time() < deadline - 5:
        h8, c8 = fine_pair_search(best_h, best_n, deadline - 2, radius=20,
                                   deltas=[0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Also try: if we have a good solution at n_high, also check n_parent version
    # Final: also try running a quick optimization at parent resolution from best
    if best_n != n_parent and time.time() < deadline - 10:
        # Downsample to check
        x_h = np.linspace(0, 2, best_n, endpoint=False)
        x_l = np.linspace(0, 2, n_parent, endpoint=False)
        h_ds = np.interp(x_l, x_h, best_h)
        h_ds = project_feasible(h_ds, n_parent)
        c_ds = compute_c5(h_ds, n_parent)
        if c_ds < best_c:
            best_c = c_ds
            best_h = h_ds.copy()
            best_n = n_parent

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
