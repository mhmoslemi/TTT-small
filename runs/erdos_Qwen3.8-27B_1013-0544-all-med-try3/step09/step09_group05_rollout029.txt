from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0≤h≤1, ∑h=n/2} using water-filling."""
    target = n / 2.0
    h = h.copy()
    
    # Binary search for the shift τ
    # h_proj[i] = clip(h[i] - τ, 0, 1), choose τ so sum = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid  # need to increase τ (decrease h_proj)
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    
    # Fix any tiny numerical issues
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference uniformly among non-boundary elements
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
        else:
            # All at boundary, this shouldn't happen if target is in [0, n]
            h_proj = np.clip(h_proj + diff / n, 0, 1)
    
    return h_proj

    for alpha in alphas:
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        top_idx = np.argsort(-c_vec)[:K]
        
        def obj(x):
            cv = compute_c5_vec(x, n)
            vals = cv[top_idx]
            m = vals.max()
            sh = alpha * (vals - m)
            sh -= sh.max()
            return float(m + (sh.max() + np.log(np.exp(sh).sum())) / alpha)
        
        def grad(x):
            cv = compute_c5_vec(x, n)
            vals = cv[top_idx]
            m = vals.max()
            sh = alpha * (vals - m)
            sh -= sh.max()
            w = np.exp(sh)
            w /= w.sum()
            g = np.zeros(n)
            for wi, idx in zip(w, top_idx):
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad_local = np.zeros(n)
                    grad_local[0:nv] = (1.0 - x[k:n]) * (2.0/n)
                    grad_local[k:n] = (-x[0:nv]) * (2.0/n)
                    g += wi * grad_local
                else:
                    kk = -k
                    nv = n - kk
                    grad_local = np.zeros(n)
                    grad_local[kk:n] = (1.0 - x[0:nv]) * (2.0/n)
                    grad_local[0:nv] = (-x[kk:n]) * (2.0/n)
                    g += wi * grad_local
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def fine_local_search(h, n, deadline):
    """Very fine 2-point local search with gradient guidance."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    base_steps = [5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    
    for base_step in base_steps:
        if time.time() > deadline:
            break
        
        # Compute gradient for guidance
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        K = min(len(active), 15)
        active = active[np.argsort(-c_vec[active])[:K]]
        
        grad = active_shift_gradient(h, n, active)
        # Center to maintain sum constraint
        grad_centered = grad - grad.mean()
        
        # Find promising indices (large |gradient|)
        prominent = np.argsort(-np.abs(grad_centered))[:80]
        
        improved = True
        max_rounds = 50
        for round_num in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            
            for ii in range(len(prominent)):
                if time.time() > deadline:
                    break
                i = prominent[ii]
                gi = grad_centered[i]
                if abs(gi) < 1e-10:
                    continue
                
                for jj in range(ii + 1, len(prominent)):
                    j = prominent[jj]
                    gj = grad_centered[j]
                    if gi * gj >= 0:
                        continue  # Same sign, moving between them won't help
                    
                    # Determine direction: move from high gradient to low gradient
                    if gi > 0 and gj < 0:
                        # Decrease h[i], increase h[j]
                        for step in [base_step, base_step * 0.5, base_step * 0.1]:
                            if h[i] - step < 0 or h[j] + step > 1:
                                continue
                            h[i] -= step
                            h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] += step
                                h[j] -= step
                                break
                    elif gi < 0 and gj > 0:
                        # Increase h[i], decrease h[j]
                        for step in [base_step, base_step * 0.5, base_step * 0.1]:
                            if h[i] + step > 1 or h[j] - step < 0:
                                continue
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= step
                                h[j] += step
                                break
                        
                        if improved:
                            # Update gradient periodically
                            if round_num % 5 == 0:
                                c_vec = compute_c5_vec(h, n)
                                max_val = c_vec.max()
                                threshold = max_val - max(1e-8, 0.001 * max_val)
                                active = np.where(c_vec >= threshold)[0]
                                active = active[np.argsort(-c_vec[active])[:K]]
                                grad = active_shift_gradient(h, n, active)
                                grad_centered = grad - grad.mean()
                                prominent = np.argsort(-np.abs(grad_centered))[:80]

                break

def projected_gradient_opt(h, n, deadline):
    """Adaptive projected gradient descent."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    lr = 0.01
    patience = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        K = min(len(active), 20)
        active = active[np.argsort(-c_vec[active])[:K]]
        
        grad = active_shift_gradient(h, n, active)
        grad_centered = grad - grad.mean()
        
        # Zero out at bounds
        grad_centered[h <= 1e-8] = 0
        grad_centered[h >= 1 - 1e-8] = 0
        
        # Line search
        found = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * lr_mult
            h_new = h - step * grad_centered
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                found = True
                break
            elif c_new < c_vec.max() - 0.5 * step * np.sum(grad_centered[h_new - h]):
                h = h_new
                found = True
                break
        
        if not found:
            lr *= 0.5
            patience += 1
            if patience > 20 or lr < 1e-10:
                break
        else:
            patience = 0
            if c_new < best_c:
                lr = min(lr * 1.1, 0.1)
    
    return best_h, best_c
