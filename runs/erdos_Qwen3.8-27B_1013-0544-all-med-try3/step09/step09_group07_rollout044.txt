import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    return float(m + (mx + np.log(np.sum(w))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h) * dx
        elif 0 < lag < n:
            grad[lag:n] += w[k] * (1.0 - h[0:n-lag]) * dx
            grad[0:n-lag] += w[k] * (-h[lag:n]) * dx
        elif -n < lag < 0:
            idx2 = -lag
            grad[0:n-idx2] += w[k] * (1.0 - h[idx2:n]) * dx
            grad[idx2:n] += w[k] * (-h[0:n-idx2]) * dx

    def obj_and_grad(x):

        c = compute_all_lags(x, n)
        val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        val += 1e8 * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * 1e8 * diff
        return val, grad
    bounds = [(0.0, 1.0)] * n
    try:
