def proximal_gradient(h, n, deadline, ...):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 1.0  # smooth max parameter
    step = 0.01
    
    while time.time() < deadline:
        c = compute_c5_vec(best_h, n)
        # Compute smooth max gradient
        m = c.max()
        shifted = alpha * (c - m)
        smx = shifted.max()
        w = np.exp(shifted - smx)
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for idx, wi in enumerate(w):
            if wi < 1e-12:
                continue
            k = idx - (n-1)
            wm = wi * (2.0/n)
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - best_h[k:])
                grad[k:] += wm * (-best_h[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - best_h[:nv])
                grad[:nv] += wm * (-best_h[kk:])
        
        # Proximal step
        new_h = project_feasible(best_h - step * grad, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < best_c:
            best_c = new_c
            best_h = new_h
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.5
        
        # Increase alpha for sharper max
        alpha *= 1.05
    
    return best_h, best_c

def make_beatty_start(n, alpha_param):

        h9, c9 = coord_descent(best_h, n, deadline - 3, max_rounds=20)

            best_c = c9
            best_h = h9

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
