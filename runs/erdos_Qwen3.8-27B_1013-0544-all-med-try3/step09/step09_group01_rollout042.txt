import numpy as np
from scipy.optimize import minimize
import time


def direct_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))


def fft_c5(h):
    return float(corr_vec_fft(h).max())


def corr_vec_fft(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    v = 1.0 - h
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    a = np.fft.rfft(h, size)
    b = np.fft.rfft(v[::-1], size)
    c = np.fft.irfft(a * b, size)[:need]
    return c * (2.0 / n)


def smooth_max_value(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    with np.errstate(over='ignore', under='ignore'):
        s = np.exp(sh - mx).sum()
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + (mx + np.log(s)) / alpha)


def softmax_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    with np.errstate(over='ignore', under='ignore'):
        w = np.exp(sh - mx)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros(c.size, dtype=float)
        w[idx] = 1.0
        return w
    return w / s


def c_grad_weights(h, w):
    """Gradient of the weighted sum of all correlation lags w.r.t. h."""
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = h.size
    center = n - 1
    dx = 2.0 / n

    # Contributions from the additive/lag-window terms.
    pos_w = w[center:2 * n - 1]
    mask_pos = np.cumsum(pos_w[::-1])[::-1]
    if n > 1:
        neg_w = w[center - 1:0:-1]
        pref = np.cumsum(neg_w)
        mask_neg = np.zeros(n, dtype=float)
        mask_neg[0:n - 1] = pref[::-1]
    else:
        mask_neg = np.zeros(1, dtype=float)
    mask = mask_pos + mask_neg

    # Contributions from the bilinear terms h_i h_{i+-lag}.
    s = np.zeros(n, dtype=float)
    s[0] = 2.0 * w[center]
    if n > 1:
        s[1:] = w[center + 1:2 * n - 1] + w[center - 1:0:-1]

    K = np.zeros(2 * n - 1, dtype=float)
    K[center:center + n] = s
    if n > 1:
        K[0:center] = s[1:][::-1]

    need = 3 * n - 1
    size = 1
    while size < need:
        size <<= 1
    hf = np.fft.rfft(h, size)
    kf = np.fft.rfft(K, size)
    conv = np.fft.irfft(hf * kf, size)[:2 * n - 1]
    shift = conv[center:center + n]

    return dx * (mask - shift)


def enforce_exact(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = h[:n] if h.size > n else np.pad(h, (0, max(0, n - h.size)))
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                room = 1.0 - h[idx]
                order = np.argsort(-room)
                rem = diff
                for p in order:
                    if rem < 1e-14:
                        break
                    i = idx[p]
                    take = min(room[p] if p < room.size else room[int(p)], rem)
                    if take > 0.0:
                        h[i] += take
                        rem -= take
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                vals = h[idx]
                order = np.argsort(vals)
                rem = -diff
                for p in order:
                    if rem < 1e-14:
                        break
                    i = idx[p]
                    take = min(vals[int(p)], rem)
                    if take > 0.0:
                        h[i] -= take
                        rem -= take

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.where((h > 1e-13) & (h < 1.0 - 1e-13))[0]
        if idx.size == 0:
            if diff > 0.0:
                idx0 = np.where(h < 1.0)[0]
                if idx0.size:
                    h[idx0[0]] = 0.5
            else:
                idx0 = np.where(h > 0.0)[0]
                if idx0.size:
                    h[idx0[0]] = 0.5
            diff = target - h.sum()
        if idx.size:
            best_i = int(idx[0])
            cand = h[best_i] + diff
            if 0.0 <= cand <= 1.0:
                h[best_i] = cand

    h = np.clip(h, 0.0, 1.0)
    return h


def consider_candidate(h, n, best_h, best_c):
    if h is None:
        return best_h, best_c
    h = np.asarray(h, dtype=float).copy()
    if n is None:
        n = h.size
    h = enforce_exact(h, n)
    c = direct_c5(h)
    if c < best_c - 1e-15:
        return h, c
    return best_h, best_c


def interpolate_to(h, n_src, n_dst):
    h = np.asarray(h, dtype=float)
    if n_dst == n_src:
        return h.copy()
    x_src = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst, dtype=float) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h)


def anti_full(a):
    a = np.asarray(a, dtype=float)
    return np.concatenate((a, 1.0 - a))


def pal_full(a):
    a = np.asarray(a, dtype=float)
    return np.concatenate((a, a[::-1]))


def anti_from_full(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_to(h, h.size, n)
    m = n // 2
    if m <= 0:
        return np.array([0.5])
    if n % 2 == 0:
        a = 0.5 * (h[:m] + (1.0 - h[m:]))
    else:
        a = 0.5 * (h[:m] + (1.0 - h[m + 1:]))
    return np.clip(a, 0.0, 1.0)


def pal_from_full(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_to(h, h.size, n)
    m = n // 2
    if m <= 0:
        return np.array([0.5])
    if n % 2 == 0:
        a = 0.5 * (h[:m] + h[n - 1::-1][:m])
    else:
        a = 0.5 * (h[:m] + h[n - 1::-1][:m - 1])
        if m >= 1:
            a = np.resize(a, m)
    return np.clip(a, 0.0, 1.0)


def make_cos_anti(n):
    m = max(1, n // 2)
    if n <= 2:
        return np.full(m, 0.5)
    x = (np.arange(m, dtype=float) + 0.5) * (2.0 / n)
    a = 0.5 + 0.5 * np.cos(np.pi * x)
    return np.clip(a, 0.0, 1.0)


def make_cos_full(n):
    a = make_cos_anti(n)
    h = anti_full(a)
    return enforce_exact(h, len(h))


def make_fourier_anti_start(n, rng, nharm=5, maxamp=0.45):
    m = max(1, n // 2)
    x = (np.arange(m, dtype=float) + 0.5) * (2.0 / n)
    nharm = max(1, int(nharm))
    rs = np.arange(1, 2 * nharm, 2, dtype=float)
    if rs.size < nharm:
        rs = rs[:1]
    amps = rng.uniform(0.0, 1.0, size=rs.size)
    s = amps.sum()
    if s <= 1e-12:
        amps = np.ones_like(rs) / rs.size
    else:
        amps = amps / s
    amps = amps * maxamp
    a = np.full(m, 0.5, dtype=float)
    for r, amp in zip(rs, amps):
        a += amp * np.cos(np.pi * r * x)
    return np.clip(a, 0.0, 1.0)


def anti_grad_weights(h, w):
    g = c_grad_weights(h, w)
    m = h.size // 2
    return g[:m] - g[m:]


def anti_obj(a, alpha):
    h = anti_full(a)
    return smooth_max_value(corr_vec_fft(h), alpha)


def anti_grad(a, alpha):
    h = anti_full(a)
    c = corr_vec_fft(h)
    w = softmax_weights(c, alpha)
    return anti_grad_weights(h, w)


def pal_obj(a, alpha, mu):
    h = pal_full(a)
    val = smooth_max_value(corr_vec_fft(h), alpha)
    if mu > 0.0:
        d = h.sum() - 0.5 * h.size
        val += mu * d * d
    return float(val)


def pal_grad(a, alpha, mu):
    h = pal_full(a)
    c = corr_vec_fft(h)
    w = softmax_weights(c, alpha)
    g = c_grad_weights(h, w)
    if mu > 0.0:
        d = h.sum() - 0.5 * h.size
        g = g + 2.0 * mu * d
    m = h.size // 2
    return g[:m] + g[h.size - 1:: - 1][:m]


def full_obj(a, alpha, mu):
    h = a
    val = smooth_max_value(corr_vec_fft(h), alpha)
    if mu > 0.0:
        d = h.sum() - 0.5 * h.size
        val += mu * d * d
    return float(val)


def full_grad(a, alpha, mu):
    h = a
    c = corr_vec_fft(h)
    w = softmax_weights(c, alpha)
    g = c_grad_weights(h, w)
    if mu > 0.0:
        d = h.sum() - 0.5 * h.size
        g = g + 2.0 * mu * d
    return g


def choose_alphas(kind, slice_time):
    if kind == 'random':
        base = [20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0]
    else:
        base = [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 500000.0]
    ts = max(0.0, float(slice_time))
    if ts < 18.0:
        return base[:1]
    if ts < 45.0:
        return base[:3]
    if ts < 120.0:
        return base[:5]
    return base


def make_maxit_list(n, time_slice):
    t = max(0.0, float(time_slice))
    if n <= 512:
        base = [400, 600, 800, 1000, 1300, 1600, 2000]
    elif n <= 1024:
        base = [300, 450, 650, 850, 1100, 1400]
    elif n <= 2048:
        base = [220, 350, 520, 700, 900, 1100]
    else:
        base = [160, 260, 400, 550, 750, 950]
    if t < 25.0:
        return base[:2]
    if t < 60.0:
        return base[:3]
    if t < 150.0:
        return base[:5]
    return base


def run_lbfgs_start(x0, mode, deadline, alphas, maxiter_per=None, mu=0.0):
    deadline = float(deadline)
    if mode == 'full':
        n = int(np.size(x0))
        h = enforce_exact(x0, n)
        x = h.copy()
    elif mode == 'pal':
        n = int(2 * np.size(x0))
        h = enforce_exact(pal_full(x0), n)
        x = pal_from_full(h, n)
        h = pal_full(x)
        h = enforce_exact(h, n)
        x = pal_from_full(h, n)
    else:
        n = int(2 * np.size(x0))
        x = np.clip(x0, 0.0, 1.0)
        h = anti_full(x)
        if abs(h.sum() - 0.5 * n) > 1e-10:
            h = enforce_exact(h, n)

    best_x = np.asarray(x, dtype=float).copy()
    best_h = np.asarray(h, dtype=float).copy()
    best_c = direct_c5(best_h)
    bounds = [(0.0, 1.0)] * best_x.size

    for ai, alpha in enumerate(alphas):
        now = time.time()
        if now >= deadline:
            break
        rem = deadline - now
        if rem < 6.0:
            break

        if maxiter_per is None:
            maxit = 300
        elif ai < len(maxiter_per):
            maxit = int(maxiter_per[ai])
        else:
            maxit = int(maxiter_per[-1])

        est = 0.05 if n <= 512 else 0.09 if n <= 1024 else 0.18 if n <= 2048 else 0.38
        maxit = max(80, min(maxit, int(rem / max(1e-6, est)) + 20))

        if mode == 'anti':
            fun = anti_obj
            jac = anti_grad
            args = (float(alpha),)
        elif mode == 'pal':
            fun = pal_obj
            jac = pal_grad
            args = (float(alpha), float(mu))
        else:
            fun = full_obj
            jac = full_grad
            args = (float(alpha), float(mu))

        try:
            res = minimize(
                fun,
                best_x,
                jac=jac,
                args=args,
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': maxit,
                    'ftol': 1e-18,
                    'gtol': 1e-10,
                    'maxfun': max(250, 5 * maxit),
                    'maxls': 20,
                    'maxcor': 40,
                },
            )
            x_new = np.asarray(res.x, dtype=float)
        except Exception:
            break

        if mode == 'anti':
            x_new = np.clip(x_new, 0.0, 1.0)
            h_new = anti_full(x_new)
            if abs(h_new.sum() - 0.5 * n) > 1e-9:
                h_new = enforce_exact(h_new, n)
        elif mode == 'pal':
            x_new = np.clip(x_new, 0.0, 1.0)
            h_new = pal_full(x_new)
            h_new = enforce_exact(h_new, n)
            x_new = pal_from_full(h_new, n)
        else:
            h_new = enforce_exact(x_new, n)
            x_new = h_new.copy()

        if h_new.size != n:
            continue

        c_new = direct_c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_x = x_new.copy()
            best_h = h_new.copy()

    if mode == 'anti':
        h_final = anti_full(best_x)
        h_final = enforce_exact(h_final, h_final.size)
    elif mode == 'pal':
        h_final = pal_full(best_x)
        h_final = enforce_exact(h_final, h_final.size)
    else:
        h_final = enforce_exact(best_x, best_x.size)

    c_final = direct_c5(h_final)
    if c_final <= best_c:
        best_h = h_final
        best_c = c_final
    return best_x, best_h, best_c, h_final.size


def polish_anti(a, n, deadline, alpha=200000.0, max_iter=900):
    a = np.clip(np.asarray(a, dtype=float), 0.0, 1.0)
    h = anti_full(a)
    if h.size != n:
        h = enforce_exact(h, h.size)
        if h.size % 2 == 0:
            a = anti_from_full(h, h.size)
            n = h.size
    h = anti_full(a)
    best_a = a.copy()
    best_h = h.copy()
    best_c = fft_c5(best_h)

    eta = 0.02
    no_improve = 0
    factors = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]

    for it in range(int(max_iter)):
        if time.time() >= float(deadline) - 2:
            break
        c = corr_vec_fft(h)
        w = softmax_weights(c, float(alpha))
        g = anti_grad_weights(h, w)
        ng = float(np.linalg.norm(g))
        if ng < 1e-13:
            break

        step = (eta / ng)
        accepted = False
        for f in factors:
            t = step * f
            if t <= 0.0:
                continue
            a_new = np.clip(a - t * g, 0.0, 1.0)
            h_new = anti_full(a_new)
            c_new = fft_c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_a = a_new.copy()
                best_h = h_new.copy()
                a = a_new.copy()
                h = h_new.copy()
                eta = min(eta * 1.02, 0.05)
                no_improve = 0
                accepted = True
                break

        if not accepted:
            eta *= 0.5
            no_improve += 1
            if no_improve > 25 or eta < 1e-9:
                break

    h_final = anti_full(best_a)
    h_final = enforce_exact(h_final, len(h_final))
    c_final = direct_c5(h_final)
    return best_a, h_final, c_final, h_final.size


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(1.0, float(budget_s))
    deadline = t0 + total - 6.0
    rng = np.random.RandomState(int(seed) % (2**32))

    h_best = None
    c_best = np.inf
    n_best = 0

    try:
        hp = np.asarray(initial_h_values, dtype=float).copy()
        if hp.size > 0:
            hp = enforce_exact(hp, hp.size)
            h_best, c_best = consider_candidate(hp, hp.size, h_best, c_best)
            n_best = hp.size
    except Exception:
        h_best = None
        c_best = np.inf

    if h_best is None:
        n0 = 512
        h_best = make_cos_full(n0)
        c_best = direct_c5(h_best)
        n_best = n0

    if total >= 800.0:
        n_list = [512, 1024, 2048, 4096]
    elif total >= 400.0:
        n_list = [512, 1024, 2048]
    else:
        n_list = [512]

    for si, n in enumerate(n_list):
        now = time.time()
        if now > deadline - 25:
            break

        if si < len(n_list) - 1:
            left = deadline - now
            frac = 0.22 if si == 0 else 0.24 if si == 1 else 0.28
            stage_end = now + frac * left
            if stage_end < now + 12.0:
                stage_end = now + max(6.0, min(12.0, left * 0.2))
            if stage_end > deadline - 30:
                stage_end = min(stage_end, deadline - 30)
        else:
            stage_end = deadline - 12.0

        if stage_end <= now + 4.0:
            if si == len(n_list) - 1:
                break
            continue

        if h_best.size == n:
            h_base = np.array(h_best, dtype=float, copy=True)
        else:
            h_base = interpolate_to(h_best, h_best.size, n)
        h_base = enforce_exact(h_base, n)
        h_best, c_best = consider_candidate(h_base, n, h_best, c_best)

        stage_dur = max(1.0, stage_end - time.time())

        # Palindromic local optimization: strong for boundary terms.
        pal_end = time.time() + 0.35 * stage_dur
        if pal_end > time.time() + 12.0 and n % 2 == 0:
            a_pal = pal_from_full(h_base, n)
            alphas = choose_alphas('parent', pal_end - time.time())
            maxits = make_maxit_list(n, pal_end - time.time())
            _, hpal, cpal, npal = run_lbfgs_start(a_pal, 'pal', pal_end, alphas, maxits, 1.0)
            h_best, c_best = consider_candidate(hpal, npal, h_best, c_best)

        # Anti-periodic starts.
        anti_end = time.time() + 0.78 * stage_dur
        if anti_end > time.time() + 10.0 and n % 2 == 0:
            starts = []
            starts.append((anti_from_full(h_base, n), 'parent'))
            if stage_dur > 25.0:
                starts.append((make_cos_anti(n), 'parent'))
            if n <= 1024 or stage_dur > 100.0:
                starts.append((make_fourier_anti_start(n, rng, 5, 0.45), 'random'))
            if total >= 800.0 and (n <= 1024 or stage_dur > 160.0):
                starts.append((make_fourier_anti_start(n, rng, 7, 0.42), 'random'))

            anti_time = max(5.0, anti_end - time.time())
            fracs = [0.45, 0.25, 0.15, 0.15][:len(starts)]
            if len(starts) == 1:
                fracs = [1.0]
            elif len(starts) == 2:
                fracs = [0.65, 0.35]
            elif len(starts) == 3:
                fracs = [0.50, 0.27, 0.23]

            for k, (a0, kind) in enumerate(starts):
                if time.time() >= anti_end - 3.0:
                    break
                if k == len(starts) - 1:
                    sub_end = min(anti_end, time.time() + anti_time)
                else:
                    sub_end = min(anti_end, time.time() + fracs[k] * anti_time)
                if sub_end <= time.time() + 4.0:
                    continue
                alphas = choose_alphas(kind, sub_end - time.time())
                maxits = make_maxit_list(n, sub_end - time.time())
                _, hf, cf, nn = run_lbfgs_start(a0, 'anti', sub_end, alphas, maxits, 0.0)
                h_best, c_best = consider_candidate(hf, nn, h_best, c_best)

        # Full / palindromic polishing at this resolution.
        if time.time() < stage_end - 15.0:
            rem = stage_end - time.time()
            pal_end = time.time() + 0.55 * rem
            if pal_end > time.time() + 8.0 and n % 2 == 0 and h_best.size == n:
                a_pal = pal_from_full(h_best, n)
                alphas = choose_alphas('parent', pal_end - time.time())
                maxits = make_maxit_list(n, pal_end - time.time())
                _, hpal, cpal, npal = run_lbfgs_start(a_pal, 'pal', pal_end, alphas, maxits, 1.0)
                h_best, c_best = consider_candidate(hpal, npal, h_best, c_best)

            if time.time() < stage_end - 15.0 and n <= 2048 and h_best.size == n:
                a_full = enforce_exact(h_best, n)
                alphas = choose_alphas('parent', stage_end - time.time())
                maxits = make_maxit_list(n, stage_end - time.time())
                _, hf, cf, nn = run_lbfgs_start(a_full, 'full', stage_end, alphas, maxits, 1.0)
                h_best, c_best = consider_candidate(hf, nn, h_best, c_best)

        if h_best.size <= 2048:
            h_rev = h_best[::-1]
            h_comp = 1.0 - h_best
            h_pal = 0.5 * (h_best + h_rev)
            for cand in (h_rev, h_comp, h_pal):
                h_best, c_best = consider_candidate(cand, cand.size, h_best, c_best)

    # Final targeted polish on the current incumbent.
    if time.time() < deadline - 25.0:
        n_rem = h_best.size
        if n_rem % 2 == 0:
            sub_end = min(deadline - 6.0, time.time() + 60.0)
            if sub_end > time.time() + 20.0:
                a = anti_from_full(h_best, n_rem)
                alphas = choose_alphas('parent', sub_end - time.time())
                maxits = make_maxit_list(n_rem, sub_end - time.time())
                _, ha, ca, na = polish_anti(a, n_rem, sub_end,
                                            alpha=300000.0,
                                            max_iter=max(100, int(make_maxit_list(n_rem, 60)[0])))
                h_best, c_best = consider_candidate(ha, na, h_best, c_best)

            sub_end = min(deadline - 3.0, time.time() + 35.0)
            if sub_end > time.time() + 12.0:
                a = pal_from_full(h_best, n_rem)
                alphas = choose_alphas('parent', sub_end - time.time())
                maxits = make_maxit_list(n_rem, sub_end - time.time())
                _, hp, cp, npf = run_lbfgs_start(a, 'pal', sub_end, alphas, maxits, 1.0)
                h_best, c_best = consider_candidate(hp, npf, h_best, c_best)
        else:
            sub_end = min(deadline - 5.0, time.time() + 50.0)
            if sub_end > time.time() + 12.0:
                a_full = enforce_exact(h_best, n_rem)
                alphas = choose_alphas('parent', sub_end - time.time())
                maxits = make_maxit_list(n_rem, sub_end - time.time())
                _, hf, cf, nn = run_lbfgs_start(a_full, 'full', sub_end, alphas, maxits, 1.0)
                h_best, c_best = consider_candidate(hf, nn, h_best, c_best)

    # Final symmetry checks and cleanup.
    if h_best is not None:
        h_rev = h_best[::-1]
        h_comp = 1.0 - h_best
        h_pal = 0.5 * (h_best + h_rev)
        for cand in (h_rev, h_comp, h_pal):
            h_best, c_best = consider_candidate(cand, cand.size, h_best, c_best)

    h_best = enforce_exact(h_best, len(h_best))
    c_best = direct_c5(h_best)
    return h_best, float(c_best), int(len(h_best))
