import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_logsumexp(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_logsumexp_grad(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_c5(h, n, alpha):
    """Compute gradient of smooth max of all correlation values."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    w = smooth_logsumexp_grad(c, alpha)
    
    # Instead of looping over all lags, use the structure
    # w[k] is the weight for lag k-(n-1)
    grad = np.zeros(n)
    n_active = np.sum(w > 1e-10)
    
    # For efficiency, only process active lags
    active_idx = np.where(w > 1e-10)[0]
    for idx in active_idx:
        lag = idx - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[m:n] += w[idx] * (1.0 - h[0:n-m])
            grad[0:n-m] += w[idx] * (-h[m:n])
        else:
            grad[0:n-m] += w[idx] * (1.0 - h[m:n])
            grad[m:n] += w[idx] * (-h[0:n-m])
    
    return grad * dx

def adam_smooth_max(h, n, alpha_start, alpha_end, deadline, lr=0.5, beta1=0.9, beta2=0.999, max_iter=50000):
    """Adam optimizer for smooth max C5."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    t = 0
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        # Anneal alpha
        frac = min(1.0, t / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        g = grad_c5(h_curr, n, alpha)
        
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * (g * g)
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_curr - step, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_new
        
        if patience > 300:
            lr *= 0.8
            m_v *= 0.5
            v_v *= 0.5
            patience = 0
        
        if it % 10000 == 9999:
            lr *= 0.9
    
    return best_h, best_c

            lag = ki - (n - 1)

                continue
            m = abs(lag)

                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        delta = min(d, (h[i] - h[j]) * 0.3)

                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-6:
                        delta = min(d, (h[j] - h[i]) * 0.3)

                        else:

                    # Break inner loop if improved

                if improved:

def binary_project_and_relax(h, n, deadline):
    """Project to binary, then do local bit-flips, then relax."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    # Create binary version
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n//2]
    h_bin[idx] = 1.0
    c_bin = compute_c5(h_bin, n)
    
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    else:
        best_h = h.copy()
    
    # Simulated annealing on binary
    h_sa = h_bin.copy()
    c_sa = c_bin
    for temp in [0.5, 0.3, 0.1, 0.05, 0.02]:
        for it in range(2000):
            if time.time() > deadline:
                return best_h, best_c
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h_sa[i] == h_sa[j]:
                continue
            # Swap
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            c_new = compute_c5(h_sa, n)
            if c_new < c_sa or np.random.random() < np.exp(-(c_new - c_sa) / temp):
                c_sa = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_sa.copy()
            else:
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000, top_m=None):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 
             'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    c = compute_all_lags(h, n)
    if top_m is not None:
        top_idx = np.argsort(c)[-top_m:]
        def obj(x):
            cc = compute_all_lags(x, n)
            active = cc[top_idx]
            return smooth_logsumexp(active, alpha)
    else:
        def obj(x):
            return smooth_logsumexp(compute_all_lags(x, n), alpha)
    
    h_proj = project_feasible(h.copy(), n)
    try:
        res = minimize(obj, h_proj, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h_proj

def slsqp_polish(h, n, alpha, maxiter=2000, top_m=5):
    n_c = len(h)
    bounds = [(0.0, 1.0)] * n_c
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac, 'args': (n_c,)}]
    
    c_all = compute_all_lags(h, n_c)
    if top_m is not None:
        top_idx = np.argsort(c_all)[-top_m:]
        obj_fn = lambda x: obj_smooth_max_top(x, n_c, alpha, top_idx)
    else:
        obj_fn = lambda x: obj_smooth_max_all(x, n_c, alpha)
    
    h0 = project_feasible(h.copy(), n_c)
    try:
        res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n_c)
        return h_new
    except Exception:
        return h0

    return best_h, best_c, len(best_h)
