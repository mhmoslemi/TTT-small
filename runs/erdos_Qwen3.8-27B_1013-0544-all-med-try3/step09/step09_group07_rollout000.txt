        obj_val += penalty_coeff * diff * diff

        remaining = deadline - 5 - time.time()
        maxit = min(3000, int(remaining * 100))
        if maxit < 50:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(remaining, 40), maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def multi_restart_optimize(h, n, deadline, rng, num_restarts=3):
    """Try several perturbations and optimize each."""
    best_h, best_c = h.copy(), compute_c5(h, n)
    per_restart_time = (deadline - time.time()) / (num_restarts + 1)

    for r in range(num_restarts):
        if time.time() > deadline - 10:
            break
        # Perturb
        scale = 0.02 * (1 + r * 0.5)
        h_pert = project_feasible(best_h + rng.normal(0, scale, n), n)
        # Optimize
        h_opt, c_opt = high_alpha_lbfgsb(h_pert, n, time.time() + per_restart_time * 0.7)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Phase 0: Interpolate to n=1000 and quick settle
    n_target = 1000
    h_cur = expand_to_resolution(h_init, n_init, n_target)
    best_h = h_cur.copy()
    best_c = compute_c5(best_h, n_target)

    # Phase 1: Quick L-BFGS-B settle at n=1000
    if time.time() < deadline - 400:
        h_opt, c_opt = high_alpha_lbfgsb(best_h, n_target, time.time() + 120)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Targeted local search
    if time.time() < deadline - 200:
        h_opt, c_opt = targeted_local_search(best_h, n_target, time.time() + 180, rng,
                                              num_hags=3, max_rounds=30000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: L-BFGS-B polish
    if time.time() < deadline - 150:
        h_opt, c_opt = high_alpha_lbfgsb(best_h, n_target, time.time() + 120)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: More local search
    if time.time() < deadline - 100:
        h_opt, c_opt = targeted_local_search(best_h, n_target, time.time() + 80, rng,
                                              num_hags=2, max_rounds=25000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Multi-restart
    if time.time() < deadline - 80:
        h_opt, c_opt = multi_restart_optimize(best_h, n_target, time.time() + 70, rng, num_restarts=2)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Gradient descent on max lags
    if time.time() < deadline - 40:
        h_opt, c_opt = gradient_descent_max(best_h, n_target, time.time() + 35,
                                             lr=0.0005, num_hags=2, momentum=0.85)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final L-BFGS-B
    if time.time() < deadline - 15:
        h_opt, c_opt = high_alpha_lbfgsb(best_h, n_target, deadline - 5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Tiny local search
    if time.time() < deadline - 3:
        h_opt, c_opt = targeted_local_search(best_h, n_target, deadline - 1, rng,
                                              num_hags=2, max_rounds=15000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Also try n=500 version and compare
    if best_c > 0.3808 and time.time() < deadline - 20:
        h_500 = project_feasible(h_init, n_init)
        h_opt, c_opt = high_alpha_lbfgsb(h_500, n_init, deadline - 10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_target = n_init

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
