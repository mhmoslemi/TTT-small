                grad[k:] += wmi * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wmi * (1.0 - x[:nv])
                grad[:nv] += wmi * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def do_lbfgs(h, n, alpha, mu, ktop=0, maxiter=3000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        if ktop > 0:
            res = minimize(topk_obj, h, jac=topk_grad,
                          args=(n, ktop, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxiter * 4, 'maxls': 40})
        else:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxiter * 4, 'maxls': 40})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return project_feasible(h, n), compute_c5(h, n)

def do_slqp(h, n, deadline, alpha=1000000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)

    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)

    def sq_jac_fn(x, na):
        return np.ones(na)

    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def reweighted_gradient(h, n, deadline, alpha, n_iter=5000):
    """Projected gradient with adaptive shift weighting."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    n_shifts = 2 * n - 1
    eta = 0.05

    for it in range(n_iter):
        if time.time() > deadline:
            break

        c = compute_c5_vec(h, n)
        # Adaptive weights emphasizing worst shifts
        m = c.max()
        log_w = alpha * (c - m)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w /= w.sum()

        # Weighted gradient
        grad = np.zeros(n)
        for idx in range(n_shifts):
            wi = w[idx]
            if wi < 1e-13:
                continue
            k = idx - center
            wmi = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[:nv] += wmi * (1.0 - h[k:])
                    grad[k:] += wmi * (-h[:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:] += wmi * (1.0 - h[:nv])
                    grad[:nv] += wmi * (-h[kk:])

        # Projected gradient step with backtracking
        current_c = compute_c5(h, n)
        h_trial = project_feasible(h - eta * grad, n)
        c_trial = compute_c5(h_trial, n)

        step_count = 0
        while c_trial >= current_c and step_count < 10:
            eta *= 0.5
            h_trial = project_feasible(h - eta * grad, n)
            c_trial = compute_c5(h_trial, n)
            step_count += 1

        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
        h = h_trial

        # Occasionally try larger alpha
        if it % 500 == 499:
            alpha = min(alpha * 1.5, 50000)

    return best_h, best_c

def targeted_refine(h, n, deadline, max_rounds=300):
    """Pinpoint adjustments to reduce worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(99)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        top_k = np.argpartition(c_vec, -5)[-5:]

        improved = False
        for widx in top_k:
            k = int(widx) - (n - 1)
            if k >= 0:
                overlap_idx = np.arange(0, n - k)
            else:
                overlap_idx = np.arange(-k, n)

            # Sample a subset of points to try
            n_try = min(80, len(overlap_idx))
            if len(overlap_idx) <= n_try:
                trial_pts = overlap_idx
            else:
                trial_pts = rng.choice(overlap_idx, n_try, replace=False)

            for i in trial_pts:
                iv = int(i)
                for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.003, -0.003]:
                    nv = best_h[iv] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    old_val = best_h[iv]
                    best_h[iv] = nv
                    new_c = compute_c5(best_h, n)
                    if new_c < best_c - 1e-15:
                        best_c = new_c
                        improved = True
                    else:
                        best_h[iv] = old_val
                    if time.time() > deadline:
                        return best_h, best_c

        if not improved and rnd > 50:
            break

    return best_h, best_c

def perturbation_search(h, n, deadline, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(2024)
    bounds = [(0.0, 1.0)] * n
    mags = [0.015, 0.03, 0.02, 0.008, 0.04, 0.025, 0.05, 0.012]

    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            h_new, c_new = do_lbfgs(h_p, n, alpha=1000, mu=3000.0, maxiter=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c

def multi_alpha_optimize(h, n, deadline):
    """Progressive alpha increase for convergence."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [
        (100, 100.0, 2000),
        (300, 300.0, 2000),
        (800, 1000.0, 2000),
        (2000, 3000.0, 2000),
        (5000, 8000.0, 2000),
        (10000, 15000.0, 1500),
        (20000, 30000.0, 1000),
    ]
    for alpha, mu, maxit in alphas:
        if time.time() > deadline - 15:
            break
        h_new, c_new = do_lbfgs(best_h, n, alpha, mu, maxiter=maxit)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: Multi-alpha L-BFGS at base resolution (~80s)
    t1 = min(t0 + 80, deadline - 700)
    if time.time() < t1:
        best_h, best_c = multi_alpha_optimize(best_h, n_base, t1)

    # Phase 2: SLSQP polish at base (~50s)
    t2 = min(time.time() + 50, deadline - 550)
    if time.time() < t2 and n_base <= 600:
        h_s, c_s = do_slqp(best_h, n_base, t2, alpha=3000000, maxiter=1500)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 3: Top-k focused optimization at base (~50s)
    t3 = min(time.time() + 50, deadline - 450)
    if time.time() < t3:
        h_tk, c_tk = do_lbfgs(best_h, n_base, alpha=8000, mu=10000.0, ktop=15, maxiter=2500)
        if c_tk < best_c:
            best_c = c_tk
            best_h = h_tk
        h_tk2, c_tk2 = do_lbfgs(best_h, n_base, alpha=20000, mu=20000.0, ktop=10, maxiter=2000)
        if c_tk2 < best_c:
            best_c = c_tk2
            best_h = h_tk2

    # Phase 4: Upsample to n=1000 and optimize (~250s)
    n1 = 1000
    if time.time() < deadline - 350:
        h1000 = interpolate_to(best_h, n_base, n1)
        c1000 = compute_c5(h1000, n1)

        # Progressive alpha at n=1000
        t4 = min(time.time() + 120, deadline - 180)
        if time.time() < t4:
            h1000, c1000 = multi_alpha_optimize(h1000, n1, t4)

        if c1000 < best_c:
            best_c = c1000
            best_h = h1000

        # Top-k at n=1000
        t5 = min(time.time() + 80, deadline - 100)
        if time.time() < t5:
            h_tk3, c_tk3 = do_lbfgs(best_h, n1, alpha=5000, mu=10000.0, ktop=20, maxiter=2000)
            if c_tk3 < best_c:
                best_c = c_tk3
                best_h = h_tk3
            h_tk4, c_tk4 = do_lbfgs(best_h, n1, alpha=15000, mu=20000.0, ktop=10, maxiter=1500)
            if c_tk4 < best_c:
                best_c = c_tk4
                best_h = h_tk4

    # Phase 5: Reweighted gradient at current resolution (~80s)
    t6 = min(time.time() + 80, deadline - 80)
    if time.time() < t6:
        h_rw, c_rw = reweighted_gradient(best_h, len(best_h), t6, alpha=500, n_iter=3000)
        if c_rw < best_c:
            best_c = c_rw
            best_h = h_rw

    # Phase 6: Perturbation search (~60s)
    t7 = min(time.time() + 60, deadline - 50)
    if time.time() < t7:
        best_h, best_c = perturbation_search(best_h, len(best_h), t7, n_trials=5)

    # Phase 7: Final targeted refinement (~40s)
    t8 = min(time.time() + 40, deadline - 5)
    if time.time() < t8:
        best_h, best_c = targeted_refine(best_h, len(best_h), t8, max_rounds=200)

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
