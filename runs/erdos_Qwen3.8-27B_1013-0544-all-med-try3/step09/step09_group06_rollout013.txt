def project_feasible(h, n):

def _lag_gradient(h, n, lag):

If lag < 0: let m = -lag > 0. i ranges from m to n-1
  c[k] = sum_{i=m}^{n-1} h[i-m] * (1-h[i])

So for lag m > 0: C(m) = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
For lag 0: C(0) = sum_{i=0}^{n-1} h[i] * (1-h[i])
For lag -m (m > 0): C(-m) = sum_{i=m}^{n-1} h[i-m] * (1-h[i]) = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])

Interesting! So C(-m) = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m]) which is the "directed" overlap: positions where h=1 now but h=0 after shift m.

And C(m) = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i]) which is: positions where h=1 after shift m but h=0 now.

So C(m) + C(-m) = sum_{i=0}^{n-1-m} [h[i+m]*(1-h[i]) + h[i]*(1-h[i+m])] = sum_{i=0}^{n-1-m} (h[i]+h[i+m]-2*h[i]*h[i+m])

For binary h, C(m) counts positions where h shifted is 1 but h is 0, and C(-m) counts positions where h is 1 but h shifted is 0.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -h[0:n-m]
    return grad

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-8:
            continue
        lag = k - (n-1)
        grad += w[k] * _lag_gradient(h, n, lag)
    return grad * dx

def grad_momentum(h, n, alpha, deadline, lr=1.0, momentum=0.90, max_iter=100000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)

        if patience > 300:
            lr *= 0.85
            v *= 0.3
            patience = 0
        if it % 10000 == 9999:
            lr = max(0.05, lr * 0.8)
    return best_h, best_c

def slsqp_active(h, n, alpha, active_indices, maxiter=2000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        active_vals = c[active_indices]
        m = active_vals.max()
        shifted = alpha * (active_vals - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binarize_and_fix(h, n):
    """Push values toward 0 or 1 while preserving sum."""
    h_bin = np.zeros(n)
    # Use a soft threshold based on the actual values
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:n//2]] = 1.0
    # Try different thresholds
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    for thresh in np.linspace(0.3, 0.7, 9):
        hb = (h >= thresh).astype(float)
        ones_c = int(hb.sum())
        target = n // 2
        if ones_c > target:
            zeros_ones = np.where(hb == 1)[0]
            order = np.argsort(h[zeros_ones])
            hb[zeros_ones[order[:ones_c-target]]] = 0
        elif ones_c < target:
            zero_zeros = np.where(hb == 0)[0]
            order = np.argsort(-h[zero_zeros])
            hb[zero_zeros[order[:target-ones_c]]] = 1
        hb = project_feasible(hb, n)
        c_t = compute_c5(hb, n)
        if c_t < best_c:
            best_c = c_t
            best_h = hb
    return best_h, best_c
