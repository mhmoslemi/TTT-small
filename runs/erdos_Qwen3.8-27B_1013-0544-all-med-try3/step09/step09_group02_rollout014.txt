    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    consecutive_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.001 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            active = [int(np.argmax(c_vec))]
        if len(active) > 25:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:25]]
        
        # Build and solve LP
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        c_vals = c_vec[active]
        n_act = len(active)
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in range(n_act):
            g = shift_grad_single(h, n, active[j])
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[j]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0; b_ub[row] = h[i]; row += 1
        for i in range(n):
            A_ub[row, i] = 1.0; b_ub[row] = 1.0 - h[i]; row += 1
        for i in range(n):
            A_ub[row, i] = -1.0; b_ub[row] = rho; row += 1
        for i in range(n):
            A_ub[row, i] = 1.0; b_ub[row] = rho; row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        bounds = [(None, None)] * n + [(0.0, 2.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                h_new = project_feasible(h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    consecutive_no_improve = 0
                    if rho < 0.08:
                        rho = min(rho * 1.2, 0.08)
                else:
                    consecutive_no_improve += 1
                    rho *= 0.5
            else:
                consecutive_no_improve += 1
                rho *= 0.7
        except Exception:
            consecutive_no_improve += 1
            rho *= 0.5
        
        if rho < 1e-10 or consecutive_no_improve > 30:
            break
    return best_h, best_c

def subgradient_method(h, n, deadline, max_iter=5000, step_init=0.005):
    """Direct subgradient method on exact max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    n_bad = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        current_c = c_vec.max()
        
        # Active set: top 50 weights
        top_k = 50
        active = np.argsort(-c_vec)[:top_k]
        # Weight by proximity to max
        spread = c_vec[active] - (c_vec.max() - 0.003)
        weights = np.maximum(spread, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights /= weights.sum()
        
        g = np.zeros(n)
        for idx, w in zip(active, weights):
            if w < 1e-10:
                continue
            g += w * shift_grad_single(h, n, idx)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Backtracking line search
        s = step
        found = False
        for _ in range(15):
            trial = project_feasible(h - s * g, n)
            trial_c = compute_c5(trial, n)
            if trial_c < current_c - 1e-10:
                h = trial
                if trial_c < best_c:
                    best_c = trial_c
                    best_h = h.copy()
                n_bad = 0
                found = True
                break
            s *= 0.6
        if not found:
            n_bad += 1
            if n_bad > 50:
                break
            step *= 0.7
        else:
            if step < step_init:
                step = min(step * 1.05, step_init)
    
    return best_h, best_c

    best_c = compute_c5(h, n)
    rng = np.random.default_rng(42)

Let me also think about a completely different angle: **What if I use a different n?**

The objective is C₅ = max_k (correlation_value * dx) where dx = 2/n.

For a given "shape" of h, increasing n changes dx but also changes the number of correlation values and their values. At n=1000, dx = 0.002, so the correlation values are halved compared to n=500 (where dx = 0.004). Wait no - the correlation values also scale.

Actually: np.correlate(h, 1-h, mode='full')[m] = sum_i h[i] * (1-h[i+m-(n-1)]) for valid i.

If I double n (and double the function, so each original sample becomes two), the new correlation at the "same" physical shift would be:
- Old: sum_{i} h[i] * (1-h[i+k]) for k physical samples
- New: sum_{i} h'[i] * (1-h'[i+2k]) ≈ 2 * sum_{i} h[i] * (1-h[i+k]) (because there are twice as many terms)

And dx halves. So the product (correlation * dx) stays the same! This means that simply upresolving doesn't change the objective value (approximately). The benefit of higher n comes from being able to represent finer features in h.
