def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    """Gradient of c_k with respect to h, for a single shift."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_val(cvec, alpha):
    m = cvec.max()
    sh = alpha * (cvec - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad_weights(cvec, alpha):
    m = cvec.max()
    sh = alpha * (cvec - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    stale = 0
    target_val = n / 2.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > max_active:
            order = np.argsort(-c_vec[active])
            active = [active[i] for i in order[:max_active]]

        n_act = len(active)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0

        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in active:
            g = shift_grad(h, n, j)
            rhs = -c_vec[j] + g.dot(h)
            A_ub[row, 0:n] = g
            A_ub[row, n] = -1.0
            b_ub[row] = rhs
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = -h[i] + rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = h[i] + rho
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = 0.0
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0
            row += 1

        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([target_val])
        bounds_lp = [(None, None)] * n + [(-0.5, 1.0)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs',
                          options={'presolve': True})
            if not res.success:
                rho *= 0.5
                stale += 1
                if rho < 1e-12 or stale > 40:
                    break
                continue
            h_lp = res.x[0:n]
        except Exception:
            rho *= 0.5
            stale += 1
            if rho < 1e-12 or stale > 40:
                break
            continue

        # Line search along direction h_lp - h
        direction = h_lp - h
        max_step = 1.0
        best_t = 0.0
        base_c = compute_c5(h, n)
        for t_try in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            h_try = h + t_try * direction
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < base_c - 1e-15:
                best_t = t_try
                break
        if best_t == 0.0:
            h_try = project_feasible(h_lp, n)
            c_try = compute_c5(h_try, n)
            if c_try < base_c - 1e-15:
                best_t = 1.0
            else:
                rho *= 0.5
                stale += 1
                if rho < 1e-12 or stale > 40:
                    break
                continue

        h_new = project_feasible(h + best_t * direction, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stale = 0
            if rho < 0.08:
                rho = min(rho * 1.4, 0.12)
        else:
            stale += 1
            rho *= 0.5
            if rho < 1e-12 or stale > 40:
                break

    return best_h, best_c

def fine_local_search(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    steps = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    target_val = n / 2.0

    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= thresh)[0]

            grad = np.zeros(n)
            for idx in active:
                grad += shift_grad(h, n, idx)
            if len(active) > 0:
                grad /= float(len(active))

            order = np.argsort(-np.abs(grad))
            top_count = min(radius * 2, n)
            for ii in order[:top_count]:
                if time.time() > deadline:
                    return h, best_c
                i = int(ii)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 + 1e-10 and h[j] - step >= -1e-10:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if h[i] - step >= -1e-10 and h[j] + step <= 1.0 + 1e-10:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def slqp_polish(h, n, deadline, alpha=3000000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        return smooth_max_val(c, alpha)
    def grad_fn(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_weights(c, alpha)
        g = np.zeros(n)
        for s in range(2 * n - 1):
            if w[s] < 1e-15:
                continue
            g += w[s] * shift_grad(x, n, s)
        return g
    def eq_fun(x):
        return x.sum() - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coordinate_descent_polish(h, n, deadline):
    """Very fine 1D coordinate descent on each variable."""
    best_c = compute_c5(h, n)
    target_val = n / 2.0
    for _ in range(5):
        if time.time() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            lo_bound = max(0.0, old_val - 0.01)
            hi_bound = min(1.0, old_val + 0.01)
            best_v = old_val
            best_c_i = best_c
            for trial in np.linspace(lo_bound, hi_bound, 11):
                h[i] = trial
                h_adjust = project_feasible(h, n)
                c_try = compute_c5(h_adjust, n)
                if c_try < best_c_i - 1e-15:
                    best_c_i = c_try
                    best_v = trial
            if best_v != old_val:
                h[i] = best_v
                h = project_feasible(h, n)
                if best_c_i < best_c:
                    best_c = best_c_i
                    improved_any = True
        if not improved_any:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(500, 0.5)

    # Start at n=1000
    n = 1000
    if len(h_init) != n:
        x_old = np.linspace(0, 2, len(h_init), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_init = np.interp(x_new, x_old, h_init)
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS warmup at n=1000 (50s)
    t1 = min(t0 + 50, deadline - 750)
    if t1 > time.time() + 5:
        h, best_c = run_lbfgs(h, n, t1, [300, 1500, 6000, 20000])
        best_h = h.copy()

    # Phase 2: SQP main optimization (200s)
    t2 = min(t0 + 250, deadline - 550)
    if t2 > time.time() + 10:
        h, best_c = sqp_with_linesearch(h, n, t2, max_iter=500, rho_init=0.05, max_active=30)
        best_h = h.copy()

    # Phase 3: Fine local search (80s)
    t3 = min(t0 + 330, deadline - 450)
    if t3 > time.time() + 5:
        h, best_c = fine_local_search(h, n, t3, radius=60)
        best_h = h.copy()

    # Phase 4: SQP refinement (120s)
    t4 = min(t0 + 450, deadline - 350)
    if t4 > time.time() + 10:
        h, best_c = sqp_with_linesearch(h, n, t4, max_iter=400, rho_init=0.02, max_active=40)
        best_h = h.copy()

    # Phase 5: Perturbation + SQP (80s)
    if time.time() < deadline - 280:
        h_pert = best_h + np.random.randn(n) * 0.008
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        t5 = min(time.time() + 70, deadline - 180)
        if t5 > time.time() + 10:
            h_pert, c_pert = sqp_with_linesearch(h_pert, n, t5, max_iter=200, rho_init=0.03, max_active=30)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()

    # Phase 6: SLSQP polish (80s)
    t6 = min(time.time() + 80, deadline - 60)
    if t6 > time.time() + 5:
        h_p, c_p = slqp_polish(best_h, n, t6, alpha=5000000)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()

    # Phase 7: Coordinate descent polish (30s)
    if time.time() < deadline - 30:
        h_cd, c_cd = coordinate_descent_polish(best_h, n, deadline - 5)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd.copy()

    # Phase 8: Try n=2000 if time permits
    if time.time() < deadline - 150:
        n_hi = 2000
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_hi = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_hi, x_old, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c_hi = compute_c5(h_hi, n_hi)

        t7 = min(time.time() + 100, deadline - 50)
        if t7 > time.time() + 10:
            h_hi, c_hi = sqp_with_linesearch(h_hi, n_hi, t7, max_iter=200, rho_init=0.04, max_active=25)

        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            n = n_hi

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
