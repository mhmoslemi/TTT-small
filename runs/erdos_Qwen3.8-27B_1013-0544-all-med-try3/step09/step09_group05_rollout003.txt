import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample_func(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def active_set_obj(x, n, active_idx, alpha, mu):
    c_vec = np.correlate(x, 1 - x, mode='full')
    vals = c_vec[active_idx] * (2.0 / n)
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh_max = sh.max()
    if sh_max > 500:
        val = mx + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(x, n, active_idx, alpha, mu):
    c_vec = np.correlate(x, 1 - x, mode='full')
    vals = c_vec[active_idx] * (2.0 / n)
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    dx = 2.0 / n
    n_half = n - 1
    grad = np.zeros(n)
    for ki, ai in enumerate(active_idx):
        if w[ki] < 1e-14:
            continue
        k = ai - n_half
        wm = w[ki] * alpha * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_lse_obj(x, n, alpha, mu):
    c_vec = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    mx = c_vec.max()
    sh = alpha * (c_vec - mx)
    sh_max = sh.max()
    if sh_max > 500:
        val = mx + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_lse_grad(x, n, alpha, mu):
    c_vec = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    mx = c_vec.max()
    sh = alpha * (c_vec - mx)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    dx = 2.0 / n
    n_half = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - n_half
        wm = w[m] * alpha * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def find_active_shifts(h, n, rel_tol=0.005, top_k=8):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val * (1.0 - rel_tol)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def lbfgs_optimize(h, n, deadline, obj_fn, grad_fn, obj_args, grad_args,
                   maxiter=5000, maxfun=50000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn,
                       args=obj_args + grad_args,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxfun, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

    bounds = [(0.0, 1.0)] * n

def subgradient_steps(h, n, deadline, n_steps=200, step_init=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    dx = 2.0 / n
    n_half = n - 1

    for s in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(c_vec)
        k = max_idx - n_half

        if k >= 0:

            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx

        h_try = h - step * grad
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)

            best_c = c_try
            best_h = h_try.copy()
            step = min(step * 1.05, 0.05)

            step *= 0.7
            if step < 1e-7:
