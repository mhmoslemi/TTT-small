import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    if mx_sh > 500:
        val = m + (mx_sh + np.log(np.sum(np.exp(sh - mx_sh)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    w = np.exp(sh - mx_sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    np1 = n - 1
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - np1
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:

    if mu > 0:

    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)

            pass
    return best_h, best_c

def active_optimize(h, n, deadline, top_k=4, alphas=None, mu=0.002, rounds=40):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    for rnd in range(rounds):
        rem = deadline - time.time()
        if rem < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(100, int(rem2 * 2.0))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, mu), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-24,
                                        'gtol': 1e-20, 'maxfun': maxit * 12,
                                        'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-18:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def single_dive(h, n, deadline, alpha=10000000, max_rounds=100):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for it in range(max_rounds):
        rem = deadline - time.time()
        if rem < 8:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        for a in [alpha, alpha * 3]:
            if time.time() > deadline - 4:
                break
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, a, 0.001), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-24,
                                        'gtol': 1e-18, 'maxfun': 1500,
                                        'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-18:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_pattern(n, pattern_type=0):
    h = np.zeros(n)
    n_ones = n // 2
    if pattern_type == 0:
        for i in range(n):
            if (i * 7 + 3) % n < n_ones:
                h[i] = 1.0
    elif pattern_type == 1:
        golden = (1 + np.sqrt(5)) / 2
        for i in range(n):
            idx = int((i * golden) % n)
            h[idx] = 1.0
        h = np.clip(h, 0, 1)
    elif pattern_type == 2:
        rng = np.random.RandomState(42 + pattern_type)
        ones = rng.choice(n, n_ones, replace=False)
        h[ones] = 1.0
    else:
        quarter = n // 4
        h[0:quarter] = 1.0
        h[quarter:2*quarter] = 0.0
        h[2*quarter:3*quarter] = 1.0
        h[3*quarter:n] = 0.0
        n_ones_actual = 2 * quarter
        if n_ones_actual != n_ones:
            diff = n_ones - n_ones_actual
            if diff > 0:
                for i in range(n):
                    if h[i] == 0 and diff > 0:
                        h[i] = 1.0
                        diff -= 1
            elif diff < 0:
                for i in range(n-1, -1, -1):
                    if h[i] == 1 and diff < 0:
                        h[i] = 0.0
                        diff += 1
    return project_feasible(h, n)

def pair_search(h, n, deadline, radius=30, max_pairs=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    pairs_tried = 0
    for d in deltas:
        improved = True
        while improved and time.time() < deadline and pairs_tried < max_pairs:
            improved = False
            step = max(1, (n * 200) // max_pairs)
            for i in range(0, n, step):
                if time.time() > deadline or pairs_tried >= max_pairs:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j or pairs_tried >= max_pairs:
                        continue
                    pairs_tried += 1
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        new_h = project_feasible(best_h, n)
                        c = compute_c5(new_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        new_h = project_feasible(best_h, n)
                        c = compute_c5(new_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        np1 = n - 1
        for idx in active:
            k = idx - np1
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.005)
        else:
            eta *= 0.95
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution
    p1_end = min(t0 + 40, deadline - 600)
    if p1_end > t0 + 5:
        h1, c1 = lbfgs_optimize(best_h, best_n, p1_end,
            alphas=[50, 500, 5000, 50000], mu=0.01,
            maxiters_list=[500, 800, 1000, 1200])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to high resolution
    n_high = 2000
    if time.time() < deadline - 500:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c + 0.002:
            p2_end = min(time.time() + 100, deadline - 450)
            h2, c2 = lbfgs_optimize(h_high, n_high, p2_end,
                alphas=[10, 50, 200, 1000, 5000, 20000],
                mu=0.003, maxiters_list=[400, 600, 800, 1000, 1200, 1500])
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_n = n_high

    # Phase 3: Refine with higher alpha at high resolution
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 150, deadline - 300)
        h3, c3 = lbfgs_optimize(best_h, best_n, p3_end,
            alphas=[50000, 200000, 1000000],
            mu=0.002, maxiters_list=[1500, 2000, 2500])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Active set optimization
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 120, deadline - 150)
        h4, c4 = active_optimize(best_h, best_n, p4_end, top_k=4,
            alphas=[100000, 500000, 2000000, 10000000, 50000000],
            mu=0.001, rounds=50)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Single worst constraint dive
    if time.time() < deadline - 130:
        p5_end = min(time.time() + 70, deadline - 100)
        h5, c5 = single_dive(best_h, best_n, p5_end, alpha=20000000, max_rounds=80)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Diverse restarts - try binary patterns at high resolution
    if time.time() < deadline - 100 and best_n >= 1000:
        for pt in range(3):
            if time.time() > deadline - 80:
                break
            h_bin = binary_pattern(best_n, pt)
            c_bin = compute_c5(h_bin, best_n)
            if c_bin < best_c + 0.05:
                p6_end = min(time.time() + 30, deadline - 60)
                h6, c6 = lbfgs_optimize(h_bin, best_n, p6_end,
                    alphas=[100, 10000, 500000],
                    mu=0.01, maxiters_list=[500, 800, 1000])
                if c6 < best_c:
                    best_c = c6
                    best_h = h6

    # Phase 7: Final active set + dive
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 30, deadline - 30)
        h7, c7 = active_optimize(best_h, best_n, p7_end, top_k=3,
            alphas=[500000, 5000000, 50000000],
            mu=0.001, rounds=30)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Subgradient refinement
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 12, deadline - 8)
        h8, c8 = subgrad_refine(best_h, best_n, p8_end, n_iter=10000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final pair search
    if time.time() < deadline - 3:
        p9_end = deadline - 1
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=20, max_pairs=30000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: one more quick active set push
    if time.time() < deadline - 0.5:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 50000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 100, 'ftol': 1e-24,
                                    'gtol': 1e-18, 'maxfun': 300})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
