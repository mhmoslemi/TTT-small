import numpy as np
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_thue_morse(n):
    """Create Thue-Morse based binary pattern with n/2 ones"""
    # Generate Thue-Morse sequence of length n
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i // 2] ^ (i % 2)
    # This gives a balanced sequence (n/2 zeros, n/2 ones for n power of 2)
    # For general n, we might not have exactly n/2 ones
    ones_in_tm = tm.sum()
    h = tm.astype(float)
    if ones_in_tm != n // 2:
        # Adjust: if too many ones, change some to 0; if too few, change some to 1
        # Prefer to change positions that are far from existing same-type neighbors
        if ones_in_tm > n // 2:
            excess = ones_in_tm - n // 2
            # Change some 1s to 0s (prefer isolated 1s)
            ones_pos = np.where(tm == 1)[0]
            np.random.shuffle(ones_pos)
            for p in ones_pos:
                if excess <= 0:
                    break
                h[p] = 0
                excess -= 1
        else:
            deficit = n // 2 - ones_in_tm
            zeros_pos = np.where(tm == 0)[0]
            np.random.shuffle(zeros_pos)
            for p in zeros_pos:
                if deficit <= 0:
                    break
                h[p] = 1
                deficit -= 1
    return h

def make_beatty(n, alpha):
    """Create Beatty sequence pattern"""
    h = np.zeros(n)
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n // 2:
                break
    for pos in positions:
        h[pos] = 1.0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
