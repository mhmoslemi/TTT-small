def make_fourier_binary(n,rng,comp=5):
  t=np.arange(n,dtype=float)
  score=np.zeros(n)
  maxf=max(2, min(100,n//5))

    phi=rng.uniform(0,2*math.pi)

  target=n//2
  h=np.zeros(n)

    idx=np.argpartition(score,target-1)[:target]
    h[idx]=1.0
  return h

import numpy as np

from scipy.optimize import minimize


def compute_c5_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if val < target:
            lo = mid
        else:
            hi = mid

    p = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(p.sum())

    if abs(diff) > 1e-15:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - p))
            add = min(diff, 1.0 - p[idx])
            if add > 0.0:
                p[idx] += add
        else:
            idx = int(np.argmax(p))
            sub = min(-diff, p[idx])
            if sub > 0.0:
                p[idx] -= sub

    diff = target - float(p.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - p
            tot = float(room.sum())
            if tot > 1e-15:
                p += (diff / tot) * room
        else:
            tot = float(p.sum())
            if tot > 1e-15:
                p += (diff / tot) * p

    return np.clip(p, 0.0, 1.0)


def top_k_indices(arr, k):
    if k <= 0:
        return np.array([], dtype=int)
    if k >= arr.size:
        return np.arange(arr.size)
    return np.argpartition(-arr, k)[:k]


def top_n_half(h, n):
    target = n // 2
    out = np.zeros(n, dtype=float)
    if target > 0 and target < n:
        idx = np.argpartition(-h, target - 1)[:target]
        out[idx] = 1.0
    return out


def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1.0
    return h


def make_beatty_binary(n, alpha, rng):
    h = np.zeros(n, dtype=float)
    target = n // 2
    count = 0
    for k in range(1, 20 * n + 1):
        pos = int((k * alpha)) % n
        if h[pos] == 0.0:
            h[pos] = 1.0
            count += 1
            if count >= target:
                break
    if count < target:
        zeros = np.where(h == 0.0)[0]
        need = target - count
        if zeros.size > 0:
            take = min(need, zeros.size)
            idx = rng.choice(zeros, size=take, replace=False)
            h[idx] = 1.0
    return h


def make_fourier_binary(n, rng, comp=5):
    t = np.arange(n, dtype=float)
    score = np.zeros(n, dtype=float)
    maxf = max(2, min(100, n // 5))
    for _ in range(comp):
        f = int(rng.randint(1, maxf + 1))
        phi = rng.uniform(0.0, 2.0 * math.pi)
        score += np.sin((2.0 * math.pi * f / float(n)) * t + phi)
    target = n // 2
    h = np.zeros(n, dtype=float)
    if target > 0:
        idx = np.argpartition(score, target - 1)[:target]
        h[idx] = 1.0
    return h


def make_block_binary(n, b, rng):
    h = (np.arange(n, dtype=float) / float(b)).astype(int) % 2
    h = h.astype(float)
    target = n // 2
    cnt = int(h.sum())
    if cnt < target:
        zeros = np.where(h == 0.0)[0]
        need = target - cnt
        if zeros.size > 0:
            idx = rng.choice(zeros, size=min(need, zeros.size), replace=False)
            h[idx] = 1.0
    elif cnt > target:
        ones = np.where(h == 1.0)[0]
        need = cnt - target
        if ones.size > 0:
            idx = rng.choice(ones, size=min(need, ones.size), replace=False)
            h[idx] = 0.0
    return h


def upsample_box(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    if new_n % n == 0:
        out = np.repeat(h, new_n // n)
    else:
        x_old = np.linspace(0.0, new_n - 1, n)
        x_new = np.arange(new_n, dtype=float)
        out = np.interp(x_new, x_old, h)
    return project_sum(out, new_n)


def downsample_linear(h, new_n):
    n = len(h)
    if new_n >= n:
        return upsample_box(h, new_n)
    x_old = np.linspace(0.0, new_n - 1, n)
    x_new = np.arange(new_n, dtype=float)
    out = np.interp(x_new, x_old, h)
    return project_sum(out, new_n)

def apply_move_to_c(c, h, g, g_rev, dx, p, q, n, a):
    off = n - 1
    da = dx * a
    aa = dx * a * a

    c[off + p - n + 1: off + p + 1] += da * g_rev
    c[off - p: off - p + n] -= da * h

    c[off + q - n + 1: off + q + 1] -= da * g_rev
    c[off - q: off - q + n] += da * h

    c[off] -= 2.0 * aa
    c[off + p - q] += aa
    c[off + q - p] += aa

            inc = np.where(h < 1.0 - 1e-6)[0]
            dec = np.where(h > 1e-6)[0]
            if inc.size > 0 and dec.size > 0:
                for _ in range(12):
                    prec = int(inc[rng.randint(0, inc.size)])
                    qdon = int(dec[rng.randint(0, dec.size)])
                    if prec == qdon:
                        continue
                    ma = min(step, 1.0 - h[prec], h[qdon])
                    if ma <= 1e-9:
                        continue
                    for fr in (1.0, 0.5):
                        a = ma * fr
                        if a <= 1e-10:
                            continue
                        temp[:] = c
                        apply_move_to_c(temp, h, g, g_rev, dx, prec, qdon, n, a)
                        val = float(temp.max())
                        if val < best_val - 1e-14:
                            best_val = val
                            best_pair = (prec, qdon)
                            best_a = a
                            break

        if best_pair is None:
            no_improve += 1
            step_idx += 1
            if no_improve > 60:
                break
            continue

        prec, qdon = best_pair
        a = best_a

        apply_move_to_c(c, h, g, g_rev, dx, prec, qdon, n, a)

        h[prec] += a
        h[qdon] -= a
        g[prec] -= a
        g[qdon] += a
        g_rev[n - 1 - prec] -= a
        g_rev[n - 1 - qdon] += a

        C = best_val
        moves += 1
        no_improve = 0

        if C < best_C - 1e-7:
            step_idx = max(0, step_idx - 1)

        if time.time() > deadline:
            break

    h = project_sum(h, n)
    c = compute_c5_vec(h, n)
    Cv = float(c.max())

    bh = project_sum(best_h, n)
    bc = compute_c5(bh, n)

    if Cv < bc:
        return h, Cv
    return bh, bc


def block_swap_search(h0, n, deadline, rng, max_trials=2000):
    h = project_sum(h0, n).copy()
    c = compute_c5_vec(h, n)
    C = float(c.max())
    best_h = h.copy()
    best_C = C

    blist = (2, 3, 4, 5, 6, 7, 8, 10, 13, 16, 20, 32)

    for _ in range(max_trials):
        if time.time() > deadline:
            break
        bl = int(blist[rng.randint(0, len(blist))])
        if 2 * bl >= n:
            continue
        a = int(rng.randint(0, n - 2 * bl))
        span = n - a - 2 * bl + 1
        b = a + bl + int(rng.randint(0, span))

        left = h[a:a + bl].copy()
        right = h[b:b + bl].copy()
        h[a:a + bl] = right
        h[b:b + bl] = left

        cv = compute_c5_vec(h, n)
        val = float(cv.max())

        if val < C - 1e-14:
            C = val
            best_h = h.copy()
            best_C = C
        else:
            h[a:a + bl] = left
            h[b:b + bl] = right

    h = project_sum(h, n)
    val = compute_c5(h, n)
    if val < best_C:
        return h, val
    bh = project_sum(best_h, n)
    bc = compute_c5(bh, n)
    return bh, bc


def smooth_top_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    e = alpha * (vals - m)
    smx = float(e.max())
    if smx > 400.0:
        sh = e - smx
        val = m + (smx + math.log(float(np.sum(np.exp(sh))))) / alpha
    else:
        val = m + math.log(float(np.sum(np.exp(e)))) / alpha
    ss = float(x.sum()) - 0.5 * float(n)
    val += mu * ss * ss
    return float(val)


def smooth_top_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    e = alpha * (vals - m)
    smx = float(e.max())
    if smx > 400.0:
        sh = e - smx
        w = np.exp(sh)
        w /= w.sum()
    else:
        w = np.exp(e)
        w /= w.sum()

    dx = 2.0 / float(n)
    off = n - 1
    grad = np.zeros(n, dtype=float)

    for ii in range(active.size):
        wi = float(w[ii])
        if wi < 1e-10:
            continue
        idx = int(active[ii])
        k = idx - off
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[k:] += wm * (1.0 - x[:nv])
                grad[:nv] -= wm * x[k:]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[kk:])
                grad[kk:] -= wm * x[:nv]

    ss = float(x.sum()) - 0.5 * float(n)
    if mu > 0.0:
        grad += (2.0 * mu * ss) * np.ones(n)
    return grad


def smooth_optimize(h0, n, deadline):
    h = project_sum(h0, n).copy()
    c = compute_c5_vec(h, n)
    C = float(c.max())
    best_h = h.copy()
    best_C = C

    bounds = [(0.0, 1.0)] * n
    L = 2 * n - 1
    configs = (
        (16, 50.0),
        (32, 500.0),
        (64, 5000.0),
        (96, 20000.0),
    )

    for asize, alpha in configs:
        if time.time() > deadline - 20:
            break
        asize = max(4, min(asize, L))
        c = compute_c5_vec(h, n)
        active = top_k_indices(c, asize)
        if active.size == 0:
            continue

        rem = max(20.0, deadline - time.time())
        maxit = int(min(300, max(70, rem * 0.18)))

        try:
            res = minimize(
                smooth_top_obj,
                h,
                jac=smooth_top_grad,
                args=(n, active, alpha, 1e-4),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-20,
                    "maxfun": max(maxit, 200) * 8,
                    "maxls": 20,
                    "maxcor": 20,
                }
            )
            if res.x is None:
                continue
            h_new = project_sum(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < C - 1e-15:
                h = h_new.copy()
                C = c_new
            elif c_new <= C + 1e-15:
                h = h_new.copy()
                C = c_new
        except Exception:
            pass

        if time.time() > deadline - 20:
            break

    h = project_sum(h, n)
    C = compute_c5(h, n)
    bh = project_sum(best_h, n)
    bc = compute_c5(bh, n)
    if C < bc:
        return h, C
    return bh, bc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    deadline = t0 + max(20.0, total_budget - 10.0)
    rng = np.random.RandomState(int(seed))

    try:
        base = np.asarray(initial_h_values, dtype=float).ravel().copy()
    except Exception:
        base = np.full(500, 0.5, dtype=float)

    if base.size < 50:
        base = np.full(500, 0.5, dtype=float)

    if base.size > 1600:
        base = downsample_linear(base, 1000)

    n = int(base.size)
    if n % 2 == 1:
        if n > 2:
            base = project_sum(base[:n - 1], n - 1)
            n = n - 1
        else:
            base = np.full(500, 0.5, dtype=float)
            n = 500

    base = project_sum(base, n)
    best_n = n

    cands = make_candidates(base, n, rng)
    if not cands:
        best_h = base.copy()
        best_C = compute_c5(best_h, best_n)
    else:
        best_h = cands[0][1].copy()
        best_C = cands[0][0]

    base_reserve = 350 if n <= 700 else 250
    base_timeout = min(deadline - base_reserve, t0 + (540 if n <= 700 else 380))
    if base_timeout < t0 + 30:
        base_timeout = max(t0 + 15, min(deadline, t0 + 30))

    phase_ends = []
    for offset in (min(300, base_timeout - t0), min(430, base_timeout - t0),
                   min(530, base_timeout - t0), int(base_timeout - t0)):
        pe = t0 + offset
        phase_ends.append(min(base_timeout, pe))

    ac_base = 40 if n <= 700 else 28
    mpg_base = 20 if n <= 700 else 14

    for i, cand in enumerate(cands[:4]):
        if i >= len(phase_ends):
            break
        phase_end = phase_ends[i]
        if time.time() > phase_end - 25:
            break
        if cand[0] > best_C + 0.02:
            break
        if cand[0] < best_C:
            best_C = cand[0]
            best_h = cand[1].copy()
        try:
            h, c = local_search(
                cand[1], n, phase_end, rng,
                active_count=ac_base,
                max_pair_grad=mpg_base,
                max_lag_pairs=8
            )
            if c < best_C:
                best_C = c
                best_h = h
                best_n = n
        except Exception:
            pass

    if time.time() < deadline - 300:
        end_block = min(deadline - 260, time.time() + 90)
        max_trials = 2500 if n <= 700 else 900
        try:
            h, c = block_swap_search(best_h, best_n, end_block, rng, max_trials=max_trials)
            if c < best_C:
                best_C = c
                best_h = h
                best_n = len(h)
        except Exception:
            pass

    if time.time() < deadline - 260:
        end_smooth = min(deadline - 220, time.time() + 120)
        try:
            h, c = smooth_optimize(best_h, best_n, end_smooth)
            if c < best_C:
                best_C = c
                best_h = h
                best_n = len(h)
        except Exception:
            pass

    fine_n = 2 * best_n
    if fine_n > 1500:
        fine_n = max(best_n + 100, 1000)
    if fine_n % 2 == 1:
        fine_n += 1

    if fine_n > best_n and time.time() < deadline - 220:
        h_fine = upsample_box(best_h, fine_n)
        c_fine = compute_c5(h_fine, fine_n)
        if c_fine < best_C:
            best_C = c_fine
            best_h = h_fine
            best_n = fine_n

        end_fine = deadline - 25
        ac_fine = min(32, max(18, fine_n // 30))
        try:
            h, c = local_search(
                best_h, fine_n, end_fine, rng,
                active_count=ac_fine,
                max_pair_grad=14,
                max_lag_pairs=6,
                fractions=(1.0, 0.5, 0.2)
            )
            if c < best_C:
                best_C = c
                best_h = h
                best_n = fine_n
        except Exception:
            pass

        if time.time() < deadline - 15:
            end_fine_block = deadline - 5
            max_trials = 700 if fine_n <= 1200 else 300
            try:
                h, c = block_swap_search(best_h, best_n, end_fine_block, rng,
                                         max_trials=max_trials)
                if c < best_C:
                    best_C = c
                    best_h = h
                    best_n = len(h)
            except Exception:
                pass

    if time.time() < deadline - 8:
        end_last = deadline - 2
        ac_last = min(24, max(12, best_n // 40))
        try:
            h, c = local_search(
                best_h, best_n, end_last, rng,
                active_count=ac_last,
                max_pair_grad=8,
                max_lag_pairs=4,
                fractions=(1.0, 0.5)
            )
            if c < best_C:
                best_C = c
                best_h = h
                best_n = len(h)
        except Exception:
            pass

    best_h = project_sum(best_h, best_n)
    best_C = compute_c5(best_h, best_n)

    return best_h, best_C, int(best_n)
