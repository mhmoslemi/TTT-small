from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def softmax_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    ncorr = len(c)
    grad = np.zeros(n)
    for mm in range(ncorr):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_obj(x, n, top_idx, alpha):

    vals = c[top_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)

    if smx > 500:

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def softmax_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def softmax_grad(x, n, alpha):

    m = c.max()

            continue

        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 10: Final top-k with highest alpha
    if time.time() < deadline - 20:
        t10 = min(time.time() + 20, deadline - 20)
        c_vec = compute_c5_vec(best_h, n_best)
        top_idx = np.argsort(-c_vec)[:3]
        h_new = lbfgs_topk(best_h, n_best, top_idx, 50000000, 500, t10)
        c_new = compute_c5(h_new, n_best)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final: also try at original resolution in case it was better
    n_orig_final = n_init
    x_fine = np.linspace(0, 2, n_best)
    x_orig = np.linspace(0, 2, n_orig_final)
    h_orig = np.interp(x_orig, x_fine, best_h)
    h_orig = project_feasible(h_orig, n_orig_final)
    c_orig = compute_c5(h_orig, n_orig_final)
    if c_orig < best_c:
        best_c = c_orig
        best_h = h_orig
        n_best = n_orig_final

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
