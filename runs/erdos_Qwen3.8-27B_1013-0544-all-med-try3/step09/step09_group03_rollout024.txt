- If j appears in the second factor (h[j+k] = h[j]): j+k ranges from 0 to n-1-k, so j ranges from k to n-1
  Contribution: dx * (-h[j-k])

Total: ∂c_k/∂h[j] = dx * [(1-h[j+k]) * 1{j ≤ n-1-k} - h[j-k] * 1{j ≥ k, j ≤ n-1}]

For k < 0, similar with -k.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_weights(c_vec, alpha):
    """Compute softmax weights for the log-sum-exp objective."""
    sh = alpha * c_vec
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c_vec)
        w[c_vec.argmax()] = 1.0
    else:
        w /= w_sum
    return w

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = compute_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(len(w)):
        if w[k_idx] < 1e-14:
            continue
        k = k_idx - center
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:k+nv])
                grad[k:k+nv] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def chase_worst_shift(h, n, deadline, n_steps=5000):
    """Gradient descent chasing the current worst shift."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.01
    center = n - 1
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k = worst_idx - center
        wm = dx
        
        # Compute gradient of the worst shift
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[0:nv] += (1.0 - h[k:k+nv])
                g[k:k+nv] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:kk+nv] += (1.0 - h[0:nv])
                g[0:nv] += (-h[kk:kk+nv])
        
        # Gradient step with line search
        best_c_here = compute_c5(h, n)
        tried = 0
        while step > 1e-8 and tried < 10:
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c_here - 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            step *= 0.5
            tried += 1
        else:

            step = max(step * 0.5, 1e-8)
            if step <= 1e-8:

                h = project_feasible(h + 0.01 * (np.random.rand(n) - 0.5), n)
                step = 0.01
            continue
        
        # Gradually reduce step
        if it % 100 == 0:

    return best_h, best_c

        alpha_list = [100, 500, 2000, 10000, 50000, 200000, 500000]

                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_rounds=200):
    """Coordinate descent focusing on worst-shift-affected coordinates."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    center = n - 1
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Get top 5 worst shifts
        top_idx = np.argsort(c_vec)[-5:][::-1]
        
        improved = False
        for idx in top_idx:
            if time.time() > deadline:
                return best_h, best_c
            k = idx - center
            # Find affected coordinates
            if k >= 0:
                affected = list(range(0, n - k)) + list(range(k, n))
            else:
                affected = list(range(-k, n)) + list(range(0, n + k))
            affected = list(set(affected))
            
            # Try small perturbations on affected coordinates
            for i in affected:
                for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                    nv = best_h[i] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    best_h[i] = nv
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] = best_h[i] - delta
                if improved:
                    break
        
        if not improved and rnd > 50:
            break
    
    return best_h, best_c

            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:kk+nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def chase_worst_shift(h, n, deadline, n_steps=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.02
    center = n - 1
    stall_count = 0
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k = worst_idx - center
        
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[0:nv] += (1.0 - h[k:k+nv])
                g[k:k+nv] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:kk+nv] += (1.0 - h[0:nv])
                g[0:nv] += (-h[kk:kk+nv])
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        g = g / g_norm * np.linalg.norm(g)
        
        improved = False
        for ls in range(12):
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stall_count = 0
                break
            step *= 0.6
        if not improved:
            stall_count += 1
            if stall_count > 50:
                h = project_feasible(h + 0.02 * (np.random.rand(n) - 0.5), n)
                step = 0.02
                stall_count = 0
            else:
                step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.1, 0.1)
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, alpha_list=None, mu=100.0):
    if alpha_list is None:
        alpha_list = [100, 500, 2000, 10000, 50000, 200000, 500000]
    
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_list:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def multi_start_lbfgs(h, n, deadline, n_starts=5):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(777)
    
    alphas = [1000, 5000, 20000]
    mags = [0.03, 0.06, 0.10, 0.02, 0.15]
    
    for s in range(n_starts):
        if time.time() > deadline - 20:
            break
        h_p = project_feasible(h + mags[s] * (rng.rand(n) - 0.5), n)
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            try:
                res = minimize(smooth_obj, h_p, jac=smooth_grad,
                              args=(n, alpha, 100.0), method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 15000, 'ftol': 1e-17,
                                       'gtol': 1e-13, 'maxfun': 50000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_rounds=300):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    center = n - 1
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        top_idx = np.argsort(c_vec)[-3:][::-1]
        
        improved = False
        for idx in top_idx:
            if time.time() > deadline:
                return best_h, best_c
            k = idx - center
            if k >= 0:
                affected = set(range(0, n - k)) | set(range(k, n))
            else:
                kk = -k
                affected = set(range(kk, n)) | set(range(0, n - kk))
            affected = list(affected)
            
            rng_local = np.random.RandomState(rnd * 31 + idx)
            rng_local.shuffle(affected)
            
            for i in affected:
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.003, -0.003, 0.0015, -0.0015, 0.0008, -0.0008]:
                    nv = best_h[i] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    old = best_h[i]
                    best_h[i] = nv
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] = old
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try larger perturbations
            if rnd % 10 == 0:
                for i in range(0, n, max(1, n // 100)):
                    if time.time() > deadline:
                        return best_h, best_c
                    for delta in [0.01, -0.01, 0.02, -0.02]:
                        nv = best_h[i] + delta
                        if nv < -1e-12 or nv > 1.0 + 1e-12:
                            continue
                        old = best_h[i]
                        best_h[i] = nv
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            best_h[i] = old
                    if improved:
                        break
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def adaptive_alpha_lbfgs(h, n, deadline, alpha_start=5000, n_phases=6):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alpha = alpha_start
    
    for phase in range(n_phases):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        sorted_c = np.sort(c_vec)[::-1]
        gap = sorted_c[0] - sorted_c[1] if len(sorted_c) > 1 else 0.1
        
        if gap < 0.005:
            alpha = max(1000, alpha * 0.5)
        elif gap < 0.02:
            alpha = min(1000000, alpha * 2)
        else:
            alpha = min(1000000, alpha * 4)
        
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha, 50.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def periodic_perturb(h, n, deadline):
    """Try structured perturbations: shift segments, flip blocks."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(999)
    
    n_attempts = min(50, max(10, int((deadline - time.time()) / 5)))
    for _ in range(n_attempts):
        if time.time() > deadline:
            break
        h_p = best_h.copy()
        # Random segment flip
        start = rng.randint(0, n - 20)
        length = rng.randint(10, min(50, n - start))
        h_p[start:start+length] = 1.0 - h_p[start:start+length]
        h_p = project_feasible(h_p, n)
        
        c = compute_c5(h_p, n)
        if c < best_c:
            best_c = c
            best_h = h_p.copy()
        
        # Try a small L-BFGS from this perturbation
        if time.time() < deadline - 10:
            try:
                res = minimize(smooth_obj, h_p, jac=smooth_grad,
                              args=(n, 5000, 100.0), method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 3000, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxfun': 10000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: L-BFGS with increasing alpha at base resolution
    t1 = min(t0 + 180, deadline - 700)
    if t1 > time.time() + 20:
        best_h, best_c = lbfgs_optimize(best_h, n_base, t1,
            alpha_list=[100, 500, 2000, 10000, 50000, 200000, 500000])
    
    # Phase 2: Chase worst shift at base
    t2 = min(t0 + 270, deadline - 580)
    if t2 > time.time() + 15:
        h_chase, c_chase = chase_worst_shift(best_h, n_base, t2, n_steps=15000)
        if c_chase < best_c:
            best_c = c_chase
            best_h = h_chase
    
    # Phase 3: Multi-start + periodic perturb at base
    t3 = min(t0 + 370, deadline - 470)
    if t3 > time.time() + 15:
        h_multi, c_multi = multi_start_lbfgs(best_h, n_base, t3, n_starts=4)
        if c_multi < best_c:
            best_c = c_multi
            best_h = h_multi
    
    # Phase 4: Periodic perturbations at base
    t4 = min(t0 + 440, deadline - 400)
    if t4 > time.time() + 10:
        h_per, c_per = periodic_perturb(best_h, n_base, t4)
        if c_per < best_c:
            best_c = c_per
            best_h = h_per
    
    # Phase 5: Upsample to high resolution and optimize
    n_high = 1500
    if time.time() < deadline - 250:
        h_high = interpolate_to(best_h, len(best_h), n_high)
        t5 = min(time.time() + 180, deadline - 120)
        if t5 > time.time() + 20:
            h_high, c_high = lbfgs_optimize(h_high, n_high, t5,
                alpha_list=[200, 1000, 5000, 20000, 100000, 400000, 1000000])
            if c_high < best_c:
                best_c = c_high
                best_h = h_high
        
        # Chase at high resolution
        t5b = min(time.time() + 80, deadline - 60)
        if t5b > time.time() + 15:
            h_ch2, c_ch2 = chase_worst_shift(best_h, len(best_h), t5b, n_steps=8000)
            if c_ch2 < best_c:
                best_c = c_ch2
                best_h = h_ch2
    
    # Phase 6: Adaptive alpha L-BFGS at current resolution
    t6 = min(time.time() + 120, deadline - 50)
    if t6 > time.time() + 15:
        h_adapt, c_adapt = adaptive_alpha_lbfgs(best_h, len(best_h), t6,
            alpha_start=20000, n_phases=5)
        if c_adapt < best_c:
            best_c = c_adapt
            best_h = h_adapt
    
    # Phase 7: Coordinate descent polish
    t7 = deadline - 10
    if t7 > time.time() + 10:
        best_h, best_c = coordinate_descent(best_h, len(best_h), t7, n_rounds=200)
    
    # Final feasibility check and verification
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c:
            best_c = float(eval_c5)
    except Exception:
        pass
    
    return best_h, best_c, n_final
