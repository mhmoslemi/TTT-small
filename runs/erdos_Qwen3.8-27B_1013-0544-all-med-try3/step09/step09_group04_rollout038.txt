def active_values(x, n, active):
    m = len(active)
    vals = np.empty(m, dtype=float)
    if m == 0:
       return vals
    pref = np.empty(n+1, dtype=float)
    pref[0]=0.0
    np.cumsum(x, out=pref[1:])
    dx=2.0/n
    center=n-1
    for ii in range(m):

    return vals

def project_fast(h):
    n=len(h)
    h=np.array(h, dtype=float, copy=True)
    np.clip(h,0.0,1.0,out=h)
    target=0.5*float(n)
    for _ in range(5):
        s=float(np.sum(h))
        diff=target-s
        if abs(diff) <= 1e-13*max(1.0,float(n)): break
        if diff > 0:
            room=1.0-h
            tr=float(np.sum(room))
            if tr <= 1e-15: break
            h += (diff/tr)*room
        else:
            if s <= 1e-15: break
            h += (diff/s)*h
        np.clip(h,0.0,1.0,out=h)
    # final small correction
    diff=target-float(np.sum(h))
    if diff > 1e-15:
       idx=np.argmax(1.0-h)
       if 1.0-h[idx] >= diff:
          h[idx] += diff
       else: # spread
          # distribute among max rooms
          idxs=np.argsort(1.0-h)[::-1]
          for idx in idxs:
             add=min(diff, max(0.0,1.0-h[idx])); h[idx]+=add; diff-=add; if diff<=1e-15: break
    elif diff < -1e-15:
       similar
    np.clip(h,0,1,out=h)
    return h

    n=len(h); target=0.5*n

            x0,
            jac=dfunc,
            args=args,
            method="L-BFGS-B",
            bounds=bounds,
            options=opts,
            callback=_timeout_callback,
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            return np.array(res.x, dtype=float, copy=True)
        return np.array(_TIMEOUT_X, dtype=float, copy=True)
    except _BoundedTimeout:
        return np.array(_TIMEOUT_X, dtype=float, copy=True)
    except Exception:
        return np.array(_TIMEOUT_X, dtype=float, copy=True)


def _sanitize(h):
    h = np.asarray(h, dtype=float)
    return np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)


def project_fast(h):
    h = _sanitize(h).copy()
    n = len(h)
    if n == 0:
        return h

    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)

    for _ in range(4):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-13 * max(1.0, float(n)):
            break

        if diff > 0.0:
            room = 1.0 - h
            tr = float(np.sum(room))
            if tr <= 1e-15:
                break
            h += (diff / tr) * room
        else:
            if s <= 1e-15:
                break
            h += (diff / s) * h

        np.clip(h, 0.0, 1.0, out=h)

    diff = target - float(np.sum(h))
    if diff > 1e-15:
        order = np.argsort(1.0 - h)[::-1]
        for idx in order:
            if diff <= 1e-15:
                break
            add = min(diff, max(0.0, 1.0 - h[idx]))
            h[idx] += add
            diff -= add
    elif diff < -1e-15:
        order = np.argsort(h)[::-1]
        for idx in order:
            if diff >= -1e-15:
                break
            sub = min(-diff, max(0.0, h[idx]))
            h[idx] -= sub
            diff += sub

    np.clip(h, 0.0, 1.0, out=h)
    return h


def project_euclid(h):
    h = _sanitize(h).copy()
    n = len(h)
    if n == 0:
        return h

    target = 0.5 * float(n)
    lo = min(0.0, float(np.min(h))) - 1.0
    hi = max(1.0, float(np.max(h))) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return project_fast(h)


def fix_sum_small(h):
    h = _sanitize(h).copy()
    n = len(h)
    if n == 0:
        return h

    target = 0.5 * float(n)
    for _ in range(4):
        diff = target - float(np.sum(h))
        if diff > 1e-15:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 1e-14:
                    break
                add = min(diff, max(0.0, 1.0 - h[idx]))
                h[idx] += add
                diff -= add
        elif diff < -1e-15:
            order = np.argsort(h)[::-1]
            for idx in order:
                if diff >= -1e-14:
                    break
                sub = min(-diff, max(0.0, h[idx]))
                h[idx] -= sub
                diff += sub

    np.clip(h, 0.0, 1.0, out=h)
    return h


def c5_full_vec(h):
    h = _sanitize(h)
    n = len(h)
    if n == 0:
        return np.array([1e300], dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_max(h):
    h = _sanitize(h)
    if len(h) == 0:
        return 1e300
    v = c5_full_vec(h)
    c = float(np.max(v))
    if not np.isfinite(c):
        return 1e300
    return c


def evaluate_candidate(h):
    h = project_fast(h)
    c = c5_max(h)
    return h, c


def active_values(x, n, active):
    m = len(active)
    vals = np.empty(m, dtype=float)
    if m == 0:
        return vals

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(x, out=pref[1:])

    dx = 2.0 / float(n)
    center = n - 1

    for ii in range(m):
        idx = int(active[ii])
        k = idx - center
        if k >= 0:
            cnt = n - k
            s = pref[cnt]
            vals[ii] = dx * (s - float(np.dot(x[:cnt], x[k:k + cnt])))
        else:
            kk = -k
            cnt = n - kk
            s = pref[n] - pref[kk]
            vals[ii] = dx * (s - float(np.dot(x[kk:kk + cnt], x[:cnt])))
    return vals


def active_obj(x, n, active, alpha, mu):
    x = _sanitize(x)
    vals = active_values(x, n, active)
    if vals.size == 0:
        return 1e300

    m = float(np.max(vals))
    sh = alpha * (vals - m)
    logsum = float(np.log(np.sum(np.exp(sh))))
    val = m + logsum / alpha

    if mu != 0.0:
        s = float(np.sum(x))
        t = s - 0.5 * float(n)
        val += mu * t * t

    if not np.isfinite(val):
        return 1e300
    return val


def active_grad(x, n, active, alpha, mu):
    x = _sanitize(x)
    vals = active_values(x, n, active)
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    w = np.exp(sh)
    ws = float(np.sum(w))
    if ws <= 1e-300:
        w = np.zeros_like(vals)
        w[np.argmax(vals)] = 1.0
    else:
        w = w / ws

    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    center = n - 1

    for ii in range(len(active)):
        wi = float(w[ii])
        if wi <= 1e-15:
            continue

        idx = int(active[ii])
        k = idx - center
        wm = wi * dx

        if k >= 0:
            cnt = n - k
            grad[:cnt] += wm
            grad[:cnt] -= wm * x[k:k + cnt]
            grad[k:k + cnt] -= wm * x[:cnt]
        else:
            kk = -k
            cnt = n - kk
            grad[kk:] += wm
            grad[kk:] -= wm * x[:cnt]
            grad[:cnt] -= wm * x[kk:kk + cnt]

    if mu != 0.0:
        s = float(np.sum(x))
        grad += 2.0 * mu * (s - 0.5 * float(n))

    return grad


def defaults_for_n(n):
    if n <= 700:
        return 30, 90
    if n <= 1600:
        return 28, 75
    return 24, 55


def alphas_for_n(n):
    if n <= 900:
        return (5000.0, 50000.0, 500000.0, 5000000.0)
    if n <= 2600:
        return (10000.0, 100000.0, 1000000.0, 10000000.0)
    return (10000.0, 150000.0, 2000000.0, 15000000.0)


def select_active(c, top_k, max_active):
    length = len(c)
    if length == 0:
        return np.array([], dtype=int)

    cmax = float(np.max(c))
    tol = max(1e-8, 4e-5)

    near = np.where(c >= cmax - tol)[0]
    top = np.argsort(-c)[:min(top_k, length)]
    act = np.unique(np.concatenate((near, top))).astype(int)

    adds = []
    rad = 3
    for idx in act:
        lo = max(0, int(idx) - rad)
        hi = min(length, int(idx) + rad + 1)
        adds.extend(range(lo, hi))

    if adds:
        act = np.unique(np.concatenate((act, np.array(adds, dtype=int))))

    if len(act) > max_active:
        order = np.argsort(-c[act])
        act = act[order[:max_active]]

    if len(act) == 0:
        act = np.array([int(np.argmax(c))], dtype=int)
    return act.astype(int, copy=False)


def active_optimize(h, n, end, top_k=None, max_active=None, alphas=None, mu=0.03):
    if n <= 0:
        return h, 1e300
    if time.time() >= end - 25.0:
        h = project_fast(h)
        return h, c5_max(h)

    if top_k is None or max_active is None:
        top_k, max_active = defaults_for_n(n)
    if alphas is None:
        alphas = alphas_for_n(n)

    work = project_fast(h)
    best = work.copy()
    best_c = c5_max(work)

    bounds = [(0.0, 1.0)] * n
    no_improve = 0
    round_idx = 0

    while time.time() < end - 28.0:
        round_idx += 1

        cv = c5_full_vec(work)
        cmax = float(np.max(cv))
        if cmax < best_c - 1e-13:
            best_c = cmax
            best = work.copy()

        active = select_active(cv, top_k, max_active)
        improved_round = False

        for idx, a in enumerate(alphas):
            remaining_alphas = len(alphas) - idx
            remain_end = end - 12.0 - time.time()
            if remain_end <= 0.0:
                break

            seg = remain_end / float(remaining_alphas)
            if seg < 7.0:
                break

            alpha_end = min(end - 6.0, time.time() + seg)

            xout = _bounded_minimize(
                active_obj,
                active_grad,
                work,
                (n, active, float(a), float(mu)),
                bounds,
                alpha_end,
            )

            cand = project_fast(xout)
            cc = c5_max(cand)

            if cc < best_c - 1e-13:
                best_c = cc
                best = cand.copy()
                improved_round = True

            if cc <= best_c + 0.0018:
                work = cand.copy()
            else:
                work = best.copy()

        if improved_round:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

        if round_idx >= 60:
            break

    best = project_fast(best)
    best_c = c5_max(best)
    return best, best_c


def resample_to(h, n_new):
    h = _sanitize(h)
    n_old = len(h)
    if n_new == n_old:
        return project_fast(h)

    if n_new > n_old and n_new % n_old == 0:
        return project_fast(np.repeat(h, n_new // n_old))

    if n_new < n_old and n_old % n_new == 0:
        g = n_old // n_new
        return project_fast(h.reshape(n_new, g).mean(axis=1))

    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h)
    return project_fast(y)


def low_freq_noise(n, rng, k, base_amp):
    if n <= 0:
        return np.zeros(0, dtype=float)
    x = np.arange(n, dtype=float)
    noise = np.zeros(n, dtype=float)
    k = min(max(1, int(k)), max(1, n // 2))
    for f in range(1, k + 1):
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = base_amp / np.sqrt(float(f))
        noise += amp * np.sin(2.0 * np.pi * f * x / float(n) + phase)
    return noise


def upsample_noisy(h, n_new, rng, noise_amp):
    y = resample_to(h, n_new)
    if noise_amp > 0.0:
        y = y + noise_amp * low_freq_noise(n_new, rng, 24, 1.0)
        y = y + 0.05 * noise_amp * np.array(rng.standard_normal(n_new), dtype=float)
    return project_fast(y)


def perturb_candidate(h, rng, amp):
    n = len(h)
    noise = low_freq_noise(n, rng, 18, amp)
    noise = noise + 0.25 * amp * np.array(rng.standard_normal(n), dtype=float)
    return project_fast(h + noise)


def de_search(size, end, rng, parent):
    P = 42
    if size < 24:
        size = 24

    starts = []
    if parent is not None and len(parent) > 0:
        starts.append(resample_to(parent, size))

    x = (np.arange(size) + 0.5) * (2.0 / float(size))
    starts.append(np.full(size, 0.5, dtype=float))
    starts.append(1.0 - np.abs(x - 1.0))
    starts.append(x / 2.0)
    starts.append(1.0 - x / 2.0)

    for freq in (1, 2, 3, 5, 7):
        starts.append(0.5 + 0.32 * np.sin(np.pi * freq * x + 0.61))
        starts.append(0.5 + 0.32 * np.cos(np.pi * freq * x + 0.37))

    while len(starts) < P:
        if time.time() > end:
            break
        noise = low_freq_noise(size, rng, 10, 0.55)
        starts.append(0.5 + noise)

    pop = []
    fits = []
    for v in starts:
        if time.time() > end - 2.0:
            break
        v = project_fast(v)
        pop.append(v)
        fits.append(c5_max(v))

    P = len(pop)
    if P < 3:
        v = project_fast(0.5 + low_freq_noise(size, rng, 8, 0.5))
        for _ in range(3 - P):
            pop.append(v.copy())
            fits.append(c5_max(v))
        P = len(pop)

    best_v = pop[int(np.argmin(fits))].copy()
    best_f = float(min(fits))
    stagnant = 0
    gen = 0

    while time.time() < end - 4.0:
        gen += 1
        if gen > 20000:
            break

        for i in range(P):
            if time.time() > end - 3.0:
                break

            a = int(rng.integers(P))
            b = int(rng.integers(P))
            while b == a:
                b = int(rng.integers(P))
            c = int(rng.integers(P))
            while c == a or c == b:
                c = int(rng.integers(P))

            F = 0.65 + 0.25 * (0.5 + 0.5 * np.sin(0.021 * gen + 0.37 * i))
            mut = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(size) < 0.82
            y = np.where(mask, mut, pop[i])
            y = project_fast(y)
            fy = c5_max(y)

            if fy < fits[i] - 1e-14:
                pop[i] = y
                fits[i] = fy

            if fy < best_f - 1e-13:
                best_v = y.copy()
                best_f = fy
                stagnant = 0
            else:
                stagnant += 1

        if stagnant > 2500:
            y = project_fast(best_v + 0.08 * low_freq_noise(size, rng, 7, 1.0))
            fy = c5_max(y)
            if fy < best_f - 1e-13:
                best_v = y.copy()
                best_f = fy
                stagnant = 0
            else:
                break

    return best_v, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if budget <= 15.0:
        budget = max(20.0, float(budget_s))
    G_deadline = t0 + budget - 6.0
    if G_deadline < t0 + 10.0:
        G_deadline = t0 + max(10.0, budget - 2.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        parent = None

    if parent is None or len(parent) == 0:
        n0 = 500
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        h0 = project_euclid(1.0 - np.abs(x - 1.0))
    else:
        h0 = project_fast(parent.copy())
        if abs(float(np.sum(h0)) - 0.5 * float(len(h0))) > 1e-7 * max(1, len(h0)):
            h0 = project_euclid(h0)

    best_h = h0.copy()
    best_n = len(best_h)
    best_c = c5_max(best_h)

    tr1 = project_fast(h0[::-1])
    c1 = c5_max(tr1)
    if c1 < best_c:
        best_c = c1
        best_h = tr1

    tr2 = project_fast(1.0 - h0)
    c2 = c5_max(tr2)
    if c2 < best_c:
        best_c = c2
        best_h = tr2

    tr3 = project_fast(0.5 * (h0 + h0[::-1]))
    c3 = c5_max(tr3)
    if c3 < best_c:
        best_c = c3
        best_h = tr3

    tr4 = project_fast(1.0 - 0.5 * (h0 + h0[::-1]))
    c4 = c5_max(tr4)
    if c4 < best_c:
        best_c = c4
        best_h = tr4

    rem = G_deadline - time.time()
    if rem > 260.0:
        end0 = min(G_deadline - 180.0, time.time() + max(25.0, min(75.0, rem * 0.10)))
        htry, ctry = active_optimize(best_h, best_n, end0)
        if ctry < best_c:
            best_c = ctry
            best_h = htry

    orig_len = len(h0)
    ds_source = best_h.copy()
    for f in (2, 3, 4, 5, 8, 10, 16, 20, 25, 50, 100):
        if orig_len % f != 0:
            continue
        nd = orig_len // f
        if nd < 20:
            continue
        if time.time() > G_deadline - 120.0:
            break
        hd = resample_to(ds_source, nd)
        cd = c5_max(hd)
        if cd < best_c:
            best_c = cd
            best_h = hd
            best_n = nd
            if G_deadline - time.time() > 180.0:
                endd = min(G_deadline - 80.0, time.time() + max(12.0, min(35.0, (G_deadline - time.time()) * 0.10)))
                hd2, cd2 = active_optimize(hd, nd, endd)
                if cd2 < best_c:
                    best_c = cd2
                    best_h = hd2
                    best_n = nd

    rem = G_deadline - time.time()
    if rem > 430.0:
        de_size = 100
        de_end = min(G_deadline - 300.0, time.time() + max(30.0, min(80.0, rem * 0.12)))
        hde, cde = de_search(de_size, de_end, rng, best_h)

        if cde < 1.0:
            for target in (500, 1000, 2000):
                if target <= best_n:
                    continue
                if time.time() > G_deadline - 220.0:
                    break
                hd = upsample_noisy(hde, target, rng, 0.002)
                cd = c5_max(hd)
                if cd <= best_c + 0.008:
                    endd = min(G_deadline - 130.0, time.time() + max(25.0, min(120.0, (G_deadline - time.time()) * 0.18)))
                    hd2, cd2 = active_optimize(hd, target, endd)
                    if cd2 < best_c:
                        best_c = cd2
                        best_h = hd2
                        best_n = hd2.shape[0]
                    elif cd2 <= best_c + 0.003 and (target == 2000 or target == 1000):
                        pass

    high_candidate = None
    high_c = 1e300

    for target in (1000, 2000, 4000):
        if best_n >= target and (high_candidate is None or len(high_candidate) >= target):
            continue

        rem = G_deadline - time.time()
        if target == 1000 and rem < 380.0:
            break
        if target == 2000 and rem < 360.0:
            break
        if target == 4000 and rem < 320.0:
            break

        if high_candidate is not None and high_c <= best_c + 0.0035 and len(high_candidate) < target:
            source = high_candidate
            source_c = high_c
        else:
            source = best_h
            source_c = best_c

        if len(source) >= target:
            continue

        htry = upsample_noisy(source, target, rng, 0.0012)
        ctry = c5_max(htry)

        if ctry > max(best_c, source_c) + 0.014:
            continue

        if target == 1000:
            dur = min(230.0, max(70.0, (G_deadline - time.time()) * 0.32))
        elif target == 2000:
            dur = min(380.0, max(110.0, (G_deadline - time.time()) * 0.42))
        else:
            dur = min(260.0, max(80.0, (G_deadline - time.time()) * 0.40))

        endh = min(G_deadline - 60.0, time.time() + dur)
        htry, ctry = active_optimize(htry, target, endh)

        if ctry < best_c:
            best_c = ctry
            best_h = htry
            best_n = target
            high_candidate = htry
            high_c = ctry
        else:
            if ctry <= best_c + 0.0045 and ctry < high_c:
                high_candidate = htry
                high_c = ctry

    rem = G_deadline - time.time()
    if rem > 170.0:
        amps = (0.08, 0.045, 0.025, 0.012)
        for amp in amps:
            if time.time() > G_deadline - 70.0:
                break
            start = best_h.copy()
            cand = perturb_candidate(start, rng, amp)
            cc = c5_max(cand)
            if cc <= best_c + 0.004:
                dur = min(55.0, max(12.0, (G_deadline - time.time()) * 0.12))
                endp = min(G_deadline - 45.0, time.time() + dur)
                htry, ctry = active_optimize(cand, len(cand), endp)
                if ctry < best_c:
                    best_c = ctry
                    best_h = htry
                    best_n = len(htry)

    rem = G_deadline - time.time()
    if rem > 45.0:
        endf = G_deadline - 4.0
        final_alphas = (100000.0, 1000000.0, 10000000.0)
        htry, ctry = active_optimize(
            best_h,
            best_n,
            endf,
            top_k=22,
            max_active=45,
            alphas=final_alphas,
            mu=0.02,
        )
        if ctry < best_c:
            best_c = ctry
            best_h = htry
            best_n = len(htry)

    h_a = fix_sum_small(best_h)
    c_a = c5_max(h_a)

    h_b = project_euclid(best_h)
    c_b = c5_max(h_b)

    if c_a <= c_b:
        final_h = h_a
        final_c = c_a
    else:
        final_h = h_b
        final_c = c_b

    final_h = fix_sum_small(final_h)
    final_c = c5_max(final_h)

    return final_h, float(final_c), int(len(final_h))
