def correlate_fft(h, n):
    # Compute full correlation of h with (1-h)
    # Using FFT: correlate(a, b, 'full') = ifft(fft(a, 2n) * fft(b_flipped, 2n))[:2n-1]
    # Actually: np.correlate(a, b, 'full')[k] = sum_i a[i+k] * b[i] for valid i
    # This equals the convolution of a with reversed b.
    # conv(a, b_rev)[k] = sum_i a[i] * b_rev[k-i] = sum_i a[i] * b[len(b)-1-(k-i)]
    # For same length n: = sum_i a[i] * b[n-1-k+i]
    # correlate(a, b)[k] = sum_i a[i+k-(n-1)] * b[i] = sum_i a[i] * b[i-(k-(n-1))]
    # Let j = i-(k-(n-1)), i = j+(k-(n-1))
    # = sum_j a[j+k-(n-1)] * b[j]
    # conv(a, b_rev)[k'] = sum_j a[j] * b[n-1-k'+j]
    # Setting k' = k-(n-1)+... hmm this is confusing.

    # fft_correlate = np.fft.ifft(np.fft.fft(h, 2*n) * np.fft.fft((1-h)[::-1], 2*n)).real[:2*n-1]
    
    size = 2 * n

    f1h = np.fft.rfft((1-h)[::-1], size)
    result = np.fft.irfft(fh * f1h, size)[:2*n-1]

        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(0, n - m):
                if time.time() > deadline:
                    break
                j = i + m if lag > 0 else i + m
                if lag > 0:
                    j = i + m
                else:
                    j = i
                    i_off = i + m
                    i, j = i_off, i
                if h[i] > h[j] + 1e-10:
                    delta = min(d, (h[i] - h[j]) * 0.5)
                    if delta < 1e-12:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-10:
                    delta = min(d, (h[j] - h[i]) * 0.5)
                    if delta < 1e-12:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
        if not improved:
            break
    return best_h, best_c

def restricted_search(h, n, deadline):
    """Push values toward {0, 0.25, 0.5, 0.75, 1} and check improvement."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    levels = [0.0, 0.25, 0.5, 0.75, 1.0]
    for _ in range(3):
        if time.time() > deadline:
            break
        h_new = np.array([min(levels, key=lambda l: abs(l - h[i])) for i in range(n)], dtype=float)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_new = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_new, n_dst)

def optimize_at_resolution(h_init, n, deadline, rng):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Gradient descent with smooth max
    gd_time = (deadline - time.time()) * 0.5
    h_new, c_new = grad_descent(h, n, time.time() + gd_time,
                                 alphas=[100, 500, 2000, 10000, 50000],
                                 base_lr=0.3, momentum=0.93)
    if c_new < best_c:
        best_c = c_new
        best_h = h_new
    h = h_new

    # Phase 2: SLSQP on top lags
    if time.time() < deadline - 20:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_top(h, n, top_m, 50000, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: Full SLSQP
    if time.time() < deadline - 15:
        h_new = slsqp_full(h, n, 200000, maxiter=2000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: Binary local search
    if time.time() < deadline - 10:
        ls_dl = min(deadline - 2, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = binary_local_search(best_h, n, ls_dl, max_rounds=80)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Restricted value search
    if time.time() < deadline - 5:
        h_new, c_new = restricted_search(best_h, n, time.time() + 10)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    # Phase A: Polish at initial resolution
    if h_init is not None:
        alloc = min(150, (deadline - t0) * 0.15)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(n_init, 0.5)
        # Add structured perturbation
        phase = 2 * np.pi * rng.randint(3, 8) / n_init * np.arange(n_init)
        h_start = 0.5 + 0.3 * np.sin(phase) + rng.normal(0, 0.05, n_init)
        h_start = project_feasible(h_start, n_init)
        alloc = min(100, (deadline - t0) * 0.1)
        h_opt, c_opt = optimize_at_resolution(h_start, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase B: Try higher resolutions
    n_targets = [700, 1000, 1400]
    for n_try in n_targets:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 30
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.2)
        if alloc < 50:
            break
        h_interp = interpolate_to(best_h, len(best_h), n_try)
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase C: Multi-start at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 80:
        n_starts = 3
        for r in range(n_starts):
            if time.time() > deadline - 40:
                break
            if time.time() - t0 > budget_s * 0.7:
                break
            scale = 0.03 * (1.0 + 0.5 * r)
            if r == 0:
                hp = best_h + rng.normal(0, scale, n_best)
            elif r == 1:
                # Shift perturbation
                shift = rng.randint(-10, 10)
                hp = np.roll(best_h, shift)
                hp = hp * 0.7 + 0.3 * rng.uniform(0, 1, n_best)
            else:
                # Block flip
                bp = n_best // 4
                hp = best_h.copy()
                hp[:bp] = 1.0 - hp[:bp]
                hp[-bp:] = 1.0 - hp[-bp:]
            hp = project_feasible(hp, n_best)
            alloc = min(60, (deadline - time.time() - 20) * 0.25)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase D: Final intensive polish
    n_f = len(best_h)
    if time.time() < deadline - 20:
        # Very high alpha SLSQP
        if time.time() < deadline - 15:
            h_new = slsqp_top(best_h, n_f, 3, 200000, maxiter=5000, ftol=1e-17)
            c_new = compute_c5(h_new, n_f)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
            h_new = slsqp_full(best_h, n_f, 500000, maxiter=3000, ftol=1e-17)
            c_new = compute_c5(h_new, n_f)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        # Final binary local search with very fine steps
        if time.time() < deadline - 5:
            best_h, best_c = binary_local_search(best_h, n_f, deadline - 2, max_rounds=200)

        # Final restricted search
        if time.time() < deadline - 3:
            best_h, best_c = restricted_search(best_h, n_f, deadline - 1)

    best_h = project_feasible(best_h, n_f)
    best_c = compute_c5(best_h, n_f)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
