import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def gradient_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        shift = k - (n - 1)
        m = abs(shift)
        if m >= n:
            continue
        if shift == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[m:n] += w[k] * (1.0 - h[0:n-m])
            grad[0:n-m] += w[k] * (-h[m:n])
        else:
            grad[0:n-m] += w[k] * (1.0 - h[m:n])
            grad[m:n] += w[k] * (-h[0:n-m])
    return grad * dx

def binary_nudge(h, n, t):
    h_nudged = h + t * (2.0 * h - 1.0)
    return project_feasible(h_nudged, n)

def adaptive_gradient_descent(h, n, deadline, alpha_init=200, lr_init=1.5):

    best_h = h_curr.copy()

    v = np.zeros(n)

    alpha = alpha_init
    momentum = 0.92
    patience = 0
    max_patience = 300
    step_count = 0

    while time.time() < deadline:
        alpha = min(alpha * 1.01, 50000)
        g = gradient_smooth_max(h_curr, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        h_new = h_curr - v

def slsqp_active(h, n, alpha, maxiter, top_m):
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_m:]

    def obj_func(x):
        dx_val = 2.0 / n
        cc = np.correlate(x, 1 - x, mode='full') * dx_val
        active = cc[top_idx]
        m_val = active.max()
        shifted = alpha * (active - m_val)
        mx = shifted.max()
        if mx > 500:
            return float(m_val + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m_val + np.log(np.sum(np.exp(shifted))) / alpha)

    def con_func(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def simulated_annealing(h, n, deadline, rng, T_init=0.08, T_min=0.003):

    best_h = h_curr.copy()

    T = T_init

        shift = k_star - (n - 1)

            j = rng.randint(n)
            if i == j:
                continue
            delta = (rng.random() - 0.5) * 0.04
        else:

            if m >= n:
                continue
            if shift > 0:
                i = rng.randint(n - m)
                j = i + m
            else:
                j = rng.randint(n - m)
                i = j + m
            if h_curr[i] > h_curr[j]:
                delta = -min(rng.random() * T, (h_curr[i] - h_curr[j]) * 0.4)
            elif h_curr[j] > h_curr[i]:
                delta = min(rng.random() * T, (h_curr[j] - h_curr[i]) * 0.4)
            else:
                continue

        if abs(delta) < 1e-10:

        h_new = h_curr.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
