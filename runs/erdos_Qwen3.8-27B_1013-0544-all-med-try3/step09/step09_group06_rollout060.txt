Let me also think about the local search more carefully.

**Vectorized local search**:

At the worst lag m, the objective is:
c(m) = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

To decrease c(m), we want to decrease the product h[i]*(1-h[i+m]) for the terms that contribute most. The derivative with respect to h[i] (for i < n-m) is (1-h[i+m]), and with respect to h[i+m] is -h[i].

So the "most impactful" moves are:
- Decrease h[i] where h[i] is high and h[i+m] is low (large (1-h[i+m]))
- Increase h[i+m] where h[i] is high and h[i+m] is low (large h[i])

Both of these are the same move: transfer mass from i to i+m.

For a batch move: select the top-B pairs (i, i+m) where h[i] - h[i+m] is largest, and transfer a small amount from each.

This is a much more efficient local search than trying one pair at a time.

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    # Only process lags with significant weight
    active = w > 1e-8
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        lag = int(k) - (n - 1)
        wk = w[k]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n-lag] += wk * (1.0 - h[lag:n])
            grad[lag:n] -= wk * h[0:n-lag]
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] -= wk * h[m:n]
    return grad * dx

    best_h = h_curr.copy()

    v = np.zeros(n)

    log_a1 = np.log(alpha_end)

        alpha = np.exp(log_a0 + frac * (log_a1 - log_a0))

        h_new = project_feasible(h_curr - v, n)

Actually, let me think about another approach: **direct optimization of the max using a "water-filling" insight**.

If we could make all c_k equal, that would be ideal (by the min-max principle). The question is: can we? And how?

At the optimum, the KKT conditions say:
- For lags k that are "active" (c_k = t): the gradient of c_k is in the span of the gradients of the constraints
- For lags k that are "inactive" (c_k < t): no condition

The constraint gradients are:
- sum(h) = n/2: gradient is (1, 1, ..., 1)

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    ssum = w.sum()
    if ssum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(s)] = 1.0
        return w
    w /= ssum
    return w

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    active = np.where(w > 1e-8)[0]
    if len(active) == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        wk = w[k]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n-lag] += wk * (1.0 - h[lag:n])
            grad[lag:n] -= wk * h[0:n-lag]
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n-m])
            grad[0:n-m] -= wk * h[m:n]
    return grad * dx

def gradient_descent(h, n, deadline, alpha_start=50.0, alpha_end=50000.0, lr_start=1.2):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    v = np.zeros(n)
    lr = lr_start
    mu = 0.92
    max_iter = 300000
    patience = 0
    best_patience = 0
    
    log_a0 = np.log(alpha_start)
    log_a1 = np.log(alpha_end)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(float(it) / max_iter, 1.0)
        alpha = np.exp(log_a0 + frac * (log_a1 - log_a0))
        
        g = grad_smooth_max(h_curr, n, alpha)
        v = mu * v + lr * g
        h_new = project_feasible(h_curr - v, n)

def thue_morse_init(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        c = 0
        v = i
        while v > 0:
            c += v & 1
            v >>= 1
        tm[i] = c & 1
    h = tm.astype(float)
    # Scale to have sum = n/2
    s = h.sum()
    if s > 0:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
