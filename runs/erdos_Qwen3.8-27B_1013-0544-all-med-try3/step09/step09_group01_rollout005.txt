- Captures the same "bottleneck" in the max

def compute_c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smax(x, n, alpha, mu, active=None):
    c = compute_c5_vec(x, n)
