    target = n / 2.0

        if abs(diff) < 1e-15:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def power_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    val = (np.sum(c**p))**(1.0/p)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return float(val)

def power_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    cp = np.sum(c**p)
    # d/dx (sum c_i^p)^(1/p) = (1/p) * (sum c_i^p)^(1/p - 1) * p * sum c_i^(p-1) * dc_i/dx
    # = (sum c_i^p)^(1/p - 1) * sum c_i^(p-1) * dc_i/dx
    base = cp**(1.0/p - 1.0)
    dx = 2.0/n
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if c[idx] < 1e-15:
            continue
        k = idx - (n-1)
        coeff = base * (c[idx]**(p-1)) * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += coeff * (1.0 - x[k:n])
            grad[k:n] += coeff * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += coeff * (1.0 - x[0:nv])
            grad[0:nv] += coeff * (-x[kk:n])

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
