    return float(c.max())

    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        diff = target - s
        if abs(diff) < 1e-15:

            if tot < 1e-15:
                break
            h += (diff / tot) * h

    """Beatty sequence: positions where floor(k*alpha) differs from floor((k-1)*alpha)."""
    h = np.zeros(n)
    # Generate Beatty sequence positions
    positions = set()
    for k in range(1, 2 * n + 100):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n // 2:
                break
    for p in positions:
        h[p] = 1.0
    return h

def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_swap_search(h, n, deadline, max_sweeps=100):
    """Local search via pair swaps on binary solution."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(best_h > 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                if time.time() > deadline:
                    break
                # Try swap
                best_h[i] = 0.0
                best_h[j] = 1.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    # Continue with new assignments for this i
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
        if not improved:
            break
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):

        sdiff = target - s
        if abs(sdiff) < 1e-14:
            break
        if sdiff > 0:
            room = 1.0 - h

            if tot < 1e-14:

            h += (sdiff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (sdiff / tot) * h

def make_beatty(n, alpha, beta=0.0):

def compute_c5_vec(h, n):

def project_feasible(h, n):

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):

        sdiff = target - s
        if abs(sdiff) < 1e-15:

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (sdiff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (sdiff / tot) * h

def make_beatty(n, alpha, beta=0.0):

    zeros_list = np.where(h < 0.5)[0].tolist()

        h[j] = 1.0

def smooth_max_obj(x, n, alpha, active_lags):
    c = compute_c5_vec(x, n)
    vals = c[active_lags]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha, active_lags):
    c = compute_c5_vec(x, n)
    vals = c[active_lags]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_lags):
        m = int(idx)
        k = m - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

                                      T0=0.005, T1=0.00005, alpha_cool=0.998)
        if c_ann < best_c:
            best_c = c_ann
            best_h = h_ann.copy()
            best_binary = h_ann.copy()
        # Also anneal a random start
        rng3 = np.random.RandomState(seed + 2)
        h_rand = make_binary_random(n, rng3)
        h_ann2, c_ann2 = anneal_binary(h_rand, n, p3_deadline, rng3,
                                        T0=0.003, T1=0.0001, alpha_cool=0.997)
        if c_ann2 < best_c:
            best_c = c_ann2
            best_h = h_ann2.copy()
            best_binary = h_ann2.copy()
    
    # Phase 4: More targeted binary search (150s)
    p4_deadline = min(time.time() + 150, deadline - 250)
    if time.time() < p4_deadline:
        rng4 = np.random.RandomState(seed + 10)
        h_bin2, c_bin2 = targeted_binary_search(best_binary, n, p4_deadline, rng4)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
            best_binary = h_bin2.copy()
    
    # Phase 5: Continuous optimization (250s)
    p5_deadline = min(time.time() + 250, deadline - 150)
    if time.time() < p5_deadline:
        h_cont, c_cont = continuous_optimize(best_h, n, p5_deadline, max_rounds=30)
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
    
    # Phase 6: Second round of binary search on continuous solution's binary part (100s)
    p6_deadline = min(time.time() + 100, deadline - 80)
    if time.time() < p6_deadline:
        # Binarize the continuous solution
        h_binarized = (best_h > 0.5).astype(float)
        # Fix sum constraint
        cnt = int(h_binarized.sum())
        if cnt != n // 2:
            sorted_idx = np.argsort(-best_h)
            h_binarized = np.zeros(n)
            h_binarized[sorted_idx[:n//2]] = 1.0
        rng5 = np.random.RandomState(seed + 20)
        h_bin3, c_bin3 = targeted_binary_search(h_binarized, n, p6_deadline, rng5)
        if c_bin3 < best_c:
            best_c = c_bin3
            best_h = h_bin3.copy()
    
    # Phase 7: Final continuous polish (80s)
    p7_deadline = min(time.time() + 80, deadline - 20)
    if time.time() < p7_deadline:
        h_cont2, c_cont2 = continuous_optimize(best_h, n, p7_deadline, max_rounds=15)
        if c_cont2 < best_c:
            best_c = c_cont2
            best_h = h_cont2.copy()
    
    # Phase 8: Very fine continuous adjustments (50s)
    p8_deadline = min(time.time() + 50, deadline - 5)
    if time.time() < p8_deadline:
        h_fine, c_fine = fine_tune_continuous(best_h, n, p8_deadline)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
