    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    
    for alpha_idx, alpha in enumerate(alphas):
        # Reset FISTA state for each alpha
        x = best_h.copy()
        y = x.copy()
        t = 1.0
        
        # Estimate Lipchitz constant (rough)
        c0 = compute_c5_vec(x, n)
        L = alpha * (2.0 / n) * (2 * n - 1)  # rough upper bound
        step = 1.0 / max(L, 1.0)
        
        iters_done = 0
        stall_count = 0
        last_improve = best_c
        
        while time.time() < deadline:
            # FISTA update
            f, g = smooth_obj_and_grad_full(y, n, alpha, mu)
            
            # Backtracking line search
            step *= 0.5

                x_new = project_feasible(x_new, n)
                f_new = smooth_obj(x_new, n, alpha, mu)
                # Check quadratic termination
                quad_bound = f + np.dot(g, x_new - y) + 0.5 / step * np.sum((x_new - y)**2)
                if f_new <= quad_bound + 1e-10:
                    break
                step *= 0.5
            
            # Nesterov update for y
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t**2)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = project_feasible(y, n)
            t = t_new
            x = x_new
            
            iters_done += 1
            
            # Periodic evaluation of true objective
            if iters_done % 20 == 0:
                c_true = compute_c5(project_feasible(x, n), n)
                if c_true < best_c - 1e-10:

                    best_h = project_feasible(x, n)

                    stall_count += 1

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_set(h, n, tol_frac=0.001, max_active=50):
    c = compute_c5_vec(h, n)
    mx = c.max()
    tol = max(1e-9, tol_frac * mx)
    active = np.where(c >= mx - tol)[0]
    if len(active) > max_active:
        order = np.argsort(-c[active])
        active = active[order[:max_active]]
    return active

def run_lbfgs_smooth(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=10, alphas=None, tol_frac=0.002):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 5000000]
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        active = get_active_set(best_h, n, tol_frac=tol_frac, max_active=top_k)
        if len(active) == 0:
            active = np.array([np.argmax(compute_c5_vec(best_h, n))])
        for alpha in alphas:
            if time.time() > deadline - 12:
                break
            maxit = max(300, int((min(deadline, time.time() + remaining * 0.3) - time.time()) * 3))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_trust_region(h, n, deadline, alpha=100000, mu=0.01, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_obj, h, jac=sm_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-15, 'maxfun': maxiter * 10,
                               'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def fista_proximal(h, n, deadline, alpha, mu=0.005, max_iters=50000):
    """Accelerated proximal gradient (FISTA) for the smoothed problem."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    x = best_h.copy()
    y = x.copy()
    t = 1.0
    step = 1.0 / (alpha * 4.0)  # initial step
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        _, g = (_ for _ in [None]).__next__ and (sm_obj(y, n, alpha, mu), sm_grad(y, n, alpha, mu))
        
        x_new = y - step * g
        x_new = project_feasible(x_new, n)
        
        if it % 25 == 0:
            c_true = compute_c5(x_new, n)
            if c_true < best_c - 1e-12:
                best_c = c_true
                best_h = x_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project_feasible(y_new, n)
        t = t_new
        x = x_new
        y = y_new
        
        if it % 200 == 0 and it > 0:
            t = 1.0  # periodic restart of momentum
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def local_search(h, n, deadline, radius=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick polish at original resolution
    p1_end = min(t0 + 120, deadline - 600)
    h1, c1 = run_lbfgs_smooth(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 1000000],
        mu=0.01, maxiter_per=[1500, 2000, 2500, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    # Active set polish at original resolution
    if time.time() < deadline - 550:
        p1b_end = min(time.time() + 60, deadline - 550)
        h1b, c1b = run_active_opt(best_h, best_n, p1b_end, top_k=8,
            alphas=[50000, 200000, 500000, 2000000], tol_frac=0.003)
        if c1b < best_c:
            best_c = c1b
            best_h = h1b
    
    # Phase 2: Upsample to n=1000 and optimize
    n_target = 1000
    h_up = upsample(best_h, best_n, n_target)
    c_up = compute_c5(h_up, n_target)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up
        best_n = n_target
    
    # Also try 1-h as a different starting point
    h_flip = project_feasible(1.0 - best_h, best_n)
    c_flip = compute_c5(h_flip, best_n)
    if c_flip < best_c:
        best_c = c_flip
        best_h = h_flip.copy()
    
    p2_end = min(time.time() + 350, deadline - 250)
    h2, c2 = run_lbfgs_smooth(h_up, n_target, p2_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 500000, 2000000],
        mu=0.005, maxiter_per=[800, 1000, 1500, 2000, 2500, 2500, 3000, 3000, 3500])
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = n_target
    
    # Phase 3: Active set optimization at n=1000
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 200, deadline - 200)
        h3, c3 = run_active_opt(best_h, best_n, p3_end, top_k=12,
            alphas=[20000, 100000, 500000, 2000000, 5000000, 20000000],
            tol_frac=0.003)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n
    
    # Phase 4: Try FISTA at n=1000
    if time.time() < deadline - 180:
        p4_end = min(time.time() + 100, deadline - 150)
        h4, c4 = fista_proximal(best_h, best_n, p4_end, alpha=200000, mu=0.003)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n
    
    # Phase 5: Try n=1500 if time permits
    if time.time() < deadline - 150:
        n_1500 = 1500
        h_1500 = upsample(best_h, best_n, n_1500)
        p5_end = min(time.time() + 100, deadline - 100)
        h5, c5 = run_lbfgs_smooth(h_1500, n_1500, p5_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000],
            mu=0.003, maxiter_per=[500, 700, 1000, 1200, 1500, 1500, 2000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_1500
    
    # Phase 6: Subgradient refinement
    if time.time() < deadline - 50:
        p6_end = min(time.time() + 40, deadline - 30)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=10000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n
    
    # Phase 7: Final active set + local search
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 20, deadline - 20)
        h7, c7 = run_active_opt(best_h, best_n, p7_end, top_k=6,
            alphas=[500000, 2000000, 10000000], tol_frac=0.005)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Local search
    if time.time() < deadline - 5:
        p8_end = max(deadline - 2, time.time() + 10)
        h8, c8 = local_search(best_h, best_n, p8_end, radius=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
