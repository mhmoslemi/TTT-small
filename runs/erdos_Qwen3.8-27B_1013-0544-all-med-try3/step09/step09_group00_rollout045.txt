            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def active_set_grad(h, n, active_idx):
    """Gradient of average of B_k over active set."""
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx_val
            grad[k:n] += (-h[0:nv]) * dx_val
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx_val
            grad[0:nv] += (-h[kk:n]) * dx_val
    grad /= len(active_idx)
    return grad

def lbfgs_smooth(h, n, alpha, mu, deadline, maxit=2000):
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n/2.0)**2
        return val
    def grad(x):
        c = compute_c5_vec(x, n)
        g = smooth_max_grad(c, alpha, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n/2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit*8,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def subgradient_step(h, n, lr, tol_frac=0.005):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - tol_frac * max_val
    active = np.where(c_vec >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    grad = active_set_grad(h, n, active)
    new_h = project_feasible(h - lr * grad, n)
    return new_h, compute_c5(new_h, n)

def subgradient_descent(h, n, deadline, lr_init=0.005, lr_min=1e-7, decay=0.9995, tol_frac=0.005):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    lr = lr_init
    stall = 0
    while time.time() < deadline and lr > lr_min:
        new_h, new_c = subgradient_step(best_h, n, lr, tol_frac)
        if new_c < best_c - 1e-12:
            best_h = new_h
            best_c = new_c
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                stall = 0
                if lr < lr_min:
                    break
        lr *= decay
    return best_h, best_c

def binary_flip_search(h, n, deadline, max_rounds=10):
    """Try flipping values toward 0 or 1 for nearly-binary solutions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try rounding each element to nearest of {0, 1}
        for i in range(n):
            if time.time() > deadline:
                break
            target = 1.0 if best_h[i] > 0.5 else 0.0
            if best_h[i] == target:
                continue
            old = best_h[i]
            best_h[i] = target
            # Need to maintain sum = n/2
            # Adjust a compensating element
            # Find the element most "in the middle" to adjust
            mid_mask = (best_h > 0.05) & (best_h < 0.95)
            if mid_mask.sum() == 0:
                best_h[i] = old
                continue
            mid_idx = np.where(mid_mask)[0]
            # Choose the one whose adjustment causes least disruption
            # Simple: just pick the first
            j = mid_idx[0]
            if j == i:
                best_h[i] = old
                continue
            adj = (old - target) + (best_h[j] - best_h[j])  # This is just old - target
            # Actually we need: best_h[i] changed by (target - old), so need best_h[j] changed by (old - target)
            best_h[j] += (old - target)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_feasible(best_h, n)
            c = compute_c5(best_h, n)
            if c < best_c - 1e-12:
                best_c = c
                improved = True
            else:
                best_h[i] = old
                best_h[j] -= (old - target)
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def interp_upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.append(h, h[0]))
    return project_feasible(h_new, n_new)

def beatty_binary(n, irrational):
    """Generate binary pattern using Beatty sequence."""
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * irrational) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def pair_exchange(h, n, i, j, amt):
    """Exchange amt from h[i] to h[j]."""
    new_h = h.copy()
    new_h[i] -= amt
    new_h[j] += amt
    if new_h[i] < -1e-12 or new_h[j] > 1.0 + 1e-12:
        return None
    new_h = np.clip(new_h, 0.0, 1.0)
    return project_feasible(new_h, n)

def local_pair_opt(h, n, deadline, deltas=None, radius=60):
    """Local pair optimization: try small exchanges between pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            if time.time() > deadline:
                break
            for i in range(0, n, 3):  # Step by 3 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if i == j:
                        continue
                    # Try moving d from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        new_h = pair_exchange(best_h, n, i, j, d)
                        if new_h is not None:
                            c = compute_c5(new_h, n)
                            if c < best_c - 1e-14:
                                best_h = new_h
                                best_c = c
                                improved = True
                    # Try moving d from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        new_h = pair_exchange(best_h, n, i, j, -d)
                        if new_h is not None:
                            c = compute_c5(new_h, n)
                            if c < best_c - 1e-14:
                                best_h = new_h
                                best_c = c
                                improved = True
    return best_h, best_c
