    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    try:
        res = minimize(_active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def multi_start_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Gradient with increasing alpha
    alphas = [(200, 0.15), (2000, 0.15), (20000, 0.20), (200000, 0.25), (2000000, 0.25)]
    total_frac = sum(f for _, f in alphas)
    phase_end = deadline - 10
    t_phase = time.time()
    
    for alpha, frac in alphas:
        if time.time() > phase_end:
            break
        phase_deadline = time.time() + frac * total_frac * (phase_end - t_phase) / total_frac
        phase_deadline = min(phase_deadline, phase_end)
        h_new, c_new = grad_momentum(h, n, alpha, phase_deadline, lr=2.5, momentum=0.90)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    # Phase 2: SLSQP
    if time.time() < deadline - 20:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(h, n, 500000, maxiter=2000, ftol=1e-16, top_m=top_m)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    
    # Phase 3: Intensive local search
    if time.time() < deadline - 8:
        ls_deadline = min(deadline - 3, time.time() + (deadline - time.time()) * 0.5)
        h_new, c_new = top_lag_local_search(best_h, n, ls_deadline, top_k=5, max_rounds=300)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 4: Binary projection
    if time.time() < deadline - 5:
        h_new, c_new = binary_project_and_refine(best_h, n, deadline - 3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    # Phase 5: Final local search
    if time.time() < deadline - 3:
        h_new, c_new = top_lag_local_search(best_h, n, deadline - 1, top_k=7, max_rounds=200)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    remaining = deadline - time.time()
    
    # Main optimization at parent resolution
    if remaining > 60:
        alloc = min(remaining * 0.45, 400)
        h_opt, c_opt = multi_start_optimize(best_h, n_init, t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Try higher resolution
    for n_try in [n_init + 50, n_init + 100, n_init + 150]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(remaining * 0.25, 120)
        if alloc < 40:
            break
        h_start = interpolate_h(best_h, len(best_h), n_try)
        h_opt, c_opt = multi_start_optimize(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Try lower resolution (sometimes coarser grid is better)
    for n_try in [n_init - 50, n_init - 100]:
        if time.time() > deadline - 70:
            break
        if n_try < 100:
            continue
        remaining = deadline - time.time() - 25
        if remaining < 40:
            break
        alloc = min(remaining * 0.3, 80)
        if alloc < 30:
            break
        h_start = interpolate_h(best_h, len(best_h), n_try)
        h_opt, c_opt = multi_start_optimize(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Multi-restart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        n_restarts = max(1, int((deadline - time.time() - 30) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            # Structured perturbation
            scale = 0.01 * (1.0 + 0.5 * r)
            if r % 2 == 0:
                # Smooth perturbation
                freq = rng.uniform(0.5, 3.0)
                t = np.linspace(0, 2, n_best)
                pert = scale * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
                hp = best_h + pert
            else:
                # Block perturbation
                block = max(10, n_best // 10)
                pert = np.zeros(n_best)
                start = rng.randint(0, n_best - block)
                pert[start:start + block] = scale * rng.uniform(-1, 1)
                hp = best_h + pert
            hp = project_feasible(hp, n_best)
            alloc = min(45, deadline - time.time() - 25)
            if alloc < 20:
                break
            h_opt, c_opt = multi_start_optimize(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Final gradient
        if time.time() < deadline - 15:
            h_new, c_new = grad_momentum(best_h, n, 5000000, deadline - 12, lr=1.0, momentum=0.85)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 3:
            best_h, best_c = top_lag_local_search(best_h, n, deadline - 1, top_k=8, max_rounds=500)
    
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    
    # Verify feasibility
    s = best_h.sum() * (2.0 / n_best)
    if abs(s - 1.0) > 1e-6:
        best_h = project_feasible(best_h, n_best)
        best_c = compute_c5(best_h, n_best)
    
    return best_h, best_c, len(best_h)
