    if abs(s - target) > 1e-10:

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

# === Projection ===

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(h - lam, 0.0, 1.0)
    # Verify and fix
    s = result.sum()
    if abs(s - target) > 1e-10:
        # Adjust non-boundary elements
        mask = (result > 1e-15) & (result < 1.0 - 1e-15)
        if mask.sum() > 0:
            result[mask] += (target - s) / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result
