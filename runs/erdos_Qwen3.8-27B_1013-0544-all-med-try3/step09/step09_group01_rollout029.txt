def corr_full_fft(a,b):
    n=len(a)
    L=1
    while L < 2*n-1: L*=2
    fa=np.fft.rfft(a,L)

def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0,1)
    target=n/2.0
    for _ in range(100):
       d=target-h.sum()
       if abs(d) < 1e-13: break
       if d>0:
          room=1.0-h; tot=room.sum()
          if tot<1e-15: break
          h += d*room/tot
       else:
          tot=h.sum()
          if tot<1e-15: break
          h += d*h/tot
    h=np.clip(h,0,1)
    # fine
    for _ in range(20):
       d=target-h.sum()
       if abs(d) <= 1e-15: break
       if d>0:
          mask=h<1.0
       else: mask=h>0.0
       if not mask.any(): break
       idxs=np.where(mask)[0]
       # choose one with largest capacity
       if d>0:
          cap=1-h[idxs]; pos=int(np.argmax(cap))
          add=min(d, cap[pos])
          h[idxs[pos]] += add
       else:
          cap=h[idxs]; pos=int(np.argmax(cap))
          sub=min(-d, cap[pos])
          h[idxs[pos]] -= sub
    # If not exact, spread to all with small
    d=target-h.sum()
    if abs(d)>0:
       if d>0: mask=h<1 else h>0; adjust first
    return h

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (d / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (d / tot) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):

        if abs(d) <= 1e-14: break

def smooth_max(vals, alpha):
    vals=np.asarray(vals, dtype=float)
    m=float(vals.max())
    if alpha <= 1e-12:
       p=np.full_like(vals, 1.0/vals.size)
       return m,p
    z=alpha*(vals-m)
    zm=float(z.max())
    ex=np.exp(z-zm)
    s=float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:

       mask=vals >= m - tol

       else: p[mask]=1.0/np.count_nonzero(mask)
       return m,p
    p=ex/s
    # clean small
    val = m + (zm + float(np.log(s))) / alpha

def select_active(c,k):

_ONE_CACHE = {}

def get_ones(n):
    if n not in _ONE_CACHE:
        _ONE_CACHE[n] = np.ones(n, dtype=float)
    return _ONE_CACHE[n]

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (d / tot) * h

    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) <= 1e-14:
            break
        if d > 0.0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            break
        if d > 0.0:
            caps = 1.0 - h[idx]
            pos = int(np.argmax(caps))
            add = min(float(d), float(caps[pos]))
            h[idx[pos]] += add
        else:
            caps = h[idx]
            pos = int(np.argmax(caps))
            sub = min(float(-d), float(caps[pos]))
            h[idx[pos]] -= sub
    return h

def compute_c5_exact(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def correlate_p_fft(h, n):
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft((1.0 - h)[::-1], L)
    out = np.fft.irfft(fa * fb, L)[:m]
    return out[::-1]

def compute_c5_fast(h, n):
    return float(np.max(correlate_p_fft(h, n)) * (2.0 / float(n)))

def smooth_max(vals, alpha):
    vals = np.asarray(vals, dtype=float)
    m = float(vals.max())
    if vals.size == 1:
        return m, np.ones(1, dtype=float)
    if alpha <= 1e-12:
        return m, np.full(vals.size, 1.0 / vals.size, dtype=float)

    z = alpha * (vals - m)
    zm = float(z.max())
    ex = np.exp(z - zm)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(vals.size, dtype=float)
        tol = max(1e-15, 1e-12 * max(1.0, abs(m)))
        mask = vals >= (m - tol)
        if not mask.any():
            p[int(np.argmax(vals))] = 1.0
        else:
            p[mask] = 1.0 / float(np.count_nonzero(mask))
        return m, p

    p = ex / s
    val = m + (zm + float(np.log(s))) / alpha
    return float(val), p

def grad_from_active_weights(h, n, active, weights):
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    for idx, w in zip(active, weights):
        if w <= 1e-16:
            continue
        k = int(idx) - (n - 1)
        wm = w * dx
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += -wm * h[:nv]
        else:
            r = -k
            nv = n - r
            grad[r:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[r:]
    return grad

def grad_from_weights_loop(h, n, p):
    nz = np.where(p > 1e-15)[0]
    if nz.size == 0:
        return np.zeros(n, dtype=float)
    return grad_from_active_weights(h, n, nz, p[nz])

def obj_dense(h, n, alpha, mu):
    c = correlate_p_fft(h, n) * (2.0 / float(n))
    val, _ = smooth_max(c, alpha)
    if mu != 0.0:
        d = h.sum() - 0.5 * n
        val += mu * d * d
    return float(val)

def grad_dense(h, n, alpha, mu):
    c = correlate_p_fft(h, n) * (2.0 / float(n))
    _, p = smooth_max(c, alpha)
    p = p.copy()
    p[np.abs(p) <= 1e-14] = 0.0
    s = p.sum()
    if s > 0.0:
        p /= s
    else:
        p[int(np.argmax(c))] = 1.0

    dx = 2.0 / float(n)
    nnz = int(np.count_nonzero(p))
    if nnz <= 80:
        nz = np.where(p > 0.0)[0]
        grad = grad_from_active_weights(h, n, nz, p[nz])
    else:
        pw = p
        pw_rev = pw[::-1]
        ones = get_ones(n)
        A = np.convolve(pw_rev, ones, mode="full")[n - 1:2 * n - 1]
        B = np.convolve(pw_rev, h, mode="full")[n - 1:2 * n - 1]
        C = np.convolve(pw, h, mode="full")[n - 1:2 * n - 1]
        grad = dx * (A - B - C)

    if mu != 0.0:
        grad += 2.0 * mu * (h.sum() - 0.5 * n)
    return grad

def obj_active(h, n, active, alpha, mu):
    c = correlate_p_fft(h, n) * (2.0 / float(n))
    vals = c[active]
    val, _ = smooth_max(vals, alpha)
    if mu != 0.0:
        d = h.sum() - 0.5 * n
        val += mu * d * d
    return float(val)

def grad_active(h, n, active, alpha, mu):
    c = correlate_p_fft(h, n) * (2.0 / float(n))
    vals = c[active]
    _, w = smooth_max(vals, alpha)
    grad = grad_from_active_weights(h, n, active, w)
    if mu != 0.0:
        grad += 2.0 * mu * (h.sum() - 0.5 * n)
    return grad

def select_active(c, k):
    k = max(1, int(k))
    m = float(c.max())
    tol = max(1e-9, 1e-6 * abs(m))
    idx = np.where(c >= m - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    if idx.size > k:
        order = np.argsort(-c[idx])
        idx = idx[order[:k]]
    elif idx.size < 3 and idx.size < k:
        top = np.argsort(-c)[:max(k, 3)]
        idx = np.unique(np.concatenate((idx, top)))
        if idx.size > k:
            order = np.argsort(-c[idx])
            idx = idx[order[:k]]
    return np.asarray(idx, dtype=int)

def random_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    target = 0.5 * float(n)
    ones = int(n // 2)
    perm = rng.permutation(n)
    h[perm[:ones]] = 1.0
    if target > ones:
        if ones < n:
            h[perm[ones]] += float(target - ones)
    return project_feasible(h, n)

            phase_end = min(deadline - 60.0, time.time() + 60.0)
            h4, c4 = run_active_lbgf(cur_h, n4, phase_end, ACTIVE_4096, 200, 0.002)
            if c4 < cur_c:
                cur_h = h4
                cur_c = c4
                cur_n = n4
            if c4 < best_c:
                best_c = c4
                best_h = h4.copy()
                best_n = n4

    if time.time() < deadline - 25.0:
        hsub, csub = subgrad_polish(
            best_h,
            best_n,
            deadline - 5.0,
            topk=6,
            alpha=100000.0,
            maxit=350,
        )
        if csub < best_c:
            best_h = hsub.copy()
            best_c = csub
            best_n = len(hsub)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_exact(best_h, best_n)
    return best_h, float(best_c), int(best_n)
