def project_feasible(h, n):

                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
                            else:
                                h = best_h.copy()
                        elif h[j] > h[i] + 1e-6:
                            delta = min(0.08, (h[j] - h[i]) * 0.4)
                            if delta < 1e-8:
                                continue
                            h[i] += delta
                            h[j] -= delta
                            h_t = project_feasible(h, n)
                            c_new = compute_c5(h_t, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
                            else:
                                h = best_h.copy()
            else:
                # lag < 0: c = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
                valid_range = np.arange(0, n - m)
                if len(valid_range) == 0:
                    continue
                # For lag=-m: grad[i] for i in [0,m): -h[i+m], grad[i] for i in [m,n): (1-h[i-m])
                gv = g.copy()
                n_tries = min(40, len(valid_range))
                top_high = np.argsort(gv)[-n_tries:]
                top_low = np.argsort(gv)[:n_tries]
                tried = set()
                for ih in top_high[:15]:
                    for il in top_low[:15]:
                        if ih == il or (ih, il) in tried:
                            continue
                        tried.add((ih, il))
                        i = int(ih)
                        j = int(il)
                        if i + m >= n or j + m >= n:
                            continue
                        if h[i] > h[j] + 1e-6:
                            delta = min(0.08, (h[i] - h[j]) * 0.4)
                            if delta < 1e-8:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            h_t = project_feasible(h, n)
                            c_new = compute_c5(h_t, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
                            else:
                                h = best_h.copy()
                        elif h[j] > h[i] + 1e-6:
                            delta = min(0.08, (h[j] - h[i]) * 0.4)
                            if delta < 1e-8:
                                continue
                            h[i] += delta
                            h[j] -= delta
                            h_t = project_feasible(h, n)
                            c_new = compute_c5(h_t, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
                            else:
                                h = best_h.copy()
        if not improved:
            # Broader random search across all top lags
            c_all = compute_all_lags(h, n)
            top10 = np.argsort(c_all)[-10:]
            improved2 = False
            rng_local = np.random.RandomState(rnd * 31 + 7)
            for k2 in top10:
                if time.time() > deadline:
                    break
                lag2 = int(k2 - (n - 1))
                if lag2 == 0:
                    continue
                m2 = abs(lag2)
                if m2 >= n:
                    continue
                n_pairs = min(100, n - m2)
                for _ in range(n_pairs):
                    if time.time() > deadline:
                        break
                    i = rng_local.randint(0, n - m2)
                    j = i + m2
                    delta = 0.02 * rng_local.random()
                    if h[i] > h[j]:
                        if delta >= h[i] * 0.5 or delta >= (1 - h[j]) * 0.5:
                            continue
                        h[i] -= delta
                        h[j] += delta
                    elif h[j] > h[i]:
                        if delta >= h[j] * 0.5 or delta >= (1 - h[i]) * 0.5:
                            continue
                        h[i] += delta
                        h[j] -= delta
                    else:
                        continue
                    h_t = project_feasible(h, n)
                    c_new = compute_c5(h_t, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved2 = True
                    else:
                        h = best_h.copy()
            if not improved2:
                break
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def _slsqp_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def _slsqp_jac(x, n, alpha, top_idx):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] < 1e-10:
            continue
        lag = int(idx - (n - 1))
        grad += w[i] * lag_gradient(x, n, lag)
    return grad * dx

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha=100000, maxiter=3000, top_m=5):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    if len(top_idx) == 0:
        return h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def anneal_local_search(h, n, deadline, max_iter=50000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(99)
    temp = 0.001
    for it in range(max_iter):
        if time.time() > deadline:
            break
        temp = max(0.0001, temp * 0.9995)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] > h[j]:
            delta = min(0.05 * rng.random() + 0.005, (h[i] - h[j]) * 0.3)
        elif h[j] > h[i]:
            delta = -(min(0.05 * rng.random() + 0.005, (h[j] - h[i]) * 0.3))
        else:
            delta = 0.01 * (rng.random() - 0.5)
        if abs(delta) < 1e-10:
            continue
        h[i] -= delta
        h[j] += delta
        h_t = project_feasible(h, n)
        c_new = compute_c5(h_t, n)
        if c_new < best_c + temp:
            h = h_t.copy()
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_t.copy()
        else:
            h = best_h.copy() if rng.random() > 0.05 else h_t.copy()
            if c_new >= best_c:
                h = best_h.copy()
    return best_h, best_c

def binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Gradient optimization at initial resolution
    if time.time() < deadline - 100:
        alloc = min(50, (deadline - t0) * 0.06)
        h1, c1 = proj_gradient_descent(best_h, n_init, t0 + alloc, alpha0=200, alpha1=20000, lr0=1.5, max_iter=8000, top_k=50)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: Local search at initial resolution
    if time.time() < deadline - 80:
        alloc = min(60, (deadline - t0) * 0.10)
        h2, c2 = gradient_local_search(best_h, n_init, t0 + alloc, top_lags=5, max_rounds=200)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: SLSQP at initial resolution
    if time.time() < deadline - 60:
        h3 = slsqp_refine(best_h, n_init, alpha=80000, maxiter=2000, top_m=5)
        c3 = compute_c5(h3, n_init)
        if c3 < best_c:
            best_c, best_h = c3, h3
        h3b = slsqp_refine(best_h, n_init, alpha=200000, maxiter=2000, top_m=8)
        c3b = compute_c5(h3b, n_init)
        if c3b < best_c:
            best_c, best_h = c3b, h3b

    # Phase 4: Move to higher resolution
    n_high = 1000
    if time.time() < deadline - 250:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c, best_h = c_high, h_high
        else:
            best_h = h_high

    # Phase 5: Gradient optimization at high resolution
    if time.time() < deadline - 180:
        n_cur = len(best_h)
        alloc = min(120, (deadline - time.time()) * 0.25)
        h5, c5v = proj_gradient_descent(best_h, n_cur, time.time() + alloc, alpha0=400, alpha1=50000, lr0=1.2, max_iter=15000, top_k=60)
        if c5v < best_c:
            best_c, best_h = c5v, h5

    # Phase 6: Local search at high resolution
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        alloc = min(150, (deadline - time.time()) * 0.30)
        h6, c6 = gradient_local_search(best_h, n_cur, time.time() + alloc, top_lags=6, max_rounds=400)
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: SLSQP at high resolution
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        h7 = slsqp_refine(best_h, n_cur, alpha=150000, maxiter=2500, top_m=5)
        c7 = compute_c5(h7, n_cur)
        if c7 < best_c:
            best_c, best_h = c7, h7
        h7b = slsqp_refine(best_h, n_cur, alpha=400000, maxiter=2500, top_m=10)
        c7b = compute_c5(h7b, n_cur)
        if c7b < best_c:
            best_c, best_h = c7b, h7b

    # Phase 8: More local search
    if time.time() < deadline - 50:
        n_cur = len(best_h)
        alloc = min(80, (deadline - time.time()) * 0.35)
        h8, c8 = gradient_local_search(best_h, n_cur, time.time() + alloc, top_lags=8, max_rounds=300)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Try even higher resolution if time
    n_veryhigh = 1500
    if time.time() < deadline - 100 and len(best_h) < n_veryhigh:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_veryhigh)
        h_vh = np.interp(x_new, x_old, best_h)
        h_vh = project_feasible(h_vh, n_veryhigh)
        c_vh = compute_c5(h_vh, n_veryhigh)
        if c_vh < best_c:
            best_c = c_vh
        best_h = h_vh
        # Quick gradient step at very high res
        if time.time() < deadline - 40:
            h9, c9 = proj_gradient_descent(best_h, n_veryhigh, deadline - 20, alpha0=500, alpha1=30000, lr0=1.0, max_iter=5000, top_k=50)
            if c9 < best_c:
                best_c, best_h = c9, h9

    # Phase 10: Final local search
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h10, c10 = gradient_local_search(best_h, n_cur, deadline - 3, top_lags=6, max_rounds=100)
        if c10 < best_c:
            best_c, best_h = c10, h10

    # Final SLSQP if time
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h11 = slsqp_refine(best_h, n_cur, alpha=300000, maxiter=1500, top_m=8)
        c11 = compute_c5(h11, n_cur)
        if c11 < best_c:
            best_c, best_h = c11, h11

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
