    """Project h to satisfy 0<=h<=1 and sum(h)*dx=1 (i.e., sum(h)=n/2)."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    """Gradient of c[shift_idx] with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def power_obj(x, n, p):
    """Power-sum approximation to max."""
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-30)  # avoid issues with negative values
    # Use c^p / p as the objective (constant doesn't matter for optimization)
    val = np.sum(c_pos ** p) / p
    return float(val)

def power_grad(x, n, p):
    """Gradient of power-sum objective."""
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-30)
    weights = c_pos ** (p - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        w = weights[mm]
        if w < 1e-30:
            continue
        k = mm - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def logsumexp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def logsumexp_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, configs=None):
    """L-BFGS with various smooth objectives."""
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    if configs is None:
        configs = [
            ('lse', 500, None),
            ('lse', 2000, None),
            ('lse', 5000, None),
            ('lse', 10000, None),
            ('pow', None, 20),
            ('pow', None, 50),
            ('pow', None, 100),
        ]
    
    for cfg in configs:
        if time.time() > deadline - 10:
            break
        method_type = cfg[0]
        if method_type == 'lse':
            alpha = cfg[1]
            res = minimize(logsumexp_obj, best_h, jac=logsumexp_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 40})
        else:
            p = cfg[2]
            res = minimize(power_obj, best_h, jac=power_grad,
                          args=(n, p), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000, 'maxls': 40})
        
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def sqp_optimize(h, n, deadline, max_iter=300, rho_init=0.05):
    """SQP with LP subproblem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rho = rho_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-10, 0.0005 * max_val)
        active = list(np.where(c_vec >= thresh)[0])
        if not active:
            break
        if len(active) > 15:
            top_order = np.argsort(-c_vec[active])
            active = [active[i] for i in top_order[:15]]
        
        # Build LP
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[-1] = 1.0
        grads = []
        c_vals = []
        for idx in active:
            g = shift_grad_single(best_h, n, idx)
            grads.append(g)
            c_vals.append(c_vec[idx])
        n_act = len(active)
        total_ineq = n_act + 4 * n
        A_ub = np.zeros((total_ineq, n_var))
        b_ub = np.zeros(total_ineq)
        row = 0
        for j in range(n_act):
            A_ub[row, 0:n] = grads[j]
            A_ub[row, n] = -1.0
            b_ub[row] = -c_vals[j]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = best_h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = 1.0 - best_h[i]
            row += 1
        for i in range(n):
            A_ub[row, i] = -1.0
            b_ub[row] = rho
            row += 1
        for i in range(n):
            A_ub[row, i] = 1.0
            b_ub[row] = rho
            row += 1
        A_eq = np.zeros((1, n_var))
        A_eq[0, 0:n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(None, None)] * n + [(0.0, 2.0)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[0:n]
                h_new = project_feasible(best_h + delta, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    no_improve = 0
                    if rho < 0.1:
                        rho = min(rho * 1.4, 0.1)
                else:
                    no_improve += 1
                    rho *= 0.5
                    if rho < 1e-10:
                        break
                    if no_improve > 30:
                        break
            else:
                no_improve += 1
                rho *= 0.5
                if rho < 1e-10:
                    break
        except Exception:
            break
    
    return best_h, best_c

def advanced_local_search(h, n, deadline, center_radius=60):
    """More powerful local search with multiple move types."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Move types and step sizes
    step_sizes = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            thresh = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= thresh)[0]
            
            # Compute combined gradient magnitude
            grad_mag = np.zeros(n)
            n_act = len(active)
            if n_act == 0:
                break
            for idx in active[:min(n_act, 30)]:
                g = shift_grad_single(best_h, n, idx)
                grad_mag += np.abs(g)
            grad_mag /= min(n_act, 30)
            
            # Strategy 1: 2-swap moves on high-gradient positions
            top_pos = np.argsort(-grad_mag)[:min(center_radius * 3, n)]
            for i_val in top_pos:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_val)
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += step, h[j] -= step
                    if best_h[i] + step <= 1.0 + 1e-12 and best_h[j] - step >= -1e-12:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
                    # Try h[i] -= step, h[j] += step
                    if best_h[i] - step >= -1e-12 and best_h[j] + step <= 1.0 + 1e-12:
                        best_h[i] -= step
                        best_h[j] += step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += step
                            best_h[j] -= step
            # Strategy 2: Smooth perturbation (Gaussian bump)
            if time.time() > deadline:
                return best_h, best_c
            # Try adding a small Gaussian-shaped perturbation at random positions
            n_tries = 20
            for _ in range(n_tries):
                if time.time() > deadline:
                    return best_h, best_c
                center = np.random.randint(0, n)
                width = np.random.choice([5, 10, 20, 40, 80])
                # Create a zero-mean perturbation
                pert = np.zeros(n)
                x = np.arange(n)
                pert = np.exp(-0.5 * ((x - center) / width) ** 2)
                # Make it zero-mean
                pert -= pert.sum() / n
                # Scale to step size
                pert *= step / (np.max(np.abs(pert)) + 1e-30)
                # Check bounds
                h_test = best_h + pert
                if h_test.min() >= -1e-12 and h_test.max() <= 1.0 + 1e-12:
                    h_test = np.clip(h_test, 0, 1)
                    h_test = project_feasible(h_test, n)
                    c = compute_c5(h_test, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h_test
                        improved = True
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=3000):
    """SLSQP polish with high alpha."""
    def obj_fn(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sq_fn(x, n):
        return float(x.sum() - n / 2.0)
    
    def sq_jac_fn(x, n):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new
