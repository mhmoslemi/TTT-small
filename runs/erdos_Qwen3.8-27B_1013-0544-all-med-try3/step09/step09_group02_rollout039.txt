5. Use bisection to find the optimal x

    # Current overlap values
    c_vec = compute_c5_vec(h, n)  # shape (2n-1,)
    
    # Compute gradient coefficients a[k] for each shift
    a = np.zeros(2*n-1)
    
    # Positive shifts: k = 0 to n-1, shift_idx = n-1+k
    for k in range(n):
        si = n - 1 + k
        ak = 0.0
        if k <= n - 1 - i:  # i is in range [0, n-1-k]
            ak += (1.0 - h[i + k])
        if k <= i:  # i-k is in range [0, n-1-k] (need i-k >= 0 and i-k <= n-1-k)
            # i-k >= 0: k <= i (satisfied)
            # i-k <= n-1-k: i <= n-1 (always true)
            ak -= h[i - k]
        a[si] = ak
    
    # Negative shifts: kk = 1 to n-1, shift_idx = n-1-kk
    for kk in range(1, n):
        si = n - 1 - kk
        ak = 0.0
        if kk <= i:  # i-kk >= 0
            ak += (1.0 - h[i - kk])
        if kk <= n - 1 - i:  # i <= n-1-kk
            ak -= h[i + kk]

```python
def compute_a_coeff(h, n, i):
    """Compute a[k] for all shifts k for coordinate i."""
    a = np.zeros(2*n-1)
    
    # Positive shifts: k = 0 to n-1, shift_idx = n-1+k
    # First term: (1-h[i+k]) valid when k <= n-1-i
    # Second term: -h[i-k] valid when k <= i
    
    max_k_first = n - 1 - i  # max k for first term
    max_k_second = i  # max k for second term
    
    # First term
    k_range = np.arange(0, min(max_k_first, n-1) + 1)
    if len(k_range) > 0:
        a[n-1 + k_range] += (1.0 - h[i + k_range])
    
    # Second term
    k_range2 = np.arange(0, min(max_k_second, n-1) + 1)
    if len(k_range2) > 0:
        a[n-1 + k_range2] -= h[i - k_range2]
    
    # Negative shifts: kk = 1 to n-1, shift_idx = n-1-kk
    # First term: (1-h[i-kk]) valid when kk <= i
    # Second term: -h[i+kk] valid when kk <= n-1-i
    
    max_kk_first = i
    max_kk_second = n - 1 - i
    
    kk_range = np.arange(1, min(max_kk_first, n-1) + 1)
    if len(kk_range) > 0:
        a[n-1 - kk_range] += (1.0 - h[i - kk_range])
    
    kk_range2 = np.arange(1, min(max_kk_second, n-1) + 1)
    if len(kk_range2) > 0:
        a[n-1 - kk_range2] -= h[i + kk_range2]
    
    return a
```

def coord_descent_transfer(h, n, deadline=None, n_sweeps=1000):
    c_vec = compute_c5_vec(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        
        improved = False
        for i in range(n):
            # Choose a partner j (random or round-robin)
            j = (i + 1) % n  # Simple: pair with next coordinate
            
            # Compute combined gradient coefficients
            a_i = compute_a_coeff(h, n, i)
            a_j = compute_a_coeff(h, n, j)
            a_diff = a_i - a_j  # Coefficient of d in each O(k)
            
            # Valid range for d
            lo_d = max(-h[i], h[j] - 1.0)
            hi_d = min(1.0 - h[i], h[j])
            
            if hi_d - lo_d < 1e-14:
                continue
            
            # Bisection
            for _ in range(50):
                d = (lo_d + hi_d) / 2.0
                vals = a_diff * d + c_vec
                max_val = vals.max()
                active = vals >= max_val - 1e-10
                sg = a_diff[active].mean()
                if sg > 1e-10:
                    hi_d = d
                elif sg < -1e-10:
                    lo_d = d
                else:
                    break
            
            d_opt = (lo_d + hi_d) / 2.0
            
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                c_vec += a_diff * d_opt
                improved = True
        
        h = np.clip(h, 0.0, 1.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
