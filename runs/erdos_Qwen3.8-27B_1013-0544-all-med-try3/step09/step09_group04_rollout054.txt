import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:

            nv = n - kk

    return grad

def de_evaluate(trial, n):
    trial = project_feasible(trial, n)
    return compute_c5(trial, n)

def de_mutation(pop, best_idx, i, n, F, rng):
    NP = len(pop)
    while True:
        a = rng.randint(0, NP - 1)
        b = rng.randint(0, NP - 1)
        c = rng.randint(0, NP - 1)
        if a != b and a != c and b != c and a != i and b != i and c != i:
            break
    mutant = pop[best_idx] + F * (pop[a] - pop[b])
    return mutant

def run_de(h_init, n, deadline, pop_size=30, max_gens=500, F=0.8, CR=0.85, rng=None):
    if rng is None:
        rng = random.Random(42)
    
    # Initialize population
    pop = [h_init.copy() for _ in range(pop_size)]
    # Diversify
    for i in range(1, pop_size):
        noise_scale = 0.1 + 0.3 * (i / pop_size)
        pop[i] = h_init + noise_scale * (rng.random() * 2 - 1) * np.ones(n)
        # Add some structured noise
        freq = rng.randint(2, 10)
        phase = rng.uniform(0, 2 * np.pi)
        pop[i] += 0.1 * np.sin(freq * np.pi * np.arange(n) / n * 2 + phase)
        pop[i] = project_feasible(pop[i], n)
    
    pop_obj = [compute_c5(pop[i], n) for i in range(pop_size)]
    best_c = min(pop_obj)
    best_h = pop[np.argmin(pop_obj)].copy()
    
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        
        best_idx = np.argmin(pop_obj)
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            
            # Mutation
            mutant = de_mutation(pop, best_idx, i, n, F, rng)
            
            # Crossover
            j_rand = rng.randint(0, n - 1)
            trial = np.zeros(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            
            trial = project_feasible(trial, n)
            obj = compute_c5(trial, n)
            
            if obj <= pop_obj[i]:
                pop[i] = trial
                pop_obj[i] = obj
                if obj < best_c:
                    best_c = obj
                    best_h = trial.copy()
        
        # Adaptive: increase CR slightly over generations
        CR = min(0.95, CR + 0.001)
    
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = max(100, maxiters[i])
        else:
            maxit = max(200, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):  # Step by 2 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 1):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.02, 0.01)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def de_diff_to_vec(a, b):
    return a - b

def run_de(h_init, n, deadline, pop_size, max_gens, F_start, F_end, CR, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    pop_obj = np.zeros(pop_size)
    pop_obj[0] = compute_c5(pop[0], n)

    for i in range(1, pop_size):
        scale = 0.05 + 0.25 * (i / float(pop_size))
        noise = (rng.random(n) * 2.0 - 1.0) * scale
        freq = rng.randint(2, 12)
        phase = rng.uniform(0, 6.2832)
        noise += 0.08 * np.sin(freq * np.pi * np.arange(n) / n * 4.0 + phase)
        pop[i] = project_feasible(h_init + noise, n)
        pop_obj[i] = compute_c5(pop[i], n)

    best_c = float(pop_obj.min())
    best_h = pop[np.argmin(pop_obj)].copy()

    n_indices = np.arange(n)
    for gen in range(max_gens):
        if time.time() > deadline - 2:
            break
        F = F_start + (F_end - F_start) * (gen / float(max(1, max_gens)))
        best_idx = int(np.argmin(pop_obj))

        for i in range(pop_size):
            if time.time() > deadline - 1:
                break
            tries = 0
            while tries < 20:
                a = rng.randint(0, pop_size - 1)
                b = rng.randint(0, pop_size - 1)
                c = rng.randint(0, pop_size - 1)
                if a != b and a != c and b != c and a != i and b != i and c != i:
                    break
                tries += 1
            if tries >= 20:
                continue
            mutant = pop[best_idx] + F * (pop[a] - pop[b])
            mask = np.array([rng.random() < CR for _ in range(n)])
            j_rand = rng.randint(0, n - 1)
            mask[j_rand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_feasible(trial, n)
            obj = compute_c5(trial, n)
            if obj <= pop_obj[i]:
                pop[i] = trial
                pop_obj[i] = obj
                if obj < best_c:
                    best_c = obj
                    best_h = trial.copy()
    return best_h, best_c, pop, pop_obj

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = max(100, maxiters[i])
        else:
            maxit = max(150, int(remaining * 2.5))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-22,
                                    'gtol': 1e-18, 'maxfun': maxit * 10,
                                    'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(80):
        remaining = deadline - time.time()
        if remaining < 12:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            maxit = max(80, int(rem2 * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                               args=(n, active, alpha, 0.002), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-22,
                                        'gtol': 1e-18, 'maxfun': maxit * 10,
                                        'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.015, 0.01)
        else:
            eta *= 0.96
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 1):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

                best_c = c4
                best_h = h4
                best_n = n_high
            if time.time() < deadline - 100:
                p4b_end = min(time.time() + 60, deadline - 80)
                h4b, c4b = run_active_opt(h4, best_n, p4b_end, top_k=5,
                                          alphas=[200000, 1000000, 5000000, 20000000])
                if c4b < best_c:
                    best_c = c4b
                    best_h = h4b

    # Phase 5: Multi-start with structured perturbation
    if time.time() < deadline - 120:
        p5_end = min(time.time() + 50, deadline - 80)
        for pert_i in range(5):
            if time.time() > p5_end - 10:
                break
            h_pert = structured_perturb(best_h, best_n, 0.03, rng)
            h_p5, c_p5 = run_lbfgs(h_pert, best_n, p5_end,
                                   alphas=[1000, 10000, 100000, 1000000],
                                   mu=0.005, maxiters=[200, 300, 400, 500])
            if c_p5 < best_c:
                best_c = c_p5
                best_h = h_p5

    # Phase 6: Subgradient refinement
    if time.time() < deadline - 40:
        p6_end = min(time.time() + 30, deadline - 20)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=10000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Pair search
    if time.time() < deadline - 15:
        p7_end = deadline - 6
        h7, c7 = pair_search(best_h, best_n, p7_end, radius=25)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final active set push
    if time.time() < deadline - 3:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 10000000, 0.001),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 300, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 800})
            h_fin = project_feasible(res.x, best_n)
            c_fin = compute_c5(h_fin, best_n)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin
        except Exception:
            pass

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
