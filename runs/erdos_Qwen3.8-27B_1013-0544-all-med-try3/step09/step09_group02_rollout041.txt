def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    b = (h >= 0.5).astype(float)
    target = n // 2
    cnt = int(b.sum())
    if cnt > target:
        idx = np.where(b == 1.0)[0]
        order = np.argsort(-h[idx])
        remove = idx[order[:cnt - target]]
        b[remove] = 0.0
    elif cnt < target:
        idx = np.where(b == 0.0)[0]
        order = np.argsort(h[idx])
        add = idx[order[:target - cnt]]
        b[add] = 1.0
    return b


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_pos_neg(h):
    n = int(h.size)
    pos = np.empty(n, dtype=float)
    neg = np.empty(n, dtype=float)
    inv = 1.0 - h
    pos[0] = float(np.dot(h, inv))
    for k in range(1, n):
        m = n - k
        pos[k] = np.dot(h[:m], inv[k:])
    neg[0] = pos[0]
    for k in range(1, n):
        m = n - k
        neg[k] = np.dot(h[k:], inv[:m])
    return pos, neg


def max_raw(pos, neg):

def make_vals(pos, neg):

def lag_gradient_pos(k, h):
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:

        g[:m] += 1.0 - h[k:]
        g[k:] -= h[:m]
    return g


def lag_gradient_neg(k, h):
    n = int(h.size)
    g = np.zeros(n, dtype=float)

def apply_single(pos, neg, h, i, d):
    if d == 0.0:
        return
    n = int(h.size)
    old = h[i]

    pos[0] += d * (1.0 - 2.0 * old) - d * d
    neg[0] += d * (1.0 - 2.0 * old) - d * d

    r1 = n - 1 - i
    if r1 >= 1:
        pos[1:r1 + 1] += d * (1.0 - h[i + 1:i + r1 + 1])
    if i >= 1:
        pos[1:i + 1] -= d * h[:i][::-1]

    if i >= 1:
        neg[1:i + 1] += d * (1.0 - h[:i][::-1])
    if r1 >= 1:
        neg[1:r1 + 1] -= d * h[i + 1:i + r1 + 1]


def apply_move(pos, neg, h, inc, dec, s):
    if s <= 0.0 or inc == dec:
        return
    apply_single(pos, neg, h, inc, s)
    apply_single(pos, neg, h, dec, -s)
    d = abs(int(inc) - int(dec))
    if d > 0:
        cs = s * s
        pos[d] += cs
        neg[d] += cs
    h[inc] += s
    h[dec] -= s


def active_gradient(h, pos, neg):
    n = int(h.size)
    mpos = float(pos.max())
    mneg = float(neg.max())
    m = max(mpos, mneg)
    tol = max(1e-9, 5e-6 * m)
    g = np.zeros(n, dtype=float)
    cnt = 0

    idx = np.where(pos >= m - tol)[0]
    if idx.size > 3:
        order = np.argsort(-pos[idx])
        idx = idx[order[:3]]
    for k in idx:
        g += lag_gradient_pos(int(k), h)
        cnt += 1

    idx = np.where(neg >= m - tol)[0]
    if idx.size > 3:
        order = np.argsort(-neg[idx])
        idx = idx[order[:3]]
    for k in idx:
        k = int(k)
        if k == 0:
            continue
        g += lag_gradient_neg(k, h)
        cnt += 1

    if cnt == 0:
        if mpos >= mneg:
            g = lag_gradient_pos(int(np.argmax(pos)), h)
        else:
            k = int(np.argmax(neg))
            if k == 0:
                g = lag_gradient_pos(0, h)
            else:
                g = lag_gradient_neg(k, h)
    else:
        g /= float(cnt)
    return g

    x_new = np.linspace(0.0, 2.0, n)
    return np.interp(x_new, x_old, np.asarray(h, dtype=float))


def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    b = (rng.rand(n) < 0.5).astype(float)
    return make_binary(b, n)


def beatty_binary(n, alpha, phase):
    x = np.arange(n, dtype=float)
    vals = (alpha * x + phase) % 1.0
    b = (vals < 0.5).astype(float)
    return make_binary(b, n)


def binary_sa(h, deadline, seed=1):
    h = make_binary(project_feasible(np.asarray(h, dtype=float).copy(), len(h)), len(h))
    n = int(h.size)
    pos, neg = compute_pos_neg(h)
    cur_raw = max_raw(pos, neg)
    best_raw = cur_raw
    best_h = h.copy()
    rng = np.random.RandomState(seed)

    T0 = max(0.1, 0.02 * best_raw)
    T = T0
    moves = 0
    stale = 0

    while time.time() < deadline - 1.0:
        valid = False
        dec = -1
        inc = -1
        for _ in range(10):
            i = int(rng.randint(0, n))
            j = int(rng.randint(0, n))
            if i != j and h[i] > 0.5 and h[j] < 0.5:
                dec = i
                inc = j
                valid = True
                break
        if not valid:
            continue

        apply_move(pos, neg, h, inc, dec, 1.0)
        new_raw = max_raw(pos, neg)
        moves += 1
        accept = False

        if new_raw < cur_raw - 1e-12:
            accept = True
        elif abs(new_raw - cur_raw) <= 1e-12:
            accept = True
        else:
            delta = new_raw - cur_raw
            if T > 1e-9 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:
                best_raw = new_raw
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        else:
            apply_move(pos, neg, h, dec, inc, 1.0)
            stale += 1

        if moves % 1000 == 0:
            T = max(1e-3, T0 * (0.995 ** (moves // 1000)))
        if moves % 10000 == 0:
            pos, neg = compute_pos_neg(h)
            cur_raw = max_raw(pos, neg)
            if cur_raw < best_raw - 1e-12:
                best_raw = cur_raw
                best_h = h.copy()

        if stale > 60000 and T < 1e-2:
            break

    return best_h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s) - 10.0)
    deadline = t0 + float(budget_s) - 3.0
    np.random.seed(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = int(len(h_parent))
    except Exception:
        h_parent = None
        n_parent = 0

    if h_parent is None or n_parent < 4:
        h_parent = random_binary(512, seed)
        n_parent = 512

    if n_parent % 2 != 0:
        h_parent = upsample(h_parent, n_parent + 1)
        n_parent = int(len(h_parent))

    h_parent = project_feasible(h_parent, n_parent)
    h_test, c_test = exact_evaluate(h_parent, n_parent)
    best_h = h_test
    best_c = c_test
    best_n = n_parent

    a_end = min(t0 + 0.20 * total, deadline - 300.0)
    if a_end > time.time() + 10.0:
        hA, rawA = local_search(best_h, a_end, seed=seed + 1)
        hA, cA = exact_evaluate(hA, len(hA))
        if cA < best_c:
            best_c = cA
            best_h = hA
            best_n = len(hA)

    n_main = 1024
    b_end = min(t0 + 0.60 * total, deadline - 200.0)
    if b_end > time.time() + 30.0:
        hB = upsample(best_h, n_main)
        hB, _ = exact_evaluate(hB, n_main)
        hB, rawB = local_search(hB, b_end, seed=seed + 2)
        hB, cB = exact_evaluate(hB, n_main)
        if cB < best_c:
            best_c = cB
            best_h = hB
            best_n = n_main

    c_end = min(t0 + 0.78 * total, deadline - 120.0)
    if c_end > time.time() + 30.0:
        nC = best_n if best_n <= 1024 else 1024
        hC = upsample(best_h, nC)
        hC, _ = exact_evaluate(hC, nC)
        hC, cC = smooth_lbfgs(hC, nC, c_end)
        hC, cC = exact_evaluate(hC, nC)
        if cC < best_c:
            best_c = cC
            best_h = hC
            best_n = nC

    d_end = min(t0 + 0.88 * total, deadline - 60.0)
    if d_end > time.time() + 20.0:
        hD, rawD = local_search(best_h, d_end, seed=seed + 3, start_step=4)
        hD, cD = exact_evaluate(hD, len(hD))
        if cD < best_c:
            best_c = cD
            best_h = hD
            best_n = len(hD)

    if time.time() < deadline - 80.0:
        nE = 512
        hE = upsample(best_h, nE)
        hE, _ = exact_evaluate(hE, nE)
        hE = make_binary(hE, nE)
        e1_end = min(time.time() + 70.0, deadline - 70.0)
        if e1_end > time.time() + 10.0:
            hE_sa = binary_sa(hE, e1_end, seed=seed + 11)
            hE_sa, cE = exact_evaluate(hE_sa, nE)
            if cE < best_c:
                best_c = cE
                best_h = hE_sa
                best_n = nE
            e2_end = min(time.time() + 30.0, deadline - 40.0)
            if e2_end > time.time() + 5.0:
                hE_ls, rawE = local_search(hE_sa, e2_end, seed=seed + 12)
                hE_ls, cE = exact_evaluate(hE_ls, nE)
                if cE < best_c:
                    best_c = cE
                    best_h = hE_ls
                    best_n = nE

    if time.time() < deadline - 60.0:
        for alpha, phase in ((0.6180339887, 0.123), (0.3819660113, 0.717)):
            nBe = 512
            hBe = beatty_binary(nBe, alpha, phase)
            hBe, _ = exact_evaluate(hBe, nBe)
            be_end = min(time.time() + 35.0, deadline - 40.0)
            if be_end > time.time() + 10.0:
                hBe_ls, rawBe = local_search(hBe, be_end, seed=seed + 21)
                hBe_ls, cBe = exact_evaluate(hBe_ls, nBe)
                if cBe < best_c:
                    best_c = cBe
                    best_h = hBe_ls
                    best_n = nBe

    if time.time() < deadline - 15.0:
        hF, rawF = local_search(best_h, deadline - 2.0, seed=seed + 100, start_step=9)
        hF, cF = exact_evaluate(hF, len(hF))
        if cF < best_c:
            best_c = cF
            best_h = hF
            best_n = len(hF)

    best_h, best_c = exact_evaluate(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
