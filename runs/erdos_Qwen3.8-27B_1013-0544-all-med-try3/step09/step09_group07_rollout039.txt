    # Compute weighted gradient

The correlation at shift k and shift -k should be related. Specifically:
I(k) = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) * dx
I(-k) = sum_{i=k}^{n-1} h[i] * (1-h[i-k]) * dx = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j]) * dx

So I(k) and I(-k) are different in general (they count different things). I(k) counts "1 at position i, 0 at position i+k" while I(-k) counts "1 at position i+k, 0 at position i". Their sum is:
I(k) + I(-k) = sum_{i=0}^{n-1-k} [h[i](1-h[i+k]) + h[i+k](1-h[i])] * dx
= sum_{i=0}^{n-1-k} [h[i] + h[i+k] - 2*h[i]*h[i+k]] * dx

This isn't immediately useful, but it tells us that the "worst" lag could be positive or negative.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()


def compute_lag_gradient(lag, h, n):
    """Compute gradient of I(lag) with respect to h, without dx factor."""
    n_inner = n - abs(lag)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n_inner] += (1.0 - h[lag:lag+n_inner])
        grad[lag:] += (-h[:n_inner])
    else:
        idx = -lag
        grad[:n_inner] += (1.0 - h[idx:idx+n_inner])
        grad[idx:] += (-h[:n_inner])
    return grad


def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        grad += w[k] * compute_lag_gradient(lag, h, n)
    return grad * dx


def level_set_obj_grad(h, n, T):
    """Objective and gradient for level-set method: sum_k max(0, I_k - T)^2"""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    excess = np.maximum(0.0, c - T)
    obj = float(np.sum(excess ** 2))
    active = np.where(excess > 1e-15)[0]
    grad = np.zeros(n)
    for k_idx in active:
        w = 2.0 * excess[k_idx]
        lag = k_idx - (n - 1)
        grad += w * compute_lag_gradient(lag, h, n)
    return obj * dx * dx, grad * dx


def subgradient_max(h, n, epsilon=0.002):
    """Compute a subgradient of max_k I_k(h) using lags within epsilon of max."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    M = c.max()
    mask = c >= M - epsilon
    count = mask.sum()
    if count == 0:
        count = 1
        mask = np.zeros_like(c, dtype=bool)
        mask[int(np.argmax(c))] = True
    grad = np.zeros(n)
    for k_idx in np.where(mask)[0]:
        lag = k_idx - (n - 1)
        grad += compute_lag_gradient(lag, h, n)
    grad /= count
    return grad * dx


def momentum_subgradient(h, n, deadline, lr0=0.002, momentum=0.9, max_iter=50000):
    """Projected subgradient method with momentum for max-k problem."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = lr0
    g = np.zeros(n)
    target_sum = n / 2.0
    no_improve = 0
    eps = 0.003

    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = subgradient_max(h, n, epsilon=eps)
        grad = grad - grad.mean()  # project to sum=0 direction
        g = momentum * g - lr * grad
        h_new = h + g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        elif c_new < compute_c5(h, n):
            no_improve += 1
        else:
            no_improve += 1
        h = h_new
        if no_improve > 500:
            lr *= 0.7
            g = np.zeros(n)
            no_improve = 0
            if lr < 1e-6:
                break
        # Adaptive epsilon
        if it % 5000 == 4999:
            eps *= 0.8

    return best_h, best_c


def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)


def lbfgsb_level_set(h, n, T, deadline, maxiter=3000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        obj_val, grad = level_set_obj_grad(x, n, T)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = grad + 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def sa_structured(h, n, deadline, rng, alpha_sa=100000, t_init=0.006,
                  t_min=0.00002, cool=0.9999):
    """SA with moves focused on top lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    no_improve = 0
    K = 5

    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        k_star = top_idx[rng.randint(K)]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue
        i = rng.randint(0, n - m)
        j = i + m

        # Determine direction: reduce overlap means if h[i]>h[j], decrease h[i] or increase h[j]
        if h[i] > h[j] + 1e-6:
            max_d = min(0.08, (h[i] - h[j]) * 0.3, h[i] * 0.4, (1.0 - h[j]) * 0.4)
            d = rng.uniform(0.001, max(0.002, max_d))
        elif h[j] > h[i] + 1e-6:
            max_d = min(0.08, (h[j] - h[i]) * 0.3, h[j] * 0.4, (1.0 - h[i]) * 0.4)
            d = -rng.uniform(0.001, max(0.002, max_d))
        else:
            T *= cool
            continue

        if abs(d) < 1e-9:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = c_new.max()

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            no_improve += 1

        T *= cool
        if no_improve > 20000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            no_improve = 0
            T = max(T, 0.003)

    return best_h, best_c

def binary_relax_and_optimize(h, n, deadline, rng):
    """Threshold to binary, optimize, then relax."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1

    h_opt = lbfgsb_optimize(h_bin, n, 50000, min(deadline, time.time() + 40), maxiter=2000)
    return h_opt

        ls_deadline2 = min(deadline - 5, time.time() + 60)
        h_opt, c_opt = exhaustive_local_search(best_h, n, ls_deadline2, rng,
                                               top_k=5, min_step=2e-5, max_step=0.001)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 11: Try higher resolution
    if time.time() < deadline - 20:
        n_high = 1000
        h_interp = interpolate_h(best_h, n, n_high)
        h_opt = lbfgsb_optimize(h_interp, n_high, 50000, time.time() + 15, maxiter=1500)
        c_opt = compute_c5(h_opt, n_high)
        if c_opt < best_c:
            h_opt2 = lbfgsb_optimize(h_opt, n_high, 5000000, time.time() + 10, maxiter=2000)
            c_opt2 = compute_c5(h_opt2, n_high)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
