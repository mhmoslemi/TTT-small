import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    """Compute all correlation values."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    """Compute C5 = max correlation."""
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = 1.0 - h
            room[~mask] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            if not mask.any():
                break
            tot = h.sum()
            if tot < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def grad_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = 2 * n - 1
    for mm in range(nn):
        wi = w[mm]
        if wi < 1e-16:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

    c_act = c[indices]
    m = c_act.max()

    if mu > 0:

    return float(val)

def grad_top(h, n, indices, alpha, mu):
    c = compute_c5_vec(h, n)
    c_act = c[indices]
    m = c_act.max()
    sh = alpha * (c_act - m)
    mx = sh.max()

    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, indices):

def sq_fun(h, n):
    return h.sum() - n / 2.0

def sq_jac(h, n):
    return np.ones(n)

def lbfgs_optimize(h, n, alpha, mu, bounds, maxiter, deadline):
    if time.time() > deadline - 3:
        return h, compute_c5(h, n)
    try:

        return h_new, c_new

def gradient_descent_refine(h, n, deadline, base_lr=0.005):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = base_lr
    improving = True
    stall = 0
    while improving and time.time() < deadline:
        improving = False
        for _ in range(200):

                return best_h, best_c
            c_vec = compute_c5_vec(best_h, n)
            m = c_vec.max()
            sh = 500000.0 * (c_vec - m)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            g = grad_full(best_h, n, 500000.0, 0.0)

            g_mean = g.mean()
            g_dir = g - g_mean  # preserves sum

            norm = np.sqrt(np.sum(g_dir**2))
            if norm < 1e-15:

                best_h = new_h
                improving = True
                stall = 0
            else:
                lr *= 0.7
                stall += 1
                if stall > 15:
                    return best_h, best_c
    return best_h, best_c
