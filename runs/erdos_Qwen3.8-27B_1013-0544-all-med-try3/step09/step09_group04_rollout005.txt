    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active = w > 1e-13
    idxs = np.where(active)[0]
    for mm in idxs:
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, deadline, alphas, mu=0.01, max_iters_list=None, bounds=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if max_iters_list is not None and i < len(max_iters_list):
            maxit = max_iters_list[i]
        else:
            maxit = max(100, int(remaining * 1.5))
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-24,
                                    'gtol': 1e-20, 'maxfun': maxit * 15,
                                    'maxls': 60, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_opt(h, n, deadline, top_k=5, alphas=None, mu=0.002):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    total_budget = deadline - time.time()
    per_round = total_budget / (len(alphas) * 3)
    for alpha in alphas:
        for _ in range(3):
            remaining = deadline - time.time()
            if remaining < 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                order = np.argsort(-c_vec[active])
                active = active[order[:top_k]]
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            maxit = max(80, int(min(remaining, per_round) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                               args=(n, active, alpha, mu), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-24,
                                        'gtol': 1e-20, 'maxfun': maxit * 12,
                                        'maxls': 50, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_perturbation(h, n, rng, mode=0, scale=0.05):
    """Create diverse perturbations of h."""
    h_new = h.copy()
    if mode == 0:
        # Gaussian noise
        h_new += rng.normal(0, scale, n)
    elif mode == 1:
        # Sinusoidal perturbation
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += scale * np.sin(2 * np.pi * freq * x + phase)
    elif mode == 2:
        # Block flip
        idx = rng.randint(0, n - 50)
        length = rng.randint(10, 50)
        h_new[idx:idx+length] = 1.0 - h_new[idx:idx+length]
    elif mode == 3:
        # Asymmetric perturbation (left half up, right half down)
        h_new[:n//2] += scale
        h_new[n//2:] -= scale
    elif mode == 4:
        # Cosine bump
        x = np.linspace(0, 2, n, endpoint=False)
        center = rng.uniform(0.2, 1.8)
        width = rng.uniform(0.1, 0.5)
        h_new += scale * np.exp(-0.5 * ((x - center) / width) ** 2)
    elif mode == 5:
        # Swap two blocks
        i1 = rng.randint(0, n - 40)
        i2 = rng.randint(0, n - 40)
        length = rng.randint(10, 30)
        if abs(i1 - i2) > length:
            h_new[i1:i1+length], h_new[i2:i2+length] = h_new[i2:i2+length].copy(), h_new[i1:i1+length].copy()
    return project_feasible(h_new, n)

def coordinate_descent(h, n, deadline, n_passes=5):
    """Simple coordinate descent: sweep over coordinates, greedily adjust."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    n_corr = 2 * n - 1
    dx = 2.0 / n
    target = n / 2.0
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for s in steps:
        for _ in range(n_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                # Try increasing h[i]
                if best_h[i] + s <= 1.0:
                    old = best_h[i]
                    best_h[i] = old + s
                    # Rebalance (only adjust one other coord)
                    # Actually just check bounds and integral approximately
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] = old
                # Try decreasing h[i]
                if best_h[i] - s >= 0.0:
                    old = best_h[i]
                    best_h[i] = old - s
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] = old
            best_h = project_feasible(best_h, n)
            if not improved:
                break
    return best_h, best_c

def multi_restart_opt(h, n, deadline, rng, n_restarts=8):
    """Multi-restart: perturb + L-BFGS for each restart."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    remaining = deadline - time.time()
    per_restart = remaining / (n_restarts + 1)
    
    # First: polish the current best
    t_end = time.time() + min(per_restart, 25)
    h0, c0 = lbfgs_run(best_h, n, t_end,
                        alphas=[1000, 10000, 100000], mu=0.005,
                        max_iters_list=[200, 300, 400])
    if c0 < best_c:
        best_c = c0
        best_h = h0
    
    # Then: diverse restarts
    for r in range(n_restarts):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        mode = r % 6
        scale = min(0.08, 0.03 + 0.01 * r)
        h_pert = make_perturbation(best_h, n, rng, mode=mode, scale=scale)
        t_end = time.time() + min(remaining - 5, per_restart * 0.9)
        h_new, c_new = lbfgs_run(h_pert, n, t_end,
                                  alphas=[500, 5000, 50000, 200000], mu=0.005,
                                  max_iters_list=[150, 200, 250, 300])
        # Also try active set on the perturbed result
        if c_new < best_c:
            remaining2 = deadline - time.time()
            if remaining2 > 15:
                h_act, c_act = active_set_opt(h_new, n, time.time() + min(remaining2*0.4, 15),
                                              top_k=4, alphas=[100000, 1000000, 5000000], mu=0.002)
                if c_act < c_new:
                    h_new, c_new = h_act, c_act
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.default_rng(seed)
    
    # Initial setup
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except (NameError, Exception):
        h0 = np.full(400, 0.5)
        n_orig = 400
    
    best_h = project_feasible(h0, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick polish at original resolution (15s)
    p1_end = min(t0 + 15, deadline - 900)
    h1, c1 = lbfgs_run(best_h, best_n, p1_end,
                        alphas=[500, 5000, 50000], mu=0.005,
                        max_iters_list=[150, 200, 250])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Upsample to n=1000, initial optimization (60s)
    n1 = 1000
    if time.time() < deadline - 600:
        h_u = upsample(best_h, best_n, n1)
        p2_end = min(time.time() + 60, deadline - 550)
        h2, c2 = lbfgs_run(h_u, n1, p2_end,
                            alphas=[50, 200, 1000, 5000, 20000], mu=0.005,
                            max_iters_list=[200, 300, 400, 400, 500])
        if c2 < best_c + 0.0005:
            best_c = min(best_c, c2)
            best_h = h2
            best_n = n1
    
    # Phase 3: Multi-restart at n=1000 (300s)
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 300, deadline - 300)
        best_h, best_c = multi_restart_opt(best_h, best_n, p3_end, rng, n_restarts=7)
    
    # Phase 4: Active set refinement at n=1000 (80s)
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 80, deadline - 180)
        h4, c4 = active_set_opt(best_h, best_n, p4_end, top_k=4,
                                 alphas=[200000, 1000000, 5000000, 20000000], mu=0.001)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Upsample to n=2000 (150s)
    n2 = 2000
    if time.time() < deadline - 150 and best_n < n2:
        h_u2 = upsample(best_h, best_n, n2)
        c_u2 = compute_c5(h_u2, n2)
        if c_u2 < best_c + 0.002:
            p5_end = min(time.time() + 120, deadline - 100)
            h5, c5 = lbfgs_run(h_u2, n2, p5_end,
                                alphas=[20, 100, 500, 2000, 10000, 50000], mu=0.003,
                                max_iters_list=[100, 200, 300, 300, 400, 500])
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n2
    
    # Phase 6: Active set at final resolution (80s)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 80, deadline - 40)
        h6, c6 = active_set_opt(best_h, best_n, p6_end, top_k=4,
                                 alphas=[500000, 2000000, 10000000, 50000000], mu=0.001)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Coordinate descent (40s)
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 40, deadline - 5)
        h7, c7 = coordinate_descent(best_h, best_n, p7_end, n_passes=3)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final active set push (15s)
    if time.time() < deadline - 3:
        p8_end = deadline - 1
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst, worst+1 if worst+1 < 2*best_n-1 else worst-1])
        active = active[active < 2*best_n-1]
        try:
            res = minimize(active_obj, best_h, jac=active_grad,
                           args=(best_n, active, 50000000, 0.0005),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 300, 'ftol': 1e-24,
                                    'gtol': 1e-20, 'maxfun': 1000, 'maxls': 40, 'maxcor': 40})
            h8 = project_feasible(res.x, best_n)
            c8 = compute_c5(h8, best_n)
            if c8 < best_c:
                best_c = c8
                best_h = h8
        except Exception:
            pass
    
    # Final: ensure feasibility and compute
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
