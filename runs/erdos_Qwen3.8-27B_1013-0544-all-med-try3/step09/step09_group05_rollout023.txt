import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, weights, center):
    """Compute gradient of weighted sum of correlations.
    weights: array of length 2n-1 (weights for each shift)
    center: n-1 (middle index of full correlation)
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if abs(w) < 1e-16:
            continue
        k = s - center
        if k >= 0:
            nv = n - k
            grad[0:nv] += w * dx * (1.0 - h[k:n])
            grad[k:n] += w * dx * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += w * dx * (1.0 - h[0:nv])
            grad[0:nv] += w * dx * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    """Smoothed max objective with log-sum-exp."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    """Gradient of smoothed max."""
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = compute_gradient(x, n, w, n - 1)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_obj(x, n, active, alpha, mu):
    """Objective focusing on active set of shifts."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(x, n, active, alpha, mu):
    """Gradient for active set objective."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w_active = np.exp(sh - mx)
    w_active /= w_active.sum()
    
    # Full weight vector (mostly zeros)
    w = np.zeros(2 * n - 1)
    for i, idx in enumerate(active):
        w[idx] = w_active[i]
    
    grad = compute_gradient(x, n, w, n - 1)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient_step(x, n, step_size):
    """One subgradient step on the actual max."""
    c = compute_c5_vec(x, n)
    m_idx = np.argmax(c)
    # Subgradient at the argmax
    grad = compute_gradient(x, n, np.zeros(2*n-1) + 0.0, n - 1)  # zeros
    # Actually, subgradient is the gradient of the max component
    grad = compute_gradient(x, n, np.eye(2*n-1)[:, m_idx] if False else (np.arange(2*n-1) == m_idx).astype(float), n - 1)
    # Project
    x_new = np.clip(x - step_size * grad, 0.0, 1.0)
    x_new = project_feasible(x_new, n)
    return x_new

def nesterov_smooth_max(x, n, alpha, mu, deadline, step_schedule='linear'):
    """Nesterov-accelerated gradient descent on smoothed max."""
    best_c = compute_c5(x, n)
    best_x = x.copy()
    
    # Initial step size
    f0 = smooth_max_obj(x, n, alpha, mu)
    g0 = smooth_max_grad(x, n, alpha, mu)
    g0_norm = np.linalg.norm(g0)
    if g0_norm < 1e-15:
        return best_x, best_c
    
    # Lipschitz constant estimate
    L = np.max(np.abs(compute_c5_vec(x, n))) + 1.0
    L = L * alpha * 4.0  # rough estimate
    lr = 1.0 / max(L, 1.0)
    
    y = x.copy()
    t_k = 1.0
    n_iter = 0
    
    while time.time() < deadline:
        n_iter += 1
        f_y = smooth_max_obj(y, n, alpha, mu)
        g_y = smooth_max_grad(y, n, alpha, mu)
        
        # Gradient step from y
        x_new = y - lr * g_y
        x_new = project_feasible(x_new, n)
        
        # Check improvement
        f_new = smooth_max_obj(x_new, n, alpha, mu)
        
        # Nesterov update
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_k * t_k))
        momentum = (t_k - 1) / t_new
        y = x_new + momentum * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        t_k = t_new
        
        # Update step size (halve if not improving)
        if f_new > f_y + 1e-16:
            lr *= 0.5
        else:
            lr = min(lr * 1.1, 1.0 / max(L * 0.1, 1e-10))
        
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_x = x_new.copy()
        
        x = x_new
    
    return best_x, best_c

def sa_optimize(h, n, deadline, T_init=0.05, T_min=0.001, cooling=0.995, max_iter=50000):
    """Simulated annealing for the C5 problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    current_c = best_c
    T = T_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Random perturbation: pick two indices, transfer mass
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        delta = np.random.uniform(0.005, 0.03)
        
        # Try moving mass from i to j
        if current_h[i] - delta >= 0 and current_h[j] + delta <= 1:
            current_h[i] -= delta
            current_h[j] += delta
            new_c = compute_c5(current_h, n)
            
            if new_c < current_c:
                current_c = new_c
                no_improve = 0
                if new_c < best_c:
                    best_c = new_c
                    best_h = current_h.copy()
            else:
                delta_c = new_c - current_c
                if np.random.random() < np.exp(-delta_c / max(T, 1e-10)):
                    current_c = new_c
                    no_improve += 1
                else:
                    current_h[i] += delta
                    current_h[j] -= delta
                    no_improve += 1
        
        # Also try direct swap
        idx = np.random.permutation(n)[:2]
        i2, j2 = idx[0], idx[1]
        if i2 != j2:
            swap_delta = np.random.uniform(-0.02, 0.02)
            if (current_h[i2] + swap_delta >= 0 and current_h[i2] + swap_delta <= 1 and
                current_h[j2] - swap_delta >= 0 and current_h[j2] - swap_delta <= 1):
                current_h[i2] += swap_delta
                current_h[j2] -= swap_delta
                new_c = compute_c5(current_h, n)
                
                if new_c < current_c:
                    current_c = new_c
                    no_improve = 0
                    if new_c < best_c:
                        best_c = new_c
                        best_h = current_h.copy()
                else:
                    delta_c = new_c - current_c
                    if np.random.random() < np.exp(-delta_c / max(T, 1e-10)):
                        current_c = new_c
                        no_improve += 1
                    else:
                        current_h[i2] -= swap_delta
                        current_h[j2] += swap_delta
                        no_improve += 1
        
        # Cool down
        T *= cooling
        if T < T_min:
            T = T_min
        
        # Reset if stuck
        if no_improve > 500:
            no_improve = 0
            T = max(T * 5, 0.01)
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_rounds=10000):
    """Coordinate descent with smart step sizes."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        # Random subset of coordinates for speed
        if n > 200:
            coords = np.random.choice(n, size=min(50, n), replace=False)
        else:
            coords = range(n)
        
        for i in coords:
            if time.time() > deadline:
                return best_h, best_c
            # Try to optimize h[i] alone (keeping sum constraint)
            # We need to adjust another coordinate to maintain sum
            # Instead, do a line search on h[i] with a partner
            j = (i + n // 3) % n
            
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
                # Increase i, decrease j
                if h[i] + step <= 1 and h[j] - step >= 0:
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] -= step
                        h[j] += step
                        break
                
                # Decrease i, increase j
                if h[i] - step >= 0 and h[j] + step <= 1:
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] += step
                        h[j] -= step
                        break
        
        if not improved and rnd > 200:
            break
    
    return best_h, best_c

def lbfgs_smooth(h, n, deadline, alphas=None, mus=None):
    """L-BFGS with multiple smoothing levels."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    if mus is None:
        mus = [1.0, 10.0, 100.0, 1000.0, 10000.0, 100000.0, 1000000.0]
    
    for a, mu in zip(alphas, mus):
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, a, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c

def active_optimization(h, n, deadline, top_k=10):

    best_c = compute_c5(h, n)

        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Find active set: shifts within small fraction of max
        threshold = max_val - max(0.001, 0.01 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        elif len(active) > top_k:

                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)

            except Exception:

def slqp_polish(h, n, deadline, alpha=500000):
    """SLSQP polish."""
    bounds = [(0.0, 1.0)] * n
    
    def slq_obj(x):
        return smooth_max_obj(x, n, alpha, 0.0)
    
    def slq_jac(x):
        return smooth_max_grad(x, n, alpha, 0.0)
    
    def eq_con(x):
        return x.sum() - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
    
    try:
        res = minimize(slq_obj, h, jac=slq_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_weighted(h, n, weights, center):
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if abs(w) < 1e-14:
            continue
        k = s - center
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[0:nv] += w * dx * (1.0 - h[k:k + nv])
            grad[k:k + nv] -= w * dx * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:kk + nv] += w * dx * (1.0 - h[0:nv])
            grad[0:nv] -= w * dx * h[kk:kk + nv]
    return grad

def smooth_max_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(sh)] = 1.0
    else:
        w /= w_sum
    grad = compute_gradient_weighted(x, n, w, n - 1)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_grad(x, n, active, alpha, mu):

def sa_optimize(h, n, deadline, T_init=0.03, T_min=0.0005, cooling=0.997):

        if i == j:

        delta = np.random.uniform(0.003, 0.025)

            cur[i] -= delta
            cur[j] += delta
            new_c = compute_c5(cur, n)

                if new_c < best_c:

        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    h += 0.15 * np.sin(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(300, 0.5)
        n_parent = 300

    best_h = project_feasible(h_parent, n_parent)
    best_c = compute_c5(best_h, n_parent)

    # Choose target resolution
    n_target = 700
    x_parent = np.linspace(0, 2, n_parent)
    x_target = np.linspace(0, 2, n_target)
    h_cur = np.interp(x_target, x_parent, best_h)
    h_cur = project_feasible(h_cur, n_target)
    c_cur = compute_c5(h_cur, n_target)

    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 1: Progressive LBFGS
    t1 = min(t0 + 120, deadline - 500)
    h_cur, c_cur = run_lbfgs(h_cur, n_target, t1,
        alphas=[50, 200, 1000, 5000, 20000],
        mus=[5.0, 20.0, 200.0, 2000.0, 20000.0],
        maxiter_list=[2000, 2000, 3000, 3000, 3000])
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 2: Active set optimization
    t2 = min(t0 + 280, deadline - 350)
    h_cur, c_cur = run_active(h_cur, n_target, t2, top_k=10,
        alpha_list=[50000, 200000, 1000000])
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 3: More LBFGS with higher alpha
    t3 = min(t0 + 420, deadline - 280)
    h_cur, c_cur = run_lbfgs(h_cur, n_target, t3,
        alphas=[50000, 200000, 1000000],
        mus=[50000.0, 200000.0, 1000000.0],
        maxiter_list=[2000, 2000, 2000])
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 4: SA exploration at medium resolution
    n_mid = 200
    x_mid = np.linspace(0, 2, n_mid)
    h_mid = np.interp(x_mid, x_target, h_cur)
    h_mid = project_feasible(h_mid, n_mid)
    # Also try binary init
    h_bin = binary_init(n_mid, seed + 7)
    h_struct = structured_init(n_mid, seed + 13)
    t4a = min(t0 + 520, deadline - 250)
    t4b = min(t0 + 590, deadline - 200)
    h_sa1, c_sa1 = sa_optimize(h_mid, n_mid, t4a, T_init=0.02, T_min=0.0005, cooling=0.998)
    h_sa2, c_sa2 = sa_optimize(h_bin, n_mid, t4b, T_init=0.05, T_min=0.001, cooling=0.997)
    h_sa3, c_sa3 = sa_optimize(h_struct, n_mid, min(t0 + 650, deadline - 150), T_init=0.04, T_min=0.001, cooling=0.997)

    # Upsample best coarse
    candidates = [(h_sa1, c_sa1), (h_sa2, c_sa2), (h_sa3, c_sa3)]
    best_mid = min(candidates, key=lambda x: x[1])
    h_up = np.interp(x_target, x_mid, best_mid[0])
    h_up = project_feasible(h_up, n_target)
    c_up = compute_c5(h_up, n_target)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up.copy()
        h_cur = h_up

    # Phase 5: LBFGS from potentially new solution
    t5 = min(t0 + 750, deadline - 180)
    h_cur, c_cur = run_lbfgs(h_cur, n_target, t5,
        alphas=[50000, 200000, 500000, 2000000],
        mus=[50000.0, 200000.0, 500000.0, 2000000.0],
        maxiter_list=[2000, 2000, 2000, 2000])
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 6: Active set refinement
    t6 = min(t0 + 850, deadline - 120)
    h_cur, c_cur = run_active(h_cur, n_target, t6, top_k=5,
        alpha_list=[200000, 1000000, 5000000])
    if c_cur < best_c:
        best_c = c_cur
        best_h = h_cur.copy()

    # Phase 7: Nesterov
    if time.time() < deadline - 80:
        t7 = min(t0 + 920, deadline - 80)
        h_n, c_n = nesterov_descend(h_cur, n_target, 500000, 50000.0, t7)
        if c_n < best_c:
            best_c = c_n
            best_h = h_n.copy()
            h_cur = h_n

    # Phase 8: Multi-direction pair search
    if time.time() < deadline - 40:
        t8 = min(t0 + 980, deadline - 40)
        h_cur, c_cur = multi_pair_search(h_cur, n_target, t8, n_trials=8000)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()

    # Phase 9: SLSQP polish
    if time.time() < deadline - 10:
        t9 = deadline - 5
        h_s, c_s = slqp_polish(h_cur, n_target, t9, alpha=2000000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()

    # Phase 10: Try at n=500 as well
    if time.time() < deadline - 3:
        h_500 = np.interp(x_parent, x_target, best_h)
        h_500 = project_feasible(h_500, n_parent)
        c_500 = compute_c5(h_500, n_parent)
        if c_500 < best_c:
            best_c = c_500
            best_h = h_500
            n_target = n_parent

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
