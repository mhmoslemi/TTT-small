def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

def coord_descent_1d(h, n, i, c_vec):
    """Find optimal h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute derivatives B_k = df_k/dh[i] for each shift
    B = np.zeros(2 * n - 1)
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        # h[i] appears in f_k with coefficient:
        # (1-h[i+k])*dx if 0 <= i+k < n, minus h[i-k]*dx if 0 <= i-k < n
        coeff = 0.0
        if 0 <= i + k < n:
            coeff += (1.0 - h[i + k]) * dx
        if 0 <= i - k < n:
            coeff += (-h[i - k]) * dx
        B[mm] = coeff
    # Current f_k values minus the linear part from h[i]
    A = c_vec - h[i] * B
    # Minimize max_k (A_k + t * B_k) for t in [0,1]
    # Convex piecewise linear; use subgradient binary search
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = A + mid * B
        g = vals.max()
        active_mask = vals >= g - 1e-14
        if not active_mask.any():
            active_mask = np.array([np.argmax(vals)])
        sg = B[active_mask].mean()
        if sg > 1e-12:
            hi = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    t_opt = 0.5 * (lo + hi)
    t_opt = np.clip(t_opt, 0.0, 1.0)
    return t_opt

                              mu=0.008, maxiters=[800, 1000, 1200, 1500, 1800])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Progressive resolution increase with L-BFGS-B
    resolutions = [1000, 1500, 2000, 2500, 3000]
    for nt in resolutions:
        remaining = deadline - time.time()
        if remaining < 100:
            break
        if nt <= best_n:
            continue

        # Allocate time for this resolution
        time_alloc = min(remaining * 0.3, 180)
        p_end = time.time() + time_alloc

        h_up = upsample(best_h, best_n, nt)
        c_up = compute_c5(h_up, nt)

        # L-BFGS-B with increasing alpha
        n_alphas = min(8, max(3, int(time_alloc / 20)))
        alphas_list = [10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000][:n_alphas]
        iters_list = [400, 600, 800, 1000, 1200, 1500, 1800, 2000, 2500, 3000][:n_alphas]

        h_new, c_new = lbfgs_polish(h_up, nt, min(p_end, deadline - 80),
                                     alphas=alphas_list, mu=0.003, maxiters=iters_list)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            best_n = nt

    # Phase 4: Active set optimization at best resolution
    remaining = deadline - time.time()
    if remaining > 80:
        p4_end = min(time.time() + remaining * 0.25, deadline - 60)
        h4, c4 = active_set_opt(best_h, best_n, p4_end, top_k=5,
                                 alphas=[50000, 200000, 1000000, 5000000, 20000000, 100000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Single constraint dive on worst shift
    remaining = deadline - time.time()
    if remaining > 40:
        p5_end = min(time.time() + remaining * 0.2, deadline - 30)
        h5, c5 = single_constraint_dive(best_h, best_n, p5_end, alpha=50000000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Coordinate descent
    remaining = deadline - time.time()
    if remaining > 20:
        p6_end = min(time.time() + remaining * 0.4, deadline - 10)
        h6, c6 = coord_descent_pass(best_h, best_n, p6_end, max_passes=10)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Another active set round + SLSQP
    remaining = deadline - time.time()
    if remaining > 15:
        p7_end = min(time.time() + remaining * 0.4, deadline - 8)
        h7, c7 = active_set_opt(best_h, best_n, p7_end, top_k=3,
                                 alphas=[500000, 2000000, 10000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final SLSQP
    remaining = deadline - time.time()
    if remaining > 12:
        p8_end = min(time.time() + remaining * 0.5, deadline - 3)
        h8, c8 = slqp_polish(best_h, best_n, p8_end, alpha=2000000, maxiter=800)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: single worst-shift push
    if time.time() < deadline - 2:
        c_vec = compute_c5_vec(best_h, best_n)
        worst = np.argmax(c_vec)
        active = np.array([worst])
        try:
            res = minimize(act_obj, best_h, jac=act_grad,
                           args=(best_n, active, 100000000, 0.0005),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 300, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': 800})
            h_final = project_feasible(res.x, best_n)
            c_final = compute_c5(h_final, best_n)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
        except Exception:
            pass

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
