    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx

    top_idx = np.argsort(c)[::-1][:top_k]
    # Compute softmax weights (only over these top_k)
    c_top = c[top_idx]
    alpha_c = alpha * (c_top - c_top.max())

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_gradient(h, n, top_k=20, alpha=500):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    cm = c_top.max()
    a = alpha * (c_top - cm)
    a -= a.max()
    w = np.exp(a)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= wsum
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = top_idx[idx]
        lag = k - (n - 1)
        m = abs(lag)
        if m == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - m] += wi * dx * (1.0 - h[m:])
            grad[m:] -= wi * dx * h[:n - m]
        else:
            grad[m:] += wi * dx * (1.0 - h[:n - m])
            grad[:n - m] -= wi * dx * h[m:]
    return grad

def gradient_descent(h, n, deadline, lr_init=1.0, top_k_init=10, top_k_max=40,
                     patience=3000, decay=0.9):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    top_k = top_k_init
    stall = 0
    v = np.zeros(n)
    momentum = 0.85
    
    for it in range(500000):
        if time.time() > deadline:
            break
        g = fast_gradient(h, n, top_k, alpha=800)
        v = momentum * v + lr * g
        h_new = project_feasible(h - v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
            lr *= 1.005
        else:
            stall += 1
            if stall > patience:
                lr *= decay
                v *= 0.5
                stall = 0
        h = h_new
        top_k = min(top_k_max, top_k_init + it // 3000)
        if lr < 0.01:
            break
    return best_h, best_c

def targeted_local_search(h, n, deadline, max_rounds=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        top_lags = np.argsort(c_all)[::-1][1:]  # skip lag 0 (middle)
        improved = False
        for k_star in top_lags[:3]:

                break
            lag = k_star - (n - 1)
            m = abs(lag)

            steps = [0.01, 0.005, 0.002, 0.001]

def fourier_perturb(h, n, scale, rng):
    """Add structured Fourier perturbation preserving sum and bounds."""
    h_new = h.copy()
    f = np.fft.rfft(h_new - 0.5)
    # Perturb high frequencies (skip DC)
    n_freq = len(f)
    if n_freq > 3:
        f[1:] += scale * (rng.randn(n_freq - 1) + 1j * rng.randn(n_freq - 1))
        h_new = np.fft.irfft(f, n=n)
    h_new = project_feasible(h_new, n)
    return h_new

def pair_swap_search(h, n, deadline, rng, max_iters=50000):
    """Randomized pair-swap search targeting worst lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_lags = np.argsort(c_all)[::-1][1:4]
        k_star = top_lags[rng.randint(len(top_lags))]
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0:
            continue
        i = rng.randint(0, n - m)
        j = i + m
        if abs(h[i] - h[j]) < 1e-8:
            continue
        # Move from larger to smaller
        if h[i] > h[j]:
            delta = min(0.03 * rng.random() + 0.002, (h[i] - h[j]) * 0.4)
            h[i] -= delta
            h[j] += delta
        else:
            delta = min(0.03 * rng.random() + 0.002, (h[j] - h[i]) * 0.4)
            h[i] += delta
            h[j] -= delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        else:
            # Revert
            if h[i] > h[j]:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
            h = project_feasible(h, n)
    return best_h, best_c

def slsqp_polish(h, n, maxiter=2000, top_m=5):
    """SLSQP on smooth max of top-m lags."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[::-1][:top_m]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        a = 5000 * (active - active.max())
        a -= a.max()
        val = active.max() + np.log(np.sum(np.exp(a))) / 5000
        return val
    
    def gfun(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        active = cc[top_idx]
        a = 5000 * (active - active.max())
        a -= a.max()
        w = np.exp(a)
        w /= w.sum()
        grad = np.zeros(len(x))
        for idx in range(len(top_idx)):
            if w[idx] < 1e-10:
                continue
            k = top_idx[idx]
            lag = k - (n - 1)
            m = abs(lag)
            if m == 0:
                grad += w[idx] * dx * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n - m] += w[idx] * dx * (1.0 - x[m:])
                grad[m:] -= w[idx] * dx * x[:n - m]
            else:
                grad[m:] += w[idx] * dx * (1.0 - x[:n - m])
                grad[:n - m] -= w[idx] * dx * x[m:]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=gfun, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or (hasattr(res, 'fun') and res.fun < compute_c5(h, n)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

    """Full optimization pipeline for a single resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    elapsed_budget = deadline - time.time()
    
    # Phase 1: Fast gradient descent
    t1 = time.time() + min(120, elapsed_budget * 0.25)
    h, c = gradient_descent(h, n, t1, lr_init=1.5, top_k_init=10, top_k_max=30)
    if c < best_c:
        best_c = c
        best_h = h
    h = h  # continue from here
    
    # Phase 2: Finer gradient with more lags
    t2 = time.time() + min(150, elapsed_budget * 0.25)
    h2, c2 = gradient_descent(h, n, t2, lr_init=0.5, top_k_init=20, top_k_max=50,
                               patience=5000, decay=0.85)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    h = h2
    
    # Phase 3: Targeted local search
    t3 = min(time.time() + 120, deadline - 60)
    h3, c3 = targeted_local_search(h, n, t3)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Pair swap search
    if time.time() < deadline - 40:
        t4 = min(time.time() + 80, deadline - 30)
        h4, c4 = pair_swap_search(best_h, n, t4, rng)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: SLSQP polish (top-5)
    if time.time() < deadline - 20:
        h5 = slsqp_polish(best_h, n, maxiter=1500, top_m=5)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: SLSQP on single worst lag
    if time.time() < deadline - 10:
        h6 = slsqp_polish_top_lambda(best_h, n, maxiter=1000)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Final local search
    if time.time() < deadline - 2:
        h7, c7 = targeted_local_search(best_h, n, deadline - 1)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize at initial resolution
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    else:
        h_start = np.full(500, 0.5)
        alloc = min(300, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_single(h_start, 500, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase B: Try higher resolution (600)
    if time.time() < deadline - 200:
        n_hi = 600
        h_hip = interpolate_h(best_h, len(best_h), n_hi)
        alloc = min(300, (deadline - time.time()) * 0.35)
        if alloc > 80:
            h_opt, c_opt = optimize_single(h_hip, n_hi, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase C: Fourier-based multistart at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 180:
        remaining = deadline - time.time() - 60
        n_starts = max(2, int(remaining / 80))
        for r in range(min(n_starts, 8)):
            if time.time() > deadline - 80:
                break
            # Fourier perturbation
            scale = 0.015 * (1.0 + 0.2 * r)
            hp = fourier_perturb(best_h, n_best, scale, rng)
            c_test = compute_c5(hp, n_best)
            alloc = min(60, (deadline - time.time() - 50) * 0.3)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_single(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase D: Switch to different resolution if it helps
    if time.time() < deadline - 150:
        n_alt = 550 if n_best >= 600 else 650
        h_alt = interpolate_h(best_h, n_best, n_alt)
        alloc = min(200, (deadline - time.time()) * 0.3)
        if alloc > 60:
            h_opt, c_opt = optimize_single(h_alt, n_alt, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase E: Final intensive local search
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        # Final targeted local search with many rounds
        h_ls, c_ls = targeted_local_search(best_h, n, deadline - 8, max_rounds=500)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
        
        # Final SLSQP
        if time.time() < deadline - 5:
            h_sp = slsqp_polish(best_h, n, maxiter=1000, top_m=3)
            c_sp = compute_c5(h_sp, n)
            if c_sp < best_c:
                best_c = c_sp
                best_h = h_sp
        
        # One more local search
        best_h, best_c = targeted_local_search(best_h, n, deadline - 1, max_rounds=200)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(500, 0.5), 500)
        best_c = compute_c5(best_h, 500)
    
    return best_h, best_c, len(best_h)
