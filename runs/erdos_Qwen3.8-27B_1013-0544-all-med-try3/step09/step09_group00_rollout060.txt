    # Compute delta_c for all k
    # Term 1: -(1-h[i+k-(n-1)]) for k where 0 <= i+k-(n-1) < n
    # Term 2: +h[i-k+(n-1)] for k where 0 <= i-k+(n-1) < n
    # Term 3: +(1-h[j+k-(n-1)]) for k where 0 <= j+k-(n-1) < n

    k_hi = min(2*n-2, i+n-1)

        idx = np.arange(k_lo, k_hi+1)
        h_idx = i - idx + (n-1)  # = i-k+(n-1)
        delta[k_lo:k_hi+1] += h[h_idx]

    """Place target_ones ones in positions based on golden ratio."""
    h = np.zeros(n)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    # Beatty sequence: positions where floor(k*golden) - floor((k-1)*golden) = 1
    # Actually, use the characteristic word of slope 1/golden ≈ 0.618
    # But we need exactly target_ones = n/2 ones...
    
    # Alternative: use a balanced sequence construction
    # Place ones at positions floor(k * n / target_ones) for k = 0, 1, ..., target_ones-1
