            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def grad_at_lag(h, n, k_idx):
    """Gradient of overlap at lag index k_idx (0 to 2n-2)."""
    dx = 2.0 / n
    g = np.zeros(n)
    k = k_idx - (n - 1)
    if k >= 0:
        nv = n - k
        g[0:nv] += (1.0 - h[k:n]) * dx
        g[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += (1.0 - h[0:nv]) * dx
        g[0:nv] += (-h[kk:n]) * dx
    return g

def momentum_optimize(h, n, deadline, initial_step=0.01, momentum=0.9):
    """Projected momentum gradient descent on max-overlap."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    step = initial_step
    step_history = []
    top_k = 5
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        
        # Adaptive top_k: start small, grow
        if c_max < 0.39:
            top_k = min(20, 5 + int((0.40 - c_max) * 200))
        else:
            top_k = 5
        
        # Get active lags
        threshold = c_max - max(1e-10, 0.001 * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        
        # Compute averaged gradient over active lags
        grad = np.zeros(n)
        for idx in active:
            grad += grad_at_lag(h, n, idx)
        grad /= len(active)
        
        # Momentum update
        v = momentum * v - step * grad
        
        # Try the step
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        if c_new < c_max - 1e-12:
            # Accept step
            h = h_new
            c_hist = c_max - c_new
            step_history.append(step)
            if len(step_history) > 10:
                step_history.pop(0)
            # Geometric mean of recent steps that worked
            if len(step_history) >= 2:
                new_step = np.exp(np.mean(np.log(step_history)))
                step = step * 0.9 + new_step * 0.1  # smooth
            step = min(step * 1.05, 0.1)
            momentum = min(momentum * 1.001, 0.95)
        else:
            # Reject, reduce step
            step *= 0.7
            v = 0.5 * v  # damp momentum
    
    return best_h, best_c

def project_feasible(h, n):

        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()

        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        if mu_val > 0:
            val += mu_val * (x.sum() - n / 2.0) ** 2
        return float(val)

        c = compute_c5_vec(x, n)

def coordinate_descent(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try move: h[i] += d, h[j] -= d
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try move: h[i] -= d, h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=1000):

        h5, c5 = active_lbfgs(best_h, best_n, p5_end, top_k=5,
            alphas=[50000, 200000, 1000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: More momentum (80s)
    p6_end = min(time.time() + 80, deadline - 60)
    if p6_end > time.time() + 15:
        h6, c6 = momentum_optimize(best_h, best_n, p6_end,
                                   base_step=0.005, max_step=0.02)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish (40s)
    p7_end = min(time.time() + 40, deadline - 20)
    if p7_end > time.time() + 10:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 15:
                break
            h7, c7 = slqp_polish(best_h, best_n,
                                min(time.time() + 20, deadline - 10),
                                alpha=alpha, maxiter=500)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Coordinate descent polish (remaining time)
    p8_end = deadline - 2
    if p8_end > time.time() + 5:
        h8, c8 = coordinate_descent(best_h, best_n, p8_end,
                                   radius=40,
                                   deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
