            # Try increasing h[i]
            old = h[i]
            for delta in [0.01, 0.005, 0.001, 0.0001]:
                h[i] = min(1.0, old + delta)
                # Adjust another index to maintain sum
                diff = target - h.sum()
                if abs(diff) < 1e-10:
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        old = h[i]
                    else:
                        h[i] = old
                    break
                else:
                    # Find a tunable j
                    if diff < 0:
                        # Need to decrease some other h
                        candidates = np.where(h > 0.5)[0]
                    else:
                        candidates = np.where(h < 0.5)[0]
                    if len(candidates) == 0:
                        h[i] = old
                        break
                    j = candidates[i % len(candidates)]
                    if j == i:
                        h[i] = old
                        break
                    adj = min(diff, h[j]) if diff > 0 else max(diff, -h[j])
                    h[j] -= diff  # this adjusts to fix sum
                    if h[j] < 0:
                        h[j] = 0
                        h[i] = old
                        break
                    if h[j] > 1:
                        h[j] = 1
                        h[i] = old
                        break
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        old = h[i]
                    else:
                        h[i] = old
                        h[j] += diff
                    break
        if not improved and passe > 3:
            break
    
    return h, best_c

def random_restart_opt(h, n, deadline, rng, num_restarts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(num_restarts):
        if time.time() > deadline - 30:
            break
        # Perturb h
        noise = rng.randn(n) * 0.02
        h_pert = h + noise
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS from perturbed start
        for alpha in [5000, 50000]:
            if time.time() > deadline - 20:
                break
            h_new = run_lbfgs_smooth(h_pert, n, alpha, 1.0, maxiter=500,
                                      ftol=1e-14, gtol=1e-12)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h_pert = h_new
    
    return best_h, best_c

def multires_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Go to coarser, optimize, come back
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50:
            continue
        if time.time() > deadline - 20:
            break
        x_fine = np.linspace(0, 2, n)
        x_c = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_c, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        # Optimize at coarse level
        t_c = min(30, max(10, (deadline - time.time()) * 0.2))
        h_c = run_lbfgs_smooth(h_c, n_coarse, 10000, 10.0, maxiter=1000,
                               ftol=1e-15, gtol=1e-12)
        
        # Upsample back
        h_f = np.interp(x_fine, x_c, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial h from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, h_init)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS smooth optimization (increasing alpha)
    phase1_configs = [
        (50, 5.0, 2000),
        (200, 20.0, 2000),
        (1000, 100.0, 2500),
        (5000, 500.0, 2500),
        (20000, 2000.0, 2000),
        (100000, 10000.0, 1500),
    ]
    for alpha, mu, maxit in phase1_configs:
        if time.time() > deadline - 350:
            break
        h_new = run_lbfgs_smooth(h, n, alpha, mu, maxiter=maxit,
                                  ftol=1e-16, gtol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
    
    # Phase 2: Active-set optimization
    if time.time() < deadline - 200:
        t2 = min(time.time() + 120, deadline - 200)
        h, best_c = run_active_set_opt(h, n, t2, max_rounds=10)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Phase 3: Higher alpha L-BFGS polish
    if time.time() < deadline - 180:
        for alpha in [200000, 1000000, 5000000]:
            if time.time() > deadline - 160:
                break
            h_new = run_lbfgs_smooth(h, n, alpha, 50.0, maxiter=1000,
                                      ftol=1e-17, gtol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 4: Nesterov projected subgradient
    if time.time() < deadline - 150:
        t4 = min(time.time() + 60, deadline - 150)
        h_ns, c_ns = nesterov_projected_subgrad(h, n, t4, lr0=0.005,
                                                 momentum=0.85, num_iters=30000)
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns.copy()
            h = h_ns
    
    # Phase 5: Local pair search (most impactful for final improvement)
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h, best_c = local_pair_search(h, n, t5, target_steps=3000)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Phase 6: Another active-set round with tighter threshold
    if time.time() < deadline - 120:
        t6 = min(time.time() + 50, deadline - 120)
        h_act, c_act = run_active_set_opt(h, n, t6, max_rounds=6)
        if c_act < best_c:
            best_c = c_act
            best_h = h_act.copy()
            h = h_act
    
    # Phase 7: Multi-resolution refinement
    if time.time() < deadline - 80:
        t7 = min(time.time() + 40, deadline - 80)
        h_mr, c_mr = multires_refine(h, n, t7)
        if c_mr < best_c:
            best_c = c_mr
            best_h = h_mr.copy()
            h = h_mr
    
    # Phase 8: Random restarts
    if time.time() < deadline - 80:
        t8 = min(time.time() + 50, deadline - 80)
        h_rr, c_rr = random_restart_opt(h, n, t8, rng, num_restarts=4)
        if c_rr < best_c:
            best_c = c_rr
            best_h = h_rr.copy()
            h = h_rr
    
    # Phase 9: Final fine local search
    if time.time() < deadline - 30:
        t9 = min(time.time() + 25, deadline - 30)
        h, best_c = local_pair_search(h, n, t9, target_steps=1500)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Phase 10: Final SLSQP-style polish (using active set with very high alpha)
    if time.time() < deadline - 10:
        active = get_active_shifts(h, n, tol_frac=0.001, max_k=4)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(active_obj, h, jac=active_grad,
                          args=(n, np.array(active), 10000000, 0.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Final validation
    best_h = project_feasible(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h, n)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, best_c, n
