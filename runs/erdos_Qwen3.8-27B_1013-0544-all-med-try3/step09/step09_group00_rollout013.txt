    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_vec(c, alpha, x, n):
    """Vectorized smooth max gradient."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorized: for each shift index mm, k = mm - (n-1)
    for mm in range(len(c)):

        if k >= 0:

            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

    val = smooth_max_val(c, alpha)

        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    grad = smooth_max_grad_vec(c, alpha, x, n)
    if mu > 0:

    return grad

    smx = sh.max()
    if smx > 400:

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha

        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    vals = c[active_idx]
    mx = vals.max()

    for shake_i in range(4):
        if time.time() > p5_end - 10:
            break
        h_shake = random_shake(h_best, n, rng, n_shake=25 + shake_i * 5)
        # Quick active set on shaken solution
        h_sk, c_sk = active_set_phase(h_shake, n, min(time.time() + 25, p5_end),
                                       top_k=5, n_rounds=4, threshold_frac=0.003)
        if c_sk < c_best:
            c_best, h_best = c_sk, h_sk
        # Also try L-BFGS from shaken
        h_sk2, c_sk2 = lbfgs_phase(h_shake, n, min(time.time() + 15, p5_end),
                                     alphas=[5000, 50000, 500000],
                                     mus=[0.01, 0.005, 0.001],
                                     maxits=[500, 800, 1000])
        if c_sk2 < c_best:
            c_best, h_best = c_sk2, h_sk2

    # Phase 6: Second round active set + gradient descent
    p6_end = min(time.time() + 100, deadline - 100)
    h6, c6 = active_set_phase(h_best, n, min(time.time() + 50, p6_end),
                               top_k=8, n_rounds=10, threshold_frac=0.001)
    if c6 < c_best:
        c_best, h_best = c6, h6
    if time.time() < p6_end:
        h6b, c6b = grad_coord_desc(h_best, n, p6_end,
            step_sizes=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=20000)
        if c6b < c_best:
            c_best, h_best = c6b, h6b

    # Phase 7: Very fine pair search
    p7_end = min(time.time() + 60, deadline - 50)
    h7, c7 = fine_pair_search(h_best, n, p7_end, radius=30,
        deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
    if c7 < c_best:
        c_best, h_best = c7, h7

    # Phase 8: SLSQP polish
    p8_end = min(time.time() + 50, deadline - 20)
    for alpha_slq in [3000000, 10000000, 50000000]:
        if time.time() > p8_end:
            break
        h8, c8 = slqp_phase(h_best, n, p8_end, alpha=alpha_slq, maxiter=400)
        if c8 < c_best:
            c_best, h_best = c8, h8

    # Phase 9: Final very fine adjustments
    h9, c9 = fine_pair_search(h_best, n, deadline, radius=20,
        deltas=[0.00001, 0.000005, 0.000002])
    if c9 < c_best:
        c_best, h_best = c9, h9

    # Final gradient descent with tiny steps
    if time.time() < deadline - 3:
        h10, c10 = grad_coord_desc(h_best, n, deadline,
            step_sizes=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
            max_steps=10000)
        if c10 < c_best:
            c_best, h_best = c10, h10

    # Final projection and verification
    final_n = len(h_best)
    h_best = project_feasible(h_best, final_n)
    c_best = compute_c5(h_best, final_n)
    assert 0.0 <= h_best.min() and h_best.max() <= 1.0
    assert abs(h_best.sum() * (2.0 / final_n) - 1.0) < 1e-10

    return h_best, c_best, final_n
