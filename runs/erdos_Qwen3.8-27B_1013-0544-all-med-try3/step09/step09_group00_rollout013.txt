import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upscale(h_coarse, n_fine):
    """Upscale by repeating and averaging."""
    n_coarse = len(h_coarse)
    factor = n_fine // n_coarse
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * factor
        end = min(start + factor, n_fine)
        h_fine[start:end] = h_coarse[i]
    # Handle remainder
    for i in range(n_coarse * factor, n_fine):
        h_fine[i] = h_coarse[i % n_coarse]
    return project_feasible(h_fine, n_fine)

def c5_grad_at_active(c_vec, active, h, n):
    """Gradient of the mean of active correlation values."""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_active = len(active)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= n_active
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(c_vec, alpha, x, n, mu):
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c_vec)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_smooth(h, n, deadline, alphas, mu=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(300, int(remaining * 3))
        try:
            def obj(x):
                return smooth_max_obj(x, n, alpha, mu)
            def grad(x):
                return smooth_max_grad(compute_c5_vec(x, n), alpha, x, n, mu)
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coord_descent_pairs(h, n, deadline, max_rounds=100):

    for rnd in range(max_rounds):

        # Find best pair move

        best_gain = 0
        best_i, best_j = -1, -1

def fine_sweep(h, n, deadline):

    best_h = h.copy()

    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:

            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.005 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            grad = c5_grad_at_active(c_vec, active, best_h, n)
            # Try moving from high-grad to low-grad positions
            idx_sorted = np.argsort(np.abs(grad))
            for i in idx_sorted[:20]:
                if time.time() > deadline:
                    break
                for j in idx_sorted[-20:]:
                    if i == j:
                        continue
                    # Try increasing h[i] and decreasing h[j]
                    if grad[i] < 0 and best_h[i] < 1.0:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = np.clip(best_h, 0, 1)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upscale_h(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    h_fine = np.zeros(n_fine)
    ratio = n_fine / n_coarse
    for i in range(n_coarse):
        lo = int(i * ratio)
        hi = int((i + 1) * ratio)
        if hi > n_fine:
            hi = n_fine
        if lo < n_fine:
            h_fine[lo:hi] = h_coarse[i]
    return project_feasible(h_fine, n_fine)

def smax_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smax_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nw = len(c)
    for mm in range(nw):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topk_sm(c_sub, alpha):
    mx = c_sub.max()
    sh = alpha * (c_sub - mx)
    smx = sh.max()
    if smx > 400:
        return mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return mx + np.log(np.sum(np.exp(sh))) / alpha

def topk_w(c_sub, alpha):
    mx = c_sub.max()
    sh = alpha * (c_sub - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_for_active(c_vec, active_list, x, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    nact = len(active_list)
    for idx in active_list:
        k = idx - (n - 1)
        inv = dx / nact
        if k >= 0:
            nv = n - k
            grad[0:nv] += inv * (1.0 - x[k:n])
            grad[k:n] += inv * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += inv * (1.0 - x[0:nv])
            grad[0:nv] += inv * (-x[kk:n])
    return grad

def lbfgs_round(h, n, alpha, mu, maxit, deadline):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        v = smax_val(c, alpha)
        if mu > 0:
            v += mu * (x.sum() - n / 2.0) ** 2
        return v
    def grad(x):
        c = compute_c5_vec(x, n)
        g = smax_grad(c, alpha, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    rem = deadline - time.time()
    if rem < 5:
        return h, compute_c5(h, n)
    maxit = min(maxit, max(50, int(rem * 2)))
    try:

                      options={'maxiter': maxit, 'ftol': 1e-25, 'gtol': 1e-18,
                               'maxfun': maxit * 10, 'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)

def coord_descent(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = 0
    for step_size in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]:
        no_improve = 0
        while steps < max_iter and time.time() < deadline:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 12:
                active = np.argsort(-c_vec[active])[:12]
            grad = grad_for_active(c_vec, active, best_h, n)
            pos = np.where(grad > 1e-10)[0]
            neg = np.where(grad < -1e-10)[0]
            if len(pos) == 0 or len(neg) == 0:
                break
            best_gain = 0
            best_i, best_j = -1, -1
            p_top = pos[:25]
            n_top = neg[:25]
            for i in p_top:
                gi = grad[i]
                for j in n_top:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i, best_j = i, j
            if best_i < 0:
                break
            amt = min(step_size, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                no_improve = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                no_improve += 1
                if no_improve > 200:
                    break
    return best_h, best_c

def slqp_polish(h, n, deadline):
    bounds = [(0.0, 1.0)] * n
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    def sq_jac(x, nn):
        return np.ones(nn)
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [1000000, 5000000, 20000000]:
        if time.time() > deadline - 5:
            break
        def obj(x):
            c = compute_c5_vec(x, n)

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

            pass
    return best_h, best_c
