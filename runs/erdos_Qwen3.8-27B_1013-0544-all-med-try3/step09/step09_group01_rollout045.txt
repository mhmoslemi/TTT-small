def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for mm in nz:
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxiter_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_list and i < len(maxiter_list):
            maxit = maxiter_list[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k=8):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50000, 200000, 1000000, 5000000]
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Find active set
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        progressed = False
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    progressed = True
            except:
                pass
        if not progressed:
            break
    return best_h, best_c

def binary_greedy(h, n, deadline):
    """Greedy bit-flip optimization."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    # First: try all single flips
    for round_idx in range(50):
        if time.time() > deadline:
            break
        improved = False
        # Try making 0s into 1s and 1s into 0s
        for i in range(n):
            if time.time() > deadline:
                break
            # Try flipping h[i]
            old_val = best_h[i]
            new_val = 1.0 - old_val
            if new_val < 0 or new_val > 1:
                continue
            # Check sum constraint
            new_sum = best_h.sum() + (new_val - old_val)
            if abs(new_sum - n/2.0) > 1e-10:
                continue
            best_h[i] = new_val
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                best_h[i] = old_val
        if not improved:
            break
    
    # Then: try pairs (one up, one down)
    for round_idx in range(20):
        if time.time() > deadline:
            break
        improved = False
        # Only try nearby pairs to save time
        radius = max(5, n // 50)
        for i in range(n):
            if time.time() > deadline:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                if best_h[i] < 1.0 and best_h[j] > 0.0:
                    d = 1.0 - best_h[i]
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
        if not improved:
            break
    
    return best_h, best_c

def local_coord_search(h, n, deadline, n_sweeps=100):
    """Gradient-directed coordinate descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c_vec = compute_c5_vec(best_h, n)
    active = np.where(c_vec >= c_vec.max() - 0.001 * c_vec.max())[0]
    
    eta = 0.01
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        # Recompute active set periodically
        if sweep % 10 == 0:
            c_vec = compute_c5_vec(best_h, n)
            active = np.where(c_vec >= c_vec.max() - 0.002 * c_vec.max())[0]
        
        # Compute approximate gradient (average over active)
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) 
                grad[k:n] += (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv])
                grad[0:nv] += (-best_h[kk:n])
        grad *= dx / len(active)
        
        # Coordinate descent: try moving each coordinate along -grad
        for i in range(n):
            if time.time() > deadline:
                break
            if abs(grad[i]) < 1e-6:
                continue
            direction = -1 if grad[i] > 0 else 1
            for step in [eta, eta*0.5, eta*0.1, eta*0.01]:
                new_val = best_h[i] + direction * step
                if new_val < 0 or new_val > 1:
                    continue
                # Need to maintain sum constraint: adjust another coordinate
                diff = best_h[i] - new_val
                if abs(diff) < 1e-15:
                    continue
                # Find a coordinate j to compensate
                # Use the one with opposite gradient or random
                candidates = np.where(np.sign(grad) != np.sign(grad[i]) | (np.abs(grad) < 1e-6))[0]
                candidates = candidates[candidates != i]
                if len(candidates) == 0:
                    break
                # Pick the one that would most improve
                j = candidates[np.argmax(-np.abs(grad[candidates]))]
                adj = -diff
                new_j = best_h[j] + adj
                if new_j < 0 or new_j > 1:
                    # Try smaller step
                    continue
                best_h[i] = new_val
                best_h[j] = new_j
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    break
                else:
                    best_h[i] -= direction * step
                    best_h[j] = best_h[j]
            # Recompute gradient occasionally
            if i % 50 == 0:
                c_vec = compute_c5_vec(best_h, n)
                active = np.where(c_vec >= c_vec.max() - 0.002 * c_vec.max())[0]
                grad = np.zeros(n)
                for idx in active:
                    k = idx - (n - 1)
                    if k >= 0:
                        nv = n - k
                        grad[0:nv] += (1.0 - best_h[k:n])
                        grad[k:n] += (-best_h[0:nv])
                    else:
                        kk = -k
                        nv = n - kk
                        grad[kk:n] += (1.0 - best_h[0:nv])
                        grad[0:nv] += (-best_h[kk:n])
                grad *= dx / max(1, len(active))
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=200000):
    bounds = [(0.0, 1.0)] * n
    def sq_fun(x):
        return x.sum() - n / 2.0
    def sq_jac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 1000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def perturbation(h, n, rng, strength=0.05):
    """Create a perturbed version of h."""
    h_p = h.copy()
    # Add noise
    noise = rng.randn(n) * strength
    h_p += noise
    # Also try random swaps
    for _ in range(n // 10):
        i, j = rng.choice(n, 2, replace=False)
        if h_p[i] < 1 and h_p[j] > 0:
            d = min(0.1, 1 - h_p[i], h_p[j])
            h_p[i] += d
            h_p[j] -= d
    return project_feasible(h_p, n)
