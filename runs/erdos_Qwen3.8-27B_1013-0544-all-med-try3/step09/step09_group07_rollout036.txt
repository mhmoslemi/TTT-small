def project_feasible(h, n):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-13
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    center = n - 1

    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top_8 = np.argsort(c_all)[-8:]
        k_star = top_8[rng.randint(8)]
        lag = k_star - (n - 1)

        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue

        i = rng.randint(0, n - m)
        j = i + m

        if h[i] > h[j] + 1e-9:
            d = rng.uniform(0.001, 0.025)
            d = min(d, (h[i] - h[j]) * 0.3)
        elif h[j] > h[i] + 1e-9:
            d = -rng.uniform(0.001, 0.025)
            d = max(d, -(h[j] - h[i]) * 0.3)
        else:
            T *= cool
            continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)

        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            it_since_improve = 0
        else:
            it_since_improve += 1
            delta = c_new - best_c
            ratio = -delta / max(T, 1e-15)
            if ratio > 30 or rng.random() >= np.exp(-ratio):
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)

        T *= cool

        if it_since_improve > 40000:
            h = best_h + rng.normal(0, 0.004, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing alpha
    alpha_schedule = [5000, 50000, 500000]
    for alpha in alpha_schedule:
        remaining = deadline - time.time()
        if remaining < 60:
            break
        alloc = min(30, remaining * 0.08)
        h_opt = lbfgsb_opt(best_h, n, alpha, t0 + budget_s - 5 - (remaining - alloc), maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Gradient refinement
    remaining = deadline - time.time()
    if remaining > 80:
        alloc = min(60, remaining * 0.15)
        h_opt, c_opt = gradient_refinement(best_h, n, t0 + budget_s - 5 - (remaining - alloc), alpha=500000, max_iter=3000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Basin hopping
    remaining = deadline - time.time()
    if remaining > 200:
        alloc = min(150, remaining * 0.3)
        h_opt, c_opt = basin_hopping(best_h, n, t0 + budget_s - 5 - (remaining - alloc), rng, alpha=100000, max_restarts=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Focused local search (major phase)
    remaining = deadline - time.time()
    if remaining > 100:
        alloc = min(300, remaining * 0.5)
        h_opt, c_opt = focused_local_search(best_h, n, t0 + budget_s - 5 - (remaining - alloc), rng, init_step=0.012)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: SA
    remaining = deadline - time.time()
    if remaining > 100:
        alloc = min(200, remaining * 0.5)
        h_opt, c_opt = sa_focused(best_h, n, t0 + budget_s - 5 - (remaining - alloc), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: L-BFGS-B polish after SA
    remaining = deadline - time.time()
    if remaining > 40:
        h_opt = lbfgsb_opt(best_h, n, 2000000, t0 + budget_s - 10, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final gradient refinement
    remaining = deadline - time.time()
    if remaining > 30:
        h_opt, c_opt = gradient_refinement(best_h, n, t0 + budget_s - 10, alpha=5000000, max_iter=2000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final local search
    remaining = deadline - time.time()
    if remaining > 20:
        h_opt, c_opt = focused_local_search(best_h, n, t0 + budget_s - 3, rng, init_step=0.005)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
