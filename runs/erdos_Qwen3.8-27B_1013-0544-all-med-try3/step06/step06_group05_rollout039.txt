import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_fgrad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f = max_c + np.log(sum_exp) / alpha
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        m = k - (n - 1)
        w = weights[k] * dx
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
        else:
            m2 = -m
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return f, grad

def lbfgs_optimize(h, n, alpha, penalty_coef=500.0, maxiter=800, deadline=None):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def objective(x):
        f, g = smooth_fgrad(x, n, alpha)
        s = x.sum() - target
        f_pen = f + penalty_coef * s * s
        g_pen = g + 2.0 * penalty_coef * s
        return f_pen, g_pen

    try:
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-15,
                                          'gtol': 1e-7, 'maxfun': 2000})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def slsqp_optimize(h, n, alpha, maxiter=3000, deadline=None):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj(x):
        f, g = smooth_fgrad(x, n, alpha)
        return f, g

    def sum_con(x):
        return x.sum() - target

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                                                max_passes=80)
        h_final = project_feasible(h_final, n)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    if best_h is None:
        best_h = project_feasible(np.full(250, 0.5), 250)
        best_c = compute_c5(best_h, 250)

    return best_h, best_c, len(best_h)
