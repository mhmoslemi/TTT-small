import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(h, i, j, n, m_range):
    """Compute the change in correlation when swapping h[i]=0 to 1 and h[j]=1 to 0."""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    
    # Initialize replicas
    h_list = [to_binary(h_init, n).copy() for _ in range(n_reps)]
    corr_list = [np.correlate(h, 1-h, mode='full').astype(float) for h in h_list]
    
    # Temperature schedule for each replica
    T0 = [0.02, 0.008, 0.003, 0.001][:n_reps]
    cool = 0.99998
    scale = 3000.0
    
    best_h = h_list[0].copy()
    best_c = corr_list[0].max() * dx
    
    for r in range(n_reps):
        c = corr_list[r].max() * dx
        if c < best_c:
            best_c = c
            best_h = h_list[r].copy()

    while time.time() < deadline:

        for r in range(n_reps):
            h = h_list[r]
            corr = corr_list[r]
            cur_c = corr.max() * dx
            
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = zeros_arr[ai]
            j = ones_arr[bi]
            
            delta = compute_delta(h, i, j, n, m_range)
            nc = (corr + delta).max() * dx

                h[i], h[j] = 1.0, 0.0
                corr += delta
                zeros_arr[ai] = j
                ones_arr[bi] = i

                    best_c = nc
                    best_h = h.copy()

        if rng.random() < 0.01:

            r2 = r1 + 1
            c1 = corr_list[r1].max() * dx
            c2 = corr_list[r2].max() * dx
            # Accept swap if c1 < c2 or with probability exp(-(c1-c2)/T[r1])

                h_list[r1], h_list[r2] = h_list[r2].copy(), h_list[r1].copy()
                corr_list[r1], corr_list[r2] = corr_list[r2].copy(), corr_list[r1].copy()
                T[r1], T[r2] = T[r2], T[r1]
    
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=300):

    h = to_binary(h_init, n).copy()

    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = None
            best_nc = cur_c
            for j in ones_list:
                delta = compute_delta(h, i, j, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j is not None and best_nc < cur_c - 1e-15:

                h[best_j] = 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

```python
def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1-h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()  # softmax weights, length 2n-1
    val = m + (sm + np.log(exp_s.sum())) / alpha
    
    # Gradient
    # grad[i] = dx * sum_d w_d * [(1-h[i+d])*1{i+d<n} - h[i-d]*1{i-d>=0}]
    grad = np.zeros(n)
    W = w  # W[m] = w_d where d = m - (n-1)
    for d in range(-(n-1), n):
        wd = W[d + (n-1)]
        if wd < 1e-15:
            continue
        # (1-h[i+d]) for i in [0,n) where i+d in [0,n): i in [max(0,-d), min(n, n-d))
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo < hi:
            grad[lo:hi] += wd * (1 - h[lo+d:hi+d])
        # -h[i-d] for i in [0,n) where i-d in [0,n): i in [max(0,d), min(n, n+d))
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wd * h[lo2-d:hi2-d]
    grad *= dx
    return val, grad
```

def slsqp_with_grad(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    
    def grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        w = exp_s / exp_s.sum()
        
        grad_out = np.zeros(n)
        for d in range(-(n-1), n):
            wd = w[d + (n-1)]
            if wd < 1e-14:
                continue
            lo = max(0, -d)
            hi = min(n, n - d)
            if lo < hi:
                grad_out[lo:hi] += wd * (1.0 - x[lo+d:hi+d])
            lo2 = max(0, d)
            hi2 = min(n, n + d)
            if lo2 < hi2:
                grad_out[lo2:hi2] -= wd * x[lo2-d:hi2-d]
        grad_out *= dx
        return grad_out
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
