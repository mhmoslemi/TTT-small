    prefix=np.empty(n+1)

def project_box_sum(v, target):

    lo = float(v.min()) - 1.0

def compute_C(h, n):
    m = n - 1
    N = 2 * n - 1
    off = m

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])

    nfl = next_pow2(2 * n - 1)
    F = np.fft.rfft(h, nfl)
    ac = np.fft.irfft(F * np.conj(F), nfl)[:n]

    C = np.empty(N, dtype=float)
    idx = np.arange(n)
    C[off:off + n] = prefix[n] - prefix[idx] - ac[idx]
    if n > 1:
        C[off - 1:0:-1] = prefix[n - 1:0:-1] - ac[1:][::-1]
    return C


def smax_from_C(C, alpha):
    m = float(C.max())
    if alpha <= 1e-12:
        return m
    z = alpha * (C - m)
    return m + float(np.log(np.exp(z).sum())) / alpha


def softmax_weights(C, alpha):
    m = float(C.max())
    if alpha <= 1e-12:
        p = np.full(C.shape, 1.0 / float(C.size))
        return p
    z = alpha * (C - m)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 0.0:
        idx = int(np.argmax(C))
        p.fill(0.0)
        p[idx] = 1.0
    else:
        p /= s
    return p


def weights_active(C, tol):
    m = float(C.max())
    act = C >= (m - tol)
    cnt = int(act.sum())
    if cnt <= 0:
        act = (C == m)
        cnt = int(act.sum())
        if cnt <= 0:
            idx = int(np.argmax(C))
            act.fill(False)
            act[idx] = True
    p = act.astype(float)
    s = float(p.sum())
    if s <= 0.0:
        idx = int(np.argmax(C))
        p.fill(0.0)
        p[idx] = 1.0
    else:
        p /= s
    return p


def gradient_from_p(p, h, n):
    m = n - 1
    N = 2 * n - 1
    off = m

    cp = np.empty(N + 1, dtype=float)
    cp[0] = 0.0
    np.cumsum(p, out=cp[1:])
    jidx = np.arange(n)
    lin = cp[jidx + n] - cp[jidx]

    s = p + p[::-1]
    nconv = next_pow2(3 * n - 2)
    Sf = np.fft.rfft(s, nconv)
    Hf = np.fft.rfft(h, nconv)
    conv = np.fft.irfft(Sf * Hf, nconv)
    return lin - conv[off:off + n]

    else:
        p = softmax_weights(C, alpha)
    g = gradient_from_p(p, h, n)
    return f, g, C

def resample_const(h_old, n_old, n_new):
    if n_new == n_old:
        return project_box_sum(h_old, float(n_new) / 2.0)
    d_old = 2.0 / float(n_old)
    d_new = 2.0 / float(n_new)
    x_old = np.linspace(0.0, 2.0, n_old + 1)
    cum_old = np.empty(n_old + 1, dtype=float)
    cum_old[0] = 0.0
    np.cumsum(h_old * d_old, out=cum_old[1:])
    x_new = np.linspace(0.0, 2.0, n_new + 1)
    cum_new = np.interp(x_new, x_old, cum_old)
    hh = np.diff(cum_new) / d_new
    return project_box_sum(hh, float(n_new) / 2.0)


def coord_deriv(h, n, j):
    m = n - 1
    N = 2 * n - 1
    b = np.zeros(N, dtype=float)
    b[j:j + n] += 1.0 - h[m::-1]
    b[m - j:m - j + n] -= h
    return b


def choose_pair(G, h, rng, k=8):
    rec = np.flatnonzero(h < (1.0 - 1e-10))
    don = np.flatnonzero(h > 1e-10)
    if rec.size == 0 or don.size == 0:
        return None
    k = int(min(k, rec.size, don.size))
    if k <= 0:
        return None

    if rec.size > k:
        rec_idx = np.argpartition(G[rec], k - 1)[:k]
    else:
        rec_idx = np.arange(rec.size)

    if don.size > k:
        don_idx = np.argpartition(G[don], -k)[:k]
    else:
        don_idx = np.arange(don.size)

    best_i = -1
    best_j = -1
    best_score = 0.0
    for a in rec_idx:
        i = int(rec[a])
        gi = float(G[i])
        for b in don_idx:
            j = int(don[b])
            if i == j:
                continue
            sc = gi - float(G[j])
            if sc < best_score:
                best_score = sc
                best_i = i
                best_j = j

    if best_i >= 0 and best_score < -1e-10:
        return best_i, best_j, best_score

    for _ in range(24):
        i = int(rec[rng.randint(rec.size)])
        j = int(don[rng.randint(don.size)])
        if i == j:
            continue
        sc = float(G[i]) - float(G[j])
        if sc < -1e-10:
            return i, j, sc
    return None


def pair_polish(h, n, deadline, alpha, seed, active_tol=0.0):
    target = float(n) / 2.0
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = c5_exact(h, n)

    C = compute_C(h, n)
    f = smax_from_C(C, alpha)
    rng = np.random.RandomState(seed)
    off = n - 1

    it = 0
    no_imp = 0
    total_no = 0
    recompute_every = 160
    max_total_no = 5000

    while time.time() < deadline:
        if active_tol > 0.0:
            p = weights_active(C, active_tol)
        else:
            p = softmax_weights(C, alpha)
        G = gradient_from_p(p, h, n)

        chosen = choose_pair(G, h, rng, 8)
        if chosen is None:
            no_imp += 6
            total_no += 6
            if total_no > max_total_no:
                break
            h = add_noise_preserve(h, n, 0.012, it + seed)
            C = compute_C(h, n)
            f = smax_from_C(C, alpha)
            it += 1
            continue

        i, j, score = chosen
        dmax = min(1.0 - h[i], h[j])
        if dmax <= 1e-12:
            no_imp += 1
            total_no += 1
            if total_no > max_total_no:
                break
            continue

        bi = coord_deriv(h, n, i)
        bj = coord_deriv(h, n, j)
        bdiff = bi - bj
        d = i - j

        base = min(dmax, 0.5 / max(1.0, abs(score)))
        cands = (
            dmax,
            0.70 * dmax,
            0.50 * dmax,
            0.25 * dmax,
            0.10 * dmax,
            0.01 * dmax,
            base,
            0.5 * base,
            0.25 * base,
        )

        best_delta = 0.0
        best_f = f
        best_arr = None
        for cand in cands:
            cand = float(cand)
            if cand <= 1e-12 or cand > dmax + 1e-12:
                continue
            cand = min(cand, dmax)
            if best_delta != 0.0 and abs(cand - best_delta) < 1e-10:
                continue

            delta = bdiff * cand
            delta[off] -= 2.0 * cand * cand
            if d != 0:
                delta[off + d] += cand * cand
                delta[off - d] += cand * cand

            Cf = C + delta
            ff = smax_from_C(Cf, alpha)
            if np.isfinite(ff) and ff < best_f - 1e-13:
                best_f = ff
                best_delta = cand
                best_arr = delta

        if best_arr is not None:
            h[i] += best_delta
            h[j] -= best_delta
            C = C + best_arr
            f = best_f
            no_imp = 0
            total_no = max(0, total_no - 4)
            if it % 8 == 0:
                cc = c5_exact(h, n)
                if cc < best_c:
                    best_c = cc
                    best_h = h.copy()
        else:
            no_imp += 1
            total_no += 1
            if no_imp > 260:
                h = add_noise_preserve(h, n, 0.022, it + 7777)
                C = compute_C(h, n)
                f = smax_from_C(C, alpha)
                no_imp = 0

        if it % recompute_every == 0:
            C = compute_C(h, n)
            f = smax_from_C(C, alpha)

        if total_no > max_total_no:
            break
        it += 1

    cc = c5_exact(h, n)
    if cc < best_c:
        best_c = cc
        best_h = h.copy()
    return best_h, best_c


def polish_resolution(h, n, seed, deadline, use_slsqp=True, use_lbfgs=True, use_pair=True):
    target = float(n) / 2.0
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = c5_exact(h, n)

    t0 = time.time()
    total = max(0.0, float(deadline - t0))
    if total < 20.0:
        return best_h, best_c

    dx = 2.0 / float(n)
    active_final = max(1e-9, 1e-4 * (float(n) / 2.0))
    active_mid = max(1e-9, 5e-5 * (float(n) / 2.0))

    if use_slsqp:
        steps = [
            ("pg", 50.0, 0.060, 0.0),
            ("pg", 150.0, 0.100, 0.0),
            ("pg", 700.0, 0.100, 0.0),
            ("slsqp", 0.0, 0.250, 0.0),
            ("lbfgs", 0.0, 0.120, 0.0),
            ("pg", 8000.0, 0.120, active_mid),
            ("pg", 50000.0, 0.100, active_final),
            ("pair", 0.0, 0.150, active_final),
        ]
    else:
        steps = [
            ("pg", 150.0, 0.100, 0.0),
            ("pg", 1500.0, 0.140, 0.0),
            ("pg", 8000.0, 0.200, active_mid),
            ("pg", 50000.0, 0.280, active_final),
            ("pair", 0.0, 0.280, active_final),
        ]

    if use_slsqp and n > 700:
        steps = (
            ("pg", 150.0, 0.10, 0.0),
            ("pg", 1500.0, 0.14, 0.0),
            ("pg", 8000.0, 0.22, active_mid),
            ("pg", 50000.0, 0.30, active_final),
            ("pair", 0.0, 0.24, active_final),
        )

    tcur = t0
    for kind, alpha, frac, atol in steps:
        dstep = min(deadline - 1.0, tcur + total * frac)
        tcur = dstep
        if dstep <= time.time() + 2.0:
            continue

        if kind == "pg":
            hh, cc = projected_gradient_polish(best_h, n, dstep, alpha, base_lr=0.16, active_tol=atol)
            if cc < best_c:
                best_c = cc
                best_h = hh
        elif kind == "slsqp" and use_slsqp:
            alphas = [100.0, 300.0, 1000.0, 3000.0]
            if dstep - time.time() > 90.0:
                alphas.append(8000.0)
            hh, cc = slsqp_polish(best_h, n, dstep, alphas)
            if cc < best_c:
                best_c = cc
                best_h = hh
        elif kind == "lbfgs" and use_lbfgs:
            hh, cc = lbfgs_polish(best_h, n, dstep, [1.0, 8.0], alpha=2000.0)
            if cc < best_c:
                best_c = cc
                best_h = hh
        elif kind == "pair" and use_pair:
            hh, cc = pair_polish(best_h, n, dstep, 20000.0, seed + 31, active_tol=atol)
            if cc < best_c:
                best_c = cc
                best_h = hh

        if time.time() >= deadline - 2.0:
            break

    hh = project_box_sum(best_h, target)
    cc = c5_exact(hh, n)
    if cc < best_c:
        best_c = cc
        best_h = hh
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    total_run = max(60.0, budget_s - 12.0)
    finish = t0 + total_run

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = np.full(400, 0.5)
        n0 = 400

    if n0 <= 1:
        h0 = np.full(400, 0.5)
        n0 = 400

    if n0 % 2 == 1:
        target0 = float(n0) / 2.0
        h0 = project_box_sum(h0, target0)
        h0 = resample_const(h0, n0, 400)
        n0 = 400
    else:
        h0 = project_box_sum(h0, float(n0) / 2.0)

    best_h = h0.copy()
    best_n = n0
    best_c = c5_exact(best_h, best_n)

    # Main refinement at parent resolution.
    d1 = t0 + 0.42 * total_run
    if d1 > time.time() + 30.0:
        bh, bc = polish_resolution(best_h, best_n, int(seed) + 1, d1,
                                   use_slsqp=(best_n <= 700), use_lbfgs=(best_n <= 900))
        if bc < best_c:
            best_c = bc
            best_h = bh
            best_n = best_n

    # Explore a slightly finer resolution.
    n2 = 600 if best_n < 600 else best_n + 200
    if n2 != best_n and n2 <= 1000 and finish - time.time() > 130.0:
        h2 = resample_const(best_h, best_n, n2)
        d2 = t0 + 0.68 * total_run
        bh, bc = polish_resolution(h2, n2, int(seed) + 17, d2,
                                   use_slsqp=(n2 <= 700), use_lbfgs=(n2 <= 900))
        if bc < best_c:
            best_c = bc
            best_h = bh
            best_n = n2

    # Explore an even finer resolution if time remains.
    n3 = 800 if best_n < 800 else best_n + 200
    if n3 != best_n and n3 <= 1000 and finish - time.time() > 110.0:
        h3 = resample_const(best_h, best_n, n3)
        d3 = t0 + 0.88 * total_run
        bh, bc = polish_resolution(h3, n3, int(seed) + 37, d3,
                                   use_slsqp=False, use_lbfgs=True)
        if bc < best_c:
            best_c = bc
            best_h = bh
            best_n = n3

    # Final polish at the best resolution found.
    if finish - time.time() > 70.0:
        bf = best_h.copy()
        bh, bc = polish_resolution(bf, best_n, int(seed) + 73, finish,
                                   use_slsqp=(best_n <= 700), use_lbfgs=False)
        if bc < best_c:
            best_c = bc
            best_h = bh
            best_n = best_n

    # Cheap symmetry transformations: reversal and complement.
    targetf = float(best_n) / 2.0
    transforms = (
        best_h[::-1],
        1.0 - best_h,
    )
    for cand in transforms:
        hq = project_box_sum(cand, targetf)
        cq = c5_exact(hq, best_n)
        if cq < best_c:
            best_c = cq
            best_h = hq

    best_h = project_box_sum(best_h, targetf)
    best_c = c5_exact(best_h, best_n)
    return best_h, float(best_c), int(best_n)
