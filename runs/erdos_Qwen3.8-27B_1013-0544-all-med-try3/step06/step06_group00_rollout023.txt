So: Δc_d = (1-h[i-d])·[0≤i-d<n] - h[i+d]·[0≤i+d<n] - (1-h[j-d])·[0≤j-d<n] + h[j+d]·[0≤j+d<n]

This is O(1) per lag d. 

Now, for the full update of corr after a swap, I need Δc_d for ALL d from -(n-1) to n-1. That's O(n) per swap. But for the EVALUATION (deciding which swap to make), I only need it at the K critical lags. That's O(K) per swap evaluation.

After making the swap, I update corr in O(n).

So the total per sweep is:
- Evaluation: O(n²/4 * K) for all pairs
- Update: O(n) for the chosen swap
- Total: O(n²K/4 + n) ≈ O(n²K/4)

At n=400, K=10: 400,000 per sweep. Very fast.
At n=800, K=10: 1,600,000 per sweep. Still fast.

I can do many sweeps in the time budget!

import time

    c = np.correlate(h, 1 - h, mode='full')

            if s < 1e-14:

    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def vectorized_best_swap(corr, h, n, crit_idx, K):
    """Find the best swap using vectorized evaluation at K critical positions."""
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    if len(zeros) == 0 or len(ones) == 0:
        return None
    
    d_vals = crit_idx - (n - 1)  # shape (K,)
    
    # Compute A[i, k] for each zero position i and critical lag k
    # A[i, k] = (1-h[i-d_k])·[0≤i-d_k<n] - h[i+d_k]·[0≤i+d_k<n]
    # Using OLD h values (before swap)
    
    # Vectorize over zeros and critical lags
    # zeros: (nz,), d_vals: (K,)
    # i_minus_d: (nz, K), i_plus_d: (nz, K)
    
    im_d = zeros[:, None] - d_vals[None, :]  # (nz, K)
    ip_d = zeros[:, None] + d_vals[None, :]  # (nz, K)
    
    mask_im = (im_d >= 0) & (im_d < n)
    mask_ip = (ip_d >= 0) & (ip_d < n)
    
    # h[im_d] where valid, 0 otherwise (but we only use where mask is True)
    h_im = np.where(mask_im, h[np.clip(im_d, 0, n-1)], 0.0)
    h_ip = np.where(mask_ip, h[np.clip(ip_d, 0, n-1)], 0.0)
    
    A = (1.0 - h_im) * mask_im - h_ip * mask_ip  # (nz, K)
    
    # Similarly for ones
    jm_d = ones[:, None] - d_vals[None, :]
    jp_d = ones[:, None] + d_vals[None, :]
    
    mask_jm = (jm_d >= 0) & (jm_d < n)
    mask_jp = (jp_d >= 0) & (jp_d < n)
    
    h_jm = np.where(mask_jm, h[np.clip(jm_d, 0, n-1)], 0.0)
    h_jp = np.where(mask_jp, h[np.clip(jp_d, 0, n-1)], 0.0)
    
    B = -(1.0 - h_jm) * mask_jm + h_jp * mask_jp  # (no, K)
    
    # Now M[i, j] = max_k (corr[crit_idx[k]] + A[i, k] + B[j, k])
    corr_crit = corr[crit_idx]  # (K,)
    
    # C[i, j, k] = A[i, k] + B[j, k] + corr_crit[k]
    # M[i, j] = max_k C[i, j, k]
    
    # Using broadcasting: A (nz, 1, K) + B (1, no, K) + corr_crit (1, 1, K)
    C = A[:, None, :] + B[None, :, :] + corr_crit[None, None, :]
    M = C.max(axis=2)  # (nz, no)
    
    # Find the best (i, j)
    min_val, min_idx = M.min(), M.argmin()
    ii, jj = divmod(min_idx, len(ones))
    return zeros[ii], ones[jj], min_val

def fast_local_search(h, n, deadline, K=20, max_sweeps=5000, allow_worsening=False, worsening_tol=2):
    """Fast binary local search with bottleneck-focus."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        crit_idx = np.argsort(corr)[::-1][:K]
        cur_max = corr.max()
        
        result = vectorized_best_swap(corr, h, n, crit_idx, K)
        if result is None:
            break
        i, j, new_max = result
        
        if allow_worsening and new_max > cur_max:
            if new_max - cur_max > worsening_tol:
                break
            # Accept worsening move (for escape)
        
        if new_max >= cur_max + 1e-10 and not allow_worsening:
            break
        
        if new_max > cur_max + 1e-10:
            break  # No improving move
        
        # Perform swap
        h[i], h[j] = 1.0, 0.0
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur_max = corr.max()
        
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

Actually, L-BFGS-B doesn't handle equality constraints well. Let me stick with SLSQP but optimize it.

Alternatively, I can eliminate the sum constraint by reparameterizing: use n-1 free variables and set the last one to satisfy the constraint. But that's awkward with box constraints.

Another approach: use the projected gradient method. At each step:
1. Compute gradient of the smoothed objective
2. Take a step in the negative gradient direction
3. Project back to the feasible set [0,1]^n ∩ {sum = n/2}

The projection onto [0,1]^n ∩ {sum = n/2} can be done efficiently (it's a quadratic optimization that can be solved in O(n log n)).

Let me think about whether this would be faster than SLSQP for n=400.

SLSQP does a line search at each iteration and solves a QP subproblem. For n=400, this is O(n³) per iteration. With 2000 iterations, that's 2000 * 400³ = 128 * 10^9 operations. That's too slow.

Actually, SLSQP is more efficient than that. It uses BFGS approximation of the Hessian and solves a smaller QP. For n=400, each iteration should be maybe 10-50ms. So 2000 iterations = 20-100 seconds. That's reasonable.

The projected gradient approach would be O(n) per iteration (for the correlate + O(n log n) for the projection). Much faster per iteration, but might need more iterations.

            h = best_h + 0.003 * rng.randn(best_n)
            h = project_feasible(h, best_n)
            for alpha in [500000, 5000000]:
                if time.time() > dl - 3:
                    break
                h = slsqp_smooth(h, best_n, alpha, 1000, 1e-15, dl - 3)
                c = compute_c5(h, best_n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        
        # Try one more binary refinement on the best solution
        if time.time() < dl - 2:
            hb, cb = fast_greedy_binary(best_h, best_n, dl - 1, K=20, max_sweeps=10000)
            if cb < best_c:
                best_c, best_h = cb, hb.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
