def _lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)

        if time.time() > deadline - 90:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 45:
            break
        alloc = min(75, remaining * 0.25)
        if alloc < 30:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.008 + 0.005 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_polish(best_h, n, 1000000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            best_h, best_c = fine_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
