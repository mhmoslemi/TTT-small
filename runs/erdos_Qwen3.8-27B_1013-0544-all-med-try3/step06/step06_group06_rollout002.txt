Introduce an auxiliary variable t and minimize t subject to:

This is a semi-infinite program (finite here, with 2n-1 constraints). The Lagrangian is:
L(h, t, λ, μ) = t + sum_k λ_k (c_k(h) - t) + μ(sum(h) - n/2)
= t(1 - sum_k λ_k) + sum_k λ_k c_k(h) + μ(sum(h) - n/2)

At optimality, sum_k λ_k = 1 (KKT for the t variable), so:
L = sum_k λ_k c_k(h) + μ(sum(h) - n/2)

This is a linear combination of the c_k's plus the constraint. The dual problem is:
max_{λ ≥ 0, sum λ = 1, μ} min_{0≤h≤1} [sum_k λ_k c_k(h) + μ(sum(h)-n/2)]

The inner minimization is: min_{0≤h≤1} [sum_k λ_k sum_i h_i(1-h_{i+d_k}) * dx + μ(sum(h)-n/2)]

= dx * sum_i h_i - dx * sum_i sum_k λ_k h_i h_{i+d_k}

The first term is linear, the second is a quadratic form. So the inner problem is:

1. Fix λ, solve the QP for h
2. Update λ based on which c_k are binding

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_smooth_objective_and_grad(h, n, alpha, rho=1e8):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    c_max = c.max()
    ac = alpha * (c - c_max)
    # For numerical stability
    max_ac = ac.max()
    ac_shifted = ac - max_ac
    sum_exp = np.exp(ac_shifted).sum()
    f_smooth = c_max + (max_ac + np.log(sum_exp)) / alpha
    w = np.exp(ac_shifted) / sum_exp
    
    # Penalty
    s = h.sum()
    f_pen = f_smooth + rho * (s - n/2)**2
    
    # Gradient of smooth part
    i_idx = np.arange(n)
    j_idx = np.arange(n)
    indices = j_idx[None,:] - i_idx[:,None] + n - 1
    W = w[indices]
    grad_smooth = dx * (W @ (1-h) - W.T @ h)
    
    # Gradient of penalty
    grad_pen = 2 * rho * (s - n/2) * np.ones(n)
    
    return f_pen, grad_smooth + grad_pen

def lbfgs_smooth_step(h, n, alpha, maxiter, rho=1e8):
    def obj_and_grad(x):
        return compute_smooth_objective_and_grad(x, n, alpha, rho)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h, method='L-BFGS-B',
                  jac=True, bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                          'maxls': 50})

def nesterov_projected_grad(h, n, deadline, lr0=0.01, gamma=0.9):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 10000
    y = h.copy()
    mom = h.copy()
    lr = lr0

        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = project_feasible(h + (mom - h), n)
        # Gradient at y
        _, grad = compute_smooth_objective_and_grad(y, n, alpha, rho=0)
        # Also add gradient of the sum constraint (penalty)
        s = y.sum()
        grad += 2e6 * (s - n/2)
        
        # Step
        h_new = project_feasible(y - lr * grad, n)
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Update momentum
        mom = h_new
        h = h_new
        lr *= gamma
        if lr < 1e-10:
            break
    
    return best_h, best_c
