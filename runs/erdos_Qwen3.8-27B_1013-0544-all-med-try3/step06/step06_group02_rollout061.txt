Let me also think about the L-BFGS approach more carefully. The smooth max approximation:
f_α(h) = (1/α) log(∑_k exp(α * c_k))

This approximates max_k c_k as α → ∞. The gradient is:

= ∑_k w_k * ∇c_k where w_k = exp(α c_k) / ∑_j exp(α c_j)

For large α, w_k is concentrated on the argmax. For small α, it's more uniform.

f_α(h) = c_max + (1/α) log(∑_k exp(α(c_k - c_max)))

This is numerically stable.

For the gradient, w_k = exp(α(c_k - c_max)) / ∑_j exp(α(c_j - c_max)).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _single_lag_val(h, k, n):
    dx = 2.0 / n
    if k >= 0:
        if k >= n:
            return 0.0
        return float(np.sum(h[:n-k] * (1.0 - h[k:])) * dx)
    else:
        kk = -k
        if kk >= n:
            return 0.0
        return float(np.sum(h[kk:] * (1.0 - h[:n-kk])) * dx)

def _single_lag_grad(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] = (1.0 - h[k:n]) * dx
            g[k:] += -h[:n-k] * dx
    else:
        kk = -k
        if kk < n:
            g[kk:] = (1.0 - h[:n-kk]) * dx
            g[:n-kk] += -h[kk:] * dx
    return g

def _epigraph_obj(x, n):
    return x[n]

def _epigraph_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _epigraph_ineq_val(x, k, n):
    h = x[:n]
    t = x[n]
    return t - _single_lag_val(h, k, n)

def _epigraph_ineq_jac(x, k, n):
    h = x[:n]
    g = _single_lag_grad(h, k, n)
    return np.append(-g, 1.0)

def _epigraph_eq_val(x, n):
    return x[:n].sum() * (2.0 / n) - 1.0

def _epigraph_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 2.0 / n
    return g

def solve_epigraph(h, n, lags, maxiter=500):
    cons = []
    for k in lags:
        cons.append({'type': 'ineq', 'fun': _epigraph_ineq_val, 'jac': _epigraph_ineq_jac, 'args': (k, n)})
    cons.append({'type': 'eq', 'fun': _epigraph_eq_val, 'jac': _epigraph_eq_jac, 'args': (n,)})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, np.inf)]
    x0 = np.append(h.copy(), compute_c5(h, n))
    
    try:
        res = minimize(_epigraph_obj, x0, jac=_epigraph_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons, args=(n,),
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        h_new = project_feasible(res.x[:n].copy(), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _single_lag_val(h, k, n):
    dx = 2.0 / n
    if k >= 0:
        if k >= n:
            return 0.0
        return float(np.sum(h[:n - k] * (1.0 - h[k:])) * dx)
    else:
        kk = -k
        if kk >= n:
            return 0.0
        return float(np.sum(h[kk:] * (1.0 - h[:n - kk])) * dx)


def _single_lag_grad(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n - k] = (1.0 - h[k:n]) * dx
            g[k:] += -h[:n - k] * dx
    else:
        kk = -k
        if kk < n:
            g[kk:] = (1.0 - h[:n - kk]) * dx
            g[:n - kk] += -h[kk:] * dx
    return g

    return x[n]


def _epi_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _epi_ineq_val(x, k, n):
    h = x[:n]
    t = x[n]
    return t - _single_lag_val(h, k, n)


def _epi_ineq_jac(x, k, n):
    h = x[:n]
    g = _single_lag_grad(h, k, n)
    return np.append(-g, 1.0)


def _epi_eq_val(x, n):
    return x[:n].sum() * (2.0 / n) - 1.0


def _epi_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 2.0 / n
    return g


def solve_epigraph(h, n, lags, maxiter=500):
    cons = []
    for k in lags:
        cons.append({'type': 'ineq', 'fun': _epi_ineq_val, 'jac': _epi_ineq_jac,
                     'args': (k, n)})
    cons.append({'type': 'eq', 'fun': _epi_eq_val, 'jac': _epi_eq_jac, 'args': (n,)})
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    x0 = np.append(h.copy(), compute_c5(h, n))
    try:
        res = minimize(_epi_obj, x0, jac=_epi_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, args=(n,),
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'eps': 1e-8})
        h_new = project_feasible(res.x[:n].copy(), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)


def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad


def _smmax_penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def _smmax_penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g


def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 10.0), (500, 50.0), (2000, 100.0),
        (10000, 500.0), (50000, 2000.0), (200000, 5000.0),
        (1000000, 10000.0), (5000000, 20000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_smmax_penalty_obj, h, jac=_smmax_penalty_grad,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': min(max_iters, 300), 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': 5000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c


def active_set_optimize(h, n, deadline, max_rounds=30):
    best_c = compute_c5(h, n)
    prev_best = best_c
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_c = c_vec.max()
        top_indices = np.argsort(c_vec)[::-1]
        lags_top = (top_indices[:min(10, 2 * n - 1)] - (n - 1)).tolist()
        threshold = max_c - 0.015
        near_indices = np.where(c_vec >= threshold)[0]
        near_lags = (near_indices - (n - 1)).tolist()
        all_lags = list(set(lags_top + near_lags))
        all_lags.sort(key=lambda k: -c_vec[k + (n - 1)])
        all_lags = all_lags[:12]
        h_new, c_new = solve_epigraph(h, n, all_lags, maxiter=400)
        if c_new < best_c - 1e-10:
            best_c = c_new
            h = h_new
        else:
            if len(all_lags) < 20:
                lags2 = (top_indices[:20] - (n - 1)).tolist()
                h2, c2 = solve_epigraph(h, n, lags2, maxiter=300)
                if c2 < best_c - 1e-10:
                    best_c = c2
                    h = h2
                else:
                    if best_c >= prev_best - 1e-12:
                        break
            else:
                if best_c >= prev_best - 1e-12:
                    break
        prev_best = best_c
    return h, best_c


def try_gurobi(h, n, lags, timeout=30):
    try:
        import cvxpy as cp
    except ImportError:
        return h, compute_c5(h, n)
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in lags:
        S = np.zeros((n, n))
        ak = np.zeros(n)
        if k == 0:
            np.fill_diagonal(S, 1.0)
            ak[:] = 1.0
            nvalid = n
        elif k > 0:
            for i in range(n - k):
                S[i, i + k] = 0.5
                S[i + k, i] = 0.5
            ak[:n - k] = 1.0
            nvalid = n - k
        else:
            kk = -k
            for i in range(n - kk):
                S[i + kk, i] = 0.5
                S[i, i + kk] = 0.5
            ak[kk:] = 1.0
            nvalid = n - kk
        constraints.append(cp.quad_form(h_var, dx * S) - dx * (ak @ h_var) + t_var >= 0)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solved = False
    for solver_name in ['GUROBI', 'SCIP']:
        if solved:
            break
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            if solver_name == 'GUROBI':
                prob.solve(solver=solver, nonconvex=2, TimeLimit=timeout)
            else:
                prob.solve(solver=solver, TimeLimit=timeout)
            if h_var.value is not None:
                hv = np.array(h_var.value).flatten()
                hv = project_feasible(hv, n)
                cv = compute_c5(hv, n)
                return hv, cv
                solved = True
        except Exception:
            continue
    return h, compute_c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Active-set SLSQP on initial resolution
    t1 = min(t0 + 300, deadline - 350)
    best_h, best_c = active_set_optimize(best_h, n_init, t1, max_rounds=30)

    # Phase 2: L-BFGS complementary
    if time.time() < deadline - 300:
        t2 = min(time.time() + 120, deadline - 300)
        h2, c2 = lbfgs_optimize(best_h, n_init, t2, max_iters=300)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        if time.time() < deadline - 250:
            t2b = min(time.time() + 100, deadline - 250)
            h2b, c2b = active_set_optimize(best_h, n_init, t2b, max_rounds=20)
            if c2b < best_c:
                best_c = c2b
                best_h = h2b

    # Phase 3: Try Gurobi on current best (if available)
    if time.time() < deadline - 200:
        c_vec = compute_c5_vec(best_h, n_init)
        top_idx = np.argsort(c_vec)[::-1][:8]
        g_lags = (top_idx - (n_init - 1)).tolist()
        g_deadline = min(time.time() + 60, deadline - 140)
        if time.time() < g_deadline:
            hg, cg = try_gurobi(best_h, n_init, g_lags, timeout=50)
            if cg < best_c:
                best_c = cg
                best_h = hg
            if time.time() < g_deadline:
                hg2, cg2 = active_set_optimize(best_h, n_init, g_deadline, max_rounds=10)
                if cg2 < best_c:
                    best_c = cg2
                    best_h = hg2

    # Phase 4: Try different resolutions
    if time.time() < deadline - 120:
        n_candidates = [n_init + 50, n_init - 50, n_init + 100]
        for n_try in n_candidates:
            if time.time() > deadline - 60:
                break
            if n_try < 100 or n_try > 600:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(70, deadline - time.time() - 30)
            if alloc < 25:
                break
            h_t, c_t = active_set_optimize(h_interp, n_try, time.time() + alloc * 0.65, max_rounds=15)
            if time.time() < deadline - 25:
                h_t, c_t = lbfgs_optimize(h_t, n_try, time.time() + min(20, deadline - time.time() - 12), max_iters=100)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 5: Final active-set pass on best
    if time.time() < deadline - 20:
        best_h, best_c = active_set_optimize(best_h, len(best_h), deadline - 10, max_rounds=10)

    # Phase 6: Pair search polishing
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 1,
                                     [0.001, 0.0005, 0.0002, 0.0001], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
