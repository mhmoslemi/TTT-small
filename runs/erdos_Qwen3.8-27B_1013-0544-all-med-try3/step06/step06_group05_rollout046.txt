from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

Total effect on c[k_max]:
d * ((1-h[i0-k_max]) - h[i0+k_max] - (1-h[j0-k_max]) + h[j0+k_max])

For this to be negative (reducing c[k_max]), we need:
(1-h[i0-k_max]) - h[i0+k_max] < (1-h[j0-k_max]) - h[j0+k_max]
i.e., h[j0-k_max] + h[j0+k_max] > h[i0-k_max] + h[i0+k_max]

So we want to choose (i0, j0) such that h[j0-k_max] + h[j0+k_max] > h[i0-k_max] + h[i0+k_max].

The best j0 maximizes h[j0-k_max] + h[j0+k_max] (this is where the "bad" contribution is largest in the correlation).
The best i0 minimizes h[i0-k_max] + h[i0+k_max] (this is where increasing h helps most).

But we also need to check that other lags don't increase above the max!

This gives us a very efficient algorithm:
1. Find k_max = argmax(c_full)
2. Compute S[i] = h[i-k_max] + h[i+k_max] for all i (with bounds, treating out-of-bounds as 0)
3. The best j0 is argmax(S)
4. The best i0 is argmin(S) (subject to constraints)
5. Try the move h[i0] += d, h[j0] -= d
6. Check if the new max is lower (using the full delta computation)
7. If yes, accept; if no, try smaller d or different pairs

This is O(n) per candidate pair! And we only need to try a few candidates.

        #   +d*(1-h[i-k]) if i-k is valid (h[i] is the first factor)
        #   -d*h[i+k] if i+k is valid (h[i] is in (1-h[n]), so increasing h[i] decreases c[k])

            P[valid_m] += (1.0 - h[idx_m[valid_m]])
            P[valid_p] -= h[idx_p[valid_p]]

    return best_h, best_c

def anneal_efficient(h, n, deadline, rng, T0=0.005, T_end=0.00005, n_iter=200000):
    """Simulated annealing with O(n) correlation updates."""
    dx = 2.0 / n
    h = h.copy()
    c_full = np.correlate(h, 1 - h, mode='full')
    best_c = float(np.max(c_full) * dx)
    best_h = h.copy()
    current_c = best_c

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / n_iter)
        if T < 1e-8:
            break

        i0 = rng.randint(n)
        j0 = rng.randint(n)
        if i0 == j0:
            continue
        d = rng.uniform(-0.03, 0.03)

        if h[i0] + d < -1e-12 or h[i0] + d > 1.0 + 1e-12:
            continue
        if h[j0] - d < -1e-12 or h[j0] - d > 1.0 + 1e-12:
            continue

        delta = get_delta_c(h, n, i0, j0, d)
        new_max = float(np.max(c_full + delta) * dx)
        delta_c = new_max - current_c

        if delta_c < 0 or (T > 0 and np.log(rng.rand() + 1e-15) < -delta_c / T):
            h[i0] = np.clip(h[i0] + d, 0.0, 1.0)
            h[j0] = np.clip(h[j0] - d, 0.0, 1.0)
            c_full = np.correlate(h, 1 - h, mode='full')
            current_c = float(np.max(c_full) * dx)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            pass

    return best_h, best_c

def multi_algorithm_optimize(h_start, n, deadline, rng):
    """Run multiple optimization strategies in sequence."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    alphas = [1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        alloc = min(60, (deadline - time.time()) * 0.15)
        if alloc < 10:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Phase 2: Targeted local search
    if time.time() < deadline - 20:
        ht, ct = targeted_local_search(best_h, n, time.time() + min(120, (deadline - time.time()) * 0.4))
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    # Phase 3: Annealing
    if time.time() < deadline - 20:
        alloc = min(80, (deadline - time.time()) * 0.4)
        if alloc > 15:
            ht, ct = anneal_efficient(best_h, n, time.time() + alloc, rng)
            if ct < best_c:
                best_c = ct
                best_h = ht.copy()

    # Phase 4: More SLSQP at high alpha
    if time.time() < deadline - 15:
        ht = slsqp_optimize(best_h, n, 50000000, maxiter=5000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()

    # Phase 5: Final targeted search with very small steps
    if time.time() < deadline - 5:
        ht, ct = targeted_local_search(best_h, n, deadline - 2,
                                       top_candidates=20,
                                       step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interp_to_resolution(h_old, n_old, n_new):
    """Interpolate from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase A: Optimize at initial resolution
    alloc_a = min(250, (deadline - t0) * 0.25)
    if alloc_a > 30:
        h_opt, c_opt = multi_algorithm_optimize(h_init, n_init, t0 + alloc_a, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase B: Try higher resolution
    resolutions = [400, 500, 750]
    for n_try in resolutions:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 30
        alloc = min(180, remaining * 0.25)
        if alloc < 40:
            break
        h_interp = interp_to_resolution(best_h, len(best_h), n_try)
        h_opt, c_opt = multi_algorithm_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c - 1e-6:
            best_c = c_opt
            best_h = h_opt

    # Phase C: Random restarts at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 50:
        restarts = [
            (0.01, 1.0), (0.02, 1.0), (0.05, 1.0),
            (0.01, 0.5), (0.03, 0.5),
        ]
        for scale, scale2 in restarts:
            if time.time() > deadline - 30:
                break
            alloc = min(60, (deadline - time.time() - 25) * 0.3)
            if alloc < 20:
                break
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            h_opt, c_opt = multi_algorithm_optimize(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase D: Final fine-tuning on best solution
    if time.time() < deadline - 10:
        n_final = len(best_h)
        alloc_final = time.time() + min(60, deadline - time.time() - 3)
        # SLSQP at very high alpha
        if time.time() < alloc_final - 10:
            ht = slsqp_optimize(best_h, n_final, 100000000, maxiter=8000, ftol=1e-17)
            c = compute_c5(ht, n_final)
            if c < best_c:
                best_c = c
                best_h = ht
        # Final targeted search with tiny steps
        if time.time() < alloc_final - 3:
            ht, ct = targeted_local_search(best_h, n_final, alloc_final - 1,
                                           top_candidates=25,
                                           step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            if ct < best_c:
                best_c = ct
                best_h = ht

    # Phase E: Ultra-fine final pass
    if time.time() < deadline - 2:
        n_final = len(best_h)
        ht, ct = targeted_local_search(best_h, n_final, deadline - 1,
                                       top_candidates=30,
                                       step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if ct < best_c:
            best_c = ct
            best_h = ht

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
