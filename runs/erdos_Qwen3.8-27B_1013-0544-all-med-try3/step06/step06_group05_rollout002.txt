    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx

def smooth_max(c, alpha):
    """Smooth approximation of max using log-sum-exp."""
    m = c.max()
    if m * alpha > 500:  # prevent overflow
        shifted = alpha * (c - m)
        return m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha
    else:
        return (m + np.log(np.sum(np.exp(alpha * (c - m))))) / alpha

def obj_and_grad(x, n, alpha, beta):
    """
    Objective: smooth_max(correlation) + beta * integral of h(1-h)
    Returns (objective, gradient)
    """
    dx = 2.0 / n
    g = 1.0 - x
    c = np.correlate(x, g, mode='full') * dx
    
    # Smooth max
    obj = smooth_max(c, alpha)
    
    # Bimodality penalty: beta * sum(x*(1-x)) * dx = beta * (sum(x) - sum(x^2)) * dx
    # Since sum(x) is fixed at n/2, this is beta * (n/2 - sum(x^2)) * dx
    bimodal = beta * (np.sum(x * (1 - x)) * dx)
    obj += bimodal
    
    # Gradient of smooth max
    # p_k = exp(alpha * c_k) / sum(exp(alpha * c_j))
    m = c.max()
    shifted = alpha * (c - m)
    if m * alpha > 500:
        shifted = shifted - shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
