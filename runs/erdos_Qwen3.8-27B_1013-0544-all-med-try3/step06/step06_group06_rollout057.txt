    c = np.zeros(2*n - 1)

    # As first factor: m in [n-1-a, 2n-2-a], term change = -(1 - h[a+m-n+1])

    idx = m_hi - m_lo + 1  # should be n

        for m in range(max(0, m_lo), min(2*n-1, m_hi + 1)):
            j = a + m - n + 1
            if 0 <= j < n:
                c[m] += -(1 - h[j])
    
    # As second factor: m in [a, a+n-1], term change = +h[a-m+n-1]
    m_lo2 = a
    m_hi2 = a + n - 1
    for m in range(max(0, m_lo2), min(2*n-1, m_hi2 + 1)):

        if 0 <= i < n:

        j = b + m - n + 1
        if 0 <= j < n:

    # As second factor: m in [b, b+n-1], term change = -h[b-m+n-1]
    m_lo2 = b
