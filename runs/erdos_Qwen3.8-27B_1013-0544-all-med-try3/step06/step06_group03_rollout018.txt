```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    z = alpha * (c - m)
    zmax = z.max()
    if zmax > 500:
        return float(m + (zmax + np.log(np.sum(np.exp(z - zmax)))) / alpha)
    return float(m + np.log(np.sum(np.exp(z))) / alpha)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h.copy(), args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n), compute_c5(res.x, n)
    except:
        return h.copy(), compute_c5(h, n)
```

**Better local search: Gradient-guided**

Instead of trying random partners, use the gradient to determine which direction to move:
1. Compute the gradient of the smooth max
2. The gradient tells us which coordinates should increase/decrease
3. For each coordinate i, the optimal move is in the direction -grad[i]
4. Find the partner j that best matches (if grad[i] > 0, we want to decrease h[i], so move mass to a j where grad[j] < 0)

This is like a "greedy matching" approach:
- Sort coordinates by gradient (most positive first = should decrease, most negative = should increase)
- Match the top positive with the bottom negative
- Move a small amount between them
- Check if C₅ improved

This is much more targeted than random partner selection.

```python
    """
    Gradient-guided local search.
    Uses the smooth max gradient to determine optimal moves.
    """
    best_c = compute_c5(h, n)
    h = h.copy()
    dx = 2.0 / n
    n_full = 2*n - 1
    lags = np.arange(n_full) - (n-1)
    
    for delta in deltas:
        while time.time() < deadline:

            c = np.correlate(h, 1-h, mode='full') * dx
            m = c.max()

            zmax = z.max()
            if zmax > 500:
                z = z - zmax
            exp_z = np.exp(z)
            w = exp_z / exp_z.sum()

            H1[mask] = h[indices[mask]]

                if grad[i] - grad[j] < 1e-10:
                    break
                # Move delta from i to j
                if h[i] - delta >= 0 and h[j] + delta <= 1:
                    h[i] -= delta
                    h[j] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                if time.time() > deadline:
                    return h, best_c
            if not improved:
                break
    
    return h, best_c
```

This is O(n²) per gradient computation (for the H1, H2 matrices) and O(n) per move evaluation. With maybe 10 moves per gradient step, and the gradient step being O(n²), the total per "iteration" is O(n²). For n=250, that's 62500 operations for the gradient + 10 * 62500 for the C₅ evaluations = ~7 * 10^5 operations per iteration. Very fast!

But wait, I'm constructing H1 and H2 which are n × (2n-1) matrices. For n=250, that's 250 × 499 = 125K elements. Creating these matrices and doing the matrix-vector product is O(n²) which is fine.

Actually, I can avoid the matrix construction by using convolution:

```python
# H1 @ w = for each j: sum_k w[k] * h[j + lags[k]]
# This is the cross-correlation of h (padded) with w, evaluated at specific points.
# Equivalently, it's a "sliding dot product" of w with windows of h.

# Using np.convolve:
# np.convolve(h_pad, w, mode='valid') where h_pad is h zero-padded to length >= n + len(w) - 1
# h_pad needs to be at least n + (2n-1) - 1 = 3n - 2 long
# The result has length n + (2n-1) - 1 - (2n-1) + 1 = n
# Wait: len(convolve(a, v, 'valid')) = max(len(a), len(v)) - min(len(a), len(v)) + 1
# If len(a) = 3n-2, len(v) = 2n-1, result = 3n-2 - (2n-1) + 1 = n. 

# But is the result what we want?
# convolve(h_pad, w, 'valid')[j] = sum_k h_pad[j+k] * w[len(w)-1-k]

# np.correlate(h_pad, w, 'valid')[j] = sum_k h_pad[j+k] * w[k]
# If h_pad[j+k] = h[j + k - (n-1)] for the right range... this is getting complicated.
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    z = alpha * (c - m)
    zmax = z.max()
    if zmax > 500:
        z = z - zmax
        zmax = 0.0
    return float(m + (zmax + np.log(np.sum(np.exp(z)))) / alpha)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h.copy(), args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = project_feasible(res.x, n)
        return hx, compute_c5(hx, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def compute_smooth_grad(h, n, alpha):
    """Compute gradient of smooth max objective w.r.t. h."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    z = alpha * (c - m)
    zmax = z.max()
    if zmax > 500:
        z = z - zmax
    exp_z = np.exp(z)
    w = exp_z / exp_z.sum()
    
    n_full = 2 * n - 1
    lags = np.arange(n_full) - (n - 1)
    
    indices_plus = np.arange(n)[:, None] + lags[None, :]
    mask_p = (indices_plus >= 0) & (indices_plus < n)
    H1 = np.zeros((n, n_full), dtype=float)
    H1[mask_p] = h[indices_plus[mask_p]]
    
    indices_minus = np.arange(n)[:, None] - lags[None, :]
    mask_m = (indices_minus >= 0) & (indices_minus < n)
    H2 = np.zeros((n, n_full), dtype=float)
    H2[mask_m] = h[indices_minus[mask_m]]
    
    grad = dx * (1.0 - H1 @ w - H2 @ w)
    return c, w, grad

def gradient_guided_search(h, n, deadline, alpha, deltas):
    """Gradient-guided local search using smooth max subgradient."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for delta in deltas:
        while time.time() < deadline:
            try:
                c, w, grad = compute_smooth_grad(h, n, alpha)
            except Exception:
                break
            
            order = np.argsort(grad)
            improved = False
            n_trials = min(20, n // 2)
            
            for trial in range(n_trials):
                if time.time() > deadline:
                    return h, best_c
                i = n - 1 - trial
                j = trial
                if grad[i] - grad[j] < 1e-10:
                    break
                if h[i] - delta >= 0 and h[j] + delta <= 1:
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                
                # Also try reverse pairing
                i2 = n - 1 - trial
                j2 = n - 1 - trial
                if trial < n // 4:
                    i2 = trial
                    j2 = n - 1 - trial
                    if grad[i2] < grad[j2] - 1e-10:
                        if h[i2] + delta <= 1 and h[j2] - delta >= 0:
                            h[i2] += delta
                            h[j2] -= delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i2] -= delta
                                h[j2] += delta
            if not improved:
                break
    
    return h, best_c

def coordinate_search(h, n, deadline, deltas, n_partners=5):
    """Coordinate-wise local search with multiple partners."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for delta in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                old_hi = h[i]
                # Choose partners spread around the circle
                partners = set()
                partners.add((i + n // 2) % n)
                partners.add((i + 1) % n)
                partners.add((i + n // 3) % n)
                partners.add((i - 1 + n) % n)
                partners.add((i + 2 * n // 3) % n)

                    if j == i:
                        continue
                    # Move from j to i
                    if old_hi + delta <= 1.0 + 1e-12 and h[j] - delta >= -1e-12:
                        h[i] = min(1.0, old_hi + delta)

                            best_c = c
                            improved = True
                            old_hi = h[i]
                        else:
                            h[i] = old_hi
                            h[j] += delta
                    # Move from i to j
                    if h[i] - delta >= -1e-12 and h[j] + delta <= 1.0 + 1e-12:
                        h[i] = max(0.0, h[i] - delta)
                        h[j] = min(1.0, h[j] + delta)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta

            if not improved:
```
