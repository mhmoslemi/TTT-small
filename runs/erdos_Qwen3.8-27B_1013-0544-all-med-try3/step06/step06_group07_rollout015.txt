```python
    # Try to reduce overlap at k_worst
    # The gradient at k_worst tells us which direction reduces it
    grad = get_subgradient(h, k_worst, n)
    
    # Take a small step in the -grad direction
    # But check that other lags don't increase too much
    for step in [0.01, 0.005, 0.002, 0.001]:
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = max(corr_new) * dx
        if c5_new < c5:
            return h_new, c5_new
    return h, c5
```

Wait, but coordinate descent with projection is problematic because the projection changes other variables too. Let me think...

Actually, the projection to maintain the sum constraint will adjust other variables. So when we change h[i] by +delta, the projection will reduce some other variables by delta total (distributed among variables that can be reduced). This makes coordinate descent less clean.

A better approach: instead of projecting after each step, use the Lagrange multiplier for the sum constraint. The problem is:
minimize f_k(h) = ∑_i h_i(1-h_{i+l}) * dx
subject to ∑h_i = n/2, 0 ≤ h_i ≤ 1

The KKT conditions give: ∂f_k/∂h_i + λ = 0 (for i not at bounds)
i.e., (1-h_{i+l}) - h_{i-l} + λ = 0 (up to the dx factor)

So at the optimum of a single lag, we have:
h_{i+l} - h_{i-l} = λ + 1 for all free i

This is a linear system! For a given lag l, the optimal h (ignoring bounds) satisfies:
h_{i+l} - h_{i-l} = c (constant) for all valid i

This is a linear recurrence. For l=1: h_{i+1} - h_{i-1} = c, which means h alternates with period 2c... actually it means h grows linearly: h_i = c*i/2 + (alternating term).

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])


def get_subgradient(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def slsqp_optimize(h0, n, beta, maxiter=3000):
    ts = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts, n)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, ts)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)


def subgrad_loop(h, n, budget_fn, max_iter, lr_scale=1.0):
    ts = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Exponential moving average for adaptive step
    ema_c5 = best_c5
    ema_gsq = 1.0
    decay = 0.995

    for it in range(max_iter):
        if budget_fn() < 20:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        c5_curr = corr[k_worst] * (2.0 / n)

        grad = get_subgradient(h, k_worst, n)
        gsq = np.sum(grad ** 2)

        ema_c5 = decay * ema_c5 + (1 - decay) * c5_curr
        ema_gsq = decay * ema_gsq + (1 - decay) * gsq

        # Adaptive step: larger when far from lower bound, smaller when close
        gap = max(c5_curr - 0.35, 0.001)
        lr = lr_scale * gap / max(np.sqrt(ema_gsq), 0.01)
        lr = min(lr, 0.03)
        lr = max(lr, 0.0001)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, ts)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new

        # Occasionally check if we should restart from best
        if it > 0 and it % 5000 == 0:
            if compute_c5(h) > best_c5 + 0.005:
                h = best_h.copy()

    return best_h, best_c5


def averaging_subgrad(h, n, budget_fn, max_iter=20000, N_avg=50):
    """Averaged subgradient method: maintain average of iterates."""
    ts = n / 2.0
    dx = 2.0 / n
    h_curr = h.copy()
    h_avg = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_best_lower = 0.30

    for it in range(max_iter):
        if budget_fn() < 20:
            break

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_worst = np.argmax(corr)
        c5_curr = corr[k_worst] * dx

        grad = get_subgradient(h_curr, k_worst, n)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break

        # Polyak step size
        gap = max(c5_curr - c5_best_lower, 0.01)
        lr = gap / (norm_g ** 2 + 1e-8)
        lr = min(lr, 0.02)

        h_curr = h_curr - lr * grad
        h_curr = project_feasible(h_curr, ts)

        h_avg = h_avg + (h_curr - h_avg) / (it + 1)
        h_avg_proj = project_feasible(h_avg, ts)

        if (it + 1) % 500 == 0:
            c5_avg = compute_c5(h_avg_proj)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg_proj.copy()
            c5_curr_check = compute_c5(h_curr)
            if c5_curr_check < best_c5:
                best_c5 = c5_curr_check
                best_h = h_curr.copy()

    c5_final = compute_c5(project_feasible(h_avg, ts))
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = project_feasible(h_avg, ts).copy()

    return best_h, best_c5


def coordinate_refine(h, n, budget_fn, max_rounds=50):
    """Local coordinate-style search: nudge individual entries."""
    ts = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if budget_fn() < 15:
            break
        improved = False
        for i in range(n):
            if budget_fn() < 15:
                break
            if best_h[i] >= 0.999 or best_h[i] <= 0.001:
                continue
            for sign in [1, -1]:
                for delta in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    h_try = best_h.copy()
                    h_try[i] += sign * delta
                    if h_try[i] < 0 or h_try[i] > 1:
                        continue
                    h_try = project_feasible(h_try, ts)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-10:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break

    return best_h, best_c5
```
