import time
import numpy as np
from scipy.optimize import minimize


def c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def c_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def lag_eval(h, m, n, dx):
    if m == 0:
        return dx * float(np.sum(h * (1.0 - h)))
    if m > 0:
        a = h[:n - m]
        b = h[m:]
        return dx * float(np.sum(a * (1.0 - b)))
    u = -m
    a = h[u:]
    b = h[:n - u]
    return dx * float(np.sum(a * (1.0 - b)))


def lag_grad(h, m, n, dx):
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
        return g * dx
    if m > 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        u = -m
        g[u:] += (1.0 - h[:n - u])
        g[:n - u] -= h[u:]
    return g * dx


def smooth_max(vals, beta):
    m = float(np.max(vals))
    sh = vals - m
    mx = float(np.max(sh))
    if mx > 500.0:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / beta)
    return float(m + np.log(np.sum(np.exp(sh))) / beta)


def top_fvec(cvec, e):
    e = min(e, len(cvec))
    arr = np.argpartition(cvec, -e)[-e:]
    order = np.argsort(cvec[arr])[::-1]
    return arr[order]


def project_feas(h, n):
    target = 0.5 * n
    v = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    lo = float(v.max()) + 1.0
    hi = float(v.min()) - 1.0
    if lo >= hi:
        hi = lo - 1.0
        lo = hi + 1.0
    F0 = float(np.sum(np.clip(v - lo, 0.0, 1.0)))
    F1 = float(np.sum(np.clip(v - hi, 0.0, 1.0)))
    F_lo, F_hi = min(F0, F1), max(F0, F1)
    for _ in range(80):
        t = 0.5 * (lo + hi)
        F = float(np.sum(np.clip(v - t, 0.0, 1.0)))
        Fc = min(F, F_hi)
        if Fc > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    v = np.clip(v - t, 0.0, 1.0)
    diff = target - float(np.sum(v))
    if abs(diff) > 1e-13:
        dm = (v > 1e-9) & (v < 1.0 - 1e-9)
        cnt = int(np.sum(dm))
        if cnt > 0:
            v[dm] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
    return v


def interp_to(h_old, n_new):
    x_old = np.arange(len(h_old), dtype=float)
    x_new = np.arange(n_new, dtype=float) * (len(h_old) - 1) / max(1, n_new - 1)
    v = np.interp(x_new, x_old, h_old)
    return project_feas(v, n_new)


def analyze_h(n, cycles, kind, amp, phase, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    w = np.pi * float(cycles)
    wave = np.cos(w * x + phase)
    if kind == 0:
        base = 0.5 + amp * wave
    elif kind == 1:
        sq = 2.0 * np.clip(wave, 0.0, 1.0) - 1.0
        base = 0.5 + 0.5 * amp * sq
    else:
        base = 0.5 * (1.0 + 2.0 * min(1.0, amp) * wave)
    base = np.clip(base, 0.0, 1.0)
    return project_feas(base, n)


def pg_refine(h_in, n, t_end, label):
    h = np.asarray(h_in, dtype=float).copy()
    mid = n - 1
    cvec = c_vec(h, n)
    jr = top_fvec(cvec, 8)
    valsE = cvec[jr]
    cmax = float(cvec.max())
    cur_loss = float(smooth_max(valsE, 6.0))
    best_loss = cur_loss
    beta = 6.0
    alpha = 0.05
    gma = np.zeros(n)
    stall = 0
    it = 0
    while time.time() < t_end:
        it += 1
        wv = valsE - valsE.max()
        ew = np.exp(np.minimum(wv * beta, 500.0))
        wsum = float(np.sum(ew))
        g = None
        for q in range(len(jr)):
            m = int(jr[q]) - mid
            gm = lag_grad(h, m, n, 1.0)
            wq = ew[q] / wsum
            g = gm * wq if g is None else (g + gm * wq)
        g = g * (2.0 / n)
        g = g - g.mean()
        gma = 0.60 * gma + 0.40 * g
        h2 = h - alpha * gma
        h2 = project_feas(h2, n)
        vals2 = np.array([lag_eval(h2, int(j) - mid, n, 2.0 / n) for j in jr])
        loss2 = smooth_max(vals2, beta)
        if loss2 < cur_loss - 1e-12:
            h = h2
            valsE = vals2
            cur_loss = loss2
            if cur_loss < best_loss:
                best_loss = cur_loss
            stall = 0
            alpha = min(0.45, alpha * 1.20)
            beta = min(90.0, beta * 1.012 + 0.004)
        else:
            alpha *= 0.55
            stall += 1
            if stall >= 600:
                break
            if alpha < 5e-9:
                break
        if it % 14 == 0 or stall > 120:
            cvec = c_vec(h, n)
            cmax = float(cvec.max())
            valsE = cvec[jr]
            cur_loss = smooth_max(valsE, beta)
            if cmax > float(valsE.max()) + 3e-7:
                jr = top_fvec(cvec, 16)
                valsE = cvec[jr]
        if cmax <= 0.38062 and beta >= 60.0 and stall >= 300:
            break
    return h


def _cb_f(xv, n, jk):
    h = xv[:n]
    m = int(jk) - (n - 1)
    return lag_eval(h, m, n, 2.0 / n) - xv[n]


def _cb_j(xv, n, jk):
    h = xv[:n]
    m = int(jk) - (n - 1)
    g = lag_grad(h, m, n, 2.0 / n)
    out = np.empty(n + 1)
    out[:n] = g
    out[n] = -1.0
    return out


def _s_obj(xv, n):
    return float(xv[n])


def slsqp_round(h_in, n, t_end, label, max_constr=12, maxiter=1500):
    cvec = c_vec(h_in, n)
    jr = top_fvec(cvec, max_constr)
    js = set(int(x) for x in jr)
    js.add(n - 1)
    js = list(js)[: max_constr + 1]
    cps = [lag_eval(h_in, int(j) - (n - 1), n, 2.0 / n) for j in js]
    r0 = max(cps) + 0.002
    x0 = np.empty(n + 1)
    x0[:n] = h_in
    x0[n] = r0
    bds = [(0.0, 1.0)] * n + [(-0.5, 5.0)]
    cons = []
    for jk in js:
        cons.append({'type': 'ineq', 'fun': _cb_f, 'jac': _cb_j, 'args': (n, jk)})
    try:
        res = minimize(_s_obj, x0, args=(n,), method='SLSQP', bounds=bds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    except Exception:
        return h_in
    if res.x is None:
        return h_in
    h2 = np.clip(res.x[:n], 0.0, 1.0)
    h2 = project_feas(h2, n)
    return h2


def scan_refine(h_in, n, t_end, label):
    h = np.asarray(h_in, dtype=float).copy()
    mid = n - 1
    cvec = c_vec(h, n)
    ccur = float(cvec.max())
    cap = 0.25
    it = 0
    while time.time() < t_end:
        it += 1
        jr = top_fvec(cvec, 8)
        moved = False
        for jj in jr:
            m = int(jj) - mid
            if m == 0:
                continue
            if m > 0:
                L = np.arange(0, n - m)
                hL = h[0:n - m]
                hR = h[m:n]
                B = -(1.0 - hR) - hL
                nkm = n - 2 * m
                if nkm > 0:
                    B[:nkm] += (1.0 - h[2 * m:n])
                    B[m:] += h[:nkm]
            else:
                u = -m
                L = np.arange(u, n)
                hL = h[u:n]
                hR = h[0:n - u]
                B = 2.0 * hR - 1.0
                nku = n - 3 * u
                if nku > 0:
                    B[:nku] += (1.0 - hL[:nku]) + hR[:nku]
            dm = np.minimum(hL, 1.0 - hR)
            dm = np.minimum(dm, cap)
            ok = (dm > 1e-9) & (B < -1e-7)
            if not np.any(ok):
                continue
            dA = np.zeros(len(L))
            dA[ok] = np.minimum(dm[ok], -0.5 * B[ok])
            D = dA * (B + dA)
            D = np.where(dA > 1e-9, D, 0.0)
            best = float(np.max(D))
            if best < 1e-9:
                continue
            qa = int(np.argmax(D))
            kk = int(L[qa])
            d1 = float(dA[qa])
            h[kk] -= d1
            h[kk + m] += d1
            moved = True
            cvec = c_vec(h, n)
            c2 = float(cvec.max())
            if c2 < ccur - 1e-12:
                ccur = c2
        if not moved:
            break
        if it > 400:
            break
    return h

        for cf in [0.618, 1.0, 1.236, 1.618, 2.0, 2.618, 3.236, 1.45, 0.786, 2.36]:
            for kd in [0, 1, 2]:
                for am in [0.45, 0.8, 1.0]:
                    for ph in [0.0, 1.26, 2.52]:
                        ha = analyze_h(n_a, cf, kd, am, ph, rng)
                        ca = c5_full(ha, n_a)
                        cands.append((ca, ha))
        cands.sort(key=lambda zv: zv[0])
        if cands:
            besth, bestc = cands[0][1].copy(), cands[0][0]
        else:
            besth = np.full(300, 0.5)
            bestc = c5_full(besth, 300)

    stages = []
    if h_p is not None:
        stages.append((n_p, 100))
    for nn in [450, 600, 750, 900]:
        if nn not in [s[0] for s in stages]:
            stages.append((nn, 120 if nn <= 600 else 140))
    done = set()
    for ns, als in stages:
        if ns in done:
            continue
        done.add(ns)
        rem = dl - time.time()
        if rem <= 95:
            break
        allow = min(als, rem - 55)
        if allow < 55:
            break
        te = time.time() + allow
        nl = len(besth) if besth is not None else 0
        if ns == nl:
            hs = besth.copy()
        else:
            hs = interp_to(besth, ns)
        hs, cs = process_stage(hs, ns, te, "n%d" % ns)
        print("stage n=%d c=%.8f remain=%.0f" % (ns, cs, dl - time.time()))
        if cs < bestc:
            bestc = cs
            besth = hs.copy()

    for rr in range(3):
        if time.time() > dl - 80:
            break
        ns = len(besth)
        hp = besth + rng.normal(0.0, 0.010, ns)
        hp = project_feas(hp, ns)
        te = min(dl - 55, time.time() + 45)
        hs, cs = process_stage(hp, ns, te, "restart%d" % rr)
        print("restart %d c=%.8f" % (rr, cs))
        if cs < bestc:
            bestc = cs
            besth = hs.copy()

    k2 = len(besth) + len(besth) // 2
    if k2 >= 200 and time.time() < dl - 120 and c5_full(besth, len(besth)) <= 0.38092:
        te = min(dl - 35, time.time() + 85)
        hi2 = interp_to(besth, k2)
        hs2, cs2 = process_stage(hi2, k2, te, "k2")
        print("upsample n=%d c=%.8f" % (k2, cs2))
        if cs2 < bestc:
            bestc = cs2
            besth = hs2.copy()

    nb = len(besth)
    for _f in range(2):
        if time.time() > dl - 20:
            break
        rem2 = dl - time.time()
        if rem2 < 14:
            break
        maxiter = 2000 if nb <= 600 else 900
        hc = slsqp_round(besth, nb, dl - 5, "polish", max_constr=12, maxiter=maxiter)
        hc = project_feas(hc, nb)
        cc = c5_full(hc, nb)
        if cc < bestc:
            bestc = cc
            besth = hc.copy()

    nfr = len(besth)
    for kk in range(nfr):
        if time.time() > dl - 6:
            break
        s_off = (kk + 0.5) + 0.5
        xb = np.arange(nfr) + s_off
        i0 = np.floor(xb).astype(int) % nfr
        i1 = (i0 + 1) % nfr
        fr = np.mod(xb, 1.0)
        hc = besth[i0] * (1.0 - fr) + besth[i1] * fr
        hc = np.clip(hc, 0.0, 1.0)
        hc = project_feas(hc, nfr)
        cc = c5_full(hc, nfr)
        if cc < bestc:
            bestc = cc
            besth = hc.copy()

    nb = len(besth)
    if time.time() < dl - 3:
        hsub = besth.copy()
        d0 = float(np.sum(hsub) - nb / 2.0)
        if abs(d0) > 0:
            msk = (hsub > 1e-10) & (hsub < 1.0 - 1e-10)
            cnt = int(np.sum(msk))
            if cnt > 0:
                hsub[msk] += d0 / cnt
        besth = np.clip(hsub, 0.0, 1.0)
        bestc = c5_full(besth, nb)

    print("final n=%d c=%.9f" % (len(besth), bestc))
    if besth is None or bestc == float('inf'):
        hfb = np.full(300, 0.5)
        return hfb, 0.5, 300
    return besth, bestc, int(len(besth))
