```python
def hard_max_gradient(h, n, tolerance=1e-4):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    active = np.where(c > m - tolerance)[0]
    # Average gradient over active lags
    grad = np.zeros(n)
    w = np.zeros(len(active))
    w[:] = 1.0 / len(active)
    for idx in active:
        d = idx - (n - 1)
        contrib = np.zeros(n)
        if d == 0:
            contrib = 1 - 2 * h
        elif d > 0:
            if d < n:
                contrib[:n-d] += (1 - h[d:])
                contrib[d:] -= h[:n-d]
        else:
            dd = -d
            if dd < n:
                contrib[dd:] += (1 - h[:n-dd])
                contrib[:n-dd] -= h[dd:]
        grad += (1.0 / len(active)) * contrib
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                h[:] = 0.5
                break
            scale = diff / tot
            h += scale * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                h[:] = 0.5
                break
            h *= (target_sum / tot)
    return np.clip(h, 0.0, 1.0)


def compute_soft_max_info(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w_sum = w.sum()
    w /= w_sum
    obj = m + (max_s + np.log(w_sum)) / alpha
    return obj, w, c


def compute_gradient(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    two_n_minus_1 = 2 * n - 1
    for k in range(two_n_minus_1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += dx * wk * (1.0 - 2.0 * h)
        elif d > 0:
            if d >= n:
                continue
            grad[:n - d] += dx * wk * (1.0 - h[d:])
            grad[d:] -= dx * wk * h[:n - d]
        else:
            dd = -d
            if dd >= n:
                continue
            grad[dd:] += dx * wk * (1.0 - h[:n - dd])
            grad[:n - dd] -= dx * wk * h[dd:]
    return grad


def nesterov_gradient_descent(h, n, deadline, alphas, max_iter_per_alpha=3000):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = 0.3
        v = np.zeros(n)
        step = 0.5
        obj, w, c = compute_soft_max_info(h, n, alpha)
        
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            grad = compute_gradient(h, n, w)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-14:
                break
            
            v = mu * v - step * grad
            best_x = h.copy()
            best_f = obj
            s = step
            success = False
            for _ls in range(25):
                x_new = h + v
                x_new = project_feasible(x_new, n)
                obj_new, w_new, _ = compute_soft_max_info(x_new, n, alpha)
                if obj_new < best_f - 1e-18:
                    best_x = x_new
                    best_f = obj_new
                    success = True
                    break
                s *= 0.5
                v = mu * v - s * grad
            
            if not success:
                step *= 0.5
                if step < 1e-8:
                    break
                continue
            
            h = best_x
            v = mu * v - step * grad
            step = min(step * 1.2, 5.0)
            w = w_new
            
            c_now = compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=None, max_passes=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if radius is not None:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                else:

                    j_hi = n
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)

                            best_c = c

                        else:

def slsqp_refine(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def obj_func(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return val
    
    def grad_func(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            d = k - (n - 1)
            if d == 0:
                g += dx * wk * (1.0 - 2.0 * x)
            elif d > 0:
                if d < n:
                    g[:n - d] += dx * wk * (1.0 - x[d:])
                    g[d:] -= dx * wk * x[:n - d]
            else:
                dd = -d
                if dd < n:
                    g[dd:] += dx * wk * (1.0 - x[:n - dd])
                    g[:n - dd] -= dx * wk * x[dd:]
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

    n_ones = int(round(target_sum))
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_ones]] = 1.0
    c_bin = compute_c5(h_bin, n)
    return h_bin, c_bin
```
