    k = np.arange(n)

    L1 = k - i
    v1 = (L1 >= -(n-1)) & (L1 <= n-1)
    flip_up[v1, L1[v1] + (n-1)] -= 1.0

                    cp3 = (j_s - i_s) + (n-1)
                    if 0 <= cp3 < 2*n-1:
                        delta[cp3] += 1.0
                    corr = corr + delta
                    h[i_s] = 1.0
                    h[j_s] = 0.0
                ones_arr = np.where(h == 1)[0]
                zeros_arr = np.where(h == 0)[0]
                cur_max = corr.max()
                stuck = 0
            else:
                break
        
        result = find_best_improving_swap(flip_up, h, corr, n, ones_arr, zeros_arr)
        if result is None:
            stuck += 1
            continue
        
        i, j, new_corr = result
        # Apply swap: update flip_up, then corr, then h
        k = np.arange(n)
        L1 = k - i
        v1 = (L1 >= -(n-1)) & (L1 <= n-1)
        flip_up[v1, L1[v1] + (n-1)] -= 1.0
        L2 = i - k
        v2 = (L2 >= -(n-1)) & (L2 <= n-1)
        flip_up[v2, L2[v2] + (n-1)] -= 1.0
        L3 = k - j
        v3 = (L3 >= -(n-1)) & (L3 <= n-1)
        flip_up[v3, L3[v3] + (n-1)] += 1.0
        L4 = j - k
        v4 = (L4 >= -(n-1)) & (L4 <= n-1)
        flip_up[v4, L4[v4] + (n-1)] += 1.0
        
        delta = flip_up[i].copy() - flip_up[j].copy()
        delta[n-1] -= 2.0
        cp2 = (i - j) + (n-1)
        if 0 <= cp2 < 2*n-1:
            delta[cp2] += 1.0
        cp3 = (j - i) + (n-1)
        if 0 <= cp3 < 2*n-1:
            delta[cp3] += 1.0
        corr = corr + delta
        h[i] = 1.0
        h[j] = 0.0
        cur_max = corr.max()
        stuck = 0
        
        if cur_max * (2.0/n) < best_c:
            best_c = cur_max * (2.0/n)
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
