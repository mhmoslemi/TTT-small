def lse_jac(x, n, lags, alpha):
   r = np.correlate(x, 1.0-x, 'full')
   vals = r[lags]
   mx = vals.max()
   e = np.exp(alpha*(vals-mx))
   w = e / e.sum()
   g = np.zeros(n)
   cnr = n-1
   for idx, wt in zip(lags, w):
      l = int(idx) - cnr
      if abs(wt) < 1e-16: continue
      if l >= 0:
         i = slice(l, n)
         j = slice(0, n-l)
         g[i] += wt * (1.0 - x[j])
         g[j] -= wt * x[i]
      else:
         m = -l
         i = slice(0, n-m)
         j = slice(m, n)
         g[i] += wt * (1.0 - x[j])
         g[j] -= wt * x[i]
   return g

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n, dtype=float)
            tmp[:h.size] = h
            h = tmp
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)

    target = 0.5 * float(n)
    for _ in range(100):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h *= (1.0 + diff / s)

    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:

def compute_c5(h, n):

    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r)) * dx


def make_top(r, k=12):
    m = len(r)
    if m <= k:
        idx = np.argsort(r)[::-1]
    else:
        idx = np.argpartition(r, -k)[-k:]
        idx = idx[np.argsort(r[idx])]
    idx = idx[::-1]
    out = []
    for i in idx:
        out.append((float(r[i]), int(i)))
    return out


def add_roots(cand, a, b, c, lo, hi):
    if abs(a) < 1e-14:
        if abs(b) > 1e-14:
            rr = -c / b
            if rr < lo:
                if lo - rr < 1e-10:
                    cand.append(lo)
            elif rr > hi:
                if rr - hi < 1e-10:
                    cand.append(hi)
            else:
                cand.append(rr)
        return

    disc = b * b - 4.0 * a * c
    if disc < -1e-10:
        return
    if disc < 0.0:
        disc = 0.0

    s = math.sqrt(disc)
    den = 2.0 * a
    r1 = (-b + s) / den
    r2 = (-b - s) / den

    for rr in (r1, r2):
        if rr < lo:
            if lo - rr < 1e-10:
                cand.append(lo)
        elif rr > hi:
            if rr - hi < 1e-10:
                cand.append(hi)
        else:
            cand.append(rr)

def pair_descent_pass(h, r, top, n, deadline, tol=1e-13):

    if len(top) == 0:
        cmax = float(np.max(r))
    else:

def pair_polish(h, n, deadline):

def lse_value(x, n, lags, alpha):
    x = np.asarray(x, dtype=float)
    r = np.correlate(x, 1.0 - x, mode="full")
    cvals = r * (2.0 / float(n))
    vals = cvals[lags]
    mx = float(np.max(vals))
    sh = float(alpha) * (vals - mx)
    sh = np.clip(sh, -700.0, 0.0)
    return (mx + float(np.log(np.sum(np.exp(sh))))) / float(alpha)


def lse_grad(x, n, lags, alpha):
    x = np.asarray(x, dtype=float)
    r = np.correlate(x, 1.0 - x, mode="full")
    cvals = r * (2.0 / float(n))
    vals = cvals[lags]
    mx = float(np.max(vals))
    sh = float(alpha) * (vals - mx)
    sh = np.clip(sh, -700.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        w = np.ones_like(e) / float(len(e))
    else:
        w = e / s

    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    cnr = n - 1

    for idx in range(len(lags)):
        wt = float(w[idx])
        if wt < 1e-14:
            continue
        lag = int(lags[idx]) - cnr
        scale = wt * dx

        if lag >= 0:
            i = slice(lag, n)
            j = slice(0, n - lag)
            g[i] += scale * (1.0 - x[j])
            g[j] -= scale * x[i]
        else:
            m = -lag
            i = slice(0, n - m)
            j = slice(m, n)
            g[i] += scale * (1.0 - x[j])
            g[j] -= scale * x[i]

    return g

        r = np.correlate(h, 1.0 - h, mode="full")
        cvals = r * (2.0 / float(n))
        maxc = float(np.max(cvals))

        lags = np.where(cvals >= maxc - margin)[0]
        if len(lags) == 0:
            lags = np.array([int(np.argmax(cvals))])
        elif len(lags) > 100:
            order = np.argsort(cvals[lags])[-100:]
            lags = lags[order]

        for alpha in alphas:
            if calls >= max_calls:
                break
            if time_func() > deadline - 12.0:
                break

            calls += 1
            try:
                bounds = [(0.0, 1.0)] * n
                constraints = [
                    {"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (n,)}
                ]
                res = minimize(
                    lse_value,
                    h,
                    args=(n, lags, float(alpha)),
                    method="SLSQP",
                    jac=lse_grad,
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": maxiter, "ftol": 1e-11, "disp": 0},
                )
                if res is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
                    ht = project_feasible(res.x, n)
                    c = compute_c5(ht, n)
                    if c < best_c - 1e-15:
                        best_h = ht.copy()
                        best_c = c
                        h = best_h.copy()
            except Exception:
                pass

    return best_h, best_c


def global_polish(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    time_func = time.time

    stuck = 0
    while time_func() < deadline - 20.0:
        r = np.correlate(h, 1.0 - h, mode="full")
        top = make_top(r, 12)

        improved, completed = pair_descent_pass(
            h, r, top, n, deadline - 20.0, tol=1e-13
        )

        if improved:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            h = best_h.copy()
            stuck = 0
            continue

        stuck += 1

        if stuck <= 6 and time_func() < deadline - 50.0:
            smooth_deadline = min(deadline - 35.0, time_func() + 28.0)
            try:
                hs, cs = slsqp_active(best_h, n, smooth_deadline, rng)
                if cs < best_c - 1e-15:
                    best_c = cs
                    best_h = hs.copy()
                    h = best_h.copy()
                    stuck = 0
                    continue
            except Exception:
                pass

        if time_func() < deadline - 30.0:
            scale = 0.02 + 0.02 * float(stuck % 4) + 0.03 * float(rng.rand())
            if scale > 0.20:
                scale = 0.20

            hp = best_h + scale * rng.normal(0.0, 1.0, n)
            hp = project_feasible(hp, n)

            polish_dl = min(deadline - 20.0, time_func() + min(25.0, 8.0 + 2.0 * stuck))
            hp, cp = pair_polish(hp, n, polish_dl)

            if cp < best_c - 1e-15:
                best_c = cp
                best_h = hp.copy()
                h = best_h.copy()
                stuck = 0
                continue

        if stuck > 10:
            break

    if time_func() < deadline - 3.0:
        hh, cc = pair_polish(best_h, n, deadline - 2.0)
        if cc < best_c:
            best_c = cc
            best_h = hh
    else:
        best_c = compute_c5(best_h, n)

    return best_h, best_c


def structural_initial(n, rng):
    x = np.linspace(0.0, 2.0, n)
    m1 = int(rng.randint(3, 9))
    m2 = int(rng.randint(10, 25))
    p1 = float(rng.uniform(0.0, 2.0 * math.pi))
    p2 = float(rng.uniform(0.0, 2.0 * math.pi))
    h = (
        0.5
        + 0.35 * np.sin(np.pi * m1 * x / 2.0 + p1)
        + 0.25 * np.sin(np.pi * m2 * x / 2.0 + p2)
        + 0.05 * rng.normal(0.0, 1.0, n)
    )
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def interpolate_to(h_src, n_src, n_dst):
    if n_src == n_dst:
        return np.array(h_src, dtype=float, copy=True)

    x_src = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / float(n_src))
    x_dst = (np.arange(n_dst, dtype=float) + 0.5) * (2.0 / float(n_dst))
    h_src = np.asarray(h_src, dtype=float)
    return np.interp(x_dst, x_src, h_src)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    if deadline < t0 + 10.0:
        deadline = t0 + max(5.0, float(budget_s))

    rng = np.random.RandomState(seed)

    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=float).copy()
        n0 = int(len(h0))
    else:
        n0 = 250
        h0 = structural_initial(n0, rng)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    target_c = 0.3808

    remaining = deadline - t0
    if remaining > 60.0:
        main_alloc = min(350.0, max(30.0, remaining * 0.35))
        main_dl = t0 + main_alloc
        if main_dl > deadline:
            main_dl = deadline
        if main_dl - t0 > 15.0:
            try:
                hnew, cnew = global_polish(h0, n0, main_dl, rng)
                if cnew < best_c:
                    best_c = cnew
                    best_h = hnew
            except Exception:
                try:
                    hnew, cnew = pair_polish(h0, n0, main_dl)
                    if cnew < best_c:
                        best_c = cnew
                        best_h = hnew
                except Exception:
                    pass

    raw_candidates = [n0]
    for delta in (1, -1, 2, -2, 4, -4, 8, -8, 12, -12):
        raw_candidates.append(n0 + delta)

    for factor in (1.15, 1.25, 1.4, 1.6, 1.8, 2.0, 0.8, 0.65):
        raw_candidates.append(int(round(n0 * factor)))

    for val in (120, 160, 200, 240, 260, 300, 350, 400, 500, 600, 700):
        raw_candidates.append(val)

    cands = []
    seen = set()
    for val in raw_candidates:
        val = int(val)
        if val < 80 or val > 720:
            continue
        if val in seen:
            continue
        seen.add(val)
        cands.append(val)

    for n_try in cands[1:]:
        if time.time() > deadline - 80.0:
            break
        if best_c <= target_c:
            break
        if n_try == len(best_h):
            continue

        alloc = min(80.0, max(20.0, (deadline - time.time()) * 0.18))
        if alloc < 25.0:
            break

        try:
            start_h = interpolate_to(best_h, len(best_h), n_try)
            start_h = project_feasible(start_h, n_try)
            c_start = compute_c5(start_h, n_try)
            if c_start > best_c + 0.05:
                continue

            hnew, cnew = global_polish(start_h, n_try, time.time() + alloc, rng)
            if cnew < best_c:
                best_c = cnew
                best_h = hnew
        except Exception:
            continue

    time_func = time.time
    while time_func() < deadline - 60.0:
        if best_c <= target_c:
            break

        n_best = int(len(best_h))
        scale = 0.02 + 0.06 * float(rng.rand())
        hp = best_h + scale * rng.normal(0.0, 1.0, n_best)
        hp = project_feasible(hp, n_best)

        alloc = min(50.0, max(12.0, (deadline - time.time()) * 0.20))
        if alloc < 15.0:
            break

        try:
            hnew, cnew = global_polish(hp, n_best, time.time() + alloc, rng)
            if cnew < best_c:
                best_c = cnew
                best_h = hnew
        except Exception:
            try:
                hnew, cnew = pair_polish(hp, n_best, time.time() + alloc)
                if cnew < best_c:
                    best_c = cnew
                    best_h = hnew
            except Exception:
                break

    if time.time() < deadline - 3.0:
        best_h, best_c = pair_polish(best_h, len(best_h), deadline - 2.0)
    else:
        best_c = compute_c5(best_h, len(best_h))

    best_h = enforce_exact_sum(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
