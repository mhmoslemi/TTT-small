```python
def simulated_annealing(h, n, deadline, seed=42, max_iters=100000):
    """Simulated annealing for binary sequences."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    m_range = np.arange(-(n-1), n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    T = 0.01
    T_min = 1e-8
    cooling = 0.999
    
    n_iters = 0
    while time.time() < deadline and T > T_min:
        if n_iters % 1000 == 0:
            T *= cooling
        
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta
        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range
        
        delta = np.zeros(2*n-1)
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        n_iters += 1
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    cons_fun = lambda x: x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-14:
                break
            h = h + diff / rs * room
        else:
            if s < 1e-14:
                break
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def thue_morse_bin(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') & 1
    return t

    phi = (1 + np.sqrt(5)) / 2
    h = np.zeros(n)
    for i in range(n):
        h[i] = int(np.floor((i+1)*phi)) - int(np.floor(i*phi))
    return h

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

    """Compute delta in correlation when swapping h[i]:0->1, h[j]:1->0."""
    delta = np.zeros(2*n-1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
```

So the problem reduces to minimizing max_k [(n-k)*0.25 - sum_i u[i+k]*u[i]] which is equivalent to minimizing max_k [-sum_i u[i+k]*u[i]] (since (n-k)*0.25 is fixed for each k).

This is getting complicated. Let me just focus on making the SLSQP work better.

**Key insight for improving SLSQP**: The log-sum-exp approximation with large alpha is essentially trying to solve a "minimize the max" problem. A better approach might be to use **epigraph formulation**:

minimize t
subject to corr_k(h) * dx ≤ t for all k
0 ≤ h[i] ≤ 1
sum(h) = n/2

This is a quadratic constraint problem (since corr_k is quadratic in h). We could potentially solve this with a QP solver or use the method of multipliers.

But corr_k(h) = sum_i h[i+k] - sum_i h[i+k]*h[i] is not a convex quadratic constraint (the -h[i+k]*h[i] term makes it non-convex).

However, we can write it as:
h[i+k] - h[i+k]*h[i] ≤ t/dx (for each i, summed over i)

The left side h[i+k]*(1-h[i]) is not convex either (it's a bilinear term).

So we're stuck with non-convex optimization. The best we can do is local optimization.

**Let me focus on making the local optimization faster and more effective:**

1. **Use a smarter parameterization**: Instead of optimizing all 250 variables, note that the optimal solution is likely close to binary. So we can:
   - Identify which positions are "definitely 0" or "definitely 1"
   - Only optimize the "border" positions (which are fractional)
   - This reduces the effective dimension

2. **Use a two-level optimization**:
   - Outer loop: choose which positions are fractional
   - Inner loop: optimize the fractional values

3. **Use the structure of the correlation**: The correlation at lag k depends on the overlap of h and its shift by k. If we know the "active" lags (those near the max), we can focus optimization on reducing those.

Let me implement approach 3:

```python
def active_lag_optimization(h, n, deadline, seed=42):
    """Optimize h by focusing on the active (max) lags."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    
    while time.time() < deadline:
        corr = np.correlate(h, 1-h, mode='full') * dx
        max_corr_idx = np.argmax(corr)
        max_val = corr[max_corr_idx]
        
        # Find all lags within epsilon of the max
        eps = 0.005 * dx  # small threshold
        active = np.where(corr > max_val - eps)[0]
        
        # For each active lag, compute gradient of corr_k with respect to h
        # And try to reduce them simultaneously
        # Use a simple gradient step with projection
        
        # Compute subgradient of max: average gradient over active lags
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n-1)
            # corr_k = sum_i h[i+k]*(1-h[i])
            # d(corr_k)/d(h[m]) = (1-h[m-k])*1_{m-k valid} - h[m+k]*1_{m+k valid}
            for m in range(n):
                g = 0.0
                if 0 <= m-k < n:
                    g += (1 - h[m-k])
                if 0 <= m+k < n:
                    g -= h[m+k]
                grad[m] += g / len(active)
        
        # Gradient descent step
        lr = 0.01
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        # Project onto sum constraint
        target = n / 2.0
        h_new = h_new + (target - h_new.sum()) / n
        h_new = np.clip(h_new, 0, 1)
        # Re-project
        for _ in range(100):
            diff = target - h_new.sum()
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                room = 1 - h_new
                s = room.sum()
                if s < 1e-10: break
                h_new = h_new + diff/s * room
            else:
                s = h_new.sum()
                if s < 1e-10: break
                h_new = h_new + diff/s * h_new
        h_new = np.clip(h_new, 0, 1)
        
        new_max = np.max(np.correlate(h_new, 1-h_new, mode='full')) * dx
        if new_max < max_val:
            h = h_new
        else:
            # Try smaller step
            lr = lr * 0.5
            if lr < 1e-8:
                break
    
    return h, np.max(np.correlate(h, 1-h, mode='full')) * dx
```

This can be computed as:
grad_k = (1-h) shifted by k - h shifted by -k (with appropriate masking)

Specifically:
grad_k[m] = (1-h)[m-k] - h[m+k] (with zero padding)

This is just two shifted arrays! So for all active lags:
grad = sum_{k in active} [(1-h) shifted by k - h shifted by -k] / |active|
