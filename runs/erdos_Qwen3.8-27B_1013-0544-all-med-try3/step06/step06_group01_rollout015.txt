def compute_delta(i, j, h, n):
    delta = np.zeros(2*n-1)
    # Position j (1→0): affects k in [j, j+n-1]
    k_range_j = np.arange(j, min(2*n-1, j+n))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(i, j, h, n):
    """Compute the change in correlation when h[i]: 0→1 and h[j]: 1→0."""
    delta = np.zeros(2 * n - 1)
    # Position j (1→0): affects corr[k] for k in [j, j+n-1]
    lo_j = max(0, j)
    hi_j = min(2 * n - 1, j + n)
    if lo_j < hi_j:
        ks = np.arange(lo_j, hi_j)
        h_idx = j + (n - 1) - ks
        delta[lo_j:hi_j] -= (1.0 - h[h_idx])
    # Position i (0→1): affects corr[k] for k in [i, i+n-1]
    lo_i = max(0, i)
    hi_i = min(2 * n - 1, i + n)
    if lo_i < hi_i:
        ks = np.arange(lo_i, hi_i)
        h_idx = i + (n - 1) - ks
        delta[lo_i:hi_i] += (1.0 - h[h_idx])
    return delta

def binary_local_search(h, n, deadline, max_time_per_sweep=0.01):
    """Fast binary local search with vectorized delta computation."""
    target = n // 2
    h = (h >= 0.5).astype(float)
    if h.sum() != target:
        idx = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_max = float(corr.max())
    
    rng = np.random.RandomState(42)
    
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Shuffle for diversity
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in zeros:
            if time.time() > deadline:
                return h, cur_max * dx
            for j in ones:
                delta = compute_swap_delta(int(i), int(j), h, n)
                new_max = (corr + delta).max()
                if new_max < cur_max - 1e-12:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_max = float(new_max)
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h, cur_max * dx

def sa_local_search_cycle(h, n, deadline, sa_time_fraction=0.3):
    """Alternating SA and local search."""
    target = n // 2
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    sa_deadline = time.time() + sa_time_fraction * (deadline - time.time())
    
    # SA phase
    h_sa = (best_h >= 0.5).astype(float)
    if h_sa.sum() != target:
        idx = np.argsort(h_sa)[::-1]
        h_sa = np.zeros(n)
        h_sa[idx[:target]] = 1.0
    
    corr_sa = np.correlate(h_sa, 1 - h_sa, mode='full').astype(float)
    cur_energy = corr_sa.max() * dx
    T = 0.005
    rng = np.random.RandomState(123)
    
    while time.time() < sa_deadline and T > 1e-7:
        ones = np.where(h_sa == 1)[0]
        zeros = np.where(h_sa == 0)[0]
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        
        delta = compute_swap_delta(i, j, h_sa, n)
        new_corr = corr_sa + delta
        new_energy = new_corr.max() * dx
        dE = new_energy - cur_energy
        
        if dE < 0 or rng.random() < np.exp(-dE / T):
            h_sa[i] = 1.0
            h_sa[j] = 0.0
            corr_sa = new_corr
            cur_energy = new_energy
            if cur_energy < best_c:
                best_c = cur_energy
                best_h = h_sa.copy()
        
        T *= 0.9995
        if T < 1e-7:
            break
    
    # Local search from SA result
    ls_deadline = deadline - 0.05 * (deadline - time.time())
    h_ls, c_ls = binary_local_search(h_sa, n, ls_deadline)
    if c_ls < best_c:
        best_c = c_ls
        best_h = h_ls.copy()
    
    return best_h, best_c

For `np.correlate(a, v, mode='full')`:
- Result length = len(a) + len(v) - 1 = 2n - 1
- out[k] = sum_{n} a[n + k - (len(v)-1)] * conj(v[n]) for n = max(0, len(v)-1-k) to min(len(a)-1, k)

out[k] = sum_n a[n] * v[n - k + (nv - 1)]

where nv = len(v).

For a = [1,2,3], v = [0,1,0], nv=3:

out[2] = sum_n a[n] * v[n-2+2] = a[0]*v[0] + a[1]*v[1] + a[2]*v[2] = 0 + 2 + 0 = 2
out[3] = sum_n a[n] * v[n-3+2] = a[1]*v[0] + a[2]*v[1] = 0 + 2 = 2
