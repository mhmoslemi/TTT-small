from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def swap_delta(h, p, q, n):
    delta = np.zeros(2 * n - 1)
    kr = np.arange(-(n - 1), n)
    ik = p + kr
    m = (ik >= 0) & (ik < n)
    delta[m] += (1.0 - h[ik[m]])
    ik2 = p - kr
    m2 = (ik2 >= 0) & (ik2 < n)
    delta[m2] -= h[ik2[m2]]
    ik = q + kr
    m = (ik >= 0) & (ik < n)
    delta[m] -= (1.0 - h[ik[m]])
    ik2 = q - kr
    m2 = (ik2 >= 0) & (ik2 < n)
    delta[m2] += h[ik2[m2]]
    return delta

def greedy_descent(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_val = float(corr.max())
    best_val = cur_val
    best_h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_new = cur_val
        best_p = best_q = -1
        best_d = None
        for p in zeros:
            for q in ones:
                d = swap_delta(h, p, q, n)
                nv = float((corr + d).max())
                if nv < best_new:
                    best_new = nv
                    best_p, best_q = p, q
                    best_d = d
        if best_p >= 0 and best_new < cur_val - 1e-14:
            h[best_p] = 1.0
            h[best_q] = 0.0
            corr = corr + best_d
            cur_val = best_new
            ones.remove(best_q)
            ones.append(best_p)
            zeros.remove(best_p)
            zeros.append(best_q)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            break
    return best_h, best_val * dx

def sa_optimize(h, n, deadline, seed, t_hi, t_lo):
    """Simulated annealing on binary swaps."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_val = float(corr.max())
    best_val = cur_val
    best_h = h.copy()
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    t_start = time.time()
    total = max(deadline - t_start, 1e-6)
    
    while time.time() < deadline:
        frac = min((time.time() - t_start) / total, 0.9999)
        T = t_hi * (t_lo / t_hi) ** frac
        
        pi = rng.randint(len(zeros_list))
        qi = rng.randint(len(ones_list))
        p = zeros_list[pi]
        q = ones_list[qi]
        
        d = swap_delta(h, p, q, n)
        new_val = float((corr + d).max())
        delta = new_val - cur_val
        
        if delta < 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
            h[p] = 1.0
            h[q] = 0.0
            corr = corr + d
            cur_val = new_val
            zeros_list[pi] = q
            ones_list[qi] = p
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
    
    return best_h, best_val * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
