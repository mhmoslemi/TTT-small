    return float(np.max(c) * dx), c * dx

def compute_corr(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_corr_k(h, n, k):
    """Compute gradient of c[k] w.r.t. h, where k is index in full correlation (0 to 2n-2)."""
    m = k - (n - 1)  # actual lag
    grad = np.zeros(n)
    if m == 0:
        # c = sum_i h[i]*(1-h[i])
        grad = 1.0 - 2.0 * h
    elif m > 0:
        # c = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        for j in range(m, n):
            grad[j] += (1.0 - h[j - m])
        for j in range(0, n - m):
            grad[j] -= h[j + m]
    else:
        p = -m
        # c = sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
        for j in range(0, n - p):
            grad[j] += (1.0 - h[j + p])
        for j in range(p, n):
            grad[j] -= h[j - p]
    return grad

def subgradient_optimize(h, n, deadline, top_k=3):
    """Projected subgradient with active set averaging."""
    h = h.copy()
    best_h = h.copy()
    corr = compute_corr(h, n)
    best_c = corr.max()
    
    step = 0.005
    n_no_improve = 0
    
    while time.time() < deadline:
        # Find worst lags
        top_lags = np.argsort(corr)[-top_k:]
        current_max = corr[top_lags[0]]
        
        if current_max < best_c - 1e-10:
            best_c = current_max
            best_h = h.copy()
            n_no_improve = 0
        else:
            n_no_improve += 1
            if n_no_improve > 100:
                step *= 0.97
                n_no_improve = 0
        
        # Average subgradient over active set
        sg = np.zeros(n)
        for k in top_lags:
            sg += grad_corr_k(h, n, int(k))
        sg /= top_k
        
        # Take step
        h_new = h - step * sg
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        new_corr = compute_corr(h_new, n)
        if new_corr.max() < corr.max():
            h = h_new
            corr = new_corr
        else:
            step *= 0.99
        
    return best_h, best_c

def fast_pair_search(h, n, deadline, deltas=None, max_passes=1000):
    """Pair search with O(n) per pair evaluation using incremental updates."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    dx = 2.0 / n
    best_c = float(corr.max() * dx)
    best_h = h.copy()
    
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n):
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        # Compute change in correlation
                        # Δc[k] = d * (grad of c[k] w.r.t. h[i]) - d * (grad of c[k] w.r.t. h[j])

To find min_t f(t):

  The minimum of such a function on [0,1] must occur at:
  - An endpoint (t=0 or t=1), or
  - A point where the concave function intersects one of the affine functions

  For the concave function g(t) = α*t - t² + γ and affine function h(t) = α'*t + γ':

  t = [(α+α') ± sqrt((α+α')² - 4(γ'-γ))] / 2

  There are at most O(n) such intersection points, so we can find the minimum in O(n) by checking all candidates.
