from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-15:
        # Distribute to non-boundary elements
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    best_cost = float(np.max(corr))
    
    for it in range(max_iters):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(n_swaps_per_iter):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            corr_new = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
            cost_new = float(np.max(corr_new))
            if cost_new < best_cost:
                best_cost = cost_new
                corr = corr_new
                improved = True
            else:
                h[i] = 1
                h[j] = 0
        if not improved:
            break
    return h, best_cost

def obj_slsqp(h, dx, beta, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def cons_eq(h, target_sum):
    return h.sum() - target_sum

def cons_eq_jac(h, target_sum):
    return np.ones(len(h))

def subgrad_direction(h, n):
    """Compute subgradient of max_k corr[k] at h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_worst = int(np.argmax(corr))
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        if len(idx) > 0:
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        if len(idx) > 0:
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad, k_worst

def slsqp_optimize(h0, n, beta, target_sum, bounds, maxiter=5000, ftol=1e-14):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_slsqp, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()
