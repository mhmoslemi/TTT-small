def compute_y_exact(h, n):
    dx = 2.0 / float(n)
    return dx * np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    y = compute_y_exact(h, n)
    return float(np.max(y))


def compute_y_opt(h, n):
    dx = 2.0 / float(n)
    if n <= 300:
        return dx * np.correlate(h, 1.0 - h, mode="full")
    return dx * safe_fftconvolve(h, (1.0 - h)[::-1], mode="full")


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = h.reshape(min(h.size, n))
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    diff = target - float(np.sum(h))
    if abs(diff) > 1.0e-13:
        lo = -1.0 - float(np.max(h)) - 1.0e-12
        hi = 1.0 - float(np.min(h)) + 1.0e-12

        f_lo = float(np.sum(np.clip(h + lo, 0.0, 1.0))) - target
        f_hi = float(np.sum(np.clip(h + hi, 0.0, 1.0))) - target

def grad_weighted_direct(x, n, w):

        else:
            cnt = n + lag
            if cnt > 0:
                g[:cnt] += wp * (1.0 - x[-lag:])
                g[-lag:] -= wp * x[:cnt]

                    if i == j:
                        continue
                    if h[i] >= 1.0 - 1.0e-12:
                        continue
                    maxd = min(1.0 - h[i], h[j])
                    if maxd <= 1.0e-5:
                        continue
                    Gi = G[i]

                    for d in PAIR_DELTAS:
                        if d >= maxd:
                            continue
                        if d < 1.0e-6:
                            break

                        np.subtract(Gi, Gj, out=tmp1)
                        np.multiply(tmp1, d, out=tmp2)
                        np.add(y, tmp2, out=base)

                        val = float(np.max(base))
                        p1 = nm1 + i - j
                        p2 = nm1 + j - i
                        add = dx * d * d

                        v1 = float(base[p1]) + add
                        if v1 > val:
                            val = v1
                        v2 = float(base[p2]) + add
                        if v2 > val:
                            val = v2

                        if val < best - 1.0e-13:
                            np.copyto(y, base)
                            y[p1] += add
                            y[p2] += add
                            h[i] += d
                            h[j] -= d
                            if h[i] > 1.0:
                                h[i] = 1.0
                            if h[j] < 0.0:
                                h[j] = 0.0
                            update_gmat_point(G, n, i, d)
                            update_gmat_point(G, n, j, -d)
                            best = val
                            accepts += 1
                            improved = True
                            break
                    if improved:
                        break
                    tested += 1
                    if tested >= test_limit:
                        break
                if improved:
                    break
                if tested >= test_limit:
                    break

        if not improved:
            rand_limit = 120 if n > 350 else 220
            for _ in range(rand_limit):
                if time.time() >= deadline:
                    break

                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue

                if gv[i] < gv[j]:
                    inc, dec = i, j
                else:
                    inc, dec = j, i

                if h[inc] >= 1.0 - 1.0e-12 or h[dec] <= 1.0e-12:
                    continue
                maxd = min(1.0 - h[inc], h[dec])
                if maxd <= 1.0e-5:
                    continue

                Gi = G[inc]
                Gj = G[dec]

                for d in (0.005, 0.001, 1.0e-4):
                    if d >= maxd:
                        continue

                    np.subtract(Gi, Gj, out=tmp1)
                    np.multiply(tmp1, d, out=tmp2)
                    np.add(y, tmp2, out=base)
                    val = float(np.max(base))

                    p1 = nm1 + inc - dec
                    p2 = nm1 + dec - inc
                    add = dx * d * d
                    v1 = float(base[p1]) + add
                    if v1 > val:
                        val = v1
                    v2 = float(base[p2]) + add
                    if v2 > val:
                        val = v2

                    if val < best - 1.0e-13:
                        np.copyto(y, base)
                        y[p1] += add
                        y[p2] += add
                        h[inc] += d
                        h[dec] -= d
                        if h[inc] > 1.0:
                            h[inc] = 1.0
                        if h[dec] < 0.0:
                            h[dec] = 0.0
                        update_gmat_point(G, n, inc, d)
                        update_gmat_point(G, n, dec, -d)
                        best = val
                        accepts += 1
                        improved = True
                        break
                if improved:
                    break

            if not improved:
                stall += 1
                if stall == 1:
                    dir_delta = 5.0e-5
                elif stall == 2:
                    dir_delta = 3.0e-3
                elif stall == 3:
                    dir_delta = 8.0e-4
                else:
                    break
            else:
                stall = 0
                dir_delta = 1.0e-4

    c_final = compute_c5(project_feasible(h, n), n)
    h = project_feasible(h, n)
    if c_final < best:
        best = c_final
    return h, best


def optimize_candidate(h0, n, deadline, rng, do_initial_local=True):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if do_initial_local and time.time() < deadline - 10:
        alloc = min(70.0, (deadline - time.time()) * 0.30)
        if alloc > 5.0:
            hh, cc = local_pair_search(h, n, time.time() + alloc, rng)
            if cc < best_c:
                best_c = cc
                best_h = hh copy.
                h = hh.copy()

    bounds = [(0.0, 1.0)] * n
    constraints = [{
        "type": "eq",
        "fun": sum_eq_fun,
        "jac": sum_eq_jac,
        "args": (n,)
    }]

    h_cur = best_h.copy()
    T_list = (0.03, 0.012, 0.005, 0.002, 0.0008, 0.0003)

    for T in T_list:
        rem = deadline - time.time()
        if rem < 25.0:
            break

        if n <= 180:
            base_iter = 320
        elif n <= 300:
            base_iter = 240
        elif n <= 420:
            base_iter = 170
        else:
            base_iter = 120

        if rem < 120.0:
            base_iter = int(0.55 * base_iter)
        if base_iter < 25:
            break

        try:
            res = minimize(
                soft_fun,
                h_cur,
                args=(n, T),
                method="SLSQP",
                jac=soft_jac,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": base_iter, "ftol": 1.0e-12}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if np.isfinite(c_new):
                    if c_new < best_c - 1.0e-14:
                        best_c = c_new
                        best_h = h_new.copy()
                    if c_new < best_c + 0.005:
                        h_cur = h_new
                    else:
                        h_cur = best_h.copy()
        except Exception:
            pass

        rem = deadline - time.time()
        if rem > 25.0:
            loc_time = min(30.0, 0.16 * rem)
            if loc_time > 5.0:
                hh, cc = local_pair_search(h_cur, n, time.time() + loc_time, rng)
                if cc < best_c - 1.0e-14:
                    best_c = cc
                    best_h = hh.copy()
                    h_cur = hh.copy()
                elif cc < best_c + 0.003:
                    h_cur = hh.copy()
                else:
                    h_cur = best_h.copy()

    rem = deadline - time.time()
    if rem > 20.0:
        hh, cc = local_pair_search(best_h, n, time.time() + min(60.0, 0.5 * rem), rng)
        if cc < best_c - 1.0e-14:
            best_h = hh.copy()
            best_c = cc

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    dx_old = 2.0 / float(n_old)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * dx_old
    dx_new = 2.0 / float(n_new)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)
    return project_feasible(h_new, n_new)


def best_shift(h, n, deadline):
    hh = h.copy()
    bc = compute_c5(hh, n)
    for s in range(1, n):
        if time.time() >= deadline:
            break
        hs = np.roll(hh, s)
        c = compute_c5(hs, n)
        if c < bc:
            bc = c
            hh = hs.copy()
    return hh, bc


def jitter_h(h, n, rng, moves, step):
    h = h.copy()
    for _ in range(moves):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = (float(rng.random_sample()) * 2.0 - 1.0) * step
        if 0.0 <= h[i] + d <= 1.0 and 0.0 <= h[j] - d <= 1.0:
            h[i] += d
            h[j] -= d
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is None or h_init.size < 10:
        n_init = 250
        x = (np.arange(n_init, dtype=float) + 0.5) * (2.0 / n_init)
        h_init = (
            0.5
            + 0.28 * np.cos(0.5 * np.pi * x)
            - 0.12 * np.cos(1.0 * np.pi * x)
            + 0.05 * np.sin(1.0 * np.pi * x)
        )
        h_init = project_feasible(h_init, n_init)
    else:
        n_init = int(len(h_init))
        h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    if n_init <= 350 and time.time() < deadline - 15:
        hs, cs = best_shift(best_h, n_init, deadline - 15)
        if cs < best_c:
            best_h, best_c = hs, cs

    if time.time() < deadline - 40:
        rem = deadline - time.time()
        main_alloc = min(520.0, 0.58 * rem)
        if main_alloc >= 45.0:
            ho, co = optimize_candidate(best_h, len(best_h), time.time() + main_alloc, rng, do_initial_local=True)
            if co < best_c:
                best_h, best_c = ho, co

    tried = set([len(best_h)])
    candidate_ns = []
    preferred = (120, 160, 200, 250, 320, 400, 500, 640)
    for m in preferred:
        if m not in tried and 60 <= m <= 800:
            candidate_ns.append(m)
    candidate_ns.sort(key=lambda m: abs(m - len(best_h)))
    candidate_ns = candidate_ns[:3]

    for m in candidate_ns:
        rem = deadline - time.time()
        if rem < 90.0:
            break
        alloc = min(180.0, 0.28 * rem)
        if alloc < 50.0:
            alloc = min(70.0, rem)
        if alloc < 35.0:
            break
        h_start = interpolate_h(best_h, len(best_h), m)
        ho, co = optimize_candidate(h_start, m, time.time() + alloc, rng, do_initial_local=True)
        if co < best_c:
            best_h, best_c = ho, co

    n_best = int(len(best_h))
    rem = deadline - time.time()
    if rem > 90.0:
        for step in (0.012, 0.030):
            rem = deadline - time.time()
            if rem < 50.0:
                break
            alloc = min(45.0, 0.40 * rem)
            if alloc < 20.0:
                break
            hj = jitter_h(best_h, n_best, rng, moves=80, step=step)
            ho, co = local_pair_search(hj, n_best, time.time() + alloc, rng)
            if co < best_c:
                best_h, best_c = ho, co

    rem = deadline - time.time()
    if rem > 12.0:
        ho, co = local_pair_search(best_h, len(best_h), min(deadline - 1.0, time.time() + min(40.0, 0.6 * rem)), rng)
        if co < best_c:
            best_h, best_c = ho, co

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
