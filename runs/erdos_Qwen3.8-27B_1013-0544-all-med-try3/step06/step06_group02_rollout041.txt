                break
            h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h


def softmax_weights(c, alpha, tol=0.0):
    m = len(c)
    if m <= 0:
        return np.array([], dtype=float)

    cmax = float(np.max(c))
    amax = int(np.argmax(c))
    if alpha <= 0.0:
        w = np.zeros(m, dtype=float)
        w[amax] = 1.0
        return w

    z = alpha * (c - cmax)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(m, dtype=float)
        w[amax] = 1.0
        return w

    w = e / s
    if tol > 0.0:
        mask = w > tol
        if np.any(mask):
            w = np.where(mask, w / float(np.sum(w[mask])), 0.0)
        else:
            w = np.zeros(m, dtype=float)
            w[amax] = 1.0
    return w


def grad_smooth_max(h, alpha, active_tol=0.0):
    n = len(h)
    if n <= 0:
        return np.array([], dtype=float)

    c = c_vec(h)
    w = softmax_weights(c, alpha, active_tol)
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)

    act = np.nonzero(w > 0.0)[0]
    n1 = n - 1
    for m in act:
        wm = float(w[m]) * dx
        if wm <= 0.0 or not np.isfinite(wm):
            continue
        k = int(m) - n1
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            g[0:nv] += wm * (1.0 - h[k:n])
            g[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            g[kk:n] += wm * (1.0 - h[0:nv])
            g[0:nv] += wm * (-h[kk:n])
    return g


def smooth_obj_val(h, alpha):
    c = c_vec(h)[:n]
    if len(c) == 0:
        return 0.0
    m = float(np.max(c))
    if alpha <= 0.0:
        return m
    z = alpha * (c - m)
    z = z - float(np.max(z))
    s = float(np.sum(np.exp(z)))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / alpha)


def penalty_obj(x, alpha, mu):
    n = len(x)
    val = smooth_obj_val(x, alpha)
    diff = float(x.sum() - 0.5 * n)
    val += 0.5 * mu * diff * diff
    return float(val)


def penalty_grad(x, alpha, mu):
    n = len(x)
    g = grad_smooth_max(x, alpha, 0.0)
    diff = float(x.sum() - 0.5 * n)
    g += mu * diff * np.ones(n, dtype=float)
    return g


def slsqp_val(x, alpha):
    return smooth_obj_val(x, alpha)


def slsqp_grad(x, alpha):
    return grad_smooth_max(x, alpha, 0.0)


def slsqp_constraint(x, n):
    return float(x.sum() - 0.5 * len(x))


def slsqp_jac(x, n):
    return np.ones(len(x), dtype=float)


def lbfgs_phase(h, n, deadline, full=True):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    if n <= 0 or time.time() >= deadline:
        return best_h, best_c

    if full:
        base = 110 if n <= 250 else (70 if n <= 500 else (35 if n <= 900 else 18))
        configs = [
            (20.0, 1.0),
            (100.0, 10.0),
            (500.0, 50.0),
            (2000.0, 200.0),
            (10000.0, 1000.0),
            (50000.0, 5000.0),
        ]
    else:
        base = max(20, int(base * 0.45))
        configs = [
            (20.0, 1.0),
            (300.0, 10.0),
            (3000.0, 100.0),
            (20000.0, 1000.0),
        ]

    bounds = [(0.0, 1.0)] * n

    for idx, (alpha, mu) in enumerate(configs):
        if time.time() > deadline - 3.0:
            break
        maxit = max(8, int(base * (1.0 - min(0.6, 0.08 * idx))))
        time_left = deadline - time.time()
        if time_left < 8.0:
            maxit = min(maxit, 12)
        try:
            res = minimize(
                penalty_obj,
                h,
                jac=penalty_grad,
                args=(alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-15,
                    "gtol": 1e-12,
                    "maxfun": max(maxit * 8, maxit + 20),
                    "maxls": 50,
                },
            )
            if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                elif c_new < best_c + 5.0e-4:
                    h = h_new.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def get_newton_data(n):
    data = _NEWTON_CACHE.get(n)
    if data is not None:
        return data

    idx = np.arange(n, dtype=int)
    off = idx[:, None] - idx[None, :] + (n - 1)
    offn = -off + 2 * (n - 1)
    starts = (n - 1) - idx
    ends = starts + n
    eye = np.eye(n, dtype=float)

    data = (off, offn, starts, ends, eye)
    _NEWTON_CACHE[n] = data
    return data


def newton_kkt_solution(w, n):
    if n < 2 or n > 1200:
        return None

    off, offn, starts, ends, eye = get_newton_data(n)
    dx = 2.0 / float(n)

    try:
        A = -dx * (w[off] + w[offn])
        A += 1.0e-10 * eye
        pref = np.concatenate(([0.0], np.cumsum(w)))
        b = dx * (pref[ends] - pref[starts])

        lhs = np.empty((n + 1, n + 1), dtype=float)
        lhs[:n, :n] = A
        lhs[:n, n] = 1.0
        lhs[n, :n] = -1.0
        lhs[n, n] = 0.0

        rhs = np.empty(n + 1, dtype=float)
        rhs[:n] = -b
        rhs[n] = -0.5 * n

        sol = np.linalg.solve(lhs, rhs)
        return sol[:n]
    except Exception:
        try:
            A = -dx * (w[off] + w[offn])
            A += 1.0e-6 * np.eye(n, dtype=float)
            pref = np.concatenate(([0.0], np.cumsum(w)))
            b = dx * (pref[ends] - pref[starts])

            lhs = np.empty((n + 1, n + 1), dtype=float)
            lhs[:n, :n] = A
            lhs[:n, n] = 1.0
            lhs[n, :n] = -1.0
            lhs[n, n] = 0.0

            rhs = np.empty(n + 1, dtype=float)
            rhs[:n] = -b
            rhs[n] = -0.5 * n

            sol = np.linalg.solve(lhs, rhs)
            return sol[:n]
        except Exception:
            return None


def active_newton_refine(h, n, deadline, full=True):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    if n <= 0:
        return best_h, best_c

    if full:
        alphas = (5.0, 20.0, 60.0, 200.0, 600.0, 2000.0, 6000.0, 20000.0, 60000.0)
        iters = 18
        stall_limit = 4
    else:
        alphas = (20.0, 100.0, 500.0, 2000.0)
        iters = 8
        stall_limit = 2

    stall = 0
    for aw in alphas:
        if time.time() > deadline - 2.0:
            break
        improved_alpha = False

        for _ in range(iters):
            if time.time() > deadline - 2.0:
                break

            h = project_fast(h, n)
            cur = compute_c5(h)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()

            c = c_vec(h)
            w = softmax_weights(c, float(aw), 0.0)
            if not np.any(w > 0.0):
                break

            sol = newton_kkt_solution(w, n)
            if sol is None:
                break
            if not np.all(np.isfinite(sol)):
                break
            sol = np.clip(sol, -2.0, 3.0)

            best_local = best_c
            best_shift = 0.0
            best_cand = None

            for s in _NEWTON_S:
                if time.time() > deadline - 1.0:
                    break
                cand = project_fast(h + float(s) * (sol - h), n)
                cv = compute_c5(cand)
                if cv < best_local - 1.0e-14:
                    best_local = cv
                    best_shift = float(s)
                    best_cand = cand.copy()

            if best_cand is not None and best_local < best_c - 1.0e-14:
                best_c = best_local
                best_h = best_cand
                h = best_cand.copy()
                improved_alpha = True
            else:
                break

        if improved_alpha:
            stall = 0
        else:
            stall += 1
            if stall >= stall_limit:
                break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def slsqp_phase(h, n, deadline, alpha=5000.0, maxiter=50):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    if n <= 0 or time.time() > deadline - 12.0:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": slsqp_constraint,
            "jac": slsqp_jac,
            "args": (n,),
        }
    ]

    try:
        res = minimize(
            slsqp_val,
            h,
            jac=slsqp_grad,
            args=(alpha,),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-14},
        )
        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    except Exception:
        pass

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def _pair_move_value(delta, h, i, j, direction):
    old_i = h[i]
    old_j = h[j]
    if direction > 0:
        h[i] = old_i + delta
        h[j] = old_j - delta
    else:
        h[i] = old_i - delta
        h[j] = old_j + delta

    h[i] = min(1.0, max(0.0, h[i]))
    h[j] = min(1.0, max(0.0, h[j]))

    val = compute_c5(h)

    h[i] = old_i
    h[j] = old_j
    return float(val)


def pair_single_step(h, n, i, j, direction, best_c):
    if n <= 1 or i == j:
        return h, best_c, False

    if direction > 0:
        hi = min(1.0 - h[i], h[j])
    else:
        hi = min(h[i], 1.0 - h[j])

    if hi <= 1e-12:
        return h, best_c, False

    best = best_c
    best_d = 0.0

    for frac in _PAIR_FRACS:
        d = float(frac) * hi
        if d <= 1.0e-16:
            continue
        val = _pair_move_value(d, h, i, j, direction)
        if val < best - 1.0e-14:
            best = val
            best_d = d

    if best_d > 0.0:
        span = max(0.05 * hi, 1.0e-4)
        lo = max(0.0, best_d - span)
        hi2 = min(hi, best_d + span)
        if hi2 > lo:
            for frac in _PAIR_REFINE_FRACS:
                d = lo + (hi2 - lo) * float(frac)
                if d <= 1.0e-16:
                    continue
                val = _pair_move_value(d, h, i, j, direction)
                if val < best - 1.0e-14:
                    best = val
                    best_d = d

    if best_d > 0.0:
        if direction > 0:
            h[i] = min(1.0, max(0.0, h[i] + best_d))
            h[j] = min(1.0, max(0.0, h[j] - best_d))
        else:
            h[i] = min(1.0, max(0.0, h[i] - best_d))
            h[j] = min(1.0, max(0.0, h[j] + best_d))
        return h, best, True

    return h, best_c, False


def get_pair_candidates(h, n, g, limit):
    if n < 2 or limit <= 0:
        return []

    diff = g[:, None] - g[None, :]
    room_p = np.minimum(1.0 - h[:, None], h[None, :])
    room_n = np.minimum(h[:, None], 1.0 - h[None, :])

    sp = np.where(diff < 0.0, (-diff) * room_p, 0.0)
    sn = np.where(diff > 0.0, diff * room_n, 0.0)
    score = np.maximum(sp, sn)
    np.fill_diagonal(score, 0.0)
    score = np.where(np.isfinite(score), score, 0.0)

    flat = score.ravel()
    total = int(flat.size)
    if total <= 1:
        return []

    k = min(int(limit), total)
    if k >= total:
        order = np.argsort(flat)[::-1]
    else:
        order = np.argpartition(flat, -k)[-k:]
        order = order[np.argsort(flat[order])[::-1]]

    out = []
    for idx in order:
        i, j = divmod(int(idx), n)
        if i == j:
            continue

        feasible_p = (g[i] < g[j] - 1.0e-15) and (room_p[i, j] > 1.0e-12)
        feasible_n = (g[i] > g[j] + 1.0e-15) and (room_n[i, j] > 1.0e-12)

        if feasible_p:
            out.append((i, j, 1))
        elif feasible_n:
            out.append((i, j, -1))

        if len(out) >= k:
            break

    return out


def pair_refine(h, n, best_c, deadline, rng, candidate_limit=250):
    h = project_fast(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()

    if n < 2 or candidate_limit <= 0:
        return project_feasible(h, n), compute_c5(h)

    if n > 1200:
        candidate_limit = min(candidate_limit, 30)
    elif n > 900:
        candidate_limit = min(candidate_limit, 50)
    elif n > 600:
        candidate_limit = min(candidate_limit, 100)
    elif n > 400:
        candidate_limit = min(candidate_limit, 180)

    no_improve_rounds = 0
    max_rounds = 220

    for _ in range(max_rounds):
        if time.time() >= deadline:
            break

        g = grad_smooth_max(h, 10000.0, 1e-10)
        if not np.all(np.isfinite(g)):
            break

        candidates = []
        if n <= 900:
            candidates = get_pair_candidates(h, n, g, candidate_limit)

        improved = False
        if candidates:
            fail = 0
            for i, j, direction in candidates:
                if time.time() >= deadline:
                    break
                h, c, ok = pair_single_step(h, n, i, j, direction, best_c)
                if ok:
                    best_c = c
                    improved = True
                    fail = 0
                else:
                    fail += 1
                if fail > 2 * len(candidates):
                    break

        if not improved:
            rem = max(0.0, deadline - time.time())
            rtries = int(min(50, max(10, 3.0 * rem)))
            random_improved = False
            for _r in range(rtries):
                if time.time() >= deadline:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if abs(g[i] - g[j]) <= 1.0e-12:
                    direction = 1 if rng.rand() < 0.5 else -1
                else:
                    direction = 1 if g[i] <= g[j] else -1
                h, c, ok = pair_single_step(h, n, i, j, direction, best_c)
                if ok:
                    best_c = c
                    random_improved = True
                    improved = True

            if not improved:
                no_improve_rounds += 1
                if no_improve_rounds >= 2:
                    break
            else:
                no_improve_rounds = 0
        else:
            no_improve_rounds = 0

    h = project_feasible(h, n)
    best_c = compute_c5(h)
    if best_c < 0.0:
        best_c = compute_c5(h)
    return h, best_c


def optimize_local(h, n, deadline, full=True):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    if n <= 0 or time.time() >= deadline:
        return best_h, best_c

    window = max(0.0, deadline - time.time())
    if full:
        lb_end = time.time() + 0.45 * window
        newt_end = time.time() + 0.85 * window
    else:
        lb_end = time.time() + 0.55 * window
        newt_end = time.time() + 0.9 * window

    if lb_end > time.time() + 1.0:
        h1, c1 = lbfgs_phase(h, n, lb_end, full)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
        h = h1 if c1 <= best_h_c_plus_1e6(best_h, c1) else h

    if time.time() < deadline - 4.0:
        h2, c2 = active_newton_refine(h, n, newt_end, full)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
        h = h2 if c2 <= best_h_c_plus_1e6(best_h, c2) else h

    if full and n <= 400 and time.time() < deadline - 15.0:
        h3, c3 = slsqp_phase(best_h, n, deadline, alpha=5000.0, maxiter=40)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    if time.time() < deadline - 5.0:
        h4, c4 = active_newton_refine(best_h, n, deadline, full=False)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def best_h_c_plus_1e6(h_ref, c_ref):
    return c_ref + 1.0e-6


def try_candidate(h, n, end_time, rng, full=False, with_pair=True):
    h = project_feasible(h, n)
    c = compute_c5(h)
    if n <= 0 or time.time() >= end_time:
        return h, c

    remaining = max(0.0, end_time - time.time())
    pair_alloc = min(35.0, max(0.0, 0.35 * remaining))
    opt_end = time.time() + max(1.0, remaining - pair_alloc)

    h, c = optimize_local(h, n, opt_end, full)

    if with_pair and time.time() < end_time - 4.0:
        lim = 80 if n <= 400 else (45 if n <= 700 else 25)
        h, c = pair_refine(h, n, c, end_time, rng, candidate_limit=lim)

    h = project_feasible(h, n)
    c = compute_c5(h)
    return h, c


def interpolate_step(old, n_old, n_new):
    if n_old <= 0 or n_new <= 0:
        return np.array([], dtype=float)
    if n_old == n_new:
        return project_feasible(old, n_old)

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    new = np.interp(x_new, x_old, old, left=old[0], right=old[-1])
    return project_feasible(new, n_new)


def make_thue_h(n):
    if n <= 0:
        return np.array([], dtype=float)
    h = np.empty(n, dtype=float)
    for i in range(n):
        x = i
        par = 0
        while x:
            par ^= x & 1
            x >>= 1
        h[i] = 0.0 if par == 0 else 1.0
    return project_feasible(h, n)


def perturb_h(h, rng, fraction=0.05, scale=0.12):
    h = h.copy()
    n = len(h)
    if n <= 0:
        return h
    mask = rng.rand(n) < fraction
    if np.any(mask):
        cnt = int(np.count_nonzero(mask))
        h[mask] += (rng.rand(cnt) - 0.5) * 2.0 * scale
    return project_feasible(h, n)


def reverse_h(h):
    return h[::-1].copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    margin = min(20.0, max(2.0, 0.02 * budget))
    deadline = t0 + max(1.0, budget - margin)
    rng = np.random.RandomState(int(seed % (2**32 - 1)))

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = make_thue_h(300)
        n0 = len(h0)

    if n0 <= 0:
        h0 = make_thue_h(300)
        n0 = len(h0)

    if budget <= 60.0:
        hb = project_feasible(h0, n0)
        cb = compute_c5(hb)
        return hb, cb, n0

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h)
    best_n = n0

    # Main optimization on the supplied construction.
    main_end = time.time() + min(300.0, max(90.0, 0.28 * (deadline - time.time())))
    h1, c1 = optimize_local(best_h, best_n, main_end, full=True)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, len(h1)

    # Main local pair line search.
    pair_end = time.time() + min(300.0, max(60.0, 0.3 * (deadline - time.time())))
    h2, c2 = pair_refine(best_h, best_n, best_c, pair_end, rng, candidate_limit=300)
    if c2 < best_c:
        best_h, best_c, best_n = h2, c2, len(h2)

    final_reserve = min(100.0, max(35.0, 0.12 * (deadline - time.time())))

    # Alternative: stochastic perturbation of the current best.
    left = deadline - time.time()
    if left > final_reserve + 80.0:
        alloc = min(110.0, left - final_reserve - 25.0)
        hp = perturb_h(best_h, rng, 0.05, 0.12)
        ha, ca = try_candidate(hp, best_n, time.time() + max(20.0, alloc), rng, full=False, with_pair=True)
        if ca < best_c:
            best_h, best_c, best_n = ha, ca, len(ha)

    # Alternative: resolution change from the current best.
    left = deadline - time.time()
    if left > final_reserve + 100.0:
        if best_n < 450:
            n_alt = 450
        elif best_n < 650:
            n_alt = 650
        elif best_n > 300:
            n_alt = 300
        else:
            n_alt = 450

        if n_alt != best_n:
            alloc = min(150.0, left - final_reserve - 25.0)
            hi = interpolate_step(best_h, best_n, n_alt)
            ha, ca = try_candidate(hi, n_alt, time.time() + max(25.0, alloc), rng, full=False, with_pair=True)
            if ca < best_c:
                best_h, best_c, best_n = ha, ca, len(ha)

    # Alternative: deterministic Thue-Morse restart.
    left = deadline - time.time()
    if left > final_reserve + 80.0:
        n_alt = 300 if best_n != 300 else 250
        alloc = min(90.0, left - final_reserve - 25.0)
        ht = make_thue_h(n_alt)
        ha, ca = try_candidate(ht, n_alt, time.time() + max(20.0, alloc), rng, full=False, with_pair=True)
        if ca < best_c:
            best_h, best_c, best_n = ha, ca, len(ha)

    # Final polishing on the global best.
    if time.time() < deadline - 25.0:
        hn = min(len(best_h), 1200)
        h3, c3 = active_newton_refine(best_h, best_n, min(deadline - 20.0, time.time() + 35.0), full=False)
        if c3 < best_c:
            best_h, best_c, best_n = h3, c3, len(h3)

    if time.time() < deadline - 6.0:
        lim = 140 if best_n <= 400 else (80 if best_n <= 700 else 35)
        h4, c4 = pair_refine(best_h, best_n, best_c, deadline, rng, candidate_limit=lim)
        if c4 < best_c:
            best_h, best_c, best_n = h4, c4, len(h4)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    # Safety: ensure we are better than the initial feasible parent if stored.
    try:
        cb_init = compute_c5(project_feasible(h0, n0))
        if cb_init < best_c:
            best_h = project_feasible(h0, n0)
            best_c = cb_init
            best_n = n0
    except Exception:
        pass

    return best_h, float(best_c), int(len(best_h))
