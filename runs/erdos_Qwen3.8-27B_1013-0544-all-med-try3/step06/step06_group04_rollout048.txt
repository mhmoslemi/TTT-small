    c = compute_all_lags(h, n)

            break
        h_new = slsqp_with_grad(best_h, n, alpha, top_m, maxiter=2000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: trust-constr
    if time.time() < deadline - 20:
        h_new = trust_constr_opt(best_h, n, 50000, 5, maxiter=1000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: More SLSQP polish
    if time.time() < deadline - 15:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_with_grad(best_h, n, alpha, 3, maxiter=3000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 5: Fast local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 1, time.time() + 30)
        h_new, c_new = fast_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 6: Multi-pair local search
    if time.time() < deadline - 3:
        mp_deadline = min(deadline - 1, time.time() + 20)
        h_new, c_new = multi_pair_local_search(best_h, n, mp_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Optimize at initial resolution
    if h_init is not None:
        alloc = min(300, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    
    # Try other resolutions
    n_candidates = [200, 250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.25)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 40) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        h_new = slsqp_with_grad(best_h, n, 1000000, 3, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final local search
        if time.time() < deadline - 5:
            h_new, c_new = fast_local_search(best_h, n, deadline - 2)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
