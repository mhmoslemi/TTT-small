The minimum of this convex function over h_j ∈ [0,1] (while maintaining the sum constraint) can be found exactly!

But we also have the constraint sum(h) = n/2. If we change h_j by delta, we need to change some other h_i by -delta. So it's not a single-coordinate problem.

Unless we use a "block" of 2 coordinates: change h_j by +delta and h_i by -delta. Then the objective becomes max_k (a_k*delta + b_k) where a_k depends on both i and j. This is still convex in delta, so we can find the optimal delta for each pair (i,j).

This is essentially what pair_search does, but instead of trying discrete deltas, we can solve for the optimal delta analytically!

For a pair (i, j) and a shift delta (h_i += delta, h_j -= delta):
R_k(delta) = R_k(0) + delta * (coeff_k)

where coeff_k = dR_k/dh_i - dR_k/dh_j (evaluated at the current point)

The new max is max_k (R_k(0) + delta * coeff_k)

We want to find delta ∈ [-h_j, 1-h_i] that minimizes max_k (R_k(0) + delta * coeff_k).

This is a 1D convex optimization! We can solve it by:
1. Finding the delta where the "active" set of k's changes
2. Or using bisection on the subgradient

The subgradient of max_k (R_k(0) + delta * coeff_k) with respect to delta is:
sum_{k in active_set} soft_max_k * coeff_k

At the optimum, 0 is in the subgradient, meaning:
min_{k active} coeff_k <= 0 <= max_{k active} coeff_k

This gives us a way to find the optimal delta. But it's still O(n) per pair evaluation (to compute the max), and O(n²) pairs, so O(n³) total. For n=250, that's about 15 million operations per sweep, which is fine.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        exp_s = np.exp(shifted - max_s)
        lse = (max_s + np.log(np.sum(exp_s))) / alpha
        weights = exp_s / np.sum(exp_s)
    else:
        exp_s = np.exp(shifted)
        lse = np.log(np.sum(exp_s)) / alpha
        weights = exp_s / np.sum(exp_s)
    
    obj = m + lse
    
    # Compute gradient
    # d(c[k])/d(h[j]) for the full correlation c = correlate(h, 1-h, 'full')
    # For k in [0, n-1]: c[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    #   d/dh[j] = (1-h[j-k]) * [k<=j<n] - h[j+k] * [j<=n-1-k]
    # For k in [n, 2n-2]: let m=k-n+1, c[k] = sum_{i=0}^{n-m-1} h[i+m]*(1-h[i])
    #   d/dh[j] = (1-h[j-m]) * [m<=j<n] - h[j+m] * [j<=n-m-1]
    
    grad = np.zeros(n)
    
    # Handle k = 0 to n-1
    for k in range(n):
        wk = weights[k] * dx
        if wk < 1e-18:
            continue
        # First term: (1-h[j-k]) for j in [k, n-1]
        if k == 0:
            grad += wk * (1.0 - h)
        else:
            grad[k:] += wk * (1.0 - h[:n-k])
        # Second term: -h[j+k] for j in [0, n-1-k]
        if k == 0:
            grad -= wk * h
        else:
            grad[:n-k] -= wk * h[k:]
    
    # Handle k = n to 2n-2 (m = 1 to n-1)
    for m in range(1, n):
        k = n + m - 1
        wk = weights[k] * dx
        if wk < 1e-18:
            continue
        grad[m:] += wk * (1.0 - h[:n-m])
        grad[:n-m] -= wk * h[m:]
    
    return obj, grad

def l_bfgsb_optimize(h, n, alpha, maxiter=500, deadline=None):
    def fun(x):
        obj, _ = smooth_objective_and_grad(x, n, alpha)
        return obj
    
    def jac(x):
        _, g = smooth_objective_and_grad(x, n, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    # Use equality constraint via penalty (L-BFGS-B doesn't support constraints)
    # Instead, we'll project after each iteration
    
    # Actually, let's use the penalized approach:
    # Add penalty for sum(h) != n/2
    
    penalty = 1000.0
    
    def fun_p(x):
        obj, g = smooth_objective_and_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return obj + penalty * s * s
    
    def jac_p(x):
        obj, g = smooth_objective_and_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return g + 2 * penalty * s * np.ones(n)
    
    res = minimize(fun_p, h, jac=jac_p, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    x = project_feasible(res.x, n)
    return x

def coordinate_descent(h, n, deadline, max_sweeps=100):
    """Coordinate descent with mean-preserving updates."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Precompute R
        R = np.correlate(h, 1.0 - h, mode='full') * dx
        
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            # If we change h[j] by delta, we adjust all other h by -delta/(n-1)
            # New h: h_new[j] = h[j] + delta
            #         h_new[i] = h[i] - delta/(n-1) for i ≠ j
            
            # R_k as function of delta:
            # R_k(delta) = R_k + delta * d_k where d_k = dR_k/dh[j] - sum_{i≠j} dR_k/dh[i] / (n-1)
            # But sum_i dR_k/dh[i] = dR_k/d(sum(h)) ... 
            # Actually, R_k = sum_i h[i]*(1-h[i+k])*dx
            # dR_k/dh[j] (keeping others fixed) = (1-h[j+k])*dx*[j+k<n] - h[j-k]*dx*[j>=k]
            # For the mean-preserving update, the effective derivative is:
            # d_eff_k = dR_k/dh[j] - (1/(n-1)) * sum_{i≠j} dR_k/dh[i]
            #         = dR_k/dh[j] - (1/(n-1)) * (sum_i dR_k/dh[i] - dR_k/dh[j])
            #         = (n/(n-1)) * dR_k/dh[j] - (1/(n-1)) * sum_i dR_k/dh[i]
            
            # sum_i dR_k/dh[i] = d/d(sum(h)) of R_k
            # R_k = sum_i h[i]*(1-h[i+k])*dx
            # = sum_i h[i]*dx - sum_i h[i]*h[i+k]*dx
            # d/d(sum(h)): the first term gives (number of terms)*dx
            # The second term: d/d(sum(h)) of sum_i h[i]*h[i+k] is not simply related to sum(h)
            # because h[i]*h[i+k] is not linear in sum(h).
            
            # Hmm, this approach doesn't simplify nicely. Let me use a different strategy.
            # Instead of mean-preserving, let me use "pair" updates: change h[j] and h[i] for a specific i.
            # But that's O(n²) which is what we had before.
            
            # Alternative: just do the single-coordinate update and then reproject.
            # The reprojection will fix the sum constraint.
            # This is less efficient but simpler.
            
            # Let me compute the optimal delta for h[j] alone (ignoring sum constraint temporarily)
            # dR_k/dh[j] for each k:
            dk = np.zeros(2*n-1)
            for k in range(n):
                if j >= k and j < n:  # first term
                    dk[k] += (1.0 - h[j-k]) * dx
                if j <= n-1-k:  # second term
                    dk[k] -= h[j+k] * dx
            for m in range(1, n):
                k = n + m - 1
                if j >= m:
                    dk[k] += (1.0 - h[j-m]) * dx
                if j <= n-m-1:
                    dk[k] -= h[j+m] * dx
            
            # R_k + delta * dk[k], minimize max over k, subject to 0 <= h[j]+delta <= 1
            # This is a 1D convex minimization
            # Use the fact that at optimum, the max is minimized
            
            new_hj = h[j]
            # Try a few candidate deltas
            # The optimal delta is where the "active" constraints balance
            # Simple approach: try delta values and pick the best
            
            # Actually, for efficiency, let me just use a scalar minimizer
            lo = -h[j]
            hi = 1.0 - h[j]
            
            # Evaluate at boundaries and a few interior points
            candidates = [lo, hi]
            # Add the point where the top 2 R values cross
            idx_order = np.argsort(R)[::-1]
            if len(idx_order) >= 2 and dk[idx_order[0]] != dk[idx_order[1]]:
                cross = (R[idx_order[1]] - R[idx_order[0]]) / (dk[idx_order[0]] - dk[idx_order[1]])
                if lo < cross < hi:
                    candidates.append(cross)
            candidates.extend([0.0])
            
            best_delta = 0.0
            best_val = best_c
            for delta in candidates:
                if delta < lo or delta > hi:
                    continue
                hj_new = h[j] + delta
                if hj_new < 0 or hj_new > 1:
                    continue
                # Compute new max R
                new_R = R + delta * dk
                new_max = new_R.max()
                # We need to reproject, but for comparison, let's use the raw value
                if new_max < best_val:
                    best_val = new_max
                    best_delta = delta
            
            if abs(best_delta) > 1e-12:
                h[j] += best_delta
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                else:
                    # Revert (approximately)
                    h[j] -= best_delta
                    h = project_feasible(h, n)
        
        if not improved:
            break
    
    return h, best_c
```

```python
def compute_dk(h, j, n):
    """Compute dR_k/dh[j] for all k, vectorized."""
    dx = 2.0 / n
    dk = np.zeros(2*n-1)
    
    # For k in [0, n-1]:
    # dk[k] = (1-h[j-k]) * [k <= j] - h[j+k] * [k <= n-1-j]
    k_range = np.arange(n)
    mask1 = k_range <= j
    mask2 = k_range <= (n-1-j)
    # (1-h[j-k]) for k where k <= j: h[j], h[j-1], ..., h[j-min(j,n-1)]
    if j >= 0:
        vals1 = 1.0 - h[max(0, j-(j)) : j+1][::-1]  # h[j], h[j-1], ..., h[0]
        # Actually: for k=0: h[j], k=1: h[j-1], ..., k=j: h[0]
        # So vals1 = 1 - h[j::-1] (if j < n)
        dk[:min(j+1, n)] += (1.0 - h[j::-1][:min(j+1, n)])
    if n-1-j >= 0:
        vals2 = h[j:j+min(n-j, n)]  # h[j], h[j+1], ..., h[min(2j, n-1)]
        dk[:min(n-j, n)] -= vals2[:min(n-j, n)]
    
    # Hmm, this is getting confusing. Let me just do it more carefully.
    
    # For k = 0, 1, ..., n-1:
    # Term 1: (1-h[j-k]) is valid when 0 <= j-k < n, i.e., k <= j and j-k >= 0
    #   So for k in [0, min(j, n-1)]: contribution is (1-h[j-k])
    # Term 2: -h[j+k] is valid when 0 <= j+k < n, i.e., k <= n-1-j
    #   So for k in [0, min(n-1-j, n-1)]: contribution is -h[j+k]
    
    max_k1 = min(j, n-1)
    if max_k1 >= 0:
        ks = np.arange(max_k1 + 1)
        dk[ks] += (1.0 - h[j - ks])
    
    max_k2 = min(n-1-j, n-1)
    if max_k2 >= 0:
        ks = np.arange(max_k2 + 1)
        dk[ks] -= h[j + ks]
    
    # For k = n, n+1, ..., 2n-2 (m = k-n+1, m from 1 to n-1):
    # Same formula with m:
    # (1-h[j-m]) for m in [1, min(j, n-1)]
    # -h[j+m] for m in [1, min(n-1-j, n-1)]
    
    max_m1 = min(j, n-1)
    if max_m1 >= 1:
        ms = np.arange(1, max_m1 + 1)
        dk[n + ms - 1] += (1.0 - h[j - ms])
    
    max_m2 = min(n-1-j, n-1)
    if max_m2 >= 1:
        ms = np.arange(1, max_m2 + 1)
        dk[n + ms - 1] -= h[j + ms]
    
    return dk * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        exp_s = np.exp(shifted - max_s)
        sse = np.sum(exp_s)
        lse = (max_s + np.log(sse)) / alpha
        weights = exp_s / sse
    else:
        exp_s = np.exp(shifted)
        sse = np.sum(exp_s)
        lse = np.log(sse) / alpha
        weights = exp_s / sse
    obj = m + lse
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k] * dx
        if wk < 1e-18:
            continue
        mk1 = min(k, n - 1 - 0)
        if k == 0:
            grad += wk * (1.0 - h)
            grad -= wk * h
        else:
            # (1-h[j-k]) for j in [k, n-1]
            grad[k:] += wk * (1.0 - h[:n-k])
            # -h[j+k] for j in [0, n-1-k]
            grad[:n-k] -= wk * h[k:]
    for moff in range(1, n):
        k = n + moff - 1
        wk = weights[k] * dx
        if wk < 1e-18:
            continue
        grad[moff:] += wk * (1.0 - h[:n-moff])
        grad[:n-moff] -= wk * h[moff:]
    return obj, grad

def lbfgs_optimize(h, n, alpha, maxiter=300, penalty=5000.0):
    def fun(x):
        obj, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return obj + penalty * s * s

    def jac(x):
        obj, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return g + 2.0 * penalty * s * np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 2})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def compute_dk_all(h, n):
    """Compute dR_k/dh[j] for all j, k. Returns (n, 2n-1) matrix."""
    dx = 2.0 / n
    J = np.zeros((n, 2 * n - 1))
    for j in range(n):
        # k in [0, n-1]
        k1 = min(j, n - 1)
        if k1 >= 0:
            ks = np.arange(k1 + 1)
            J[j, ks] += (1.0 - h[j - ks])
        k2 = min(n - 1 - j, n - 1)
        if k2 >= 0:
            ks = np.arange(k2 + 1)
            J[j, ks] -= h[j + ks]
        # k in [n, 2n-2], moff = k-n+1
        m1 = min(j, n - 1)
        if m1 >= 1:
            ms = np.arange(1, m1 + 1)
            J[j, n + ms - 1] += (1.0 - h[j - ms])
        m2 = min(n - 1 - j, n - 1)
        if m2 >= 1:
            ms = np.arange(1, m2 + 1)
            J[j, n + ms - 1] -= h[j + ms]
    J *= dx
    return J

def coord_descent_anneal(h, n, deadline, T0=0.01, T_rate=0.95, max_sweeps=200):
    """Coordinate descent with simulated annealing and incremental R update."""
    dx = 2.0 / n
    h = h.copy()
    R = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c = float(R.max())
    J = compute_dk_all(h, n)
    T = T0
    order = list(range(n))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        T *= T_rate
        improved = False
        rng_local = np.random.RandomState(sweep * 1000 + 42)
        rng_local.shuffle(order)
        for j in order:
            if time.time() > deadline:
                break
            dk = J[j]
            # Find optimal delta: minimize max_k(R_k + delta*dk_k) s.t. 0<=h[j]+delta<=1
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-15:
                continue
            # Evaluate at candidate points
            # The function g(delta) = max_k(R_k + delta*dk_k) is convex piecewise linear
            # Find minimum by checking where subgradient changes sign
            # Use ternary search on [lo, hi] with ~30 iterations
            a = lo
            b = hi
            for _ in range(40):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                g1 = (R + m1 * dk).max()
                g2 = (R + m2 * dk).max()
                if g1 < g2:
                    b = m2
                else:
                    a = m1
            delta = (a + b) / 2.0
            # Also check if a small random perturbation helps (annealing)
            if T > 1e-6 and rng_local.random() < 0.1:
                delta_rand = rng_local.uniform(lo, hi)
                g_rand = (R + delta_rand * dk).max()
                g_det = (R + delta * dk).max()
                if g_rand < g_det + T:
                    delta = delta_rand
            g_new = (R + delta * dk).max()
            g_old = R.max()
            if g_new < g_old - 1e-12 or (T > 1e-6 and rng_local.random() < np.exp(-(g_new - g_old) / max(T, 1e-10))):
                h[j] += delta
                # Update R incrementally
                R += delta * dk
                # Recompute J for row j (since h changed)
                J[j] = 0.0
                k1 = min(j, n - 1)
                if k1 >= 0:
                    ks = np.arange(k1 + 1)
                    J[j, ks] += (1.0 - h[j - ks])
                k2 = min(n - 1 - j, n - 1)
                if k2 >= 0:
                    ks = np.arange(k2 + 1)
                    J[j, ks] -= h[j + ks]
                m1 = min(j, n - 1)
                if m1 >= 1:
                    ms = np.arange(1, m1 + 1)
                    J[j, n + ms - 1] += (1.0 - h[j - ms])
                m2 = min(n - 1 - j, n - 1)
                if m2 >= 1:
                    ms = np.arange(1, m2 + 1)
                    J[j, n + ms - 1] -= h[j + ms]
                J[j] *= dx
                # Update J for all other rows (since h[j] changed, affects other dks)
                # J[i, k] depends on h[j] if j = i+k or j = i-k or j = i+moff etc.
                # This is complex, so we just recompute J periodically
                improved = True
                c_current = float(R.max())
                if c_current < best_c:
                    best_c = c_current
                # Periodically recompute J to avoid drift
                if sweep % 20 == 0:
                    J = compute_dk_all(h, n)
                    R = np.correlate(h, 1.0 - h, mode='full') * dx
        # Project to maintain feasibility
        h = project_feasible(h, n)
        R = np.correlate(h, 1.0 - h, mode='full') * dx
        c_actual = float(R.max())
        if c_actual < best_c:
            best_c = c_actual
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def make_binary_random(n, rng):
    """Create a random binary pattern with n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_sawtooth(n, period):
    """Create a sawtooth pattern."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    phase = (x * period / 2.0) % 1.0
    h[phase < 0.5] = 1.0
    return project_feasible(h, n)

def make_sine(n, freq):
    """Create a sine-based pattern."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
    return project_feasible(h, n)

def optimize_single(h, n, deadline, alphas=None):
    """Run L-BFGS-B with increasing alphas then coordinate descent."""
    if alphas is None:
        alphas = [200, 500, 1000, 3000, 8000, 20000]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = lbfgs_optimize(h, n, a, maxiter=200, penalty=10000.0)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    # Coordinate descent refinement
    if time.time() < deadline - 5:
        h, best_c = coord_descent_anneal(h, n, deadline, T0=0.005, T_rate=0.9, max_sweeps=100)
    return h, best_c
```
