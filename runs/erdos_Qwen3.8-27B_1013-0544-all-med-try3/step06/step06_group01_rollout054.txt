def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones"""
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def lbfgs_smooth(h, n, alpha, deadline, maxiter=300):
    """L-BFGS-B optimization with smooth max and penalty"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        # log-sum-exp
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -100, 100)
        sm = m + np.log(np.exp(shifted).sum()) / alpha
        pen = 1000.0 * (x.sum() - target) ** 2
        return sm + pen
    
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                              'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

```python
def compute_delta(h, i, j, n):
    """Compute the change in correlation when h[i] goes 0->1 and h[j] goes 1->0.
    Returns the delta array of length 2n-1 (same as np.correlate output)."""
    delta = np.zeros(2*n - 1)
    g = 1.0 - h  # current g values (before swap)
    
    # After swap: h'[i]=1, h'[j]=0, g'[i]=0, g'[j]=1
    
    # For each lag m (corresponding to index n-1+m in the output):
    # delta[m] = g'[i+m] - g'[j+m] - h'[i-m] + h'[j-m]
    # (with appropriate bounds and special cases)
    
    # g'[i+m]: 
    #   if i+m in [0,n): 
    #     if i+m == j: g'[j] = 1
    #     else: g[i+m] = 1-h[i+m]
    #   else: 0
    # So the lags m where i+m in [0,n) are: m in [-i, n-1-i]
    
    for m in range(-i, n - i):
        idx = n - 1 + m
        ip = i + m
        if ip == j:
            val_a = 1.0
        else:
            val_a = 1.0 - h[ip]
        delta[idx] += val_a
    
    # -g'[j+m]:
    #   if j+m in [0,n):
    #     if j+m == i: g'[i] = 0
    #     else: g[j+m] = 1-h[j+m]
    #   else: 0
    # Lags m where j+m in [0,n): m in [-j, n-1-j]
    for m in range(-j, n - j):
        idx = n - 1 + m
        jp = j + m
        if jp == i:
            val_b = 0.0
        else:
            val_b = 1.0 - h[jp]
        delta[idx] -= val_b
    
    # -h'[i-m]:
    #   if i-m in [0,n):
    #     if i-m == i (m=0): h'[i] = 1
    #     if i-m == j: h'[j] = 0
    #     else: h[i-m]
    #   else: 0
    # Lags m where i-m in [0,n): m in [i-(n-1), i]
    for m in range(max(-(n-1), i-(n-1)), min(n-1, i)+1):
        idx = n - 1 + m
        im = i - m
        if im == i:  # m = 0
            val_c = 1.0
        elif im == j:
            val_c = 0.0
        else:
            val_c = h[im]
        delta[idx] -= val_c
    
    # +h'[j-m]:
    #   if j-m in [0,n):
    #     if j-m == i: h'[i] = 1
    #     if j-m == j (m=0): h'[j] = 0
    #     else: h[j-m]
    #   else: 0
    # Lags m where j-m in [0,n): m in [j-(n-1), j]
    for m in range(max(-(n-1), j-(n-1)), min(n-1, j)+1):
        idx = n - 1 + m
        jm = j - m
        if jm == i:
            val_d = 1.0
        elif jm == j:  # m = 0
            val_d = 0.0
        else:
            val_d = h[jm]
        delta[idx] += val_d
    
    return delta

For p=j-m: need j-m in [0,n), so m in [j-(n-1), j]. Also p+m=j which is always in [0,n) if j is.

So the affected lags are the union of:
[-i, n-1-i] ∪ [-j, n-1-j] ∪ [i-(n-1), i] ∪ [j-(n-1), j]

Each of these is an interval of length n, so the union covers at most 4n lags (but they overlap, so it's at most about 2n-1 lags, which is all of them).

For each affected lag m, the contribution is:
sum over p in {i, j, i-m, j-m} ∩ [0,n) with p+m in [0,n):

def compute_delta_fast(h, i, j, n):
    delta = np.zeros(2*n-1)
    g = 1.0 - h
    
    # Determine range of affected m values
    m_start = max(-i, -j, i-(n-1), j-(n-1))
    m_end = min(n-1-i, n-1-j, i, j) + 1  # exclusive
    
    # Actually the union is more complex. Let me just iterate over all m in [-(n-1), n-1]
    # and check if any p is affected.
    
    for m in range(-(n-1), n):
        idx = n - 1 + m
        contrib = 0.0
        
        # Check p=i
        p = i
        q = p + m  # = i+m
        if 0 <= q < n:
            hp = 1.0  # h'[i]
            gq = 1.0 if q == j else (0.0 if q == i else g[q])  # g'[q]
            contrib += hp * gq - h[p] * g[q]
        
        # Check p=j
        p = j
        q = p + m  # = j+m
        if 0 <= q < n:
            hp = 0.0  # h'[j]
            gq = 1.0 if q == j else (0.0 if q == i else g[q])
            # Wait, gq = g'[q] = 1 - h'[q]
            # h'[q] = 0 if q==j, 1 if q==i, h[q] otherwise
            # So g'[q] = 1 if q==j, 0 if q==i, g[q] otherwise
            contrib += hp * gq - h[p] * g[q]
            # = 0 - 1*g[q] = -g[q] (since h[j]=1)
            # But if q==i: g'[q]=g'[i]=0, so contribution is 0 - 1*g[i] = -g[i]
            #   But g[i] = 1-h[i] = 1-0 = 1, so contribution = -1
            #   And 0 - 1*1 = -1 ✓
            # If q==j: g'[q]=g'[j]=1, contribution is 0 - 1*g[j] = -g[j] = -0 = 0
            #   But wait, if q==j then m=j-j=0, and h[j]*g[j] = 1*0 = 0
            #   So the original term was 0, and the new term is 0*1 = 0. Change = 0. ✓
        
        # Check p=i-m (only if i-m != i and i-m != j to avoid double counting)
        p2 = i - m
        if p2 != i and p2 != j and 0 <= p2 < n:
            q = p2 + m  # = i
            # q = i, which is in [0,n)
            hp = h[p2]  # h'[p2] = h[p2] since p2 not in {i,j}
            gq = 0.0  # g'[i] = 0
            contrib += hp * gq - h[p2] * g[i]
            # = 0 - h[p2]*g[i] = -h[p2]*(1-h[i]) = -h[p2]*1 = -h[p2]
            
        # Check p=j-m (only if j-m != i and j-m != j)
        p3 = j - m
        if p3 != i and p3 != j and 0 <= p3 < n:
            q = p3 + m  # = j
            hp = h[p3]
            gq = 1.0  # g'[j] = 1
            contrib += hp * gq - h[p3] * g[j]
            # = h[p3]*1 - h[p3]*0 = h[p3]
        
        delta[idx] = contrib
    
    return delta
```

Actually, for the simulated annealing, I can use a vectorized approach:

For a given swap (i, j), the delta at each lag m is a sum of up to 4 terms, each of which is a simple function of m. I can compute all 4 "components" as arrays and then add them (with the appropriate validity masks and overlap corrections).

Let me define:
- A[m] = contribution from p=i (valid when i+m in [0,n))
- B[m] = contribution from p=j (valid when j+m in [0,n))
- C[m] = contribution from p=i-m (valid when i-m in [0,n), i-m≠i, i-m≠j)
- D[m] = contribution from p=j-m (valid when j-m in [0,n), j-m≠i, j-m≠j)

Then delta = A + B + C + D (with the understanding that when there are overlaps, we've already excluded them from C and D).

Let me compute each:

A[m] for m in [-i, n-1-i]:
  q = i+m
  g'[q] = 1 if q==j, 0 if q==i, g[q] otherwise
  A[m] = 1 * g'[q] - 0 * g[q] = g'[q]
  = 1 if q==j (i.e., m==j-i)
  = 0 if q==i (i.e., m==0) [but h'[i]=1, g'[i]=0, so 1*0 - 0*1 = 0] ✓
  = g[q] otherwise = 1-h[q]

So A is a vector: for m in [-i, n-1-i], A[m] = 1-h[i+m], except at m=j-i where A=1, and at m=0 where A=0.

Wait, at m=0: q=i, g'[q]=g'[i]=0, so A[0]=0. And 1-h[i]=1-0=1. So the exception is: at m=0, A=0 instead of 1.
At m=j-i: q=j, g'[q]=g'[j]=1, so A[j-i]=1. And 1-h[j]=1-1=0. So the exception is: at m=j-i, A=1 instead of 0.

So: A[m] = 1-h[i+m] for m in [-i, n-1-i], with corrections:
  A[0] = 0 (instead of 1-h[i] = 1)
  A[j-i] = 1 (instead of 1-h[j] = 0)
  (These corrections only apply if the indices are in the valid range AND not equal to each other)

This is getting complex but implementable with numpy.

Similarly:
B[m] for m in [-j, n-1-j]:
  q = j+m
  B[m] = 0 * g'[q] - 1 * g[q] = -g[q]
  = -(1-h[j+m])
  Exceptions:
  At m=j-j=0: q=j, contribution = 0*g'[j] - 1*g[j] = 0 - 0 = 0. And -g[j] = -(1-h[j]) = 0. So no exception needed!
  At m=i-j: q=i, contribution = 0*g'[i] - 1*g[i] = 0 - 1 = -1. And -g[i] = -(1-h[i]) = -1. So no exception either!

C[m] for m in [i-(n-1), i], with i-m in [0,n) and i-m≠i (m≠0) and i-m≠j (m≠i-j):
  p = i-m
  C[m] = h[p] * 0 - h[p] * g[i] = -h[p] * (1-h[i]) = -h[i-m] * 1 = -h[i-m]
  (since h[i]=0, so g[i]=1)
  Valid when: 0 ≤ i-m < n, m≠0, m≠i-j
  i.e., m in [i-(n-1), i], m≠0, m≠i-j

D[m] for m in [j-(n-1), j], with j-m in [0,n) and j-m≠i (m≠j-i) and j-m≠j (m≠0):
  p = j-m
  D[m] = h[p] * 1 - h[p] * g[j] = h[p] - h[p]*0 = h[p] = h[j-m]
  (since h[j]=1, so g[j]=0)
  Valid when: 0 ≤ j-m < n, m≠0, m≠j-i
  i.e., m in [j-(n-1), j], m≠0, m≠j-i

So the full delta is:
delta[m] = A[m] + B[m] + C[m] + D[m]

where:
A[m] = (1-h[i+m]) for m in [-i, n-1-i], with A[0]=0 and A[j-i]=1 (if valid)
B[m] = -(1-h[j+m]) for m in [-j, n-1-j]
C[m] = -h[i-m] for m in [i-(n-1), i], m≠0, m≠i-j
D[m] = h[j-m] for m in [j-(n-1), j], m≠0, m≠j-i

And delta[m] = 0 for m outside the union of the valid ranges.

This can be computed vectorized! Let me implement it:

```python
def compute_delta_vec(h, i, j, n):
    delta = np.zeros(2*n-1)
    g = 1.0 - h
    L = n - 1  # lag m = idx - L
    
    # A: m in [-i, n-1-i]
    m_start = -i
    m_end = n - 1 - i
    idx_start = L + m_start
    idx_end = L + m_end + 1
    # A[m] = 1 - h[i+m] = g[i+m]
    # i+m ranges from i+m_start = i-i = 0 to i+m_end = i+n-1-i = n-1
    delta[idx_start:idx_end] += g[0:n]  # g[i+m] for m from -i to n-1-i is g[0] to g[n-1]
    # Correction at m=0: A[0] should be 0, but we added g[i]
    # m=0 corresponds to idx = L
    # g[i] was added at position idx_start + (0-m_start) = L + i
    # Wait, idx_start = L + (-i) = L-i. Position for m=0 is L. Offset = L - (L-i) = i.
    # So we added g[i] at delta[L]. We need to set it to 0, so subtract g[i].
    delta[L] -= g[i]  # which is 1-h[i] = 1
    # Correction at m=j-i: A[j-i] should be 1, but we added g[j]
    # m=j-i corresponds to idx = L + j - i. Offset from idx_start = j-i+i = j.
    # We added g[j] at delta[L+j-i]. g[j] = 0. We need it to be 1, so add 1.
    mi = j - i
    if -i <= mi <= n-1-i:
        delta[L + mi] += 1.0  # g[j]=0, we need 1, so add 1
    
    # B: m in [-j, n-1-j]
    m_start = -j
    m_end = n - 1 - j
    idx_start = L + m_start
    idx_end = L + m_end + 1
    # B[m] = -(1-h[j+m]) = -g[j+m]
    # j+m ranges from 0 to n-1
    delta[idx_start:idx_end] -= g[0:n]
    # No corrections needed (as shown above)
    
    # C: m in [i-(n-1), i], m≠0, m≠i-j
    m_start = i - (n - 1)
    m_end = i
    idx_start = L + m_start
    idx_end = L + m_end + 1
    # C[m] = -h[i-m]
    # i-m ranges from i-m_end=0 to i-m_start=n-1
    # So -h[i-m] for m from m_start to m_end is -h[n-1], -h[n-2], ..., -h[0] = -h[::-1]
    delta[idx_start:idx_end] += -h[::-1]
    # Correction at m=0: should be excluded
    # m=0 at idx=L. Offset = L - idx_start = L - (L+i-n+1) = n-1-i
    # We added -h[i-0] = -h[i] = 0 at that position. So no correction needed (it's already 0)!
    # Wait, -h[i] = -0 = 0. So the correction is automatically 0. Good.
    # Correction at m=i-j: should be excluded
    mi = i - j
    if m_start <= mi <= m_end and mi != 0:
        pos = L + mi
        # We added -h[i-mi] = -h[i-(i-j)] = -h[j] = -1
        # We need to remove it: add 1
        delta[pos] += 1.0
    
    # D: m in [j-(n-1), j], m≠0, m≠j-i
    m_start = j - (n - 1)
    m_end = j
    idx_start = L + m_start
    idx_end = L + m_end + 1
    # D[m] = h[j-m]
    # j-m ranges from j-m_end=0 to j-m_start=n-1
    # So h[j-m] for m from m_start to m_end is h[n-1], h[n-2], ..., h[0] = h[::-1]
    delta[idx_start:idx_end] += h[::-1]
    # Correction at m=0: should be excluded
    # m=0 at idx=L. We added h[j-0] = h[j] = 1. Need to remove: subtract 1.
    delta[L] -= 1.0
    # Correction at m=j-i: should be excluded
    mi = j - i
    if m_start <= mi <= m_end and mi != 0:
        pos = L + mi
        # We added h[j-mi] = h[j-(j-i)] = h[i] = 0. So correction is 0. No action needed.
    
    return delta
```
