def smooth_max_hgrad(h, corr, alpha, dx):
    n = len(h)
    # Compute softmax weights
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    # weights[k] corresponds to shift k-(n-1) where k is index in corr array (0 to 2n-2)
    
    # Gradient: dL/dh[j] = dx * sum_k weights[k] * [(1-h[j-k]) - h[j+k]]
    # where k = index - (n-1), and we need valid indices
    n_1 = n - 1
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        k = k_idx - n_1  # actual shift
        w = weights[k_idx]
        if w < 1e-15:
            continue
        # Term 1: (1-h[j-k]) for j such that 0<=j-k<n, i.e., k<=j<k+n
        # Term 2: -h[j+k] for j such that 0<=j+k<n, i.e., -k<=j<n-k
        # For term 1: j ranges over max(0,k) to min(n-1, k+n-1) = min(n-1, n-1) if k>=0, or 0 to n-1+k if k<0
        # Actually: 0 <= j-k < n means k <= j < k+n and 0<=j<n

        j_start = max(0, k)
        j_end = min(n-1, k + n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end+1)
            h_idx = idx - k
            grad[idx] += w * (1 - h[h_idx])
        
        # For term 2: 0 <= j+k < n means -k <= j < n-k and 0<=j<n
        # So j in [max(0,-k), min(n-1, n-1-k)]

        j_end = min(n-1, n-1-k)
        if j_start <= j_end:

    return grad * dx

    # Actually with N=2n padding, there's no wrap-around
    # conv result indices 0 to 2n-2 are valid
    return result[:2*n-1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_lse_grad(h, corr, alpha, dx):
    n = len(h)
    n_corr = len(corr)
    n_1 = n - 1
    
    max_corr = corr.max()
    exp_vals = np.exp(alpha * (corr - max_corr))
    p = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for s_idx in range(n_corr):
        s = s_idx - n_1
        w = p[s_idx]
        if w < 1e-16:
            continue
        t_start = max(0, s)
        t_end = min(n - 1, s + n - 1)
        if t_start <= t_end:
            t_idx = np.arange(t_start, t_end + 1)
            h_idx = t_idx - s
            grad[t_idx] += w * (1.0 - h[h_idx])
        t_start = max(0, -s)
        t_end = min(n - 1, n - 1 - s)
        if t_start <= t_end:
            t_idx = np.arange(t_start, t_end + 1)
            h_idx = t_idx + s
            grad[t_idx] -= w * h[h_idx]
    
    return grad * dx

def pgd_adam(h, n, deadline, seed=42, alpha_start=100, alpha_end=1000000,
             eta=0.1, max_stagnation=2000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = compute_corr(h, n)
    cur_c = corr.max()
    best_h = h.copy()
    best_c = cur_c
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    stagnation = 0
    
    while time.time() < deadline:
        t_count += 1
        frac = min(1.0, t_count / 20000)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        grad = compute_lse_grad(h, corr, alpha, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_feasible(h - eta * update, n)
        h = h_new
        corr = compute_corr(h, n)
        cur_c = corr.max()
        
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > max_stagnation:
            pert = 0.02 * rng.randn(n)
            h = project_feasible(best_h + pert, n)
            corr = compute_corr(h, n)
            cur_c = corr.max()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            stagnation = 0
            max_stagnation = max(500, max_stagnation // 2)
    
    return best_h, best_c

def upsample(h, n, new_n):
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n)

def lbfgs_refine(h, n, deadline, alpha=50000, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        c = np.correlate(x_clipped, 1 - x_clipped, mode='full') * dx
        max_c = c.max()
        shifted = alpha * (c - max_c)
        lse = max_c + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha
        penalty = 50.0 * (x_clipped.sum() - target) ** 2
        return lse + penalty
    
    def gradient(x):
        x_clipped = np.clip(x, 0, 1)
        c = np.correlate(x_clipped, 1 - x_clipped, mode='full') * dx
        max_c = c.max()
        exp_vals = np.exp(alpha * (c - max_c))
        p = exp_vals / exp_vals.sum()
        g = compute_lse_grad(x_clipped, c, alpha, dx)
        g += 100.0 * (x_clipped.sum() - target)
        # Zero out gradient where x is at boundary
        mask = (x > 0) & (x < 1)
        g = g * mask
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, method='L-BFGS-B', jac=gradient,
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

dL/dh = conv(p, 1-h) - conv(p_rev, h)

where p is the softmax weights (length 2n-1, centered) and p_rev is the reversed p.

Using FFT:
- Pad p to length ≥ 2n (for linear convolution)
- Pad h and 1-h to the same length
- Compute ifft(fft(p) * fft(1-h)) for the first term
- Compute ifft(fft(p_rev) * fft(h)) for the second term

def compute_lse_grad_fft(h, corr, alpha, dx):

    # Let u = t - k + n_1, so k = t + n_1 - u

    # This is the cross-correlation of q and (1-h) at lag (t + n_1)

    # So correlate(q, 1-h)[m] = sum_i q[i+m] * (1-h)[i]
    # We want conv1[t] = sum_i q[i + t + n_1 - i + i]...

    # where p[s] = p[s + n_1] in array form (let's call it P[s+n_1])

    # B[u] = 1 - h[u] for u in [0, n-1]
