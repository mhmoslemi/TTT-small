                    h_new = projected_gradient_step(h, n, step, deadline)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_new.copy()
                        h = h_new
                    else:
                        break
    return best_h, best_c

def active_set_slsqp(h, n, top_m, alpha, maxiter, deadline):
    lags = compute_all_lags(h, n)
    top_idx = np.argsort(lags)[-top_m:]
    active_c = lags[top_idx]
    
    def active_obj(x):
        c = compute_all_lags(x, n)
        a = c[top_idx]
        m = a.max()
        shifted = alpha * (a - m)
        sm = shifted.max()
        if sm > 500:
            return float(m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha)
        return float(m + np.log(np.exp(shifted).sum()) / alpha)
    
    def active_grad(x):
        c = compute_all_lags(x, n)
        a = c[top_idx]
        m = a.max()
        shifted = alpha * (a - m)
        sm = shifted.max()
        if sm > 500:
            shifted = shifted - sm
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            wk = w[i] * dx
            if wk < 1e-14:
                continue
            if k >= n:
                mm = k - n + 1
                if mm < n:
                    grad[:n-mm] -= wk * x[mm:]
                    grad[mm:] += wk * (1.0 - x[:n-mm])
            elif k == n - 1:
                grad[:] += wk * (1.0 - 2.0 * x)
            else:
                mm = n - 1 - k
                if mm < n:
                    grad[:n-mm] += wk * (1.0 - x[mm:])
                    grad[mm:] -= wk * x[:n-mm]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    actual_maxiter = maxiter
    if deadline is not None:
        remaining = deadline - time.time()
        if remaining < 5:
            return h.copy()
        actual_maxiter = max(50, min(maxiter, int(remaining / 0.3)))
    try:
        res = minimize(active_obj, h, jac=active_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': actual_maxiter, 'ftol': 1e-14})
        h_res = project_feasible(res.x, n)
        if compute_c5(h_res, n) < compute_c5(h, n):
            return h_res
    except Exception:
        pass
    return h.copy()

def multi_alpha_slsqp(h, n, deadline, alpha_schedule=None):
    if alpha_schedule is None:
        alpha_schedule = [500, 2000, 10000, 50000, 200000, 500000, 1000000]
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alpha_schedule:
        if time.time() > deadline - 10:
            break
        alloc = min(80, (deadline - time.time()) * 0.3)
        h_new = slsqp_run(h, n, alpha, maxiter=2000, ftol=1e-15, deadline=time.time() + alloc)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def local_search_pairs(h, n, deadline, max_rounds=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        lags = compute_all_lags(h, n)
        k_star = int(np.argmax(lags))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        n_tries = min(100, n - m)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = rng.randint(0, n - m)
            j = i + m
            if lag > 0:
                if h[i] > h[j] + 1e-4:
                    delta = min(0.005, (h[i] - h[j]) * 0.4)
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-4:
                    delta = min(0.005, (h[j] - h[i]) * 0.4)
                    h[j] -= delta
                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] += delta
                        h[i] -= delta
            else:
                if h[j] > h[i] + 1e-4:
                    delta = min(0.005, (h[j] - h[i]) * 0.4)
                    h[j] -= delta
                    h[i] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] += delta
                        h[i] -= delta
                elif h[i] > h[j] + 1e-4:
                    delta = min(0.005, (h[i] - h[j]) * 0.4)
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
        if not improved and r > 100:
            break
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Multi-alpha SLSQP (main optimization)
    if time.time() < deadline - 30:
        h1, c1 = multi_alpha_slsqp(h, n, deadline, 
                                   alpha_schedule=[500, 2000, 10000, 50000, 200000, 500000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Active set SLSQP with different top_m values
    if time.time() < deadline - 20:
        for top_m in [3, 5, 8, 12]:
            if time.time() > deadline - 15:
                break
            alloc = min(40, (deadline - time.time()) * 0.2)
            h_new = active_set_slsqp(best_h, n, top_m, 100000, 1500, time.time() + alloc)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Very high alpha SLSQP polish
    if time.time() < deadline - 10:
        alloc = min(50, (deadline - time.time()) * 0.3)
        h_new = slsqp_run(best_h, n, 2000000, maxiter=3000, ftol=1e-16, deadline=time.time() + alloc)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Local search
    if time.time() < deadline - 5:
        h_new, c_new = local_search_pairs(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Primary optimization at parent resolution
    if h_init is not None:
        alloc_primary = min(600, (deadline - t0) * 0.55)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc_primary)
        best_h, best_c = h_opt, c_opt
    
    # Try other resolutions
    other_ns = []
    if best_h is not None:
        if n_init != 200:
            other_ns.append(200)
        if n_init != 400:
            other_ns.append(400)
        if n_init != 250:
            other_ns.append(250)
    else:
        other_ns = [200, 300, 400]
    
    for n_try in other_ns:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, min(5, int(remaining / 50)))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Final high-alpha SLSQP
        h_new = slsqp_run(best_h, n, 5000000, maxiter=3000, ftol=1e-16, deadline=deadline - 5)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Final local search
        h_new, c_new = local_search_pairs(best_h, n, deadline - 1, max_rounds=3000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
