def bb_gradient_descent(h, n, alpha, deadline, max_iter=10000):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    h_curr = h.copy()
    prev_grad = None
    prev_h = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute gradient of smooth max
        c = compute_all_lags(h_curr, n)
        m = c.max()
        weights = np.exp(alpha * (c - m)) / np.exp(alpha * (c - m)).sum()
        grad = np.zeros(n)
        for w, k_idx in zip(weights, range(len(c))):
            if w > 1e-10:
                lag = k_idx - (n - 1)
                grad += w * _compute_lag_gradient(h_curr, n, lag)

        # and box constraints
        grad -= grad.mean()  # project onto sum=0 hyperplane
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            s = h_curr - prev_h
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-15:
                alpha_bb_plus = ss / sy if sy > 1e-15 else 1.0
                alpha_bb_minus = sy / yy
                # Alternate or average
                step = 0.5 * (alpha_bb_plus + alpha_bb_minus)
                step = max(1e-8, min(step, 0.1))
            else:
                step = 0.01
        else:
            step = 0.01
            # Take a trial step to initialize BB
            trial = h_curr - step * grad
            trial = project_feasible(trial, n)
            if compute_c5(trial, n) < best_c:
                best_c = compute_c5(trial, n)
                best_h = trial.copy()

        prev_grad = grad.copy()
        
        h_curr = h_curr - step * grad
        h_curr = project_feasible(h_curr, n)
        
        c_curr = compute_c5(h_curr, n)

        return float(m + (shifted.max() + np.log(np.sum(np.exp(s2)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def optimize_slq(h, n, active_idx, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, active_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def subgrad_local(h, n, deadline, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.05
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        c_max = c.max()
        # Active lags: within 0.5% of max
        active = np.where(c >= c_max * 0.995)[0]
        
        grad = np.zeros(n)
        for idx in active:
            lag = int(idx) - (n - 1)
            grad += lag_gradient(h, n, lag)
        grad /= max(1, len(active))
        grad -= grad.mean()
        
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-12:
            break
        
        # Try a few step sizes
        for s in [step, step * 0.3, step * 0.01]:
            h_try = h - s * grad / gnorm
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h_try.copy()
                step = max(0.001, step * 0.99)
                break
        else:
            step *= 0.8
            if step < 0.0001:
                break
    
    return best_h, best_c

def pair_local_search(h, n, deadline, max_iter=100000):
    """Try moving mass between pairs to reduce the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        
        # For this lag, find the best (i, j) pair to swap
        # We want to reduce c[k_star]
        # Moving δ from i to j (i in "high" part, j in "low" part)
        
        if lag > 0:
            # c = sum h[i]*(1-h[i+m])
            # To reduce: decrease h[i] where h[i] is high and (1-h[i+m]) is high
            # i.e., where h[i] is high and h[i+m] is low
            # Or increase h[i+m] where h[i] is high and h[i+m] is low
            
            # Gradient: dc/dh[i] = (1-h[i+m]) for i<n-m, dc/dh[i+m] = -h[i] for i<n-m
            # So we want to move mass from high-gradient to low-gradient
            # High gradient at position i means decreasing h[i] reduces c
            # Low gradient (or negative) at position j means increasing h[j] reduces c
            
            # Simple: move from the position with highest (1-h[i+m])*h[i] to adjacent
            # Better: just use the gradient
            g = np.zeros(n)
            g[:n-m] += (1.0 - h[m:]) * h[:n-m]  # not quite right
            # Actually gradient of c w.r.t h:
            # dc/dh[i] = (1-h[i+m]) for i < n-m (from h[i]*(1-h[i+m]))
            #           + (-h[i-m]) for i >= m (from (1-h[i-m])*h[i])
            # So total: dc/dh[i] = (1-h[i+m]) (if i < n-m) + (-h[i-m]) (if i >= m)
            
            g = np.zeros(n)
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
            g -= g.mean()
            
            # Move from high g to low g
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if g[i] - g[j] < 1e-10:
                break
            
            max_delta = min(h[i], 1.0 - h[j])
            for delta in [0.01 * max_delta, 0.005 * max_delta, 0.001 * max_delta]:
                if delta < 1e-12:
                    continue
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
        else:
            # Similar for lag < 0
            m = -lag
            g = np.zeros(n)
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
            g -= g.mean()
            
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if g[i] - g[j] < 1e-10:
                break
            
            max_delta = min(h[i], 1.0 - h[j])
            for delta in [0.01 * max_delta, 0.005 * max_delta, 0.001 * max_delta]:
                if delta < 1e-12:
                    continue
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = project_feasible(h, n)
    
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Stage 1: SLSQP with all lags (medium alpha)
    if time.time() < deadline - 20:
        h_new = optimize_slq(h, n, np.arange(2*n-1), alpha=5000, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Stage 2: Active set SLSQP (top 5 lags, high alpha)
    for top_m in [5, 10, 3]:
        if time.time() < deadline - 15:
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_m:]
            for alpha in [50000, 100000, 500000]:
                if time.time() < deadline - 10:
                    h_new = optimize_slq(best_h, n, top_idx, alpha, maxiter=3000)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_new
    
    # Stage 3: Subgradient local search
    if time.time() < deadline - 10:
        h_new, c_new = subgrad_local(best_h, n, min(deadline - 5, time.time() + 60))
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Stage 4: Pair local search
    if time.time() < deadline - 5:
        h_new, c_new = pair_local_search(best_h, n, deadline - 2)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Final check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

    n_init = len(h_init)

    # Try initial resolution
    alloc = min(200, (deadline - t0) * 0.25)

    best_h, best_c = h_opt, c_opt
    
    # Try higher resolutions
    for n_new in [400, 500, 600, 800]:
        if time.time() > deadline - 100:

            break

        h_opt, c_opt = optimize_at_resolution(h_start, n_new, time.time() + alloc)
