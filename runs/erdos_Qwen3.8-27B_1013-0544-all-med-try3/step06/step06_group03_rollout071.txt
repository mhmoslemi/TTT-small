    m1 = np.where(valid1)[0]
    m2 = np.where(valid2)[0]
    U[k1[valid1], m1] += delta
    U[k2[valid2], m2] -= delta


def fast_pair_search(h, n, deadline, step_sizes):
    dx = 2.0 / n
    lags = compute_lags(n)
    c = full_correlation(h)
    current_max = float(np.max(c))
    U = compute_U(h, n, lags)

    for d in step_sizes:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return project_feasible(h, n), compute_c5(h, n)
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        diff_vec = d * (U[j] - U[i])
                        new_max = float(np.max(c + diff_vec))
                        if new_max < current_max - 1e-13:
                            h[i] += d
                            h[j] -= d
                            c += diff_vec
                            current_max = new_max
                            update_U_single(U, n, lags, i, d)
                            update_U_single(U, n, lags, j, -d)
                            improved = True
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        diff_vec = -d * (U[j] - U[i])
                        new_max = float(np.max(c + diff_vec))
                        if new_max < current_max - 1e-13:
                            h[i] -= d
                            h[j] += d
                            c += diff_vec
                            current_max = new_max
                            update_U_single(U, n, lags, i, -d)
                            update_U_single(U, n, lags, j, d)
                            improved = True
            if not improved:
                break

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = full_correlation(x)
    a = alpha * c
    a -= a.max()
    val = np.sum(np.exp(a))
    if val > 1e300:
        return float('inf')
    return float(dx * np.log(val) / alpha)


def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = full_correlation(x)
    a = alpha * c
    a -= a.max()
    ea = np.exp(a)
    s = ea.sum()
    if s > 1e300:
        return np.ones(n)
    p = ea / s
    lags = compute_lags(n)
    i_range = np.arange(n)[:, None]
    idx_p = i_range + lags[None, :]
    idx_m = i_range - lags[None, :]
    valid_p = (idx_p >= 0) & (idx_p < n)
    valid_m = (idx_m >= 0) & (idx_m < n)
    clipped_p = np.clip(idx_p, 0, n - 1)
    clipped_m = np.clip(idx_m, 0, n - 1)
    mat_p = np.where(valid_p, 1.0 - x[clipped_p], 0.0)
    mat_m = np.where(valid_m, x[clipped_m], 0.0)
    return dx * (mat_p.dot(p) - mat_m.dot(p))


def slsqp_smooth(h, n, alpha, maxiter=1500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x, nn, aa):
        return smooth_obj(x, nn, aa)

    def obj_grad(x, nn, aa):
        return smooth_obj_grad(x, nn, aa)

    def sum_con_fun(x, nn):
        return x.sum() - nn / 2.0

    def sum_con_jac(x, nn):
        return np.ones(nn)

    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_func, h.copy(), args=(n, alpha),
                       jac=lambda x: obj_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or compute_c5(project_feasible(res.x, n), n) < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def subgradient_hill_climb(h, n, deadline, max_iter=20000):
    dx = 2.0 / n
    lags = compute_lags(n)
    c = full_correlation(h)
    current_max = float(np.max(c) * dx)
    lr = 0.005
    patience = 0
    best_h = h.copy()
    best_c = current_max

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_max_val = np.max(c)
        active = np.where(c >= c_max_val - 0.02)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = np.zeros(n)
        for k in active:
            lag = lags[k]
            idx_p = np.arange(n) + lag
            idx_m = np.arange(n) - lag
            valid_p = (idx_p >= 0) & (idx_p < n)
            valid_m = (idx_m >= 0) & (idx_m < n)
            grad_contrib = np.zeros(n)
            grad_contrib[valid_p] += (1.0 - h[idx_p[valid_p]])
            grad_contrib[valid_m] -= h[idx_m[valid_m]]
            g += grad_contrib
        g /= len(active)
        g *= dx
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = full_correlation(h_new)
        new_max = float(np.max(c_new) * dx)
        if new_max < current_max - 1e-14:
            h = h_new
            c = c_new
            current_max = new_max
            if current_max < best_c:
                best_c = current_max
                best_h = h.copy()
            patience = 0
            lr = min(lr * 1.02, 0.05)
        else:
            patience += 1
            lr *= 0.92
            if lr < 1e-7:
                break
            if patience > 500:
                break
    return best_h, best_c


def optimize_at_resolution(h_start, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_smooth(h, n, a, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    # Fast pair search
    if time.time() < deadline - 10:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        h, best_c = fast_pair_search(h, n, time.time() + min(60, (deadline - time.time()) * 0.5), steps)

    # Subgradient
    if time.time() < deadline - 10:
        alloc = min(40, (deadline - time.time()) * 0.3)
        ht, ct = subgradient_hill_climb(h, n, time.time() + alloc)
        if ct < best_c:
            best_c = ct
            h = ht

    # Final pair search
    if time.time() < deadline - 3:
        h, best_c = fast_pair_search(h, n, deadline - 2,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    # Final SLSQP
    if time.time() < deadline - 3:
        ht = slsqp_smooth(h, n, 100000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except (NameError, Exception):
        n_init = 100
        h_init = project_feasible(np.full(n_init, 0.5), n_init)

    best_h = project_feasible(h_init.copy(), len(h_init))
    best_c = compute_c5(best_h, len(best_h))

    # Phase 1: Optimize initial at its resolution
    alloc1 = min(150, (deadline - t0) * 0.2)
    if alloc1 > 30:
        h1, c1 = optimize_at_resolution(best_h, len(best_h), t0 + alloc1, seed=seed)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Try different resolutions
    n_base = len(best_h)
    resolutions = []
    for nr in [n_base, 200, 300, 400, 500]:
        if nr != n_base and nr not in resolutions:
            resolutions.append(nr)
    resolutions.sort()

    for nr in resolutions:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time() - 60
        alloc = min(120, remaining * 0.3)
        if alloc < 30:
            break
        # Interpolate
        x_old = np.linspace(0, 2, n_base)
        x_new = np.linspace(0, 2, nr)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, nr)
        h_new, c_new = optimize_at_resolution(h_interp, nr, time.time() + alloc, seed=seed + nr)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            n_base = nr

    # Phase 3: Random restarts
    n_best = len(best_h)
    n_restarts = max(2, int((deadline - time.time() - 50) / 80))
    for r in range(n_restarts):
        if time.time() > deadline - 50:
            break
        scale = 0.05 + 0.1 * (r / max(1, n_restarts))
        hp = best_h + rng.normal(0, scale, n_best)
        hp = project_feasible(hp, n_best)
        alloc = min(60, (deadline - time.time() - 30) * 0.4)
        if alloc < 20:
            break
        h_r, c_r = optimize_at_resolution(hp, n_best, time.time() + alloc, seed=seed + 100 + r)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase 4: Extended pair search on best
    n_final = len(best_h)
    if time.time() < deadline - 10:
        steps_final = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        best_h, best_c = fast_pair_search(best_h, n_final, deadline - 8, steps_final)

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 5:
        ht = slsqp_smooth(best_h, n_final, 200000, maxiter=2000)
        c = compute_c5(ht, n_final)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Tiny pair search polish
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h, len(best_h), deadline - 1,
                                          [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
