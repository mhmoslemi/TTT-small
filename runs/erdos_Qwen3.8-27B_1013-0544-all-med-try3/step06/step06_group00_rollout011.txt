            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        if time.time() > deadline - 2:
            break
    return best_h, best_c

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(len(zeros_list)):
            if time.time() > deadline:
                break
            zi = zeros_list[i]
            best_delta = cur_c
            best_j = -1
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = zi + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = zi - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - zi)
                delta[k1] += 1.0
                k2 = n - 1 + (zi - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_delta - 1e-14:
                    best_delta = nc
                    best_j = j
            if best_j >= 0:
                delta = np.zeros(2 * n - 1)
                im = zi + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = zi - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - zi)
                delta[k1] += 1.0
                k2 = n - 1 + (zi - best_j)
                delta[k2] += 1.0
                h[zi] = 1.0
                h[best_j] = 0.0
                corr += delta
                cur_c = best_delta
                ones_list.remove(best_j)
                zeros_list.remove(zi)
                ones_list.append(zi)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary_fast(h_init, n, deadline, seed=42, max_iters=2000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.99998
    scale = 2000.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-12:
            ratio = scale * (cur_c - nc) / T
            if ratio > 50:
                accept = True
            elif rng.random() < np.exp(min(ratio, 50)):
                accept = True
        if accept:
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def slsqp_refine(h_init, n, alpha, maxiter=500, ftol=1e-18, deadline=None):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj(h_init):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init

def structured_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 0:
        block = max(2, n // 40)
        pos = 0
        for i in range(n_ones):
            if pos >= n:
                pos = rng.randint(n)
            h[pos] = 1.0
            pos += block
            if pos >= n:
                pos = 0
            if h.sum() >= n_ones:
                break
    while h.sum() < n_ones:
        idx = rng.randint(n)
        if h[idx] == 0:
            h[idx] = 1.0
    while h.sum() > n_ones:
        idx = rng.randint(n)
        if h[idx] == 1:
            h[idx] = 0.0
    return h

def downsample(h_old, n_old, n_new):
    if n_new >= n_old:
        x_old = np.linspace(0, 2, n_old)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, h_old)
    else:
        ratio = n_old / n_new
        h_new = np.zeros(n_new)
        for i in range(n_new):
            s = int(i * ratio)
            e = int((i + 1) * ratio)
            if e <= s:
                e = s + 1
            h_new[i] = h_old[s:min(e, n_old)].mean()
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Projected gradient descent (fast, many iterations)
    p1_end = t0 + 0.12 * total
    if time.time() < p1_end:
        h1, c1 = proj_grad_descent(best_h, n0, p1_end,
                                    alpha_start=10.0, alpha_end=20000.0,
                                    n_alphas=15, lr0=0.8, momentum=0.85)
        if c1 < best_c:
            best_c, best_h, best_n = c1, h1, n0

    # Phase 2: SLSQP refinement with smooth max
    p2_end = t0 + 0.30 * total
    if time.time() < p2_end:
        for alpha in [500, 2000, 10000, 50000]:
            if time.time() > p2_end - 5:
                break
            h2 = slsqp_refine(best_h, n0, alpha, maxiter=800, ftol=1e-16, deadline=p2_end)
            c2 = compute_c5(h2, n0)
            if c2 < best_c:
                best_c, best_h, best_n = c2, h2, n0

    # Phase 3: Binary local search on current best
    p3_end = t0 + 0.50 * total
    if time.time() < p3_end:
        hb, cb = binary_local_search(best_h, n0, p3_end, max_sweeps=300)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb, n0

    # Phase 4: Short anneal from binary
    p4_end = t0 + 0.62 * total
    if time.time() < p4_end:
        ha, ca = anneal_binary_fast(best_h, n0, p4_end, seed=seed, max_iters=1500000)
        if ca < best_c:
            best_c, best_h, best_n = ca, ha, n0

    # Phase 5: Re-smooth after binary work
    p5_end = t0 + 0.72 * total
    if time.time() < p5_end:
        h5 = slsqp_refine(best_h, n0, 20000, maxiter=1000, ftol=1e-17, deadline=p5_end)
        c5 = compute_c5(h5, n0)
        if c5 < best_c:
            best_c, best_h, best_n = c5, h5, n0
        h5b = proj_grad_descent(best_h, n0, p5_end,
                                 alpha_start=1000, alpha_end=200000,
                                 n_alphas=10, lr0=0.3, momentum=0.8)
        c5b = compute_c5(h5b, n0)
        if c5b < best_c:
            best_c, best_h, best_n = c5b, h5b, n0

    # Phase 6: Try different resolutions
    p6_end = t0 + 0.88 * total
    for nh in [200, 300, 600]:
        if time.time() > p6_end - 20:
            break
        hh = downsample(best_h, best_n, nh)
        # Quick binary search at this resolution
        hb2, cb2 = binary_local_search(hh, nh, min(time.time() + 30, p6_end - 15), max_sweeps=100)
        if cb2 < compute_c5(hh, nh):
            hh = hb2
        # SLSQP refine
        if time.time() < p6_end - 10:
            hh = slsqp_refine(hh, nh, 5000, maxiter=500, ftol=1e-15, deadline=p6_end - 8)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh

    # Phase 7: Final polish
    p7_end = dl - 3
    if time.time() < p7_end:
        # Try a few noise restarts
        for trial in range(5):
            if time.time() > p7_end - 5:
                break
            h7 = best_h + 0.003 * rng.randn(best_n)
            h7 = project_feasible(h7, best_n)
            h7 = proj_grad_descent(h7, best_n, min(time.time() + 10, p7_end - 3),
                                    alpha_start=5000, alpha_end=200000,
                                    n_alphas=8, lr0=0.2, momentum=0.7)
            c7 = compute_c5(h7, best_n)
            if c7 < best_c:
                best_c, best_h = c7, h7

        # Final SLSQP
        if time.time() < p7_end - 2:
            best_h = slsqp_refine(best_h, best_n, 50000, maxiter=500, ftol=1e-18, deadline=p7_end - 1)
            best_c = compute_c5(best_h, best_n)

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    # Force exact sum
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - best_h
            mask = room > 1e-6
            if mask.sum() > 0:
                best_h[mask] += diff / mask.sum()
        else:
            mask = best_h > 1e-6
            if mask.sum() > 0:
                best_h[mask] += diff / mask.sum()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
