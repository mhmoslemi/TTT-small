```
def project_feasible(x, n):
```

```
def compute_f(h,n):
    dx = 2.0 / n
    autoc = np.correlate(h, h, mode='full')
    H = autoc[n-1:2*n-1]
    total = float(h.sum())
    pref = np.empty(n+1, dtype=float)
    pref[0]=0.0
    np.cumsum(h, out=pref[1:])
    if n > 1:
       p = np.arange(1, n)
       fpos = dx * (pref[n - p] - H[1:])
       fneg = dx * (total - pref[p] - H[1:])
    else:
       fpos = np.zeros(0); fneg=np.zeros(0)
    f0 = dx * (total - H[0])
    return f0, fpos, fneg
```

```
def gradient_from_weights(h,n,w0,wpos,wneg):
    dx=2.0/n
    A=np.full(n, float(w0), dtype=float)
    if n > 1:
       total_wpos = float(wpos.sum())
       tail = np.zeros(n, dtype=float)
       if n > 1:
          tail[1:] = np.cumsum(wpos[::-1])
       A += total_wpos - tail
       prefix_w = np.zeros(n, dtype=float)
       prefix_w[1:] = np.cumsum(wneg)
       A += prefix_w
       wtot = np.empty(n, dtype=float)
       wtot[0] = 2.0*float(w0)
       wtot[1:] = wpos + wneg
       kernel = np.empty(2*n - 1, dtype=float)
       kernel[n-1] = wtot[0]
       kernel[n:] = wtot[1:]
       kernel[:n-1] = wtot[1:][::-1]
       B = np.convolve(h, kernel, mode='full')[n-1:n-1+n]
    else:
       B = 2.0*float(w0)*h
    return dx * (A - B)
```

```python
import time
import numpy as np
from scipy.optimize import minimize


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).reshape(-1)
    if x.size != n:
        if x.size > n:
            x = x[:n]
        else:
            x = np.concatenate([x, np.zeros(int(n - x.size), dtype=float)])
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)

    target = 0.5 * n
    s = float(x.sum())
    if abs(s - target) <= 1e-13 and x.min() >= -1e-14 and x.max() <= 1.0 + 1e-14:
        return np.clip(x, 0.0, 1.0)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - float(y.sum())
        if abs(diff) <= 1e-15:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] = y[mask] + diff / float(mask.sum())
            y = np.clip(y, 0.0, 1.0)
        else:
            y = y + diff / float(n)
            y = np.clip(y, 0.0, 1.0)
    return y


def exact_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def compute_f(h, n):
    dx = 2.0 / n
    autoc = np.correlate(h, h, mode="full")
    H = autoc[n - 1:2 * n - 1]
    total = float(h.sum())

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])

    if n > 1:
        p = np.arange(1, n)
        fpos = dx * (pref[n - p] - H[1:])
        fneg = dx * (total - pref[p] - H[1:])
    else:
        fpos = np.zeros(0, dtype=float)
        fneg = np.zeros(0, dtype=float)

    f0 = dx * (total - H[0])
    return f0, fpos, fneg


def exp_weights(f0, fpos, fneg, alpha):
    m = float(f0)
    if fpos.size:
        mp = float(np.max(fpos))
        if mp > m:
            m = mp
    if fneg.size:
        mn = float(np.max(fneg))
        if mn > m:
            m = mn

    a = float(alpha)
    cnt = 1.0 + float(fpos.size) + float(fneg.size)

    if a <= 1e-12:
        u = 1.0 / cnt
        w0 = u
        wpos = np.full(fpos.size, u, dtype=float)
        wneg = np.full(fneg.size, u, dtype=float)
        phi = m
        return w0, wpos, wneg, phi, m

    e0 = float(np.exp((float(f0) - m) * a))
    ep = np.exp((fpos - m) * a)
    en = np.exp((fneg - m) * a)

    s = e0 + float(ep.sum()) + float(en.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        u = 1.0 / cnt
        w0 = u
        wpos = np.full(fpos.size, u, dtype=float)
        wneg = np.full(fneg.size, u, dtype=float)
        phi = m
        return w0, wpos, wneg, phi, m

    w0 = e0 / s
    wpos = ep / s
    wneg = en / s
    phi = m + float(np.log(s)) / a
    return w0, wpos, wneg, phi, m


def gradient_from_weights(h, n, w0, wpos, wneg):
    dx = 2.0 / n
    A = np.full(n, float(w0), dtype=float)

    if n > 1:
        total_wpos = float(wpos.sum())
        tail = np.zeros(n, dtype=float)
        tail[1:] = np.cumsum(wpos[::-1])
        A += total_wpos - tail

        prefix_w = np.zeros(n, dtype=float)
        prefix_w[1:] = np.cumsum(wneg)
        A += prefix_w

        wtot = np.empty(n, dtype=float)
        wtot[0] = 2.0 * float(w0)
        wtot[1:] = wpos + wneg

        kernel = np.empty(2 * n - 1, dtype=float)
        kernel[n - 1] = wtot[0]
        kernel[n:] = wtot[1:]
        kernel[:n - 1] = wtot[1:][::-1]

        B = np.convolve(h, kernel, mode="full")[n - 1:n - 1 + n]
    else:
        B = 2.0 * float(w0) * h

    return dx * (A - B)


def smooth_value(h, n, alpha):
    f0, fpos, fneg = compute_f(h, n)
    w0, wpos, wneg, phi, m = exp_weights(f0, fpos, fneg, alpha)
    return phi, m


def smooth_gradient(h, n, alpha):
    f0, fpos, fneg = compute_f(h, n)
    w0, wpos, wneg, phi, m = exp_weights(f0, fpos, fneg, alpha)
    return gradient_from_weights(h, n, w0, wpos, wneg)


def smooth_obj_grad(h, n, alpha):
    f0, fpos, fneg = compute_f(h, n)
    w0, wpos, wneg, phi, m = exp_weights(f0, fpos, fneg, alpha)
    grad = gradient_from_weights(h, n, w0, wpos, wneg)
    return phi, grad, m


def vertex_from_grad(grad, n):
    s = np.zeros(n, dtype=float)
    target = n // 2
    if target <= 0:
        return s
    if target >= n:
        s[:] = 1.0
        return s
    idx = np.argpartition(grad, target - 1)[:target]
    s[idx] = 1.0
    return s


def interpolate_h(h, nnew):
    old_n = len(h)
    if old_n == nnew:
        return project_feasible(h, nnew)
    xold = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xnew = np.linspace(0.0, 2.0, nnew, endpoint=False)
    return project_feasible(np.interp(xnew, xold, h), nnew)


def default_start(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.30 * np.cos(np.pi * x) + 0.10 * np.sin(2.0 * np.pi * x)
    return project_feasible(h, n)


def transform_best(h, n):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c5(h, n)

    r = h[::-1].copy()
    cands = [
        project_feasible(r, n),
        project_feasible(1.0 - h, n),
        project_feasible(1.0 - r, n),
        project_feasible(0.5 * (h + r), n),
    ]

    for cand in cands:
        cc = exact_c5(cand, n)
        if cc < best_c - 1e-15:
            best_c = cc
            best_h = cand.copy()
    return best_h, best_c


def random_perturb(h, n, rng, deadline, max_evals=200):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c5(h, n)

    sigmas = (0.02, 0.05, 0.10, 0.20)
    evals = 0
    for sig in sigmas:
        for _ in range(max_evals // 4):
            if time.time() > deadline or evals >= max_evals:
                break
            cand = project_feasible(best_h + sig * rng.randn(n), n)
            c = exact_c5(cand, n)
            evals += 1
            if c < best_c - 1e-14:
                best_c = c
                best_h = cand.copy()
    return best_h, best_c


def fw_polish(h, n, deadline, alpha_list, max_iter_per_alpha=4000, rng=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = exact_c5(h, n)

    for alpha_idx, alpha in enumerate(alpha_list):
        if time.time() > deadline:
            break

        it = 0
        no_impr = 0
        phi_stag = 0
        escape = 0

        while it < max_iter_per_alpha:
            if time.time() > deadline:
                break

            phi, grad, cur_c = smooth_obj_grad(h, n, alpha)

            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_impr = 0
                phi_stag = 0

            if not np.all(np.isfinite(grad)):
                break

            gmax = float(np.max(np.abs(grad)))
            if gmax <= 1e-14:
                break

            vbin = vertex_from_grad(grad, n)

            sdir = 1.2 / gmax
            if sdir < 5.0:
                sdir = 5.0
            if sdir > 800.0:
                sdir = 800.0
            vpg = project_feasible(h - sdir * grad, n)

            best_trial = None
            best_tc = cur_c + 1e-13
            best_tp = phi + 1e-13

            for g in (1.0, 0.5, 0.25):
                tri = (1.0 - g) * h + g * vbin
                tc, tp = smooth_value(tri, n, alpha)
                if tc < best_tc - 1e-14 or (abs(tc - best_tc) <= 1e-14 and tp < best_tp - 1e-14):
                    best_tc = tc
                    best_tp = tp
                    best_trial = tri

            for g in (1.0, 0.5, 0.25):
                tri = (1.0 - g) * h + g * vpg
                tc, tp = smooth_value(tri, n, alpha)
                if tc < best_tc - 1e-14 or (abs(tc - best_tc) <= 1e-14 and tp < best_tp - 1e-14):
                    best_tc = tc
                    best_tp = tp
                    best_trial = tri

            if best_trial is not None:
                if best_tc < cur_c - 1e-14:
                    h = best_trial.copy()
                    no_impr = 0
                    phi_stag = 0
                elif abs(best_tc - cur_c) <= 1e-12 and best_tp < phi - 1e-14:
                    h = best_trial.copy()
                    phi_stag += 1
                    if phi_stag > 800:
                        break
                else:
                    no_impr += 1
                    if no_impr > 1200:
                        if rng is not None and escape < 4:
                            sigma = max(0.004, 0.12 / float(alpha_idx + 2))
                            h = project_feasible(h + sigma * rng.randn(n), n)
                            nc = exact_c5(h, n)
                            if nc < best_c - 1e-14:
                                best_c = nc
                                best_h = h.copy()
                            no_impr = 0
                            escape += 1
                        else:
                            break
            else:
                no_impr += 1
                if no_impr > 1200:
                    break

            it += 1
            if it % 200 == 0:
                h = project_feasible(h, n)

        h = project_feasible(h, n)

    best_h = project_feasible(best_h, n)
    best_c = exact_c5(best_h, n)
    return best_h, best_c


def make_block_start(rng, m, kind):
    if kind == 0:
        b = np.full(m, 0.5, dtype=float)
    elif kind == 1:
        b = rng.rand(m)
        b = project_feasible(b, m)
    elif kind == 2:
        b = (rng.rand(m) > 0.5).astype(float)
        b = project_feasible(b, m)
    elif kind == 3:
        b = np.zeros(m, dtype=float)
        b[0::2] = 1.0
    elif kind == 4:
        b = 0.5 + 0.35 * rng.rand(m)
        b = project_feasible(b, m)
    else:
        b = rng.rand(m)
        b = project_feasible(b, m)
    return project_feasible(b, m)


def block_penalty(b, L, alpha):
    b = np.asarray(b, dtype=float)
    b = np.clip(b, 0.0, 1.0)
    h = np.repeat(b, L)
    phi, c = smooth_value(h, len(h), alpha)
    s = float(b.sum()) - 0.5 * len(b)
    return phi + 1000.0 * s * s


def block_search(rng, deadline):
    best_h = None
    best_c = 1e100
    L = 20
    ms = [8, 12, 16, 20, 24]
    kinds = [1, 2, 3, 0, 4]

    for m in ms:
        if time.time() > deadline:
            break
        n = m * L
        for kind in kinds:
            if time.time() + 8.0 > deadline:
                break
            b0 = make_block_start(rng, m, kind)
            h0 = np.repeat(b0, L)
            c0 = exact_c5(h0, n)
            if c0 < best_c:
                best_c = c0
                best_h = h0.copy()

            if time.time() + 10.0 > deadline:
                continue

            try:
                res = minimize(
                    block_penalty,
                    b0,
                    method="Nelder-Mead",
                    args=(L, 800.0),
                    options={"maxiter": 150, "xatol": 1e-5, "fatol": 1e-7},
                )
                b = np.clip(res.x, 0.0, 1.0)
                b = project_feasible(b, m)
                h = np.repeat(b, L)
                h = project_feasible(h, n)
                c = exact_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            except Exception:
                pass

    return best_h, best_c


def slsqp_sum_eq(x, n):
    return float(np.sum(x) - 0.5 * n)


def slsqp_sum_jac(x, n):
    return np.ones_like(x, dtype=float)


def slsqp_objective(x, n, alpha):
    phi, c = smooth_value(x, n, alpha)
    return phi


def slsqp_gradient(x, n, alpha):
    return smooth_gradient(x, n, alpha)


def slsqp_polish(h, n, deadline, alphas, maxiter):
    best_h = project_feasible(h, n)
    best_c = exact_c5(best_h, n)

    for alpha in alphas:
        if time.time() + 60.0 > deadline:
            break

        bounds = np.zeros((n, 2), dtype=float)
        bounds[:, 1] = 1.0
        cons = [
            {
                "type": "eq",
                "fun": slsqp_sum_eq,
                "jac": slsqp_sum_jac,
                "args": (n,),
            }
        ]

        try:
            res = minimize(
                slsqp_objective,
                best_h.copy(),
                jac=slsqp_gradient,
                args=(n, float(alpha)),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
            )
            h2 = project_feasible(res.x, n)
            c2 = exact_c5(h2, n)
            if c2 < best_c - 1e-15:
                best_h = h2.copy()
                best_c = c2
        except Exception:
            pass

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s) - 5.0
    if total <= 20.0:
        total = max(5.0, 0.5 * float(budget_s))
    dl = t0 + total
    rng = np.random.RandomState(int(seed))

    parent_n = 250
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        parent_n = int(len(hp))
        if parent_n <= 0:
            raise ValueError("empty parent")
        if parent_n % 2 != 0:
            parent_n += 1
            hp = interpolate_h(hp, parent_n)
        hp = project_feasible(hp, parent_n)
    except Exception:
        hp = default_start(250)
        parent_n = 250

    best_h, best_c = transform_best(hp, parent_n)
    best_n = parent_n

    alpha_full = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 60000.0, 150000.0]
    alpha_high = [5000.0, 20000.0, 60000.0, 150000.0]

    t_parent = t0 + 0.20 * total
    if time.time() < t_parent - 90.0:
        h_r, c_r = random_perturb(best_h, best_n, rng, t_parent - 120.0, 200)
        if c_r < best_c - 1e-15:
            best_h = h_r
            best_c = c_r
            best_n = len(h_r)

    if time.time() < t_parent - 10.0:
        h0, c0 = fw_polish(best_h, best_n, t_parent, alpha_full, 5000, rng)
        if c0 < best_c - 1e-15:
            best_h = h0
            best_c = c0
            best_n = len(h0)

    t_block = t0 + 0.32 * total
    if time.time() + 10.0 < t_block:
        hb, cb = block_search(rng, t_block)
        if hb is not None and cb < best_c - 1e-15:
            best_h = hb
            best_c = cb
            best_n = len(hb)

    t_ladder_end = t0 + 0.90 * total
    n_min = min(parent_n, best_n)
    n_max = best_n + 320
    if n_max < n_min + 120:
        n_max = n_min + 120
    if n_max > 780:
        n_max = 780
    if n_max % 2:
        n_max += 1

    res_list = []
    for i in range(5):
        frac = float(i) / 4.0
        nn = int(round(n_min + (n_max - n_min) * frac))
        if nn % 2:
            nn += 1
        if nn < 50:
            nn = 50
        if nn > 800:
            nn = 800
        if nn not in res_list:
            res_list.append(nn)

    if parent_n not in res_list and parent_n <= n_max and parent_n % 2 == 0:
        res_list.append(parent_n)

    res_list = sorted(set(res_list))
    if len(res_list) > 5:
        idxs = [0, len(res_list) // 4, len(res_list) // 2, (3 * len(res_list)) // 4, len(res_list) - 1]
        res_list = sorted(set(res_list[i] for i in idxs))
    if not res_list:
        res_list = [best_n if best_n % 2 == 0 else best_n + 1]

    ladder_start = max(time.time(), t0)
    ladder_len = max(0.0, t_ladder_end - ladder_start)
    if ladder_len > 0.0:
        slice_time = ladder_len / float(len(res_list))
        for resort_idx, nn in enumerate(res_list):
            slice_dead = min(t_ladder_end, ladder_start + (resort_idx + 1) * slice_time)
            if time.time() + 20.0 >= slice_dead:
                continue

            h_start = interpolate_h(best_h, nn)
            h_tr, c_tr = fw_polish(h_start, nn, slice_dead, alpha_full, 3500, rng)
            if c_tr < best_c - 1e-15:
                best_h = h_tr
                best_c = c_tr
                best_n = len(h_tr)

            if time.time() + 45.0 < slice_dead:
                h_pert = interpolate_h(best_h, nn)
                sig = 0.06 + 0.02 * float(resort_idx)
                h_pert = project_feasible(h_pert + sig * rng.randn(nn), nn)
                h_b, c_b = fw_polish(h_pert, nn, slice_dead, alpha_high, 1500, rng)
                if c_b < best_c - 1e-15:
                    best_h = h_b
                    best_c = c_b
                    best_n = len(h_b)

    if time.time() + 25.0 < dl:
        h_f, c_f = fw_polish(best_h, best_n, dl - 20.0, alpha_high, 4000, rng)
        if c_f < best_c - 1e-15:
            best_h = h_f
            best_c = c_f
            best_n = len(h_f)

    for sig in (0.03, 0.01):
        if time.time() + 45.0 < dl:
            hpert = project_feasible(best_h + sig * rng.randn(best_n), best_n)
            h_s, c_s = fw_polish(hpert, best_n, dl - 20.0, alpha_high, 1500, rng)
            if c_s < best_c - 1e-15:
                best_h = h_s
                best_c = c_s
                best_n = len(h_s)

    if time.time() + 100.0 < dl and best_n <= 320:
        h_sl, c_sl = slsqp_polish(best_h, best_n, dl - 20.0, alpha_high[:1], 40)
        if c_sl < best_c - 1e-15:
            best_h = h_sl
            best_c = c_sl
            best_n = len(h_sl)

    best_h = project_feasible(best_h, best_n)
    best_c = exact_c5(best_h, best_n)
    return best_h, best_c, best_n
```
