import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def topk_mean(x, k):
    k = min(k, len(x))
    sorted_x = np.sort(x)[::-1][:k]
    return float(np.mean(sorted_x))


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.arange(len(h_old))
    x_new = np.linspace(0, len(h_old) - 1, n_new)
    return np.interp(x_new, x_old, h_old)


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")


def subgrad_for_lag(h, k_idx, n):
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def subgrad_multi(h, top_indices, n):
    grad = np.zeros(n)
    for ki in top_indices:
        grad += subgrad_for_lag(h, ki, n)
    return grad / len(top_indices)


def obj_topk(h, dx, k):
    corr = compute_correlations(h)
    return topk_mean(corr, k) * dx


def obj_topk_grad(h, dx, k):
    corr = compute_correlations(h)
    n = len(h)
    top_idx = np.argsort(corr)[::-1][:k]
    grad = subgrad_multi(h, top_idx, n)
    return grad * dx


def con_sum(h, target_sum):
    return h.sum() - target_sum


def con_sum_jac(h, target_sum):
    return np.ones(len(h))


def obj_active(h, dx, active_indices, n):
    val = 0.0
    for ki in active_indices:
        lag = ki - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            val += np.sum(h[idx] * (1.0 - h[idx + lag]))
        else:
            la = -lag
            idx = np.arange(n - la)
            val += np.sum(h[idx + la] * (1.0 - h[idx]))
    return val * dx / len(active_indices)


def obj_active_grad(h, dx, active_indices, n):
    grad = np.zeros(n)
    for ki in active_indices:
        grad += subgrad_for_lag(h, ki, n)
    return grad * dx / len(active_indices)
