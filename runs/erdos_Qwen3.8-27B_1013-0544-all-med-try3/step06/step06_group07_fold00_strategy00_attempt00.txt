def subgrad_step(h, k_worst, n):

    lag = k_worst - (n - 1)  # Lag from the previous part
    grad = np.zeros(n)

        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:

        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad
