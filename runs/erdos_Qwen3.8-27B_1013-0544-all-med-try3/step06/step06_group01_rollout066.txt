A swap (set h[i]=1, h[j]=0) is beneficial if it reduces the max correlation. The delta for this swap is a fixed vector (depending only on i, j, and the current h). The new max is max(corr + delta). If the current max is at position k*, then the new max is at most max(corr[k*] + delta[k*], max of other positions).

A quick filter: if delta[k*] >= 0 (the max position increases), the swap is unlikely to help (though it might shift the max to a lower position). If delta[k*] < 0, it's promising.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            lo, hi = 0.0, 1.0
            for _ in range(60):
                mid = (lo + hi) / 2.0
                if np.clip(h + mid, 0, 1).sum() < target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h + (lo + hi) / 2.0, 0, 1)
        else:
            lo, hi = 0.0, 1.0
            for _ in range(60):
                mid = (lo + hi) / 2.0
                if np.clip(h - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h - (lo + hi) / 2.0, 0, 1)
    # Final exact correction
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s

        else:
            ss = h.sum()
            if ss > 1e-14:

    return np.clip(h, 0.0, 1.0)

def get_gradient(h, n, alpha):
    """Compute gradient of smooth max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr_shift = corr - corr.max()
    exp_c = np.exp(alpha * corr_shift)
    p = exp_c / exp_c.sum()
    
    N = n
    grad = np.zeros(N)
    one_mh = 1.0 - h
    
    # For each lag l (from -(N-1) to (N-1)), weight is p[l + (N-1)]
    # Contribution to grad[j]:
    #   + p[l+(N-1)] * (1-h[j+l]) if 0<=j+l<N
    #   + p[l+(N-1)] * 1 if j+l < 0 or j+l >= N  (since h=0 out of range, 1-h=1)
    #   - p[l+(N-1)] * h[j-l] if 0<=j-l<N
    #   (h[j-l] = 0 out of range, so no contribution)
    
    for l in range(-(N-1), N):
        w = p[l + (N-1)]
        if w < 1e-16:
            continue
        # Part 1: w * (1-h[j+l])
        # Valid j: 0 <= j+l < N => -l <= j < N-l
        j_lo = max(0, -l)
        j_hi = min(N, N - l)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * one_mh[j_lo+l:j_hi+l]
        if j_lo > 0:
            grad[:j_lo] += w
        if j_hi < N:
            grad[j_hi:] += w
        # Part 2: -w * h[j-l]
        # Valid j: 0 <= j-l < N => l <= j < N+l
        j2_lo = max(0, l)
        j2_hi = min(N, N + l)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= w * h[j2_lo-l:j2_hi-l]
    
    return p, corr, dx * grad

def projected_gradient_step(h, n, alpha, lr, deadline):
    """One step of projected gradient with backtracking."""
    p, corr, grad = get_gradient(h, n, alpha)
    smooth_val = corr.max() + np.log(np.exp(alpha * (corr - corr.max())).sum()) / alpha
    
    # Backtracking line search
    for trial in range(30):
        h_new = project_feasible(h - lr * grad, n)
        p2, corr2, _ = get_gradient(h_new, n, alpha)
        smooth_val2 = corr2.max() + np.log(np.exp(alpha * (corr2 - corr2.max())).sum()) / alpha
        if smooth_val2 < smooth_val - 1e-16:
            return h_new, smooth_val2
        lr *= 0.5
    return h, smooth_val

def optimize_projected_gradient(h, n, alpha_list, max_iters_per_alpha, deadline):
    """Run projected gradient with increasing alpha."""
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    for alpha in alpha_list:
        h_cur = best_h.copy()
        lr = 1.0 / (n * alpha + 1)
        for it in range(max_iters_per_alpha):
            if time.time() > deadline:
                break
            h_new, val = projected_gradient_step(h_cur, n, alpha, lr, deadline)
            if np.allclose(h_new, h_cur, atol=1e-14):
                break
            h_cur = h_new
            actual = compute_c5(h_cur, n)
            if actual < best_val:
                best_val = actual
                best_h = h_cur.copy()
    return best_h, best_val
