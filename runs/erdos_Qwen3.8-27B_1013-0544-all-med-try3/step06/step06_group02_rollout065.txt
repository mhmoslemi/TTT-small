    h, best_c = lbfgs_phase(h, n_init, t1, max_iters=300)
    best_h = h.copy()
    
    # Phase 2: Subgradient method
    if time.time() < deadline - 300:
        t2 = min(time.time() + 150, deadline - 300)
        h2, c2 = subgradient_method(h, n_init, t2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            h = h2
    
    # Phase 3: Gradient-guided search
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h3, c3 = gradient_guided_search(best_h, best_n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            h = h3
    
    # Phase 4: Try different resolutions
    if time.time() < deadline - 200:
        n_candidates = [400, 500, 350]
        for n_try in n_candidates:
            if time.time() > deadline - 100:
                break
            t_rem = deadline - time.time() - 50
            if t_rem < 60:
                break
            h_t, c_t, n_t = try_new_resolution(best_h, best_n, n_try, time.time() + min(150, t_rem))
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_t
                h = h_t
    
    # Phase 5: More L-BFGS on best
    if time.time() < deadline - 100:
        t5 = min(time.time() + 100, deadline - 100)
        h5, c5 = lbfgs_phase(best_h, best_n, t5, max_iters=200)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            h = h5
    
    # Phase 6: Subgradient polishing
    if time.time() < deadline - 80:
        t6 = min(time.time() + 80, deadline - 80)
        h6, c6 = subgradient_method(best_h, best_n, t6)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Random perturbations + re-optimize
    if time.time() < deadline - 60:
        t7 = min(time.time() + 100, deadline - 60)
        h7, c7 = random_perturb_hopt(best_h, best_n, t7, rng, n_trials=15)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final pair search
    if time.time() < deadline - 10:
        t8 = deadline - 5
        best_h, best_c = pair_search(best_h, best_n, t8,
                                     [0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=80)
    
    # Phase 9: Gradient-guided final polish
    if time.time() < deadline - 3:
        t9 = deadline - 2
        h9, c9 = gradient_guided_search(best_h, best_n, t9)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
