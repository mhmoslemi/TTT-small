def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def coord_cd_step(h, i, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_lags = len(corr)
    
    # Compute coefficients a and b for coordinate i
    # corr[idx] = a[idx] * h[i] + b[idx]
    # where idx = L + (n-1) and L is the lag
    a = np.ones(n_lags)
    for idx in range(n_lags):
        L = idx - (n - 1)
        if 0 <= i + L < n:
            a[idx] -= h[i + L]
        if 0 <= i - L < n:
            a[idx] -= h[i - L]
    
    b = corr - a * h[i]
    
    # Find optimal t in [0,1] minimizing max_k (a_k * t + b_k)
    # Compute all pairwise intersections
    denom = a[:, None] - a[None, :]
    numer = b[None, :] - b[:, None]
    safe_denom = np.where(np.abs(denom) > 1e-14, denom, 1.0)
    t_mat = numer / safe_denom
    
    valid = (np.abs(denom) > 1e-14) & (t_mat >= -1e-10) & (t_mat <= 1.0 + 1e-10)
    t_mat = np.where(valid, np.clip(t_mat, 0.0, 1.0), -1.0)
    
    candidates = t_mat[t_mat >= 0].flatten()
    candidates = np.concatenate([candidates, [0.0, 1.0]])
    
    # Evaluate g(t) = max_k (a_k * t + b_k) at all candidates
    vals = candidates[:, None] * a[None, :] + b[None, :]
    g = np.max(vals, axis=1)
    best_idx = np.argmin(g)
    t_opt = candidates[best_idx]

    h[i] = t_opt

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def coord_cd_step(h, i, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_lags = len(corr)

    a = np.ones(n_lags)
    for idx in range(n_lags):
        L = idx - (n - 1)
        if 0 <= i + L < n:
            a[idx] -= h[i + L]
        if 0 <= i - L < n:
            a[idx] -= h[i - L]

    b = corr - a * h[i]

    denom = a[:, None] - a[None, :]
    numer = b[None, :] - b[:, None]
    safe_denom = np.where(np.abs(denom) > 1e-14, denom, 1.0)
    t_mat = numer / safe_denom

    valid = (np.abs(denom) > 1e-14) & (t_mat >= -1e-10) & (t_mat <= 1.0 + 1e-10)
    t_mat = np.where(valid, np.clip(t_mat, 0.0, 1.0), -1.0)

    candidates = t_mat[t_mat >= 0].flatten()
    candidates = np.concatenate([candidates, [0.0, 1.0]])

    vals = candidates[:, None] * a[None, :] + b[None, :]
    g = np.max(vals, axis=1)
    best_idx = np.argmin(g)
    t_opt = candidates[best_idx]

    h = h.copy()
    h[i] = t_opt
    h = project_feasible(h, target_sum)
    return h


def slsqp_obj(h, dx_val, beta_val):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta_val) * dx_val


def eq_con_fun(h, target_sum):
    return h.sum() - target_sum


def eq_con_jac(h, n_val):
    return np.ones(n_val)


def do_coord_descent(h, target_sum, max_sweeps, time_limit_fn):
    n = len(h)
    for sweep in range(max_sweeps):
        if time_limit_fn() < 5:
            break
        for i in range(n):
            h = coord_cd_step(h, i, target_sum)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:

    n = len(h_init)

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial solution
    phase1_deadline = budget_s * 0.45
    h = best_h.copy()
    stagnant_count = 0
    max_stagnant = 5

    while remaining() > budget_s - phase1_deadline + 50:
        h = do_coord_descent(h, target_sum, 200, remaining)
        c5 = compute_c5(h)

        if c5 < best_c5 - 1e-12:

            best_c5 = c5
            stagnant_count = 0

                h[idx] += np.random.randn(n_pert) * 0.2
                h = project_feasible(h, target_sum)
                stagnant_count = 0
                max_stagnant = max(3, max_stagnant // 2)

            best_c5 = c5_t
            n = n_try

            # Continue coordinate descent on this resolution

            c5_t = compute_c5(h_t)

                best_c5 = c5_t

                if c5 < best_c5:

                    best_c5 = c5

    # Phase 5: Final coordinate descent polish
    if remaining() > 10:
        h = best_h.copy()
        h = do_coord_descent(h, target_sum, 200, remaining)
        c5 = compute_c5(h)
        if c5 < best_c5:

            best_c5 = c5

    if remaining() > 15:

    # Final coordinate descent

        h = do_coord_descent(h, target_sum, 50, remaining)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
