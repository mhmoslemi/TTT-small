    indices = np.argsort(-h)
    binary = np.zeros(n)
    binary[indices[:target]] = 1.0
    return binary

def greedy_pass(h, c, offsets, ones, zeros, early_stop=0.5):
    """One greedy pass: try all (a,b) pairs, apply the best swap.
    Returns (h, c, ones, zeros, improved, current_max)"""
    n = len(h)
    current_max = c.max()
    best_improvement = 0.0
    best_a, best_b = -1, -1
    best_new_c = None
    
    for a in ones:
        da = np.zeros(2 * n - 1)
        vf = (a + offsets >= 0) & (a + offsets < n)
        da[vf] = -(1.0 - h[a + offsets[vf]])
        vs = (a - offsets >= 0) & (a - offsets < n)
        da[vs] += h[a - offsets[vs]]
        
        base = c + da  # c + da, reuse for all b
        
        for b in zeros:
            db = np.zeros(2 * n - 1)
            vfb = (b + offsets >= 0) & (b + offsets < n)
            db[vfb] = (1.0 - h[b + offsets[vfb]])
            vsb = (b - offsets >= 0) & (b - offsets < n)
            db[vsb] += -h[b - offsets[vsb]]
            
            new_max = (base + db).max()
            improvement = current_max - new_max
            if improvement > best_improvement:
                best_improvement = improvement
                best_a, best_b = a, b
                best_new_c = base + db
            if improvement > early_stop:
                break
        if best_improvement > early_stop:
            break
    
    if best_a >= 0 and best_improvement > 1e-12:
        h[best_a], h[best_b] = h[best_b], h[best_a]
        c = best_new_c
        ones.remove(best_a)
        zeros.remove(best_b)
        ones.append(best_b)
        zeros.append(best_a)
        return h, c, ones, zeros, True, c.max()
    return h, c, ones, zeros, False, current_max

def iterated_local_search(h, n, deadline, rng, n_restarts=20, perturb_size=8):
    """Iterated local search: greedy + random perturbation."""
    h = binary_quantize(h, n)
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    best_c5 = c.max() * dx
    best_h = h.copy()
    
    offsets = np.arange(2 * n - 1) - (n - 1)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        # Greedy passes until convergence
        passes = 0
        while time.time() < deadline - 10:
            h, c, ones, zeros, improved, cur_max = greedy_pass(
                h, c, offsets, ones, zeros, early_stop=0.3)
            if not improved:
                break
            passes += 1
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()
            if passes > 5000:
                break
        
        # Check if we should continue
        if time.time() > deadline - 10:
            break
        
        # If converged to a good solution, do a final recompute
        c = np.correlate(h, 1 - h, mode='full')
        if c.max() * dx < best_c5:
            best_c5 = c.max() * dx
            best_h = h.copy()
        
        # Random perturbation
        h = best_h.copy()
        n_perturb = rng.randint(3, perturb_size + 1)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for _ in range(n_perturb):
            if not ones or not zeros:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = h[b], h[a]
            ones.remove(a)
            zeros.remove(b)
            ones.append(b)
            zeros.append(a)
    
    return best_h, best_c5

def multi_start_optimize(h_init, n, deadline, rng):
    """Multiple starting points + iterated local search."""
    best_h = None
    best_c5 = float('inf')
    
    # Starting point 1: quantize h_init
    h1 = binary_quantize(h_init, n)
    h_opt, c_opt = iterated_local_search(h1, n, time.time() + (deadline - time.time()) * 0.5, rng)
    best_h, best_c5 = h_opt, c_opt
    
    if time.time() > deadline - 30:
        return best_h, best_c5
    
    # Starting point 2: random binary
    h2 = np.zeros(n)
    perm = rng.permutation(n)
    h2[perm[:n//2]] = 1.0
    remaining = deadline - time.time() - 10
    alloc = remaining * 0.25
    h_opt, c_opt = iterated_local_search(h2, n, time.time() + alloc, rng)
    if c_opt < best_c5:
        best_h, best_c5 = h_opt, c_opt
    
    if time.time() > deadline - 20:
        return best_h, best_c5
    
    # Starting point 3: perturbed best
    h3 = best_h.copy()
    ones = list(np.where(h3 == 1)[0])
    zeros = list(np.where(h3 == 0)[0])
    for _ in range(15):
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h3[a], h3[b] = h3[b], h3[a]
        ones.remove(a); zeros.remove(b)
        ones.append(b); zeros.append(a)
    
    remaining = deadline - time.time() - 5
    alloc = remaining * 0.4
    h_opt, c_opt = iterated_local_search(h3, n, time.time() + alloc, rng)
    if c_opt < best_c5:
        best_h, best_c5 = h_opt, c_opt
    
    return best_h, best_c5

def smooth_refinement(h, n, deadline):

    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
