```python
def binary_local_search(h_init, n, max_iters, seed):
    # h_init is binary with exactly n/2 ones
    h = h_init.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    best_max = np.max(corr) * dx
    best_h = h.copy()
    
    # Simulated annealing
    T = 0.01
    T_min = 0.0001
    cooling = (T_min / T) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        T *= cooling
        # Pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1-h, mode='full')
        max_new = np.max(corr_new) * dx
        h[i], h[j] = h[j], h[i]  # revert
        
        # Accept?
        if max_new < best_max or np.random.random() < np.exp(-(max_new - best_max) / T):
            h[i], h[j] = h[j], h[i]  # accept
            corr = corr_new
            if max_new < best_max:
                best_max = max_new
                best_h = h.copy()
    
    return best_h, best_max
```

```python
        if time_limit_fn is not None and time_limit_fn() < 15:
            break

        if it > 0:
            T = T_start * (T_end / T_start) ** (it / max_iters)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = h[j], h[i]
        new_mc = binary_max_count(h)

        delta = new_mc - curr_mc
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            curr_mc = new_mc
            if new_mc < best_mc:
                best_mc = new_mc
                best_h = h.copy()
                # Reset temperature on improvement
                T = max(T * 1.5, T_start * 0.5)
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_mc


def continuous_refine(h_init, n, time_limit_fn):
    """Refine binary/near-binary solution to continuous using SLSQP"""
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]

    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for beta in [50, 200, 1000, 5000, 20000]:
        if time_limit_fn() < 10:
            break
        for trial in range(2):
            if time_limit_fn() < 8:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.003
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5


def subgrad_polish(h, n, max_iters, time_limit_fn):
    """Subgradient method on the non-smooth max"""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr_base = 0.003

    for it in range(max_iters):
        if time_limit_fn() < 10:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        lag = k_worst - (n - 1)

        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]

        lr = lr_base / (1.0 + it * 0.001)
        h = h - lr * grad
        h = project_feasible(h, target_sum)

        if it % 5000 == 4999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0
    best_binary_h = None
    best_binary_n = 0
    best_binary_mc = float('inf')

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        c5_parent = compute_c5(h_parent)
        best_h = h_parent.copy()
        best_c5 = c5_parent
        best_n = n_parent

    # Phase 1: Binary search at multiple resolutions
    # Focus on n=70 first (matching parent), then others
    resolutions = [70, 80, 90, 100, 110, 120, 140]
    time_per_res = max(20, (budget_s * 0.45) / len(resolutions))

    for n in resolutions:
        if tleft() < 40:
            break

        # Multiple initializations
        n_starts = max(8, min(40, int(time_per_res / 2)))

        for start_idx in range(n_starts):
            if tleft() < 20:
                break

            # Choose initialization strategy
            if start_idx == 0 and n == (best_n if best_n in resolutions else 70) and best_h is not None:
                # Binarize the current best
                h_bin = (best_h > 0.5).astype(float)
                ones_c = int(h_bin.sum())
                if ones_c > n // 2:
                    ones = np.where(h_bin == 1)[0]
                    remove = rng.choice(ones, size=ones_c - n // 2, replace=False)
                    h_bin[remove] = 0
                elif ones_c < n // 2:
                    zeros = np.where(h_bin == 0)[0]
                    add = rng.choice(zeros, size=n // 2 - ones_c, replace=False)
                    h_bin[add] = 1
            elif start_idx == 1:
                h_bin = make_quadratic_binary(n, rng)
            elif start_idx == 2:
                h_bin = make_golden_binary(n, rng)
            else:
                h_bin = make_binary(n, rng)

            # Quick hill climb
            h_bin, mc = hill_climb_binary(h_bin, n, max_iters=10000, rng=rng,
                                           time_limit_fn=tleft)
            c5_bin = mc * 2.0 / n

            if mc < best_binary_mc:
                best_binary_mc = mc
                best_binary_h = h_bin.copy()
                best_binary_n = n

            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n

    # Phase 2: Extended SA from best binary
    if best_binary_h is not None and tleft() > 80:
        n = best_binary_n
        h_sa = best_binary_h.copy()
        _, mc = simulated_annealing_binary(h_sa, n, max_iters=500000, rng=rng,
                                            T_start=2.0, T_end=0.005,
                                            time_limit_fn=tleft)
        c5_sa = mc * 2.0 / n
        if mc < best_binary_mc:
            best_binary_mc = mc
            best_binary_h = h_sa.copy()
            best_binary_n = n
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    # Phase 3: Hill climb from best binary again (with more iterations)
    if best_binary_h is not None and tleft() > 40:
        n = best_binary_n
        h_hc = best_binary_h.copy()
        h_hc, mc = hill_climb_binary(h_hc, n, max_iters=50000, rng=rng,
                                      time_limit_fn=tleft)
        c5_hc = mc * 2.0 / n
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()
            best_n = n

    # Phase 4: Continuous relaxation from best binary
    if best_binary_h is not None and tleft() > 30:
        n = best_binary_n
        h_cont, c5_cont = continuous_refine(best_binary_h, n, tleft)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            best_n = n

    # Phase 5: Subgradient polish on continuous
    if best_h is not None and tleft() > 20:
        n = best_n
        h_sg, c5_sg = subgrad_polish(best_h, n, max_iters=50000, time_limit_fn=tleft)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n

    # Phase 6: Try higher resolution continuous
    if tleft() > 30 and best_h is not None:
        for n_new in [best_n + 30, best_n + 70, best_n * 2]:
            if n_new < 50 or n_new > 300 or tleft() < 15:
                continue
            h_interp = resample_h(best_h, n_new)
            h_interp = project_feasible(h_interp, n_new / 2.0)
            h_new, c5_new = continuous_refine(h_interp, n_new, tleft)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_new

    # Phase 7: Final SLSQP polish
    if tleft() > 10 and best_h is not None:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [50000, 100000]:
            if tleft() < 5:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
```
