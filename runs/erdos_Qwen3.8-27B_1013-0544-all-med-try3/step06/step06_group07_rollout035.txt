import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_overlap_array(x, n):
    """Compute overlap O(k) for all lags k from -(n-1) to n-1."""
    return np.correlate(x, 1.0 - x, mode="full")


def make_balanced_binary(x, n):
    """Adjust binary vector to have exactly n//2 ones."""
    target = n // 2
    x = (x > 0.5).astype(float)
    while x.sum() > target:
        idx = np.where(x == 1)[0]
        if len(idx) == 0:
            break
        i = np.random.randint(len(idx))
        x[idx[i]] = 0
    while x.sum() < target:
        idx = np.where(x == 0)[0]
        if len(idx) == 0:
            break
        i = np.random.randint(len(idx))
        x[idx[i]] = 1

def compute_dO(x, a, b, n):
    """Compute change in overlap array when x[a]:1->0, x[b]:0->1. Uses OLD x values."""
    k = np.arange(-(n-1), n)
    dO = np.zeros(2 * n - 1)
    
    # x[a]: 1 -> 0
    mask1 = (a + k >= 0) & (a + k < n)
    dO[mask1] += x[a + k[mask1]] - 1.0
    mask2 = (a - k >= 0) & (a - k < n)
    dO[mask2] += -x[a - k[mask2]]
    
    # x[b]: 0 -> 1
    mask3 = (b + k >= 0) & (b + k < n)
    dO[mask3] += 1.0 - x[b + k[mask3]]
    mask4 = (b - k >= 0) & (b - k < n)
    dO[mask4] += -x[b - k[mask4]]
    
    # Correction for same-term lags
    idx_ab = (n - 1) + (b - a)
    idx_ba = (n - 1) + (a - b)
    if 0 <= idx_ab < 2 * n - 1:
        dO[idx_ab] += 1.0
    if 0 <= idx_ba < 2 * n - 1:
        dO[idx_ba] += 1.0
    
    return dO

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    return np.interp(x_new, x_old, h_old)

    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h, target_sum)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    
    for beta in [50, 200, 1000, 5000, 20000, 80000]:
        if tl() < 30:
            break
        for trial in range(2):
            if tl() < 15:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.003
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    
    # Phase 5: Subgradient refinement on exact max
    if tl() > 40:
        h = best_h.copy()
        for it in range(50000):
            if tl() < 20:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                idx = np.arange(n - lag)
                grad[idx] += (1.0 - h[idx + lag])
                grad[idx + lag] -= h[idx]
            else:
                la = -lag
                idx = np.arange(n - la)
                grad[idx + la] += (1.0 - h[idx])
                grad[idx] -= h[idx + la]
            lr = 0.002 / (1.0 + it * 0.001)
            h = h - lr * grad
            h = project_feasible(h, target_sum)
            if it % 5000 == 4999:
                c5_check = compute_c5(h)
                if c5_check < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_check
        c5_check = compute_c5(h)
        if c5_check < best_c5:
            best_h = h.copy()
            best_c5 = c5_check
    
    # Phase 6: Try resampling best to nearby resolutions and re-optimizing
    if tl() > 80:
        for n_try in [best_n - 5, best_n + 5, best_n + 10, best_n - 10]:
            if n_try < 20 or tl() < 40:
                continue
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, n_try / 2.0)
            dx_t = 2.0 / n_try
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n_try / 2.0, n_try)}]
            for beta in [2000, 20000]:
                if tl() < 20:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, n_try / 2.0)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        best_n = n_try
                except Exception:
                    pass
    
    # Phase 7: Final high-beta SLSQP polish
    n = best_n
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    if tl() > 15:
        for beta in [50000, 200000]:
            if tl() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
