import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_simplex_box(x, target):
    lo = -np.max(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        exp_shifted = np.exp(shifted - mx)
        logsum = mx + np.log(exp_shifted.sum())
    else:
        exp_shifted = np.exp(shifted)
        logsum = np.log(exp_shifted.sum())
    weights = exp_shifted / exp_shifted.sum()
    obj = m + logsum / alpha
    # Gradient of c_k w.r.t. h[m]
    # c_lag = sum_i h[i]*(1-h[i+lag]) * dx
    # dc_lag/dh[m] = ((1-h[m+lag]) - h[m-lag]) * dx (with validity)
    lags = np.arange(-(n-1), n)
    T = len(lags)
    grad_c = np.zeros((n, T))
    for m in range(n):
        ip = m + lags
        mask_p = (ip >= 0) & (ip < n)
        grad_c[m, mask_p] += (1 - h_val(x, ip[mask_p]))
        im = m - lags
        mask_m = (im >= 0) & (im < n)
        grad_c[m, mask_m] -= x[im[mask_m]]
    grad_c *= dx
    grad = (weights[None, :] * grad_c).sum(axis=1)
    return obj, grad

def h_val(x, idx):
    out = np.zeros(len(idx))
    mask = (idx >= 0) & (idx < len(x))
    out[mask] = x[idx[mask]]
    return out

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        return smooth_obj(x, n, alpha)

    def obj_grad_fn(x):
        lags = np.arange(-(n-1), n)
        T = len(lags)
        c = np.correlate(x, 1 - x, mode='full')
        dx = 2.0 / n
        m = c.max() * dx
        shifted = alpha * (c * dx - m)
        mx = shifted.max()
        if mx > 500:
            exp_shifted = np.exp(shifted - mx)
            logsum = mx + np.log(exp_shifted.sum())
        else:
            exp_shifted = np.exp(shifted)
            logsum = np.log(exp_shifted.sum())
        weights = exp_shifted / exp_shifted.sum()
        obj = m + logsum / alpha
        # grad of each c_k w.r.t. h
        grad = np.zeros(n)
        for m_idx in range(n):
            ip = m_idx + lags
            mask_p = (ip >= 0) & (ip < n)
            vals_p = 1 - x[ip[mask_p]] if mask_p.any() else np.array([])
            im = m_idx - lags
            mask_m = (im >= 0) & (im < n)
            vals_m = x[im[mask_m]] if mask_m.any() else np.array([])
            grad_k = np.zeros(T)
            grad_k[mask_p] = vals_p
            grad_k[mask_m] -= vals_m
            grad_k *= dx
            grad[m_idx] = (weights * grad_k).sum()
        return obj, grad

    def sum_con(x):
        return float(x.sum() - n / 2.0)

    def sum_con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj_grad_fn, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, deadline, alpha=500, lr=0.001, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    lags = np.arange(-(n-1), n)
    T = len(lags)
    dx = 2.0 / n

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 100 == 0:
            lr = max(0.0001, 0.001 * (0.999 ** it))
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            exp_s = np.exp(shifted - mx)
        else:
            exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()

        grad = np.zeros(n)
        for m_idx in range(n):
            ip = m_idx + lags
            mask_p = (ip >= 0) & (ip < n)
            im = m_idx - lags
            mask_m = (im >= 0) & (im < n)
            g = np.zeros(T)
            g[mask_p] = 1 - h[ip[mask_p]]
            g[mask_m] -= h[im[mask_m]]
            grad[m_idx] = (weights * g).sum() * dx

        h_new = h - lr * grad
        h_new = project_simplex_box(h_new, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c

def fast_local_search(h, n, deadline, deltas):
    dx = 2.0 / n
    c = compute_c5_all(h, n)
    best_c = c.max()
    lags = np.arange(-(n-1), n)
    T = len(lags)

    A = np.zeros((n, T))
    B = np.zeros((n, T))
    for m in range(n):
        ip = m + lags
        mask = (ip >= 0) & (ip < n)
        A[m, mask] = 1 - h[ip[mask]]
        im = m - lags
        mask2 = (im >= 0) & (im < n)
        B[m, mask2] = h[im[mask2]]

    for d in deltas:
        for _pass in range(500):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        delta_c = d * (A[i] - B[i] - A[j] + B[j])
                        new_max = (c + delta_c).max()
                        if new_max < best_c - 1e-17:
                            h[i] += d
                            h[j] -= d
                            c += delta_c
                            best_c = new_max
                            ip = i + lags
                            msk = (ip >= 0) & (ip < n)
                            A[i] = 0; A[i, msk] = 1 - h[ip[msk]]
                            im = i - lags
                            msk2 = (im >= 0) & (im < n)
                            B[i] = 0; B[i, msk2] = h[im[msk2]]
                            ip = j + lags
                            msk = (ip >= 0) & (ip < n)
                            A[j] = 0; A[j, msk] = 1 - h[ip[msk]]
                            im = j - lags
                            msk2 = (im >= 0) & (im < n)
                            B[j] = 0; B[j, msk2] = h[im[msk2]]
                            improved = True
                    if 0 <= h[i] - d <= 1 and 0 <= h[j] + d <= 1:
                        delta_c = d * (-A[i] + B[i] + A[j] - B[j])
                        new_max = (c + delta_c).max()
                        if new_max < best_c - 1e-17:
                            h[i] -= d
                            h[j] += d
                            c += delta_c
                            best_c = new_max
                            ip = i + lags
                            msk = (ip >= 0) & (ip < n)
                            A[i] = 0; A[i, msk] = 1 - h[ip[msk]]
                            im = i - lags

                            ip = j + lags

                            im = j - lags
                            msk2 = (im >= 0) & (im < n)
                            B[j] = 0; B[j, msk2] = h[im[msk2]]
                            improved = True
            if not improved:
                break
    return h, best_c

def triple_local_search(h, n, deadline, deltas, radius=30):
    dx = 2.0 / n
    c = compute_c5_all(h, n)
    best_c = c.max()
    lags = np.arange(-(n-1), n)
    T = len(lags)

    A = np.zeros((n, T))
    B = np.zeros((n, T))
    for m in range(n):
        ip = m + lags
        mask = (ip >= 0) & (ip < n)
        A[m, mask] = 1 - h[ip[mask]]
        im = m - lags
        mask2 = (im >= 0) & (im < n)
        B[m, mask2] = h[im[mask2]]

    for d in deltas:
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        delta_c = d * (A[i] - B[i] - A[j] + B[j])
                        new_max = (c + delta_c).max()
                        if new_max < best_c - 1e-17:
                            h[i] += d
                            h[j] -= d
                            c += delta_c
                            best_c = new_max
                            ip2 = i + lags; msk = (ip2 >= 0) & (ip2 < n)
                            A[i] = 0; A[i, msk] = 1 - h[ip2[msk]]
                            im2 = i - lags; msk2 = (im2 >= 0) & (im2 < n)
                            B[i] = 0; B[i, msk2] = h[im2[msk2]]
                            ip3 = j + lags; msk3 = (ip3 >= 0) & (ip3 < n)
                            A[j] = 0; A[j, msk3] = 1 - h[ip3[msk3]]
                            im3 = j - lags; msk4 = (im3 >= 0) & (im3 < n)
                            B[j] = 0; B[j, msk4] = h[im3[msk4]]
                            # Also update pairs within radius of j
                            improved = True
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [200, 1000, 5000, 20000, 100000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        alloc = min(25, (deadline - time.time()) * 0.15)
        if alloc < 3:
            break

        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()

    if time.time() < deadline - 10:

        if ct < best_c:

            best_h = ht

    if time.time() < deadline - 3:

        best_h, best_c = fast_local_search(best_h, n, deadline - 1, deltas)

    best_h = project_feasible(best_h, n)

    return best_h, best_c
