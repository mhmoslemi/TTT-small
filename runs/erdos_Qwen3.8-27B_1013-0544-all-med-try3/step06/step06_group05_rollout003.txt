def project_feasible(h, n):
    # Project onto {0 ≤ h ≤ 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-15:

**Better approach: Exact max with active set**

2. Identify the top lag k*

c(k*) = sum_i h[i]*(1-h[i+k*]) = sum_i h[i] - sum_i h[i]*h[i+k*]

The first term is constant (for interior lags). The second term is sum_i h[i]*h[i+k*], which is a quadratic form in h. It's not concave in general.

But if we restrict to binary h (h[i] ∈ {0,1}), then h[i]*h[i+k*] = 1 iff both h[i]=1 and h[i+k*]=1. So maximizing sum_i h[i]*h[i+k*] is equivalent to maximizing the number of "agreements" at lag k*, which is a combinatorial problem.

For continuous h, the problem is non-convex. But we can use **alternating minimization**:
- Fix all h[j] except h[i], optimize h[i] (1D problem)
- Cycle through all i

For fixed h[j] (j≠i), c(k) as a function of h[i] is:
c(k) = h[i]*(1-h[i+k]) + [terms not involving h[i]] ... for lag k where i and i+k are both in range.

Actually, h[i] appears in c(k) for multiple lags k:
- As the first factor: c(k) includes h[i]*(1-h[i+k]) for lag k
- As the second factor: c(k) includes h[i-k]*(1-h[i]) for lag k

So c(k) = [h[i]*(1-h[i+k])] + [h[i-k]*(1-h[i])] + [other terms]
= h[i] - h[i]*h[i+k] + h[i-k] - h[i]*h[i-k] + [other]
= h[i]*(1 - h[i+k] - h[i-k]) + h[i-k] + [other]

This is linear in h[i]! (For fixed other h values.)

So for each lag k, c(k) is linear in h[i] (when we fix all other h values). The max of linear functions is convex in h[i]. So minimizing max_k c(k) over h[i] (with all other h fixed) is a 1D convex optimization problem!

This means coordinate descent works! And it's exact (no smoothing needed).

Actually, the 1D problem is simpler. We have lines f_k(x) = a_k * x + b_k. We want min_{x∈[0,1]} max_k f_k(x).

The maximum of lines is a convex piecewise-linear function. The minimum is at a "kink" point (intersection of two lines) or at a boundary (0 or 1).

To find it:
1. Evaluate max_k f_k(x) at x=0 and x=1
2. Find all intersection points of pairs of lines in [0,1]
3. Evaluate max_k f_k(x) at each intersection
4. Take the minimum

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_correlations(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def objective(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_obj_val(h, n, i, x):
    """Compute max_k c(k) with h[i] temporarily set to x."""
    old = h[i]
    h[i] = x
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    val = float(c.max())
    h[i] = old
    return val

def optimize_coord(h, n, i, iters=45):
    """Minimize max c(k) over h[i] using ternary search on [0,1]."""
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_val = hi - gr * (hi - lo)
    d_val = lo + (hi - lo) - (hi - c_val)
    fc = coord_obj_val(h, n, i, c_val)
    fd = coord_obj_val(h, n, i, d_val)
    for _ in range(iters):
        if fc < fd:
            hi = d_val
            d_val = c_val
            fd = fc
            c_val = hi - gr * (hi - lo)
            fc = coord_obj_val(h, n, i, c_val)
        else:
            lo = c_val
            c_val = d_val
            fc = fd
            d_val = lo + (hi - lo) - (hi - c_val)
            fd = coord_obj_val(h, n, i, d_val)
    return (lo + hi) / 2.0

def coordinate_descent(h, n, deadline, max_passes=100, rng=None):
    """Cyclic coordinate descent on the true (unsmoothed) objective."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            x_opt = optimize_coord(h, n, i, iters=35)
            h[i] = x_opt
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def local_search(h, n, deadline, deltas, max_passes=100):
    """Pair-move local search with full correlation evaluation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

def upsample_h(h, n_old, n_new):
    """Upsample h from n_old to n_new via linear interpolation (periodic)."""
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    # Tile for periodic interpolation
    h_tiled = np.concatenate([h, h[:n_new - n_old + 1] if n_new - n_old + 1 > 0 else []])
    x_tiled = np.concatenate([x_old, x_old + 2.0 * np.arange(1, max(1, (n_new - n_old) // n_old + 1))])
    # Simple: use np.interp with the original points, wrapping
    h_new = np.interp(x_new, x_old, h)
    # Handle wrap-around: for x_new near 2, interpolate using h values near 0
    # np.interp doesn't do periodic, so do it manually
    idx = np.searchsorted(x_old, x_new)
    idx = np.clip(idx, 1, n_old)
    t = (x_new - x_old[idx - 1]) / (x_old[idx] - x_old[idx - 1])
    t = np.clip(t, 0, 1)
    h_new = (1 - t) * h[idx - 1] + t * h[idx % n_old]
    return project_feasible(h_new, n_new)
