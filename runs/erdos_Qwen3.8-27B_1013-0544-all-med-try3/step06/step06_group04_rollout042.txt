def smooth_max(f,beta):
   if beta <= 0: return f.max()
   fmax = f.max()
   a = beta*(f - fmax)
   # underflow safe: shift by a.max=0
   z = np.sum(np.exp(a))
   return float(fmax + np.log(z)/beta)

   e=np.exp(a) # a<=0
   z=e.sum()
   return e/z

def fk_values(h, n):
    dx = 2.0 / float(n)
    omh = 1.0 - h
    if _HAVE_SCIPY_CORRELATE and n >= 350:

    c = np.asarray(c, dtype=float)

import time
import numpy as np
from scipy.optimize import minimize

try:
    from scipy.signal import correlate
    _HAVE_SCIPY_CORRELATE = True
except Exception:
    _HAVE_SCIPY_CORRELATE = False


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(np.asarray(h, dtype=float), 1.0 - np.asarray(h, dtype=float), mode='full')
    return float(np.max(c) * dx)


def fk_values(h, n):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    om = 1.0 - h
    if _HAVE_SCIPY_CORRELATE and n >= 512:
        c = correlate(h, om, mode='full', method='fft')
    else:
        c = np.correlate(h, om, mode='full')
    return np.asarray(c, dtype=float) * dx


def smooth_max(f, beta):
    fmax = float(np.max(f))
    if beta <= 1e-12:
        return fmax
    a = beta * (f - fmax)
    z = float(np.sum(np.exp(a)))
    if z <= 0.0 or not np.isfinite(z):
        return fmax
    return float(fmax + np.log(z) / beta)


def softmax_weights(f, beta):
    w = np.zeros(f.shape, dtype=float)
    fmax = float(np.max(f))
    if beta <= 1e-12:
        w[int(np.argmax(f))] = 1.0
        return w
    a = beta * (f - fmax)
    e = np.exp(a)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        w[int(np.argmax(f))] = 1.0
        return w
    return e / z


def grad_from_weights(h, n, w):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float)
    g = np.zeros(n, dtype=float)
    center = n - 1
    thr = 1e-14

    if w[center] > thr:
        g += dx * w[center] * (1.0 - 2.0 * h)

    nz = np.flatnonzero(w > thr)
    if nz.size == 0:
        return g

    if nz.size < 2 * n:
        rs = np.unique(np.abs(nz - center))
        for rk in rs:
            r = int(rk)
            if r <= 0 or r >= n:
                continue
            wp = w[center + r]
            wn = w[center - r]
            s = wp + wn
            if s <= thr:
                continue
            sdx = dx * s
            m = n - r
            g[:m] += dx * wn - sdx * h[r:]
            g[r:] += dx * wp - sdx * h[:m]
    else:
        for r in range(1, n):
            wp = w[center + r]
            wn = w[center - r]
            s = wp + wn
            if s <= thr:
                continue
            sdx = dx * s
            m = n - r
            g[:m] += dx * wn - sdx * h[r:]
            g[r:] += dx * wp - sdx * h[:m]
    return g


def smooth_fun(x, n, beta):
    return smooth_max(fk_values(x, n), beta)


def smooth_jac(x, n, beta):
    return grad_from_weights(x, n, softmax_weights(fk_values(x, n), beta))


def pen_fun(x, n, beta, pen):
    s = float(np.sum(x) - 0.5 * float(n))
    return smooth_max(fk_values(x, n), beta) + pen * s * s


def pen_jac(x, n, beta, pen):
    f = fk_values(x, n)
    g = grad_from_weights(x, n, softmax_weights(f, beta))
    s = float(np.sum(x) - 0.5 * float(n))
    if abs(s) > 0.0:
        g = g + (2.0 * pen * s) * np.ones(n, dtype=float)
    return g


def sum_con_fun(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(int(n), dtype=float)


def project_close(x, n):
    target = 0.5 * float(n)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    for _ in range(30):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - x
            tot = float(np.sum(room))
            if tot < 1e-15:
                break
            x += (diff / tot) * room
        else:
            tot = float(np.sum(x))
            if tot < 1e-15:
                break
            x += (diff / tot) * x
    return np.clip(x, 0.0, 1.0)


def project_sum_box(x, n):
    target = 0.5 * float(n)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if x.size == 0:
        return x

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.clip(x - tau, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            room = 1.0 - x
            order = np.argsort(-room)
            rem = diff
            for idx in order:
                if rem <= 0.0:
                    break
                add = min(float(room[idx]), float(rem))
                if add > 0.0:
                    x[idx] += add
                    rem -= add
        else:
            mass = x.copy()
            order = np.argsort(-mass)
            rem = -diff
            for idx in order:
                if rem <= 0.0:
                    break
                sub = min(float(mass[idx]), float(rem))
                if sub > 0.0:
                    x[idx] -= sub
                    rem -= sub
    return x


def optimize_slsqp_once(h, n, beta, maxiter):
    h = np.asarray(h, dtype=float).copy()
    h = project_close(h, n)
    c0 = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (int(n),)}]
    try:
        res = minimize(
            smooth_fun,
            h,
            args=(int(n), float(beta)),
            method='SLSQP',
            jac=smooth_jac,
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': 0, 'iprint': 0}
        )
        if res.x is not None:
            h2 = project_close(np.asarray(res.x, dtype=float), n)
            c2 = compute_c5(h2, n)
            if np.isfinite(c2):
                return h2, c2
    except Exception:
        pass
    return h, c0


def optimize_lbfgs_once(h, n, beta, pen, maxiter):
    h = np.asarray(h, dtype=float).copy()
    h = project_close(h, n)
    c0 = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * int(n)
    try:
        res = minimize(
            pen_fun,
            h,
            args=(int(n), float(beta), float(pen)),
            method='L-BFGS-B',
            jac=pen_jac,
            bounds=bounds,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 20, 'disp': 0}

            h2 = project_close(np.asarray(res.x, dtype=float), n)
            c2 = compute_c5(h2, n)

    except Exception:
        pass
    return h, c0
