import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha <= 0:
        return float(m)
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_local_search(h0, n, deadline, max_stall=5):
    """Local search on 0-1 sequences: best-improvement then first-improvement."""
    rng = np.random.RandomState(0)
    h = h0.copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    num_ones = int(h.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h == 1)[0]
        rng.shuffle(one_idx)
        for idx in one_idx[:num_ones - target]:
            h[idx] = 0
    elif num_ones < target:
        zero_idx = np.where(h == 0)[0]
        rng.shuffle(zero_idx)
        for idx in zero_idx[:target - num_ones]:
            h[idx] = 1

    best_c = compute_c5(h, n)
    stall = 0

    while time.time() < deadline and stall < max_stall:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        best_move_c = best_c
        best_move = None

        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best_move_c - 1e-14:
                    best_move_c = c
                    best_move = (i, j)
                h[i] = 1.0
                h[j] = 0.0

        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            best_c = best_move_c
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            if stall < max_stall - 1 and len(ones) > 0 and len(zeros) > 0:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                best_c = compute_c5(h, n)

    return h, best_c

def continuous_local_search(h, n, deadline, deltas, radius=None, max_passes=200):
    """Continuous pair-move local search."""
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n

    for d in deltas:
        passes = 0
        while time.time() < deadline and passes < max_passes:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
            if not improved:
                break
            passes += 1

    return h, best_c

def active_set_optimize(h, n, deadline, m=20):
    """Iteratively optimize the active set (top-m correlations)."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)

    while time.time() < deadline - 5:
        corr = np.correlate(h, 1 - h, mode='full')
        target = best_c / dx
        # Find active set
        active = np.where(corr > target * 0.7)[0]
        if len(active) == 0:
            break
        active = active[np.argsort(corr[active])[-m:]]

        # Build sub-gradient: average gradient of active k values
        # c[k] = sum_i h[i]*a_k[i] where a_k depends on k
        # Instead, use smoothed objective restricted to active set
        def obj_active(x, act, n, dx, m):
            c = np.correlate(x, 1 - x, mode='full')
            vals = c[act] * dx
            return float(np.max(vals))

        # Use SLSQP on the restricted problem with high alpha
        alpha = 50000
        h_new = slsqp_opt(h, n, alpha, maxiter=1000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            h = h_new
        else:
            break

    return h, best_c

def coarse_to_fine(h_init, n_init, n_fine, deadline, rng):
    """Optimize at coarse resolution, then refine to fine resolution."""
    # Downsample
    x_coarse = np.linspace(0, 2, n_init)
    x_fine = np.linspace(0, 2, n_fine)
    h_coarse = np.interp(x_coarse, x_coarse, h_init)  # already at coarse
    h_coarse = project_feasible(h_coarse, n_init)

    # Optimize at coarse
    alloc_coarse = min(60, (deadline - time.time()) * 0.3)
    h_coarse_opt = h_coarse.copy()
    for alpha in [1000, 5000, 20000]:
        if time.time() > time.time() + alloc_coarse:
            break
        ht = slsqp_opt(h_coarse_opt, n_init, alpha, maxiter=1000)
        c = compute_c5(ht, n_init)
        c_prev = compute_c5(h_coarse_opt, n_init)
        if c < c_prev - 1e-10:
            h_coarse_opt = ht

    # Upsample to fine
    h_fine = np.interp(x_fine, x_coarse, h_coarse_opt)
    h_fine = project_feasible(h_fine, n_fine)
    return h_fine

def multi_restart(h_best, n, deadline, rng, n_restarts=5):
    """Try perturbed versions of the best solution."""
    best_c = compute_c5(h_best, n)
    best_h = h_best.copy()

    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        scale = 0.05 + 0.15 * (r / max(1, n_restarts - 1))
        hp = h_best + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)

        # Quick SLSQP
        ht = slsqp_opt(hp, n, 20000, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-10:
            best_c = c
            best_h = ht

    return best_h, best_c
