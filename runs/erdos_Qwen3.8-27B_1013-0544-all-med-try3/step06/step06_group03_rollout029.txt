    a = np.zeros(2*n-1)
    # h[i] appears in F_j (shift j) when:
    # As first factor: 0 <= i+j < n => j in [-i, n-1-i], a_j += (1-h[i+j])*dx
    lo1 = max(-(n-1), -i)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def correlate_c5(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha, active_idx=None):
    if active_idx is not None:
        c = correlate_c5(x, n)[active_idx]
    else:
        c = correlate_c5(x, n)
    m = c.max()
    if alpha < 1:
        return float(m)
    sh = alpha * (c - m)
    sh_max = sh.max()
    if sh_max > 500:
        val = m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha
    else:
        val = m + float(np.log(np.sum(np.exp(sh)))) / alpha
    return float(val)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000, active_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha, active_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    best_c = compute_c5(h, n)
    corr = correlate_c5(h, n)
    target = n / 2.0

            if old_val < 1e-12 or old_val > 1 - 1e-12:
                continue
            
            # Compute a_j = dF_j/dh[i] for all shifts j

            # For j == 0: F_0 = sum_p h[p](1-h[p])*dx, dF_0/dh[i] = dx*(1-2*h[i])
            
            a = np.zeros(2 * n - 1)

            j_hi = min(n - 1, n - 1 - i)

            # As function of x: 1 - dx*(x^2 + S) where S = sum_{p!=i} h[p]^2

            best_idx = np.argmin(f_max)
            best_x = xs[best_idx]
            
            # Refine with golden section

                gr = (np.sqrt(5.0) - 1.0) / 2.0

                    m2 = lo + gr * (hi - lo)

                    f1 = a * m1 + b

                    v1 = f1.max()

                    f0_2 = 1.0 - dx * (m2 * m2 + S)
                    f2 = a * m2 + b
                    f2[n - 1] = f0_2
                    v2 = f2.max()
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1
                best_x = (lo + hi) / 2.0

def near_pair_search(h, n, deadline, radius=50, top_m=30):

    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        while time.time() < deadline:
            c = correlate_c5(h, n)
            active_idx = np.argsort(c)[-top_m:]
            active_shifts = active_idx - (n - 1)
            
            improved = False
            for i in range(0, n, max(1, n // 80)):
                if time.time() > deadline:
                    return h, best_c
                for j_shift in active_shifts:
                    if abs(j_shift) > radius:
                        continue
                    j = i + int(j_shift)
                    if j < 0 or j >= n or j == i:
                        continue
                    for direction in [1, -1]:
                        di = d * direction
                        dj = -d * direction
                        hi, hj = h[i], h[j]
                        if 0 <= hi + di <= 1 and 0 <= hj + dj <= 1:

                                improved = True

                                h[j] = hj

    return h, best_c

    if time.time() < deadline - 50:
        alloc4 = min(150, (deadline - time.time()) * 0.25)
        best_h, best_c = coord_descent(best_h, n_init, time.time() + alloc4, grid_size=200)
    
    # Phase 5: Try higher resolution
    if time.time() < deadline - 100:
        n_high = 500
        x_init = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_init, best_h)
        h_high = project_feasible(h_high, n_high)
        c_high = compute_c5(h_high, n_high)
        
        if c_high < best_c + 0.01:  # Promising
            # SLSQP on higher res (fewer iterations)
            ht = slsqp_opt(h_high, n_high, 5000, maxiter=500)
            c_test = compute_c5(ht, n_high)
            if c_test < best_c:
                best_c = c_test
                best_h = ht
            ht = slsqp_opt(h_high, n_high, 50000, maxiter=400)
            c_test = compute_c5(ht, n_high)
            if c_test < best_c:
                best_c = c_test
                best_h = ht
            # Coordinate descent on high res
            if time.time() < deadline - 60:
                h_high = best_h if len(best_h) == n_high else h_high
                n_curr = len(h_high)
                best_h, best_c = coord_descent(h_high, n_curr, deadline - 60, grid_size=100)
    
    # Phase 6: Refine at original resolution with best found
    if len(best_h) != n_init:
        # Downsample back to n_init if high res was used
        x = np.linspace(0, 2, len(best_h))
        h_back = np.interp(np.linspace(0, 2, n_init), x, best_h)
        h_back = project_feasible(h_back, n_init)
        if compute_c5(h_back, n_init) < best_c:
            best_h = h_back
            best_c = compute_c5(best_h, n_init)
        else:
            # Keep high res version
            pass
    
    n_best = len(best_h)
    
    # Phase 7: Final active set SLSQP
    if time.time() < deadline - 30:
        best_h, best_c = active_set_slsqp(best_h, n_best, time.time() + min(100, deadline - time.time() - 30), top_m=25)
    
    # Phase 8: Final coordinate descent
    if time.time() < deadline - 20:
        best_h, best_c = coord_descent(best_h, n_best, time.time() + min(80, deadline - time.time() - 20), grid_size=200)
    
    # Phase 9: Pair search polish
    if time.time() < deadline - 5:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2, radius=60, top_m=25)
        else:
            best_h, best_c = pair_search_small(best_h, n, deadline - 2)
    
    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 2:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 1000000, maxiter=500)
        c_val = compute_c5(ht, n)
        if c_val < best_c:
            best_c = c_val
            best_h = ht
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    # Fallback
    if not np.isfinite(best_c) or best_c > 0.5:
        best_h = np.full(n_init, 0.5)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
