def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    
    sh2 = sh - mx
    w = np.exp(sh2)
    wsum = w.sum()
    w = w / wsum
    
    grad = np.zeros(n)
    for idx in range(len(c)):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-15:
            continue
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:] += wi * (1.0 - x[:nk])
                grad[:nk] -= wi * x[k:]
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                grad[:nk] += wi * (1.0 - x[kk:])
                grad[kk:] -= wi * x[:nk]
    grad *= dx
    return val, grad

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

    return np.ones(n)

def slsqp_with_grad(h, n, alpha, maxiter=2000):

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    
    def func(x):
        return obj_and_grad(x, n, alpha)[0]
    
    def jac(x):
        return obj_and_grad(x, n, alpha)[1]
    
    try:
        res = minimize(func, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def proj_gradient_descent(h, n, deadline, lr=0.01, momentum=0.9, alpha=1000):
    """Projected gradient descent with momentum on smooth objective"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    t0 = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        _, g = obj_and_grad(h, n, alpha)
        # Remove component along constraint (mean)
        g = g - np.mean(g)
        
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):

The idea: at each iteration, linearize the problem around the current h:

This is essentially a **cutting plane / LP approximation** method. Each iteration solves an LP to get a direction, then takes a (possibly damped) step.

This could be much more effective than SLSQP because:
1. Each subproblem is exact (LP solved to optimality)
2. The direction is the exact linearized direction to reduce the max
3. No convergence issues with nonlinear solvers

```python
def lp_step(h, n, max_lags=None):
    """Solve LP to find direction that minimizes linearized max correlation"""
    from scipy.optimize import linprog
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(c)
    
    if max_lags is not None and max_lags < n_lags:
        # Focus on top lags
        top_idx = np.argsort(c)[-max_lags:]
    else:
        top_idx = range(n_lags)
    
    # LP: min t
    # s.t. c[k] + (grad_k @ d) * ... <= t  for k in top_idx
    #      0 <= h[i] + d[i] <= 1  for all i
    #      sum(d) = 0  (to maintain sum constraint)
    
    # Variables: d[0..n-1], t
    # x = [d_0, ..., d_{n-1}, t]
    # n_vars = n + 1
    
    # Objective: minimize t → c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each k in top_idx: grad_k^T d - t <= -c[k]
    # (grad_k is the gradient of C_k/h w.r.t. h, which we computed)
    
    # Also: h[i] + d[i] <= 1 → d[i] <= 1 - h[i]
    #       h[i] + d[i] >= 0 → -d[i] <= h[i]
    
    n_ineq = len(top_idx) + 2 * n
    
    A_ineq = np.zeros((n_ineq, n + 1))
    b_ineq = np.zeros(n_ineq)
    
    row = 0
    for idx in top_idx:
        k = idx - (n - 1)
        # Compute grad_k
        g = np.zeros(n)
        if k >= 0:
            nk = n - k
            if nk > 0:
                g[k:] = (1.0 - h[:nk])
                g[:nk] -= h[k:]
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                g[:nk] = (1.0 - h[kk:])
                g[kk:] -= h[:nk]
        g *= dx
        
        A_ineq[row, :n] = g
        A_ineq[row, n] = -1.0
        b_ineq[row] = -c[idx]
        row += 1
    
    # d[i] <= 1 - h[i]
    for i in range(n):
        A_ineq[row, i] = 1.0
        b_ineq[row] = 1.0 - h[i]
        row += 1
    
    # -d[i] <= h[i]
    for i in range(n):
        A_ineq[row, i] = -1.0
        b_ineq[row] = h[i]
        row += 1
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        return d
    return None
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_k(h, n, k):
    """Compute d(C_k)/dh vector (includes dx factor)"""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            g[k:] += (1.0 - h[:nk])
            g[:nk] -= h[k:]
    else:
        kk = -k
        nk = n - kk
        if nk > 0:
            g[:nk] += (1.0 - h[kk:])
            g[kk:] -= h[:nk]
    return g * dx

def lp_solve(h, n, c, top_idx):
    """Solve the LP subproblem. Returns direction d (length n) or None."""
    n_lags = len(top_idx)
    
    # Variables: [d_0, ..., d_{n-1}, t], total n+1
    n_vars = n + 1
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1) For each lag k in top_idx: grad_k^T d - t <= -c[k]
    # 2) d[i] <= 1 - h[i]
    # 3) -d[i] <= h[i]
    
    n_ineq = n_lags + 2 * n
    A_ub = np.zeros((n_ineq, n_vars))
    b_ub = np.zeros(n_ineq)
    
    row = 0
    for idx in top_idx:
        k = idx - (n - 1)
        g = get_grad_k(h, n, k)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -c[idx]
        row += 1
    
    # d[i] <= 1 - h[i]
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    
    # -d[i] <= h[i]
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def lp_optimize(h, n, deadline, max_lags=None, max_iters=5000):
    """LP-based optimization"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 1.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        n_total_lags = len(c)
        current_max = c.max()
        
        if max_lags is not None and max_lags < n_total_lags:
            top_idx = np.argsort(c)[-max_lags:]
        else:
            top_idx = np.arange(n_total_lags)
        
        d = lp_solve(h, n, c, top_idx)
        if d is None:
            break
        
        # Check if direction is meaningful
        if np.max(np.abs(d)) < 1e-12:
            break
        
        # Take step with damping
        h_new = h + alpha * d
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            alpha = min(1.0, alpha * 1.1)  # increase step
        elif c_new < current_max:
            best_c = min(best_c, c_new)
            best_h = h_new.copy()
            alpha = min(1.0, alpha * 1.05)
        else:
            alpha = max(0.05, alpha * 0.5)  # reduce step
            if alpha < 0.05:
                break
        
        h = h_new
    
    return best_h, best_c
