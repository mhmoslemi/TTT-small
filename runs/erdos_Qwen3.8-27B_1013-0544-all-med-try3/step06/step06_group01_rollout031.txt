import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, n, alpha, mu, maxiter, deadline):
    """Optimize using L-BFGS-B with analytical gradient."""
    def fun(x):
        return smooth_max_obj_and_grad(x, n, alpha, mu)[0]
    
    def jac(x):
        return smooth_max_obj_and_grad(x, n, alpha, mu)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', 
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)

def binary_refine(h, n, deadline, max_sweeps=200):

    target = n // 2

    zeros = set(np.where(h_bin == 0)[0].tolist())

                jmn = j - m_range

                msk = (imn >= 0) & (imn < n)

                    break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w_sum = w.sum()
    obj = m + np.log(w_sum) / alpha
    w = w / w_sum
    first_term = np.correlate(w, 1.0 - h, mode='full')[n-1 : 2*n-1]
    second_term = np.correlate(w, h, mode='full')[n-1 : -1 : -1]
    grad = dx * (first_term - second_term)
    return obj, grad, w

def projected_gradient_step(h, n, alpha, step):
    dx = 2.0 / n
    obj, grad, w = smooth_max_and_grad(h, n, alpha)
    min_val = (h - step * grad).min()
    max_val = (h - step * grad).max()
    lo = min_val - 1.0
    hi = max_val + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x_trial = np.clip(h - step * grad + mid, 0.0, 1.0)
        s = x_trial.sum()
        if s < n / 2.0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_new = np.clip(h - step * grad + lam, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, obj

def continuous_optimize(h, n, deadline, max_iter=10000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    alpha = 500.0
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new, obj = projected_gradient_step(h, n, alpha, step)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-10:
            best_h = h_new.copy()
            best_c = c_new
            stall_count = 0
            step = min(step * 1.05, 0.3)
        else:
            stall_count += 1
            step *= 0.8
            
        h = h_new
        
        if it % 300 == 299:
            alpha = min(alpha * 1.8, 50000.0)
        
        if stall_count > 200:
            step = max(step * 0.5, 0.005)
            if step < 0.005:
                break
    
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def binary_local_search(h, n, deadline, max_sweeps=500, seed=42):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def lbfgs_polish(h, n, alpha, mu, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w_sum = w.sum()
        obj = m + np.log(w_sum) / alpha
        w = w / w_sum
        first_term = np.correlate(w, 1.0 - x, mode='full')[n-1 : 2*n-1]
        second_term = np.correlate(w, x, mode='full')[n-1 : -1 : -1]
        grad = dx * (first_term - second_term)
        penalty_val = mu * (x.sum() - target) ** 2
        grad_pen = 2.0 * mu * (x.sum() - target)
        return obj + penalty_val, grad + grad_pen
    
    def fun(x):
        return obj_grad(x)[0]
    
    def jac(x):
        return obj_grad(x)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h, n)

def multi_restart_continuous(h, n, deadline, n_restarts=3, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(n_restarts):
        t_rem = deadline - time.time()
        if t_rem < 30:
            break
        if r == 0:
            h_start = h.copy()
        else:
            pert = rng.uniform(-0.15, 0.15, n)
            h_start = project_feasible(h.copy() + pert, n)
        
        t_alloc = deadline - 30 - r * 5
        h_res, c_res = continuous_optimize(h_start, n, t_alloc, max_iter=5000)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Continuous optimization at n=500 (upsampled from parent)
    p1 = t0 + 0.20 * (budget_s - 5)
    nh = 500
    h500 = upsample(best_h, n, nh)
    h_opt, c_opt = continuous_optimize(h500, nh, p1 - 10, max_iter=8000)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt.copy()
        n = nh
    
    # Phase 2: Binary refinement at n=500
    p2 = t0 + 0.35 * (budget_s - 5)
    if time.time() < p2 - 10:
        h_bin, c_bin = binary_local_search(best_h, n, p2, max_sweeps=1000, seed=seed)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
    
    # Phase 3: L-BFGS polish
    p3 = t0 + 0.45 * (budget_s - 5)
    if time.time() < p3:
        h_lb = lbfgs_polish(best_h, n, 20000, 50.0, 3000, p3)
        c_lb = compute_c5(h_lb, n)
        if c_lb < best_c:
            best_c = c_lb
            best_h = h_lb.copy()
    
    # Phase 4: Upsample to n=1000 and optimize
    p4 = t0 + 0.65 * (budget_s - 5)
    nh2 = 1000
    if time.time() < p4 - 60:
        h1000 = upsample(best_h, n, nh2)
        h_opt2, c_opt2 = continuous_optimize(h1000, nh2, p4 - 30, max_iter=10000)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2.copy()
            n = nh2
    
    # Phase 5: Binary refinement at current resolution
    p5 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p5 - 10:
        h_bin2, c_bin2 = binary_local_search(best_h, n, p5, max_sweeps=1500, seed=seed+10)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
    
    # Phase 6: Final continuous polish with restarts
    p6 = dl - 20
    if time.time() < p6 - 30:
        h_final, c_final = multi_restart_continuous(best_h, n, p6, n_restarts=2, seed=seed+20)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
    
    # Phase 7: Final binary polish
    if time.time() < dl - 5:
        h_fin, c_fin = binary_local_search(best_h, n, dl, max_sweeps=500, seed=seed+30)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
