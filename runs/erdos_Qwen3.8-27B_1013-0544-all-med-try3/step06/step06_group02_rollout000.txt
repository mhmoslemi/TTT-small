import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-boundary elements
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu=0.0):
    g = grad_smooth_max(x, n, alpha)
    if mu > 0:
        g = g + 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 1.0), (500, 5.0), (1000, 10.0),
        (5000, 50.0), (20000, 100.0), (100000, 500.0),
        (500000, 2000.0), (2000000, 5000.0), (10000000, 10000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fista_method(h, n, deadline, alpha=1000, max_iters=5000):
    """FISTA with projected gradient and backtracking."""
    h = project_feasible(h, n)
    y = h.copy()
    h_prev = h.copy()
    t = 1.0
    L = 10.0  # Initial Lipschitz estimate
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        g = grad_smooth_max(y, n, alpha)
        # Project gradient to satisfy sum constraint
        g = g - np.mean(g)
        
        # Backtracking line search
        step = 1.0 / L
        y_new = y - step * g
        h_new = project_feasible(y_new, n)
        
        # Check sufficient decrease (simplified)
        f_new = compute_c5(h_new, n)
        if f_new < best_c - 1e-15:
            best_c = f_new
            best_h = h_new.copy()
        
        # Momentum update
        h_curr = h_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_curr + momentum * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        h_prev = h_curr
        h = h_curr
        t = t_new
        
        # Adaptive L
        if f_new > best_c:
            L *= 1.5
        elif f_new < best_c - 1e-6:
            L = max(1.0, L * 0.9)
    
    return best_h, best_c

def gradient_descent_hc(h, n, deadline, lr_init=0.05, alpha_init=500.0):
    """Heavy-ball momentum gradient descent."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    alpha = alpha_init
    beta = 0.9  # Momentum
    
    for it in range(20000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g = g - np.mean(g)  # Project to null space of constraint
        
        v = beta * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            v = beta * v  # Keep some momentum
            h_new = project_feasible(h - 0.1 * lr * g, n)
            c_new2 = compute_c5(h_new, n)
            if c_new2 < best_c - 1e-15:
                best_c = c_new2
                best_h = h_new.copy()
                h = h_new
                v = 0.5 * np.zeros(n)
            else:
                lr *= 0.8
                alpha *= 1.2
                if lr < 1e-14:
                    break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, T_init=0.01, T_min=1e-8, cooling=0.9995, n_sweeps=500):
    """Simulated annealing local search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T = T_init
    rng = np.random.RandomState(42)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        if T < T_min:
            break
        
        improved_this_sweep = 0
        for _ in range(n):  # n random moves per sweep
            if time.time() > deadline:
                return best_h, best_c
            
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            # Try moving a small amount of mass from j to i (or vice versa)
            delta = 0.01
            if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                h[i] += delta
                h[j] -= delta
                c_new = compute_c5(h, n)
                delta_c = c_new - best_c
                if delta_c < 0 or (T > T_min and rng.random() < np.exp(-delta_c / T)):
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                        improved_this_sweep += 1
                else:
                    h[i] -= delta
                    h[j] += delta
            
            # Also try the reverse
            delta = 0.01
            if h[j] + delta <= 1.0 and h[i] - delta >= 0.0:
                h[j] += delta
                h[i] -= delta
                c_new = compute_c5(h, n)
                delta_c = c_new - best_c
                if delta_c < 0 or (T > T_min and rng.random() < np.exp(-delta_c / T)):
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                        improved_this_sweep += 1
                else:
                    h[j] -= delta
                    h[i] += delta
        
        T *= cooling
    
    return best_h, best_c
