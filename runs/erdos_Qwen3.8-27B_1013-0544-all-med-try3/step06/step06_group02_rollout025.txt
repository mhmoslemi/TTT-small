def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()

        top_k = np.argsort(c)[-k_active:]
        w = np.zeros_like(c)
        c_top = c[top_k]
        sh = alpha * (c_top - c_top.max())
        w_top = np.exp(sh)
        w_top /= w_top.sum()
        w[top_k] = w_top
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for m_idx in top_k:
            if w[m_idx] < 1e-14:
                continue
            k = m_idx - (n - 1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                if nv <= 0:
                    continue
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv <= 0:
                    continue
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        step = lr
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - step * grad / g_norm, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
                improved = True
                break
            step *= 0.5
            if step < 1e-14:
                break
        if not improved:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n
    
    # Phase 1: Quick L-BFGS-B warm start (60s)
    t1 = min(t0 + 60, deadline - 600)
    h, best_c = lbfgs_optimize(h, n, t1, alphas=[200, 1000, 5000, 20000], max_iters=150)
    best_h = h.copy()
    
    # Phase 2: SLSQP refinement (80s)
    if time.time() < deadline - 500:
        t2 = min(time.time() + 80, deadline - 500)
        for a in [50000, 200000, 1000000]:
            if time.time() > t2 - 5:
                break
            h2, c2 = slsqp_optimize(best_h, best_n, min(time.time() + 30, t2), alpha=a, maxiter=800)
            if c2 < best_c:
                best_c = c2
                best_h = h2
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # Phase 3: Active set optimization (120s)
    if time.time() < deadline - 380:
        t3 = min(time.time() + 120, deadline - 380)
        for ka in [5, 10, 20, 50]:
            if time.time() > t3 - 10:
                break
            alloc = min(30, t3 - time.time())
            if alloc < 5:
                break
            h3, c3 = active_set_optimize(best_h, best_n, time.time() + alloc,
                                         k_active=ka, alpha=5000)
            if c3 < best_c:
                best_c = c3
                best_h = h3
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # Phase 4: Multi-start with random perturbations (150s)
    if time.time() < deadline - 230:
        t4 = min(time.time() + 150, deadline - 230)
        best_h, best_c = multi_start_search(best_h, best_n, t4, n_starts=10, sigma=0.02)
    
    # Phase 5: More L-BFGS + SLSQP on the best (120s)
    if time.time() < deadline - 110:
        t5 = min(time.time() + 120, deadline - 110)
        best_h, best_c = lbfgs_optimize(best_h, best_n, min(time.time() + 60, t5 - 30),
                                         alphas=[50000, 200000, 1000000, 5000000], max_iters=200)
        if time.time() < t5 - 5:
            h5, c5 = slsqp_optimize(best_h, best_n, min(time.time() + 40, t5),
                                     alpha=2000000, maxiter=1000)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: Active set + gradient descent (100s)
    if time.time() < deadline - 60:
        t6 = min(time.time() + 100, deadline - 60)
        h6, c6 = active_set_optimize(best_h, best_n, min(time.time() + 40, t6 - 30),
                                      k_active=10, alpha=20000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        if time.time() < t6 - 5:
            h6b, c6b = gradient_descent_optimize(best_h, best_n, t6,
                                                  lr_init=0.002, alpha_init=1000)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b
    
    # Phase 7: Pair search (100s)
    if time.time() < deadline - 20:
        t7 = min(time.time() + 100, deadline - 10)
        best_h, best_c = pair_search(best_h, best_n, t7,
                                      deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                      radius=80)
    
    # Phase 8: Try different resolutions (remaining time)
    if time.time() < deadline - 15:
        for n_try in [296, 302, 298, 304, 294, 306]:
            if time.time() > deadline - 12:
                break
            h_interp = upsample(best_h, best_n, n_try)
            alloc = min(20, deadline - time.time() - 5)
            if alloc < 8:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.7,
                                       alphas=[10000, 100000], max_iters=100)
            if time.time() < deadline - 8:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 10, deadline - 5),
                                        deltas=[0.001, 0.0005, 0.0001], radius=50)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
    
    # Final: small pair search polish
    if time.time() < deadline:
        best_h, best_c = pair_search(best_h, best_n, deadline,
                                      deltas=[0.0005, 0.0002, 0.0001, 0.00005],
                                      radius=40)
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
