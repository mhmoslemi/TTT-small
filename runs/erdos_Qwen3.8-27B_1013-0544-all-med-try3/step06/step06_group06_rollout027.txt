import numpy as np
from scipy.optimize import minimize
import time

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_softmax_gradient(h, n, alpha):
    g = compute_all_g(h, n)
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / exp_g.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_shifts = len(g)
    # Only process shifts with significant weight
    active = np.where(p > 1e-12)[0]
    for kk in active:
        w = p[kk]
        m = kk - (n - 1)
        if m >= 0:
            if m < n:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
        else:
            ps = -m
            if ps < n:
                grad[:n - ps] += w * (1.0 - h[ps:])
                grad[ps:] -= w * h[:n - ps]
    grad *= dx
    return grad, g, g_max

def soft_max_descent(h, n, deadline, alphas, max_iter_per_alpha):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha_idx, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        h_curr = best_h.copy()
        lr = 2.0 / alpha
        stalled = 0
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            grad, g, g_max = compute_softmax_gradient(h_curr, n, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            improved = False
            for lr_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
                step = lr * lr_factor / gnorm
                h_try = project_feasible(h_curr - step * grad, n)
                c_try = compute_c5(h_try, n)
                if c_try < g_max - 1e-13:
                    h_curr = h_try
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h_try.copy()
                    improved = True
                    stalled = 0
                    break
            if not improved:
                lr *= 0.5
                stalled += 1
                if lr < 1e-12 or stalled > 30:
                    break
        
        h_curr = project_feasible(h_curr, n)
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
    
    return best_h, best_c

def subgradient_active(h, n, deadline, max_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        g_max = g.max()
        # Find active set: k values within 0.01 of max
        active = np.where(g >= g_max - 0.01)[0]
        
        # Average gradient over active set
        grad = np.zeros(n)
        for kk in active:
            m = kk - (n - 1)
            if m >= 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
            else:
                ps = -m
                grad[:n - ps] += (1.0 - h[ps:])
                grad[ps:] -= h[:n - ps]
        grad *= dx / len(active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        improved = False
        for lr_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * lr_factor / gnorm
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)

                h = h_try
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                improved = True
                lr = min(lr * 1.1, 0.5)

def binary_local_search(h, n, deadline, max_sweeps=80):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0].tolist()
        zeros = np.where(h_bin == 0)[0].tolist()
        
        for zi in zeros:
            if time.time() > deadline:
                break
            best_swap = None
            best_swap_c = best_c
            for oi in ones:
                h_bin[zi] = 1.0
                h_bin[oi] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_swap_c:
                    best_swap_c = c
                    best_swap = (zi, oi)
                h_bin[zi] = 0.0
                h_bin[oi] = 1.0
            
            if best_swap is not None:
                zi, oi = best_swap
                h_bin[zi] = 1.0
                h-bin[oi] = 0.0 if False else h_bin[oi] = 0.0
                best_c = best_swap_c
                best_h = h_bin.copy()
                improved = True
                ones.remove(oi)
                ones.append(zi)
                zeros = np.where(h_bin == 0)[0].tolist()
                break
        if not improved:
            break
    
    return best_h, best_c

def pair_swap_refine(h, n, deadline, delta):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n - 1):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * delta
                    hj = h[j] - s * delta
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - s * delta, hj + s * delta
    return best_h, best_c

def random_binary_starts(rng, n, num_starts, deadline):
    target = int(n // 2)
    best_h = None
    best_c = float('inf')
    for _ in range(num_starts):
        if time.time() > deadline:
            break
        positions = rng.choice(n, size=target, replace=False)
        h_rand = np.zeros(n)
        h_rand[positions] = 1.0
        c = compute_c5(h_rand, n)
        if c < best_c:
            best_c = c
            best_h = h_rand.copy()
    return best_h, best_c
