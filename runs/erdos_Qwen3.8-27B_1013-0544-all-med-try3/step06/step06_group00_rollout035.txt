def subgradient_step(h, n, step_size, top_m=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Find top m shifts
    top_idx = np.argsort(corr)[-top_m:]
    weights = np.zeros(2*n-1)
    top_vals = corr[top_idx]
    weights[top_idx] = top_vals - top_vals.min()  # or softmax
    if weights.sum() > 0:
        weights /= weights.sum()
    else:
        weights[top_idx[len(top_idx)//2]] = 1.0
    # Compute subgradient
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n-1)  # actual shift
        w = weights[k_idx]
        # ∂c[k]/∂h[j] = (1-h[j+k])·[j+k valid] - h[j-k]·[j-k valid]
        # For j from 0 to n-1:
        # (1-h[j+k]) valid when 0 <= j+k < n, i.e., -k <= j < n-k
        # -h[j-k] valid when 0 <= j-k < n, i.e., k <= j < n+k
        for j in range(n):
            g = 0.0
            if 0 <= j+k < n:
                g += 1.0 - h[j+k]
            if 0 <= j-k < n:
                g -= h[j-k]
            grad[j] += w * dx * g
    # Update
    h_new = h - step_size * grad
    return project_onto_simplex_box(h_new, n/2.0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Fix any remaining error
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - result
            s = room.sum()
            if s > 1e-14:
                result += diff / s * room
        else:
            s = result.sum()
            if s > 1e-14:

    return h_b

    """One step of projected subgradient descent on max_k corr_k * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top m shifts
    top_idx = np.argsort(corr)[-top_m:]
    top_vals = corr[top_idx]
    # Softmax weights
    shifted = top_vals - top_vals.max()
    weights = np.exp(50.0 * shifted)
    weights /= weights.sum()
    # Compute subgradient
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        k = int(idx) - (n - 1)
        if k >= 0:

                grad[:nk] += w * (-h[kp:])
    grad *= dx

def subgradient_descent(h, n, deadline, initial_step=0.01, decay=0.9999, top_m=10):
    """Run subgradient descent until deadline"""
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.time() < deadline:
        h_curr, c = subgradient_step(h_curr, n, step, top_m)
        # Momentum
        momentum = beta * momentum + (1 - beta) * (h_curr - h)
        h_curr = project_feasible(h + 0.5 * momentum, n)
        c = compute_c5(h_curr, n)
        if c < best_c:
            best_c = c
            best_h = h_curr.copy()
        step *= decay
        h = h_curr
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:

            if s > 1e-14:

def sg_step(h, n, step, top_m):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -top_m)[-top_m:]
    top_vals = corr[top_idx]
    mx = top_vals.max()
    w = np.exp(40.0 * (top_vals - mx))

    if wsum < 1e-30:
        w = np.zeros(len(top_idx))

            if nc <= cur_c:
                accept = True
            elif T > 1e-13:
                ratio = (cur_c - nc) / T
                if ratio < 500 and rng.random() < np.exp(-ratio):
                    accept = True
            if accept:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool ** batch_size
        if T < 1e-13:
            T = 0.01
    return best_h, best_c

def randomized_greedy(h_init, n, deadline, rng, n_pairs_per_iter=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    while time.time() < deadline:
        for _ in range(n_pairs_per_iter):
            if time.time() > deadline:
                break
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = zeros_arr[ai]
            j = ones_arr[bi]
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=50000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        t_limit = max(5.0, deadline - time.time())
        maxiter = int(t_limit * 50)
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Subgradient descent at base resolution
    p1_end = t0 + 0.18 * total
    h1, c1 = sg_descent(best_h, n, p1_end, step0=0.06, decay=0.99996, top_m=10, beta=0.85)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Binary greedy local search
    p2_end = t0 + 0.38 * total
    h2, c2 = greed_binary(best_h, n, p2_end, max_sweeps=500)
    if c2 < best_c:
        best_c, best_h = c2, h2

    # Phase 3: Annealing for exploration
    p3_end = t0 + 0.52 * total
    h3, c3 = anneal_binary(best_h, n, p3_end, seed=seed)
    if c3 < best_c:
        best_c, best_h = c3, h3

    # Phase 4: Subgradient from annealed solution
    p4_end = t0 + 0.65 * total
    h4, c4 = sg_descent(best_h, n, p4_end, step0=0.04, decay=0.99998, top_m=8, beta=0.9)
    if c4 < best_c:
        best_c, best_h = c4, h4

    # Phase 5: SLSQP polish
    p5_end = t0 + 0.72 * total
    if time.time() < p5_end - 10:
        h5 = slsqp_polish(best_h, n, p5_end, alpha=100000)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Multi-resolution search
    p6_end = deadline - 20
    for n_new in [600, 800]:
        remaining = p6_end - time.time()
        if remaining < 50:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        c_interp = compute_c5(h_interp, n_new)

        sub_dl = min(p6_end - 20, time.time() + remaining * 0.45)
        h_sub, c_sub = sg_descent(h_interp, n_new, sub_dl, step0=0.04, decay=0.99997, top_m=10, beta=0.85)
        if c_sub < c_interp:
            h_interp, c_interp = h_sub, c_sub

        rand_dl = min(p6_end - 10, time.time() + remaining * 0.35)
        h_rg, c_rg = randomized_greedy(h_interp, n_new, rand_dl, rng, n_pairs_per_iter=500)
        if c_rg < c_interp:
            h_interp, c_interp = h_rg, c_interp

        if c_interp < best_c:
            best_c, best_h, n = c_interp, h_interp, n_new

    # Phase 7: Final perturb-and-reoptimize cycles
    if time.time() < deadline - 10:
        for trial in range(5):
            if time.time() > deadline - 8:
                break
            perturbation = 0.003 * (trial + 1)
            h_pert = best_h + perturbation * rng.randn(n)
            h_pert = project_feasible(h_pert, n)
            short_dl = min(deadline - 5, time.time() + (deadline - time.time()) * 0.2)
            h_new, c_new = sg_descent(h_pert, n, short_dl, step0=0.01, decay=0.99999, top_m=5, beta=0.95)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
                if time.time() < deadline - 5:
                    h_bin, c_bin = greed_binary(best_h, n, deadline - 3, max_sweeps=50)
                    if c_bin < best_c:
                        best_c, best_h = c_bin, h_bin

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
