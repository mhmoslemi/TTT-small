    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_constraint(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def simulated_annealing(h_bin, n, start_temp, end_temp, n_iter, deadline, rng):
    """Simulated annealing on binary solutions"""
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    current_h = h_bin.copy()
    current_c = best_c
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Cooling
        progress = it / max(1, n_iter - 1)
        temp = start_temp * (end_temp / start_temp) ** progress
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Make swap
        current_h[i], current_h[j] = 0.0, 1.0
        new_c = compute_c5(current_h, n)
        
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            # Accept
            current_c = new_c
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            # Reject
            current_h[i], current_h[j] = 1.0, 0.0
    
    return best_h, best_c

For the continuous optimization, let me try a **projected subgradient method** with a twist:
- Compute g(k) for all k
- Find k* = argmax g(k)
- Compute subgradient of g(k*) with respect to h
- Take a step in the negative subgradient direction
- Project onto the feasible set

The subgradient of g(k*) = sum_i h[i]*(1-h[i+k*]) (times dx) with respect to h[i] is:
∂g(k*)/∂h[i] = (1 - h[i+k*]) - h[i] (for the terms where i and i+k* are both in range)

More precisely, for k* > 0:
g(k*) = dx * sum_{i=0}^{n-1-k*} h[i]*(1-h[i+k*])

∂g/∂h[i] = dx * (1 - h[i+k*]) for i = 0, ..., n-1-k*-1 (from the first factor)
           + dx * (-h[i-k*]) for i = k*, ..., n-1 (from the second factor, where i = j+k*)

So: ∂g/∂h[i] = dx * [(1 - h[i+k*]) * 1_{i < n-k*} - h[i-k*] * 1_{i >= k*}]

This is easy to compute!

But we want to minimize the MAX of all g(k), so we should use the subgradient of the max, which is a convex combination of the subgradients of the active constraints. At the optimum, multiple k values are active.

A simple approach: use the subgradient of the single worst k. This is a "supporting hyperplane" method.

A better approach: use a "smooth max" approximation and compute its gradient:
f(h) = (1/alpha) * log(sum_k exp(alpha * g(k)))
∇f = sum_k exp(alpha * g(k)) * ∇g(k) / sum_k exp(alpha * g(k))

This gives a smooth gradient that weights all k values, with more weight on the worst ones.

def smooth_max_gradient(g, alpha):
    """Compute gradient of smooth max of g(k) over k"""
    # g is a vector of values
    shifted = alpha * (g - g.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    return weights

def compute_grad_g(h, n, k):
    """Compute gradient of g(k) = dx * sum h[i]*(1-h[i+k]) with respect to h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad = dx * (1 - 2*h)  # since g(0) = dx * sum h(1-h), dg/dh = dx*(1-2h)
    elif k > 0:
        # g(k) = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # dg/dh[i] = dx * (1-h[i+k]) for i = 0..n-1-k
        # dg/dh[i] += dx * (-h[i-k]) for i = k..n-1
        grad[:n-k] += dx * (1 - h[k:])
        grad[k:] += dx * (-h[:n-k])
    else:
        # k < 0, let m = -k > 0
        m = -k
        # g(-m) = dx * sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # = dx * sum_{j=0}^{n-1-m} h[j+m]*(1-h[j])
        # dg/dh[j+m] = dx*(1-h[j]) for j=0..n-1-m, i.e., h[i] for i=m..n-1 gets dx*(1-h[i-m])
        # dg/dh[j] = dx*(-h[j+m]) for j=0..n-1-m, i.e., h[i] for i=0..n-1-m gets dx*(-h[i+m])
        grad[m:] += dx * (1 - h[:n-m])
        grad[:n-m] += dx * (-h[m:])
    return grad

def optimize_smooth_max(h0, n, alpha, deadline, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + np.log(np.exp(shifted - sm).sum()) / alpha
    
    def obj_grad(x):
        return smooth_max_grad_fast(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
                   'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h0

    while time.time() < p1_deadline - 3:
        r = rng.random()
        if r < 0.35:
            h_start = to_binary(best_h, n_init)
        elif r < 0.65:
            h_start = np.zeros(n_init)
            ones_pos = rng.choice(n_init, n_init // 2, replace=False)
            h_start[ones_pos] = 1.0
        elif r < 0.80:
            h_start = np.zeros(n_init)
            for i in range(n_init):
                if (i * phi) % 1.0 < 0.5:
                    h_start[i] = 1.0
            h_start = to_binary(h_start, n_init)
        else:
            # Perturbed best binary
            h_temp = best_h.copy()
            for _ in range(5):
                o = np.where(h_temp == 1)[0]
                z = np.where(h_temp == 0)[0]
                oi = o[rng.randint(len(o))]
                zi = z[rng.randint(len(z))]
                h_temp[oi], h_temp[zi] = 0.0, 1.0
            h_start = h_temp

        h_sa, c_sa = simulated_annealing(h_start, n_init, p1_deadline - 3, rng, n_iter=30000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        run_count += 1

    # Quick binary local search polish
    if time.time() < p1_deadline + 10:
        h_bl, c_bl = binary_local_search(best_h.copy(), n_init, min(p1_deadline + 10, t0 + 0.47 * effective_budget))
        if c_bl < best_c:
            best_c = c_bl
            best_h = h_bl.copy()

    # Phase 2: Continuous optimization at n_init (30% budget)
    p2_deadline = t0 + 0.78 * effective_budget

    h_cont = best_h.copy()
    h_cont, c_cont = optimize_continuous_slsqp(
        h_cont, n_init, p2_deadline,
        alpha_list=[200, 1000, 5000, 20000, 100000, 500000],
        maxiter=200)
    if c_cont < best_c:
        best_c = c_cont
        best_h = h_cont.copy()

    # Projected gradient polish
    if time.time() < p2_deadline - 20:
        h_pg, c_pg = projected_gradient_method(best_h.copy(), n_init, p2_deadline - 20,
                                                alpha=5000, lr_init=0.3, max_iter=10000)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg.copy()

    # Another SLSQP from perturbed start
    if time.time() < p2_deadline - 15:
        h_pert = project_feasible(best_h + 0.03 * rng.randn(n_init), n_init)
        h_pert, c_pert = optimize_continuous_slsqp(
            h_pert, n_init, p2_deadline,
            alpha_list=[1000, 5000, 50000], maxiter=150)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()

    # Phase 3: Higher resolution (20% budget)
    p3_deadline = deadline - 10

    for nh in [300, 400, 500]:
        if time.time() > p3_deadline - 50:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, nh)
        h_new = project_feasible(np.interp(x_new, x_old, best_h), nh)

        h_new, c_new = optimize_continuous_slsqp(
            h_new, nh, p3_deadline - 20,
            alpha_list=[500, 3000, 20000, 100000], maxiter=150)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            n_best = nh

    # Final polish
    if time.time() < deadline - 5:
        h_fin, c_fin = optimize_continuous_slsqp(
            best_h, n_best, deadline,
            alpha_list=[200000, 1000000, 5000000], maxiter=100)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
