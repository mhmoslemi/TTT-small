    j = np.arange(2*n-1)
    for i in range(n):

        mask1 = (j >= i) & (j <= i + n - 1)

            m1 = i - j[mask1] + n - 1
            G[i, mask1] += (1.0 - h[m1])
        
        # Second term: j in [n-1-i, n-1+i], m2 = i+j-n+1 in [0, n-1]

So d* is where min{G[i,j] : j active} ≤ 0 ≤ max{G[i,j] : j active}.

1. Start at d = 0

5. The boundary of the active set changes at d = (corr[j1] - corr[j2]) / (G[i,j2] - G[i,j1]) for pairs (j1, j2)

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_func(h, dx, beta, target_sum, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def constraint_fun(h, target_sum, n):
    return h.sum() - target_sum


def constraint_jac(h, target_sum, n):
    return np.ones(n)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        count = bin(i).count('1')
        h[i] = count % 2
    return h


def compute_grad_row(h, i, n):
    j = np.arange(2 * n - 1)
    grads = np.zeros(2 * n - 1)
    mask1 = (j >= i) & (j <= i + n - 1)
    if np.any(mask1):
        m1 = i - j[mask1] + n - 1
        grads[mask1] += (1.0 - h[m1])
    mask2 = (j >= n - 1 - i) & (j <= n - 1 + i)
    if np.any(mask2):
        m2 = i + j[mask2] - n + 1
        valid = m2 < n
        vals = np.where(valid, -h[m2], 0.0)
        grads[mask2] += vals
    return grads


def solve_1d_minmax(grads, corr, x0):
    b = corr - grads * x0
    xs = np.linspace(0.0, 1.0, 41)
    vals = np.max(grads[:, None] * xs[None, :] + b[:, None], axis=0)
    best_idx = np.argmin(vals)
    best_x = xs[best_idx]
    best_val = vals[best_idx]
    step = 0.025
    for _ in range(8):
        improved = True
        while improved:
            improved = False
            for dx_ in [-step, step]:
                x = best_x + dx_
                if 0.0 <= x <= 1.0:
                    val = np.max(grads * x + b)
                    if val < best_val - 1e-12:
                        best_x = x
                        best_val = val
                        improved = True
        step /= 2.0
    return best_x


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = None
    best_c5 = np.inf

    def remaining():
        return budget_s - (time.time() - t0)

    resolutions = [70, 80, 100, 120, 140]

    for n in resolutions:
        if remaining() < 60:
            break

        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        initializations = []
        h_resampled = resample_h(h_init, n)
        h_resampled = project_feasible(h_resampled, target_sum)
        initializations.append(h_resampled.copy())

        if n % 2 == 0:
            h_tm = thue_morse(n)
            if h_tm.sum() == n / 2:
                initializations.append(h tm.copy() if False else h_tm.copy())
            else:
                h_adj = h_tm.astype(float).copy()
                h_adj = project_feasible(h_adj, target_sum)
                initializations.append(h_adj)

        h_rand = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand = project_feasible(h_rand, target_sum)
        initializations.append(h_rand)

        for h_init_n in initializations:
            if remaining() < 40:
                break

            h = h_init_n.copy()
            c5 = compute_c5(h)

            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

            # SLSQP continuation
            for beta in [10, 50, 200, 800, 3000, 12000, 50000]:
                if remaining() < 25:
                    break
                try:
                    res = minimize(obj_func, h, args=(dx, beta, target_sum, n),
                                   method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_fun,
                                                 'jac': constraint_jac,
                                                 'args': (target_sum, n)}],
                                   options={'maxiter': 3000, 'ftol': 1e-15, 'maxls': 50})
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5:
                        h = h_new
                        c5 = c5_new
                        if c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = c5
                except Exception:
                    pass

            # Coordinate descent on worst lags
            if remaining() > 40:
                num_rounds = min(300, int(remaining() * 0.3))
                top_k = 3
                for rnd in range(num_rounds):
                    if remaining() < 15:
                        break
                    corr = get_corr(h)
                    top_lags = np.argsort(corr)[-top_k:]
                    for lag_idx in top_lags:
                        lag = lag_idx - (n - 1)
                        if lag >= 0:
                            indices = np.arange(0, n - lag)
                            indices2 = np.arange(lag, n)
                        else:
                            la = -lag
                            indices = np.arange(0, n - la)
                            indices2 = np.arange(la, n)
                        all_idx = np.unique(np.concatenate([indices, indices2]))
                        for i in all_idx:
                            grads = compute_grad_row(h, i, n)
                            corr_now = get_corr(h)
                            new_val = solve_1d_minmax(grads, corr_now, h[i])
                            h[i] = new_val
                        h = project_feasible(h, target_sum)
                    if rnd % 20 == 19:
                        c5_check = compute_c5(h)
                        if c5_check < c5:
                            c5 = c5_check
                            if c5 < best_c5:
                                best_h = h.copy()
                                best_c5 = c5
                c5_check = compute_c5(h)
                if c5_check < c5:
                    c5 = c5_check
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5

            # Subgradient with momentum
            if remaining() > 30:
                h_sub = h.copy()
                velocity = np.zeros(n)
                lr = 0.003
                momentum = 0.9
                for it in range(20000):
                    if remaining() < 10:
                        break
                    corr = get_corr(h_sub)
                    k_worst = np.argmax(corr)
                    grad = compute_grad_row(h_sub, 0, n)
                    full_grad = np.zeros(n)
                    for i in range(n):
                        full_grad[i] = compute_grad_row(h_sub, i, n)[k_worst]
                    velocity = momentum * velocity - lr * full_grad
                    h_sub = h_sub + velocity
                    h_sub = project_feasible(h_sub, target_sum)
                    if it % 500 == 499:
                        lr *= 0.7
                        c5_sub = compute_c5(h_sub)
                        if c5_sub < c5:
                            h = h_sub.copy()
                            c5 = c5_sub
                            if c5 < best_c5:
                                best_h = h.copy()
                                best_c5 = c5
                c5_sub = compute_c5(h_sub)
                if c5_sub < c5:
                    h = h_sub.copy()
                    c5 = c5_sub
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5

            # Final SLSQP polish from current best at this resolution
            if remaining() > 20:
                for beta in [20000, 80000]:
                    if remaining() < 12:
                        break
                    try:
                        res = minimize(obj_func, h, args=(dx, beta, target_sum, n),
                                       method='SLSQP', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': constraint_fun,
                                                     'jac': constraint_jac,
                                                     'args': (target_sum, n)}],
                                       options={'maxiter': 5000, 'ftol': 1e-16})
                        h_new = project_feasible(res.x, target_sum)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5:
                            h = h_new
                            c5 = c5_new
                            if c5 < best_c5:
                                best_h = h.copy()
                                best_c5 = c5
                    except Exception:
                        pass

    # Final global polish on best solution found
    if best_h is not None and remaining() > 15:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        ts_best = n_best / 2.0
        bnds_best = Bounds(np.zeros(n_best), np.ones(n_best))
        for beta in [50000, 200000]:
            if remaining() < 8:
                break
            try:
                res = minimize(obj_func, best_h, args=(dx_best, beta, ts_best, n_best),
                               method='SLSQP', bounds=bnds_best,
                               constraints=[{'type': 'eq', 'fun': constraint_fun,
                                             'jac': constraint_jac,
                                             'args': (ts_best, n_best)}],
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
