import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Correct gradient of the smooth max (log-sum-exp) approximation."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[k:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[k:n])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    return grad

def subgrad_topk(h, n, topk=20):
    """Subgradient using top-k active constraints with soft weighting."""
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    idx = np.argsort(c)[-topk:]
    w_vals = c[idx]
    w_max = w_vals.max()
    w_shift = 100.0 * (w_vals - w_max)
    w = np.exp(w_shift - w_shift.max())
    w /= w.sum()
    grad = np.zeros(n)
    for ii, m in enumerate(idx):
        if w[ii] < 1e-10:
            continue
        k = m - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[k:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[k:n])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas=None, mus=None):
    """L-BFGS-B with increasing alpha for smooth max approximation."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0, 2000000.0]
    if mus is None:

    for i in range(len(alphas)):
        if time.time() > deadline - 2:
            break
        alpha = alphas[i]

        try:

                                   'gtol': 1e-14, 'maxfun': 5000,
                                   'maxls': 50, 'time_limit': max(t_sec, 5)})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new

        except Exception:

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def subgrad_descent(h, n, deadline, lr_init=0.005):
    """Projected subgradient descent with backtracking line search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = subgrad_topk(h, n, topk=20)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
        if lr < 1e-15:
            break
    
    return best_h, best_c

def smart_pair_search(h, n, deadline, step_sizes=None, topk_select=25):
    """Gradient-guided pair search: move from high-gradient to low-gradient positions."""
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    best_c = compute_c5(h, n)
    
    for d in step_sizes:
        while time.time() < deadline:
            improved = False
            g = subgrad_topk(h, n, topk=15)
            order = np.argsort(g)
            n_try = min(topk_select, n // 3)
            sources = order[-n_try:]
            targets = order[:n_try]
            
            for si in sources:
                if time.time() > deadline:
                    return h, best_c
                for ti in targets:
                    if si == ti:
                        continue
                    if h[si] - d >= 0.0 and h[ti] + d <= 1.0:
                        h[si] -= d
                        h[ti] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[si] += d
                            h[ti] -= d
                    if h[ti] - d >= 0.0 and h[si] + d <= 1.0:
                        h[ti] -= d
                        h[si] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[ti] += d
                            h[si] -= d
            if not improved:
                break
    return h, best_c

def block_move_search(h, n, deadline, step=0.01, block_sizes=None):
    """Try moving contiguous blocks of mass."""
    if block_sizes is None:
        block_sizes = [3, 5, 8, 12, 20]
    best_c = compute_c5(h, n)
    
    for bs in block_sizes:
        while time.time() < deadline:
            improved = False
            for start in range(0, n - bs + 1, max(1, bs // 2)):
                if time.time() > deadline:
                    return h, best_c
                # Try shifting block [start, start+bs) by various amounts
                for shift in range(1, min(20, n - start - bs + 1)):
                    src = h[start:start+bs]
                    dst_start = start + shift
                    if dst_start + bs > n:
                        continue
                    # Check feasibility
                    new_vals = src + step
                    if new_vals.max() > 1.0 or (h[dst_start:dst_start+bs] - step).min() < 0.0:

                    h[dst_start:dst_start+bs] -= step
                    c = compute_c5(h, n)

                        best_c = c
                        improved = True
                    else:
                        h[start:start+bs] = src
                        h[dst_start:dst_start+bs] += step
            if not improved:
                break
    return h, best_c

        t5 = min(time.time() + 80, deadline - 150)
        best_h, best_c = smart_pair_search(best_h, n, t5,
                                            step_sizes=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    
    # Phase 6: Random restarts with structured perturbations
    if time.time() < deadline - 120:
        t6 = deadline - 120
        n_restarts = 6
        per_r = (t6 - time.time()) / n_restarts
        for r in range(n_restarts):
            if time.time() > t6 - 10:
                break
            if r % 2 == 0:
                h_p = perturb_structured(best_h, n, rng)
            else:
                h_p = perturb_h(best_h, n, rng, magnitude=0.015)
            t_r = time.time() + per_r * 0.85
            h_r, c_r = lbfgs_optimize(h_p, n, t_r,
                                       alphas=[500, 5000, 50000, 500000],
                                       mus=[0.1, 0.05, 0.01, 0.001])
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    
    # Phase 7: Try higher resolution
    if time.time() < deadline - 100:
        for n_try in [400, 500]:
            if time.time() > deadline - 80:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 50)
            if alloc < 15:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.7,
                                       alphas=[500, 5000, 50000, 200000],
                                       mus=[0.1, 0.05, 0.01, 0.001])
            if time.time() < deadline - 30:
                h_t, c_t = smart_pair_search(h_t, n_try, min(time.time() + 15, deadline - 20),
                                              step_sizes=[0.01, 0.005, 0.002, 0.001])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try
    
    # Phase 8: Final L-BFGS + SLSQP
    if time.time() < deadline - 20:
        t8 = min(time.time() + 40, deadline - 10)
        best_h, best_c = lbfgs_optimize(best_h, n, t8,
                                         alphas=[50000, 200000, 1000000],
                                         mus=[0.005, 0.001, 0.0005])
    if time.time() < deadline - 10:
        t9 = min(time.time() + 20, deadline - 5)
        best_h, best_c = slsqp_optimize(best_h, n, t9, alpha=2000000, maxiter=2000)
    
    # Phase 9: Final local refinement
    if time.time() < deadline - 3:
        best_h, best_c = local_pair_search(best_h, n, deadline - 2,
                                            [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                            radius=30)
    if time.time() < deadline - 1:
        best_h, best_c = smart_pair_search(best_h, n, deadline - 0.5,
                                            step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                            topk_select=20)
    
    final_h = project_feasible(best_h, n)
    final_c = compute_c5(final_h, n)
    return final_h, final_c, n
