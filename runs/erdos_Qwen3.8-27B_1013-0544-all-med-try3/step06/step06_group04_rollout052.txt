import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    result = np.clip(h - lam, 0, 1)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = (1.0 - result)
            mask = room > 1e-10
            if mask.sum() > 0:
                result[mask] += diff / mask.sum()
        else:
            mask = result > 1e-10
            if mask.sum() > 0:
                result[mask] += diff / mask.sum()
    return np.clip(result, 0, 1)

def get_active_idx(c, n, gap=0.005, min_count=3, max_count=50):
    threshold = c.max() - gap
    idx = np.where(c >= threshold)[0]
    if len(idx) < min_count:
        idx = np.argsort(c)[-min_count:]
    if len(idx) > max_count:
        idx = np.argsort(c)[-max_count:]
    return idx

def compute_lag_gradient(h, n, lag):
    """Gradient of c_lag with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] += (-h[:n - m])
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += (-h[m:])
    return grad

def smooth_grad(h, n, alpha, active_idx):
    """Gradient of smooth max over active lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_active = c[active_idx]
    m = c_active.max()
    shifted = alpha * (c_active - m)
    shifted_max = shifted.max()
    exps = np.exp(shifted - shifted_max)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for i, k_idx in enumerate(active_idx):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = k_idx - (n - 1)
        grad += w * compute_lag_gradient(h, n, lag) * dx
    return grad

def proj_grad_step(h, n, alpha, active_idx, eta, max_bt=20):
    """One projected gradient step with backtracking."""
    grad = smooth_grad(h, n, alpha, active_idx)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return h.copy(), 0.0
    
    step = eta
    for _ in range(max_bt):
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        # Check if this is a descent direction
        # Use the smooth objective as a proxy
        c = compute_all_lags(h_new, n)
        c_active_new = c[active_idx]
        c_old = compute_all_lags(h, n)[active_idx]
        
        # Simple check: is the max of active lags decreased?
        if c_active_new.max() < c_old.max() - 1e-14:
            return h_new, step
        step *= 0.5
    
    return project_feasible(h - step * grad, n), step

def _smooth_obj(x, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_active = c[active_idx]
    m = c_active.max()
    shifted = alpha * (c_active - m)
    shifted_max = shifted.max()
    if shifted_max > 500:
        return float(m + (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    c = compute_all_lags(h, n)
    active_idx = get_active_idx(c, n, gap=0.003, min_count=3, max_count=20)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, active_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2

        if time.time() > deadline - 3:

        h_curr = h_new

def local_search(h, n, deadline, max_rounds=2000):

    return h, best_c

    m = c_active.max()
    shifted = alpha * (c_active - m)
    shifted_max = shifted.max()
    exps = np.exp(shifted - shifted_max)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        w = weights[i]
        if w < 1e-14:
            continue
        k_idx = active_idx[i]
        lag = k_idx - (n - 1)
        grad += w * compute_lag_gradient(h, n, lag) * dx
    return grad

def smooth_obj_value(h, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_active = c[active_idx]
    m = c_active.max()
    shifted = alpha * (c_active - m)
    shifted_max = shifted.max()
    return float(m + (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))) / alpha)

def proj_grad_optimize(h, n, alpha_schedule, deadline, max_iters_per_alpha=5000):
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    h_prev = h_curr.copy()
    momentum = 0.3
    eta = 0.2

    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        active_idx = get_active_idx(c, n, gap=0.004, min_count=3, max_count=30)
        stagnant = 0
        for it in range(max_iters_per_alpha):
            if time.time() > deadline:
                break
            grad = smooth_grad(h_curr, n, alpha, active_idx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            velocity = momentum * (h_curr - h_prev)
            h_proposed = h_curr - eta * grad + velocity
            h_proposed = project_feasible(h_proposed, n)
            obj_trial = smooth_obj_value(h_proposed, n, alpha, active_idx)
            obj_curr = smooth_obj_value(h_curr, n, alpha, active_idx)
            if obj_trial < obj_curr - 1e-14:
                h_prev = h_curr.copy()
                h_curr = h_proposed
                c_now = compute_c5(h_curr, n)
                if c_now < best_c - 1e-14:
                    best_c = c_now
                    best_h = h_curr.copy()
                    stagnant = 0
                    eta = min(eta * 1.05, 1.0)
                else:
                    stagnant += 1
                    eta = min(eta * 1.02, 1.0)
            else:
                eta = max(eta * 0.5, 1e-10)
                h_prev = h_curr.copy()
                h_curr = project_feasible(h_curr - eta * grad, n)
                stagnant += 1
            if stagnant > 400:
                c = compute_all_lags(h_curr, n)
                active_idx = get_active_idx(c, n, gap=0.004, min_count=3, max_count=30)
                stagnant = 0
                eta = 0.2
                momentum = 0.5
            if stagnant > 1500:
                break
        momentum = 0.5
        eta = 0.15
    return best_h, best_c

def _smooth_obj(x, n, alpha, active_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_active = c[active_idx]
    m = c_active.max()
    shifted = alpha * (c_active - m)
    shifted_max = shifted.max()
    if shifted_max > 500:
        return float(m + (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    c = compute_all_lags(h, n)
    active_idx = get_active_idx(c, n, gap=0.003, min_count=3, max_count=15)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, active_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=40):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.15
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-8:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.85)
    return best_h, best_c

def local_search(h, n, deadline, max_rounds=3000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    rng_local = np.random.RandomState(99999)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            i = rng_local.randint(0, n - m)
            j = i + m
            if h[i] <= h[j]:
                continue
            delta = min(d, (h[i] - h[j]) * 0.4, h[i], 1.0 - h[j])
            if delta < 1e-8:
                continue
            h[i] -= delta
            h[j] += delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[i] += delta
                h[j] -= delta
    return h, best_c

def fourier_start(n, K, rng, scale=0.3):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = rng.uniform(-scale, scale)
        b = rng.uniform(-scale, scale)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n)

def block_start(n, rng):
    h = np.full(n, 0.5)
    n_blocks = rng.randint(4, 12)
    block_size = n // n_blocks
    for b in range(n_blocks):
        val = rng.choice([0.0, 1.0, 0.2, 0.8]) * rng.uniform(0.5, 1.0)
        start = b * block_size
        end = min(start + block_size, n)
        h[start:end] = val
    return project_feasible(h, n)

def multi_phase_optimize(h_start, n, deadline, rng):
    best_h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Outer approximation (fast, good for coarse structures)
    if time.time() < deadline - 40:
        oo_dl = min(deadline - 40, time.time() + 50)
        h_oo, c_oo = outer_approx(best_h, n, oo_dl, max_iters=30)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo

    # Phase 2: Projected gradient with alpha schedule
    if time.time() < deadline - 30:
        alpha_sched = [50, 200, 1000, 5000, 20000, 100000]
        pg_dl = min(deadline - 30, time.time() + (deadline - time.time() - 30) * 0.6)
        h_pg, c_pg = proj_grad_optimize(best_h, n, alpha_sched, pg_dl)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase 3: SLSQP polish
    if n <= 700 and time.time() < deadline - 10:
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 5:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new

    # Phase 4: Local search
    if time.time() < deadline - 2:
        h_new, c_new = local_search(best_h, n, deadline - 1, max_rounds=2000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
    except Exception:
        h_init = None
        n_init = 300

    # Phase A: Optimize at initial resolution
    if h_init is not None:
        alloc = min(200, (deadline - t0) * 0.2)
        h_opt, c_opt = multi_phase_optimize(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase B: Try multiple starting points at n=300
    if time.time() < deadline - 200:
        starts = []
        starts.append(fourier_start(n_init, 5, rng, scale=0.2))
        starts.append(fourier_start(n_init, 10, rng, scale=0.15))
        starts.append(block_start(n_init, rng))
        if h_init is not None:
            starts.append(h_init + rng.normal(0, 0.03, n_init))
        for s in starts:
            if time.time() > deadline - 150:
                break
            s = project_feasible(s, n_init)
            alloc = min(80, (deadline - time.time()) * 0.15)
            if alloc < 25:
                break
            h_opt, c_opt = multi_phase_optimize(s, n_init, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase C: Upsample to n=600 and optimize
    n_up = 600
    if time.time() < deadline - 150:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        alloc = min(250, (deadline - time.time()) * 0.35)
        if alloc > 60:
            h_opt, c_opt = multi_phase_optimize(h_up, n_up, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase D: Try n=900 if time permits
    n_up2 = 900
    if time.time() < deadline - 200:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_up2)
        h_up2 = np.interp(x_new, x_old, best_h)
        h_up2 = project_feasible(h_up2, n_up2)
        alloc = min(200, (deadline - time.time()) * 0.3)
        if alloc > 80:
            h_opt, c_opt = multi_phase_optimize(h_up2, n_up2, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase E: Multi-start at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scales = [0.01, 0.02, 0.04, 0.08]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(65, (deadline - time.time() - 40) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = multi_phase_optimize(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase F: Final high-precision polish
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 500 and time.time() < deadline - 20:
            h_new = slsqp_optimize(best_h, n, 500000, maxiter=4000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        if time.time() < deadline - 3:
            h_new, c_new = local_search(best_h, n, deadline - 1, max_rounds=3000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
