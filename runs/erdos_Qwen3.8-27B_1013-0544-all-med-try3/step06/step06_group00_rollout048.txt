        delta = make_delta(i, j, h, n, m_range)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_grad_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    f_val = c_max + np.log(exp_vals.sum()) / alpha
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        l = k - (n - 1)
        pk = p[k]
        if l >= 0:
            idx = np.arange(0, n - l)
            grad[idx] += pk * (1.0 - h[idx + l])
            idx2 = np.arange(l, n)
            grad[idx2] -= pk * h[idx2 - l]
        else:
            idx = np.arange(-l, n)
            grad[idx] += pk * (1.0 - h[idx + l])
            idx2 = np.arange(0, n + l)
            grad[idx2] -= pk * h[idx2 - l]
    grad *= dx
    return f_val, grad

def projected_gradient_descent(h_init, n, alpha, deadline, max_iters=500000):
    h = h_init.copy()
    f0, g0 = compute_grad_smooth(h, n, alpha)
    c0 = compute_c5(h, n)
    best_h = h.copy()
    best_c = c0
    lr = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        f, g = compute_grad_smooth(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g_normed = g / g_norm
        step_lr = lr
        accepted = False
        for _ in range(40):
            h_new = h - step_lr * g
            h_new = project_feasible(h_new, n)
            f_new, _ = compute_grad_smooth(h_new, n, alpha)
            c_new = compute_c5(h_new, n)
            if f_new < f * (1 - 1e-10) or c_new < best_c:
                accepted = True
                break
            step_lr *= 0.5
        if accepted:
            h = h_new
            lr = min(step_lr * 1.3, 1.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP at original resolution
    p1_end = t0 + 0.12 * total
    h = best_h.copy()
    for alpha in [500, 5000, 50000, 500000]:
        if time.time() > p1_end - 2:
            break
        h = slsqp_smooth(h, n0, alpha, 1000, 1e-14)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Exhaustive binary at original resolution
    p2_end = t0 + 0.22 * total
    hb, cb = greed_binary_exhaustive(best_h, n0, p2_end, max_sweeps=200)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: Upsample to higher resolution
    n_high = 800
    x_old = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
    x_new = np.linspace(0, 2, n_high, endpoint=False) + 1.0 / n_high
    h_high = np.interp(x_new, x_old, best_h, period=None)
    h_high = project_feasible(h_high, n_high)
    c_high = compute_c5(h_high, n_high)
    if c_high < best_c:
        best_c = c_high
        best_h = h_high.copy()
        best_n = n_high

    # Phase 4: Projected gradient at high resolution (multiple alpha levels)
    p4_end = t0 + 0.55 * total
    h = h_high.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000]
    time_per_alpha = (p4_end - time.time()) / len(alphas)
    for i, alpha in enumerate(alphas):
        alpha_deadline = time.time() + time_per_alpha
        if alpha_deadline > p4_end:
            alpha_deadline = p4_end - 1
        h, c = projected_gradient_descent(h, n_high, alpha, alpha_deadline, max_iters=100000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n_high
        if time.time() > p4_end:
            break

    # Phase 5: Binary random search at high resolution
    p5_end = t0 + 0.75 * total
    hb, cb = binary_random_greedy(best_h, n_high, p5_end, seed=seed, max_iters=3000000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = n_high

    # Phase 6: Annealing at high resolution
    p6_end = t0 + 0.88 * total
    ha, ca = binary_anneal(best_h, n_high, p6_end, seed=seed + 1, max_iters=3000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
        best_n = n_high

    # Phase 7: Gradient polish from best binary
    p7_end = deadline - 3
    h = best_h.copy()
    if best_n == n_high:
        for alpha in [1000, 10000, 100000]:
            if time.time() > p7_end - 2:
                break
            h, c = projected_gradient_descent(h, n_high, alpha, min(p7_end - 1, time.time() + total * 0.03), max_iters=50000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n_high
    else:
        # Resample to high res and polish
        x_old2 = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        x_new2 = np.linspace(0, 2, n_high, endpoint=False) + 1.0 / n_high
        h = np.interp(x_new2, x_old2, best_h)
        h = project_feasible(h, n_high)
        for alpha in [1000, 10000, 100000]:
            if time.time() > p7_end - 2:
                break
            h, c = projected_gradient_descent(h, n_high, alpha, min(p7_end - 1, time.time() + total * 0.03), max_iters=50000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n_high

    # Phase 8: Try intermediate resolution n=600
    if time.time() < deadline - 30:
        n_mid = 600
        x_m = np.linspace(0, 2, n_mid, endpoint=False) + 1.0 / n_mid
        x_best = np.linspace(0, 2, best_n, endpoint=False) + 1.0 / best_n
        h_mid = np.interp(x_m, x_best, best_h)
        h_mid = project_feasible(h_mid, n_mid)
        p8_end = deadline - 5
        # Quick binary at mid
        hb_m, cb_m = binary_random_greedy(h_mid, n_mid, min(p8_end - 30, t0 + total * 0.82), seed=seed+2, max_iters=1000000)
        if cb_m < best_c:
            best_c = cb_m
            best_h = hb_m.copy()
            best_n = n_mid
        if time.time() < p8_end - 5:
            h_m = best_h.copy()
            for alpha in [5000, 50000]:
                if time.time() > p8_end - 3:
                    break
                h_m, c_m = projected_gradient_descent(h_m, n_mid, alpha, p8_end - 3, max_iters=30000)
                if c_m < best_c:
                    best_c = c_m
                    best_h = h_m.copy()
                    best_n = n_mid

    # Final: ensure feasibility and compute exact C5
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
