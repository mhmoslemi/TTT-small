def build_E(h, n):
    m_vals = np.arange(-(n-1), n, dtype=int)  # length 2n-1
    E = np.zeros((n, 2*n-1))
    for p in range(n):
        # First term: (1-h[p+m]) where -p <= m <= n-1-p
        mask1 = (m_vals >= -p) & (m_vals <= n-1-p)
        if mask1.any():

        if mask2.any():

            E[p, mask2] -= h[idx2]
    
    return E

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_E(h, n):
    m_vals = np.arange(-(n - 1), n, dtype=int)
    E = np.zeros((n, 2 * n - 1))
    for p in range(n):
        mask1 = (m_vals >= -p) & (m_vals <= n - 1 - p)
        if mask1.any():
            idx = p + m_vals[mask1]
            E[p, mask1] = 1.0 - h[idx]
        mask2 = (m_vals >= p - n + 1) & (m_vals <= p)
        if mask2.any():
            idx2 = p - m_vals[mask2]
            E[p, mask2] -= h[idx2]
    return E

        mx = shifted.max()

            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n

        if alloc < 30:
            break
        h_opt, c_opt = optimize_resolution(h_s, n_init, time.time() + alloc, rng, do_pair=True)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Higher resolution (n=500)
    n_high = 500
    remaining_for_high = deadline - time.time() - 30
    if remaining_for_high > 60:
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)

        alloc_high = min(120, remaining_for_high * 0.5)
        h_opt, c_opt = optimize_resolution(h_high, n_high, time.time() + alloc_high, rng)

        # Also try Thue-Morse at high res
        if time.time() < deadline - 60:
            h_tm_high = thue_morse_init(n_high)
            alloc_tm = min(60, deadline - time.time() - 40)
            if alloc_tm > 30:
                h_opt2, c_opt2 = optimize_resolution(h_tm_high, n_high,
                                                      time.time() + alloc_tm, rng)
                if c_opt2 < c_opt:
                    h_opt, c_opt = h_opt2, c_opt2

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Multi-restart with perturbations on best
    n_best = len(best_h)
    if time.time() < deadline - 30:
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scales = [0.01, 0.03, 0.05, 0.08, 0.12]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc_r = min(40, (deadline - time.time() - 15) * 0.4)
            if alloc_r < 15:
                break
            h_opt, c_opt = optimize_resolution(hp, n_best, time.time() + alloc_r, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive pair search
    n_f = len(best_h)
    if time.time() < deadline - 5:
        if n_f <= 300:
            best_h, best_c = fast_pair_search(best_h, n_f, deadline - 2,
                                              [0.01, 0.005, 0.002, 0.001,
                                               0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = fast_pair_search(best_h, n_f, deadline - 2,
                                              [0.005, 0.002, 0.001, 0.0005,
                                               0.0002, 0.0001])

    # Phase 6: Final SLSQP polish
    if best_h is not None and time.time() < deadline - 3:
        n_p = len(best_h)
        ht = slsqp_smooth(best_h, n_p, 200000, maxiter=500)
        c = compute_c5(ht, n_p)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            if n_p <= 300:
                best_h, best_c = fast_pair_search(best_h, n_p, deadline - 0.5,
                                                  [0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = fast_pair_search(best_h, n_p, deadline - 0.5,
                                                  [0.0002, 0.0001])

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
