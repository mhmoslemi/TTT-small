        else:
            tot = h.sum()
            if tot < 1e-15:

        idx1 = i + r
        idx2 = i - r
        v[i, :] = (np.where((idx1 >= 0) & (idx1 < n), h[idx1], 0.0) +
                   np.where((idx2 >= 0) & (idx2 < n), h[idx2], 0.0))
    return v

def _update_v_for_change(v, h, i, delta, n):
    r = np.arange(-(n - 1), n)
    r_val = r.astype(float)
    m_first = (n - 1) + i - np.arange(n)
    valid_f = (m_first >= 0) & (m_first < 2 * n - 1)
    v[np.arange(n)[valid_f], m_first[valid_f]] += delta
    m_second = (n - 1) + np.arange(n) - i
    valid_s = (m_second >= 0) & (m_second < 2 * n - 1)
    v[np.arange(n)[valid_s], m_second[valid_s]] += delta

def efficient_local_search(h, n, deadline, delta_schedule=None, start_radius=None):
    if delta_schedule is None:
        delta_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    if start_radius is None:
        start_radius = min(150, n // 2)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    c5 = compute_c5_vec(h, n)
    current_max = c5.max()
    dx = 2.0 / n
    v = _compute_v_matrix(h, n)
    
    radius = start_radius
    for d_idx, d in enumerate(delta_schedule):
        r = max(5, int(radius * (1.0 - 0.08 * d_idx)))
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - r)
                hi = min(n, i + r + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d > 1.0 or h[j] - d < -1e-15:
                        continue
                    v_diff = v[i] - v[j]
                    new_max = (c5 - d * dx * v_diff).max()
                    if new_max < current_max - 1e-13:
                        h[i] += d
                        h[j] -= d
                        c5 -= d * dx * v_diff
                        current_max = new_max
                        _update_v_for_change(v, h, i, d, n)
                        _update_v_for_change(v, h, j, -d, n)
                        improved = True
                        if current_max < best_c - 1e-15:
                            best_c = current_max
                        break
                if not improved:
                    continue
        if not improved:
            break
        radius = max(10, radius - 5)
    
    return h, best_c

def simulated_annealing_search(h, n, deadline, n_iter=50000, T_init=0.01, T_min=0.0001):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c5 = compute_c5_vec(h, n)
    current_max = c5.max()
    dx = 2.0 / n
    v = _compute_v_matrix(h, n)
    rng = np.random.RandomState(12345)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_min / T_init) ** (it / n_iter)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.005
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                d = -d
            else:
                continue
        v_diff = v[i] - v[j]
        new_max = (c5 - d * dx * v_diff).max()
        delta_obj = new_max - current_max
        if delta_obj < 0 or rng.rand() < np.exp(-delta_obj / max(T, 1e-10)):
            h[i] += d
            h[j] -= d
            c5 -= d * dx * v_diff
            current_max = new_max
            _update_v_for_change(v, h, i, d, n)
            _update_v_for_change(v, h, j, -d, n)
            if current_max < best_c - 1e-15:
                best_c = current_max
                best_h = h.copy()
    
    return best_h, best_c

def multi_start_optimize(h_init, n, deadline, n_starts=3):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    rng = np.random.RandomState(999)
    
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        if s == 0:
            h_s = h_init.copy()
        elif s == 1:
            h_s = h_init + rng.randn(n) * 0.1
            h_s = project_feasible(h_s, n)
        else:
            h_s = project_feasible(rng.rand(n) * 1.2, n)
        
        alloc = min(120, (deadline - time.time()) / max(1, n_starts - s))
        t_sub = time.time() + alloc
        
        h_s, c_s = lbfgs_phase(h_s, n, min(time.time() + alloc * 0.4, t_sub), max_passes=5)
        if time.time() < t_sub - 10:
            h_s, c_s = efficient_local_search(h_s, n, t_sub, delta_schedule=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005], start_radius=80)
        
        if c_s < best_c - 1e-15:
            best_c = c_s
            best_h = h_s.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    n_best = n_init
    
    # Phase 1: L-BFGS-B with increasing smoothness
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_best, t1, max_passes=6)
    n_best = len(best_h)
    
    # Phase 2: Efficient local search (main improvement over parent)
    t2 = min(time.time() + 250, deadline - 200)
    best_h, best_c = efficient_local_search(best_h, n_best, t2,
        delta_schedule=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
        start_radius=120)
    n_best = len(best_h)
    
    # Phase 3: Another L-BFGS pass with high alpha
    t3 = min(time.time() + 60, deadline - 120)
    if time.time() < deadline - 120:
        h3, c3 = lbfgs_phase(best_h, n_best, t3, max_passes=4)
        if c3 < best_c - 1e-15:
            best_c = c3
            best_h = h3
    
    # Phase 4: More local search with fine deltas
    t4 = min(time.time() + 150, deadline - 80)
    best_h, best_c = efficient_local_search(best_h, n_best, t4,
        delta_schedule=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
        start_radius=60)
    n_best = len(best_h)
    
    # Phase 5: Try simulated annealing for escape
    if time.time() < deadline - 80:
        t5 = min(time.time() + 80, deadline - 60)
        h5, c5 = simulated_annealing_search(best_h, n_best, t5, n_iter=30000, T_init=0.005, T_min=0.0001)
        if c5 < best_c - 1e-15:
            best_c = c5
            best_h = h5
        # Follow up with local search after SA
        if time.time() < deadline - 40:
            t5b = min(time.time() + 40, deadline - 20)
            h5b, c5b = efficient_local_search(best_h, n_best, t5b,
                delta_schedule=[0.002, 0.001, 0.0005, 0.0002, 0.0001], start_radius=50)
            if c5b < best_c - 1e-15:
                best_c = c5b
                best_h = h5b
                n_best = len(best_h)
    
    # Phase 6: Try different resolution
    if time.time() < deadline - 60:
        for n_try in [n_best + 50, n_best - 50, n_best + 100]:
            if n_try < 100 or n_try > 800:
                continue
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n_best)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 30)
            if alloc < 20:
                break
            ht, ct = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_passes=4)
            if time.time() < deadline - 15:
                ht, ct = efficient_local_search(ht, n_try, deadline - 10,
                    delta_schedule=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], start_radius=60)
            if ct < best_c - 1e-15:
                best_c = ct
                best_h = ht
                n_best = n_try
    
    # Phase 7: Final polish
    if time.time() < deadline - 5:
        best_h, best_c = efficient_local_search(best_h, n_best, deadline - 2,
            delta_schedule=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            start_radius=40)
        n_best = len(best_h)
    
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    n_best = len(best_h)
    return best_h, best_c, n_best
