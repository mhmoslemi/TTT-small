```python
def swap_delta_vec(h, n, i, j):
    """Vectorized computation of delta g for swap (i:1->0, j:0->1)."""
    delta = np.zeros(2*n - 1)
    
    # For position i (going 1->0):
    # Effect on lag d: -(1-h[i+d]) for d where i+d in [0,n-1], i.e., d in [-i, n-1-i]
    #                  -h[i-d] for d where i-d in [0,n-1], i.e., d in [i-(n-1), i]
    
    # d range for i+d in [0, n-1]: d in [-i, n-1-i]
    d_start = -i
    d_end = n - 1 - i
    if d_start < -(n-1): d_start = -(n-1)
    if d_end > n-1: d_end = n-1
    if d_start <= d_end:
        idx_start = n - 1 + d_start
        idx_end = n - 1 + d_end
        delta[idx_start:idx_end+1] -= (1 - h[i+d_start : i+d_end+1])
    
    # d range for i-d in [0, n-1]: d in [i-(n-1), i]
    d_start = i - (n-1)
    d_end = i
    if d_start < -(n-1): d_start = -(n-1)
    if d_end > n-1: d_end = n-1
    if d_start <= d_end:
        idx_start = n - 1 + d_start
        idx_end = n - 1 + d_end
        delta[idx_start:idx_end+1] -= h[i-d_end : i-d_start-1 : -1]
    
    # For position j (going 0->1):
    d_start = -j
    d_end = n - 1 - j
    if d_start < -(n-1): d_start = -(n-1)
    if d_end > n-1: d_end = n-1
    if d_start <= d_end:
        idx_start = n - 1 + d_start
        idx_end = n - 1 + d_end
        delta[idx_start:idx_end+1] += (1 - h[j+d_start : j+d_end+1])
    
    d_start = j - (n-1)
    d_end = j
    if d_start < -(n-1): d_start = -(n-1)
    if d_end > n-1: d_end = n-1
    if d_start <= d_end:
        idx_start = n - 1 + d_start
```

Better approach: For a given target lag d*, find the best i (with h[i]=1) to remove and the best j (with h[j]=0) to add, such that g[d*] is most reduced.

For lag d*, the change in g[d*] from removing h[i] (1→0) is:
-(1-h[i+d*]) - h[i-d*] (with boundary checks)

And from adding h[j] (0→1) is:
+(1-h[j+d*]) + h[j-d*]

So the total change for swap (i,j) at lag d* is:
Δg[d*] = h[i+d*] - h[i-d*] - h[j+d*] + h[j-d*] (with boundary = 0)

To minimize g[d*], we want to maximize -Δg[d*] = h[i-d*] - h[i+d*] + h[j+d*] - h[j-d*]

So we want:
- i with h[i]=1 that maximizes h[i-d*] - h[i+d*] (i.e., i is preceded by 1s and followed by 0s at distance d*)
- j with h[j]=0 that maximizes h[j+d*] - h[j-d*] (i.e., j is followed by 1s and preceded by 0s at distance d*)

This gives O(n) candidate generation for a specific d*.

1. Generate top-K candidates for the worst lag
2. For each candidate (i,j), compute the full delta (O(n))
3. Check if max(g + delta) < current max
4. If yes, make the swap

```python
def local_search_binary(h, n, deadline, max_iter=10000):
    """Intensive local search on binary sequence."""
    h = h.copy()
    g = compute_g(h, n)
    current_max = g.max()
    best_h = h.copy()
    best_max = current_max
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Find worst lag
        worst_idx = np.argmax(g)
        worst_d = worst_idx - (n - 1)  # actual lag value
        
        # Generate candidates for this lag
        # i candidates: h[i]=1, maximize h[i-d] - h[i+d]
        i_scores = []
        for i in ones:
            score = 0
            if 0 <= i - worst_d < n:
                score += h[i - worst_d]
            if 0 <= i + worst_d < n:
                score -= h[i + worst_d]
            i_scores.append((score, i))
        i_scores.sort(reverse=True)
        
        # j candidates: h[j]=0, maximize h[j+d] - h[j-d]
        j_scores = []
        for j in zeros:
            score = 0
            if 0 <= j + worst_d < n:
                score += h[j + worst_d]
            if 0 <= j - worst_d < n:
                score -= h[j - worst_d]
            j_scores.append((score, j))
        j_scores.sort(reverse=True)
        
        # Try top candidates
        best_new_max = current_max
        best_swap = None
        
        top_i = i_scores[:20]
        top_j = j_scores[:20]
        
        for score_i, i in top_i:
            for score_j, j in top_j:
                if i == j:
                    continue
                delta = swap_delta_vec(h, n, i, j)
                new_g = g + delta
                new_max = new_g.max()
                if new_max < best_new_max - 1e-10:
                    best_new_max = new_max
                    best_swap = (i, j, delta, new_g)
        
        if best_swap is not None:
            i, j, delta, new_g = best_swap
            h[i] = 0
            h[j] = 1
            g = new_g
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            current_max = best_new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Try random swaps to escape
            if it % 50 == 0 and len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                delta = swap_delta_vec(h, n, i, j)
                new_g = g + delta
                new_max = new_g.max()
                # Annealing: accept with some probability
                if new_max < current_max or np.random.rand() < 0.01:
                    h[i] = 0
                    h[j] = 1
                    g = new_g
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    current_max = new_max
        
        # Check for convergence
        if it > 100 and it % 100 == 0:
            # Verify
            g_check = compute_g(h, n)
            if not np.allclose(g, g_check, atol=0.1):
                g = g_check
                current_max = g.max()
    
    return best_h, best_max * (2.0/n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    g = compute_g(h, n)
    return float(np.max(g) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def swap_delta_vec(h, n, i, j):
    delta = np.zeros(2 * n - 1)
    # h[i]: 1->0
    ds = max(-i, -(n-1))
    de = min(n - 1 - i, n - 1)
    if ds <= de:
        idx_s = n - 1 + ds
        idx_e = n - 1 + de
        delta[idx_s:idx_e + 1] -= (1 - h[i + ds: i + de + 1])
    ds = max(i - (n - 1), -(n - 1))
    de = min(i, n - 1)
    if ds <= de:
        idx_s = n - 1 + ds
        idx_e = n - 1 + de
        delta[idx_s:idx_e + 1] -= h[i - de: i - ds - 1: -1]
    # h[j]: 0->1
    ds = max(-j, -(n-1))
    de = min(n - 1 - j, n - 1)
    if ds <= de:
        idx_s = n - 1 + ds
        idx_e = n - 1 + de
        delta[idx_s:idx_e + 1] += (1 - h[j + ds: j + de + 1])
    ds = max(j - (n - 1), -(n - 1))
    de = min(j, n - 1)
    if ds <= de:
        idx_s = n - 1 + ds
        idx_e = n - 1 + de
        delta[idx_s:idx_e + 1] += h[j - de: j - ds - 1: -1]
    return delta

def binary_local_search(h, n, deadline, max_iter=50000, top_k=30):

    g = compute_g(h, n)
    current_max = float(g.max())
    best_h = h.copy()
    best_max = current_max
    rng = np.random.RandomState(12345)
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    no_improve = 0
    patience = 5000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        worst_idx = int(np.argmax(g))
        worst_d = worst_idx - (n - 1)
        
        # Score i candidates (h[i]=1 to remove)
        i_scores = np.zeros(len(ones))
        for idx, i in enumerate(ones):
            s = 0.0
            if 0 <= i - worst_d < n:
                s += h[i - worst_d]
            if 0 <= i + worst_d < n:
                s -= h[i + worst_d]
            i_scores[idx] = s
        
        # Score j candidates (h[j]=0 to add)
        j_scores = np.zeros(len(zeros))
        for idx, j in enumerate(zeros):
            s = 0.0
            if 0 <= j + worst_d < n:
                s += h[j + worst_d]
            if 0 <= j - worst_d < n:
                s -= h[j - worst_d]
            j_scores[idx] = s
        
        top_i_idx = np.argsort(i_scores)[::-1][:top_k]
        top_j_idx = np.argsort(j_scores)[::-1][:top_k]
        
        best_new_max = current_max

            zeros.remove(j)

def binary_2swap_search(h, n, deadline, max_iter=20000):
    """Try 2-swap moves (change two 1s to 0s and two 0s to 1s)."""
    h = h.copy()
    g = compute_g(h, n)
    current_max = float(g.max())
    best_h = h.copy()
    best_max = current_max
    rng = np.random.RandomState(999)
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        worst_idx = int(np.argmax(g))
        worst_d = worst_idx - (n - 1)
        
        # Try random 2-swaps biased towards good candidates
        best_new_max = current_max
        best_move = None
        
        n_tries = min(200, len(ones) * len(zeros))
        for _ in range(n_tries):
            i1 = int(rng.choice(ones))
            i2 = int(rng.choice(ones))
            if i1 == i2:
                continue
            j1 = int(rng.choice(zeros))
            j2 = int(rng.choice(zeros))
            if j1 == j2 or j1 == i1 or j1 == i2 or j2 == i1 or j2 == i2:
                continue
            
            delta = swap_delta_vec(h, n, i1, j1) + swap_delta_vec(h, n, i2, j2)
            new_max = float(np.max(g + delta))
            if new_max < best_new_max - 1e-10:
                best_new_max = new_max
                best_move = (i1, i2, j1, j2, g + delta)
        
        if best_move is not None:
            i1, i2, j1, j2, new_g = best_move
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            g = new_g
            for i in [i1, i2]:
                ones.remove(i)
            for j in [j1, j2]:
                zeros.remove(j)
            ones.extend([j1, j2])
            zeros.extend([i1, i2])
            current_max = best_new_max
            no_improve = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                break
    
    return best_h, float(best_max) * (2.0 / n)

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        soft = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return m + soft
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and res.success or (res.fun < obj(h)):
            return project_feasible(res.x, n)
        return h
    except Exception:
        return h

def projected_grad_step(h, n, alpha, n_steps, deadline):
    """Projected gradient with log-sum-exp smoothing."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    n_sub = 2 * n - 1
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        weights = np.exp(sh - sm) / np.exp(sh - sm).sum()
        
        # Gradient of sum_k weights[k] * g[k]
        # g[k] = sum_i h[i]*(1-h[i+k]) (for appropriate k)
        # d(g[k])/d(h[i]) = (1-h[i+k]) - h[i-k] (with boundary=0)
        grad = np.zeros(n)
        for d in range(-(n-1), n):
            w = weights[n - 1 + d]
            if w < 1e-15:
                continue
            for i in range(n):
                val = 0.0
                if 0 <= i + d < n:
                    val += (1 - h[i + d])
                if 0 <= i - d < n:
                    val -= h[i - d]
                grad[i] += w * val
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.8
    
    return best_h, best_c

def thue_morse_balanced(length, seed=42):
    N = 1
    while N < length:
        N *= 2
    t = np.zeros(N, dtype=int)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    t = t[:length]
    rng = np.random.RandomState(seed)
    n_ones = int(t.sum())
    target = length // 2
    if n_ones > target:
        ones = np.where(t == 1)[0]
        flip = rng.choice(ones, n_ones - target, replace=False)
        t[flip] = 0
    elif n_ones < target:
        zeros = np.where(t == 0)[0]
        flip = rng.choice(zeros, target - n_ones, replace=False)
        t[flip] = 1
    return t.astype(float)

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def periodic_binary(n, period, seed=42):
    """Create a periodic binary pattern."""
    rng = np.random.RandomState(seed)
    if period % 2 == 0:
        half = period // 2
        pattern = np.zeros(period)
        ones_in_period = rng.choice(period, half, replace=False)
        pattern[ones_in_period] = 1.0
    else:
        # Adjust: use period+1 then trim
        period2 = period + 1
        half = period2 // 2
        pattern = np.zeros(period2)
        ones_in_period = rng.choice(period2, half, replace=False)
        pattern[ones_in_period] = 1.0
        pattern = pattern[:period]
    
    h = np.tile(pattern, n // period + 1)[:n]
    # Fix to have exactly n//2 ones
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        flip = rng.choice(ones, n_ones - target, replace=False)
        h[flip] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        flip = rng.choice(zeros, target - n_ones, replace=False)
        h[flip] = 1
    return h
```
