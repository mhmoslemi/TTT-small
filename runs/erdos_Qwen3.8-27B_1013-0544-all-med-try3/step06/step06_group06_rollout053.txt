        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h

    target = n // 2
    # Pick n/2 positions
    for i in range(target):
        pos = (offset + i * step) % n
        h[pos] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def simulated_annealing(h_init, n, deadline, seed=42, t_start=None, t_end=None, max_iter=None):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c

    if t_start is None:
        t_start = 0.008
    if t_end is None:
        t_end = 1e-7
    if max_iter is None:
        max_iter = 3000000

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    T = t_start
    cool_rate = (t_end / t_start) ** (1.0 / max_iter)
    stagnant = 0
    stagnation_threshold = 50000

    for it in range(max_iter):
        if time.time() > deadline:
            break
        T *= cool_rate

        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]

        h[a] = 0.0
        h[b] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - current_c

        if delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            current_c = new_c
            ones[ia] = b
            zeros[ib] = a
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stagnant += 1

        # Restart with perturbation if stuck too long
        if stagnant > stagnation_threshold:
            # Perturb: swap 5 random pairs
            for _ in range(5):
                ia = rng.randint(len(ones))
                ib = rng.randint(len(zeros))
                a = ones[ia]
                b = zeros[ib]
                h[a], h[b] = h[b], h[a]
                ones[ia] = b
                zeros[ib] = a
            current_c = compute_c5(h, n)
            T = t_start * 0.5
            stagnant = 0

    return best_h, best_c

def genetic_algorithm(h_init, n, deadline, seed=42, pop_size=40, max_gen=5000):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_rand = to_binary(h_init + 0.3 * rng.randn(n), n)
        population.append(h_rand)

    fitness = [compute_c5(h, n) for h in population]

    for gen in range(max_gen):
        if time.time() > deadline:
            break

        # Sort by fitness
        order = np.argsort(fitness)
        population = [population[i] for i in order]
        fitness = [fitness[i] for i in order]

        if fitness[0] < best_c:
            best_c = fitness[0]
            best_h = population[0].copy()

        new_pop = [population[0].copy()]  # elitism

        while len(new_pop) < pop_size:
            if time.time() > deadline:
                break
            # Tournament selection
            i1 = rng.randint(0, min(10, pop_size))
            i2 = rng.randint(0, min(10, pop_size))
            parent1 = population[i1] if fitness[i1] < fitness[i2] else population[i2]

            i3 = rng.randint(0, min(10, pop_size))
            i4 = rng.randint(0, min(10, pop_size))
            parent2 = population[i3] if fitness[i3] < fitness[i4] else population[i4]

            # Crossover: uniform
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2).astype(float)
            # Fix count
            child = to_binary(child, n)

            # Mutation
            n_mut = rng.randint(1, 6)
            ones = np.where(child == 1)[0]
            zeros = np.where(child == 0)[0]
            for _ in range(n_mut):
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    child[a], child[b] = child[b], child[a]

            new_pop.append(child)

        population = new_pop
        fitness = [compute_c5(h, n) for h in population]

    return best_h, best_c

def slsqp_optimize(h_init, n, deadline, alpha_list=None, maxiter_each=800):
    if alpha_list is None:
        alpha_list = [500, 2000, 10000, 50000, 200000, 1000000]

    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    for alpha in alpha_list:
        if time.time() > deadline:
            break

        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            m = np.max(g)
            sh = alpha * (g - m)
            sm = np.max(sh)
            val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
            return val

        try:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=[(0, 1)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2}],
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c

def subgradient_polish(h_init, n, deadline, max_iter=3000):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_full = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        m = c_full.max()
        ks = int(np.argmax(c_full))

        # Numerical gradient at the active lag
        eps = 1e-6
        n_terms = 2 * n - 1
        valid = ks  # index in full correlation
        # The full correlation index for shift s is n-1+s
        # So shift = ks - (n-1)
        shift = ks - (n - 1)

        grad = np.zeros(n)
        for i in range(n):
            h[i] += eps
            c_new = np.correlate(h, 1 - h, mode='full')[ks] * (2.0 / n)
            grad[i] = (c_new - c_full[ks]) / eps
            h[i] -= eps

        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break

        step = lr / ng
        h_try = project_feasible(h - step * grad, n)
        c_try = compute_c5(h_try, n)

        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
        else:
            lr *= 0.9
            if lr < 1e-10:
                break

    return best_h, best_c

def pair_local_search(h_init, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001]

    h = h_init.copy()
    best_c = compute_c5(h, n)

    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-12:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = h[i] - s * d, h[j] + s * d
    return h, best_c

def make_startup(n, method='random', seed=0):
    rng = np.random.RandomState(seed)
    if method == 'random':
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        return h
    elif method == 'arithmetic':
        h = np.zeros(n)
        step = 2
        while np.gcd(step, n) != 2 and step < n:
            step += 1
        for i in range(n // 2):
            h[(i * step) % n] = 1.0
        return h
    elif method == 'clustered':
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        return h
