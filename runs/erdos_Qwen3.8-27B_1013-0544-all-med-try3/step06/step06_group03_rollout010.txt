            if tot < 1e-15:

def subgrad_hessian(h, n, active_shifts):
    """Compute subgradient and squared norm for Polyak step."""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for s_idx in active_shifts:
        s = s_idx - n_minus_1
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            grad[-s:] += (1.0 - h[:n+s])
            grad[:n+s] += (-h[-s:])
    grad *= dx / len(active_shifts)
    return grad

def subgradient_descent(h, n, max_iters, deadline, lower_bound=0.25):
    """Polyak step subgradient descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_best = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corrs = compute_all_corrs(h, n)
        c_cur = corrs.max()
        
        # Active shifts: those within tolerance of max
        tol = max(0.005, (c_cur - best_c) * 0.5)
        active = np.where(corrs > c_cur - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corrs)])
        
        grad = subgrad_hessian(h, n, active)
        grad_norm_sq = np.dot(grad, grad)
        
        if grad_norm_sq < 1e-20:
            break
        
        # Polyak step size
        gap = c_cur - lower_bound
        if gap < 1e-6:
            alpha = 0.01
        else:
            alpha = gap / (grad_norm_sq + 1e-10)
            alpha = min(alpha, 1.0)  # cap step size
        
        # Try the step
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < c_cur:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                t_best = c_new
        else:
            # Halve the step
            alpha /= 4
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    t_best = c_new
    
    return best_h, best_c

    for m in range(len(corrs)):
        w = weights[m]
        if w < 1e-8:
            continue
        s = m - n_minus_1
        if s >= 0:
            grad[:n-s] += w * (1.0 - x[s:])
            grad[s:] += w * (-x[:n-s])
        else:
            grad[-s:] += w * (1.0 - x[:n+s])
            grad[:n+s] += w * (-x[-s:])
    grad *= dx
    return grad

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_with_grad(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]

    def obj(x):
        return smooth_objective(x, n, alpha)

    def jac(x):
        return smooth_gradient(x, n, alpha)

    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, max_iters, deadline, lr_init=0.5, lr_decay=0.999):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        alpha = 500.0 + 1000.0 * it
        g = smooth_gradient(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new
            stall_count = 0
        else:
            h_new2 = project_feasible(h - (lr / 4) * g, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c - 1e-14:
                h = h_new2
                best_h = h_new2.copy()
                best_c = c_new2
                stall_count = 0
                lr = lr / 2
            else:
                stall_count += 1
                lr *= lr_decay
                if stall_count > 50:
                    break
    return best_h, best_c

def stochastic_local_search(h, n, deadline, rng, deltas=(0.02, 0.01, 0.005, 0.002, 0.001, 0.0005)):
    best_c = compute_c5(h, n)
    no_improve = 0
    max_no_improve = 5000
    d_idx = 0
    while time.time() < deadline and no_improve < max_no_improve:
        d = deltas[d_idx % len(deltas)]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
            h[i] += d
            h[j] -= d
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                no_improve = 0
            else:
                h[i] -= d
                h[j] += d
                no_improve += 1
        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
            h[i] -= d
            h[j] += d
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                no_improve = 0
            else:
                h[i] += d
                h[j] -= d
                no_improve += 1
        if no_improve % 2000 == 0:
            d_idx += 1
    return h, best_c

def pair_search_small(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng, do_pair=True):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    remaining = deadline - time.time()
    t_grad_end = time.time() + min(remaining * 0.25, 60)
    h, best_c = gradient_descent(h, n, max_iters=5000, deadline=t_grad_end)
    if time.time() < deadline - 5:
        t_slsqp = min(remaining * 0.3, 80)
        alphas = [500, 2000, 10000, 50000]
        for a in alphas:
            if time.time() > deadline - 20:
                break
            ht = slsqp_with_grad(h, n, a, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    if time.time() < deadline - 10:
        h, best_c = stochastic_local_search(h, n, min(deadline - 10, time.time() + 60), rng)
    if do_pair and n <= 350 and time.time() < deadline - 5:
        h, best_c = pair_search_small(h, n, deadline - 2, [0.005, 0.002, 0.001, 0.0005])
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def sinusoidal_init(n, freq, amp, phase, rng=None):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction at its native resolution
    alloc1 = min(180, (deadline - t0) * 0.2)
    h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc1, rng, do_pair=True)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try structured sinusoidal initializations
    if time.time() < deadline - 120:
        for freq in [1, 2, 3, 4]:
            for amp in [0.15, 0.25, 0.35]:
                if time.time() > deadline - 100:
                    break
                h_sine = sinusoidal_init(n_init, freq, amp, 0.0)
                c_sine = compute_c5(h_sine, n_init)
                if c_sine < best_c:
                    alloc = min(80, (deadline - time.time()) * 0.15)
                    if alloc > 30:
                        h_opt, c_opt = optimize_at_resolution(h_sine, n_init, time.time() + alloc, rng, do_pair=False)
                        if c_opt < best_c:
                            best_c = c_opt
                            best_h = h_opt

    # Phase 3: Multi-resolution - try finer grids
    n_best = len(best_h)
    for n_try in [300, 350, 400, 500]:
        if time.time() > deadline - 100:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=True)
        x_new = np.linspace(0, 2, n_try, endpoint=True)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng, do_pair=(n_try <= 350))
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_best = n_try

    # Phase 4: Multi-restart with perturbation on best solution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 25) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng, do_pair=False)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # SLSQP with high alpha
        if time.time() < deadline - 8:
            ht = slsqp_with_grad(best_h, n, 100000, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        # Stochastic local search
        if time.time() < deadline - 5:
            best_h, best_c = stochastic_local_search(best_h, n, deadline - 4, rng,
                                                     deltas=(0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001))
        # Pair search if small enough
        if n <= 300 and time.time() < deadline - 3:
            best_h, best_c = pair_search_small(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        elif time.time() < deadline - 3:
            best_h, best_c = stochastic_local_search(best_h, n, deadline - 2, rng,
                                                     deltas=(0.001, 0.0005, 0.0002, 0.0001))

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
