    """Subgradient/mirror descent method for the min-max problem."""
    target = n / 2.0
    dx = 2.0 / n
    L = 2 * n - 1  # number of lags
    w = np.ones(L) / L  # uniform initial distribution over lags
    
    best_c = np.inf
    best_h = h.copy()

    for t in range(1, 10000000):
        if time.time() > deadline:
            break
        
        # Update distribution
        w = w * np.exp(eta * corr / dx)  # corr already has dx factor
        w = w / w.sum()
        
        # Compute weighted gradient
        # g[p] = sum_k w_k * ∂f_k/∂h[p]
        g = np.zeros(n)
        for idx in range(L):
            if w[idx] < 1e-10:

            # ∂f_k/∂h[p] = (1-h[p+ell])*1[p+ell valid] - h[p-ell]*1[p-ell valid]

            if p_lo2 <= p_hi2:

        # Gradient step
        step = eta  # or adaptive

        h = project_feasible(h, n)
        
        # Update correlation

- Can handle the box constraints natively
- The penalty handles the equality constraint approximately
- We can use many more iterations in the same time

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max of correlation*dx and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    L = len(c)  # 2n-1
    
    # Softmax for p_k
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    p = exp_c / exp_c.sum()
    
    # Smooth max value
    val = m + np.log(exp_c.sum()) / alpha
    
    # Gradient: g[p] = sum_k p[k] * df_k/dh[p]
    # df_k/dh[p] = (1-h[p+ell_k])*1[0<=p+ell_k<n] - h[p-ell_k]*1[0<=p-ell_k<n]
    # where ell_k = k - (n-1)
    g = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for k in range(L):
        if p[k] < 1e-12:
            continue
        ell = lags[k]
        p_val = p[k]
        # First term: (1-h[p+ell]) for valid p
        p_lo = max(0, -ell)
        p_hi = min(n-1, n-1-ell)
        if p_lo <= p_hi:
            g[p_lo:p_hi+1] += p_val * (1.0 - h[p_lo+ell:p_hi+1+ell])
        # Second term: -h[p-ell] for valid p
        p_lo2 = max(0, ell)
        p_hi2 = min(n-1, n-1+ell)
        if p_lo2 <= p_hi2:
            g[p_lo2:p_hi2+1] -= p_val * h[p_lo2-ell:p_hi2+1-ell]
    
    # Scale by dx (since we computed c with dx factor)
    # Actually, the gradient of f_k = sum_m h[m]*(1-h[m+ell]) * dx with respect to h[p]
    # is dx * [(1-h[p+ell])*1 - h[p-ell])*1]
    # But I already multiplied c by dx, so the gradient should also be scaled by dx
    g *= dx
    
    return val, g
