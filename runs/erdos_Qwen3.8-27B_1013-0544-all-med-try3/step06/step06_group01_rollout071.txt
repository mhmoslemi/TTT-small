    # Formula: (1-h[b]) + h[a] = 0+0 = 0.
    # Correction: +1.
    #
    # Are there other coincidences at k=a-b?
    # "a second": first=k+a=a-b+a=2a-b. n=a. I2 = 1 if 2a-b in [0,n-1].
    # "b first": n=b-k=b-(a-b)=2b-a. I3 = 1 if 2b-a in [0,n-1].
    # These are different n values (a vs 2b-a), unless 2b-a=a, i.e., b=a. Not the case.
    # And they're different from the coincident pair (n=b).
    # So no additional coincidence.
    
    # At k=b-a (assuming b>a, so k>0):
    # "a second": first=k+a=b. n=a. I2=1.
    # "b first": n=b-k=b-(b-a)=a. I3=1.

    # Formula: -h[b] + (h[a]-1) = -1+(-1) = -2.
    # Correction: +1.

    # 1. Compute formula delta at all affected k
    # 2. Apply corrections:
    #    - If k=0: subtract 2 (if both a and b are in [0,n-1], which they always are)
    #    - If k=a-b: add 1
    #    - If k=b-a: add 1

    # I1: a-k=a ∈ [0,n-1] ✓
    # I2: k+a=a ∈ [0,n-1] ✓
    # I3: b-k=b ∈ [0,n-1] ✓
    # I4: k+b=b ∈ [0,n-1] ✓
    # So all four are present, and the correction of -2 always applies at k=0.
    
    # At k=a-b: I1 and I4 are present (as shown). Correction +1 applies.
    # At k=b-a: I2 and I3 are present. Correction +1 applies.
    
    # But what if a-b = 0? That means a=b, which is impossible.
    # What if a-b = b-a? That means a=b, impossible.
    # Can a-b = 0? No (a≠b).
    # Can a-b = b-a? Only if a=b.
    
    # So the three corrections are at distinct k values (0, a-b, b-a) since a≠b.
    
    # Great! So the algorithm is:
    
    for k in range(-(n-1), n):
        val = 0
        # "a first": n_val = a-k
        if 0 <= a-k <= n-1:
            val += (1 - h[a-k])
        # "a second": first = k+a
        if 0 <= k+a <= n-1:
            val -= h[k+a]
        # "b first": n_val = b-k
        if 0 <= b-k <= n-1:
            val -= (1 - h[b-k])
        # "b second": first = k+b
        if 0 <= k+b <= n-1:
            val += h[k+b]
        
        # Corrections for coincidences
        if k == 0:
            val -= 2
        if k == a - b:
            val += 1
        if k == b - a:
            val += 1
        
        delta[k + n - 1] = val
    
    return delta

def compute_swap_delta(h, a, b, n):
    """Vectorized computation of correlation delta for swap (a: 0->1, b: 1->0)"""
    delta = np.zeros(2*n - 1)
    
    # A[m] = 1-h[a-m+(n-1)] for m in [a+1, a+n]
    lo_a = a + 1
    hi_a = a + n  # inclusive
    if lo_a <= hi_a:
        idx = np.arange(lo_a, hi_a + 1)
        src = a - idx + (n - 1)  # source indices into h
        delta[lo_a:hi_a+1] += (1 - h[src])
    
    # B[m] = h[m-(n-1)+a] for m in [n-1-a, 2n-2-a]

    hi_b = 2*n - 2 - a
    delta[lo_b:hi_b+1] -= h[np.arange(lo_b, hi_b+1) - (n-1) + a]
    
    # C[m] = 1-h[b-m+(n-1)] for m in [b+1, b+n]
    lo_c = b + 1
    hi_c = b + n
    idx = np.arange(lo_c, hi_c + 1)
    src = b - idx + (n - 1)
    delta[lo_c:hi_c+1] -= (1 - h[src])
    
    # D[m] = h[m-(n-1)+b] for m in [n-1-b, 2n-2-b]
    lo_d = n - 1 - b
    hi_d = 2*n - 2 - b
    delta[lo_d:hi_d+1] += h[np.arange(lo_d, hi_d+1) - (n-1) + b]
    
    # Corrections
    k0 = n - 1  # m for k=0
    delta[k0] -= 2
    
    ka = (a - b) + (n - 1)  # m for k=a-b
    if 0 <= ka < 2*n-1:
        delta[ka] += 1
    
    kb = (b - a) + (n - 1)  # m for k=b-a
    if 0 <= kb < 2*n-1:
        delta[kb] += 1
    
    return delta
