        # d/dx of (1/alpha)*log(sum exp(alpha*(g_k - m))) = sum w_k * dg_k/dx
        # Total gradient = sum w_k * dg_k/dx (since dm/dx = sum w_k * dg_k/dx when alpha->inf)
        # For finite alpha: grad = sum w_k * dg_k/dx
        # Actually: dL/dx = dm/dx + (1/alpha) * d/dx log(sum exp(alpha*(g_k-m)))
        # = dm/dx + (1/alpha) * sum_j exp(alpha*(g_j-m)) / sum_j exp(alpha*(g_j-m)) * alpha * (dg_j/dx - dm/dx)
        # = dm/dx + sum_j w_j * (dg_j/dx - dm/dx)
        # = dm/dx + sum_j w_j * dg_j/dx - dm/dx * sum_j w_j
        # = dm/dx + sum_j w_j * dg_j/dx - dm/dx
        # = sum_j w_j * dg_j/dx
        # So the gradient is just the weighted sum of individual gradients!
        
        # g_k = corr_k * dx = dx * sum_n h[n+k] * (1-h[n]) for the k-th lag
        # But we have the correlation array, so let's use the indexing:
        # corr[m] = sum_n h[n+m] * (1-h[n]) for m = 0, ..., 2n-2
        # The "shift" corresponds to m, and g[m] = corr[m] * dx
        
        # ∂g[m]/∂h[i] = dx * [(1-h[i-m])*[0<=i-m<n] - h[i+m]*[0<=i+m<n]]
        
        # So grad[i] = dx * sum_m w[m] * [(1-h[i-m])*[0<=i-m<n] - h[i+m]*[0<=i+m<n]]
        # = dx * (A[i] - B[i] - C[i]) where A, B, C are as before
        
        w_cumsum = np.cumsum(np.concatenate([[0], w]))
        A = np.zeros(n)
        for i in range(n):
            lo = max(0, i - n + 1)
            hi = min(i, 2*n-2) + 1
            A[i] = w_cumsum[hi] - w_cumsum[lo]
        
        B = np.convolve(h, w, 'full')[:n]
        C = np.correlate(h, w, 'full')[:n]
        
        return dx * (A - B - C)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

            if s < 1e-14:

def compute_swap_delta(h, i, j, n):
    """Compute delta in correlation vector when swapping h[i]:0->1 and h[j]:1->0"""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greed_binary_fast(h_init, n, deadline, max_sweeps=100):
    """Greedy local search: for each 0, find the best swap with any 1"""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros_arr)):
            if time.time() > deadline:
                break
            i = zeros_arr[ii]
            best_nc = cur_c
            best_j = -1
            for jj in range(len(ones_arr)):
                j = ones_arr[jj]
                delta = compute_swap_delta(h, i, j, n)
                nc = (corr + delta).max() * dx

                    best_j = jj

                delta = compute_swap_delta(h, i, j, n)
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                # Update lists

                ones_arr[best_j] = i
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def anneal_binary_v2(h_init, n, deadline, seed=42):

    # cool = (T_min/T)^(1/N_iters)

    n_est = 500000
    cool = (T_min / T) ** (1.0 / n_est)
    
    no_improve = 0
    improve_patience = 50000
    
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        
        delta = compute_swap_delta(h, i, j, n)
        nc = (corr + delta).max() * dx
        
        accept = False
        if nc <= cur_c:

                no_improve = 0

        if no_improve > improve_patience:

            no_improve = 0

def smooth_gradient_descent(h_init, n, deadline, alpha=10000, lr=0.01, momentum=0.9):
    """Projected gradient descent with momentum on log-sum-exp objective"""
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    
    while time.time() < deadline:
        corr_raw = np.correlate(h, 1 - h, mode='full')
        corr = corr_raw * dx
        c_max = corr.max()
        shifted = alpha * (corr - c_max)
        shifted -= shifted.max()  # for numerical stability
        w = np.exp(shifted)
        w /= w.sum()
        
        # Compute gradient
        # A[i] = sum of w[m] for m in [max(0,i-n+1), min(i, 2n-2)]
        w_full = w  # shape (2n-1,)
        w_cumsum = np.concatenate([[0], np.cumsum(w_full)])
        A = np.zeros(n)
        for i in range(n):
            lo = max(0, i - n + 1)
            hi = min(i, 2*n - 2)
            A[i] = w_cumsum[hi + 1] - w_cumsum[lo]
        
        # B[i] = convolve(h, w, 'full')[i]
        B = np.convolve(h, w_full, 'full')[:n]
        
        # C[i] = correlate(h, w, 'full')[i]
        C = np.correlate(h, w_full, 'full')[:n]
        
        grad = dx * (A - B - C)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        h = h - lr * v
        h = project_feasible(h, n)
    
    return h
