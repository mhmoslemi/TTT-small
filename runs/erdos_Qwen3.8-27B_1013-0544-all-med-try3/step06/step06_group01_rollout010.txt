    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n

    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = hb.copy()
    
    # Build position tracking
    pos = np.full(n, -1, dtype=int)  # pos[i] = position of i in ones/zeros array
    ones_list = []
    zeros_list = []
    for idx in range(n):
        if hb[idx] == 1:
            pos[idx] = len(ones_list)
            ones_list.append(idx)
        else:
            pos[idx] = len(zeros_list)
            zeros_list.append(idx)
    
    ones_arr = np.array(ones_list, dtype=int)
    zeros_arr = np.array(zeros_list, dtype=int)
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    rng = np.random.RandomState(seed)
    n_iter_target = 1000000
    T = T0
    cooling = (T_min / T0) ** (1.0 / n_iter_target)
    
    for it in range(n_iter_target + 2000000):
        if time.time() > deadline:
            break
        i = zeros_arr[rng.randint(n_zeros)]
        j = ones_arr[rng.randint(n_ones)]
        d = delta_swap(hb, i, j, n, m_range)
        new_c = (corr + d).max() * dx
        delta_e = new_c - cur_c
        if delta_e < 0:
            accept = True
        elif T > T_min:
            accept = np.log(rng.rand() + 1e-300) > -delta_e / T
        else:
            accept = False
        
        if accept:
            hb[i] = 1.0
            hb[j] = 0.0
            corr += d
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = hb.copy()
            # Update tracking: i moves from zeros to ones, j moves from ones to zeros
            pi = pos[i]  # position of i in zeros_arr
            pj = pos[j]  # position of j in ones_arr
            zeros_arr[pi] = j
            ones_arr[pj] = i
            pos[i] = pj
            pos[j] = pi
            # pos for the swapped elements
            # i is now at position pj in ones_arr
            # j is now at position pi in zeros_arr
        T *= cooling
    
    return best_h, best_c
