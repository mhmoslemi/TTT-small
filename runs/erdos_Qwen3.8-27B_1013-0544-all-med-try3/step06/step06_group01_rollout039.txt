    rng = np.random.RandomState(seed)

    ones_list = np.where(h == 1)[0].copy()

    while time.time() < deadline:
        # Find max lag(s)
        max_val = np.max(corr)
        max_lags = np.where(corr >= max_val - 1e-10)[0]
        m_star = max_lags[rng.randint(len(max_lags))]
        offset = m_star - (n - 1)
        
        # Find good candidates for swap
        # We want a (from zeros) such that h[a+offset] = 1 (so flipping a to 1 reduces corr[m_star])
        # And b (from ones) such that h[b+offset] = 0 (so flipping b to 0 reduces corr[m_star])

        #   If h[a+offset]=1: adds 0 (no change to m_star)
        #   If h[a+offset]=0: adds 1 (worsens m_star)
        # Flipping b from 1 to 0: removes (1-h[b+offset]) from corr[m_star]
        #   If h[b+offset]=1: removes 0 (no change)
        #   If h[b+offset]=0: removes 1 (improves m_star)

    dx = 2.0 / n
    h = np.clip(x, 0, 1)  # Should already be in bounds

    log_sum = np.max(alpha * corr) + np.log(np.exp(shifted).sum())

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def compute_VW(h, n):
    nr = 2 * n - 1
    V = np.zeros((n, nr))
    W = np.zeros((n, nr))
    offsets = np.arange(-(n - 1), n)
    for i in range(n):
        pos = i + offsets
        valid = (pos >= 0) & (pos < n)
        V[i, valid] = 1.0 - h[pos[valid]]
        pos2 = i - offsets
        valid2 = (pos2 >= 0) & (pos2 < n)
        W[i, valid2] = h[pos2[valid2]]
    return V, W

def update_VW(V, W, n, a, b):
    nr = 2 * n - 1
    rows = np.arange(n)
    cols_va = a - rows + n - 1
    m = (cols_va >= 0) & (cols_va < nr)
    V[rows[m], cols_va[m]] -= 1.0
    cols_vb = b - rows + n - 1
    m = (cols_vb >= 0) & (cols_vb < nr)
    V[rows[m], cols_vb[m]] += 1.0
    cols_wa = rows - a + n - 1
    m = (cols_wa >= 0) & (cols_wa < nr)
    W[rows[m], cols_wa[m]] += 1.0
    cols_wb = rows - b + n - 1
    m = (cols_wb >= 0) & (cols_wb < nr)
    W[rows[m], cols_wb[m]] -= 1.0

def fast_binary_opt(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    nr = 2 * n - 1
    nz = n // 2

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(np.max(corr)) * dx

    best_h = h.copy()
    best_c = cur_c
    best_corr = corr.copy()

    zeros = np.where(h == 0)[0].copy()
    ones = np.where(h == 1)[0].copy()

    V, W = compute_VW(h, n)

    T = 0.012 * dx
    T_decay = 0.999997
    T_min = 5e-7 * dx
    no_improve = 0
    max_no_improve = 100000

    while time.time() < deadline:
        ia = rng.randint(nz)
        ib = rng.randint(nz)
        a = zeros[ia]
        b = ones[ib]

        delta = V[a] - W[a] - V[b] + W[b]
        m1 = b - a + n - 1
        if 0 <= m1 < nr:
            delta[m1] += 2.0
        m2 = a - b + n - 1
        if 0 <= m2 < nr:
            delta[m2] += 1.0

        new_max = np.max(corr + delta)
        nc = float(new_max) * dx
        delta_c = nc - cur_c

        accept = False
        if nc <= cur_c + 1e-15:
            accept = True
        elif delta_c < T:
            if rng.random() < np.exp(-delta_c / T):
                accept = True

        if accept:
            h[a] = 1.0
            h[b] = 0.0
            corr = corr + delta
            cur_c = nc
            zeros[ia] = b
            ones[ib] = a
            update_VW(V, W, n, a, b)

            if nc < best_c - 1e-14:
                best_c = nc
                best_h = h.copy()
                best_corr = corr.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            h = best_h.copy()
            corr = best_corr.copy()
            cur_c = best_c
            zeros = np.where(h == 0)[0].copy()
            ones = np.where(h == 1)[0].copy()
            V, W = compute_VW(h, n)
            no_improve = 0
            T = 0.012 * dx

        T = max(T_min, T * T_decay)

    return best_h, best_c

def slsqp_polish(h, n, deadline, max_alpha=2000000):
    dx = 2.0 / n
    nr = 2 * n - 1

    h = project_feasible(h, n)
    V, W = compute_VW(h, n)

    def obj_and_grad(x, alpha):
        h_x = np.clip(x, 0, 1)
        corr = np.correlate(h_x, 1 - h_x, mode='full').astype(np.float64) * dx
        a = alpha * corr
        a_max = np.max(a)
        exp_a = np.exp(a - a_max)
        log_sum = a_max + np.log(exp_a.sum())
        f = log_sum / alpha
        p = exp_a / exp_a.sum()
        g = dx * (V @ p - W @ p)
        return f, g

    current = h.copy()
    current_V, current_W = compute_VW(h, n)

    alphas = [2000, 10000, 50000, 200000, 500000, max_alpha]
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline:
            break
        V, W = compute_VW(current, n)

        def obj(x):
            return obj_and_grad(x, alpha)[0]

        def grad(x):
            return obj_and_grad(x, alpha)[1]

        def cons_fun(x):
            return x.sum() - n / 2.0

        cons = [{'type': 'eq', 'fun': cons_fun}]
        try:
            res = minimize(obj, current, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14,
                                    'disp': False})
            if res.success or res.fun < compute_c5(current, n):
                new_h = project_feasible(res.x, n)
                c = compute_c5(new_h, n)
                if c < compute_c5(current, n):
                    current = new_h
        except Exception:
            pass

    return current

def targeted_search(h, n, deadline, seed=42):
    """Local search targeting the max lag for faster convergence."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    nr = 2 * n - 1
    nz = n // 2

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c = cur_c
    best_corr = corr.copy()

    zeros = np.where(h == 0)[0].copy()
    ones = np.where(h == 1)[0].copy()
    V, W = compute_VW(h, n)

    no_improve = 0
    max_no_improve = 60000

    while time.time() < deadline:
        max_val = np.max(corr)
        max_lags = np.where(np.abs(corr - max_val) < 1e-10)[0]
        m_star = max_lags[rng.randint(len(max_lags))]
        offset = m_star - (n - 1)

        # Find b candidates: ones where h[b+offset]=0 (reduces corr[m_star])
        b_valid = []
        for ib in range(nz):
            b = ones[ib]
            bp = b + offset
            if 0 <= bp < n and h[bp] == 0:
                b_valid.append(ib)

        if len(b_valid) > 0:
            ib = b_valid[rng.randint(len(b_valid))]
            b = ones[ib]
        else:
            ib = rng.randint(nz)
            b = ones[ib]

        # Find a candidates: zeros where h[a+offset]=1 (neutral for m_star)
        a_preferred = []
        a_all = []
        for ia in range(nz):
            a = zeros[ia]
            ap = a + offset
            if 0 <= ap < n:
                if h[ap] == 1:
                    a_preferred.append(ia)
                else:
                    a_all.append(ia)
            else:
                a_preferred.append(ia)

        if len(a_preferred) > 0 and rng.random() < 0.7:
            ia = a_preferred[rng.randint(len(a_preferred))]
        elif len(a_all) > 0:
            ia = a_all[rng.randint(len(a_all))]
        else:
            ia = rng.randint(nz)
        a = zeros[ia]

        delta = V[a] - W[a] - V[b] + W[b]
        m1 = b - a + n - 1
        if 0 <= m1 < nr:
            delta[m1] += 2.0
        m2 = a - b + n - 1
        if 0 <= m2 < nr:
            delta[m2] += 1.0

        new_max = np.max(corr + delta)
        nc = float(new_max) * dx

        if nc < cur_c - 1e-14:
            h[a] = 1.0
            h[b] = 0.0
            corr = corr + delta
            cur_c = nc
            zeros[ia] = b
            ones[ib] = a
            update_VW(V, W, n, a, b)

            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                best_corr = corr.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            h = best_h.copy()
            corr = best_corr.copy()
            cur_c = best_c
            zeros = np.where(h == 0)[0].copy()
            ones = np.where(h == 1)[0].copy()
            V, W = compute_VW(h, n)
            no_improve = 0

    return best_h, best_c

def batch_local_search(h, n, deadline, seed=42, batch_size=8):

    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    nr = 2 * n - 1
    nz = n // 2

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c = cur_c
    best_corr = corr.copy()

    zeros = np.where(h == 0)[0].copy()
    ones = np.where(h == 1)[0].copy()
    V, W = compute_VW(h, n)

    no_improve = 0
    max_no_improve = 80000

    while time.time() < deadline:
        best_nc = cur_c
        best_a, best_b = -1, -1
        best_ia, best_ib = -1, -1
        best_delta_arr = None

        for _ in range(batch_size):
            ia = rng.randint(nz)
            ib = rng.randint(nz)
            a = zeros[ia]
            b = ones[ib]

            delta = V[a] - W[a] - V[b] + W[b]
            m1 = b - a + n - 1
            if 0 <= m1 < nr:
                delta[m1] += 2.0
            m2 = a - b + n - 1
            if 0 <= m2 < nr:
                delta[m2] += 1.0

            new_max = np.max(corr + delta)
            nc = float(new_max) * dx

            if nc < best_nc:
                best_nc = nc
                best_a, best_b = a, b
                best_ia, best_ib = ia, ib
                best_delta_arr = delta

        if best_a >= 0 and best_nc < cur_c:
            h[best_a] = 1.0
            h[best_b] = 0.0
            corr = corr + best_delta_arr
            cur_c = best_nc
            zeros[best_ia] = best_b
            ones[best_ib] = best_a
            update_VW(V, W, n, best_a, best_b)

            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_corr = corr.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # SA fallback
            ia = rng.randint(nz)
            ib = rng.randint(nz)
            a = zeros[ia]
            b = ones[ib]
            delta = V[a] - W[a] - V[b] + W[b]
            m1 = b - a + n - 1
            if 0 <= m1 < nr:
                delta[m1] += 2.0
            m2 = a - b + n - 1
            if 0 <= m2 < nr:
                delta[m2] += 1.0
            new_max = np.max(corr + delta)
            nc = float(new_max) * dx
            T = 0.008 * dx
            if nc - cur_c < T and rng.random() < np.exp(-(nc - cur_c) / T):
                h[a] = 1.0
                h[b] = 0.0
                corr = corr + delta
                cur_c = nc
                zeros[ia] = b
                ones[ib] = a
                update_VW(V, W, n, a, b)

        if no_improve > max_no_improve:
            h = best_h.copy()
            corr = best_corr.copy()
            cur_c = best_c
            zeros = np.where(h == 0)[0].copy()
            ones = np.where(h == 1)[0].copy()
            V, W = compute_VW(h, n)
            no_improve = 0

    return best_h, best_c
