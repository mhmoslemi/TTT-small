def project_feasible(h, n):

        if w[m_idx] < 1e-15:
            continue

            nv = n - k
            # d/dc w.r.t. h[j]:
            # From h[i] term: j = i, contribution (1-h[j+k]) for j in [0, nv-1]

            grad[k:n] += wm * (-h[0:nv])

            nv = n - kk

            # c[m] = sum_{i=max(0, kk)}^{n-1} h[i] * (1-h[i-kk]) * dx
            # = sum_{i=kk}^{n-1} h[i] * (1-h[i-kk]) * dx

            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):

        diff = target - s
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            if s < 1e-14:
                break
            h += (diff / s) * h

                            h[i] -= d
                            h[j] += d
                    if h[i] >= d and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            sweeps += 1
            if not improved:
                break
    return h, best_c


def sinusoidal_init(n, mode=1, amp=0.3):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(np.pi * mode * x)
    return project_feasible(h, n)


def random_smooth_init(n, seed, sigma=8.0):
    rng = np.random.RandomState(seed)
    h = 0.5 + 0.4 * rng.randn(n)
    h = gaussian_filter1d(h, sigma=sigma)
    return project_feasible(h, n)


def block_init(n, num_blocks=6):
    """Create a block-structured initialization"""
    block_size = n // num_blocks
    h = np.zeros(n)
    for b in range(num_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        if b % 2 == 0:
            h[start:end] = 1.0
        else:
            h[start:end] = 0.0
    # Smooth boundaries
    h = gaussian_filter1d(h, sigma=2.0)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = np.full(300, 0.5)
        n_init = 300

    # Primary and secondary resolutions
    n_primary = 600
    n_secondary = 400

    # Interpolate parent to primary resolution
    x_old = np.linspace(0, 2, n_init)
    x_new600 = np.linspace(0, 2, n_primary)
    x_new400 = np.linspace(0, 2, n_secondary)
    h_600 = project_feasible(np.interp(x_new600, x_old, h_init), n_primary)
    h_400 = project_feasible(np.interp(x_new400, x_old, h_init), n_secondary)
    h_orig = project_feasible(h_init, n_init)

    c_600 = compute_c5(h_600, n_primary)
    c_400 = compute_c5(h_400, n_secondary)
    c_orig = compute_c5(h_orig, n_init)

    # Track best overall
    best_h = h_600.copy()
    best_c = min(c_600, c_400, c_orig)
    best_n = n_primary
    if c_400 < best_c:
        best_h, best_c, best_n = h_400, c_400, n_secondary
    elif c_orig < best_c:
        best_h, best_c, best_n = h_orig, c_orig, n_init

    # Phase 1: L-BFGS-B on n=600 (main workhorse)
    phase1_end = min(t0 + budget_s * 0.30, deadline - 300)
    if time.time() < phase1_end:
        h_600, c_600 = run_lbfgs(h_600, n_primary, phase1_end, num_alphas=8)
        if c_600 < best_c:
            best_h, best_c, best_n = h_600, c_600, n_primary

    # Phase 2: Multi-lag focused on best
    phase2_end = min(time.time() + 120, deadline - 250)
    if time.time() < phase2_end:
        h_best = project_feasible(best_h.copy(), best_n)
        h_best, c_best = multi_lag_focused(h_best, best_n, phase2_end, top_k=5, lr_init=0.003)
        if c_best < best_c:
            best_h, best_c = h_best, c_best

    # Phase 3: Coordinate descent on worst lag
    phase3_end = min(time.time() + 100, deadline - 200)
    if time.time() < phase3_end:
        h_best = project_feasible(best_h.copy(), best_n)
        h_best, c_best = coordinate_descent(h_best, best_n, phase3_end, lr_init=0.005)
        if c_best < best_c:
            best_h, best_c = h_best, c_best

    # Phase 4: Alternative initializations at n=600
    if time.time() < deadline - 200:
        alts = []
        alts.append(sinusoidal_init(n_primary, 1, 0.3))
        alts.append(sinusoidal_init(n_primary, 2, 0.25))
        alts.append(random_smooth_init(n_primary, seed, sigma=10.0))
        alts.append(random_smooth_init(n_primary, seed + 100, sigma=5.0))
        alts.append(block_init(n_primary, 4))

        remaining = deadline - 200 - time.time()
        per_alt = max(30, remaining / len(alts))

        for h_alt in alts:
            if time.time() > deadline - 180:
                break
            alloc = min(per_alt, deadline - 180 - time.time())
            if alloc < 20:
                break
            h_opt, c_opt = run_lbfgs(h_alt, n_primary, time.time() + alloc, num_alphas=5)
            if time.time() < deadline - 170:
                h_opt, c_opt = multi_lag_focused(h_opt, n_primary, time.time() + min(30, deadline - 170 - time.time()), top_k=3, lr_init=0.002)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, n_primary

    # Phase 5: L-BFGS-B on n=400 (different resolution may find different basin)
    if time.time() < deadline - 150:
        h_400, c_400 = run_lbfgs(h_400, n_secondary, deadline - 150, num_alphas=6)
        if c_400 < best_c:
            best_h, best_c, best_n = h_400, c_400, n_secondary

    # Phase 6: Final L-BFGS-B polish on best
    phase6_end = min(time.time() + 80, deadline - 60)
    if time.time() < phase6_end:
        best_h, best_c = run_lbfgs(best_h, best_n, phase6_end, num_alphas=4)

    # Phase 7: Multi-lag focused final
    phase7_end = min(time.time() + 60, deadline - 40)
    if time.time() < phase7_end:
        best_h, best_c = multi_lag_focused(best_h, best_n, phase7_end, top_k=3, lr_init=0.001)

    # Phase 8: Pair search for final polish
    if time.time() < deadline - 10:
        phase8_end = deadline - 5
        best_h, best_c = pair_search(best_h, best_n, phase8_end,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                     radius=25)

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
