```python
def minimize_1d_minimax(a, b, x_lo=0.0, x_hi=1.0):
    """Find x in [x_lo, x_hi] that minimizes max_k (a[k]*x + b[k])"""
    n_lags = len(a)
    # Evaluate at boundaries
    vals = [x_lo, x_hi]
    # Find all crossing points in [x_lo, x_hi]
    for i in range(n_lags):
        for j in range(i+1, n_lags):
            da = a[i] - a[j]
            if abs(da) < 1e-15:
                continue
            x_cross = (b[j] - b[i]) / da
            if x_lo < x_cross < x_hi:
                vals.append(x_cross)
    # Evaluate g(x) at all candidate points
    best_x = x_lo
    best_val = np.inf
    for x in vals:
        g = np.max(a * x + b)
        if g < best_val:
            best_val = g
            best_x = x
    return best_x
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

    For L corresponding to lag 0, we use the linearization."""

    g = np.max(a[np.newaxis, :] * x[:, np.newaxis] + b[np.newaxis, :], axis=1)
    idx = np.argmin(g)
    return x[idx]

    n = len(h)
    dx = 2.0 / n

            h[j] = x_new

    return h
```
