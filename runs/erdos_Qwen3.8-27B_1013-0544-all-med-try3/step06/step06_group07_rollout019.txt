    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break

            mask = h > 0.0

        h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]

        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]

    # Phase 1: SLSQP continuation (coarse to fine)
    for beta in [5, 20, 100, 400]:
        if tl() < 120:
            break
        h = slsqp_pass(h, n, target_sum, beta, max_iter=3000, ftol=1e-13)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Projected subgradient with backtracking
    if tl() > 120:
        h_sg, c5_sg = projected_subgrad(h, target_sum, max_iter=80000,
                                         remaining_time_func=tl, min_time=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: FISTA from current best
    if tl() > 100:
        h_f, c5_f = fista_optimize(best_h, target_sum, max_iter=60000,
                                    remaining_time_func=tl, min_time=60)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 4: SLSQP high-beta polish
    if tl() > 80:
        for beta in [1000, 5000, 20000]:
            if tl() < 30:
                break
            h_polish = best_h.copy() if len(best_h) == n else resample_h(best_h, n)
            h_polish = project_feasible(h_polish, target_sum)
            h_polish = slsqp_pass(h_polish, n, target_sum, beta,
                                   max_iter=5000, ftol=1e-15)
            c5 = compute_c5(h_polish)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    # Phase 5: Interleave subgradient + SLSQP
    if tl() > 80:
        h_mid = best_h.copy()
        h_sg2, c5_sg2 = projected_subgrad(h_mid, target_sum, max_iter=30000,
                                           remaining_time_func=tl, min_time=50)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
        if tl() > 30:
            for beta in [5000, 50000]:
                if tl() < 20:
                    break
                h_polish2 = best_h.copy()
                h_polish2 = slsqp_pass(h_polish2, n, target_sum, beta,
                                        max_iter=4000, ftol=1e-15)
                c5 = compute_c5(h_polish2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_polish2.copy()

    # Phase 6: Try n=150
    if tl() > 100:
        n2 = 150
        ts2 = n2 / 2.0
        h2 = resample_h(best_h, n2)
        h2 = project_feasible(h2, ts2)
        for beta in [50, 300, 1500]:
            if tl() < 40:
                break
            h2 = slsqp_pass(h2, n2, ts2, beta, max_iter=3000, ftol=1e-14)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
        if tl() > 40:
            h2b, c5_2b = projected_subgrad(best_h, ts2 if len(best_h) == n2 else n / 2.0,
                                            max_iter=20000, remaining_time_func=tl, min_time=30)
            if len(best_h) == n2:
                if c5_2b < best_c5:
                    best_c5 = c5_2b
                    best_h = h2b.copy()

    # Phase 7: Try n=300
    if tl() > 100:
        n3 = 300
        ts3 = n3 / 2.0
        h3 = resample_h(best_h, n3)
        h3 = project_feasible(h3, ts3)
        for beta in [30, 200, 1000]:
            if tl() < 40:
                break
            h3 = slsqp_pass(h3, n3, ts3, beta, max_iter=2000, ftol=1e-14)
            c5 = compute_c5(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()

    # Phase 8: Random restarts at n=200
    if tl() > 60:
        n_r = len(best_h)
        ts_r = n_r / 2.0
        for restart in range(8):
            if tl() < 30:
                break
            scale = 0.015 / (restart + 1)
            h_r = best_h.copy() + np.random.randn(n_r) * scale
            h_r = project_feasible(h_r, ts_r)
            h_r = slsqp_pass(h_r, n_r, ts_r, 500, max_iter=2000, ftol=1e-14)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                if tl() > 30:
                    h_r2, c5_r2 = projected_subgrad(h_r, ts_r, max_iter=10000,
                                                     remaining_time_func=tl, min_time=20)
                    if c5_r2 < best_c5:
                        best_c5 = c5_r2
                        best_h = h_r2.copy()

    # Phase 9: Final FISTA + SLSQP
    if tl() > 40:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_f2, c5_f2 = fista_optimize(best_h, ts_f, max_iter=30000,
                                      remaining_time_func=tl, min_time=20)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
        if tl() > 15:
            h_f3 = slsqp_pass(best_h, n_f, ts_f, 10000, max_iter=3000, ftol=1e-16)
            c5 = compute_c5(h_f3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_f3.copy()

    # Ensure final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
