```python
def compute_smoothed_grad(h, n, alpha):
    """Compute gradient of smoothed max of cross-correlation.*
    f(h) = g_max + (1/alpha) * log(sum(exp(alpha*(g - g_max))))
    grad f = sum_k softmax_k * grad g_k
    """
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    g_max = c.max()
    shifted = alpha * (c - g_max)
    # Softmax weights
    shifted -= shifted.max()  # for numerical stability
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad g_k w.r.t. h[j]:
    # g_k = dx * sum_i h[i+k-(n-1)] * (1-h[i])
    # dg_k/dh[j] = dx * [(1-h)[j-(k-(n-1))] - h[j+(k-(n-1))]]
    #             = dx * [(1-h)[j-n+1+k] - h[j+n-1-k]]
    # Let s = k - (n-1), then: dg_k/dh[j] = dx * [(1-h)[j-s] - h[j+s]]
    
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        # (1-h)[j-s]: valid when 0 <= j-s < n
        # h[j+s]: valid when 0 <= j+s < n
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            t1[s:] = 1 - h[:n-s]  # (1-h)[j-s] for j >= s
            t2[:n-s] = h[s:]       # h[j+s] for j < n-s
        else:
            t1[:n+s] = 1 - h[-s:]  # (1-h)[j-s] = (1-h)[j+|s|] for j < n-|s|
            t2[-s:] = h[:n+s]      # h[j+s] = h[j-|s|] for j >= |s|
        grad += w[k] * dx * (t1 - t2)
    
    return grad
```

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothed_objective(x, n, alpha):
    """Smoothed max of cross-correlation."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    log_sum = np.log(np.exp(shifted).sum())
    return m + log_sum / alpha

def smoothed_gradient(x, n, alpha):
    """Gradient of smoothed max of cross-correlation."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    n_full = 2 * n - 1
    grad = np.zeros(n)
    one_minus_x = 1 - x
    
    for k in range(n_full):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            t1[s:] = one_minus_x[:n - s]
            t2[:n - s] = x[s:]
        else:
            t1[:n + s] = one_minus_x[-s:]
            t2[-s:] = x[:n + s]
        grad += wk * dx * (t1 - t2)
    
    return grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP optimization with analytic gradient."""
    def obj(x):
        return smoothed_objective(x, n, alpha)
    
    def grad(x):
        return smoothed_gradient(x, n, alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_gradient_step(h, n, alpha, lr, deadline):
    """One step of projected gradient descent."""
    g = smoothed_gradient(h, n, alpha)
    # Remove mean to maintain sum constraint (approximately)
    g_centered = g - g.mean()
    h_new = project_feasible(h - lr * g_centered, n)
    return h_new

def gradient_descent(h, n, alpha, deadline, max_iter=1000, lr_init=0.01):
    """Projected gradient descent with backtracking."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = smoothed_gradient(h, n, alpha)
        g_centered = g - g.mean()
        
        # Backtracking line search
        c_current = compute_c5(h, n)
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            h_try = project_feasible(h - step * g_centered, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_current - 1e-12:
                h = h_try
                found = True
                break
        if not found:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.1, 1.0)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def smoothed_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    log_sum = max_sh + np.log(np.exp(shifted - max_sh).sum())
    return m + log_sum / alpha


def smoothed_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    w = np.exp(shifted - max_sh)
    w /= w.sum()

    n_full = 2 * n - 1
    grad = np.zeros(n)
    one_minus_x = 1.0 - x

    for k in range(n_full):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            t1[s:] = one_minus_x[:n - s]
            t2[:n - s] = x[s:]
        else:
            t1[:n + s] = one_minus_x[-s:]
            t2[-s:] = x[:n + s]
        grad += wk * dx * (t1 - t2)

    return grad


def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smoothed_objective(x, n, alpha)

    def grad(x):
        return smoothed_gradient(x, n, alpha)

    def cons_fun(x):
        return x.sum() - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h


def subgrad_polish(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_arr = np.correlate(h, 1 - h, mode='full') * dx
        m = c_arr.max()
        ks = int(np.argmax(c_arr))
        s = ks - (n - 1)
        one_minus_h = 1.0 - h
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            t1[s:] = one_minus_h[:n - s]
            t2[:n - s] = h[s:]
        else:
            t1[:n + s] = one_minus_h[-s:]
            t2[-s:] = h[:n + s]
        grad = dx * (t1 - t2)
        grad -= grad.mean()

        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if found:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            lr *= 0.9
            if lr < 1e-12:
                break

    return best_h, best_c


def binary_local_search(h, n, deadline, max_sweeps=500):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = sorted(set(np.where(h_bin == 1.0)[0].tolist()))
    zeros = sorted(set(np.where(h_bin == 0.0)[0].tolist()))

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    ones.append(i)
                    ones.sort()
                    zeros = [x for x in range(n) if x not in set(ones)]
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break

    return h_bin, best_c


def pair_swap_search(h, n, deadline, max_sweeps=100):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = sorted(set(np.where(h_bin == 1.0)[0].tolist()))
    zeros = sorted(set(np.where(h_bin == 0.0)[0].tolist()))

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i1_idx in range(len(zeros)):
            if time.time() > deadline:
                break
            i1 = zeros[i1_idx]
            for i2_idx in range(i1_idx + 1, len(zeros)):
                i2 = zeros[i2_idx]
                for j1_idx in range(len(ones)):
                    for j2_idx in range(j1_idx + 1, len(ones)):
                        j1 = ones[j1_idx]
                        j2 = ones[j2_idx]
                        h_bin[i1] = 1.0
                        h_bin[i2] = 1.0
                        h_bin[j1] = 0.0
                        h_bin[j2] = 0.0
                        c = compute_c5(h_bin, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                            ones = [x for x in ones if x != j1 and x != j2]
                            ones.extend([i1, i2])
                            ones.sort()
                            zeros = [x for x in range(n) if x not in set(ones)]
                        else:
                            h_bin[i1] = 0.0
                            h_bin[i2] = 0.0
                            h_bin[j1] = 1.0
                            h_bin[j2] = 1.0
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return h_bin, best_c
