  - Since j ∈ [0, n-1]: j >= max(0, -m_k) is always true (j >= 0)
  - j <= min(n-1, n-1-m_k): if m_k > 0, j <= n-1-m_k; if m_k <= 0, j <= n-1 (always true)
  - Also need -m_k <= j: if m_k > 0, always true (j >= 0 > -m_k); if m_k < 0, j >= -m_k
  - So valid2: (m_k >= 0 or j >= -m_k) AND (m_k <= 0 or j <= n-1-m_k)

Let me code this up properly:

```python
def coord_descent_fast(h, n, max_sweeps=100, deadline=None, rng=None):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    c_curr = np.correlate(h, 1 - h, mode='full')
    
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 1:
            break
        
        # Coordinate order
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            # Compute slope for each lag
            m = lags  # actual lag values
            
            # valid1: h[j] appears as h[n+m] where n = j-m
            # Condition: max(m,0) <= j and (m >= 0 or j <= n-1+m)
            valid1 = np.zeros(2*n-1, dtype=bool)
            valid1[m <= 0] |= True  # for m <= 0, first condition always true
            valid1[m > 0] |= (j >= m[m > 0])
            # Second part: for m > 0, second condition (j <= n-1) always true
            # for m <= 0, need j <= n-1+m
            neg_mask = m <= 0
            valid1[neg_mask] &= (j <= n - 1 + m[neg_mask])
            
            # valid2: h[j] appears as h[n] where n = j
            # Condition: (m >= 0 or j >= -m) and (m <= 0 or j <= n-1-m)
            valid2 = np.zeros(2*n-1, dtype=bool)
            valid2[m >= 0] |= True
            valid2[m < 0] |= (j >= -m[m < 0])
            pos_mask = m > 0
            valid2[pos_mask] &= (j <= n - 1 - m[pos_mask])
            
            slope = np.zeros(2*n-1)
            # For valid1: contribution (1-h[j-m])*dx, where j-m is the n index
            idx1 = np.where(valid1)[0]
            n_idx1 = j - m[idx1]  # should be in [0, n-1]
            slope[idx1] += dx * (1 - h[n_idx1])
            
            # For valid2: contribution -h[j+m]*dx, where j+m is the n+m index
            idx2 = np.where(valid2)[0]
            nm_idx2 = j + m[idx2]  # should be in [0, n-1]
            slope[idx2] -= dx * h[nm_idx2]
            
            # const = c_curr - slope * h[j]
            const = c_curr - slope * h[j]
            
            # Minimize max_k (slope[k]*t + const[k]) over t in [0,1]
            # Use golden section
            def f_of_t(t):
                return np.max(slope * t + const)
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = f_of_t(c_pt)
            fd = f_of_t(d_pt)
            for _ in range(50):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = f_of_t(c_pt)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = f_of_t(d_pt)
            
            t_opt = (a + b) / 2.0
            f0 = f_of_t(0.0)
            f1 = f_of_t(1.0)
            fmid = f_of_t(t_opt)
            if f0 <= fmid and f0 <= f1:
                t_opt = 0.0
            elif f1 <= fmid and f1 <= f0:
                t_opt = 1.0
            
            h[j] = t_opt
            # Update c_curr: c_k changes by slope[k] * (t_opt - old_h_j)
            c_curr += slope * (t_opt - h[j])  # Wait, h[j] is already t_opt
            # Actually: c_curr was computed with old h[j]. New c = c_curr + slope * (t_opt - old_val)
            # But I already set h[j] = t_opt. Let me fix this.
        
        h = project_feasible(h, n)
        c_curr = np.correlate(h, 1 - h, mode='full')
        c_new = float(np.max(c_curr) * dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

Hmm, I'm making this too complicated with the c_curr update. Let me just recompute c_curr at the start of each coordinate update. It's O(n) each time (np.correlate of two length-n arrays is O(n log n) with FFT or O(n^2) direct). Actually np.correlate uses direct computation which is O(n^2) for 'full' mode. With n=300, that's 90K operations per coordinate, 300 coords per sweep, so 27M per sweep. With 100 sweeps, 2.7B total. At ~1ns per op, that's ~3 seconds per sweep set. Should be fine.

Actually wait, np.correlate for 'full' mode with two arrays of length n computes 2n-1 output values, each requiring up to n multiplications. So it's O(n * (2n-1)) ≈ O(2n^2). For n=300, that's 180K multiplications per call. With 300 calls per sweep, that's 54M per sweep. 100 sweeps = 5.4B. At 1ns per op, ~5.4 seconds. Fine.

But wait, I don't need to recompute the full correlation at each coordinate! I only need it at the start of the sweep (or I can maintain it incrementally). Let me just recompute at the start of each sweep and update incrementally within the sweep.

Actually, the incremental update is:
c_new[k] = c_old[k] + slope[k] * (t_new - t_old)

This is exact because c_k is linear in h[j] (for j ≠ the m=0 case). For m=0, it's quadratic, so the linear approximation is slightly off. But the error is small (proportional to the change in h[j] squared).

Let me just use the incremental update and recompute exactly at the end of each sweep.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        return smooth_max_obj(x, n, alpha, top_idx)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, max_sweeps=50, deadline=None, rng=None, verbose=False):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Precompute lag values
    lags = np.arange(-(n-1), n)
    two_n_minus_1 = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 2:
            break
        
        # Recompute full correlation at start of sweep
        c_arr = np.correlate(h, 1 - h, mode='full')  # not multiplied by dx yet
        
        order = rng.permutation(n) if rng is not None else np.arange(n)
        
        for j in order:
            m = lags  # shape (2n-1,))
            
            # Compute slope of c_k (before dx multiplication) w.r.t. h[j]
            # valid1: j-m is in [0,n-1] and in valid n-range for lag m
            #   n-index = j-m, need 0 <= j-m < n
            #   Also need j-m in [max(0,-m), min(n-1,n-1-m)]
            # valid2: j is in valid n-range for lag m, and j+m in [0,n-1]
            #   Need 0 <= j+m < n
            #   Also need j in [max(0,-m), min(n-1,n-1-m)]
            
            # Simplified conditions (since j is always in [0, n-1]):
            # valid1: 0 <= j-m < n AND max(0,-m) <= j-m <= min(n-1,n-1-m)
            #   j-m >= 0 => m <= j
            #   j-m < n => m > j-n (always true since m >= -(n-1) and j < n)
            #   j-m >= max(0,-m): if m>=0, j-m>=0 (already have); if m<0, j-m>=-m => j>=0 (true)
            #   j-m <= min(n-1, n-1-m): if m>=0, j-m<=n-1-m => j<=n-1 (true); if m<0, j-m<=n-1 => j<=n-1+m
            # So valid1: m <= j AND (m >= 0 OR j <= n-1+m)
            
            v1 = (m <= j) & ((m >= 0) | (j <= n - 1 + m))
            # n_index for valid1: j - m
            n_idx1 = j - m[v1]
            
            # valid2: 0 <= j+m < n AND max(0,-m) <= j <= min(n-1,n-1-m)
            #   j+m >= 0 => m >= -j
            #   j+m < n => m < n-j
            #   j >= max(0,-m): if m>=0, j>=0 (true); if m<0, j>=-m
            #   j <= min(n-1, n-1-m): if m>=0, j<=n-1-m; if m<0, j<=n-1 (true)
            # So valid2: m >= -j AND m < n-j AND (m >= 0 OR j >= -m) AND (m <= 0 OR j <= n-1-m)
            # Simplify: m >= -j AND m < n-j AND (m >= 0 OR j >= -m) AND (m <= 0 OR j <= n-1-m)
            # Note: m >= -j implies j >= -m, so (m >= 0 OR j >= -m) is always true
            # So valid2: m >= -j AND m < n-j AND (m <= 0 OR j <= n-1-m)
            
            v2 = (m >= -j) & (m < n - j) & ((m <= 0) | (j <= n - 1 - m))
            # n+m index for valid2: j + m
            nm_idx2 = j + m[v2]
            
            # slope (before dx): for valid1, (1-h[j-m]); for valid2, -h[j+m]
            slope = np.zeros(two_n_minus_1)
            slope[v1] += (1 - h[n_idx1])
            slope[v2] -= h[nm_idx2]
            
            # const (before dx) = c_arr - slope * h[j]
            const = c_arr - slope * h[j]
            
            # Minimize max_k (slope[k]*t + const[k]) * dx over t in [0,1]
            # Equivalent to minimizing max_k (slope[k]*t + const[k]) over t in [0,1]
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, 1.0
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            fc = np.max(slope * c1 + const)
            fd = np.max(slope * d1 + const)
            for _ in range(45):
                if fc < fd:
                    b = d1
                    d1 = c1
                    fd = fc
                    c1 = b - gr * (b - a)
                    fc = np.max(slope * c1 + const)
                else:
                    a = c1
                    c1 = d1
                    fc = fd
                    d1 = a + gr * (b - a)
                    fd = np.max(slope * d1 + const)
            
            t_opt = (a + b) / 2.0
            f0 = np.max(const)  # t=0
            f1 = np.max(slope + const)  # t=1
            fmid = np.max(slope * t_opt + const)
            if f0 <= fmid and f0 <= f1:
                t_opt = 0.0
            elif f1 <= fmid and f1 <= f0:
                t_opt = 1.0
            
            # Update c_arr incrementally
            c_arr += slope * (t_opt - h[j])
            h[j] = t_opt
        
        # Project to feasible set
        h = project_feasible(h, n)
        # Recompute exactly
        c_new = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            if verbose:
                print(f"  sweep {sweep}: c5 = {c_new:.8f}")
    
    return best_h, best_c

def local_search(h, n, deadline):

            break

                break
