import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def precompute_A(h, ones_idx, n):
    """Precompute A matrix for all one positions."""
    k_lags = np.arange(-(n-1), n)
    A = np.zeros((len(ones_idx), 2*n-1))
    for bi, i in enumerate(ones_idx):
        im = i + k_lags
        imn = i - k_lags
        vp = (im >= 0) & (im < n)
        vm = (imn >= 0) & (imn < n)
        if vp.any():
            A[bi, vp] += h[im[vp]]
        if vm.any():
            A[bi, vm] -= (1.0 - h[imn[vm]])
    return A

def precompute_B(h, zeros_idx, n):
    """Precompute B matrix for all zero positions."""
    k_lags = np.arange(-(n-1), n)
    B = np.zeros((len(zeros_idx), 2*n-1))
    for bj, j in enumerate(zeros_idx):
        jm = j + k_lags
        jmn = j - k_lags
        vp = (jm >= 0) & (jm < n)
        vm = (jmn >= 0) & (jmn < n)
        if vm.any():
            B[bj, vm] += (1.0 - h[jmn[vm]])
        if vp.any():
            B[bj, vp] -= h[jm[vp]]
    return B

def best_improvement_search(h_init, n, deadline, max_iter=5000, seed=42):
    """Best-improvement binary local search."""
    rng = np.random.RandomState(seed)
    target = n // 2
    
    # Start from h_init (should be binary)
    h = to_binary(h_init, n).copy()
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Precompute A and B
        A = precompute_A(h, ones_idx, n)  # (|ones|, 2n-1)
        B = precompute_B(h, zeros_idx, n)  # (|zeros|, 2n-1)
        
        # Find best (i, j) pair
        best_delta = 0.0
        best_pair = None
        
        for bi in range(len(ones_idx)):
            # For this i, compute new_max for all j
            # new_corr[j, m] = corr[m] + A[bi, m] + B[j, m]
            # = (corr + A[bi])[m] + B[j, m]
            base = corr + A[bi]  # (2n-1,)
            new_max = (base[None, :] + B).max(axis=1) * dx  # (|zeros|,)
            
            bj_best = new_max.argmin()
            if new_max[bj_best] < cur_c - best_delta - 1e-15:
                best_delta = cur_c - new_max[bj_best]
                best_pair = (bi, bj_best)
                best_new_corr = base + B[bj_best]
        
        if best_pair is None:
            break  # No improving move found
        
        bi, bj = best_pair
        h[ones_idx[bi]] = 0.0
        h[zeros_idx[bj]] = 1.0
        corr = best_new_corr.copy()
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def simulated_annealing(h_init, n, deadline, max_iter=5000, T0=0.01, T_end=0.0001, seed=42):

    rng = np.random.RandomState(seed)
    target = n // 2
    
    h = to_binary(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    k_lags = np.arange(-(n-1), n)
    n_iter = max_iter
    cool_rate = (T_end / T0) ** (1.0 / max(n_iter, 1))
    
    for iteration in range(n_iter):
        if time.time() > deadline:
            break
        
        T = T0 * (cool_rate ** iteration)
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Random swap
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        # Compute new correlation
        old_i, old_j = h[i], h[j]
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_c = new_corr.max() * dx
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

A faster version: at each step, only consider O(n) positions and compute the effect in O(n) each, for a total of O(n) * O(n) per step, and n/2 steps, so O(n³) total. For n=250: 15.6 million. Fast!

```python
def greedy_binary_construction(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    corr = np.zeros(2*n-1)
    k_lags = np.arange(-(n-1), n)
    
    for step in range(n // 2):
        best_pos = -1
        best_val = np.inf
        for pos in range(n):
            # Effect of adding 1 at pos:
            # In h[n+k] term: adds 1*(1-h[pos+k-k']) ... 
            # Actually, the change in corr[m] when we set h[pos] from 0 to 1:
            # corr[m] = sum_n h[n+k_m]*(1-h[n])
            # When h[pos] changes from 0 to 1:
            # - In h[n+k] term: if n+k_m = pos, i.e., n = pos-k_m, the term changes from 0 to (1-h[pos-k_m])
            #   Change: +(1-h[pos-k_m])
            # - In (1-h[n]) term: if n = pos, the term changes from h[pos+k_m]*(1-0) to h[pos+k_m]*(1-1) = 0
            #   Change: -h[pos+k_m]
            im = pos + k_lags
            imn = pos - k_lags
            vp = (im >= 0) & (im < n)
            vm = (imn >= 0) & (imn < n)
            delta = np.zeros(2*n-1)
            if vm.any():
                delta[vm] += (1.0 - h[imn[vm]])
            if vp.any():
                delta[vp] -= h[im[vp]]
            new_corr = corr + delta
            new_max = new_corr.max()
            if new_max < best_val:
                best_val = new_max
                best_pos = pos
        
        h[best_pos] = 1.0
        # Update corr
        im = best_pos + k_lags
        imn = best_pos - k_lags
        vp = (im >= 0) & (im < n)
        vm = (imn >= 0) & (imn < n)
        if vm.any():
            corr[vm] += (1.0 - h[imn[vm]])
        if vp.any():
            corr[vp] -= h[im[vp]]
    
    return h
```

This is O(n/2 * n * 2n) = O(n³) which for n=250 is about 7.8 million operations. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def greedy_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    corr = np.zeros(2*n-1)
    k_lags = np.arange(-(n-1), n)
    
    for step in range(n // 2):
        best_pos = -1
        best_val = np.inf
        # Try all positions
        for pos in range(n):
            im = pos + k_lags
            imn = pos - k_lags
            vp = (im >= 0) & (im < n)
            vm = (imn >= 0) & (imn < n)
            delta = np.zeros(2*n-1)
            if vm.any():
                delta[vm] += (1.0 - h[imn[vm]])
            if vp.any():
                delta[vp] -= h[im[vp]]
            new_max = (corr + delta).max()
            if new_max < best_val:

                best_pos = pos
        
        h[best_pos] = 1.0
        im = best_pos + k_lags
        imn = best_pos - k_lags
        vp = (im >= 0) & (im < n)
        vm = (imn >= 0) & (imn < n)
        if vm.any():
            corr[vm] += (1.0 - h[imn[vm]])
        if vp.any():
            corr[vp] -= h[im[vp]]
    
    return h

def best_improvement_search(h_init, n, deadline, max_iter=5000, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    h = to_binary(h_init, n).copy()
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    k_lags = np.arange(-(n-1), n)
    
    no_improve_count = 0
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Precompute A (for ones) and B (for zeros)
        n_corr = 2*n - 1
        A = np.zeros((len(ones_idx), n_corr))
        for bi, i in enumerate(ones_idx):
            im = i + k_lags
            imn = i - k_lags
            vp = (im >= 0) & (im < n)
            vm = (imn >= 0) & (imn < n)
            if vp.any():
                A[bi, vp] += h[im[vp]]
            if vm.any():
                A[bi, vm] -= (1.0 - h[imn[vm]])
        
        B = np.zeros((len(zeros_idx), n_corr))
        for bj, j in enumerate(zeros_idx):
            jm = j + k_lags
            jmn = j - k_lags
            vp = (jm >= 0) & (jm < n)
            vm = (jmn >= 0) & (jmn < n)
            if vm.any():
                B[bj, vm] += (1.0 - h[jmn[vm]])
            if vp.any():
                B[bj, vp] -= h[jm[vp]]
        
        # Find best (i, j) pair
        best_delta = 0.0
        best_bi = -1
        best_bj = -1
        best_new_corr = None
        
        for bi in range(len(ones_idx)):
            base = corr + A[bi]
            # new_max[j] = max_m (base[m] + B[j, m])
            new_max = (base[None, :] + B).max(axis=1) * dx
            bj_min = new_max.argmin()
            if new_max[bj_min] < cur_c - best_delta - 1e-15:
                best_delta = cur_c - new_max[bj_min]
                best_bi = bi
                best_bj = bj_min
                best_new_corr = base + B[bj_min]
        
        if best_bi < 0:
            break
        
        h[ones_idx[best_bi]] = 0.0
        h[zeros_idx[best_bj]] = 1.0
        corr = best_new_corr.copy()
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 100:
                break
    
    return best_h, best_c

def simulated_annealing(h_init, n, deadline, max_iter=10000, T0=0.005, T_end=0.00001, seed=42):
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    cool_rate = (T_end / T0) ** (1.0 / max(max_iter, 1))
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (cool_rate ** iteration)
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_c = new_corr.max() * dx
        
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        try:
            g = np.correlate(x, 1.0 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        except Exception:
            return 1e10
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def greedy_binary(n, seed=42):
    h = np.zeros(n)
    corr = np.zeros(2 * n - 1)
    k_lags = np.arange(-(n - 1), n)
    for step in range(n // 2):
        best_pos = -1
        best_val = np.inf
        for pos in range(n):
            im = pos + k_lags
            imn = pos - k_lags
            vp = (im >= 0) & (im < n)
            vm = (imn >= 0) & (imn < n)
            delta = np.zeros(2 * n - 1)
            if vm.any():
                delta[vm] += (1.0 - h[imn[vm]])
            if vp.any():
                delta[vp] -= h[im[vp]]
            new_max = (corr + delta).max()
            if new_max < best_val:
                best_val = new_max

        h[best_pos] = 1.0
        im = best_pos + k_lags
