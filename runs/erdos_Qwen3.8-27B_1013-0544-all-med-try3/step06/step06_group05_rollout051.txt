```python
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vector(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, active_threshold=0.001):
    """Compute subgradient of the max and take a step."""
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    c_max = c.max()
    # Active lags: those within active_threshold of max
    active = np.where(c >= c_max - active_threshold)[0]
    
    # Compute subgradient (average of gradients of active lags)
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)  # shift
        for i in range(n):
            if 0 <= i - s <= n - 1:
                grad[i] += (1.0 - h[i - s])
            if 0 <= i + s <= n - 1:
                grad[i] -= h[i + s]
    grad /= len(active)
    
    return grad, c_max, active

def subgradient_optimize(h, n, deadline, initial_step=0.01, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step

        h_new = project_feasible(h_new, n)

            step = min(step * 1.05, 0.1)  # increase step slightly on success

    return best_h, best_c
```

Specifically: find direction d (with sum(d) = 0 for feasibility, ||d||_∞ ≤ ε) that maximizes min_m (∇c[m] · d) over the active lags. This is a linear program!

If the optimal value of this LP is negative, we have a descent direction.

```python
from scipy.optimize import linprog

def find_descent_direction(h, n, K=10, eps=0.01):
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    c_max = c.max()
    active = np.argsort(c)[-K:]  # top K lags
    
    # Compute gradients of active lags
    grads = []
    for m in active:
        s = m - (n - 1)
        grad = np.zeros(n)
        # (1-h[i-s]) for valid i
        lo = max(0, s)
        hi = min(n, n + s)
        if lo < hi:
            grad[lo:hi] += (1 - h[lo-s:hi-s])
        # -h[i+s] for valid i
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+s:hi2+s]
        grads.append(grad)
    
    grads = np.array(grads)  # shape (K, n)
    
    # LP: maximize t subject to grads @ d >= t, sum(d) = 0, -eps <= d <= eps
    # Variables: d[0], ..., d[n-1], t
    # Minimize -t
    # Constraints: -grads @ d + t <= 0 (i.e., grads @ d >= t)
    #              sum(d) = 0
    #              -eps <= d_i <= eps
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1  # minimize -t = maximize t
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = -grads  # -grads @ d + t <= 0
    A_ub[:, -1] = 1
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1  # sum(d) = 0
    b_eq = [0]
    
    bounds = [(0, 2*eps)] * n + [(None, None)]
    # Wait, d can be negative, so bounds should be (-eps, eps) for d, and (None, None) for t
    bounds = [(-eps, eps)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0 and res.fun < -1e-10:  # t > 1e-10, so we have descent
        return res.x[:n]
    return None
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_lag(h, s, n):
    """Compute gradient of c[m]*dx where s = m-(n-1) is the shift."""
    grad = np.zeros(n)
    # (1-h[i-s]) for valid i: i-s in [0,n-1] means i in [s, s+n-1], intersect [0,n-1]
    lo = max(0, s)
    hi = min(n, n + s)
    if lo < hi:
        grad[lo:hi] += (1.0 - h[lo - s:hi - s])
    # -h[i+s] for valid i: i+s in [0,n-1] means i in [-s, n-1-s], intersect [0,n-1]
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return grad * (2.0 / n)  # multiply by dx

def find_lpm_direction(h, n, K=20, eps=0.01):
    """Find steepest descent direction using LP."""
    c = compute_c5_vector(h, n)
    # Get top K lags
    active = np.argsort(c)[-K:]
    
    grads = np.array([compute_grad_lag(h, m - (n-1), n) for m in active])
    
    # LP: maximize t s.t. grads @ d >= t, sum(d)=0, |d_i| <= eps
    # Minimize -t
    c_obj = np.concatenate([np.zeros(n), [-1.0]])
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = -grads
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(-eps, eps)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.status == 0:
            t_val = -res.fun
            if t_val > 1e-10:
                return res.x[:n], t_val
    except Exception:
        pass
    return None, 0.0

                break
            elif c_new < c_new and c_new < compute_c5(h, n):
                h = h_new  # accept even if not improving best
                no_improve_count = 0
                improved = True
                break
        if not improved:
            no_improve_count += 1
            eps *= 0.95
            if no_improve_count > 100:
                break
    
    return best_h, best_c
```

```python
def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

        h = best_h.copy()
        improved = True

            improved = False

                    return best_h, best_c

                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)

                        continue
```
