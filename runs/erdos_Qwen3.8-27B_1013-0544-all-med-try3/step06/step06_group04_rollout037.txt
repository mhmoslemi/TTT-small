from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(lag, h, n):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    if lag > 0:
        left = np.zeros(n)
        right = np.zeros(n)
        left[:n-m] = 1.0 - h[m:]
        right[m:] = h[:n-m]
        return dx * (left - right)
    else:
        left = np.zeros(n)
        right = np.zeros(n)
        left[m:] = 1.0 - h[:n-m]
        right[:n-m] = h[m:]
        return dx * (left - right)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_max_grad(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    return e / e.sum()

def active_obj_and_grad(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    val = smooth_max_val(active, alpha)
    w = smooth_max_grad(active, alpha)
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        lag = idx - (n - 1)
        grad += wi * lag_grad(lag, x, n)
    return val, grad

def active_slsqp(h, n, top_idx, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    
    def obj(x):
        return active_obj_and_grad(x, n, top_idx, alpha)[0]
    
    def grad(x):
        return active_obj_and_grad(x, n, top_idx, alpha)[1]
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def full_smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    val = smooth_max_val(c, alpha)
    w = smooth_max_grad(c, alpha)
    grad = np.zeros(n)

        if w[i] > 1e-10:  # Only compute for non-negligible weights
            lag = idx - (n - 1)
            grad += w[i] * lag_grad(lag, x, n)
    return val, grad

Actually, the full objective max_k f_k(h) is non-convex because f_k is bilinear in h (for k ≠ 0) and also non-convex for k=0 (since f_0 = n/2 - sum h² is concave). So the problem is fundamentally non-convex, and we're finding local minima.

The key to finding a good local minimum is:
1. Good starting point (the parent solution)
2. Sufficient optimization time
3. Multiple restarts if time allows

Let me also think about the local search more carefully. The previous local search tries all pairs (i, j) for the worst lag, which is O(n) per iteration and O(n²) total (since we try all i). This is very slow for n=300.

A faster approach: for the worst lag k*, the correlation is:
f_{k*} = dx * sum_{i=0}^{n-1-|k*|} h[i](1-h[i+k*])

To decrease this, I want to find the pair (i, j=i+k*) that gives the largest decrease per unit of "budget" (where budget is the change in sum(h), which must be 0).

For a swap: decrease h[i] by δ and increase h[j] by δ (or vice versa).
Change in f_{k*}: 
  If h[i] > h[j] and we decrease h[i], increase h[j]:
  Δf = -δ(1-h[j+k*]... this is getting complicated.

Actually, for a transfer of mass δ from position p to position q:
h[p] decreases by δ, h[q] increases by δ.
Change in f_k: dx * [(1-h[q+k])*(-δ) + h[q-k]*(δ) + ... ] (depends on which slots p and q fall into).

This is complex. Let me instead use a simpler heuristic:
- For the worst lag k*, iterate through all positions i.
- For each i, compute the "descent direction" for the pair (i, i+k*).
- If h[i] > h[i+k*], try transferring mass from i to i+k*.
- If h[i] < h[i+k*], try transferring mass from i+k* to i.
- Accept if it reduces the overall C₅.

This is what the previous code does, but it's slow because it evaluates the full C₅ after each trial.

A faster approach: only evaluate the change in the top few lags (not all 2n-1 lags). This gives an approximate but much faster evaluation.

def local_search_fast(h, n, best_c, deadline, top_lags_count=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for epoch in range(50):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        worst_idx = np.argsort(c)[-top_lags_count:]
        worst_lags = worst_idx - (n-1)
        
        improved = False
        step_size = 0.001 * (0.9 ** epoch)
        
        for lag in sorted(worst_lags, key=abs):
            m = abs(int(lag))
            if m == 0 or m >= n:
                continue
            if int(lag) > 0:
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        delta = min(step_size, (h[i] - h[j]) * 0.5)
                        # Try moving mass from i to j
                        h[i] -= delta
                        h[j] += delta
                        # Quick check: only check the top lags
                        c_check = compute_all_lags(h, n)
                        c_quick = c_check[worst_idx].max() * (n / (n))  # already scaled
                        if c_quick < best_c - 1e-12:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-6:
                        delta = min(step_size, (h[j] - h[i]) * 0.5)
                        h[i] += delta
                        h[j] -= delta
                        c_check = compute_all_lags(h, n)
                        c_quick = c_check[worst_idx].max()
                        if c_quick < best_c - 1e-12:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
            else:
                # lag < 0, meaning we look at pairs (i+m, i)
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    # For negative lag, the correlation involves h[i+m] and h[i]
                    # Same as positive lag case in terms of which pairs to adjust

                        h[j] += delta
