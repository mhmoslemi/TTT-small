def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_run(h, n, alpha, maxiter, ftol, barrier_eps=0.0):
    """SLSQP with log-sum-exp smoothing and optional barrier."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -700, 700)
        logsum = shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())
        val = m + logsum / alpha
        if barrier_eps > 0:
            # Barrier: -eps * sum log(x*(1-x)), with safety clip
            xc = np.clip(x, 1e-10, 1 - 1e-10)
            val -= barrier_eps * np.sum(np.log(xc * (1 - xc)))
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(res.x >= -1e-10):

    c = np.correlate(h, 1 - h, mode='full') * dx
    cur_max = c.max()
    # Polyak-like step: use current suboptimality
    gap = cur_max - 0.37
    if gap < 1e-4:
        gap = 1e-4
    step_init = gap * dx / np.linalg.norm(c.max() - c, 2)**2
    if step_init < 1e-6:
        step_init = 1e-4
    step = step_init
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Compute subgradient at active lag
        k_max = int(np.argmax(c))
        k = k_max - (n - 1)  # actual lag
        # Gradient of f_k: for each j, d/dh[j] sum_i h[i](1-h[i+k])
        # = (1-h[j+k]) if j+k valid, minus h[j-k] if j-k valid
        grad = np.zeros(n)
        if k >= 0:
            # j ranges from 0 to n-1-k for h[j+k] term
            # j ranges from k to n-1 for h[j-k] term... wait
            # f_k = sum_{j=0}^{n-1-k} h[j] * (1-h[j+k])
            # d/dh[j] = (1-h[j+k]) for j < n-k, plus -h[j-k] for j >= k (from term where i=j-k, i+k=j)
            # Actually: f_k = sum_i h[i](1-h[i+k])
            # d/dh[j]: j appears as i in term j (coefficient 1-h[j+k])
            #           and j appears as i+k where i=j-k (coefficient -h[j-k])
            j_max = n - 1 - k
            grad[:j_max + 1] += (1.0 - h[k:k + j_max + 1])
            if k > 0:
                grad[k:] -= h[:j_max + 1]
        else:
            k_abs = -k
            # f_{-k_abs} = sum_i h[i](1-h[i-k_abs]) = sum_i h[i](1-h[i+(-k_abs)])
            # Valid: i-k_abs >= 0, so i >= k_abs
            # = sum_{i=k_abs}^{n-1} h[i](1-h[i-k_abs])
            # d/dh[j]: j appears as i in term j (if j >= k_abs): coefficient (1-h[j-k_abs])
            #          j appears as i-k_abs where i=j+k_abs (if j+k_abs < n): coefficient -h[j+k_abs]
            j_min = k_abs
            grad[j_min:] += (1.0 - h[:n - k_abs])
            if k_abs > 0:
                grad[:n - k_abs] -= h[k_abs:]
        # Normalize and update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = max(step_init * (gap / (gap + it * 1e-7)), 1e-7)
        h = h - step * grad / gnorm
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full') * dx
        new_max = c.max()
        if new_max < best_c:
            best_c = new_max
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 8
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary local search + annealing (50s)
    p1_end = t0 + 0.05 * total
    hb, cb = greed_binary_fast(h0, n, p1_end, max_sweeps=400)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Short annealing restarts
    for restart in range(8):
        if time.time() > p1_end - 1:
            break
        seg_end = min(p1_end, time.time() + 4)
        ha, ca = anneal_binary(best_h, n, seg_end, seed=seed + restart * 7919, n_iters=300000)
        if ca < best_c:
            best_c, best_h = ca, ha.copy()

    # Phase 2: Subgradient exploration (60s)
    p2_end = t0 + 0.105 * total
    hs, cs = subgradient_explore(best_h, n, p2_end, n_iters=8000000, seed=seed)
    if cs < best_c:
        best_c, best_h = cs, hs.copy()

    # Phase 3: Coarse SLSQP (150s)
    p3_end = t0 + 0.24 * total
    h = best_h.copy()
    alphas_coarse = [30, 100, 500, 2000, 10000]
    for alpha in alphas_coarse:
        if time.time() > p3_end - 5:
            break
        h = slsqp_run(h, n, alpha, 5000, 1e-15, 0.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Fine SLSQP (400s) - main workhorse
    p4_end = t0 + 0.61 * total
    h = best_h.copy()
    alphas_fine = [50000, 200000, 1000000]
    for alpha in alphas_fine:
        for restart in range(4):
            if time.time() > p4_end - 8:
                break
            h = slsqp_run(h, n, alpha, 12000, 1e-16, 0.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
            if restart < 3:
                h = best_h + 0.001 * rng.randn(n)
                h = project_feasible(h, n)

    # Phase 5: Ultra-fine SLSQP (300s)
    p5_end = t0 + 0.89 * total
    h = best_h.copy()
    alphas_ultra = [5000000, 20000000, 100000000]
    for alpha in alphas_ultra:
        if time.time() > p5_end - 10:
            break
        h = slsqp_run(h, n, alpha, 20000, 1e-17, 0.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Barrier-assisted (50s)
    p6_end = dl
    if time.time() < p6_end:
        h = best_h.copy()
        for eps in [0.005, 0.001, 0.0005]:
            if time.time() > p6_end - 8:
                break
            h = slsqp_run(h, n, 50000000, 8000, 1e-16, eps)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        # Final ultra-sharp polish
        if time.time() < p6_end - 3:
            h = best_h.copy()
            h = slsqp_run(h, n, 200000000, 10000, 1e-17, 0.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Final projection and exact evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    if time.time() > dl:
        pass
    return best_h, best_c, n
