import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (500, 10.0), (2000, 50.0), (10000, 200.0),
        (50000, 500.0), (200000, 2000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def coord_descent(h, n, deadline, grid_size=100, penalty=1000.0):
    dx = 2.0 / n
    h = project_feasible(h, n)
    f = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(f.max())
    best_h = h.copy()
    n_lags = 2 * n - 1
    t_grid = np.linspace(0.0, 1.0, grid_size)
    target = n / 2.0
    sweeps = 0
    rng = np.random.RandomState(42)
    current_sum = h.sum()
    a_arr = np.empty(n_lags)

    while time.time() < deadline:
        improved = False
        order = rng.permutation(n)
        for _idx in range(n):
            i = order[_idx]
            t0 = h[i]

            # Compute a_k for all lags
            for k in range(-n + 1, n):
                if k >= 0:
                    ak = 0.0
                    if i >= k:
                        ak += 1.0 - h[i - k]
                    if i + k < n:
                        ak -= h[i + k]
                else:
                    kp = -k
                    ak = 0.0
                    if i + kp < n:
                        ak += 1.0 - h[i + kp]
                    if i >= kp:
                        ak -= h[i - kp]
                a_arr[k + n - 1] = ak

            # Objective on grid: g(t) = max(quadratic, linear, penalty)
            b = f - t0 * a_arr
            b[n - 1] = -1e18
            L = (b[:, None] + t_grid[None, :] * a_arr[:, None]).max(axis=0)
            q = f[n - 1] + (t_grid - t0) * (1.0 - t_grid - t0)
            dev_base = current_sum - target + t0
            pen = penalty * (dev_base - t_grid) ** 2
            g = np.maximum(np.maximum(q, L), pen)
            best_idx = g.argmin()
            t_new = t_grid[best_idx]

            if 0 < best_idx < grid_size - 1:
                step = 1.0 / (grid_size - 1)
                lo = max(0.0, t_grid[best_idx] - 3.0 * step)
                hi = min(1.0, t_grid[best_idx] + 3.0 * step)
                t_fine = np.linspace(lo, hi, 60)
                L_f = (b[:, None] + t_fine[None, :] * a_arr[:, None]).max(axis=0)
                q_f = f[n - 1] + (t_fine - t0) * (1.0 - t_fine - t0)
                pen_f = penalty * (dev_base - t_fine) ** 2
                g_f = np.maximum(np.maximum(q_f, L_f), pen_f)
                t_new = t_fine[g_f.argmin()]

            if abs(t_new - t0) > 1e-12:
                delta = t_new - t0
                f += delta * a_arr
                f[n - 1] = f[n - 1] + (t_new * (1.0 - t_new) - t0 * (1.0 - t0)) - delta * a_arr[n - 1]
                h[i] = t_new
                current_sum += delta
                improved = True

        sweeps += 1
        if sweeps % 50 == 0:
            h = project_feasible(h, n)
            current_sum = h.sum()
            f = np.correlate(h, 1 - h, mode='full') * dx

        c = float(f.max())
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()

        if not improved:
            break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c
