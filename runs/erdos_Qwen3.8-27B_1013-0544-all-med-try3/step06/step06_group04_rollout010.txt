                        h[j] += delta
                        h = project_feasible(h, n)
                    if pairs_moved > 200:
                        break
    return h, best_c

def multi_lag_local_search(h, n, deadline):
    """Try to reduce multiple lags simultaneously."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for phase in range(4):
        if time.time() > deadline:
            break
        delta = [0.01, 0.005, 0.002, 0.001][phase]
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            # Get top 3 lags
            top3 = np.argsort(c)[-3:]
            for k_idx in top3:
                if time.time() > deadline:
                    break
                lag = k_idx - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                sign = 1 if lag > 0 else -1
                for i in range(0, n, 3):  # stride 3 for speed
                    j = i + sign * m
                    if j < 0 or j >= n:
                        continue
                    if h[i] > h[j] + 0.02:
                        d = min(delta, (h[i] - h[j]) * 0.4)
                        if d < 1e-10:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > h[i] + 0.02:
                        d = min(delta, (h[j] - h[i]) * 0.4)
                        if d < 1e-10:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
    return h, best_c

def differential_evolution_step(h_best, population, n, F, deadline):
    """One DE mutation: h_best + F*(h_a - h_b)."""
    a = np.random.randint(len(population))
    b = np.random.randint(len(population))
    while b == a and len(population) > 1:
        b = np.random.randint(len(population))
    h_trial = h_best + F * (population[a] - population[b])
    h_trial = project_feasible(h_trial, n)
    c_trial = compute_c5(h_trial, n)
    return h_trial, c_trial

def refinement_pipeline(h, n, deadline, rng):
    """Multi-stage refinement of a solution."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Stage 1: Gradient descent with increasing alpha
    if time.time() < deadline - 5:
        alloc1 = min(20, (deadline - time.time()) * 0.25)
        h1, c1 = gradient_descent(best_h, n, 500, 5000, 200, 0.01, time.time() + alloc1)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Stage 2: Higher alpha gradient
    if time.time() < deadline - 5:
        alloc2 = min(20, (deadline - time.time()) * 0.25)
        h2, c2 = gradient_descent(best_h, n, 5000, 50000, 200, 0.005, time.time() + alloc2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Stage 3: Very high alpha (nearly exact max)
    if time.time() < deadline - 5:
        alloc3 = min(20, (deadline - time.time()) * 0.25)
        h3, c3 = gradient_descent(best_h, n, 50000, 200000, 200, 0.002, time.time() + alloc3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Stage 4: Tight local search
    if time.time() < deadline - 3:
        alloc4 = min(25, (deadline - time.time()) * 0.4)
        h4, c4 = tight_local_search(best_h, n, time.time() + alloc4)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Stage 5: Multi-lag local search
    if time.time() < deadline - 3:
        alloc5 = min(15, (deadline - time.time()) * 0.4)
        h5, c5 = multi_lag_local_search(best_h, n, time.time() + alloc5)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Stage 6: Final fine gradient + local
    if time.time() < deadline - 2:
        h6, c6 = gradient_descent(best_h, n, 100000, 500000, 100, 0.001, time.time() + 10)
        if c6 < best_c:
            best_c = c6
            best_h = h6
        if time.time() < deadline - 1:
            h7, c7 = tight_local_search(best_h, n, deadline - 0.5)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def try_resolution(h_best, n_best, n_target, deadline, rng, alloc):
    """Interpolate to target resolution and optimize."""
    if n_target == n_best:
        return refinement_pipeline(h_best.copy(), n_best, deadline, rng)
    x_best = np.linspace(0, 2, n_best)
    x_target = np.linspace(0, 2, n_target)
    h_start = np.interp(x_target, x_best, h_best)
    h_start = project_feasible(h_start, n_target)
    return refinement_pipeline(h_start, n_target, deadline, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        best_h = h_init.copy()
        best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Refine the initial solution at its native resolution (40% of budget)
    if best_h is not None:
        phase1_deadline = t0 + min(300, (deadline - t0) * 0.4)
        best_h, best_c = refinement_pipeline(best_h, n_init, phase1_deadline, rng)
    
    # Phase 2: Try different resolutions (30% of budget)
    if best_h is not None:
        n_counts = [200, 250, 300, 350, 400, 500]
        if n_init not in n_counts:
            n_counts.append(n_init)
        n_counts = [x for x in n_counts if x != n_init or len(n_counts) == 1]
        # Sort by proximity to current best resolution
        n_counts.sort(key=lambda x: abs(x - n_init))
        
        phase2_deadline = t0 + min(600, (deadline - t0) * 0.7)
        for n_try in n_counts:
            if time.time() > phase2_deadline - 30:
                break
            alloc = min(60, (phase2_deadline - time.time()) * 0.3)
            if alloc < 25:
                break
            h_try, c_try = try_resolution(best_h, len(best_h), n_try, time.time() + alloc, rng, alloc)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    # Phase 3: Population-based exploration (20% of budget)
    if best_h is not None:
        n = len(best_h)
        phase3_deadline = min(deadline - 60, t0 + (deadline - t0) * 0.9)
        
        # Build population
        population = [best_h.copy()]
        for _ in range(5):
            hp = best_h + rng.normal(0, 0.02, n)
            hp = project_feasible(hp, n)
            population.append(hp)
            # Also try structured perturbation: shift a "block"
            shift = rng.randint(5, max(6, n // 10))
            h_shifted = np.roll(best_h, shift)
            h_shifted = project_feasible(h_shifted, n)
            population.append(h_shifted)
        
        # Run DE-like iterations
        n_de_iters = 0
        while time.time() < phase3_deadline - 10 and n_de_iters < 8:
            n_de_iters += 1
            F = 0.3 + 0.1 * n_de_iters
            for i in range(len(population)):
                if time.time() > phase3_deadline - 10:
                    break
                h_best_pop, c_best_pop = compute_c5(population[i], n), population[i].copy()
                h_trial, c_trial = differential_evolution_step(population[i], population, n, F, deadline)
                if c_trial < c_best_pop:
                    population[i] = h_trial
                # Also try gradient refinement on the best
                if i == 0 or (c_trial < c_best_pop):
                    if time.time() < phase3_deadline - 20:
                        h_ref, c_ref = gradient_descent(population[i], n, 10000, 50000, 100, 0.003, time.time() + 15)
                        if c_ref < compute_c5(population[i], n):
                            population[i] = h_ref
            c_best_pop = compute_c5(population[0], n)
            if c_best_pop < best_c:
                best_c = c_best_pop
                best_h = population[0].copy()
            # Re-seed population around best
            population = [best_h.copy()]
            for _ in range(3):
                hp = best_h + rng.normal(0, 0.015 + 0.005 * n_de_iters, n)
                hp = project_feasible(hp, n)
                population.append(hp)
    
    # Phase 4: Final intensive optimization (remaining budget)
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        remaining = deadline - time.time() - 5
        
        # Multiple refinement passes
        n_passes = max(1, int(remaining / 15))
        for p in range(n_passes):
            if time.time() > deadline - 3:
                break
            pass_deadline = time.time() + min(15, remaining * 0.15)
            best_h, best_c = refinement_pipeline(best_h, n, pass_deadline, rng)
            # Small perturbation between passes
            if p < n_passes - 1:
                best_h = best_h + rng.normal(0, 0.003, n)
                best_h = project_feasible(best_h, n)
    
    # Final evaluation
    if best_h is not None:
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
