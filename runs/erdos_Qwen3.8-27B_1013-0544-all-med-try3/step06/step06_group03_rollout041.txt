    target = 0.5 * float(n)
    s = float(x.sum())
    if abs(s - target) <= 1e-13:
        return x

    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if float(y.sum()) < target:
            lo = mid
        else:
            hi = mid

    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior))
        if cnt > 0:
            x[interior] += diff / float(cnt)
            x = np.clip(x, 0.0, 1.0)
            diff = target - float(x.sum())
            if abs(diff) > 1e-12 and np.count_nonzero(interior) > 0:
                ix = np.where(interior)[0]
                if ix.size:
                    x[ix[0]] += diff
                    x = np.clip(x, 0.0, 1.0)
    return x


def lag_value(h, n, d):
    hm1 = 1.0 - h
    if d == 0:
        return float(np.dot(h, hm1))
    if d > 0:
        if d >= n:
            return 0.0
        return float(np.dot(h[d:], hm1[:-d]))
    else:
        o = -d
        if o >= n:
            return 0.0
        return float(np.dot(h[:n - o], hm1[o:]))


def lag_gradient(h, n, d):
    grad = np.zeros(n, dtype=float)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        o = d
        if o >= n:
            return grad
        grad[o:] += 1.0 - h[:-o]
        grad[:n - o] -= h[o:]
        return grad
    else:
        o = -d
        if o >= n:
            return grad
        grad[:n - o] += 1.0 - h[o:]
        grad[o:] -= h[:-o]
        return grad


def fast_corr(h, g, n):
    h = np.asarray(h, dtype=float)
    g = np.asarray(g, dtype=float)
    if n <= 512 or fftconvolve is None:
        return np.correlate(h, g, mode='full')
    return fftconvolve(h, g[::-1], mode='full')


def smooth_lse_obj_grad(h, n, beta):
    h = np.asarray(h, dtype=float)
    g = 1.0 - h
    c = fast_corr(h, g, n)
    c = np.asarray(c, dtype=float)
    cmax = float(np.max(c))

    if beta <= 0.0:
        k = int(np.argmax(c))
        return cmax, lag_gradient(h, n, k - (n - 1))

    z = beta * (c - cmax)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 0.0 or not np.isfinite(s):
        k = int(np.argmax(c))
        return cmax, lag_gradient(h, n, k - (n - 1))

    p = ez / s
    f = cmax + float(np.log(s)) / beta

    if n == 1:
        return f, p[0] * (1.0 - 2.0 * h)

    center = n - 1
    P = np.zeros(n, dtype=float)
    N = np.zeros(n, dtype=float)
    P[1:] = p[center + 1:center + n]
    if center > 0:
        N[1:] = p[0:center][::-1]

    S = P + N
    off = np.abs(np.arange(-(n - 1), n))
    K = S[off]

    if n <= 600:
        y = np.convolve(h, K, mode='full')
    elif fftconvolve is not None:
        y = fftconvolve(h, K, mode='full')
    else:
        y = np.convolve(h, K, mode='full')
    neighbor = y[n - 1:n - 1 + n]

    prefP = np.concatenate(([0.0], np.cumsum(P[1:])))
    prefN = np.concatenate(([0.0], np.cumsum(N[1:])))
    const = prefP + prefN[::-1]

    grad = p[center] * (1.0 - 2.0 * h) + const - neighbor
    return float(f), grad


def lbfgs_objective(x, n, beta, pen):
    f, _ = smooth_lse_obj_grad(x, n, beta)
    err = float(x.sum()) - 0.5 * float(n)
    return float(f + 0.5 * pen * err * err)


def lbfgs_gradient(x, n, beta, pen):
    _, g = smooth_lse_obj_grad(x, n, beta)
    err = float(x.sum()) - 0.5 * float(n)
    return g + pen * err * np.ones(n, dtype=float)


def active_sum_values(h, n, idx):
    m = len(idx)
    if m == 0:
        return 0.0
    hm1 = 1.0 - h
    s = 0.0
    for k in idx:
        d = int(k) - n + 1
        if d == 0:
            s += float(np.dot(h, hm1))
        elif d > 0:
            if d < n:
                s += float(np.dot(h[d:], hm1[:-d]))
        else:
            o = -d
            if o < n:
                s += float(np.dot(h[:n - o], hm1[o:]))
    return s / float(m)


def active_sum_gradient(h, n, idx):
    m = len(idx)
    if m == 0:
        return np.zeros(len(h), dtype=float)
    g = np.zeros(len(h), dtype=float)
    for k in idx:
        d = int(k) - n + 1
        g += lag_gradient(h, n, d)
    return g / float(m)


def active_obj_pen(x, n, idx, pen):
    val = active_sum_values(x, n, idx)
    err = float(x.sum()) - 0.5 * float(n)
    return float(val + 0.5 * pen * err * err)


def active_grad_pen(x, n, idx, pen):
    g = active_sum_gradient(x, n, idx)
    err = float(x.sum()) - 0.5 * float(n)
    return g + pen * err * np.ones(len(x), dtype=float)


def lbfgs_maxiter(n):
    if n <= 150:
        return 350
    if n <= 300:
        return 220
    if n <= 600:
        return 120
    if n <= 1000:
        return 70
    return 40


def optimize_lbfgsb(h, n, beta, pen, maxiter):
    x0 = project_feasible(h, n)
    c0 = exact_counts(x0)
    if maxiter <= 0:
        return x0, c0

    bounds = [(0.0, 1.0)] * n
    maxfun = max(80, 4 * maxiter)
    try:
        res = minimize(
            lbfgs_objective,
            x0,
            method='L-BFGS-B',
            jac=lbfgs_gradient,
            args=(n, beta, pen),
            bounds=bounds,
            options={
                'maxiter': maxiter,
                'ftol': 1e-12,
                'gtol': 1e-6,
                'maxfun': maxfun,
                'iprint': 0,
            }
        )
        x1 = np.asarray(res.x, dtype=float)
    except Exception:
        return x0, c0

    x1 = project_feasible(x1, n)
    c1 = exact_counts(x1)
    if c1 < c0:
        return x1, c1
    return x0, c0


def optimize_active_subset(h, n, pen, maxiter, active_top, max_rounds, deadline=None):
    x = project_feasible(np.asarray(h, dtype=float), n)
    c = exact_counts(x)
    best_x = x.copy()
    best_c = c

    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline - 8.0:
            break

        carr = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(carr)[::-1][:max(1, active_top)]
        near = np.where(carr >= carr.max() - 0.5)[0]
        if near.size > 0:
            if near.size > 25:
                near = near[np.argsort(carr[near])[::-1][:25]]
            idx = np.unique(np.concatenate((top_idx, near))).astype(int)
        else:
            idx = top_idx.astype(int)

        bounds = [(0.0, 1.0)] * n
        maxfun = max(50, 6 * maxiter)
        try:
            res = minimize(
                active_obj_pen,
                x,
                method='L-BFGS-B',
                jac=active_grad_pen,
                args=(n, idx, pen),
                bounds=bounds,
                options={
                    'maxiter': maxiter,
                    'ftol': 1e-12,
                    'gtol': 1e-6,
                    'maxfun': maxfun,
                    'iprint': 0,
                }
            )
            x1 = np.asarray(res.x, dtype=float)
        except Exception:
            break

        x1 = project_feasible(x1, n)
        c1 = exact_counts(x1)
        if c1 < c - 1e-14:
            x = x1
            c = c1
            if c < best_c:
                best_c = c
                best_x = x.copy()
        else:
            break

    return best_x, best_c


def make_pair_deltas(max_d):
    if max_d <= 1e-12:
        return np.array([0.0], dtype=float)
    lo = np.log10(1e-6)
    hi = np.log10(max(1e-6, max_d))
    vals = np.logspace(lo, hi, 28)
    vals = vals[vals <= max_d + 1e-12]
    vals = np.append(vals, [0.0, max_d])
    vals.sort()
    return vals


def pair_line_search_best(h, n, i, j, max_d, cur_count):
    if max_d <= 1e-12:
        return None, None

    ht = np.asarray(h, dtype=float).copy()
    gt = 1.0 - ht
    oi = ht[i]
    oj = ht[j]
    gi = gt[i]
    gj = gt[j]

    best_count = float(cur_count)
    best_d = 0.0

    deltas = make_pair_deltas(max_d)
    for d in deltas:
        d = float(d)
        if d <= 1e-14:
            continue
        ht[i] = oi + d
        ht[j] = oj - d
        gt[i] = gi - d
        gt[j] = gj + d
        c = correlate_counts(ht, gt)
        if c < best_count:
            best_count = c
            best_d = d

    if best_d <= 0.0:
        return None, None

    width = max(1e-5, 0.1 * max_d)
    for _ in range(2):
        lo = max(0.0, best_d - width)
        hi = min(max_d, best_d + width)
        grid = np.linspace(lo, hi, 13)
        for d in grid:
            d = float(d)
            d = min(max(d, 0.0), max_d)
            ht[i] = oi + d
            ht[j] = oj - d
            gt[i] = gi - d
            gt[j] = gj + d
            c = correlate_counts(ht, gt)
            if c < best_count:
                best_count = c
                best_d = d
        width *= 0.35
        if width < 1e-7:
            break

    if best_count < cur_count - 1e-12:
        ht[i] = oi + best_d
        ht[j] = oj - best_d
        return ht, best_count
    return None, None


def active_search(h, n, deadline, max_candidates, active_top, rng):
    x = project_feasible(np.asarray(h, dtype=float), n)
    count = exact_counts(x)
    best_x = x.copy()
    best_count = count
    no_improve = 0

    while time.time() < deadline:
        if time.time() >= deadline - 3.0:
            break

        c = np.correlate(x, 1.0 - x, mode='full')
        count = float(np.max(c))
        if count < best_count:
            best_count = count
            best_x = x.copy()

        top_idx = np.argsort(c)[::-1][:max(1, active_top)]
        g = np.zeros(n, dtype=float)
        for k in top_idx:
            g += lag_gradient(x, n, int(k) - (n - 1))

        if n <= 800:
            try:
                _, gs = smooth_lse_obj_grad(x, n, 32.0)
                g = g + 0.25 * gs
            except Exception:
                pass

        cands = []
        K = 8 if n > 500 else 12
        low = np.argsort(g)[:K]
        high = np.argsort(g)[-K:]
        for ii in range(K):
            i0 = int(low[ii])
            for jj in range(K):
                j0 = int(high[jj])
                if i0 != j0:
                    cands.append((float(g[i0] - g[j0]), i0, j0))

        rad = 4 if n <= 400 else 0
        if rad > 0:
            for i0 in range(n):
                lo = max(0, i0 - rad)
                hi = min(n, i0 + rad + 1)
                for j0 in range(lo, hi):
                    if i0 == j0:
                        continue
                    if g[i0] < g[j0]:
                        cands.append((float(g[i0] - g[j0]), i0, j0))

        for _ in range(80):
            i0 = int(rng.randint(n))
            j0 = int(rng.randint(n))
            if i0 != j0 and g[i0] < g[j0]:
                cands.append((float(g[i0] - g[j0]), i0, j0))

        if not cands:
            break

        cands.sort()
        seen = set()
        chosen = 0
        improved = False

        for _, i0, j0 in cands:
            if chosen >= max_candidates:
                break
            if time.time() >= deadline - 3.0:
                break
            key = i0 * n + j0
            if key in seen:
                continue
            seen.add(key)
            chosen += 1

            md = min(1.0 - x[i0], x[j0])
            if md <= 1e-13:
                continue

            nx, nc = pair_line_search_best(x, n, i0, j0, md, count)
            if nx is not None:
                x = nx
                count = nc
                if count < best_count:
                    best_count = count
                    best_x = x.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                success = False
                for sigma in (0.02, 0.05, 0.10):
                    if time.time() >= deadline - 3.0:
                        break
                    hp = project_feasible(best_x + rng.normal(0.0, sigma, n), n)
                    hc = exact_counts(hp)
                    if hc < best_count:
                        x = hp
                        count = hc
                        best_x = hp.copy()
                        best_count = hc
                        no_improve = 0
                        success = True
                        break
                if not success:
                    break

    return best_x, best_count


def alpha_candidate(h, a, n, alpha):
    ha = 0.5 + alpha * a
    ha = np.clip(ha, 0.0, 1.0)
    if abs(float(ha.sum()) - 0.5 * float(n)) > 1e-8:
        ha = project_feasible(ha, n)
    return ha


def amplitude_search(h, n):
    hp = project_feasible(np.asarray(h, dtype=float), n)
    a = hp - 0.5
    m = float(np.max(np.abs(a)))
    best_count = exact_counts(hp)
    best_x = hp.copy()
    best_alpha = 0.0

    if m < 1e-14:
        return best_x, best_count

    alpha_max = 0.5 / m
    if not np.isfinite(alpha_max):
        alpha_max = 1.0
    if alpha_max > 5.0:
        alpha_max = 5.0

    alphas = np.linspace(-alpha_max, alpha_max, 45)
    for al in alphas:
        al = float(al)
        if abs(al) * m > 0.5 + 1e-10:
            continue
        ha = alpha_candidate(hp, a, n, al)
        c = exact_counts(ha)
        if c < best_count - 1e-13:
            best_count = c
            best_alpha = al
            best_x = ha

    width = max(1e-4, alpha_max / 25.0)
    lo = max(-alpha_max, best_alpha - width)
    hi = min(alpha_max, best_alpha + width)
    for al in np.linspace(lo, hi, 25):
        al = float(al)
        if abs(al) * m > 0.5 + 1e-10:
            continue
        ha = alpha_candidate(hp, a, n, al)
        c = exact_counts(ha)
        if c < best_count - 1e-13:
            best_count = c
            best_alpha = al
            best_x = ha

    final = alpha_candidate(hp, a, n, best_alpha)
    return final, exact_counts(final)


def cyclic_shift_search(h, n):
    x = project_feasible(np.asarray(h, dtype=float), n)
    best_x = x.copy()
    best_c = exact_counts(x)
    if n > 450:
        return best_x, best_c

    for s in range(1, n):
        hs = np.roll(x, s)
        c = correlate_counts(hs, 1.0 - hs)
        if c < best_c - 1e-14:
            best_c = c
            best_x = hs.copy()
    return best_x, best_c


def upsample_h(h, n_target):
    h = np.asarray(h, dtype=float)
    n0 = len(h)
    if n_target == n0:
        return project_feasible(h, n_target)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_target)
    y = np.interp(x1, x0, h)
    return project_feasible(y, n_target)


def repeat_upsample(h, n_target):
    h = np.asarray(h, dtype=float)
    n0 = len(h)
    if n_target == n0:
        return project_feasible(h, n_target)
    k = max(1, int(round(float(n_target) / float(n0))))
    hr = np.repeat(h, k)
    if hr.size != n_target:
        x0 = np.linspace(0.0, 1.0, hr.size)
        x1 = np.linspace(0.0, 1.0, n_target)
        hr = np.interp(x1, x0, hr)
    return project_feasible(hr, n_target)


def make_smooth_start(n, rng):
    x = np.linspace(0.0, 1.0, n)
    p1 = rng.uniform(0.0, 2.0 * np.pi)
    p2 = rng.uniform(0.0, 2.0 * np.pi)
    p3 = rng.uniform(0.0, 2.0 * np.pi)
    h = 0.5 + (
        0.40 * np.sin(2.0 * np.pi * x + p1) +
        0.20 * np.sin(4.0 * np.pi * x + p2) +
        0.10 * np.sin(8.0 * np.pi * x + p3) +
        0.02 * rng.normal(size=n)
    )
    return project_feasible(h, n)


def better_solution(h, n, count, best_h, best_n, best_c):
    c = count * (2.0 / float(n))
    if c < best_c - 1e-15:
        return h.copy(), n, c
    return best_h, best_n, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        b = float(budget_s)
    except Exception:
        b = 1100.0
    if b <= 10.0:
        b = 1100.0
    deadline = t0 + max(30.0, b) - 18.0
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        h_parent = None

    if h_parent is None or h_parent.size < 2:
        n0 = 250
        h_parent = make_smooth_start(n0, rng)
    else:
        n0 = int(h_parent.size)

    best_h = project_feasible(h_parent, n0)
    best_n = n0
    best_count = exact_counts(best_h)
    best_c = best_count * (2.0 / float(best_n))

    for variant in (1.0 - best_h, best_h[::-1]):
        hv = project_feasible(variant, best_n)
        cv = exact_counts(hv)
        best_h, best_n, best_c = better_solution(hv, best_n, cv, best_h, best_n, best_c)

    if time.time() < deadline - 40.0:
        h1, c1 = amplitude_search(best_h, best_n)
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if best_n <= 450 and time.time() < deadline - 60.0:
        h1, c1 = cyclic_shift_search(best_h, best_n)
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if time.time() < deadline - 90.0:
        h1, c1 = optimize_active_subset(
            best_h, best_n, pen=10.0, maxiter=60,
            active_top=12, max_rounds=6, deadline=deadline - 90.0
        )
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    for beta in (4.0, 16.0, 64.0, 256.0):
        if time.time() > deadline - 180.0:
            break
        h1, c1 = optimize_lbfgsb(best_h, best_n, beta, pen=10.0,
                                 maxiter=lbfgs_maxiter(best_n))
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

        if time.time() < deadline - 140.0:
            h1, c1 = optimize_active_subset(
                best_h, best_n, pen=10.0, maxiter=35,
                active_top=10, max_rounds=3, deadline=deadline - 140.0
            )
            best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if best_n <= 700 and time.time() < deadline - 70.0:
        end1 = min(deadline - 70.0, time.time() + 240.0)
        h1, c1 = active_search(best_h, best_n, end1,
                               max_candidates=25, active_top=15, rng=rng)
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if best_n < 800 and time.time() < deadline - 150.0:
        n_high = min(max(2 * best_n, 512), 800)
        starts = (upsample_h(best_h, n_high), repeat_upsample(best_h, n_high))
        for st in starts:
            if time.time() > deadline - 110.0:
                break
            st = project_feasible(st, n_high)
            c0 = exact_counts(st)
            best_h, best_n, best_c = better_solution(st, n_high, c0, best_h, best_n, best_c)

            for beta in (8.0, 32.0, 128.0):
                if time.time() > deadline - 100.0:
                    break
                h1, c1 = optimize_lbfgsb(st, n_high, beta, pen=10.0,
                                         maxiter=lbfgs_maxiter(n_high))
                if c1 < c0:
                    st, c0 = h1, c1
                best_h, best_n, best_c = better_solution(h1, n_high, c1, best_h, best_n, best_c)

            if time.time() < deadline - 95.0:
                h1, c1 = optimize_active_subset(
                    st, n_high, pen=10.0, maxiter=30,
                    active_top=8, max_rounds=3, deadline=deadline - 95.0
                )
                best_h, best_n, best_c = better_solution(h1, n_high, c1, best_h, best_n, best_c)

            if time.time() < deadline - 85.0:
                h1, c1 = amplitude_search(st, n_high)
                best_h, best_n, best_c = better_solution(h1, n_high, c1, best_h, best_n, best_c)

    if best_n <= 700 and time.time() < deadline - 20.0:
        h1, c1 = active_search(best_h, best_n, deadline - 8.0,
                               max_candidates=35, active_top=20, rng=rng)
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if time.time() < deadline - 8.0:
        h1, c1 = optimize_active_subset(
            best_h, best_n, pen=10.0, maxiter=40,
            active_top=10, max_rounds=3, deadline=deadline - 4.0
        )
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    if time.time() < deadline - 2.0:
        h1, c1 = amplitude_search(best_h, best_n)
        best_h, best_n, best_c = better_solution(h1, best_n, c1, best_h, best_n, best_c)

    best_h = project_feasible(best_h, best_n)
    best_c = exact_counts(best_h) * (2.0 / float(best_n))
    return best_h, float(best_c), int(best_n)
