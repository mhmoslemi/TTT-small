def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    target = n / 2.0
    sd = x.sum() - target
    penalty = mu * sd * sd
    
    cmax = c.max()
    cshift = alpha * (c - cmax)
    ec = np.exp(cshift)
    w = ec / ec.sum()
    
    fval = cmax + np.log(ec.sum()) / alpha + penalty
    
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        p = n - 1 + l
        wp = w[p]
        if wp < 1e-16:
            continue
        # (1-x[i-l]) contribution
        ilo = max(0, l)
        ihi = min(n-1, l + n - 1)
        if ilo <= ihi:
            grad[ilo:ihi+1] += wp * (1.0 - x[ilo-l:ihi-l+1])
        # -x[i+l] contribution
