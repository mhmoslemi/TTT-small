def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = np.exp(shifted).sum()
    return float(m + np.log(s) / alpha)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    p = np.exp(shifted)
    p /= p.sum()
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        pi = p[i]
        if pi < 1e-18:
            continue
        if s == 0:
            grad += pi * (1.0 - 2.0 * x)
        elif s > 0:
            grad[s:] += pi * (1.0 - x[:n - s])
            grad[:n - s] -= pi * x[s:]
        else:
            t = -s
            grad[:n - t] += pi * (1.0 - x[t:])
            grad[t:] -= pi * x[:n - t]
    return grad

    return x.sum() - n / 2.0

    return np.ones(n)

def slsqp_with_grad(h, n, alpha, maxiter=3000):

    try:

                      jac=lambda x: smooth_grad(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    s = np.exp(shifted).sum()
    return m + np.log(s) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    p = np.exp(shifted)
    p_sum = p.sum()
    p /= p_sum
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        pi = p[i]
        if pi < 1e-18:
            continue
        if s == 0:
            grad += pi * (1.0 - 2.0 * x)
        elif s > 0:
            grad[s:] += pi * (1.0 - x[:n - s])
            grad[:n - s] -= pi * x[s:]
        else:
            t = -s
            grad[:n - t] += pi * (1.0 - x[t:])
            grad[t:] -= pi * x[:n - t]
    return grad

def slsqp_con_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha),
                      jac=lambda x: smooth_grad(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_refine(h, n, deadline, max_passes=1000):
    """Targeted pair search focusing on active shifts."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d_scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            c_vals = compute_c_full(h, n)
            max_val = c_vals.max()
            threshold = max_val - 0.02  # focus on shifts within 0.02 of max
            active = np.where(c_vals >= threshold)[0]
            
            if len(active) == 0:
                active = np.array([np.argmax(c_vals)])
            
            improved = False
            for m in active:
                if time.time() > deadline:
                    break
                s = int(m) - (n - 1)
                # Gradient of c[m]
                grad = np.zeros(n)
                if s == 0:
                    grad[:] = 1.0 - 2.0 * h
                elif s > 0:
                    grad[s:] = (1.0 - h[:n-s])
                    grad[:n-s] = -h[s:]
                else:
                    t = -s
                    grad[:n-t] = (1.0 - h[t:])
                    grad[t:] = -h[:n-t]
                
                # We want to increase c_reduction = -delta * (grad[i] - grad[j])
                # i.e., decrease c[m] by moving h[i] up and h[j] down
                # Best: i = argmin(grad), j = argmax(grad)
                # But we need to consider ALL shifts, not just m
                
                # Try several candidate pairs
                candidates = []
                for rank in range(5):
                    idx_neg = np.argsort(grad)[:5]
                    idx_pos = np.argsort(grad)[-5:][::-1]
                    for ii in idx_neg[:max(1, rank+1)]:
                        for jj in idx_pos[:max(1, rank+1)]:
                            if ii != jj:
                                candidates.append((int(ii), int(jj)))
                
                for (i, j) in candidates[:20]:
                    if h[i] >= 1.0 or h[j] <= 0.0:
                        continue
                    max_delta = min(1.0 - h[i], h[j]) * d_scale
                    if max_delta < 1e-10:
                        continue
                    h[i] += max_delta
                    h[j] -= max_delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[i] -= max_delta
                        h[j] += max_delta
                
                if improved:
                    break
            
            if not improved:
                break
    
    return h, best_c
