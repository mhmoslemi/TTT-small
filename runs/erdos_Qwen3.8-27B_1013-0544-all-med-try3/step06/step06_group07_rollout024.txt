def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        diff = target - float(v.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
        else:
            idx = np.where(v > 0.0)[0]
        if idx.size == 0:
            break
        v[idx] = v[idx] + diff / float(idx.size)
        v = np.clip(v, 0.0, 1.0)
    diff = target - float(v.sum())
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size:
                j = idx[np.argmax(1.0 - v[idx])]
                v[j] += min(diff, 1.0 - v[j])
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size:
                j = idx[np.argmax(v[idx])]
                v[j] += max(diff, -v[j])
    return v
```

import numpy as np
import time
from scipy.optimize import minimize, Bounds, differential_evolution


def time_left(t0, budget):
    return float(budget) - (time.time() - t0)


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        diff = target - float(v.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = np.where(v < 1.0)[0]
        else:
            idx = np.where(v > 0.0)[0]
        if idx.size == 0:
            break
        v[idx] = v[idx] + diff / float(idx.size)
        v = np.clip(v, 0.0, 1.0)

    diff = target - float(v.sum())
    if abs(diff) > 1e-15:
        if diff > 0.0:
            idx = np.where(v < 1.0)[0]
            if idx.size:
                j = idx[np.argmax(1.0 - v[idx])]
                v[j] += min(diff, 1.0 - v[j])
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size:
                j = idx[np.argmax(v[idx])]
                v[j] += max(diff, -v[j])
    return v


def build_h_sym(a, n):
    m = n // 2
    h = np.empty(n, dtype=float)
    h[:m] = a
    h[m:] = a[::-1]
    return h


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def c5_sym_a(a, n):
    return compute_c5(build_h_sym(a, n))


def eval_exact(x, n, mode):
    if mode == 0:
        return compute_c5(x)
    return c5_sym_a(x, n)


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float)
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def to_sym_a(h, n):
    if n % 2 != 0:
        n += 1
    if len(h) != n:
        h = resample_h(h, n)
    else:
        h = np.asarray(h, dtype=float).copy()
    h = 0.5 * (h + h[::-1])
    m = n // 2
    a = h[:m]
    return project_box_sum(a, 0.25 * n)


def default_h(n=80):
    if n % 2 != 0:
        n += 1
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    u = np.sin(1.5 * np.pi * x) - (1.0 / 3.0) * np.sin(0.5 * np.pi * x)
    norm = float(np.max(np.abs(u)))
    if norm > 1e-15:
        u /= norm
    else:
        u = np.zeros(n)
    h = 0.5 + 0.75 * u
    h = np.clip(h, 0.0, 1.0)
    h = 0.5 * (h + h[::-1])
    return project_box_sum(h, 0.5 * n)


def soft_stats(corr, beta):
    m = float(np.max(corr))
    z = np.clip(beta * (corr - m), -700.0, 700.0)
    sumexp = float(np.exp(z).sum())
    val = m + np.log(sumexp) / beta
    w = np.exp(z) / sumexp
    return val, w


def single_lag_grad(h, s):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if s == 0:
        g = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        g[:m] -= h[s:]
        g[s:] += 1.0 - h[:m]
    else:
        l = -s
        m = n - l
        g[:m] += 1.0 - h[l:]
        g[l:] -= h[:m]
    return g


def corr_grad_weighted_full(h, weights, tol=1e-12):
    n = len(h)
    g = np.zeros(n, dtype=float)
    off = n - 1

    w0 = weights[off]
    if w0 > tol:
        g += w0 * (1.0 - 2.0 * h)

    for s in range(1, n):
        wp = weights[off + s]
        if wp > tol:
            m = n - s
            g[:m] -= wp * h[s:]
            g[s:] += wp * (1.0 - h[:m])

        wn = weights[off - s]
        if wn > tol:
            m = n - s
            g[:m] += wn * (1.0 - h[s:])
            g[s:] -= wn * h[:m]

    return g


def corr_grad_weighted_nonneg(h, weights, tol=1e-12):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for s in range(n):
        w = weights[s]
        if w <= tol:
            continue
        if s == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            m = n - s
            g[:m] -= w * h[s:]
            g[s:] += w * (1.0 - h[:m])
    return g


def con_sum(x, target):
    return float(np.sum(x) - target)


def con_sum_jac(x, target):
    return np.ones(len(x), dtype=float)


def slsqp_obj_only(x, n, target, beta, mode):
    if mode == 0:
        h = np.asarray(x, dtype=float)
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr
    else:
        h = build_h_sym(x, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr[n - 1:n + n - 1]

    val, _ = soft_stats(vals, float(beta))
    return float(val * (2.0 / n))


def slsqp_grad(x, n, target, beta, mode):
    if mode == 0:
        h = np.asarray(x, dtype=float)
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr
        val, w = soft_stats(vals, float(beta))
        gh = corr_grad_weighted_full(h, w)
        g = (2.0 / n) * gh
    else:
        h = build_h_sym(x, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr[n - 1:n + n - 1]
        val, w = soft_stats(vals, float(beta))
        gh = corr_grad_weighted_nonneg(h, w)
        m = len(x)
        g = (2.0 / n) * (gh[:m] + gh[m:][::-1])
    return g


def slsqp_refine(x0, n, target, mode, beta_list, t0, budget, max_iter):
    x = project_box_sum(x0, target)
    best = x.copy()
    best_c = eval_exact(x, n, mode)

    if max_iter <= 0 or len(x) == 0:
        return best, best_c

    bounds = Bounds(np.zeros(len(x)), np.ones(len(x)))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac, 'args': (target,)}]
    opts = {'maxiter': int(max_iter), 'ftol': 1e-12, 'iprint': 0}

    for beta in beta_list:
        if time_left(t0, budget) < 35:
            break
        for trial in range(2):
            if time_left(t0, budget) < 28:

            if trial == 1:

                    start,

                    constraints=cons,
                    options=opts

                res = None

def make_basis(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    basis = []

    for k in range(1, 5):
        m = 2 * k + 1
        b = np.sin(m * np.pi * x / 2.0) - (1.0 / float(m)) * np.sin(0.5 * np.pi * x)
        b = b - b.mean()
        norm = float(np.max(np.abs(b)))
        if norm < 1e-14:
            norm = 1.0
        basis.append(b / norm)

    for j in (1, 2):
        b = np.cos(j * np.pi * (x - 1.0))
        b = b - b.mean()
        norm = float(np.max(np.abs(b)))
        if norm < 1e-14:
            norm = 1.0
        basis.append(b / norm)

    return basis


def de_build(coeffs, basis, n):

        u += float(c) * b
    h = 0.5 + u
    h = np.clip(h, 0.0, 1.0)

def mm_con_fun(z, n, mode, idx):

    return J

            gh = corr_grad_weighted_nonneg(h, w)
            m = len(x)
            grad = gh[:m] + gh[m:][::-1]

        grad = grad - grad.mean()
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break

        step = float(base_step) / (1.0 + 0.0012 * float(it))
        best_cand = x
        best_cand_c = cur_c + 1e-15

        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = project_box_sum(x - step * factor * grad / gnorm, target)
            c = eval_exact(cand, n, mode)
            if c < best_cand_c:
                best_cand_c = c
                best_cand = cand

        x = best_cand
        cur_c = best_cand_c

        improved = cur_c < best_c - 1e-13
        if improved:
            best = x.copy()
            best_c = cur_c
            stall = 0
        else:
            stall += 1
            if stall > 90:
                x = x + 0.004 * np.random.randn(len(x))
                x = project_box_sum(x, target)
                cur_c = eval_exact(x, n, mode)
                stall = 0

    if cur_c < best_c:
        best = x.copy()
        best_c = cur_c

    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    budget = float(budget_s)

    iv = globals().get('initial_h_values')
    h_init = None
    if iv is not None:
        try:
            if callable(iv):
                iv = iv()
            arr = np.asarray(iv, dtype=float)
            if arr.size > 0:
                h_init = arr.reshape(-1).astype(float).copy()
        except Exception:
            h_init = None

    if h_init is None or len(h_init) < 8:
        h_init = default_h(80)

    h_init = project_box_sum(h_init, 0.5 * len(h_init))
    best_h = h_init.copy()
    best_c = compute_c5(best_h)
    best_n = len(best_h)

    if best_n % 2 == 0:
        a0 = to_sym_a(best_h, best_n)
        c0 = eval_exact(a0, best_n, 1)
        if c0 < best_c:
            best_h = build_h_sym(a0, best_n)
            best_c = c0

    if best_n % 2 == 0 and time_left(t0, budget) > 70:
        bmax = 120 if best_n <= 80 else 70
        betas = [20.0, 80.0, 300.0, 1200.0]
        a_try = to_sym_a(best_h, best_n)
        a_try, c_try = slsqp_refine(a_try, best_n, 0.25 * best_n, 1, betas, t0, budget, bmax)
        if c_try < best_c:
            best_h = build_h_sym(a_try, best_n)
            best_c = c_try

    if time_left(t0, budget) > 55:
        bmax = 90 if best_n <= 80 else 60
        betas = [30.0, 150.0, 600.0]
        x_try = project_box_sum(best_h, 0.5 * best_n)
        x_try, c_try = slsqp_refine(x_try, best_n, 0.5 * best_n, 0, betas, t0, budget, bmax)
        if c_try < best_c:
            best_h = x_try
            best_c = c_try
            best_n = len(x_try)

    if time_left(t0, budget) > 170:
        n_de = 128
        if best_n >= 160:
            n_de = 160
        basis = make_basis(n_de)
        bounds = [(-0.8, 0.8)] * len(basis)
        maxde = 60 if time_left(t0, budget) > 240 else 35

        try:
            res_de = differential_evolution(
                de_obj,
                bounds,
                args=(basis, n_de),
                maxiter=int(maxde),
                popsize=10,
                tol=1e-6,
                seed=int(seed),
                polish=False
            )
            if hasattr(res_de, 'x'):
                h_de = de_build(res_de.x, basis, n_de)
                c_de = compute_c5(h_de)
                if c_de < best_c:
                    best_h = h_de
                    best_c = c_de
                    best_n = n_de

                a_de = to_sym_a(h_de, n_de)
                bmax = 100 if n_de <= 128 else 70
                a_de, c_de = slsqp_refine(a_de, n_de, 0.25 * n_de, 1,
                                          [50.0, 200.0, 800.0], t0, budget, bmax)
                if c_de < best_c:
                    best_h = build_h_sym(a_de, n_de)
                    best_c = c_de
                    best_n = n_de
        except Exception:
            pass

    n_candidates = []
    raw_candidates = [
        best_n, best_n + 8, 96, 112, 128, 144, 160, 176, 192, best_n + 16, 208
    ]
    for nt in raw_candidates:
        nt = int(nt)
        if nt % 2 == 0 and 40 <= nt <= 220 and nt not in n_candidates:
            n_candidates.append(nt)

    for n_try in n_candidates[:5]:
        if time_left(t0, budget) < 75:
            break
        if n_try == best_n and best_c < 0.38085:
            continue

        h_try = resample_h(best_h, n_try)
        a_try = to_sym_a(h_try, n_try)
        bmax = 90 if n_try <= 128 else 60
        if n_try <= 128:
            betas = [50.0, 200.0, 800.0, 3000.0]
        else:
            betas = [100.0, 400.0, 1500.0]

        a_new, c_new = slsqp_refine(a_try, n_try, 0.25 * n_try, 1, betas, t0, budget, bmax)
        if c_new < best_c:
            best_h = build_h_sym(a_new, n_try)
            best_c = c_new
            best_n = n_try

    if time_left(t0, budget) > 70 and best_n > 0:
        mode_mm = 0
        x_mm = project_box_sum(best_h, 0.5 * best_n)

        if best_n % 2 == 0:
            a_mm = to_sym_a(best_h, best_n)
            c_mm0 = eval_exact(a_mm, best_n, 1)
            if c_mm0 <= best_c + 1e-7:
                mode_mm = 1
                x_mm = a_mm

        if mode_mm == 0:
            cap = 64 if best_n > 120 else 96
            k_list = []
            for k in (48, cap):
                kk = int(min(k, 2 * best_n - 1))
                if kk > 0 and kk not in k_list:
                    k_list.append(kk)
            if best_n <= 80 and time_left(t0, budget) > 120:
                k_all = 2 * best_n - 1
                if k_all not in k_list:
                    k_list.append(k_all)
        else:
            cap = 64 if best_n > 120 else 96
            k_list = []
            for k in (48, cap):
                kk = int(min(k, best_n))
                if kk > 0 and kk not in k_list:
                    k_list.append(kk)
            if best_n <= 110 and time_left(t0, budget) > 120:
                k_all = best_n
                if k_all not in k_list:
                    k_list.append(k_all)

        if k_list:
            x_mm, c_mm = minmax_refine(x_mm, best_n, mode_mm, t0, budget, k_list)
            h_mm = x_mm if mode_mm == 0 else build_h_sym(x_mm, best_n)
            c_mm = compute_c5(h_mm)
            if c_mm < best_c:
                best_h = project_box_sum(h_mm, 0.5 * best_n)
                best_c = compute_c5(best_h)

            if time_left(t0, budget) > 60:
                other_mode = 1 if mode_mm == 0 else 0
                if other_mode == 0:
                    x_o = project_box_sum(best_h, 0.5 * best_n)
                    k_o = [int(min(48, 2 * best_n - 1))]
                else:
                    x_o = to_sym_a(best_h, best_n)
                    k_o = [int(min(64, best_n))]
                x_o, c_o = minmax_refine(x_o, best_n, other_mode, t0, budget, k_o)
                h_o = x_o if other_mode == 0 else build_h_sym(x_o, best_n)
                c_o = compute_c5(h_o)
                if c_o < best_c:
                    best_h = project_box_sum(h_o, 0.5 * best_n)
                    best_c = compute_c5(best_h)

    if time_left(t0, budget) > 40:
        x_f = project_box_sum(best_h, 0.5 * best_n)
        x_f, c_f = subgrad_refine(x_f, best_n, 0, t0, budget, max_iter=350, base_step=0.02)
        if c_f < best_c:
            best_h = x_f
            best_c = c_f

    if best_n % 2 == 0 and time_left(t0, budget) > 40:
        x_s = to_sym_a(best_h, best_n)
        x_s, c_s = subgrad_refine(x_s, best_n, 1, t0, budget, max_iter=350, base_step=0.02)
        if c_s < best_c:
            best_h = build_h_sym(x_s, best_n)
            best_c = c_s

    if time_left(t0, budget) > 25:
        x_f = project_box_sum(best_h, 0.5 * best_n)
        x_f, c_f = slsqp_refine(x_f, best_n, 0.5 * best_n, 0, [2000.0, 8000.0], t0, budget, 45)
        if c_f < best_c:
            best_h = x_f
            best_c = c_f

    if best_n % 2 == 0 and time_left(t0, budget) > 20:
        x_s = to_sym_a(best_h, best_n)
        x_s, c_s = slsqp_refine(x_s, best_n, 0.25 * best_n, 1, [2000.0, 8000.0], t0, budget, 45)
        if c_s < best_c:
            best_h = build_h_sym(x_s, best_n)
            best_c = c_s

    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
