            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, alphas=None, mu=1.0):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 500000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        t_alloc = min(40, deadline - time.time() - 2)
        if t_alloc < 5:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 2000,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_gradient_descent(h, n, deadline, alpha_start=50000, lr_start=0.002):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = alpha_start
    lr = lr_start
    stall_count = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            step = lr * factor
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            stall_count += 1
            lr *= 0.5
            alpha *= 1.5
            if lr < 1e-14:
                break
        else:
            stall_count = max(0, stall_count - 1)
            if stall_count == 0 and lr < 0.01:
                lr *= 1.1
    return best_h, best_c

def gradient_pair_search(h, n, deadline, max_radius=40):
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000]
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for alpha in alphas:
        g = smooth_max_grad(h, n, alpha)
        for d in deltas:
            while time.time() < deadline:
                improved = False
                for i in range(n):
                    if time.time() > deadline:
                        return h, best_c
                    lo = max(0, i - max_radius)
                    hi = min(n, i + max_radius + 1)
                    for j in range(lo, hi):
                        if i == j:
                            continue
                        move = -d * np.sign(g[i] - g[j])
                        if abs(move) < 1e-12:
                            continue
                        if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                           h[j] - move >= 0.0 and h[j] - move <= 1.0:
                            h[i] += move
                            h[j] -= move
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= move
                                h[j] += move
                if not improved:
                    break
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=30):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def active_set_search(h, n, deadline):
    """Focus on the shifts that are at or near the maximum."""
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        c = compute_corr(h, n)
        c_max = c.max()
        # Find shifts within 0.5% of max
        threshold = c_max - 0.005 * c_max
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient of the sum of active overlaps
        dx = 2.0 / n
        grad = np.zeros(n)
        for m_idx in active:
            k = m_idx - (n - 1)
            weight = (c[m_idx] - threshold) / (0.005 * c_max + 1e-15)
            weight = weight * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += weight * (1.0 - h[k:n])
                grad[k:n] += weight * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += weight * (1.0 - h[0:nv])
                grad[0:nv] += weight * (-h[kk:n])
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        
        # Try small moves in the -gradient direction
        improved = False
        for lr in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                improved = True
                break
        
        if not improved:
            # Try individual coordinate moves guided by gradient
            grad_sorted = np.argsort(grad)
            tried = 0
            for rank in range(n - 1, -1, -1):
                j = grad_sorted[rank]  # high gradient
                for rank2 in range(0, min(30, n)):
                    i = grad_sorted[rank2]  # low gradient
                    if i == j:
                        continue
                    d = 0.0005
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    d = 0.0002
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    tried += 1
                    if tried > 3000:
                        break
                if tried > 3000:
                    break
            
            if not improved:
                break
    
    return h, best_c

def make_sine_init(n, freq, phase):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_binary_init(n, rng):
    num_ones = n // 2
    h = np.zeros(n)
    positions = rng.choice(n, size=num_ones, replace=False)
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: L-BFGS on parent resolution
    t1_end = min(t0 + 120, deadline - 400)
    if time.time() < t1_end:
        best_h, best_c = lbfgs_optimize(best_h, n_init, t1_end)
    
    # Phase 2: Fine gradient descent
    t2_end = min(time.time() + 100, deadline - 300)
    if time.time() < t2_end:
        h2, c2 = fine_gradient_descent(best_h, n_init, t2_end, alpha_start=50000, lr_start=0.001)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Gradient-guided pair search
    t3_end = min(time.time() + 100, deadline - 200)
    if time.time() < t3_end:
        best_h, best_c = gradient_pair_search(best_h, n_init, t3_end, max_radius=30)
    
    # Phase 4: Active set search
    t4_end = min(time.time() + 80, deadline - 150)
    if time.time() < t4_end:
        best_h, best_c = active_set_search(best_h, n_init, t4_end)
    
    # Phase 5: Fine pair search with very small steps
    t5_end = min(time.time() + 100, deadline - 80)
    if time.time() < t5_end:
        best_h, best_c = pair_search(best_h, n_init, t5_end,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                     radius=25)
    
    # Phase 6: Try alternative resolutions
    x_old = np.linspace(0, 2, len(best_h))
    for n_try in [250, 350, 400]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(60, deadline - time.time() - 50)
        if alloc < 30:
            break
        
        h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5,
                                   alphas=[1000, 10000, 50000, 200000])
        if time.time() < deadline - 30:
            h_t, c_t = fine_gradient_descent(h_t, n_try, time.time() + min(20, deadline - time.time() - 20),
                                             alpha_start=50000, lr_start=0.001)
        if time.time() < deadline - 20:
            h_t, c_t = pair_search(h_t, n_try, time.time() + min(15, deadline - time.time() - 15),
                                    deltas=[0.001, 0.0005, 0.0002, 0.0001], radius=15)
        
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    
    # Phase 7: Try a few diverse initializations at best resolution
    n_best = len(best_h)
    diverse_starts = []
    for freq in [2, 3, 4, 6]:
        diverse_starts.append(make_sine_init(n_best, freq, 0))
        diverse_starts.append(make_sine_init(n_best, freq, np.pi / 2))
    for _ in range(3):
        diverse_starts.append(make_binary_init(n_best, rng))
    
    remaining = deadline - time.time() - 20
    per_start = max(5, remaining / len(diverse_starts))
    
    for ds in diverse_starts:
        if time.time() > deadline - 15:
            break
        c_try = compute_c5(ds, n_best)
        if c_try < best_c:
            best_c = c_try
            best_h = ds
            continue
        t_alloc = min(per_start, deadline - time.time() - 10)
        if t_alloc < 5:
            continue
        h_t, c_t = lbfgs_optimize(ds, n_best, time.time() + t_alloc * 0.6,
                                   alphas=[1000, 10000, 50000])
        if c_t < best_c and time.time() < deadline - 10:
            h_t, c_t = fine_gradient_descent(h_t, n_best, deadline - 10,
                                             alpha_start=50000, lr_start=0.0005)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    
    # Phase 8: Final intensive polish
    n_final = len(best_h)
    if time.time() < deadline - 3:
        # Very fine gradient descent
        t_final = deadline - 2
        h_f, c_f = fine_gradient_descent(best_h, n_final, t_final,
                                         alpha_start=200000, lr_start=0.0003)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    # Final pair search with tiny steps
    if time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 0.5,
                                     deltas=[0.0001, 0.00005, 0.00002],
                                     radius=15)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
