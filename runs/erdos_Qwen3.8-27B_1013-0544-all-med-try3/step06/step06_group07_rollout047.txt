Let me think about the exact coordinate update:
For h[j], the objective is max_k f_k(h_j) where each f_k is linear in h_j.
We want to find h_j ∈ [0,1] that minimizes max_k f_k(h_j).

For a set of linear functions, the minimax point is either:
- Where two lines cross (and the maximum is achieved)
- At a boundary (h_j = 0 or h_j = 1)

Actually, for min max of linear functions, the optimum is where the upper envelope is minimized. This is a 1D linear programming problem:
minimize t
subject to: a_k * h_j + b_k ≤ t for all k
           0 ≤ h_j ≤ 1

This can be solved by finding the point where the upper envelope of the lines is minimized.

The upper envelope is a convex piecewise-linear function. Its minimum is at a "kink" point (intersection of two lines) or at a boundary.

We can solve this by:
1. Computing all the lines f_k(h_j) = a_k * h_j + b_k
2. Finding the minimum of the upper envelope

For efficiency, we can just evaluate at all intersection points of pairs of lines (within [0,1]) and at boundaries, then pick the best. But with O(n) lags, that's O(n²) per coordinate, and O(n³) total per sweep. For n=100, that's 10^6 operations per sweep - very fast.

Actually, a simpler approach: bisection on t. For a given t, check if there exists h_j ∈ [0,1] such that all a_k * h_j + b_k ≤ t. This is a set of linear inequalities in one variable, which can be checked in O(n). Then binary search on t for O(log(1/eps)) iterations.

Even simpler: the lines are all of the form a_k * h_j + b_k. The upper envelope minimum can be found by:
- The constraint a_k * h_j + b_k ≤ t for all k means h_j ≤ (t - b_k)/a_k for a_k > 0 and h_j ≥ (t - b_k)/a_k for a_k < 0.
- So for a given t, the feasible range for h_j is [max_{a_k<0} (t-b_k)/a_k, min_{a_k>0} (t-b_k)/a_k] ∩ [0,1].
- We want the smallest t such that this range is non-empty.

This gives us an exact 1D binary search.

                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    return best_h, best_c5


def subgradient_refine(h, n, time_fn, n_iters=20000):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr_base = 0.003

    for it in range(n_iters):
        if time_fn() < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
        lr = lr_base / (1.0 + it * 0.001)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        if it % 5000 == 4999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(100, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at initial resolution
    n_init = len(h_init)
    if tl() > 60:
        h_cd, c5_cd = coord_descent(h_init, n_init, tl, max_sweeps=50000)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 2: Try higher resolutions via resampling
    resolutions = [100, 150, 200, 300]
    for n_try in resolutions:
        if tl() < 80:
            break
        if n_try == n_init:
            h_try = best_h.copy()
        else:
            h_try = resample_h(best_h, n_try)
        h_try = project_feasible(h_try, n_try / 2.0)

        # Coordinate descent
        h_cd, c5_cd = coord_descent(h_try, n_try, tl, max_sweeps=200000)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

        # Subgradient refinement
        if tl() > 40:
            h_sg, c5_sg = subgradient_refine(h_cd, n_try, tl, n_iters=30000)
            if c5_sg < best_c5:
                best_h = h_sg.copy()
                best_c5 = c5_sg

        # Another CD pass after subgradient
        if tl() > 40:
            h_cd2, c5_cd2 = coord_descent(best_h, n_try, tl, max_sweeps=100000)
            if c5_cd2 < best_c5:
                best_h = h_cd2.copy()
                best_c5 = c5_cd2

    # Phase 3: Try different starting points at best resolution
    n_best = len(best_h)
    if tl() > 60:
        starts = []
        # Sinusoidal start
        x = np.linspace(0, 2, n_best, endpoint=False)
        h_sin = 0.5 + 0.3 * np.cos(2 * np.pi * x)
        h_sin = project_feasible(h_sin, n_best / 2.0)
        starts.append(h_sin)
        # Random
        h_rand = np.random.rand(n_best)
        h_rand = project_feasible(h_rand, n_best / 2.0)
        starts.append(h_rand)
        # Block pattern
        n_blocks = max(2, n_best // 20)
        h_block = np.zeros(n_best)
        for b in range(n_blocks):
            s = b * n_best // n_blocks
            e = (b + 1) * n_best // n_blocks
            if b % 2 == 0:
                h_block[s:e] = 1.0
        h_block = project_feasible(h_block, n_best / 2.0)
        starts.append(h_block)

        for h_start in starts:
            if tl() < 40:
                break
            h_cd, c5_cd = coord_descent(h_start, n_best, tl, max_sweeps=100000)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd

    # Phase 4: SLSQP polish on best solution
    if tl() > 30:
        h_pol, c5_pol = slsqp_polish(best_h, len(best_h), tl)
        if c5_pol < best_c5:
            best_h = h_pol.copy()
            best_c5 = c5_pol

    # Phase 5: Final CD polish
    if tl() > 20:
        h_final, c5_final = coord_descent(best_h, len(best_h), tl, max_sweeps=50000)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
