import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_smooth_fn(h, dx, beta, target):
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    sm = smooth_max_val(corr, beta)
    pen = 100.0 * (h_c.sum() - target) ** 2
    return sm * dx + pen


def sum_constraint(h, target):
    return h.sum() - target


def sum_constraint_jac(h, target):
    return np.ones(len(h))


def binary_sim_anneal(n, rng, max_iter, t_start, t_end, time_check):
    pos = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones_set = set(pos.tolist())
    
    for it in range(max_iter):
        if time_check() < 90:
            break
        frac = it / float(max_iter)
        T = t_start * (t_end / t_start) ** frac
        oi = rng.randint(n)
        zj = rng.randint(n)
        if oi == zj:
            continue
        oi_one = oi in ones_set
        zj_one = zj in ones_set
        if oi_one == zj_one:
            continue
        h[oi], h[zj] = h[zj], h[oi]
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T) if delta > 0 else True):
            c5_cur = c5_new
            if h[oi] == 1.0:
                ones_set.add(oi)
                ones_set.discard(zj)
            else:
                ones_set.add(zj)
                ones_set.discard(oi)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[oi], h[zj] = h[zj], h[oi]
    return h_best, c5_best


def cobyla_run(h_start, n, target, maxiter):
    bounds_list = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h_start, method='COBYLA',
                       bounds=bounds_list,
                       constraints=[{'type': 'eq', 'fun': sum_constraint,
                                     'args': (target,)}],
                       options={'maxiter': maxiter, 'rhobeg': 0.05,
                                'rhoend': 1e-8, 'catol': 1e-10})
        h_new = project_to_sum(res.x, target)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_start.copy(), compute_c5(h_start)


def slsqp_run(h_start, n, target, dx, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
             'args': (target,)}]
    try:
        res = minimize(obj_smooth_fn, h_start, args=(dx, beta, target),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_to_sum(res.x, target)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_start.copy(), compute_c5(h_start)


def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)
