            if np.any(valid1):
                a[valid1] = 1.0 - h[idx1[valid1]]
            if np.any(valid2):
                a[valid2] += -h[idx2[valid2]]
            b = c - a * h[j]
            t_new = _solve_1d(a, b)
            if t_new != h[j]:
                delta = t_new - h[j]
                c += delta * a
                h[j] = t_new
        if sweep % 50 == 0:
            cur = c.max()
            if cur < best_c:
                best_c = cur
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                h = project_feasible(h, n)
                c = np.correlate(h, 1 - h, mode='full')
                no_improve = 0
        else:
            if sweep % 200 == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1 - h, mode='full')
    h = project_feasible(h, n)
    c = np.correlate(h, 1 - h, mode='full')
    best_c = c.max()
    return h, best_c

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, deadline, alpha=50000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new
    except Exception:
        pass
    return h, compute_c5(h, n)

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(0, n - m, max(1, (n - m) // 500)):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-8:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Coordinate descent (main workhorse)
    cd_deadline = min(deadline - 30, time.time() + (deadline - time.time()) * 0.6)
    h_cd, c_cd = coord_descent(best_h, n, cd_deadline, rng=rng)
    if c_cd * (2.0/n) < best_c:
        best_c = c_cd * (2.0/n)
        best_h = h_cd
    # SLSQP refinement
    if time.time() < deadline - 20:
        h_s, c_s = slsqp_refine(best_h, n, deadline, alpha=100000, maxiter=2000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
        if time.time() < deadline - 15:
            h_s, c_s = slsqp_refine(best_h, n, deadline, alpha=500000, maxiter=3000)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    # Local search
    if time.time() < deadline - 5:
        h_ls, c_ls = local_search(best_h, n, min(deadline - 1, time.time() + 30))
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = None
    best_c = float('inf')
    # First optimize at initial resolution
    alloc1 = min(200, (deadline - t0) * 0.2)
    h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt
    # Try higher resolutions
    for n_try in [500, 700, 1000, 1400]:
        if time.time() > deadline - 80:
            break
        alloc = min(250, (deadline - time.time()) * 0.35)
        if alloc < 40:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Restart with perturbations at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_final, c_final = slsqp_refine(best_h, n, deadline - 2, alpha=1000000, maxiter=5000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        h_final, c_final = local_search(best_h, n, deadline - 1)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
