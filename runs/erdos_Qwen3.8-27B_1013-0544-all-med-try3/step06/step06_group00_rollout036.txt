```python
def nesterov_smooth(h, n, deadline, alpha_start=100, alpha_end=100000, n_iters=100000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    h_v = h.copy()  # current point
    h_y = h.copy()  # Nesterov y-point
    
    beta = 0.9  # momentum
    eta = 0.001  # step size
    
    for t in range(n_iters):
        if time.time() > deadline:
            break
        
        # Evaluate at y
        g_y = np.correlate(h_y, 1 - h_y, mode='full') * dx
        m = g_y.max()
        shifted = alpha * (g_y - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute gradient at y
        grad = np.zeros(n)
        active = np.where(weights > 1e-6)[0]
        for k_idx in active:
            k = k_idx - (n - 1)  # actual lag
            for i in range(n):
                im = i - k
                ip = i + k
                val = 1.0
                if 0 <= im < n:
                    val -= h_y[im]
                if 0 <= ip < n:
                    val -= h_y[ip]
                grad[i] += weights[k_idx] * val
        
        # Nesterov update
        h_new = h_v - eta * grad
        h_new = project_feasible(h_new, n)
        
        # Update y
        beta_new = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        h_y = h_new + ((beta - 1) / beta_new) * (h_new - h_v)
        h_y = np.clip(h_y, 0, 1)
        
        h_v = h_new
        beta = beta_new
        
        # Check improvement
        c = compute_c5(h_v, n)
        if c < best_c:
            best_c = c
            best_h = h_v.copy()
        
        # Adaptive step size / alpha
        eta *= 0.99999
        alpha *= 1.00001
    
    return best_h, best_c
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_values(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, k, n):
    """Compute gradient of g_k with respect to h, where g_k = sum_i h[i+k](1-h[i]) dx"""
    # g_k = sum_i h[i+k](1-h[i])
    # d/dh[j]: (1-h[j-k]) - h[j+k] (with boundary=0)
    grad = np.ones(n)

        grad[:n-kk] -= h[kk:]
        # h[j+k] = h[j-kk]: for j >= kk
        grad[kk:] -= h[:n-kk]
    return grad
```

```python
    eta = eta_init
    stall = 0
    
    while time.time() < deadline:
        c_vals = np.correlate(h_v, 1.0 - h_v, mode='full')
        cur_c = c_vals.max() * (2.0 / n)
        
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h_v.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            break
        
        top_idx = np.argsort(c_vals)[-top_k:]
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = int(k_idx) - (n - 1)
            grad += compute_grad_for_lag(h_v, k, n)
        grad /= top_k
        
        h_new = h_v - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full').max() * (2.0 / n)
        
        if c_new < cur_c:
            h_v = h_new
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
            h_new = h_v - eta * grad
            h_new = project_feasible(h_new, n)
            h_v = h_new
    
    c = compute_c5(h_v, n)
    if c < best_c:
        best_c = c
        best_h = h_v.copy()
    
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=500):
    """Greedy binary local search with efficient delta computation."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(map(int, np.where(h == 1)[0]))
    zeros_list = list(map(int, np.where(h == 0)[0]))
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - best_j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                h[i] = 1.0
                h[best_j] = 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=3000, ftol=1e-18):
    """SLSQP with softmax smoothing for refinement."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted = np.clip(shifted, -50.0, 50.0)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Binary greedy (quick)
    p1 = t0 + 0.08 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    # Phase 2: Nesterov smooth (main workhorse)
    p2 = t0 + 0.45 * total
    h = best_h.copy()
    alphas = [30, 50, 100, 200, 500, 1000, 3000, 10000, 30000, 100000, 300000]
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        h = nesterov_smooth(h, n, p2 - 5, alpha_init=alpha)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 3: SLSQP refinement on best
    p3 = t0 + 0.55 * total
    h = best_h.copy()
    for alpha in [5000, 50000, 200000, 1000000]:
        if time.time() > p3 - 3:
            break
        h = slsqp_refine(h, n, alpha, maxiter=2000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 4: Exact subgradient polish
    p4 = t0 + 0.68 * total
    h = best_h.copy()
    h, c = exact_subgradient(h, n, p4, top_k=5, eta_init=0.002)
    if c < best_c:
        best_c, best_h = c, h
    
    # Phase 5: Multi-resolution search
    p5 = t0 + 0.85 * total
    resolutions = [300, 350, 450, 500]
    for nh in resolutions:
        if time.time() > p5 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        # Quick greedy at new resolution
        hh, ch = greed_binary(hh, nh, min(p5 - 10, time.time() + 8), max_sweeps=100)
        # Nesterov at new resolution
        hh = nesterov_smooth(hh, nh, min(p5 - 5, time.time() + 10), alpha_init=200)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
    
    # Phase 6: Final polish at best resolution
    p6 = dl - 3
    if time.time() < p6:
        h = best_h.copy()
        h = nesterov_smooth(h, n, min(p6 - 5, time.time() + 15), alpha_init=5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Final exact subgradient
    if time.time() < dl - 1:
        h = best_h.copy()
        h, c = exact_subgradient(h, n, dl - 1, top_k=3, eta_init=0.001)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
