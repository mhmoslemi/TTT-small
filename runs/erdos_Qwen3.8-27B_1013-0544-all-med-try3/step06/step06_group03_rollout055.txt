def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    # Water-filling: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1

        if s > target:

        else:

    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_proj < 1.0
            if mask.sum() > 0:
                h_proj[mask] += diff / mask.sum()
        else:

            if mask.sum() > 0:

    max_C = C.max()

    if max_scaled > 500:
        return max_C + (max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))) / alpha
    return max_C + np.log(np.sum(np.exp(scaled))) / alpha

def project_feasible(h, n):

        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    if abs(diff) > 1e-13:

    if ms > 500:

def smooth_gradient(h, alpha):

    C = np.correlate(h, 1 - h, mode='full') * dx
    max_C = C.max()
    scaled = alpha * (C - max_C)

    exp_C = np.exp(scaled - ms)

    shifts = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)

    for j in range(n):
        v1 = (j + shifts >= 0) & (j + shifts < n)
        if v1.any():

        v2 = (j - shifts >= 0) & (j - shifts < n)
        if v2.any():
