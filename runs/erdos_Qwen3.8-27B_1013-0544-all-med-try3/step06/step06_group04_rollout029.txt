    prob = cp.Problem(cp.Minimize(t_var), cons)

        except Exception:

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:

    return project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
import warnings
warnings.filterwarnings('ignore')


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if m >= n:
        return grad
    if lag > 0:
        for j in range(n - m):
            grad[j] += (1.0 - h_curr[j + m])
        for j in range(m, n):
            grad[j] -= h_curr[j - m]
    else:
        nm = -lag
        for j in range(nm, n):
            grad[j] += (1.0 - h_curr[j - nm])
        for j in range(n - nm):
            grad[j] -= h_curr[j + nm]
    return grad


def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_constraint_jac(x, n):
    return np.ones(n)


def opt_full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)


def opt_active_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    return smooth_max(active, alpha)


def slsqp_full(h, n, alpha, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(opt_full_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def slsqp_active(h, n, alpha, top_idx, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(opt_active_obj, h, args=(n, alpha, top_idx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)


def outer_approx_cuts(h, n, deadline, max_iters=25):
    """Outer approximation with accumulated cutting planes."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    cuts = []  # list of (c_val, grad, h_point)
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            grad = compute_lag_gradient(h_curr, n, lag)
            cuts.append((c[k], grad, h_curr.copy()))
        
        # Solve LP with all cuts
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        dx = 2.0 / n
        for c_val, grad, h_pt in cuts:
            cons.append(c_val + dx * (grad @ (h_var - h_pt)) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
    
    return best_h, best_c


def outer_approx_trust(h, n, deadline, max_iters=20):
    """Outer approximation with trust region (shrinking)."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.15
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.82)
    return best_h, best_c


def local_search(h, n, deadline):
    """Focused local search on the active lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for delta in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(0, n - m, 1):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    if h[i] > h[j] + 0.005:
                        d = min(delta, (h[i] - h[j]) * 0.4, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    elif h[j] > h[i] + 0.005:
                        d = min(delta, (h[j] - h[i]) * 0.4, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                        if d < 1e-9:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                else:
                    nm = -lag
                    if h[j] > h[i] + 0.005:
                        d = min(delta, (h[j] - h[i]) * 0.4, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                        if d < 1e-9:
                            continue
                        h[j] -= d
                        h[i] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
                    elif h[i] > h[j] + 0.005:
                        d = min(delta, (h[i] - h[j]) * 0.4, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c


def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Trust-region outer approximation
    if time.time() < deadline - 25:
        oo_dead = min(time.time() + 100, deadline - 25)
        h_oo, c_oo = outer_approx_trust(best_h, n, oo_dead, max_iters=20)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    
    # Phase 2: Cutting plane outer approximation
    if time.time() < deadline - 25:
        cp_dead = min(time.time() + 80, deadline - 25)
        h_cp, c_cp = outer_approx_cuts(best_h, n, cp_dead, max_iters=15)
        if c_cp < best_c:
            best_c = c_cp
            best_h = h_cp
    
    # Phase 3: Full SLSQP with increasing alpha
    if time.time() < deadline - 15:
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_full(best_h, n, alpha, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: Adaptive active set SLSQP
    if time.time() < deadline - 15:
        for _ in range(6):
            if time.time() > deadline - 15:
                break
            c = compute_all_lags(best_h, n)
            c_max = c.max()
            # Adaptive threshold: include lags within 3% of max
            threshold = c_max - 0.03 * c_max
            top_idx = np.where(c >= threshold)[0]
            if len(top_idx) > 20:
                top_idx = np.argsort(c)[-20:]
            if len(top_idx) < 3:
                top_idx = np.argsort(c)[-5:]
            for alpha in [100000, 300000]:
                if time.time() > deadline - 15:
                    break
                h_new = slsqp_active(best_h, n, alpha, top_idx, maxiter=4000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    
    # Phase 5: Final high-precision SLSQP
    if time.time() < deadline - 8:
        h_new = slsqp_full(best_h, n, 1000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 6: Local search
    if time.time() < deadline - 1:
        best_h, best_c = local_search(best_h, n, deadline - 0.5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 300
    
    best_h = None
    best_c = float('inf')
    
    # === Phase A: Optimize at parent resolution ===
    if h_parent is not None and time.time() < deadline - 100:
        alloc = min(180, (deadline - t0) * 0.22)
        h_opt, c_opt = optimize_at_resolution(h_parent, n_parent, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # === Phase B: Try higher resolutions ===
    n_candidates = [400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 70:
            break
        alloc = min(140, remaining * 0.35)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_parent is not None:
            x_p = np.linspace(0, 2, n_parent)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_p, h_parent)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # === Phase C: Multi-start at best resolution ===
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        n_restarts = max(2, int((deadline - time.time() - 50) / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 55:
                break
            if n_restarts > 3 and r >= 3:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 40) * 0.4)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # === Phase D: Final polishing ===
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Very high alpha SLSQP
        h_new = slsqp_full(best_h, n, 2000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Active set refinement
        if time.time() < deadline - 5:
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-10:]
            h_new = slsqp_active(best_h, n, 500000, top_idx, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 1:
            best_h, best_c = local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    
    return best_h, best_c, len(best_h)
