    valid_pos = (idx_pos >= 0) & (idx_pos < n)  # shape (n, 2n-1)
    
    t1 = (p[None, :] * valid_pos).sum(axis=1)  # shape (n,)
    
    # For t2: h[i + lag] where valid_pos
    # We need to index h safely
    idx_pos_safe = np.clip(idx_pos, 0, n-1)
    h_pos = h[idx_pos_safe]  # shape (n, 2n-1)
    t2 = (p[None, :] * h_pos * valid_pos).sum(axis=1)
    
    # For term 3: h[i - lag] where valid
    idx_neg = i_idx - lags[None, :]  # shape (n, 2n-1)
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    
    idx_neg_safe = np.clip(idx_neg, 0, n-1)
    h_neg = h[idx_neg_safe]
    t3 = (p[None, :] * h_neg * valid_neg).sum(axis=1)
    
    grad = dx * (t1 - t2 - t3)
    return grad

- In the term where m+k=i, i.e., m=i-k: contribution changes by h_{i-k} * (-δ) = -δ * h_{i-k}
So Δc_k (from changing h_i by δ) = δ * [(1-h_{i+k}) - h_{i-k}] * dx (with boundary corrections)

For changing both h_i and h_j:
Δc_k = dx * {δ * [(1-h_{i+k}) - h_{i-k}] - δ * [(1-h_{j+k}) - h_{j-k}]}
= dx * δ * [(1-h_{i+k}) - h_{i-k} - 1 + h_{j+k} + h_{j-k}]
= dx * δ * [h_{j+k} + h_{j-k} - h_{i+k} - h_{i-k}]

(with boundary corrections for out-of-range indices)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    p = np.exp(shifted - mx)
    p /= p.sum()
    
    # Vectorized gradient computation
    lags = np.arange(-(n-1), n)
    i_idx = np.arange(n)[:, None]
    
    idx_pos = i_idx + lags[None, :]
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    idx_pos_safe = np.clip(idx_pos, 0, n-1)
    
    idx_neg = i_idx - lags[None, :]
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    idx_neg_safe = np.clip(idx_neg, 0, n-1)
    
    t1 = (p[None, :] * valid_pos).sum(axis=1)
    t2 = (p[None, :] * x[idx_pos_safe] * valid_pos).sum(axis=1)
    t3 = (p[None, :] * x[idx_neg_safe] * valid_neg).sum(axis=1)
    
    return dx * (t1 - t2 - t3)

def slsqp_opt(h0, n, alpha, maxiter=2000, tol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        return project_feasible(res.x, n)
    except:
        return h0.copy()

def nesterov_grad(h0, n, alpha, deadline, lr=0.02):
    h = h0.copy()
    n_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Nesterov momentum
        momentum = 0.9
        y = h + momentum * (h - n_prev)
        y_p = project_feasible(y, n)
        g = smooth_max_grad(y_p, n, alpha)
        n_prev = h
        h = project_feasible(y_p - lr * g, n)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if it % 200 == 0:
            lr *= 0.9
    
    return best_h, best_c

def interpolate_h(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
