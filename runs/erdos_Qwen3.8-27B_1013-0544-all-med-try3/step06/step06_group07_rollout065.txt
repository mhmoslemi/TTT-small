```python
def compute_grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx
```

This is O(n * number_of_nonzero_weights). If beta is large, most weights are concentrated on the top few, so it's fast. If beta is small, it's slower but still O(n²).

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx


def subgrad_step(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient, continuation on beta
    for beta in [20, 100, 500, 2000, 10000]:
        if remaining() < 80:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(4):
            if remaining() < 25:
                break
            x0 = best_h.copy()
            if trial > 0:
                noise_scale = 0.005 / trial
                x0 += np.random.randn(n) * noise_scale
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(
                    obj_smooth, x0, args=(dx, beta),
                    jac=lambda x, dx_, b_: grad_smooth(x, dx_, b_),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 5000, 'ftol': 1e-15, 'iprint': 0}
                )
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Projected gradient with smooth max, adaptive beta
    if remaining() > 80:
        h = best_h.copy()
        best_pg_c5 = best_c5
        best_pg_h = h.copy()
        max_iters = 20000
        for it in range(max_iters):
            if remaining() < 40:
                break
            beta = 200 + 100 * it / max_iters
            grad = grad_smooth(h, dx, beta)
            # Step size with decay
            lr = 0.02 / (1.0 + it * 0.0005)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            if it % 200 == 199:
                c5 = compute_c5(h_new)
                if c5 < best_pg_c5:
                    best_pg_c5 = c5
                    best_pg_h = h_new.copy()
                    h = h_new.copy()
                else:
                    h = h_new.copy()
            else:
                h = h_new.copy()
        c5 = compute_c5(h)
        if c5 < best_pg_c5:
            best_pg_c5 = c5
            best_pg_h = h.copy()
        if best_pg_c5 < best_c5:
            best_h = best_pg_h.copy()
            best_c5 = best_pg_c5

    # Phase 3: Multi-subgradient (averaging top lags)
    if remaining() > 80:
        h = best_h.copy()
        best_ms_c5 = best_c5
        best_ms_h = h.copy()
        for it in range(30000):
            if remaining() < 40:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            # Use top 5% of lags
            sorted_idx = np.argsort(corr)[::-1]
            top_k = sorted_idx[:max(5, len(corr) // 20)]
            grad = np.zeros(n)
            for k in top_k:
                grad += subgrad_step(h, k, n)
            grad /= len(top_k)
            lr = 0.005 / (1.0 + it * 0.001)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            h = h_new.copy()
            if it % 1000 == 999:
                c5 = compute_c5(h)
                if c5 < best_ms_c5:
                    best_ms_c5 = c5
                    best_ms_h = h.copy()
        c5 = compute_c5(h)
        if c5 < best_ms_c5:
            best_ms_c5 = c5
            best_ms_h = h.copy()
        if best_ms_c5 < best_c5:
            best_h = best_ms_h.copy()
            best_c5 = best_ms_c5

    # Phase 4: Higher resolution optimization
    if remaining() > 100:
        for n_new in [140, 200, 280]:
            if remaining() < 70:
                break
            dx_new = 2.0 / n_new
            ts_new = n_new / 2.0
            h_new = resample_h(best_h, n_new)
            h_new = project_feasible(h_new, ts_new)
            bnds_new = Bounds(np.zeros(n_new), np.ones(n_new))
            cons_new = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_new, n_new)}]
            best_new_c5 = compute_c5(h_new)
            best_new_h = h_new.copy()

            for beta in [100, 500, 2000, 8000]:
                if remaining() < 30:
                    break
                for trial in range(2):
                    if remaining() < 15:
                        break
                    x0 = best_new_h.copy()
                    if trial > 0:
                        x0 += np.random.randn(n_new) * 0.003
                        x0 = project_feasible(x0, ts_new)
                    try:
                        res = minimize(
                            obj_smooth, x0, args=(dx_new, beta),
                            jac=lambda x, dx_, b_: grad_smooth(x, dx_, b_),
                            method='SLSQP', bounds=bnds_new, constraints=cons_new,
                            options={'maxiter': 8000, 'ftol': 1e-15, 'iprint': 0}
                        )
                        h_opt = project_feasible(res.x, ts_new)
                        c5 = compute_c5(h_opt)
                        if c5 < best_new_c5:
                            best_new_c5 = c5
                            best_new_h = h_opt.copy()
                    except Exception:
                        pass

            if best_new_c5 < best_c5:
                best_h = best_new_h.copy()
                best_c5 = best_new_c5
                n = n_new
                dx = dx_new
                target_sum = ts_new
                bounds = bnds_new

    # Phase 5: Projected gradient at current (possibly higher) resolution
    if remaining() > 60:
        h = best_h.copy()
        best_pg2_c5 = best_c5
        best_pg2_h = h.copy()
        for it in range(15000):
            if remaining() < 30:
                break
            beta = 500
            grad = grad_smooth(h, dx, beta)
            lr = 0.015 / (1.0 + it * 0.0008)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            h = h_new.copy()
            if it % 300 == 299:
                c5 = compute_c5(h)
                if c5 < best_pg2_c5:
                    best_pg2_c5 = c5
                    best_pg2_h = h.copy()
        c5 = compute_c5(h)
        if c5 < best_pg2_c5:
            best_pg2_c5 = c5
            best_pg2_h = h.copy()
        if best_pg2_c5 < best_c5:
            best_h = best_pg2_h.copy()
            best_c5 = best_pg2_c5

    # Phase 6: Final SLSQP polish with very high beta
    if remaining() > 30:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [20000, 80000]:
            if remaining() < 15:

            try:

                    obj_smooth, best_h, args=(dx, beta),
                    jac=lambda x, dx_, b_: grad_smooth(x, dx_, b_),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 10000, 'ftol': 1e-16, 'iprint': 0}
                )
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Tiny perturbations for final escape
    if remaining() > 20:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(8):
            if remaining() < 10:
                break
            scale = 0.002 * (1.0 - trial / 8.0)
            x0 = best_h + np.random.randn(n) * scale
            x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(
                    obj_smooth, x0, args=(dx, 50000),
                    jac=lambda x, dx_, b_: grad_smooth(x, dx_, b_),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0}

                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Final validation and return
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
