import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def eq_sum_fun(x, target_sum):
    return x.sum() - target_sum


def eq_sum_jac(x, target_sum):
    return np.ones(len(x))


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def subgrad_for_lag(h, idx_worst, n):
    lag = idx_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx1 = np.arange(n - lag)
        grad[idx1] += (1.0 - h[idx1 + lag])
        idx2 = np.arange(lag, n)
        grad[idx2] -= h[idx2 - lag]
    else:
        la = -lag
        idx1 = np.arange(n - la)
        grad[idx1 + la] += (1.0 - h[idx1])
        idx2 = np.arange(la, n)
        grad[idx2] -= h[idx2 - la]
    return grad

    return x[n]

    return x[:n].sum() - target_sum

    jac = np.zeros(n + 1)

    return jac


def aug_ineq_fun(x, n, active_indices):
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr[active_indices] * dx

    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    return smooth_max(corr, beta) * dx + pen
