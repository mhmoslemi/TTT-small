                else:
                    best_h = hh.copy()
                n = n if nh != n else nh
    
    # Phase 5: Simulated annealing (8%)
    p5 = t0 + 0.88 * (budget_s - 3)
    if time.time() < p5:
        ha, ca = simulate_anneal_binary(best_h, n, p5, seed=seed + 500)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
        # Polish the annealed solution
        if ca < best_c + 0.001 and time.time() < p5:
            hp = slsqp_optimize(best_h.copy() + 0.01 * rng.randn(n), n, 200000, 1000, p5)
            cp = compute_c5(hp, n)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
    
    # Phase 6: Final polish (remaining time)
    if time.time() < dl - 2:
        h_final = best_h.copy()
        for alpha in [500000, 2000000, 10000000]:
            if time.time() > dl - 3:
                break
            h_final = slsqp_optimize(h_final, n, alpha, 2000, dl - 2)
            c = compute_c5(h_final, n)
            if c < best_c:
                best_c = c
                best_h = h_final.copy()
        # One more binary search from the polished solution
        if time.time() < dl - 1:
            hb, cb = binary_local_search(best_h, n, dl, max_sweeps=200, seed=seed + 999, best_improve=True)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                if time.time() < dl:
                    best_h = slsqp_optimize(best_h.copy() + 0.005 * rng.randn(n), n, 5000000, 500, dl)
                    best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
