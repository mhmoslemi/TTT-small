**Subgradient method for the minimax problem:**

The problem is: minimize f(h) = max_k r_k(h) subject to sum(h) = n/2, 0 ≤ h ≤ 1.

The subgradient of f at h is: (1/K) * sum_{k ∈ active} ∇r_k where active = {k : r_k = f(h)} and K = |active|.

∇r_k / ∂h[i] = coefficient of h[i] in r_k = a_i[k] (the same a vector I computed above!)

So the subgradient step is:
h_new[i] = clip(h[i] - α * (1/K) * sum_{k active} a_i[k], 0, 1)

Where α is a step size.

This is interesting because it uses the same a vectors I'm already computing! And it's "global" in the sense that it considers all active shifts simultaneously.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_a(h, n, i):
    """Compute coefficient vector a for coordinate i."""
    a = np.zeros(2*n-1)
    a[i:i+n] += (1-h)[::-1]
    a[n-1-i:2*n-1-i] += -h
    return a

def coord_minimex(r_const, a):
    """Find t in [0,1] minimizing max(r_const + t*a)."""
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        val = r_const + mid * a
        C = val.max()
        tol = 1e-10
        active = val >= C - tol
        if not active.any():
            break
        a_act = a[active]
        if a_act.min() >= -tol and a_act.max() <= tol:
            break
        elif a_act.min() > 0:
            hi = mid
        elif a_act.max() < 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, n_sweeps, seed, time_limit, t_start):
    """Run coordinate descent for n_sweeps or until time limit."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    raw_r = np.correlate(h, 1-h, mode='full').copy()
    dx = 2.0 / n
    best_c = raw_r.max() * dx
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() - t_start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            a = compute_a(h, n, i)
            t_old = h[i]
            r_const = raw_r - t_old * a
            t_new = coord_minimex(r_const, a)
            delta = t_new - t_old
            if abs(delta) > 1e-14:
                raw_r += delta * a
                h[i] = t_new
        
        # Project
        h = project_feasible(h, n)
        raw_r = np.correlate(h, 1-h, mode='full')
        c = raw_r.max() * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```
