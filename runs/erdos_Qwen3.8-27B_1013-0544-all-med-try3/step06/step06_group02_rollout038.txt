    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        alpha = 1000
        g = grad_smooth_max(h, n, alpha)
        # Find the best pair move
        # We want to increase h[i] and decrease h[j] where g[i] < g[j]
        # The gain is proportional to (g[j] - g[i]) * step
        i = np.argmin(g)
        j = np.argmax(g)
        if h[i] >= 1.0 or h[j] <= 0.0:
            break
        # Try different step sizes
        best_step = 0
        best_c_new = best_c
        for frac in [0.1, 0.5, 1.0]:
            step = frac * min(1.0 - h[i], h[j])
            h[i] += step
            h[j] -= step

            c_new = compute_c5(h, n)
            if c_new < best_c_new:
                best_c_new = c_new
                best_step = step
            h[i] -= step
            h[j] += step
        if best_step > 0:
            h[i] += best_step
            h[j] -= best_step
            best_c = best_c_new
        else:
            # No improvement with this pair, try more pairs
            # Sort by gradient and try top/bottom pairs
            idx = np.argsort(g)
            found = False
            for ii in range(10):
                for jj in range(n-10, n):
                    i_try = idx[ii]
                    j_try = idx[jj]

                    if step < 1e-12:
                        continue
                    h[i] += step
                    h[j] -= step
                    c_try = compute_c5(h, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                        improved = True
                    else:
                        h[i] -= step
                        h[j] += step
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try random pairs as a secondary strategy
            for _ in range(200):
                if time.time() > deadline:
                    return best_h_local, best_c
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if i == j:
                    continue
                max_step = min(1.0 - h[i], h[j])
                if max_step < 1e-10:
                    continue
                step = 0.5 * max_step
                h[i] += step
                h[j] -= step
                c_try = compute_c5(h, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    improved = True
                else:
                    h[i] -= step
                    h[j] += step
            if not improved:
                break
    return h, best_c


def fourier_perturb_optimize(h, n, bounds, deadline, rng):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = np.linspace(0, 2, n)
    modes = [1, 2, 3, 4, 5, 7, 10]
    for mode in modes:
        if time.time() > deadline:
            break
        for _ in range(3):
            if time.time() > deadline:
                break
            phase = rng.uniform(0, 2 * np.pi)
            for eps in [0.03, 0.05, 0.08]:
                if time.time() > deadline:
                    break
                h_try = best_h + eps * np.sin(np.pi * mode * x + phase)
                h_try = project_feasible(h_try, n)
                h_new = lbfgs_optimize(h_try, n, 5000, bounds,
                                       maxiter=200, time_limit=30)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
    return best_h, best_c


def binary_restart_optimize(n, bounds, deadline, rng):
    best_c = compute_c5(np.full(n, 0.5), n)
    best_h = np.full(n, 0.5)
    x = np.linspace(0, 2, n)
    for trial in range(5):
        if time.time() > deadline:
            break
        mask = rng.uniform(size=n) < 0.5
        mask = mask.astype(float)
        # Adjust to get exactly n/2 ones
        n_ones = mask.sum()
        if n_ones > n / 2:
            idx_ones = np.where(mask == 1)[0]
            to_flip = idx_ones[:int(n_ones - n / 2)]
            mask[to_flip] = 0
        elif n_ones < n / 2:
            idx_zeros = np.where(mask == 0)[0]
            to_flip = idx_zeros[:int(n / 2 - n_ones)]
            mask[to_flip] = 1
        # Smooth
        h_try = mask.copy()
        for _ in range(3):
            h_try = np.convolve(h_try, np.ones(5) / 5.0, mode='same')
        h_try = project_feasible(h_try, n)
        h_new = lbfgs_optimize(h_try, n, 1000, bounds,
                               maxiter=200, time_limit=20)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c


def local_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        passes = 0
        while improved and passes < 3 and time.time() < deadline:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c_try = compute_c5(h, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Load initial
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing alpha (main optimization)
    alpha_schedule = [30, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    for alpha in alpha_schedule:
        if remaining() < 80:
            break
        alloc = min(60, remaining() - 40)
        if alloc < 15:
            break
        h_new = lbfgs_optimize(h, n, alpha, bounds, maxiter=400, time_limit=alloc)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = best_h.copy()

    # Phase 2: Gradient-guided pair search
    if remaining() > 80:
        deadline_ps = t0 + budget_s - (remaining() - 120)
        deadline_ps = max(deadline_ps, time.time() + 30)
        h = best_h.copy()
        h, c = gradient_pair_search(h, n, deadline_ps, alpha=50000, max_rounds=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Fourier perturbations
    if remaining() > 150:
        deadline_fp = t0 + budget_s - (remaining() - 100)
        h, c = fourier_perturb_optimize(best_h, n, bounds, deadline_fp, rng)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Binary restart (different starting point)
    if remaining() > 120:
        deadline_br = t0 + budget_s - (remaining() - 80)
        h_br, c_br = binary_restart_optimize(n, bounds, deadline_br, rng)
        if c_br < best_c:
            best_c = c_br
            best_h = h_br.copy()

    # Phase 5: Higher resolution attempt
    if remaining() > 180:
        for n_try in [400, 350]:
            if remaining() < 150:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            bounds_new = [(0.0, 1.0)] * n_try
            for alpha in [1000, 10000, 50000]:
                if remaining() < 80:
                    break
                alloc = min(40, remaining() - 50)
                if alloc < 15:
                    break
                h_new = lbfgs_optimize(h_interp, n_try, alpha, bounds_new,
                                       maxiter=300, time_limit=alloc)
                c_new = compute_c5(h_new, n_try)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_interp = h_new

    # Phase 6: Final L-BFGS polishing on best
    if remaining() > 40:
        n_final = len(best_h)
        bounds_final = [(0.0, 1.0)] * n_final
        for alpha in [50000, 200000]:
            if remaining() < 25:
                break
            h_new = lbfgs_optimize(best_h, n_final, alpha, bounds_final,
                                   maxiter=200, time_limit=min(30, remaining() - 15))
            c_new = compute_c5(h_new, n_final)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 7: Final local pair search
    if remaining() > 25:
        n_f = len(best_h)
        h, c = local_pair_search(best_h, n_f,
                                 deadline=t0 + budget_s - 5,
                                 deltas=[0.005, 0.002, 0.001, 0.0005],
                                 radius=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
