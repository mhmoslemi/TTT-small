def compute_grads(h):
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m = k - (n-1)
        g = np.zeros(n)

            j_idx = np.arange(m, n)
            g[j_idx] += (1 - h[j_idx - m])
            # Second part: grad[j] += (-h[j+m]) for j in [0, n-1-m]
            j_idx2 = np.arange(0, n - m)

            j_idx = np.arange(0, n - p)
            g[j_idx] += (1 - h[j_idx + p])

            j_idx2 = np.arange(p, n)
            g[j_idx2] += (-h[j_idx2 - p])

    values = np.correlate(h, 1.0-h, mode='full')  # (2n-1,)

    c_obj[n] = 1.0
    
    n_constr = 2*n - 1
    A_ub = np.zeros((n_constr, n_vars))
    b_ub = np.zeros(n_constr)
    
    d = values - grads @ h  # (2n-1,)
    
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -d
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])

import numpy as np
from scipy.optimize import minimize, linprog, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grads(h):
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m = k - (n-1)
        g = np.zeros(n)
        if m >= 0:
            j1 = np.arange(m, n)
            g[j1] += (1 - h[j1 - m])
            j2 = np.arange(0, n - m)
            g[j2] += (-h[j2 + m])
        else:
            p = -m
            j1 = np.arange(0, n - p)
            g[j1] += (1 - h[j1 + p])
            j2 = np.arange(p, n)
            g[j2] += (-h[j2 - p])
        grads[k] = g
    return grads

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def solve_sca_lp(h, target_sum):
    n = len(h)
    grads = compute_grads(h)
    values = np.correlate(h, 1.0 - h, mode='full')
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    n_constr = 2 * n - 1
    d = values - grads @ h
    A_ub = np.empty((n_constr, n_vars))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -d
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds_lp = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        return res.x[:n].copy()
    return h.copy()

def run_sca(h_init, target_sum, t_start, time_limit_s):
    h = h_init.copy()
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.8
    stale = 0
    
    deadline = t_start + time_limit_s
    for it in range(200000):
        if time.time() > deadline:
            break
        
        h_lp = solve_sca_lp(h, target_sum)
        h_try = (1.0 - lam) * h + lam * h_lp
        h_try = project_feasible(h_try, target_sum)
        
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-13:
            h = h_try.copy()
            best = h_try.copy()
            best_c5 = c5_try
            lam = min(1.0, lam + 0.05)
            stale = 0
        elif c5_try <= best_c5 + 1e-12:
            h = h_try.copy()
            stale += 1
            if stale % 50 == 0:
                lam = max(0.3, lam - 0.05)
        else:
            lam = max(0.1, lam * 0.9)
            stale += 1
        
        if stale > 1000:
            break
    
    return best, best_c5

def run_slsqp_polish(h_init, target_sum, t_start, time_limit_s):
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(h)
    
    deadline = t_start + time_limit_s
    for beta in [100, 500, 2000, 10000]:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        try:
            res = minimize(obj_smooth, best, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
    
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Phase 1: SCA at initial resolution (main optimization)
    time_for_sca = min(300, budget_s * 0.4)

    if c5_sca < best_c5:
        best_h = h_sca
        best_c5 = c5_sca
    
    # Phase 2: SLSQP polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 30:
        h_pol, c5_pol = run_slsqp_polish(best_h, target_sum, time.time(), min(60, budget_s - elapsed - 10))
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Phase 3: Try different resolutions
    elapsed = time.time() - t0
    if budget_s - elapsed > 50:
        for n_try in [n + 10, n + 20, n - 10, n + 5, n + 30, n - 20]:
            if n_try < 20:
                continue
            elapsed = time.time() - t0
            if budget_s - elapsed < 30:
                break
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            remaining = budget_s - (time.time() - t0) - 10
            if remaining < 10:
                break
            # SCA at new resolution
            h_opt, c5_opt = run_sca(h_t, ts_t, time.time(), min(remaining * 0.6, 60))
            # SLSQP polish
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                n = n_try
                target_sum = ts_t
                # Quick SLSQP
                elapsed2 = time.time() - t0
                if budget_s - elapsed2 > 15:
                    h_pol, c5_pol = run_slsqp_polish(best_h, target_sum, time.time(), min(20, budget_s - elapsed2 - 5))
                    if c5_pol < best_c5:
                        best_h = h_pol
                        best_c5 = c5_pol
    
    # Phase 4: Final SCA at best resolution
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        n = len(best_h)
        target_sum = n / 2.0

        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return best_h, best_c5, len(best_h)
