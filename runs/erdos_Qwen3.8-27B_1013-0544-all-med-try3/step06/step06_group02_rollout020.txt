import numpy as np
from scipy.optimize import minimize
import time

def corr_fft(h, dh):
    """Compute cross-correlation using FFT. Equivalent to np.correlate(h, dh, mode='full')."""
    n = len(h)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h, fft_size)
    D = np.fft.rfft(dh, fft_size)
    corr = np.fft.irfft(H * np.conj(D), fft_size)[:size]
    return corr

def compute_c5(h, n):
    dx = 2.0 / n
    c = corr_fft(h, 1.0 - h)
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = corr_fft(h, 1.0 - h)
    return c * dx

def grad_smooth_max_fast(h, n, alpha, top_k=50):
    """Compute gradient of smooth max using FFT and sparse weights."""
    c = corr_fft(h, 1.0 - h)  # length 2n-1
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    c_max_shift = c_shift.max()  # should be 0
    w = np.exp(c_shift - c_max_shift)
    w_sum = w.sum()
    w /= w_sum
    
    # Only keep top_k weights
    if top_k > 0 and len(w) > top_k:
        # Find indices of top weights
        top_indices = np.argpartition(w, -top_k)[-top_k:]
        w_sparse = np.zeros_like(w)
        w_sparse[top_indices] = w[top_indices]
        # Renormalize
        w_sparse /= w_sparse.sum()
    else:
        w_sparse = w
    
    dx = 2.0 / n
    n_shifts = len(w_sparse)
    grad = np.zeros(n)
    
    for idx in np.where(w_sparse > 1e-15)[0]:
        k = idx - (n - 1)
        wm = w_sparse[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    
    return grad

def corr_fft(a, b):
    """Cross-correlation via FFT. Equivalent to np.correlate(a, b, mode='full')."""
    n1, n2 = len(a), len(b)
    N = n1 + n2 - 1
    fft_size = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b[::-1], fft_size)
    return np.fft.irfft(A * B, fft_size)[:N]

def compute_c5(h, n):
    dx = 2.0 / n
    c = corr_fft(h, 1.0 - h)
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return corr_fft(h, 1.0 - h) * dx

def project_feasible(h, n):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    if max_shifted > 500:

    # Phase 8: Final L-BFGS polish
    if time.time() < deadline - 3:
        h_opt, c_opt = lbfgs_optimize(best_h, best_n, deadline - 1)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
