```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_lag_k_gradient(h, n, k):
    """Gradient of the k-th lag value w.r.t. h."""
    lag = k - (n - 1)
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        else:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max value and gradient."""
    c = compute_all_lags(h, n)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # Handle overflow
    shifted_max = shifted.max()
    if shifted_max > 500:
        shifted = shifted - shifted_max
        log_sum = shifted_max + np.log(np.sum(np.exp(shifted)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    val = c_max + log_sum / alpha
    
    w = np.exp(shifted)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        lag = k - (n - 1)
        m = abs(lag)
        wk = w[k]
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m < n:
            if lag > 0:
                grad[:n-m] += wk * (-h[m:])
                grad[m:] += wk * (1.0 - h[:n-m])
            else:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] += wk * (-h[:n-m])
    return val, grad

def objective_with_penalty(h, n, alpha, penalty_w):
    val, grad = smooth_max_and_grad(h, n, alpha)
    penalty = penalty_w * (h.sum() - n/2.0) ** 2
    grad = grad + 2 * penalty_w * (h.sum() - n/2.0) * np.ones(n)
    return val + penalty, grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_refine(h, n):
    """Try to find a better binary solution near h."""
    # Start from threshold
    b = (h >= 0.5).astype(float)
    # Adjust to meet sum constraint
    target = n // 2
    current_sum = int(b.sum())
    if current_sum > target:
        # Remove some 1s (those with lowest h values among the 1s)
        ones = np.where(b == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for i in range(current_sum - target):
            b[ones_sorted[i]] = 0
    elif current_sum < target:
        # Add some 1s (those with highest h values among the 0s)
        zeros = np.where(b == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        for i in range(target - current_sum):
            b[zeros_sorted[i]] = 1
    return b

def binary_local_search(b, n, deadline, max_rounds=50):
    """Local search on binary solutions."""
    best_c = compute_c5(b, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        c = compute_all_lags(b, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if m >= n:
            break
        # Try flipping pairs to reduce the max lag
        if lag > 0:
            # c_m = sum h[i]*(1-h[i+m]) for i=0..n-1-m
            # We want to reduce this
            # If h[i]=1 and h[i+m]=0, flipping both (h[i]->0, h[i+m]->1) reduces c_m by 1
            # But also affects other lags
            for i in range(n - m):
                if time.time() > deadline:
                    break
                if b[i] == 1 and b[i + m] == 0:
                    b[i] = 0
                    b[i + m] = 1
                    c_new = compute_c5(b, n)
                    if c_new < best_c:
                        best_c = c_new
                        improved = True
                    else:
                        b[i] = 1
                        b[i + m] = 0
        else:
            # lag < 0, m = |lag|
            # For negative lag, the correlation is: sum h[i+m]*(1-h[i]) for i=0..n-1-m
            # i.e., sum h[j]*(1-h[j-m]) for j=m..n-1
            # To reduce: if h[j]=1 and h[j-m]=0, flip both
            for j in range(m, n):
                if time.time() > deadline:
                    break
                if b[j] == 1 and b[j - m] == 0:
                    b[j] = 0
                    b[j - m] = 1
                    c_new = compute_c5(b, n)
                    if c_new < best_c:
                        best_c = c_new
                        improved = True
                    else:
                        b[j] = 1
                        b[j - m] = 0
        if not improved:
            # Try random flips (simulated annealing style)
            # Pick a random pair (i, j) and swap
            for _ in range(100):
                if time.time() > deadline:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j or b[i] == b[j]:
                    continue
                old_i, old_j = b[i], b[j]
                b[i], b[j] = old_j, old_i
                c_new = compute_c5(b, n)
                if c_new < best_c:
                    best_c = c_new
                    improved = True
                else:
                    b[i], b[j] = old_i, old_j
                if improved:
                    break
            if not improved:
                break
    return b, best_c

def optimize_at_n(h_start, n, deadline):
    """Full optimization pipeline at given n."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: L-BFGS-B with penalty (good for large n)
    if time.time() < deadline - 30:
        for alpha in [200, 1000, 5000]:
            if time.time() > deadline - 30:
                break
            h_new = lfgbbs_optimize(best_h, n, deadline - 30, alpha=alpha, penalty_w=500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
                h = h_new
    
    # Phase 2: SLSQP with full smooth max
    if time.time() < deadline - 20:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 20:
                break

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 15:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-10:]
        for alpha in [10000, 100000]:
            if time.time() > deadline - 15:
                break
            h_new = active_set_slsqp(best_h, n, top_idx, alpha, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:

                best_h = h_new
                # Update top set
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-10:]
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_gradient(h, n, k):
    """Gradient of the k-th correlation value (before dx scaling)."""
    lag = k - (n - 1)
    m = abs(lag)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]
            g[m:] += (1.0 - h[:n-m])
        else:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

def smooth_max_val_grad(h, n, alpha):
    """Value and gradient of the log-sum-exp smooth max of all lag values."""
    c = compute_all_lags(h, n)  # already includes dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    s_max = shifted.max()
    if s_max > 500:
        log_sum = s_max + np.log(np.sum(np.exp(shifted - s_max)))
        w = np.exp(shifted - s_max)
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
        w = np.exp(shifted)
    w /= w.sum()
    val = c_max + log_sum / alpha
    
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        lag = k - (n - 1)
        m = abs(lag)
        wk = w[k]
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m < n:
            if lag > 0:
                grad[:n-m] += wk * (-h[m:])
                grad[m:] += wk * (1.0 - h[:n-m])
            else:
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] += wk * (-h[:n-m])
    return val, dx * grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lfbbc_step(h, n, alpha, penalty_w, maxiter, time_limit):
    """One L-BFGS-B optimization step."""
    t_start = time.time()
    
    def obj(x):
        v, _ = smooth_max_val_grad(x, n, alpha)
        return v + penalty_w * (x.sum() - n/2.0)**2
    
    def grad(x):
        _, g = smooth_max_val_grad(x, n, alpha)
        return g + 2 * penalty_w * (x.sum() - n/2.0)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_step(h, n, alpha, maxiter, top_idx=None):
    """SLSQP optimization step with smooth max."""
    if top_idx is None:
        top_idx = np.arange(len(compute_all_lags(h, n)))
    
    def obj(x):
        c = compute_all_lags(x, n)
        a = c[top_idx]
        m = a.max()
        sh = alpha * (a - m)
        sm = sh.max()
        if sm > 500:

        return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def grad(x):
        c = compute_all_lags(x, n)
        a = c[top_idx]

            w = np.exp(sh)

    def sum_fun(x):
        return np.sum(x) - n / 2.0
    def sum_jac(x):
        return np.ones(n)

    try:

def binary_refine(h, n):
    """Project to binary and adjust sum."""
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(b.sum())
    if s > target:
        ones = np.where(b == 1)[0]
        order = ones[np.argsort(h[ones])]
        for i in range(s - target):
            b[order[i]] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for i in range(target - s):
            b[order[i]] = 1
    return b
