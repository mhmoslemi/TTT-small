def bang_bang_project(h, n):
    """Project h to bang-bang (0/1) while maintaining sum = n/2."""
    n_ones = n // 2
    # Sort indices by h value, put top n_ones to 1, rest to 0
    idx = np.argsort(h)[::-1]
    h_bb = np.zeros(n)
    h_bb[idx[:n_ones]] = 1.0
    return h_bb

def local_search_bb(h_bb, n, deadline, rng, max_iters=100000):
    """Local search on bang-bang configurations using swap moves."""
    h = h_bb.copy()
    best_c = compute_c5(h, n)
    ones_positions = set(np.where(h == 1)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:

        # Pick a random 1 to move
        ones_list = list(ones_positions)
        i = ones_list[rng.randint(len(ones_list))]
        # Pick a random 0 to swap with

        j = zeros_list[rng.randint(len(zeros_list))]

def best_swap_c5(C, h, n):
    """Find the best swap (a: 1→0, b: 0→1) that minimizes C₅.
    Returns (a, b, new_c5) or (None, None, C5) if no improvement."""
    dx = 2.0 / n
    c5_curr = C.max() * dx
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_new_c5 = c5_curr
    best_a, best_b = None, None
    
    # Precompute Q[b, m] for all b
    # m ranges from -(n-1) to (n-1), index = m + (n-1)
    num_lags = 2*n - 1
    m_vals = np.arange(-(n-1), n)  # m values
    
    for a in ones:
        # Compute P_a[m]
        P = np.zeros(num_lags)
        # Part 1: h[a+m] for -a ≤ m ≤ n-1-a
        m1_lo = max(0, (n-1) - a)  # index where m = -a
        m1_hi = min(num_lags-1, (n-1) + (n-1-a))  # index where m = n-1-a
        for idx in range(m1_lo, m1_hi+1):
            m = idx - (n-1)
            if -a <= m <= n-1-a:
                P[idx] += h[a+m]
        # Part 2: -(1-h[a-m]) for a-(n-1) ≤ m ≤ a
        m2_lo = max(0, (n-1) + (a-(n-1)))  # index where m = a-(n-1)
        m2_hi = min(num_lags-1, (n-1) + a)  # index where m = a
        for idx in range(m2_lo, m2_hi+1):
            m = idx - (n-1)
            if a-(n-1) <= m <= a:
                P[idx] -= (1 - h[a-m])
        
        base = C + P  # C[m] + P_a[m]
        # For each b in zeros, compute max(base - Q[b])
        for b in zeros:
            Q = np.zeros(num_lags)
            # h[b+m] for -b ≤ m ≤ n-1-b
            for idx in range(max(0, n-1-b), min(num_lags-1, n-1+n-1-b)+1):
                m = idx - (n-1)
                if -b <= m <= n-1-b:
                    Q[idx] += h[b+m]
            # -(1-h[b-m]) for b-(n-1) ≤ m ≤ b
            for idx in range(max(0, n-1+b-(n-1)), min(num_lags-1, n-1+b)+1):
                m = idx - (n-1)
                if b-(n-1) <= m <= b:
                    Q[idx] -= (1 - h[b-m])
            
            new_max = (base - Q).max()
            new_c5 = new_max * dx
            if new_c5 < best_new_c5 - 1e-15:
                best_new_c5 = new_c5
                best_a, best_b = a, b
        if best_new_c5 < c5_curr * (1 - 0.001):  # early exit if good enough
            break
    
    return best_a, best_b, best_new_c5
