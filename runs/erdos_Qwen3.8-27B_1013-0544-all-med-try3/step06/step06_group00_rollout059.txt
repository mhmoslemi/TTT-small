from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def gradient_multishift(h, n, n_active=5, eps=1e-3):
    """Compute gradient for minimizing max correlation, using active set."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # Active shifts: those within eps of max
    active = np.where(corr >= max_val - eps)[0]
    # Use top n_active
    top = active[np.argsort(corr[active])[::-1][:n_active]]
    
    g = np.zeros(n)
    for j in top:
        s = j - (n - 1)
        if s >= 0:
            # A(s) = sum_{m=s}^{N-1} h[m]
            # dA/dh[p] = 1 if p >= s
            # B(s) = sum_{n=0}^{N-1-s} h[n]*h[n+s]
            # dB/dh[p] = h[p-s] (if p >= s) + h[p+s] (if p+s < N)
            mask_a = np.arange(n) >= s
            mask_b1 = np.arange(n) >= s
            mask_b2 = np.arange(n) + s < n
            g += mask_a * (1 - h) - np.where(mask_b1, h[np.clip(np.arange(n) - s, 0, n-1)], 0) - np.where(mask_b2, h[np.arange(n) + s], 0)
        else:
            sp = -s  # positive shift magnitude
            # A(s) = sum_{m=0}^{N-1-sp} h[m]
            # dA/dh[p] = 1 if p < N-sp
            # B(s) = B(sp) = sum_{n=0}^{N-1-sp} h[n]*h[n+sp]
            # dB/dh[p] = h[p+sp] (if p < N-sp) + h[p-sp] (if p >= sp)
            mask_a = np.arange(n) < n - sp
            mask_b1 = np.arange(n) < n - sp
            mask_b2 = np.arange(n) >= sp
            g += mask_a * (1 - h) - np.where(mask_b1, h[np.arange(n) + sp], 0) - np.where(mask_b2, h[np.arange(n) - sp], 0)
    return g / len(top)
