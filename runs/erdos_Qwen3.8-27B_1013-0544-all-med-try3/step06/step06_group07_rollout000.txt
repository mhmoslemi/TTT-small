```
    F[n-1]=np.dot(h,1.0-h)
    for d in range(1,n):
       a=h[:-d]; b=h[d:]; ab=a*b
       F[n-1+d]=np.sum(a)-np.sum(ab)
       F[n-1-d]=np.sum(b)-np.sum(ab)
    return F

def raw_grad_matrix(h,n):
    J=np.zeros((2*n-1,n))
    J[n-1]=1.0-2.0*h
    for d in range(1,n):
       a=h[:-d]; b=h[d:]
       row=J[n-1+d]
       row[:-d] += 1.0 - b
       row[d:] -= a
       row=J[n-1-d]
       row[d:] += 1.0 - a
       row[:-d] -= b
    return J
```

```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds

GL_DEADLINE = 0.0


def global_callback(x):
    if time.time() > GL_DEADLINE:
        raise InterruptedError


def compute_c5_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def raw_all_F(h, n):
    F = np.empty(2 * n - 1, dtype=float)
    F[n - 1] = float(np.dot(h, 1.0 - h))
    for d in range(1, n):
        a = h[:-d]
        b = h[d:]
        ab = a * b
        sab = float(np.sum(ab))
        F[n - 1 + d] = float(np.sum(a) - sab)
        F[n - 1 - d] = float(np.sum(b) - sab)
    return F


def compute_c5_fast(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(raw_all_F(h, n)) * (2.0 / n))


def raw_grad_matrix(h, n):
    J = np.zeros((2 * n - 1, n), dtype=float)
    J[n - 1] = 1.0 - 2.0 * h
    for d in range(1, n):
        a = h[:-d]
        b = h[d:]
        rowp = J[n - 1 + d]
        rowp[:-d] += 1.0 - b
        rowp[d:] -= a
        rown = J[n - 1 - d]
        rown[d:] += 1.0 - a
        rown[:-d] -= b
    return J


def project_feasible(h, target):
    h = np.asarray(h, dtype=float).ravel().copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-15:
        lo = -1.0
        hi = 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
            if s > target:

            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

        else:

    h_old = np.asarray(h_old, dtype=float).ravel()
    n_old = h_old.size
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

    k_top = min(m, max(12, max_lags // 6))
    act = order[-k_top:]
    above = np.where(Fdx >= c_ref - 1.0e-3)[0]
    if above.size:
        act = np.union1d(act, above)
    if act.size > max_lags:
        ord_act = np.argsort(Fdx[act])
        act = act[ord_act[-max_lags:]]
    amax = int(order[-1])
    if amax not in act:
        act = np.append(act, amax)
    return np.asarray(act, dtype=int)


def nlp_obj(x, n):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    return float(x[n])


def nlp_obj_grad(x, n):
    g = np.zeros(n + 1, dtype=float)
    g[n] = 1.0
    return g


def nlp_eq_fun(x, n):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    return float(x[:n].sum() - 0.5 * n)


def nlp_eq_jac(x, n):
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def nlp_sub_ineq(x, n, lags):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    Fdx = raw_all_F(h, n) * dx
    return t - Fdx[lags]


def nlp_sub_jac(x, n, lags):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    h = x[:n]
    dx = 2.0 / n
    G = raw_grad_matrix(h, n)
    J = np.empty((lags.size, n + 1), dtype=float)
    J[:, :n] = -dx * G[lags]
    J[:, n] = 1.0
    return J


def run_slsqp_active_norm(h0, t0, n, lags, maxiter, ftol):
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = t0

    lower = np.zeros(n + 1, dtype=float)
    upper = np.ones(n + 1, dtype=float)
    upper[n] = 1.0
    bounds = Bounds(lower, upper)

    cons = [
        {"type": "ineq", "fun": nlp_sub_ineq, "jac": nlp_sub_jac, "args": (n, np.asarray(lags, dtype=int))},
        {"type": "eq", "fun": nlp_eq_fun, "jac": nlp_eq_jac, "args": (n,)},
    ]
    opts = {
        "maxiter": int(maxiter),
        "ftol": float(ftol),
        "iprint": 0,
        "maxls": 100,
    }
    return minimize(
        nlp_obj, x0, args=(n,), method="SLSQP",
        bounds=bounds, jac=nlp_obj_grad, constraints=cons,
        callback=global_callback, options=opts
    )


def improve_nlp(h_in, n, max_rounds, stop_time):
    h_in = np.asarray(h_in, dtype=float).ravel()
    target = 0.5 * n
    best_h = project_feasible(h_in, target)
    best_c = compute_c5_exact(best_h)
    work_h = best_h.copy()
    work_c = compute_c5_fast(work_h)

    no_improve = 0
    for rnd in range(int(max_rounds)):
        if time.time() > stop_time or time.time() > GL_DEADLINE - 25:
            break

        h_cur = work_h
        Fdx = raw_all_F(h_cur, n) * (2.0 / n)
        c_work = float(np.max(Fdx))
        if c_work < best_c - 1e-13:
            c_chk = compute_c5_exact(h_cur)
            if c_chk < best_c:
                best_h = h_cur.copy()
                best_c = c_chk

        final = (rnd == int(max_rounds) - 1) or (no_improve >= 2)
        if final:
            if 2 * n - 1 <= 220:
                lags = np.arange(2 * n - 1, dtype=int)
            else:
                lags = choose_lags(Fdx, c_work, max_lags=160)
        else:
            lags = choose_lags(Fdx, c_work, max_lags=60)

        if n <= 100:
            maxiter = 500
        elif n <= 180:
            maxiter = 300
        else:
            maxiter = 180
        ftol = 1e-12 if final else 1e-10

        improved_best = False
        try:
            res = run_slsqp_active_norm(h_cur, c_work + 1e-8, n, lags, maxiter, ftol)
            if hasattr(res, "x") and res.x is not None:
                h_new = np.asarray(res.x[:n], dtype=float)
                h_new = project_feasible(h_new, target)
                c_new_fast = compute_c5_fast(h_new)
                c_new_exact = compute_c5_exact(h_new)
                if c_new_exact < best_c - 1e-13:
                    best_h = h_new
                    best_c = c_new_exact
                    improved_best = True
                if c_new_exact <= c_work + 1.0e-3:
                    work_h = h_new
                    work_c = c_new_fast
        except Exception:
            pass

        if improved_best:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                work_h = best_h.copy()
                work_c = compute_c5_fast(work_h)
                no_improve = 1

    c_final = compute_c5_exact(project_feasible(best_h, target))
    return project_feasible(best_h, target), c_final


def anti_expand(y, n):
    y = np.asarray(y, dtype=float).ravel()
    p = n // 2
    h = np.full(n, 0.5, dtype=float)
    if p > 0:
        y = y[:p]
        idx = np.arange(p)
        h[idx] = 0.5 + 0.5 * y
        h[n - 1 - idx] = 0.5 - 0.5 * y
    return h


def anti_y_from_h(h, n):
    h = np.asarray(h, dtype=float).ravel()
    p = n // 2
    if p == 0:
        return np.zeros(0, dtype=float)
    idx = np.arange(p)
    y = h[idx] - h[n - 1 - idx]
    return np.clip(y, -1.0, 1.0)


def anti_nlp_obj(x, p, n):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    return float(x[p])


def anti_nlp_obj_grad(x, p, n):
    g = np.zeros(p + 1, dtype=float)
    g[p] = 1.0
    return g


def anti_nlp_sub_ineq(x, p, n, lags):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    y = x[:p]
    t = x[p]
    h = anti_expand(y, n)
    dx = 2.0 / n
    Fdx = raw_all_F(h, n) * dx
    return t - Fdx[lags]


def anti_nlp_sub_jac(x, p, n, lags):
    if time.time() > GL_DEADLINE:
        raise InterruptedError
    y = x[:p]
    h = anti_expand(y, n)
    dx = 2.0 / n
    G = raw_grad_matrix(h, n)
    idx = np.arange(p)
    rev = n - 1 - idx
    Gy = 0.5 * (G[:, idx] - G[:, rev])
    J = np.empty((lags.size, p + 1), dtype=float)
    J[:, :p] = -dx * Gy[lags]
    J[:, p] = 1.0
    return J


def run_slsqp_active_anti(y0, t0, p, n, lags, maxiter, ftol):
    x0 = np.empty(p + 1, dtype=float)
    x0[:p] = y0
    x0[p] = t0

    lower = np.empty(p + 1, dtype=float)
    upper = np.empty(p + 1, dtype=float)
    lower[:p] = -1.0
    upper[:p] = 1.0
    lower[p] = 0.0
    upper[p] = 1.0
    bounds = Bounds(lower, upper)

    cons = [
        {"type": "ineq", "fun": anti_nlp_sub_ineq, "jac": anti_nlp_sub_jac,
         "args": (p, n, np.asarray(lags, dtype=int))}
    ]
    opts = {
        "maxiter": int(maxiter),
        "ftol": float(ftol),
        "iprint": 0,
        "maxls": 100,
    }
    return minimize(
        anti_nlp_obj, x0, args=(p, n,), method="SLSQP",
        bounds=bounds, jac=anti_nlp_obj_grad, constraints=cons,
        callback=global_callback, options=opts
    )


def improve_anti_nlp(h_in, n, max_rounds, stop_time):
    h_in = np.asarray(h_in, dtype=float).ravel()
    p = n // 2
    if p <= 0:
        return h_in, compute_c5_exact(h_in)

    best_y = anti_y_from_h(project_feasible(h_in, 0.5 * n), n)
    best_h = anti_expand(best_y, n)
    best_c = compute_c5_exact(best_h)
    work_y = best_y.copy()
    work_h = best_h.copy()
    work_c = compute_c5_fast(work_h)

    no_improve = 0
    for rnd in range(int(max_rounds)):
        if time.time() > stop_time or time.time() > GL_DEADLINE - 25:
            break

        h_cur = work_h
        Fdx = raw_all_F(h_cur, n) * (2.0 / n)
        c_work = float(np.max(Fdx))
        if c_work < best_c - 1e-13:
            c_chk = compute_c5_exact(h_cur)
            if c_chk < best_c:
                best_h = h_cur
                best_y = anti_y_from_h(h_cur, n)
                best_c = c_chk

        final = (rnd == int(max_rounds) - 1) or (no_improve >= 2)
        if final:
            if 2 * n - 1 <= 240:
                lags = np.arange(2 * n - 1, dtype=int)
            else:
                lags = choose_lags(Fdx, c_work, max_lags=160)
        else:
            lags = choose_lags(Fdx, c_work, max_lags=60)

        if n <= 100:
            maxiter = 500
        elif n <= 180:
            maxiter = 300
        else:
            maxiter = 180
        ftol = 1e-12 if final else 1e-10

        improved_best = False
        try:
            res = run_slsqp_active_anti(work_y, c_work + 1e-8, p, n, lags, maxiter, ftol)
            if hasattr(res, "x") and res.x is not None:
                y_new = np.asarray(res.x[:p], dtype=float)
                y_new = np.clip(y_new, -1.0, 1.0)
                h_new = anti_expand(y_new, n)
                c_new_fast = compute_c5_fast(h_new)
                c_new_exact = compute_c5_exact(h_new)
                if c_new_exact < best_c - 1e-13:
                    best_h = h_new
                    best_y = y_new
                    best_c = c_new_exact
                    improved_best = True
                if c_new_exact <= c_work + 1.0e-3:
                    work_y = y_new
                    work_h = h_new
                    work_c = c_new_fast
        except Exception:
            pass

        if improved_best:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                work_y = best_y.copy()
                work_h = best_h.copy()
                work_c = compute_c5_fast(work_h)
                no_improve = 1

    h_final = anti_expand(best_y, n)
    h_final = project_feasible(h_final, 0.5 * n)
    return h_final, compute_c5_exact(h_final)


def polish_gradient(h_in, n, stop_time, rng, max_iter=2000):
    h = project_feasible(np.asarray(h_in, dtype=float).ravel(), 0.5 * n)
    dx = 2.0 / n
    c = compute_c5_fast(h)
    idle = 0

    for _ in range(int(max_iter)):
        if time.time() > stop_time or time.time() > GL_DEADLINE - 12:
            break

        Fdx = raw_all_F(h, n) * dx
        c_work = float(np.max(Fdx))
        if c_work < c - 1e-15:
            c = c_work
            idle = 0

        A = np.where(Fdx >= c_work - 1.0e-5)[0]
        if A.size == 0:
            A = np.array([int(np.argmax(Fdx))], dtype=int)
        G = raw_grad_matrix(h, n)
        g = G[A].sum(axis=0)
        d = -(g - g.mean())
        d = d - d.mean()
        dmax = float(np.max(np.abs(d)))
        if dmax <= 1e-14:
            d = rng.normal(size=n)
            d = d - d.mean()
            dmax = float(np.max(np.abs(d)))
            if dmax <= 1e-12:
                break
        d = d / dmax * 0.02

        best_h = h
        best_c = c
        for step in (0.25, 0.5, 0.75, 1.0, 0.12, 0.05, 0.02):
            h2 = h + step * d
            h2 = np.clip(h2, 0.0, 1.0)
            h2 = project_feasible(h2, 0.5 * n)
            c2 = compute_c5_fast(h2)
            if c2 < best_c - 1e-15:
                best_c = c2
                best_h = h2.copy()

        if best_c < c - 1e-14:
            h = best_h
            c = best_c
            idle = 0
        else:
            idle += 1
            if idle > 80:
                d = rng.normal(size=n)
                d = d - d.mean()
                dmax = float(np.max(np.abs(d)))
                if dmax > 0:
                    d = d / dmax * 0.06
                    for step in (0.25, 0.5, 0.75, 1.0):
                        h2 = h + step * d
                        h2 = np.clip(h2, 0.0, 1.0)
                        h2 = project_feasible(h2, 0.5 * n)
                        c2 = compute_c5_fast(h2)
                        if c2 < best_c - 1e-15:
                            best_c = c2
                            best_h = h2.copy()
                    if best_c < c - 1e-14:
                        h = best_h
                        c = best_c
                        idle = 0
                    else:
                        idle += 60
                else:
                    break
            if idle > 250:
                break

    h = project_feasible(h, 0.5 * n)
    return h, compute_c5_exact(h)


def run(seed=42, budget_s=1100, **kwargs):
    global GL_DEADLINE
    t0 = time.time()
    budget_s = float(budget_s)
    reserve = min(30.0, max(5.0, 0.10 * budget_s))
    GL_DEADLINE = t0 + budget_s - reserve

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.RandomState(seed)

    h_best = None
    n_best = None
    c_best = np.inf

    try:
        parent = initial_h_values
    except NameError:
        parent = None

    if parent is not None:
        hp = np.asarray(parent, dtype=float).ravel()
        if hp.size > 0:
            n_best = int(hp.size)
            h_best = project_feasible(hp, 0.5 * n_best)
            c_best = compute_c5_exact(h_best)

    if h_best is None:
        n_best = 70
        base_cands = [
            np.full(n_best, 0.5),
            np.clip(0.5 + 0.75 * np.sin(2.0 * np.pi * (np.arange(n_best) + 0.5) / n_best), 0.0, 1.0),
            np.linspace(0.0, 1.0, n_best),
            0.5 + 0.35 * rng.normal(size=n_best),
        ]
        for cand in base_cands:
            hc = project_feasible(cand, 0.5 * n_best)
            cc = compute_c5_exact(hc)
            if cc < c_best:
                c_best = cc
                h_best = hc

    # Main normal-space active-constraint solver.
    remaining = GL_DEADLINE - time.time()
    if remaining > 80:
        alloc = min(max(30.0, 0.45 * budget_s), 420.0, max(0.0, remaining - 260.0))
        if alloc > 25:
            hh, cc = improve_nlp(h_best, n_best, 6, time.time() + alloc)
            if cc < c_best:
                h_best, c_best = hh, cc

    # Anti-symmetric subspace: h_i + h_{n-1-i} = 1.
    remaining = GL_DEADLINE - time.time()
    if remaining > 220:
        alloc = min(max(30.0, 0.22 * budget_s), 260.0, max(0.0, remaining - 200.0))
        if alloc > 25:
            hh, cc = improve_anti_nlp(h_best, n_best, 5, time.time() + alloc)
            if cc < c_best:
                h_best, c_best = hh, cc

    # A few anti-symmetric random restarts.
    anti_trials = 0
    while anti_trials < 10 and time.time() < GL_DEADLINE - 220:
        p = n_best // 2
        if p <= 0:
            break
        scale = [0.05, 0.15, 0.4, 1.0][anti_trials % 4] * rng.uniform(0.6, 1.4)
        y0 = anti_y_from_h(h_best, n_best) + scale * rng.normal(size=p)
        y0 = np.clip(y0, -1.0, 1.0)
        alloc = min(45.0, max(0.0, GL_DEADLINE - time.time() - 180.0))
        if alloc < 20:
            break
        hh, cc = improve_anti_nlp(anti_expand(y0, n_best), n_best, 3, time.time() + alloc)
        if cc < c_best:
            h_best, c_best = hh, cc
        anti_trials += 1

    # Basin hopping in the full space.
    normal_trials = 0
    scales = [0.008, 0.02, 0.06, 0.18, 0.45]
    while normal_trials < 35 and time.time() < GL_DEADLINE - 190:
        scale = scales[normal_trials % len(scales)] * rng.uniform(0.6, 1.4)
        h_try = project_feasible(h_best + scale * rng.normal(size=n_best), 0.5 * n_best)
        c_try_fast = compute_c5_fast(h_try)
        if c_try_fast < max(c_best + 0.004, c_best * 1.005):
            alloc = min(45.0, max(0.0, GL_DEADLINE - time.time() - 150.0))
            if alloc < 20:
                break
            hh, cc = improve_nlp(h_try, n_best, 3, time.time() + alloc)
            if cc < c_best:
                h_best, c_best = hh, cc
        normal_trials += 1

    # Final normal-space polish with larger active sets.
    remaining = GL_DEADLINE - time.time()
    if remaining > 130:
        alloc = min(110.0, max(0.0, remaining - 70.0))
        if alloc > 25:
            hh, cc = improve_nlp(h_best, n_best, 4, time.time() + alloc)
            if cc < c_best:
                h_best, c_best = hh, cc

    # Deterministic active-gradient polish.
    remaining = GL_DEADLINE - time.time()
    if remaining > 70:
        alloc = min(55.0, max(0.0, remaining - 25.0))
        if alloc > 15:
            hh, cc = polish_gradient(h_best, n_best, time.time() + alloc, rng, max_iter=3000)
            if cc < c_best:
                h_best, c_best = hh, cc

    # Try neighboring and coarser/finer resolutions.
    res_list = [n_best + 10, n_best - 10, 2 * n_best, n_best + 25]
    for n_try in res_list:
        if n_try < 40:
            continue
        remaining = GL_DEADLINE - time.time()
        if remaining < 95:
            break
        h_try = resample_h(h_best, n_try)
        h_try = project_feasible(h_try, 0.5 * n_try)
        c_try = compute_c5_exact(h_try)
        if c_try < c_best + 0.003:
            alloc = min(75.0, max(0.0, remaining - 45.0))
            if alloc < 20:
                continue
            hh, cc = improve_nlp(h_try, n_try, 4, time.time() + alloc)
            if cc < c_best:
                h_best, c_best, n_best = hh, cc, n_try

    # Last chance final solver.
    remaining = GL_DEADLINE - time.time()
    if remaining > 45:
        hh, cc = improve_nlp(h_best, n_best, 3, GL_DEADLINE - 8)
        if cc < c_best:
            h_best, c_best = hh, cc

    h_best = project_feasible(h_best, 0.5 * n_best)
    c_fast = compute_c5_fast(h_best)
    c_exact = compute_c5_exact(h_best)
    c_out = float(max(c_fast, c_exact))
    return h_best, c_out, int(n_best)
```
