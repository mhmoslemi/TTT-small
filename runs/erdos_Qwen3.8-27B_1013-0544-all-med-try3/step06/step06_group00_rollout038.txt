        p_hi = 2 * n1 - j
        if p_hi < 2 * n:
            grad[j] += np.dot(p[p_lo:min(p_hi + 1, 2 * n)], 1.0 - h[0:n - j])
        p_lo2 = j
        p_hi2 = j + n1
        if p_hi2 < 2 * n:
            grad[j] -= np.dot(p[p_lo2:min(p_hi2 + 1, 2 * n)], h[n - 1:n - 1 - (n - j):-1])
    return L, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        ev = np.exp(shifted - max_s)
        return m + np.log(ev.sum()) / alpha
    def obj_grad(x):
        L, g = smooth_loss_grad(x, n, alpha)
        return L, g
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj_grad, h, method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_grad(h_init, n, deadline, alphas=None, momentum=0.85):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 5.0 / (alpha * n)
        v = np.zeros(n)
        for it in range(2000):
            if time.time() > deadline:
                break
            L, g = smooth_loss_grad(h, n, alpha)
            v = momentum * v - lr * g
            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            if it % 200 == 0:
                lr *= 0.8
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, T0=0.012, T_end=5e-5, cool=0.99995, batch=50):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].astype(np.int64)
    zeros_arr = np.where(h == 0)[0].astype(np.int64)
    T = T0
    while time.time() < deadline and T > T_end:
        for _ in range(batch):
            if time.time() > deadline:
                break
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = int(zeros_arr[ai])
            j = int(ones_arr[bi])
            if i == j:
                continue
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            if 0 <= k1 < 2 * n - 1:
                delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            if 0 <= k2 < 2 * n - 1:
                delta[k2] += 1.0
            nc = (corr + delta).max() * dx
            if nc <= cur_c:
                accept = True
            else:
                diff = nc - cur_c
                if T > 1e-14:
                    accept = rng.random() < np.exp(-diff / T)
                else:
                    accept = False
            if accept:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        T *= cool
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            for j in list(ones_list):
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP smooth at parent resolution (15% time)
    p1_end = t0 + 0.15 * total
    h = best_h.copy()
    for alpha in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() > p1_end - 5:
            break
        h = slsqp_smooth(h, n0, alpha, 300, 1e-14, p1_end)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Greedy binary at parent (5% time)
    p2_end = t0 + 0.20 * total
    if time.time() < p2_end:
        hg, cg = greed_binary(best_h, n0, p2_end, max_sweeps=100)
        if cg < best_c:
            best_c, best_h = cg, hg.copy()
            best_n = n0

    # Phase 3: Proper annealing at parent resolution (30% time)
    p3_end = t0 + 0.50 * total
    if time.time() < p3_end:
        ha, ca = anneal_binary(best_h, n0, p3_end, seed=seed,
                               T0=0.015, T_end=1e-5, cool=0.99993, batch=30)
        if ca < best_c:
            best_c, best_h, best_n = ca, ha.copy(), n0
        # Second round with different seed
        remaining = p3_end - time.time()
        if remaining > 30:
            ha2, ca2 = anneal_binary(best_h, n0, p3_end, seed=seed + 1,
                                     T0=0.008, T_end=1e-6, cool=0.99995, batch=30)
            if ca2 < best_c:
                best_c, best_h, best_n = ca2, ha2.copy(), n0

    # Phase 4: Try higher resolution (25% time)
    p4_end = t0 + 0.75 * total
    for nh in [800, 1000]:
        if time.time() > p4_end - 20:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        # Quick SLSQP at new resolution
        alpha_list = [1000, 10000, 100000]
        sub_deadline = min(p4_end - 10, time.time() + 0.5 * (p4_end - time.time()))
        for alpha in alpha_list:
            if time.time() > sub_deadline:
                break
            hh = slsqp_smooth(hh, nh, alpha, 200, 1e-12, sub_deadline)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh
        # Anneal at new resolution
        if time.time() < p4_end - 15:
            ha, ca = anneal_binary(best_h, nh, min(p4_end - 5, time.time() + 40),
                                   seed=seed + 10, T0=0.01, T_end=1e-5,
                                   cool=0.99995, batch=20)
            if ca < best_c:
                best_c, best_h, best_n = ca, ha.copy(), nh

    # Phase 5: Re-smooth and final polish (15% time)
    p5_end = deadline
    if time.time() < p5_end - 5:
        if best_h.max() - best_h.min() > 1e-10:
            h = best_h.copy()
            for alpha in [50000, 500000, 5000000]:
                if time.time() > p5_end - 3:
                    break
                h = slsqp_smooth(h, best_n, alpha, 500, 1e-14, p5_end)
                c = compute_c5(h, best_n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        # Final greedy on best
        if time.time() < p5_end - 3:
            hg, cg = greed_binary(best_h, best_n, p5_end, max_sweeps=50)
            if cg < best_c:
                best_c, best_h = cg, hg.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
