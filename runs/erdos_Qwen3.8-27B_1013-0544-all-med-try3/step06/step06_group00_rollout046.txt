    # Constraints: 0 <= z[j] <= 1 for j < n-1, and 0 <= h[n-1] <= 1
    
    def obj_and_grad(z):
        h = np.empty(n)
        h[:-1] = z
        h[-1] = n / 2.0 - z.sum()
        if h[-1] < -1e-10 or h[-1] > 1 + 1e-10:
            return 1e10, np.zeros(n - 1)
        # Clip for numerical safety
        h_safe = np.clip(h, 0, 1)
        dx = 2.0 / n
        c = np.correlate(h_safe, 1 - h_safe, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        # Gradient
        w = np.exp(shifted - sm)
        w /= w.sum()
        nfull = 2 * n - 1
        lags = np.arange(nfull) - (n - 1)
        grad_h = np.zeros(n)
        for i in range(n):
            idx1 = i + lags
            mask1 = (idx1 >= 0) & (idx1 < n)
            grad_h[i] += np.dot(w[mask1], 1.0 - h_safe[idx1[mask1]])
            idx2 = i - lags
            mask2 = (idx2 >= 0) & (idx2 < n)
            grad_h[i] -= np.dot(w[mask2], h_safe[idx2[mask2]])
        grad_h *= dx
        # Chain rule: dF/dz[j] = dF/dh[j] - dF/dh[n-1] for j < n-1
        grad_z = grad_h[:-1] - grad_h[-1]
        return val, grad_z
    
    z0 = h[:-1].copy()
    bounds = [(0.0, 1.0)] * (n - 1)
    
    try:
        res = minimize(obj_and_grad, z0, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 50})
        h_new = np.empty(n)
        h_new[:-1] = res.x
        h_new[-1] = n / 2.0 - res.x.sum()
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        return h_new
    except Exception as e:
        return h
