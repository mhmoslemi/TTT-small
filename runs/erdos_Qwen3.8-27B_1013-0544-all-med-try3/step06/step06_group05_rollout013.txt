def compute_G(h, n, i, j):
    """Compute G[k] for pair (i,j)."""
    k_vals = np.arange(2*n-1)
    s_vals = k_vals - (n-1)  # shift values
    
    # For position i:
    p1 = i - s_vals  # = i - k + n - 1
    mask1 = (p1 >= 0) & (p1 < n)
    term1 = np.zeros(2*n-1)
    term1[mask1] = 1 - h[p1[mask1]]
    
    p2 = i + s_vals  # = i + k - n + 1
    mask2 = (p2 >= 0) & (p2 < n)
    term2 = np.zeros(2*n-1)
    term2[mask2] = -h[p2[mask2]]
    
    # For position j:
    p3 = j - s_vals
    mask3 = (p3 >= 0) & (p3 < n)
    term3 = np.zeros(2*n-1)
    term3[mask3] = -(1 - h[p3[mask3]])
    
    p4 = j + s_vals
    mask4 = (p4 >= 0) & (p4 < n)
    term4 = np.zeros(2*n-1)
    term4[mask4] = h[p4[mask4]]
    
    return term1 + term2 + term3 + term4
