def init_f(h, n):

    f[k][s] = (1-h[k+s]) * I[valid] - h[k-s] * I[valid]

        valid_right = (k - m_range >= 0) & (k - m_range < n)
        f[k, valid_right] -= h[k - m_range[valid_right]]

    h = h.copy()

    cur_max = corr.max() * dx
    f = init_f(h, n)
    k_arr = np.arange(n)
    
    best_h = h.copy()
    best_c = cur_max
    T = T0
    rng = np.random.RandomState(42)
    
    zero_set = np.where(h == 0)[0]
    one_set = np.where(h == 1)[0]
    
    while time.time() < deadline:
        # Pick random swap
        i = rng.choice(zero_set)
        j = rng.choice(one_set)

        new_corr_max = (corr + f[i] - f[j]).max() * dx

2. Then, do a "kick" (random perturbation of a few positions)
3. Repeat hill climbing

**Revised algorithm:**

Phase 1: Greedy hill climbing from initial solution
Phase 2: Iterated local search (kick + hill climb)
Phase 3: Multi-resolution exploration
Phase 4: Continuous optimization (SLSQP) from best binary solution

Let me also consider: what if the optimal solution is NOT binary? The problem allows h(x) ∈ [0,1], not just {0,1}. Maybe a non-binary solution gives a lower C₅?

For the continuous problem, the objective is:
C₅ = max_k ∫ h(x)(1-h(x+k)) dx

If h is not binary, then h(x)(1-h(x+k)) can take values in [0, 0.25] (max when h=0.5 and h(x+k)=0.5). The integral is bounded by 0.25 * (length of overlap).

But the constraint is ∫h = 1, so the "average" value of h is 1/2. If h is constant = 1/2, then h(x)(1-h(x+k)) = 1/4 for all x, and the integral is 1/4 * (length of valid range). For large shifts, the valid range is small, but for small shifts k near 0, the valid range is nearly 2. So C₅ ≈ 1/4 * 2 = 1/2. That's worse than the binary solution!

So binary (or near-binary) solutions are likely optimal. This makes sense because the function x(1-x) is maximized at x=1/2, so we want h to be as far from 1/2 as possible, i.e., binary.

OK so I'll focus on binary optimization.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def init_f(h, n):
    f = np.zeros((n, 2*n-1))
    m_range = np.arange(-(n-1), n)
    for k in range(n):
        vl = (k + m_range >= 0) & (k + m_range < n)
        f[k, vl] += (1.0 - h[k + m_range[vl]])
        vr = (k - m_range >= 0) & (k - m_range < n)
        f[k, vr] -= h[k - m_range[vr]]
    return f

def update_f(f, n, i, j, k_arr):
    nm1 = n - 1
    f[k_arr, i+nm1-k_arr] -= 1.0
    f[k_arr, nm1-i+k_arr] -= 1.0
    f[k_arr, j+nm1-k_arr] += 1.0
    f[k_arr, nm1-j+k_arr] += 1.0

def greedy_hill_climb(h, n, deadline, max_iters=500000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    f = init_f(h, n)
    k_arr = np.arange(n)
    nm1 = n - 1
    
    best_c = cur_max * dx
    best_h = h.copy()
    
    zero_set = set(np.where(h == 0)[0].tolist())
    one_set = set(np.where(h == 1)[0].tolist())
    zero_list = list(zero_set)
    one_list = list(one_set)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Try to find an improving swap
        improved = False
        # Shuffle for diversity
        rng_order = np.random.permutation(len(zero_list))
        
        for idx in rng_order:
            if time.time() > deadline:
                break
            i = zero_list[idx]
            if i not in zero_set:
                continue
            g_i = corr + f[i]
            # For each j in one_set, check if max(g_i - f[j]) < cur_max
            for j in one_list:
                if j not in one_set:
                    continue
                new_max = (g_i - f[j]).max()
                if new_max < cur_max - 1e-12:
                    # Apply swap
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + f[i] - f[j]
                    update_f(f, n, i, j, k_arr)
                    cur_max = corr.max()
                    
                    zero_set.discard(i)
                    zero_set.add(j)
                    one_set.discard(j)
                    one_set.add(i)
                    
                    if cur_max * dx < best_c:
                        best_c = cur_max * dx
                        best_h = h.copy()
                    
                    improved = True
                    break
            if improved:
                # Rebuild lists (expensive but needed occasionally)
                # Actually, we can just continue with old lists and skip invalid entries
                break
        
        if not improved:
            # No improving swap found, try best-improvement
            # Check all pairs for any improvement
            found = False
            for i in list(zero_set):
                if time.time() > deadline:
                    break
                g_i = corr + f[i]

                    new_max = (g_i - f[j]).max()
                    if new_max < cur_max - 1e-12:
                        h[i], h[j] = 1.0, 0.0
