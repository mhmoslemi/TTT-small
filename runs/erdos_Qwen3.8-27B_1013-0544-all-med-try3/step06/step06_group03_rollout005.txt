def obj_topk_logsumexp(h, n, alpha, k=20):
    """Smooth max focusing on top-k lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Get top k values
    if k < len(c):
        idx = np.argpartition(c, -k)[-k:]
        c_top = c[idx]

        c_top = c
    m = c_top.max()
    # Log-sum-exp for numerical stability
    a = alpha * (c_top - m)
    return float(m + np.log(np.sum(np.exp(a))) / alpha)

    num = 2 * n - 1

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    num = len(c)
    
    if alpha is None:
        return c, None
    
    if k is None or k >= num:
        idx = np.arange(num)
        c_top = c
    else:
        idx = np.argpartition(c, -k)[-k:]
        c_top = c[idx]

    a_max = a.max()

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_onto_simplex_box(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # We want: sum(clip(h - lambda, 0, 1)) = target
    # f(lambda) = sum(clip(h - lambda, 0, 1)) is decreasing in lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha, k=None):
    """Compute log-sum-exp smooth max and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    num = len(c)
    
    if k is not None and k < num:
        idx = np.argpartition(c, -k)[-k:]
        c_top = c[idx]
    else:
        idx = np.arange(num)
        c_top = c
    
    m = c_top.max()
    a = alpha * (c_top - m)
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    exp_sum = exp_a.sum()
    weights = exp_a / exp_sum
    
    val = m + (a_max + np.log(exp_sum)) / alpha
    # Since m = c_top.max(), a_max = 0, so val = m + log(exp_sum)/alpha
    
    n_half = n - 1

    grad = np.zeros(n)

        lag = lags[t]
        w = weights[t] * dx
        if lag >= 0:
            L = n - lag
            if L > 0:
                grad[:L] += w * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= w * h[0:L]
            else:
                grad[:] -= w * h[0:n]  # lag=0, L=n
        else:
            lag = -lag
            L = n - lag
            if L > 0:
                grad[lag:] += w * (1.0 - h[0:L])
            if lag > 0:
                grad[:L] -= w * h[lag:n]
    
    return val, grad

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha, k=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    num = len(c)
    if k is not None and k < num:
        idx = np.argpartition(c, -k)[-k:]
        c_top = c[idx]
    else:
        idx = np.arange(num)
        c_top = c
    m = c_top.max()
    a = alpha * (c_top - m)
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    exp_sum = exp_a.sum()
    weights = exp_a / exp_sum
    val = m + (a_max + np.log(exp_sum)) / alpha
    n_half = n - 1
    lags = (idx - n_half).astype(np.int32)
    grad = np.zeros(n)
    for t in range(len(idx)):
        lag = int(lags[t])
        w = weights[t] * dx
        if lag >= 0:
            L = n - lag
            if L > 0:
                grad[:L] += w * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= w * h[0:L]
            else:
                grad[:L] -= w * h[0:L]
        else:
            lag = -lag
            L = n - lag
            if L > 0:
                grad[lag:] += w * (1.0 - h[0:L])
            if lag > 0:
                grad[:L] -= w * h[lag:n]
    return val, grad

        s = x.sum() - n / 2.0
        return val + 0.5 * mu * s * s

        _, grad = smooth_max_val_grad(x, n, alpha, k)
        s = x.sum() - n / 2.0
        return grad + mu * s
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    maxiter = 400 if n <= 200 else 250
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': maxiter * 10})
