            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 300:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alpha_mu_pairs, maxiter_per=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alpha_mu_pairs:
        if time.time() > deadline - 3:
            break
        t_left = deadline - time.time()
        if t_left < 2:
            break
        maxit = min(maxiter_per, max(20, int(t_left * 3)))
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def projected_gradient(h, n, deadline, alpha_init=500, alpha_end=200000, n_iter=500):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    lr = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        step = 0.05
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                step = min(step * 1.3, 0.1)
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-14:
                break
    return best_h, best_c

def pair_search_fine(h, n, deadline, deltas, radius=100):
    best_c = compute_c5(h, n)
    target = n / 2.0
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_guided_pairs(h, n, deadline, alpha=10000, n_trials=50):
    best_c = compute_c5(h, n)
    target = n / 2.0
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            rng = np.random.RandomState(int(time.time() * 1000) % 100000)
            candidates = rng.choice(n, size=min(n, 60), replace=False)
            for i in candidates:
                if time.time() > deadline:
                    return h, best_c
                # Determine best partner based on gradient
                move_dir = -g[i]
                if move_dir > 0:
                    # Want to increase h[i], decrease h[j] where g[j] is most negative
                    partners = np.argsort(g)[:30]
                else:
                    partners = np.argsort(g)[-30:]
                for j in partners:
                    if i == j:
                        continue
                    if move_dir > 0:
                        if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                            h[i] += step; h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; improved = True
                            else:
                                h[i] -= step; h[j] += step
                    else:
                        if h[i] - step >= -1e-12 and h[j] + step <= 1.0 + 1e-12:
                            h[i] -= step; h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; improved = True
                            else:
                                h[i] += step; h[j] -= step
            if not improved:
                break
    return h, best_c

def random_perturb(h, n, strength, rng):
    h_pert = h.copy()
    mask = rng.random(n) < 0.3
    h_pert[mask] += rng.uniform(-strength, strength, size=mask.sum())
    return project_feasible(h_pert, n)

def multi_resolution_optimize(h_best, n_best, deadline, rng):
    best_c = compute_c5(h_best, n_best)
    best_h = h_best.copy()
    best_n = n_best
    
    for n_try in [n_best * 2, n_best * 2 + 100, n_best // 2 + 50]:
        if time.time() > deadline - 60:
            break
        if n_try < 100 or n_try > 2000:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(90, deadline - time.time() - 40)
        if alloc < 30:
            break
        
        phases = [
            [(500, 10.0), (2000, 50.0), (10000, 200.0), (50000, 1000.0)],
            [(10000, 500.0), (50000, 2000.0), (200000, 5000.0)],
        ]
        
        t_phase1 = time.time() + alloc * 0.55
        t_phase2 = deadline - 40
        h_interp, c_mid = lbfgs_phase(h_interp, n_try, t_phase1, phases[0], maxiter_per=150)
        if time.time() < t_phase2:
            h_interp, c_final = lbfgs_phase(h_interp, n_try, t_phase2, phases[1], maxiter_per=100)
            c_mid = c_final
        
        if c_mid < best_c - 1e-10:
            best_c = c_mid
            best_h = h_interp.copy()
            best_n = n_try
    
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except (NameError, Exception):
        h_init = np.full(300, 0.5)
        n_init = 300
    
    h = project_feasible(h_init, n_init)
    best_c = compute_c5(h, n_init)
    best_h = h.copy()
    best_n = n_init
    
    # Phase 1: L-BFGS-B at original resolution (0-80s)
    alloc1 = min(80, deadline - 500)
    phases1 = [
        [(100, 1.0), (500, 5.0), (2000, 20.0)],
        [(5000, 50.0), (20000, 200.0), (100000, 1000.0)],
        [(500000, 5000.0), (2000000, 20000.0)],
    ]
    t1 = time.time() + alloc1 * 0.4
    h, best_c = lbfgs_phase(h, n_init, t1, phases1[0], maxiter_per=200)
    if best_c < compute_c5(best_h, best_n):
        best_c = best_c
        best_h = h.copy()
    t1b = time.time() + min(alloc1 * 0.3, deadline - time.time() - 350)
    h, c_tmp = lbfgs_phase(h, n_init, t1b, phases1[1], maxiter_per=200)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    if time.time() < deadline - 300:
        h, c_tmp = lbfgs_phase(h, n_init, min(time.time() + 40, deadline - 300), phases1[2], maxiter_per=150)
        if c_tmp < best_c:
            best_c = c_tmp
            best_h = h.copy()
    best_n = n_init
    
    # Phase 2: Random restarts to escape local optima
    if time.time() < deadline - 300:
        for strength in [0.05, 0.02, 0.01]:
            if time.time() > deadline - 250:
                break
            h_pert = random_perturb(best_h, best_n, strength, rng)
            alloc_r = min(40, deadline - time.time() - 200)
            if alloc_r < 15:
                break
            h_r, c_r = lbfgs_phase(h_pert, best_n, time.time() + alloc_r,
                                   [(1000, 50.0), (10000, 500.0), (100000, 5000.0)],
                                   maxiter_per=100)
            if c_r < best_c - 1e-10:
                best_c = c_r
                best_h = h_r.copy()
                best_n = best_n
    
    # Phase 3: Multi-resolution optimization
    if time.time() < deadline - 200:
        best_h, best_c, best_n = multi_resolution_optimize(best_h, best_n, deadline - 150, rng)
    
    # Phase 4: Projected gradient at best resolution
    if time.time() < deadline - 100:
        alloc_pg = min(100, deadline - time.time() - 80)
        if alloc_pg > 20:
            best_h, best_c = projected_gradient(best_h, best_n, time.time() + alloc_pg,
                                                 alpha_init=2000, alpha_end=1000000,
                                                 n_iter=300)
    
    # Phase 5: High-alpha L-BFGS
    if time.time() < deadline - 60:
        nf = len(best_h)
        alloc5 = min(60, deadline - time.time() - 40)
        if alloc5 > 15:
            phases5 = [(100000, 5000.0), (500000, 20000.0), (2000000, 100000.0)]
            best_h, best_c = lbfgs_phase(best_h, nf, time.time() + alloc5, phases5, maxiter_per=100)
    
    # Phase 6: Gradient-guided pair search
    if time.time() < deadline - 30:
        nf = len(best_h)
        best_h, best_c = gradient_guided_pairs(best_h, nf, deadline - 25, alpha=50000)
    
    # Phase 7: Fine pair search
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = pair_search_fine(best_h, nf, deadline - 2,
                                           [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                           radius=60)
    
    # Final: one more L-BFGS + pair search
    if time.time() < deadline - 1:
        nf = len(best_h)
        h_final, c_final = lbfgs_phase(best_h, nf, deadline - 1, [(1000000, 50000.0)], maxiter_per=50)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
