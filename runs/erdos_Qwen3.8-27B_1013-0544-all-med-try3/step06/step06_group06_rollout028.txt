    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, max_sweeps=2000):
    best_c = compute_c5(h_bin, n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0].tolist()
        zeros = np.where(h_bin == 0)[0].tolist()
        for j in zeros:
            if time.time() > deadline:
                return h_bin, best_c
            for i in ones:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, best_c

    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()

        if l >= 0:
            grad[l:n] += -dx * h[0:n-l]
        else:
            grad[0:n+l] += -dx * h[-l:n]

def subgradient_optimize(h, n, deadline, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad, m = compute_subgradient(h, n)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        found = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-10:
                h = ht
                found = True
                if ct < best_c:
                    best_c = ct
                    best_h = h.copy()
                    stall = 0
                break
        
        if not found:
            lr *= 0.8
            stall += 1
            if lr < 1e-10 or stall > 100:
                break
    
    return best_h, best_c

def slsqp_refine(h, n, deadline, alphas, maxiter_each):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        
        def smooth_obj(x, alpha=alpha, n=n):
            dx = 2.0 / n
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            if sm > 500:
                sm = 500
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_sum, 'args': (n,)}]
        
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_candidate = project_feasible(res.x, n)
            c_candidate = compute_c5(h_candidate, n)
            if c_candidate < best_c:
                best_c = c_candidate
                best_h = h_candidate.copy()
        except Exception:
            pass
    
    return best_h, best_c
