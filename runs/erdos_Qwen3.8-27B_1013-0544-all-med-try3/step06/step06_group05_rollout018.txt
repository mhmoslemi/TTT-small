import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h -= (-diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def pg_step(h, n, alpha, lr):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    grad = dx * (np.correlate(1 - h, w, mode='full')[:n] 
                 - np.convolve(w, h, mode='full')[:n])
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, cmax

def coord_descent(h0, n, t_end, n_sweeps=5000):
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    rng = np.random.RandomState(123)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end - 3:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        order = rng.permutation(n)
        for j in order:
            if time.time() > t_end - 3:
                return best_h, best_c
            v = h[j]
            # a[k] = dc/dh_j / dx
            a = np.zeros(n)
            a[:n-j] = 1.0 - h[j:]
            a[:j+1] -= h[j::-1]
            # minimize max_k (c[k] + dx*(t-v)*a[k]) over t in [0,1]
            m = dx * a
            b = c - dx * v * a
            # ternary search
            tlo, thi = 0.0, 1.0
            for _ in range(40):
                t1 = tlo + (thi - tlo) * 0.3333
                t2 = thi - (thi - tlo) * 0.3333
                if np.max(b + m * t1) <= np.max(b + m * t2):
                    thi = t2
                else:
                    tlo = t1
            t_opt = (tlo + thi) * 0.5

                cmax = c.max()
                if cmax < best_c - 1e-15:

                    best_h = h.copy()
    return best_h, best_c
