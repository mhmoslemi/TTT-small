    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    # Bisection on Lagrange multiplier
    target = n / 2.0
    h = h.copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h + mid, 0.0, 1.0)
        s = h_test.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # Fix any small numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    exp_c = np.exp(alpha * (corr - c_max))
    sum_exp = exp_c.sum()
    obj = (c_max + np.log(sum_exp)) / alpha
    weights = exp_c / sum_exp  # softmax weights
    
    # Compute gradient
