    target = n / 2.0

        if abs(diff) < 1e-14:

            if tot < 1e-15:

def grad_g_for_lag(h, k, n):
    """Gradient of g(h, k) = sum h[i](1-h[i+k]) for the given lag k."""
    grad = np.zeros(n)
    if k >= 0:

        grad[k:n] += (-h[0:nv])

            best_c = c_max
            best_h = h.copy()
        
        # Top-m active lags
        if top_m >= len(c):
            top_indices = np.arange(len(c))
        else:
            top_indices = np.argsort(c)[-top_m:]
        
        # Average gradient
        grad = np.zeros(n)
        for idx in top_indices:
            k = int(idx) - (n - 1)
            grad += gradient_at_lag(h, k, n)
        grad /= float(len(top_indices))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Backtracking
        step = lr
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                lr = min(step * 1.05, 0.5)
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            if lr < 1e-12:
                break
    
    return best_h, best_c

def local_pair_search(h, n, deadline, steps=None, radius=50):
    """Pair search for local improvement."""
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    
    return h, best_c

def single_point_search(h, n, deadline, steps=None):
    """Try adjusting individual points."""
    if steps is None:
        steps = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    target = n / 2.0
    
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try increasing h[i] by d (must decrease sum elsewhere)
                # This is handled by projection, so just try it
                h[i] += d
                h_new = project_feasible(h, n)
                h[i] -= d
                
                c = compute_c5(h_new, n)
                if c < best_c - 1e-14:
                    h = h_new
                    best_c = c
                    improved = True
                
                h[i] -= d
                h_new = project_feasible(h, n)
                h[i] += d
                
                c = compute_c5(h_new, n)
                if c < best_c - 1e-14:
                    h = h_new
                    best_c = c
                    improved = True
            if not improved:
                break
    
    return h, best_c
