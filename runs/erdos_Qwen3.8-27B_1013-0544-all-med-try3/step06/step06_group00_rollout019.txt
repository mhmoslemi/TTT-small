    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_shift(h, n, k, dx):
    """Gradient of dx*sum h[i]*(1-h[i+k]) wrt h."""
    n_minus_1 = n - 1
    m = n_minus_1 + k
    # We need the gradient of corr[m] wrt h
    # corr[m] = dx * sum_i h[i] * (1-h[i+k]) where h[j]=0 if j<0 or j>=n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif k > 0:
        # d/dh[j] = (1-h[j+k]) - h[j-k]
        # (1-h[j+k]) valid for j=0..n-k-1
        grad[:n-k] += (1.0 - h[k:])
        # -h[j-k] valid for j=k..n-1
        grad[k:] += (-h[:n-k])
    else:

        # (1-h[j-ka]) valid for j=ka..n-1
        grad[ka:] += (1.0 - h[:n-ka])
        # -h[j+ka] valid for j=0..n-ka-1
        grad[:n-ka] += (-h[ka:])
    return grad * dx

    """Subgradient descent with adaptive step size."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = step_lr
    stall = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c = corr.max()
        
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        
        # Find top-k shifts
        top_idx = np.argsort(corr)[-top_k:]
        
        # Weighted gradient (weight by correlation value)
        weights = corr[top_idx] - corr[top_idx].min() + 1e-8
        weights = np.exp(weights * 100.0)  # softmax with temp=0.01
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            k = k_idx - (n - 1)
            grad += weights[i] * grad_single_shift(h, n, k, dx)
        
        # Take step
        h_new = project_feasible(h - lr * grad, n)
        
        # Accept if it doesn't make things too much worse
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k, dx):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] += -h[:n - k]
    else:
        ka = -k
        grad[ka:] += 1.0 - h[:n - ka]
        grad[:n - ka] += -h[ka:]
    return grad * dx

def subgrad_optimize(h, n, deadline, top_k=30, init_lr=0.3, temperature=0.01):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = init_lr
    stall = 0
    stall_threshold = 300

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c = corr.max()
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= stall_threshold:
                lr *= 0.5
                stall = 0
                if lr < 1e-9:
                    break

        top_idx = np.argsort(corr)[-top_k:]
        scores = corr[top_idx]
        max_s = scores.max()
        w = np.exp((scores - max_s) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum

        grad = np.zeros(n)
        nbf = n - 1
        for i in range(len(top_idx)):
            k = top_idx[i] - nbf
            grad += w[i] * grad_shift(h, n, k, dx)

        cnorm = np.linalg.norm(grad)
        if cnorm < 1e-15:
            break
        grad = grad / cnorm

        h_new = project_feasible(h - lr * grad, n)
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        if c_new <= c + 0.001 * lr:
            h = h_new
        else:
            h_half = project_feasible(h - 0.5 * lr * grad, n)
            c_half = np.max(np.correlate(h_half, 1.0 - h_half, mode='full')) * dx
            if c_half <= c + 0.001 * lr * 0.5:
                h = h_half
                lr *= 0.8
            else:
                lr *= 0.5

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coordinate_refine(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001]

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        step = 2
        for i in range(0, n, step):
            if time.time() > deadline:
                break
            if i >= n:
                break
            h_orig = h[i]
            for d in deltas:
                for sgn in [1.0, -1.0]:
                    val = h_orig + sgn * d
                    if val < 0.0 or val > 1.0:
                        continue
                    h[i] = val
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] = h_orig
                if improved:
                    break
            if not improved:
                h[i] = h_orig
        if not improved:
            if step > 1:
                step = 1
            else:
                break

    return best_h, best_c

def adam_optimize(h, n, deadline, top_k=30, lr=0.05, beta1=0.9, beta2=0.999, decay=0.99999):
    dx = 2.0 / n
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_lr = lr
    stall = 0

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c = corr.max()
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        top_idx = np.argsort(corr)[-top_k:]
        nbf = n - 1
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = k_idx - nbf
            grad += grad_shift(h, n, k, dx)
        grad /= top_k

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h - step, n)
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        if c_new <= c + 0.01 * cur_lr:
            h = h_new
        else:
            cur_lr *= 0.9
        cur_lr *= decay

    return best_h, best_c
