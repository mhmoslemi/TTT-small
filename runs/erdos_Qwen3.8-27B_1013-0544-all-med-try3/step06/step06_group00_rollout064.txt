def compute_grad(h, p, n, dx):
    """Compute gradient of log-sum-exp objective.
    h: length n
    p: length 2n-1 (softmax probabilities)
    Returns: gradient of length n
    """
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)  # physical shifts
    for k_idx in range(2*n-1):
        j = k_idx - (n-1)  # physical shift
        pj = p[k_idx]
        if pj < 1e-18:
            continue
        # A part: -(1 - h[i+j]) for valid i+j
        # B part: -h[i-j] for valid i-j
        # grad[i] += dx * pj * [(1-h[i+j]) - h[i-j]]
        
        # (1-h[i+j]): i+j in [0,n) => i in [-j, n-j) ∩ [0,n)
        lo_a = max(0, -j)
        hi_a = min(n, n - j)
        if hi_a > lo_a:
            grad[lo_a:hi_a] += pj * (1.0 - h[lo_a + j:hi_a + j])
        
        # -h[i-j]: i-j in [0,n) => i in [j, n+j) ∩ [0,n)
        lo_b = max(0, j)
        hi_b = min(n, n + j)
        if hi_b > lo_b:
            grad[lo_b:hi_b] -= pj * h[lo_b - j:hi_b - j]
    
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_lse_and_p(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = alpha * c - alpha * c.max()
    expt = np.exp(shifted)
    s = expt.sum()
    if s < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = expt / s
    val = c.max() + np.log(s) / alpha
    return val, p

def compute_grad_lse(h, n, p, dx):
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        j = k_idx - (n-1)
        pj = p[k_idx]
        if pj < 1e-18:
            continue
        lo_a = max(0, -j)
        hi_a = min(n, n - j)
        if hi_a > lo_a:
            grad[lo_a:hi_a] += pj * (1.0 - h[lo_a + j:hi_a + j])
        lo_b = max(0, j)
        hi_b = min(n, n + j)
        if hi_b > lo_b:
            grad[lo_b:hi_b] -= pj * h[lo_b - j:hi_b - j]
    grad *= dx
    return grad

def slsqp_analytic(h, n, alpha, maxiter, ftol, deadline):
    dx = 2.0 / n
    def obj(x):
        if x.sum() != n/2.0:
            x = project_feasible(x, n)
        val, _ = compute_lse_and_p(x, n, alpha)
        return val + 0.5 * (x.sum() - n/2.0)**2 * 100
    def obj_grad(x):
        if x.sum() != n/2.0:
            x = project_feasible(x, n)
        val, p = compute_lse_and_p(x, n, alpha)
        g = compute_grad_lse(x, n, p, dx)
        g += 100 * (x.sum() - n/2.0)
        return val, g
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return x.sum() - n/2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    best_x = h.copy()
    best_f = obj(h)
    hs = h
    for it in range(maxiter):

        try:

                best_f = res.fun

def greed_binary(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):

            break

        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:

                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range

                delta[n-1] -= 2.0
