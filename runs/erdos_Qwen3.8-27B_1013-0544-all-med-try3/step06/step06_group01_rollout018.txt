  def compute_swap_delta(h, i, j, n):
      """
      h[i] = 0, h[j] = 1 before swap.
      After swap: h[i] = 1, h[j] = 0.
      Returns the change in the full correlation vector.
      """
      # c[m] = sum_t h[t] * (1 - h[t-k]) where k = m - (n-1)
      # After swap, only terms involving h[i] or h[j] change.
      
      # Terms that change:
      # 1. t=i: h[i]*(1-h[i-k]) changes from 0 to 1*(1-h_new[i-k])
      # 2. t-k=i (t=i+k): h[i+k]*(1-h[i]) changes from h[i+k]*1 to h_new[i+k]*0
      # 3. t=j: h[j]*(1-h[j-k]) changes from 1*(1-h[j-k]) to 0
      # 4. t-k=j (t=j+k): h[j+k]*(1-h[j]) changes from h[j+k]*0 to h_new[j+k]*1
      
      delta = np.zeros(2*n-1)
      k = np.arange(-(n-1), n)  # lag
      m = k + (n-1)  # index in correlation array
      
      # 1. t=i contribution: +1*(1-h_new[i-k]) where i-k valid
      idx_ik = i - k  # = i - (-(n-1)) to i-(n-1) = i+n-1 to i-n+1
      mask = (idx_ik >= 0) & (idx_ik < n)
      # h_new[idx_ik] = h[idx_ik] except: h_new[i]=1, h_new[j]=0
      vals = h[idx_ik[mask]].copy()
      # Adjust for positions where idx_ik == i or idx_ik == j
      ai = mask & (idx_ik == i)
      aj = mask & (idx_ik == j)
      vals[ai] = 1  # h_new[i] = 1
      vals[aj] = 0  # h_new[j] = 0
