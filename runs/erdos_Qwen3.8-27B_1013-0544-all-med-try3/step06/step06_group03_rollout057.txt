import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Fast coordinate descent with incremental correlation updates."""
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    dx = 2.0 / n
    best_c = float(np.max(c) * dx)

            break
        any_change = False
        for p in range(n):
            hp = h[p]
            
            m_lo = max(0, n - 1 - p)
            m_hi = min(n_full - 1, n - 1 + p)
            
            # Unaffected max
            c_unaff = 0.0
            if m_lo > 0:
                c_unaff = max(c_unaff, float(np.max(c[:m_lo])))
            if m_hi < n_full - 1:
                c_unaff = max(c_unaff, float(np.max(c[m_hi + 1:])))

            idx2 = p - ms + (n - 1)

            valid2 = (idx2 >= 0) & (idx2 < n)

            vals = t_grid[:, None] * coeff[None, :] + intercepts[None, :]

import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, max_sweeps=2000, grid_size=200):
    """Fast coordinate descent with incremental correlation updates."""
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    dx = 2.0 / n
    best_c = float(np.max(c) * dx)
    n_full = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_change = False
        for p in range(n):
            hp = h[p]
            m_lo = max(0, n - 1 - p)
            m_hi = min(n_full - 1, n - 1 + p)
            
            c_unaff = 0.0
            if m_lo > 0:
                c_unaff = max(c_unaff, float(np.max(c[:m_lo])))
            if m_hi < n_full - 1:
                c_unaff = max(c_unaff, float(np.max(c[m_hi + 1:])))
            
            ms = np.arange(m_lo, m_hi + 1)
            c_aff = c[ms]
            
            idx1 = p + ms - (n - 1)
            idx2 = p - ms + (n - 1)
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            coeff = np.zeros(ms.shape)
            if np.any(valid1):
                coeff[valid1] += (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                coeff[valid2] += (-h[idx2[valid2]])
            
            if np.all(coeff == 0):
                continue
            
            intercepts = c_aff - coeff * hp
            t_grid = np.linspace(0.0, 1.0, grid_size)
            vals = t_grid[:, None] * coeff[None, :] + intercepts[None, :]
            vals = np.maximum(vals, c_unaff)
            max_vals = np.max(vals, axis=1)
            
            best_idx = np.argmin(max_vals)
            t_new = t_grid[best_idx]
            
            if abs(t_new - hp) > 1e-8:
                delta = t_new - hp
                c[ms] += coeff * delta
                h[p] = t_new
                any_change = True
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full').astype(float)
        curr_c = float(np.max(c) * dx)
        if curr_c < best_c:
            best_c = curr_c
        elif not any_change:
            break
    
    return h, best_c

def binary_init(n, rng):
    """Create a random binary initialization with exactly n//2 ones."""
    h = np.zeros(n)
    n_ones = n // 2
    if n_ones * 2 != n:

    ones_idx = rng.choice(n, n_ones, replace=False)
    h[ones_idx] = 1.0
    h[~np.isin(np.arange(n), ones_idx)] = 0.0
    return h
