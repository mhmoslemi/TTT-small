```
def project_feasible(h, n):
    target = n/2.0
    h = np.array(h, dtype=float, copy=True)
    np.clip(h,0,1,out=h)
    # iterative until diff
    for _ in range(50):
       s=h.sum()
       diff=target-s
       if abs(diff)<1e-13: break
       if diff>0:
          avail=1-h; sum_avail=avail.sum()
          if sum_avail <1e-14: break
          h += (diff/sum_avail)*avail
       else:
          if s<1e-14: break
          h += (diff/s)*h
    np.clip(h,0,1,out=h)
    # final small correction on interior
    s=h.sum(); diff=target-s
    if abs(diff)>1e-12:
       interior = (h>1e-12)&(h<1-1e-12)
       if interior.sum()>0:
           h[interior] += diff
           np.clip(h,0,1,out=h)
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_PRE_CACHE = {}


def compute_c(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def compute_raw(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def exact_sum_adjust(h, n):
    target = 0.5 * float(n)
    res = float(target - h.sum())
    if abs(res) < 1e-15:
        return
    interior = np.flatnonzero((h > 1e-10) & (h < 1.0 - 1e-10))
    for idx in interior[:12]:
        if res > 0.0:
            add = min(res, 1.0 - h[idx])
        else:
            add = max(res, -h[idx])
        h[idx] += add
        res -= float(add)
        if abs(res) < 1e-15:
            break
    np.clip(h, 0.0, 1.0, out=h)


def light_project(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(8):
        diff = float(target - h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            avail = 1.0 - h
            s = float(avail.sum())
            if s < 1e-14:
                break
            h += (diff / s) * avail
        else:
            s = float(h.sum())
            if s < 1e-14:
                break
            h += (diff / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    exact_sum_adjust(h, n)
    return h


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(120):
        diff = float(target - h.sum())

        if diff > 0.0:
            avail = 1.0 - h
            s = float(avail.sum())
            if s < 1e-14:
                break
            h += (diff / s) * avail
        else:
            s = float(h.sum())
            if s < 1e-14:
                break

def build_precomp(n):
    center = n - 1
    L = 2 * n - 1
    dx = 2.0 / float(n)
    empty = np.empty(0, dtype=np.int64)
    pos_left = [empty] * n
    neg_left = [empty] * n
    left_idx = [empty] * n
    pos_right = [empty] * n
    neg_right = [empty] * n
    right_idx = [empty] * n
    for p in range(n):
        if p > 0:
            ar = np.arange(1, p + 1, dtype=np.int64)
            pos_left[p] = center + ar
            neg_left[p] = center - ar
            left_idx[p] = np.arange(p - 1, -1, -1, dtype=np.int64)
        if p < n - 1:
            ar = np.arange(1, n - p, dtype=np.int64)
            pos_right[p] = center + ar
            neg_right[p] = center - ar
            right_idx[p] = np.arange(p + 1, n, dtype=np.int64)
    pre = {
        "n": n,
        "center": center,
        "length": L,
        "dx": dx,
        "lin": np.zeros(L, dtype=float),
        "pos_left": pos_left,
        "neg_left": neg_left,
        "left_idx": left_idx,
        "pos_right": pos_right,
        "neg_right": neg_right,
        "right_idx": right_idx,
    }
    return pre


def get_precomp(n):
    if n in _PRE_CACHE:
        return _PRE_CACHE[n]
    pre = build_precomp(n)
    _PRE_CACHE[n] = pre
    return pre


def add_var_to_lin(p, sign, h, lin, pre):
    center = pre["center"]
    lin[center] += sign * (1.0 - 2.0 * h[p])
    idx = pre["left_idx"][p]
    if idx.size:
        vals = h[idx]
        lin[pre["pos_left"][p]] += sign * (1.0 - vals)
        lin[pre["neg_left"][p]] += -sign * vals
    idx = pre["right_idx"][p]
    if idx.size:
        vals = h[idx]
        lin[pre["pos_right"][p]] += -sign * vals
        lin[pre["neg_right"][p]] += sign * (1.0 - vals)


def apply_move_inplace(h, raw, a, b, delta, pre):
    if delta == 0.0:
        return
    lin = pre["lin"]
    lin.fill(0.0)
    add_var_to_lin(int(a), 1.0, h, lin, pre)
    add_var_to_lin(int(b), -1.0, h, lin, pre)
    raw += delta * lin
    d2 = delta * delta
    center = pre["center"]
    raw[center] -= 2.0 * d2
    m = abs(int(a) - int(b))
    if m > 0:
        raw[center - m] += d2
        raw[center + m] += d2
    h[a] += delta
    h[b] -= delta


def pair_delta_for_lag(h, n, lag, a, b):
    center = n - 1
    r = int(lag) - center
    if r == 0:
        return 0.0
    m = abs(r)
    if m >= n:
        return 0.0
    a = int(a)
    b = int(b)
    ha = float(h[a])
    hb = float(h[b])
    if r > 0:
        H_a = -hb
        if a >= m:
            H_a += 1.0 - h[a - m]
        H_b = 1.0 - ha
        if b + m < n:
            H_b -= h[b + m]
    else:
        H_a = 1.0 - hb
        if a >= m:
            H_a -= h[a - m]
        H_b = 1.0 - ha
        if b + m < n:
            H_b -= h[b + m]
    lin = H_a - H_b
    delta = -0.5 * lin
    dmin = -ha
    dmax = 1.0 - hb
    if delta > dmax:
        delta = dmax
    elif delta < dmin:
        delta = dmin
    if abs(delta) < 1e-11:
        return 0.0
    return float(delta)


def sweep_lag(lag, h, raw, C, n, pre, deadline, fracs, tol, max_moves, top_candidates):

    accepted = 0

    if r == 0:

            return C, 0

                if accepted >= max_moves:
                    break

                bi = int(bi)
                if ai == bi:

                if h[ai] <= h[bi]:
                    continue

                        accepted += 1
                        break
                    else:

                if accepted >= max_moves:
                    break
        return C, accepted

    m = abs(r)
    if m >= n:
        return C, 0
    cands = []
    hm = h
    if r > 0:
        for i in range(n - m):
            b = i + m
            ha = hm[i]
            hb = hm[b]
            H_a = -hb
            if i >= m:

            H_b = 1.0 - ha
            if b + m < n:
                H_b -= hm[b + m]
            lin = H_a - H_b
            delta = -0.5 * lin
            dmin = -ha
            dmax = 1.0 - hb
            if delta > dmax:
                delta = dmax
            elif delta < dmin:

            if abs(delta) < 1e-9:

            if dec > 1e-11:

    else:
        for i in range(n - m):
            b = i + m
            ha = hm[i]
            hb = hm[b]
            H_a = 1.0 - hb
            if i >= m:
                H_a -= hm[i - m]
            H_b = 1.0 - ha
            if b + m < n:
                H_b -= hm[b + m]
            lin = H_a - H_b
            delta = -0.5 * lin
            dmin = -ha
            dmax = 1.0 - hb
            if delta > dmax:
                delta = dmax
            elif delta < dmin:
                delta = dmin
            if abs(delta) < 1e-9:
                continue
            dec = -(delta * lin + delta * delta)
            if dec > 1e-11:
                cands.append((float(dec), int(i), int(b), float(delta)))
    if not cands:
        return C, 0
    cands.sort(reverse=True)
    limit = min(top_candidates, len(cands))

        if accepted >= max_moves or time.time() >= deadline:

def pair_local_search(h, n, deadline, tol=1e-13, active_K=None):

    raw = compute_raw(h, n)

                break

            continue
        no_cycles = 0

def add_lag_grad(h, grad, lag, n, center, w):
    r = int(lag) - center
    m = abs(r)
    if m == 0:
        grad += w * (1.0 - 2.0 * h)
    elif r > 0:
        grad[:n - m] -= w * h[m:]
        grad[m:] += w * (1.0 - h[:n - m])
    else:
        grad[:n - m] += w * (1.0 - h[m:])
        grad[m:] -= w * h[:n - m]


def gradient_active_search(h, n, deadline):

    best_C = C

    h = light_project(h, n)
    C = compute_c(h, n)
    if C < best_C:
        best_C = C

def _sum_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def _sum_eq_jac(x, n):
    return np.ones((1, n), dtype=float)


def _active_smooth_obj(x, n, alpha, idx):
    raw = np.correlate(x, 1.0 - x, mode="full")
    a = raw[idx]
    am = float(a.max())
    sh = alpha * (a - am)
    return float(am + np.log(np.sum(np.exp(sh))) / alpha)


def slsqp_active(h, n, deadline):
    h = light_project(np.array(h, dtype=float, copy=True), n)
    C = compute_c(h, n)
    best_h = h.copy()
    best_C = C
    alphas = (40.0, 400.0, 4000.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_eq, "jac": _sum_eq_jac, "args": (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        raw = compute_raw(h, n)
        idx = np.argsort(raw)[-6:]
        try:
            res = minimize(
                _active_smooth_obj,
                h,
                args=(n, float(alpha), idx),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 90, "ftol": 1e-12},
            )
            hnew = project_feasible(res.x, n)
            Cnew = compute_c(hnew, n)
            if Cnew < C - 1e-14:
                h = hnew
                C = Cnew
                if C < best_C:
                    best_C = C
                    best_h = h.copy()
        except Exception:
            pass
    h = light_project(h, n)
    C = compute_c(h, n)
    if C < best_C:
        best_C = C
        best_h = h.copy()
    return best_h, best_C


def micro_random_search(h, n, deadline, seed=12345):
    rng = np.random.RandomState(int(seed) % (2**32 - 1))
    h = light_project(np.array(h, dtype=float, copy=True), n)
    raw = compute_raw(h, n)
    dx = 2.0 / float(n)
    C = float(raw.max() * dx)
    best_h = h.copy()
    best_C = C
    pre = get_precomp(n)
    T = 5e-4
    scale = 0.06
    moves = 0

    while time.time() < deadline:
        a = int(rng.randint(0, n))
        b = int(rng.randint(0, n))
        if a == b:
            continue
        d = (float(rng.rand()) * 2.0 - 1.0) * scale
        dmin = -h[a]
        dmax = 1.0 - h[b]
        if d > dmax:
            d = dmax
        if d < dmin:
            d = dmin
        if abs(d) < 1e-10:
            continue
        apply_move_inplace(h, raw, a, b, float(d), pre)
        Cnew = float(raw.max() * dx)
        if Cnew < C - 1e-14:
            C = Cnew
            if C < best_C:
                best_C = C
                best_h = h.copy()
        else:
            accept = False
            if Cnew <= C + T:
                diff = Cnew - C
                if diff <= 0.0:
                    accept = True
                else:
                    if T > 1e-15:
                        prob = float(np.exp(-diff / T))
                    else:
                        prob = 0.0
                    accept = float(rng.rand()) <= prob
            if accept:
                C = Cnew
            else:
                apply_move_inplace(h, raw, a, b, -float(d), pre)
                C = max(C, float(raw.max() * dx))
        moves += 1
        if moves % 5000 == 0:
            T *= 0.94
            scale *= 0.985
            h = light_project(h, n)
            raw = compute_raw(h, n)
            C = float(raw.max() * dx)
            if C > best_C + 1e-5:
                h = best_h.copy()
                raw = compute_raw(h, n)
                C = best_C

    h = light_project(h, n)
    raw = compute_raw(h, n)
    C = float(raw.max() * dx)
    if C < best_C:
        best_C = C
        best_h = h.copy()
    return best_h, best_C


def resample_h(h, n, new_n):
    x_old = np.linspace(0.0, 2.0, int(n))
    x_new = np.linspace(0.0, 2.0, int(new_n))
    return np.interp(x_new, x_old, np.asarray(h, dtype=float))


def make_starts(h_base, n, rng):
    starts = []
    for scale in (0.010, 0.020, 0.045, 0.080):
        h = h_base + rng.normal(0.0, float(scale), n)
        starts.append(light_project(h, n))
    x = np.linspace(0.0, 2.0, n)
    for _ in range(4):
        h = np.full(n, 0.5, dtype=float)
        nterms = int(rng.randint(2, 5))
        for _j in range(nterms):
            freq = float(int(rng.randint(1, 9)))
            ph = 2.0 * np.pi * float(rng.rand())
            amp = 0.35 * float(rng.uniform(0.25, 1.0)) / np.sqrt(max(1.0, freq))
            h = h + amp * np.sin(2.0 * np.pi * freq * x + ph)
            if float(rng.rand()) < 0.55:
                h = h + 0.7 * amp * np.cos(2.0 * np.pi * freq * x + 0.5 * ph)
        h = h + rng.normal(0.0, 0.03, n)
        starts.append(light_project(h, n))
    if n > 1:
        starts.append(light_project(h_base[::-1], n))
    return starts


def refine_candidate(h_start, n, end, use_slsqp=False):
    h = project_feasible(np.array(h_start, dtype=float, copy=True), n)
    C = compute_c(h, n)
    if time.time() >= end:
        return h, C

    now = time.time()
    total = max(1.0, end - now)
    end1 = min(end, now + 0.45 * total)
    if time.time() < end1:
        h, C = pair_local_search(h, n, end1)

    if time.time() < end:
        total2 = max(1.0, end - time.time())
        end2 = min(end, time.time() + 0.25 * total2)
        if time.time() < end2:
            h, C = gradient_active_search(h, n, end2)

    if time.time() < end:
        total3 = max(1.0, end - time.time())
        end3 = min(end, time.time() + 0.30 * total3)
        if time.time() < end3:
            h, C = pair_local_search(h, n, end3)

    if use_slsqp and n <= 450 and time.time() < end - 25:
        end4 = min(end, time.time() + 45.0)
        h, C = slsqp_active(h, n, end4)
        if time.time() < end:
            h, C = pair_local_search(h, n, end)

    if time.time() < end:
        h, C = pair_local_search(h, n, end)

    h = project_feasible(h, n)
    C = compute_c(h, n)
    return h, C


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 6.0
    if deadline <= t0 + 60.0:
        deadline = t0 + float(budget_s) - 2.0
    rng = np.random.RandomState(int(seed))

    try:
        hp = np.asarray(initial_h_values, dtype=float)
        n0 = len(hp)
        if n0 < 20:
            n0 = 300
            hp = np.full(n0, 0.5, dtype=float)
    except Exception:
        hp = None
        n0 = 300

    if hp is None:
        hp = np.full(n0, 0.5, dtype=float)

    h_base = project_feasible(hp.copy(), n0)
    best_h = h_base.copy()
    best_C = compute_c(best_h, n0)

    if time.time() < deadline - 100.0:
        parent_frac = 0.38 * max(60.0, float(budget_s) - 60.0)
        parent_cap = 400.0
        parent_end = min(deadline - 120.0, t0 + min(parent_cap, parent_frac))
        if parent_end < time.time() + 25.0:
            parent_end = min(deadline - 100.0, time.time() + 120.0)
        hopt, copt = refine_candidate(h_base, n0, parent_end, use_slsqp=True)
        if copt < best_C:
            best_C = copt
            best_h = hopt

    if time.time() < deadline - 90.0:
        start_list = []
        for sh in make_starts(best_h, n0, rng):
            sh = light_project(sh, n0)
            sc = compute_c(sh, n0)
            if sc <= best_C + 0.09:
                start_list.append((sc, sh))
            if len(start_list) >= 8:
                break
        if start_list:
            cvals = np.empty(len(start_list), dtype=float)
            for ii, it in enumerate(start_list):
                cvals[ii] = it[0]
            order = np.argsort(cvals)
            for oi in order:
                if time.time() > deadline - 80.0:
                    break
                ii = int(oi)
                _, sh = start_list[ii]
                alloc = 50.0
                remaining_alloc = deadline - time.time() - 80.0
                if alloc > remaining_alloc:
                    alloc = remaining_alloc
                if alloc < 25.0:
                    break
                hend = time.time() + alloc
                hopt, copt = refine_candidate(sh, n0, hend, use_slsqp=False)
                if copt < best_C:
                    best_C = copt
                    best_h = hopt

    if best_h is not None and len(best_h) < 500 and time.time() < deadline - 140.0:
        n2 = 600
        h2 = resample_h(best_h, len(best_h), n2)
        h2 = project_feasible(h2, n2)
        c2 = compute_c(h2, n2)
        if c2 <= best_C + 0.04:
            halloc = min(deadline - 70.0, time.time() + 160.0)
            h2, c2 = refine_candidate(h2, n2, halloc, use_slsqp=False)
            if c2 < best_C:
                best_C = c2
                best_h = h2

    if best_h is not None:
        nb = len(best_h)
        if nb <= 450 and time.time() < deadline - 60.0:
            sl_end = min(deadline - 30.0, time.time() + 55.0)
            h_s, c_s = slsqp_active(best_h, nb, sl_end)
            if c_s < best_C:
                best_C = c_s
                best_h = h_s
        if time.time() < deadline - 25.0:
            rand_end = min(deadline - 15.0, time.time() + 35.0)
            h_r, c_r = micro_random_search(best_h, nb, rand_end, seed=int(seed) + 777)
            if c_r < best_C:
                best_C = c_r
                best_h = h_r
        if time.time() < deadline - 5.0:
            pair_end = min(deadline - 2.0, time.time() + 35.0)
            h_p, c_p = pair_local_search(best_h, nb, pair_end)
            if c_p < best_C:
                best_C = c_p
                best_h = h_p

    if best_h is None:
        best_h = project_feasible(np.full(max(n0, 30), 0.5), max(n0, 30))
        best_C = compute_c(best_h, len(best_h))

    best_h = project_feasible(best_h, len(best_h))
    best_C = compute_c(best_h, len(best_h))
    return best_h, float(best_C), int(len(best_h))
```
