def block_search(h, n, deadline, block_sizes=[2, 3, 5]):
    """Try moving blocks of consecutive points."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for bs in block_sizes:
        for _ in range(50):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            for start in range(0, n - bs + 1):
                if time.time() > deadline:
                    return best_h, best_c
                # Try moving mass between block [start, start+bs) and a random other block
                end = start + bs
                block_mass = h[start:end].sum()
                if block_mass < 1e-10:
                    continue
                
                # Find a target block
                for t_start in range(0, n - bs + 1, bs):
                    if t_start >= start and t_start < end:
                        continue
                    t_end = t_start + bs
                    if t_end > n:
                        continue
                    
                    # Try moving a fraction of mass
                    for frac in [0.1, 0.25, 0.5]:
                        d = block_mass * frac / bs
                        # Check feasibility
                        if np.any(h[start:end] - d < 0) or np.any(h[t_start:t_end] + d > 1):
                            continue
                        h[start:end] -= d
                        h[t_start:t_end] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[start:end] += d
                            h[t_start:t_end] -= d
            
            if not improved:
                break
    
    return best_h, best_c

Another idea: **exploiting the structure of the correlation**. The correlation `np.correlate(h, 1-h, mode='full')[m]` for m = n-1+k (k≥0) is:
sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])

= sum_{i=0}^{n-1-k} h[i+k] - sum_{i=0}^{n-1-k} h[i+k]*h[i]

The first term is linear, the second is quadratic. The max is over all k from -(n-1) to n-1.

For the symmetric case k → -k:
sum_{i=0}^{n-1-k} h[i] * (1-h[i-k]) = sum_{i=k}^{n-1} h[i] - sum_{i=k}^{n-1} h[i]*h[i-k]

So the problem has a certain symmetry between positive and negative lags, but they're not identical.

import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max_full(h, n, alpha):
    c = compute_corr(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max_full(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, deadline, alpha=500, mu=10, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    try:
        res = minimize(penalty_obj, h, jac=penalty_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def guided_pair_search(h, n, deadline, top_k=40, base_step=0.01):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = base_step
    
    for iteration in range(10000):
        if time.time() > deadline:
            break
        
        g = grad_smooth_max_full(h, n, 200.0)
        order = np.argsort(g)
        
        tried_pairs = set()
        improved = False
        
        # Build candidate pairs: high grad with low grad
        candidates = []
        for idx_p in range(n - top_k, n):
            for idx_n in range(0, top_k):
                if idx_p != idx_n:
                    pair = (idx_p, idx_n)
                    if pair not in tried_pairs:
                        candidates.append(pair)
                        tried_pairs.add(pair)
        
        np.random.shuffle(candidates)
        
        for i, j in candidates:
            if time.time() > deadline:
                return best_h, best_c
            
            # Direction: move mass to reduce objective
            # df/dd ≈ g[i] for h[i] increase, g[j] for h[j] increase
            # For h[i]+d, h[j]-d: change ≈ d*(g[i] - g[j])
            # To decrease: if g[i] > g[j], use d < 0 (i.e., h[i]-|d|, h[j]+|d|)
            
            for mult in [1.0, 0.5, 0.2, 0.1, 0.05]:
                d = step * mult
                # Try both directions
                for sign in [1, -1]:

                    if h[i] + dd < 0 or h[i] + dd > 1:

                    if h[j] - dd < 0 or h[j] - dd > 1:

                    h[i] += dd

                        best_h = h.copy()
                        improved = True
                        break

                        h[i] -= dd

                if improved:

            else:
