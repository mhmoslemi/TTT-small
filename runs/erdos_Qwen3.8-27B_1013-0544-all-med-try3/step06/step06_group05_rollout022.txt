    c = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * c
    shifted -= shifted.max()

    weights /= weights.sum()

    # dc[m]/dh[i] = (1-h[i+m-(n-1)]) * 1_{valid} - h[i-m+(n-1)] * 1_{valid}

        i_start = max(0, n-1-m)
        i_end = min(n, n-1-m+n)  # = min(n, 2n-1-m)

    h, c = gradient_optimize(h, n, grad_deadline)
    if time.time() < deadline - 5:
        if n <= 350:
            h, c = pair_search(h, n, deadline - 2,
                               [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005], max_passes=250)
        else:
            h, c = near_pair_search(h, n, deadline - 2,
                                    [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=60)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(250, 0.5)
        n_parent = 250
    best_h = None
    best_c = float('inf')
    x_parent = np.linspace(0, 2, n_parent)
    n_list = [250, 400, 550]
    allocs = [0.38, 0.30, 0.22]
    available = deadline - t0 - 80
    for idx, n in enumerate(n_list):
        if time.time() > deadline - 80:
            break
        alloc = available * allocs[idx]
        sub_dl = time.time() + alloc
        if n == n_parent:
            h_start = h_parent.copy()
        else:
            h_start = np.interp(np.linspace(0, 2, n), x_parent, h_parent)
        h_opt, c_opt = optimize_at_resolution(h_start, n, sub_dl, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        scales = [0.008, 0.02, 0.04, 0.06]
        for scale in scales:
            if time.time() > deadline - 25:
                break
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001], max_passes=500)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.0005, 0.0002, 0.0001, 0.00005], radius=80)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)
        best_c = compute_c5(best_h, n_parent)
    return best_h, best_c, len(best_h)
