from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + np.log(np.exp(sh - sm + 1e-300).sum()) / alpha
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def simulated_annealing(h, n, deadline, t_initial, t_final, seed=42, max_iters=None):
    """SA on binary representation."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    k_range = np.arange(-(n-1), n).astype(int)
    
    h = h.copy()
    if max_iters is None:
        max_iters = 10_000_000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Compute temperature (geometric cooling)
        if it < max_iters:
            T = t_initial * (t_final / t_initial) ** (it / max_iters)
        else:
            T = t_final
        
        # Pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        i_int = int(i)
        j_int = int(j)
        
        # Compute delta
        idx1 = i_int - k_range
        valid1 = (idx1 >= 0) & (idx1 < n)
        clamped1 = np.clip(idx1, 0, n-1)
        t1 = np.where(valid1, 1.0 - h[clamped1], 0.0)
        
        idx2 = i_int + k_range
        valid2 = (idx2 >= 0) & (idx2 < n)
        clamped2 = np.clip(idx2, 0, n-1)
        t2 = np.where(valid2, -h[clamped2], 0.0)
        
        idx3 = j_int - k_range
        valid3 = (idx3 >= 0) & (idx3 < n)
        clamped3 = np.clip(idx3, 0, n-1)
        t3 = np.where(valid3, -(1.0 - h[clamped3]), 0.0)
        
        idx4 = j_int + k_range
        valid4 = (idx4 >= 0) & (idx4 < n)
        clamped4 = np.clip(idx4, 0, n-1)
        t4 = np.where(valid4, h[clamped4], 0.0)
        
        delta = t1 + t2 + t3 + t4
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        
        # Accept?
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
            h[i_int], h[j_int] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c

def random_restarts_search(h_init, n, deadline, n_restarts=5, seed=42):

        if c_opt < best_c:

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + np.log(np.exp(sh - sm + 1e-300).sum()) / alpha
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def steepest_descent(h, n, max_sweeps, deadline):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    kr = np.arange(-(n-1), n)  # integer array of lags
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Precompute term_ones (n_ones, 2n-1)
        o2d = ones.astype(np.int32)[:, None]
        kr2d = kr.astype(np.int32)[None, :]
        
        idx3 = o2d - kr2d
        v3 = (idx3 >= 0) & (idx3 < n)
        t3 = np.where(v3, 0.0, 0.0)
        t3[v3] = -(1.0 - h[np.clip(idx3[v3], 0, n-1)])
        
        idx4 = o2d + kr2d
        v4 = (idx4 >= 0) & (idx4 < n)
        t4 = np.where(v4, 0.0, 0.0)
        t4[v4] = h[np.clip(idx4[v4], 0, n-1)]
        
        term_ones = t3 + t4
        
        best_nm = cur_c
        best_i = -1
        best_j = -1
        best_delta = None
        
        for ii in range(n_zeros):
            i = int(zeros[ii])
            idx1 = i - kr
            v1 = (idx1 >= 0) & (idx1 < n)
            t1 = np.zeros(2*n - 1)
            t1[v1] = 1.0 - h[np.clip(idx1[v1], 0, n-1)]
            
            idx2 = i + kr
            v2 = (idx2 >= 0) & (idx2 < n)
            t2 = np.zeros(2*n - 1)
            t2[v2] = -h[np.clip(idx2[v2], 0, n-1)]
            
            term_i = t1 + t2
            
            # (n_ones, 2n-1)
            nc = corr[None, :] + term_i[None, :] + term_ones
            nmax = nc.max(axis=1)
            bj = nmax.argmin()
            if nmax[bj] < best_nm - 1e-14:
                if nmax[bj] >= cur_c - 1e-10:
                    # No meaningful improvement
                    pass
                best_nm = nmax[bj]
                best_i = i
                best_j = int(ones[bj])
                best_delta = term_i + term_ones[bj]
        
        if best_i < 0:
            break
        if best_nm >= cur_c - 1e-12:
            break
        
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr += best_delta
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def sa_search(h, n, deadline, t_init, t_final, seed=42, max_iters=50000000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    kr = np.arange(-(n-1), n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_init * (t_final / t_init) ** (float(it) / float(max_iters))
        if T < 1e-15:
            T = 1e-15
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(zeros[rng.randint(len(zeros))])
        j = int(ones[rng.randint(len(ones))])
        
        # Compute delta
        idx1 = i - kr
        v1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.zeros(2*n - 1)
        t1[v1] = 1.0 - h[np.clip(idx1[v1], 0, n-1)]
        
        idx2 = i + kr
        v2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.zeros(2*n - 1)
        t2[v2] = -h[np.clip(idx2[v2], 0, n-1)]
        
        idx3 = j - kr
        v3 = (idx3 >= 0) & (idx3 < n)
        t3 = np.zeros(2*n - 1)
        t3[v3] = -(1.0 - h[np.clip(idx3[v3], 0, n-1)])
        
        idx4 = j + kr
        v4 = (idx4 >= 0) & (idx4 < n)
        t4 = np.zeros(2*n - 1)
        t4[v4] = h[np.clip(idx4[v4], 0, n-1)]
        
        delta = t1 + t2 + t3 + t4
        new_c = (corr + delta).max() * dx
        
        if new_c <= cur_c:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = new_c
    
    return best_h, best_c
