def solve_lp(h, n, corr, grads):

    # s.t. corr[k] + grads[k,:] @ (x - h) <= t  for all k

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s_room = room.sum()
            if s_room < 1e-14:
                break
            h += diff / s_room * room
        else:
            s_h = h.sum()
            if s_h < 1e-14:
                break
            h += diff / s_h * h
    return np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    """Compute gradient matrix (2n-1, n) for raw correlations."""
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        j_lo1 = max(0, -s)
        j_hi1 = min(n, n - s)
        if j_hi1 > j_lo1:
            grads[k, j_lo1:j_hi1] += (1.0 - h[j_lo1 + s : j_hi1 + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grads[k, j_lo2:j_hi2] -= h[j_lo2 - s : j_hi2 - s]
    return grads

def lp_step(h, n):
    """One LP cutting-plane step. Returns new h or None if no improvement direction."""
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    raw_C = raw_corr.max()
    
    grads = compute_grads(h, n)
    
    # LP: min t s.t. raw_corr[k] + grads[k]@(x-h) <= t, 0<=x<=1, sum(x)=n/2
    n_k = 2 * n - 1

    c_obj[-1] = 1.0

    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]

                  bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    return res.x[:n]

    C = compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16, deadline=None):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

        return h
    except Exception:
        return h

    n_k = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.empty((n_k, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = raw - grads.dot(h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, direction, n, current_C):
    """Find best step alpha in [0,1]: h_new = (1-a)*h + a*direction."""
    best_a = 0.0
    best_C = current_C
    # Coarse
    for a in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        h_new = (1 - a) * h + a * direction
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_C:
            best_C = c_new
            best_a = a
    # Fine around best
    if best_a > 0:
        lo = max(0, best_a - 0.15)
        hi = min(1, best_a + 0.15)
        for a in np.linspace(lo, hi, 11):
            h_new = (1 - a) * h + a * direction
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_C:
                best_C = c_new
                best_a = a
    return best_a, best_C

def cut_plane_optimize(h, n, max_iter=100, deadline=None):
    """LP cutting-plane optimization loop."""
    h = project_feasible(h, n).copy()
    C = compute_c5(h, n)
    best_h = h.copy()
    best_C = C
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        x_lp = lp_cutting_step(h, n)
        if x_lp is None:
            break
        if np.max(np.abs(x_lp - h)) < 1e-7:
            break
        
        a, C_new = line_search(h, x_lp, n, C)
        if a == 0 or C_new >= C - 1e-10:
            break
        
        h = (1 - a) * h + a * x_lp
        h = project_feasible(h, n)
        C = compute_c5(h, n)
        if C < best_C:
            best_C = C
            best_h = h.copy()
    
    return best_h, best_C
