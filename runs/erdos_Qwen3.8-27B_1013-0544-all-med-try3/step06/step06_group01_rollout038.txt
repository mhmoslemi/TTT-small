def compute_swap_delta(h, i, j, n):
    delta = np.zeros(2*n-1)
    
    for k in range(-(n-1), n):
        # Find all affected n values for this k
        affected = set()
        if 0 <= i < n and 0 <= i+k < n:
            affected.add(i)
        if 0 <= j < n and 0 <= j+k < n:
            affected.add(j)
        n_b = i - k
        if 0 <= n_b < n and 0 <= n_b+k < n:
            affected.add(n_b)
        n_d = j - k
        if 0 <= n_d < n and 0 <= n_d+k < n:
            affected.add(n_d)
        
        for n_val in affected:
            nk = n_val + k
            old_c = h[nk] * (1 - h[n_val])
            new_nk = 1.0 if nk == i else (0.0 if nk == j else h[nk])
            new_nv = 1.0 if n_val == i else (0.0 if n_val == j else h[n_val])
            new_c = new_nk * (1 - new_nv)
            delta[n-1+k] += new_c - old_c
    
    return delta

            if s < 1e-15:

    c[k] = sum_n h[n+k]*(1-h[n]) for valid n. Index in result: n-1+k."""

        else:
            delta[idx] += -h[ik]

    k_hi_b = min(n - 1, i)
    skip_b = i - j
    for k in range(k_lo_b, k_hi_b + 1):
        if k == 0:
            continue
        if k == skip_b:
            continue
        idx = n - 1 + k
        n_val = i - k
        if n_val == j:
            delta[idx] += 1.0
        else:
            delta[idx] += 1.0 - h[n_val]
    k_lo_c = max(-(n - 1), -j)
    k_hi_c = min(n - 1, n - 1 - j)
    for k in range(k_lo_c, k_hi_c + 1):
        idx = n - 1 + k
        jk = j + k
        if jk == i:
            delta[idx] += 1.0
        else:
            delta[idx] += h[jk]
    k_lo_d = max(-(n - 1), j - n + 1)
    k_hi_d = min(n - 1, j)
    skip_d = j - i
    for k in range(k_lo_d, k_hi_d + 1):
        if k == 0:
            continue
        if k == skip_d:
            continue
        idx = n - 1 + k
        n_val = j - k
        if n_val == i:
            delta[idx] += 0.0
        else:
            delta[idx] += -(1.0 - h[n_val])
    return delta

def binary_local_search(h_init, n, deadline, n_start=3, K=200, tabu_len=50, cool_rate=0.95, T_init=0.01):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c = float(np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx)
    rng = np.random.RandomState(12345)
    
    for start in range(n_start):
        if time.time() > deadline:
            break
        if start == 0:
            h = h_init.copy()
        elif start == 1:
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            fracs = np.array([((i * phi) % 1.0) for i in range(n)])
            order = np.argsort(fracs)
            h = np.zeros(n)
            h[order[:n // 2]] = 1.0
        else:
            h = best_h.copy()
            for _ in range(rng.randint(3, 8)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a, b = int(rng.choice(oi)), int(rng.choice(zi))
                    h[a], h[b] = 0.0, 1.0
        
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_c = float(corr.max() * dx)
        T = T_init * (1.0 + start * 0.5)
        tabu = set()
        no_improve = 0
        
        while time.time() < deadline:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            best_swap = None
            best_new_c = cur_c + 1e-14
            best_new_corr = None
            
            sample_K = min(K, len(zeros) * len(ones))
            si = rng.choice(zeros, size=sample_K, replace=False)
            sj = rng.choice(ones, size=sample_K, replace=False)
            
            for idx_s in range(sample_K):
                i = int(si[idx_s])
                j = int(sj[idx_s])
                key = (i, j)
                if key in tabu:
                    continue
                delta = swap_delta(h, i, j, n)
                new_corr = corr + delta
                nc = float(new_corr.max() * dx)
                if nc < best_new_c:
                    best_new_c = nc
                    best_swap = (i, j)
                    best_new_corr = new_corr
            
            if best_swap is None:
                if T > 1e-8 and rng.random() < 0.05:
                    i = int(rng.choice(zeros))
                    j = int(rng.choice(ones))
                    key = (i, j)
                    if key not in tabu:
                        delta = swap_delta(h, i, j, n)
                        new_corr = corr + delta
                        nc = float(new_corr.max() * dx)
                        delta_e = nc - cur_c
                        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(T, 1e-12)):
                            best_swap = (i, j)
                            best_new_c = nc
                            best_new_corr = new_corr
            
            if best_swap is None:
                T *= cool_rate
                no_improve += 1
                if no_improve > 100:
                    break
                continue
            
            i, j = best_swap
            h[i] = 1.0
            h[j] = 0.0
            corr = best_new_corr
            cur_c = best_new_c
            
            tabu.add((i, j))
            tabu.add((j, i))
            if len(tabu) > tabu_len * 2:
                items = list(tabu)
                for t in items[:len(items) - tabu_len * 2]:
                    tabu.discard(t)
            
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 300:
                break
            T *= cool_rate
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        sh_clipped = np.clip(sh - sm, -700, 0)
        return m + (sm + np.log(np.exp(sh_clipped).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def golden_ratio_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    fracs = np.array([((i * phi) % 1.0) for i in range(n)])
    order = np.argsort(fracs)
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    return h

def interpolated_init(n_from, h_from, n_to):
    x_f = np.linspace(0, 2, n_from)
    x_t = np.linspace(0, 2, n_to)
    h_interp = np.interp(x_t, x_f, h_from)
    return project_feasible(h_interp, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = golden_ratio_init(n0)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0
    
    # Phase 1: SLSQP from parent (10%)
    p1 = t0 + 0.10 * (budget_s - 5)
    h_work = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p1:
            break
        h_work = slsqp_smooth(h_work, n0, alpha, 600, p1)
        c = compute_c5(h_work, n0)
        if c < best_c:
            best_c = c
            best_h = h_work.copy()
            n_best = n0
    
    # Phase 2: Binary local search main (45%)
    p2 = t0 + 0.55 * (budget_s - 5)
    h_bin = to_binary(best_h, n0)
    hb, cb = binary_local_search(h_bin, n0, p2, n_start=4, K=300, tabu_len=100,
                                  cool_rate=0.97, T_init=0.005)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        n_best = n0
    
    # Phase 3: SLSQP polish (8%)
    p3 = t0 + 0.63 * (budget_s - 5)
    if time.time() < p3:
        h_pol = best_h.copy()
        for alpha in [5000, 50000, 500000]:
            if time.time() > p3:
                break
            h_pol = slsqp_smooth(h_pol, n_best, alpha, 1500, p3)
            c = compute_c5(h_pol, n_best)
            if c < best_c:
                best_c = c
                best_h = h_pol.copy()
    
    # Phase 4: Multi-resolution (22%)
    p4 = t0 + 0.85 * (budget_s - 5)
    resolutions = [300, 200, 350, 400, 280, 320]
    for nh in resolutions:
        if time.time() > p4 - 15:
            break
        if nh == n_best:
            continue
        try:
            hh = interpolated_init(n_best, best_h, nh)
        except Exception:
            continue
        hh = slsqp_smooth(hh, nh, 5000, 300, p4 - 10)
        ch = compute_c5(hh, nh)
        hb2 = to_binary(hh, nh)
        hb2, cb2 = binary_local_search(hb2, nh, p4 - 5, n_start=3, K=150,
                                        tabu_len=50, cool_rate=0.96, T_init=0.003)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n_best = nh
        elif ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n_best = nh
    
    # Phase 5: Extended binary search (10%)
    p5 = deadline - 8
    if time.time() < p5:
        h_bin_f = to_binary(best_h, n_best)
        hb, cb = binary_local_search(h_bin_f, n_best, p5, n_start=2, K=250,
                                      tabu_len=80, cool_rate=0.98, T_init=0.002)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            n_best = n_best
    
    # Final SLSQP
    if time.time() < deadline - 2:
        h_fin = slsqp_smooth(best_h, n_best, 2000000, 500, deadline - 1)
        cf = compute_c5(h_fin, n_best)
        if cf < best_c:
            best_c = cf
            best_h = h_fin.copy()
    
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
