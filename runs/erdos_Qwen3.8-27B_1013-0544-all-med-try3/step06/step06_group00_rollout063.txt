import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max_eval(h, n, alpha):
    """Evaluate smooth max of correlate(h, 1-h, 'full') * dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    val = m + (max_s + np.log(np.exp(shifted - max_s).sum())) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    """Evaluate (F, grad) for smooth max."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    p = exp_s / exp_s.sum()
    F = m + (max_s + np.log(exp_s.sum())) / alpha
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for ell in range(-(n-1), n):
        k = ell + (n-1)
        pk = p[k]
        if pk < 1e-18:
            continue
        if ell >= 0:
            grad[ell:] += pk * one_minus_h[:n-ell]
            grad[:n-ell] -= pk * h[ell:]
        else:
            m_ = -ell
            grad[:n-m_] += pk * one_minus_h[m_:]
            grad[m_:] -= pk * h[:n-m_]
    grad *= dx
    return F, grad

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if abs(current - target) < 1e-12:
        return h
    # Bisection for lambda
    if current > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

def projected_gradient(h, n, alpha, max_iter, lr_initial, deadline):
    """Projected gradient descent with backtracking line search."""
    h = h.copy()
    dx = 2.0 / n
    F, grad = smooth_max_grad(h, n, alpha)
    best_F = F
    best_h = h.copy()
    lr = lr_initial
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Projected gradient step with backtracking
        for _ in range(30):

            F_try = smooth_max_eval(h_try, n, alpha)
            if F_try <= F - 0.5 * lr * np.dot(grad, grad - (smooth_max_grad(h_try, n, alpha)[1] - grad)) or F_try < F:

        h = h_try
        F, grad = smooth_max_grad(h, n, alpha)
        if F < best_F - 1e-15:
            best_F = F
            best_h = h.copy()
