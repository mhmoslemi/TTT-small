def grad_single_lag(h, n, k_idx):
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        grad[k:] = (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
        grad *= dx
    else:
        k2 = -k
        grad[:n - k2] = (1.0 - h[k2:])
        grad[k2:] -= h[:n - k2]
        grad *= dx
    return grad

def grad_weighted(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            k2 = -k
            grad[:n - k2] += w * (1.0 - h[k2:])
            grad[k2:] -= w * h[:n - k2]

    # Phase 6: One more Nesterov pass with high alpha
    if time.time() < deadline - 3:
        best_h, c_new = nesterov_optimize(best_h, n, deadline, alpha=200000.0, lr_init=0.0005, n_max_steps=5000)
        if c_new < best_c:
            best_c = c_new
            best_h = best_h  # nesterov returns best
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
