   from scipy.optimize import minimize
   import time

   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)

   def project_feasible(h, n):
       """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
       target = n / 2.0
       h = np.clip(h, 0.0, 1.0)
       for _ in range(10000):
           diff = target - h.sum()
           if abs(diff) < 1e-15:
               break
           if diff > 0:
               room = 1.0 - h
               s = room.sum()
               if s < 1e-15:
                   break
               h = h + diff / s * room
           else:
               s = h.sum()
               if s < 1e-15:
                   break
               h = h + diff / s * h
       return np.clip(h, 0.0, 1.0)

   def lse_objective(x, n, alpha):
       """Compute log-sum-exp smoothed max and gradient."""
       dx = 2.0 / n
       c = np.correlate(x, 1 - x, mode='full') * dx
       m = c.max()
       sh = alpha * (c - m)
       sm = sh.max()
       exp_sh = np.exp(sh - sm)
       sum_exp = exp_sh.sum()
       f = m + (sm + np.log(sum_exp)) / alpha
       
       # Weights (softmax)
       w = exp_sh / sum_exp  # shape (2n-1,)
       
       # Gradient: grad[i] = (dx/alpha) * sum_k w[k] * ((1-x[i+k]) - x[i-k])
       # = (dx/alpha) * sum_k w[k] * (1 - 2*x[i+k] + x[i-k] - x[i-k])  ... no
       # Actually: (1-x[i+k]) - x[i-k] ... these have different indices
       
       # Let me just compute it directly
       grad = np.zeros(n)
       for k_idx in range(2*n-1):
           k = k_idx - (n-1)
           wk = w[k_idx] * dx / alpha
           if abs(wk) < 1e-18:
               continue
           if k >= 0:
               # (1-x[i+k]) for i in [0, n-1-k]: x[k:n]
               grad[:n-k] += wk * (1.0 - x[k:])
               # -x[i-k] for i in [k, n-1]: x[:n-k]
               grad[k:] += -wk * x[:n-k]
           else:
               kk = -k
               # (1-x[i+k]) = (1-x[i-kk]) for i in [kk, n-1]: x[:n-kk]
               grad[kk:] += wk * (1.0 - x[:n-kk])
               # -x[i-k] = -x[i+kk] for i in [0, n-1-kk]: x[kk:]
               grad[:n-kk] += -wk * x[kk:]
       
       return f, grad

           m = c.max()

           sm = sh.max()

                          constraints=cons,

       except Exception:
