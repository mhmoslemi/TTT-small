def slsqp_fast(h_init, n, alpha, maxiter, deadline):
    """Faster SLSQP using explicit gradient."""
    dx = 2.0 / n
    target = n / 2.0
    m_range = np.arange(-(n - 1), n)
    
    def corr_and_grad(x):
        # corr[k + n - 1] = sum_j x[j+k] * (1 - x[j]) for valid j, j+k
        corr = np.correlate(x, 1 - x, mode='full') * dx
        # Gradient: d corr[m]/d x[i] where m = k + n - 1
        # corr[m] = sum_j x[j + (m-n+1)] * (1 - x[j])
        # Let k = m - n + 1
        # = sum_j x[j+k] * (1-x[j])
        # d/dx[i]: 
        #   term where j+k = i (j = i-k): contribution (1-x[i-k]) if valid
        #   term where j = i: contribution -x[i+k] if valid
        grad = np.zeros(n)
        for k in m_range:
            k_idx = k + n - 1
            # j ranges such that 0<=j<n and 0<=j+k<n
            if k >= 0:
                j_start, j_end = 0, n - k
            else:
                j_start, j_end = -k, n
            # d corr[k_idx] / d x[i] for i in [j_start+k, j_end+k) from first factor
            # and for i in [j_start, j_end) from second factor
            # First factor: x[j+k], so i = j+k, i in [j_start+k, j_end+k)
            #   contribution: (1-x[j]) = (1-x[i-k])
            # Second factor: (1-x[j]), so i = j, i in [j_start, j_end)
            #   contribution: -x[j+k] = -x[i+k]
            
            # For first factor: i = j+k, j in [j_start, j_end)
            #   i in [j_start+k, j_end+k), contribution (1-x[i-k])
            i_start1 = j_start + k
            i_end1 = j_end + k
            i_start1 = max(0, i_start1)
            i_end1 = min(n, i_end1)
            if i_end1 > i_start1:
                idx = np.arange(i_start1, i_end1)
                j_idx = idx - k
                grad[idx] += (1 - x[j_idx])
            
            # For second factor: i = j, i in [j_start, j_end)
            #   contribution: -x[i+k]
            i_start2 = j_start
            i_end2 = j_end
            if i_end2 > i_start2:
                idx = np.arange(i_start2, i_end2)
                k_idx2 = idx + k
                valid = (k_idx2 >= 0) & (k_idx2 < n)
                grad[idx[valid]] += -x[k_idx2[valid]]
        
        return corr, grad
    
    def obj(x):
        g, _ = corr_and_grad(x)
        m = g.max()
        sh = alpha * (g - m)
        sh_max = sh.max()
        if sh_max > 500:
            return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha
        return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha
    
    def obj_grad(x):
        g, gc = corr_and_grad(x)
        m = g.max()
        sh = alpha * (g - m)
        sh_max = sh.max()
        ex = np.exp(sh - sh_max)

        # df/dx = dm/dx + (1/alpha) * [dsh_max/dx + d(log sum)/dx]
        # This is complex. Let's use a simpler approach.
        # For large alpha, the log-sum-exp approximates max, so grad ≈ grad of max + small correction
        return fval, grad_f

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    try:

            a1 = min(a0 + batch, n_ones)
            Oi = ones[a0:a1]
            
            # M shape: (a1-a0, n_zeros, 2n-1)
            M = corr[None, None, :] + V[Oi][:, None, :] - V[zeros][None, :, :]
            
            # Add cross terms
            for a_idx in range(a1 - a0):
                i = int(Oi[a_idx])
                for b_idx in range(n_zeros):
                    j = int(zeros[b_idx])
                    c1 = n - 1 + j - i
                    c2 = n - 1 + i - j
                    if 0 <= c1 < 2 * n - 1:
                        M[a_idx, b_idx, c1] += 1.0
                    if 0 <= c2 < 2 * n - 1:
                        M[a_idx, b_idx, c2] += 1.0
            
            M_max = M.max(axis=2)
            fi = np.argmin(M_max)
            la = fi // n_zeros
            lb = fi % n_zeros
            if M_max[la, lb] < best_max_val:
                best_max_val = M_max[la, lb]
                best_a = a0 + la
                best_b = lb
                best_delta = M[la, lb, :] - corr.copy()
        
        new_c = float(best_max_val) * dx
        if new_c < cur_c - 1e-14:
            i = int(ones[best_a])
            j = int(zeros[best_b])
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + best_delta
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: Quick binary search at n=250 (15% time)
    p1 = t0 + 0.15 * (budget_s - 5)
    hb, cb = binary_search_fast(best_h, n0, p1, max_sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        best_n = n0
    
    # Phase 2: Upsample to n=500 and binary search (35% time)
    p2 = t0 + 0.50 * (budget_s - 5)
    n2 = 500
    h2 = upsample(best_h, best_n, n2)
    h2b = to_binary(h2, n2)
    hb2, cb2 = binary_search_fast(h2b, n2, p2, max_sweeps=2000)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()
        best_n = n2
    
    # Phase 3: SLSQP on best binary solution (25% time)
    p3 = t0 + 0.75 * (budget_s - 5)
    rng = np.random.RandomState(seed)
    for alpha in [1000, 5000, 20000, 100000, 500000]:
        if time.time() > p3 - 30:
            break
        if time.time() > p3:
            break
        hp = best_h.copy()
        hp = slsqp_fast(hp, best_n, alpha, 800, p3 - 30)
        cp = compute_c5(hp, best_n)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    
    # Phase 4: Final binary polish (20% time)
    p4 = dl
    h4 = to_binary(best_h, best_n)
    hb4, cb4 = binary_search_fast(h4, best_n, p4, max_sweeps=5000)
    if cb4 < best_c:
        best_c = cb4
        best_h = hb4.copy()
        best_n = best_n
    
    # Final: try SLSQP one more time if time
    if time.time() < dl - 5:
        hp = best_h + 0.01 * rng.randn(best_n)
        hp = project_feasible(hp, best_n)
        hp = slsqp_fast(hp, best_n, 1000000, 1000, dl - 3)
        cp = compute_c5(hp, best_n)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # Ensure validity
    assert best_c < 1.0
    assert 0.0 <= best_h.min() and best_h.max() <= 1.0
    assert abs(best_h.sum() - best_n / 2.0) < 1e-8
    
    return best_h, best_c, best_n
