    grad = np.zeros(n)
    lag = k_worst - (n - 1)

            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]

        constraints.append(M >= (sum_s - sum_t) * dx)

    # Upper bound: we'll compute after. For the LP, we add M >= 0 (trivially true)
    # Actually, we should add a better lower bound. Since s_i ∈ [0,1], s_i(1-s_i) ≤ 1/4,

    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_values(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def get_subgradient(h, k_worst, n):
    grad = np.zeros(n)
    lag = k_worst - (n - 1)
    if lag >= 0:
        count = n - lag
        if count > 0:
            idx = np.arange(count)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        count = n - la
        if count > 0:
            idx = np.arange(count)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def multi_lag_direction(h, n, num_lags=5, sharpness=100):
    """Compute a weighted subgradient direction from top K worst lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = min(num_lags, len(corr))
    top_indices = np.argsort(corr)[-top_k:][::-1]
    gaps = corr[top_indices] - corr.max()
    weights = np.exp(sharpness * gaps)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k_idx in zip(weights, top_indices):
        grad += w * get_subgradient(h, int(k_idx), n)
    return grad

def nesterov_multi(h0, target_sum, n, t0, time_limit, max_steps=200000):

        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

def population_optimize(h0, target_sum, n, t0, time_limit, pop_size=8):
    """Simple evolutionary optimization."""
    h = h0.astype(float).copy()
    best_h = h.copy()

    population = [h.copy()]
