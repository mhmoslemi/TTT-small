import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c5(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize_keep_sum(h, n):
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def penalty_slsqp(h, n, mu, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        pen = np.sum(np.maximum(0, g - m) ** 2)
        return m + mu * pen
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)

        return h

def subgrad_polish(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_c5(h, n)
        m = g.max()
        ks = int(np.argmax(g))
        eps = 1e-6
        grad = np.zeros(n)
        for i in range(n):
            old = h[i]
            h[i] = old + eps
            grad[i] = (float(np.correlate(h, 1 - h, 'full')[ks]) * (2.0 / n) - g[ks]) / eps
            h[i] = old
        ng = np.linalg.norm(grad)
        if ng < 1e-13:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:

            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.85
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

        if time.time() > p4 - 1:
            break
        h_pert = penalty_slsqp(h_pert, n, mu, 500, p4 - 1)
        c = compute_c5(h_pert, n)
        if c < best_c:
            best_c = c
            best_h = h_pert.copy()

    # Phase 5: Final subgradient + polish (10% time)
    h_sg, c_sg = subgrad_polish(best_h.copy(), n, dl - 3, max_iter=8000)
    if c_sg < best_c:
        best_c = c_sg
        best_h = h_sg.copy()

    # One final SLSQP
    if time.time() < dl - 2:
        h_final = slsqp_optimize(best_h.copy(), n, 50000000, 500, dl - 2)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
