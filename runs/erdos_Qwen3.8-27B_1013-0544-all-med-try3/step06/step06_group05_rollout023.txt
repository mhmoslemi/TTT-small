def optimize_resolution(h_start, n, deadline, rng, skip_gradient=False):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [5000, 50000, 500000, 5000000, 20000000]
    for a in alphas:
        if time.time() > deadline - 25:
            break
        ht = slsqp_optimize(h, n, a, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()
            h = ht.copy()

    if not skip_gradient and time.time() < deadline - 15:
        ht, ct = gradient_search(best_h, n, deadline - 15,
                                  n_iter=15000, lr_init=0.002, lr_min=1e-7,
                                  active_frac=0.03)
        if ct < best_c:
            best_c = ct
            best_h = ht

    if time.time() < deadline - 5:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                          [0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                          max_passes=120)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                               [0.001, 0.0005, 0.0002],
                                               radius=60, max_passes=60)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(250, 0.5)
        n_parent = 250

    best_h = project_feasible(h_parent.copy(), n_parent)
    best_c = compute_c5(best_h, n_parent)

    phase1_end = t0 + (deadline - t0) * 0.30
    h1, c1 = optimize_resolution(best_h, n_parent, phase1_end, rng)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    x_parent = np.linspace(0, 2, n_parent)
    for n_try in [350, 500]:
        if time.time() > deadline - 70:
            break
        remaining = deadline - time.time()
        alloc = min(remaining * 0.28, 130)
        if alloc < 35:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_parent, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 45:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 45))
        scales = [0.005, 0.012, 0.025, 0.05]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 25) * 0.35)
            if alloc < 18:
                break
            h_opt, c_opt = optimize_resolution(hp, n, time.time() + alloc, rng,
                                               skip_gradient=True)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if time.time() < deadline - 4:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                          [0.00005, 0.00002, 0.00001], max_passes=200)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                               [0.0001, 0.00005, 0.00002],
                                               radius=80, max_passes=100)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
