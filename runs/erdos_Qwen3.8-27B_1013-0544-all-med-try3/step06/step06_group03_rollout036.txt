    """Projected gradient descent on smooth max."""
    dx = 2.0 / n
    if lr is None:
        lr = 0.05 / alpha  # initial learning rate heuristic

    best_obj = compute_c5(h, n)
    
    for it in range(max_iter):
        obj, grad = smooth_obj_and_grad(h, n, alpha)
        
        # Backtracking line search
        step = lr
        obj_new = None
        for _ in range(20):
            h_try = h - step * grad
            h_try = project_feasible(h_try, n)
            obj_try, _ = smooth_obj_and_grad(h_try, n, alpha)

        c = compute_c5(h, n)
        if c < best_obj - 1e-14:
            best_obj = c
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    weights = exp_s / sum_exp
    obj = (m + (max_s + np.log(sum_exp)) / alpha) * dx
    grad = np.zeros(n)
    for m_idx in range(len(c)):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            term = np.ones(n)
            term[k:] = 1.0 - h[:n - k]
            term[:n - k] -= h[k:]
            grad += w * term
        else:
            k_a = -k
            term = np.ones(n)
            term[:n - k_a] = 1.0 - h[k_a:n]
            term[k_a:] -= h[:n - k_a]
            grad += w * term
    grad *= dx
    return obj, grad

                obj = obj_try
                grad = grad_try

def compute_grad_vectors(h, n):
    n_lags = 2 * n - 1
    g = np.zeros((n, n_lags))
    for m_idx in range(n_lags):
        k = m_idx - (n - 1)
        if k == 0:
            g[:, m_idx] = 1.0 - 2.0 * h
        elif k > 0:
            term = np.ones(n)
            term[k:] = 1.0 - h[:n - k]
            term[:n - k] -= h[k:]
            g[:, m_idx] = term
        else:
            k_a = -k
            term = np.ones(n)
            term[:n - k_a] = 1.0 - h[k_a:n]
            term[k_a:] -= h[:n - k_a]
            g[:, m_idx] = term
    return g

def slsqp_polish(h, n, alpha, maxiter=2000):
    target = n / 2.0
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sum_c(x):
        return float(x.sum() - target)
    def sum_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c, 'jac': sum_j}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or compute_c5(res.x, n) < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()
