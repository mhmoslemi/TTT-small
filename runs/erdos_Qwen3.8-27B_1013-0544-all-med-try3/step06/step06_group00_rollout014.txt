    """Compute smooth max objective and its gradient."""
    dx = 2.0 / n
    g = np.correlate(grad_h, 1.0 - grad_h, mode='full') * dx
    # Smooth max value
    shifted = alpha * g
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p /= p.sum()
    val = (sm + np.log(p.sum())) / alpha  # = (sm + log(sum exp(shifted-sm))) / alpha
    
    # Gradient: sum_k p[k] * d g_k / d h
    # d g_k / d h[i] = dx * [(1-h[i+k])*1_{0<=i+k<n} - h[i-k]*1_{0<=i-k<n}]

```python
def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = alpha * g
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    p = exps / exps.sum()
    val = (sm + np.log(exps.sum())) / alpha
    
    # Vectorized gradient computation
    # For each i, grad[i] = dx * (sum_k p[k_idx]*(1-h[i+k])*1_{valid} - sum_k p[k_idx]*h[i-k]*1_{valid})
    
    # Create index matrix: idx[i, k_idx] = i + k where k = k_idx - (n-1)
    k = np.arange(2 * n - 1) - (n - 1)  # shape (2n-1,)
    i_idx = np.arange(n)  # shape (n,)
    
    # j1[i, k_idx] = i + k[k_idx], valid if 0 <= j1 < n
    j1 = i_idx[:, None] + k[None, :]  # shape (n, 2n-1)
    mask1 = (j1 >= 0) & (j1 < n)
    
    # j2[i, k_idx] = i - k[k_idx], valid if 0 <= j2 < n
    j2 = i_idx[:, None] - k[None, :]
    mask2 = (j2 >= 0) & (j2 < n)
    
    # Part 1: sum_k_idx p[k_idx] * (1-h[i+k]) * mask1
    # = sum_k_idx p[k_idx] * (1-h[j1]) * mask1
    h_j1 = np.where(mask1, h[j1], 0.0)  # h[j1] where valid, 0 otherwise
    part1 = (p[None, :] * (1.0 - h_j1) * mask1).sum(axis=1)
    
    # Part 2: sum_k_idx p[k_idx] * h[i-k] * mask2
    h_j2 = np.where(mask2, h[j2], 0.0)
    part2 = (p[None, :] * h_j2 * mask2).sum(axis=1)
    
    grad = (part1 - part2) * dx
    return val, grad
```
