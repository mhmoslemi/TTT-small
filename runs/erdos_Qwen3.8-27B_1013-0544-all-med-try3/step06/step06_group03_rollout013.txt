def compute_c5_array(h, n):
    """Compute all correlation values (without multiplying by dx)."""
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def to_binary(h, n):
    """Convert h to a {0,1} vector with exactly n/2 ones."""
    # Sort and threshold
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin
