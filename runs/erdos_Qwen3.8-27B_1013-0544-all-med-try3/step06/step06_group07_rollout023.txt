from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m == -np.inf:
        return 0.0
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.arange(len(h_old)) * 2.0 / len(h_old)
    x_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def binary_energy(h):
    """For binary h, compute max overlap (unscaled)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def simulated_annealing_binary(n, t0, time_limit, rng):
    """Find good binary solution via simulated annealing."""
    # Start with random arrangement
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    
    best_h = h.copy()
    best_e = binary_energy(h)
    current_e = best_e
    
    T = 5.0  # initial temperature
    cooling = 0.9995
    
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        if it % 1000 == 0:
            T *= cooling
        
        # Pick two positions to swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        
        # Swap
        h[i], h[j] = h[j], h[i]
        new_e = binary_energy(h)
        
        delta = new_e - current_e
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
        else:
            # Undo swap
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_e

def subgrad_step(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def slsqp_refine(h_init, n, dx, target_sum, t0, time_limit, beta_start=100, beta_end=50000, n_betas=6):
    """Multi-beta SLSQP refinement."""
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    betas = np.geomspace(beta_start, beta_end, n_betas)
    per_beta_time = time_limit / n_betas
    
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum, 
                'jac': lambda h: np.ones(n)}]
        try:
            res = minimize(
                lambda h: smooth_max(np.correlate(h, 1.0-h, mode='full'), beta) * dx,
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 5000, 'ftol': 1e-15}
            )
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
    
    return best_h, best_c5

def subgrad_step(h, k_worst, n):

    else:

        it += 1
        if it % 500 == 0:
            T *= cooling

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_e = binary_energy(h)
        
        delta = new_e - current_e
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            h[i], h[j] = h[j], h[i]
            stuck_count += 1

        if stuck_count > 5000:

            current_e = binary_energy(h)
            stuck_count = 0
            T = max(T, 2.0)
    
    return best_h, best_e

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()

    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

                                    'rhobeg': 0.05, 'rhoend': 1e-8, 'catol': 1e-8,
                                    'disp': False})
            h_new = project_feasible(res.x, tgt_c)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    # Phase 6: Final high-beta SLSQP polish
    if budget_s - (time.time() - t0) > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        tgt_f = n_f / 2.0
        h_ref_f, c5_ref_f = slsqp_phase(best_h, n_f, dx_f, tgt_f,
                                         [50000, 200000, 500000], 8000, t0, budget_s, 2)
        if c5_ref_f < best_c5:
            best_h = h_ref_f.copy()
            best_c5 = c5_ref_f

    # Final feasibility check
    n_final = len(best_h)
    tgt_final = n_final / 2.0
    best_h = project_feasible(best_h, tgt_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
