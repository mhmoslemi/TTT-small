                new_h = h + 0  # no change yet
                new_val = h[idx] + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Apply and check sum constraint
                # We need to maintain sum(h) = n/2
                # So we need to adjust another coordinate
                # For simplicity, let's skip the sum constraint for now and project at the end
                h[idx] = new_val
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    break
                else:
                    h[idx] = h[idx] - delta  # revert
                    h = project_feasible(h, n)
            if improved:
                break
        
        if not improved:
            # No improvement with current alpha, try a smaller step
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    """Returns (objective, gradient) for the smooth max"""
    dx = 2.0 / n
    C = np.correlate(x, 1 - x, mode='full')
    c = C * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum_exp = np.log(exp_shifted.sum())
    obj = c_max + log_sum_exp / alpha
    
    # Weights for the gradient
    weights = exp_shifted / exp_shifted.sum()
    
    # Gradient computation
    # P[m] = sum_j (1-x[j]) * w[m-j+n-1] where indices are valid
    # This is convolve(1-x, w)[m+n-1]
    ones_minus_x = 1.0 - x
    P_full = np.convolve(ones_minus_x, weights, mode='full')
    P = P_full[n-1:n-1+n]
    
    # Q[m] = sum_j x[j] * w[j-m+n-1] where indices are valid
    # Using correlate: np.correlate(x, w, 'full')[k] = sum_t x[t] * w[t+k-len(w)+1]
    # len(w) = 2n-1, so: sum_t x[t] * w[t+k-2n+2]
    # I want: sum_j x[j] * w[j-m+n-1]
    # So I need t+k-2n+2 = j-m+n-1 where t=j
    # k-2n+2 = -m+n-1
    # k = -m+n-1+2n-2 = 3n-m-3
    # Hmm, that gives k in [3n-(n-1)-3, 3n-0-3] = [2n-2, 3n-3]
    # But the full correlate has length n + (2n-1) - 1 = 3n-2, indexed 0 to 3n-3.
    # So k = 3n-m-3 ranges from 3n-(n-1)-3 = 2n-2 to 3n-3. That's within range!
    
    Q_full = np.correlate(x, weights, mode='full')
    # Q[m] = Q_full[3n-3-m] ... let me verify
    # Q_full[k] = sum_t x[t] * w[t+k-(2n-1)+1] = sum_t x[t] * w[t+k-2n+2]
    # I want Q[m] = sum_j x[j] * w[j-m+n-1]
    # Set t=j: Q_full[k] at k=3n-m-3 gives sum_t x[t] * w[t+3n-m-3-2n+2] = sum_t x[t] * w[t+n-m-1]
    # I want w[j-m+n-1] = w[j+(n-1)-m]. And I got w[t+n-m-1] = w[t+(n-1)-m]. ✓
    
    Q = Q_full[3*n-3 : 3*n-3-n : -1]  # This gives indices 3n-3, 3n-4, ..., 2n-2 (reversed)
    # Wait, Q[m] = Q_full[3n-3-m] for m=0,...,n-1
    # So Q = Q_full[3n-3 : 2n-3 : -1] which is Q_full[3n-3], Q_full[3n-4], ..., Q_full[2n-2]
    # That's n elements: indices from 3n-3 down to 2n-3 (inclusive) = n elements? 
    # 3n-3 - (2n-3) + 1 = n. Yes!
    Q = Q_full[2*n-2 : 3*n-2][::-1]  # indices 2n-2, 2n-1, ..., 3n-3, reversed = 3n-3, ..., 2n-2
    
    # Hmm, let me just do it directly to avoid indexing errors.
    # Actually, let me just verify with a small example in my head.
    # n=2, so x has 2 elements, w has 3 elements.
    # Q_full = correlate(x, w, 'full') has length 2+3-1 = 4.
    # Q_full[0] = x[1]*w[0-0] = x[1]*w[0]... 
    
    # This is getting error-prone. Let me just use the direct O(n^2) computation for Q.
    # For n=500, it's 500*500 = 250K ops, which is fast.
    
    Q = np.zeros(n)
    for m in range(n):
        j_lo = max(0, m - (n-1))
        j_hi = min(n-1, m + (n-1))
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            ks = js - m + n - 1
            Q[m] = np.dot(x[js], weights[ks])
    
    grad = dx * (P - Q)
    return obj, grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)
