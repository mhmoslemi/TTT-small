    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:

        h[mask] += diff / cnt

        b[mask1] += (1.0 - h[idxs])
    # Term 2: valid for m in [k_start2, k_end2], contributes -h[i-k] where k = m-(n-1)
    mask2 = (m_arr >= k_start2) & (m_arr <= k_end2)
    if np.any(mask2):
        idxs2 = i - (m_arr[mask2] - (n - 1))
        b[mask2] -= h[idxs2]
    return b


def solve_1d_minmax(corr, b, lo, hi):
    if lo >= hi:
        return 0.0
    if np.all(b <= 0):
        return hi
    if np.all(b >= 0):
        return lo
    lo_s = lo
    hi_s = hi
    for _ in range(80):
        mid = 0.5 * (lo_s + hi_s)
        vals = corr + mid * b
        vmax = vals.max()
        active = np.abs(vals - vmax) < 1e-13
        sub_lo = b[active].min()
        sub_hi = b[active].max()
        if sub_lo <= 0.0 <= sub_hi:
            return mid
        elif sub_lo > 0.0:
            hi_s = mid
        else:
            lo_s = mid
    return 0.5 * (lo_s + hi_s)


def bcd_optimize(h, target_sum, max_sweeps, time_fn):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time_fn() < 30:
            break
        for i in range(n):
            b = compute_slopes(h, i, n)
            delta = solve_1d_minmax(corr, b, -h[i], 1.0 - h[i])
            if abs(delta) < 1e-15:
                continue
            h[i] += delta
            corr += delta * b
            h = project_feasible(h, target_sum)
            # Recompute corr after projection if needed
            # (projection changes multiple entries, so we recompute)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_curr = float(np.max(corr) * dx)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
    return best_h, best_c5


def obj_slsqp(h_in, dx_in, beta_in):
    corr = np.correlate(h_in, 1.0 - h_in, mode='full')
    return smooth_max(corr, beta_in) * dx_in


def con_fun(h_in, ts):
    return h_in.sum() - ts


def con_jac(h_in, ts):
    return np.ones(len(h_in))


def slsqp_optimize(h, target_sum, n, bounds, beta, max_iter):
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    dx = 2.0 / n
    try:
        res = minimize(obj_slsqp, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(100, 0.5)

    best_h = project_feasible(h_init.copy(), len(h_init) / 2.0)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Resolution list
    base_n = best_n
    resolutions = sorted(set([r for r in [base_n, base_n + 20, base_n + 50,
                                           base_n + 80, base_n + 120, base_n + 200]
                              if r >= 50]))

    for n in resolutions:
        if remaining() < 60:
            break

        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        # Resample from current best
        h_curr = resample_h(best_h, n)
        h_curr = project_feasible(h_curr, target_sum)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_curr
            best_n = n

        # BCD optimization (primary method)
        if remaining() > 40:
            n_sweeps = 200
            h_bcd, c5_bcd = bcd_optimize(h_curr, target_sum, n_sweeps, remaining)
            if c5_bcd < best_c5:
                best_h = h_bcd.copy()
                best_c5 = c5_bcd
                best_n = n
                h_curr = h_bcd.copy()

        # SLSQP polish
        if remaining() > 40:
            for beta in [200, 1000, 5000]:
                if remaining() < 15:
                    break
                h_new = slsqp_optimize(h_curr, target_sum, n, bounds, beta, 3000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    best_n = n
                    h_curr = h_new.copy()

        # Second BCD pass from SLSQP result
        if remaining() > 30:
            n_sweeps2 = 100
            h_bcd2, c5_bcd2 = bcd_optimize(h_curr, target_sum, n_sweeps2, remaining)
            if c5_bcd2 < best_c5:
                best_h = h_bcd2.copy()
                best_c5 = c5_bcd2
                best_n = n
                h_curr = h_bcd2.copy()

    # Random restarts at best resolution
    if remaining() > 120:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        n_restarts = int(min(5, remaining() // 40))
        for r in range(n_restarts):
            if remaining() < 40:
                break
            # Generate diverse restart
            if r == 0:
                h_r = best_h.copy() + np.random.randn(n) * 0.05
            elif r == 1:
                h_r = best_h.copy() + np.random.randn(n) * 0.1
            elif r == 2:
                # Try a different structure: alternating
                h_r = np.where(np.arange(n) % 2 == 0, 0.7, 0.3)
            elif r == 3:
                # Try cosine structure
                x = np.linspace(0, 2, n, endpoint=False)
                h_r = 0.5 + 0.3 * np.cos(2 * np.pi * x)
            else:
                h_r = np.random.uniform(0.2, 0.8, n)

            h_r = project_feasible(h_r, target_sum)

            # BCD from restart
            h_bcdr, c5_dr = bcd_optimize(h_r, target_sum, 150, remaining)
            if c5_dr < best_c5:
                best_h = h_bcdr.copy()
                best_c5 = c5_dr
                best_n = n

            # SLSQP polish on best restart result
            if remaining() > 20:
                h_sl = slsqp_optimize(h_bcdr, target_sum, n, bounds, 2000, 2000)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_h = h_sl.copy()
                    best_c5 = c5_sl
                    best_n = n

    # Final high-precision polish
    if remaining() > 20:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        # BCD polish
        h_p, c5_p = bcd_optimize(best_h, target_sum, 200, remaining)
        if c5_p < best_c5:
            best_h = h_p.copy()
            best_c5 = c5_p

        # SLSQP with very high beta
        if remaining() > 10:
            h_f = slsqp_optimize(best_h, target_sum, n, bounds, 20000, 5000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_h = h_f.copy()
                best_c5 = c5_f

        # One more BCD
        if remaining() > 10:
            h_f2, c5_f2 = bcd_optimize(best_h, target_sum, 100, remaining)
            if c5_f2 < best_c5:
                best_h = h_f2.copy()
                best_c5 = c5_f2

    # Final projection to ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
