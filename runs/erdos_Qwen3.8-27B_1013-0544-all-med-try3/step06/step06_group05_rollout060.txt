def pair_search_fast(h, n, deadline, dets):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    best_c = c.max() * dx
    
    for d in dets:
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                for j in range(i+1, n):
                    # Try h[i]+d, h[j]-d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        # Recompute c (expensive but necessary for correctness)
                        c_new = np.correlate(h, 1-h, mode='full')
                        max_new = c_new.max()
                        if max_new < best_c / dx - 1e-15:
                            best_c = max_new * dx
                            c = c_new

Let me use the incremental update approach:

When h[i] changes by δ, the correlation at lag k (where k = m - (n-1)) changes by:
For k ≥ 0: δ * [(1-h[i+k]) * 1_{k ≤ n-1-i} - h[i-k] * 1_{1 ≤ k ≤ i}]
For k < 0 (k = -k'): δ * [(1-h[i-k']') * 1_{k' ≤ i} - h[i+k'] * 1_{i+k' ≤ n-1}]

Wait, I had k' = -k > 0 for the negative case:
For k = -k' (k' > 0): δ * [(1-h[i-k']) * 1_{k' ≤ i} - h[i+k'] * 1_{i+k' ≤ n-1}]

Note the symmetry: the formula for k and -k are similar but with i and n-1-i swapped in some sense.

For a pair move (h[i] += δ, h[j] -= δ), the total change at lag k is:
Δk = δ * [f(i, k) - f(j, k)]

where f(i, k) is the "influence" of position i on lag k.

Let me precompute a matrix F of shape (n, 2n-1) where F[i, m] = influence of h[i] on result[m].

Actually, F[i, m] = ∂result[m]/∂h[i] (the partial derivative).

Then for a pair move, the new correlation is:
c_new[m] = c[m] + δ * (F[i, m] - F[j, m])

And the new max is:
max_m (c[m] + δ * (F[i, m] - F[j, m]))

If we precompute F, we can compute this in O(n) per pair (since 2n-1 ≈ 2n lags). But that's still O(n² * n) = O(n³) per pass, same as before.

Unless... we can vectorize. For a fixed i and varying j, the update is:
c_new[m] = c[m] + δ * F[i, m] - δ * F[j, m]

The max of this over m... it's not easy to vectorize over j.

Alternative: for a fixed i, the best j is the one that minimizes max_m (c[m] + δ*(F[i,m] - F[j,m])). This is a "closest vector" type problem in the L∞ sense, which is hard.

        if abs(diff) < 1e-13:

            tot = h.sum()
            if tot < 1e-14:

            mask = h > 1e-10
            if not mask.any():
                break
            h[mask] -= (diff * h[mask]) / h[mask].sum()

    dx = 2.0 / n
    c = np.correlate(x, 1-x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def active_set_gradient(h, n, top_m=10):
    """Compute gradient of the sum of the top M correlations."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1-h, mode='full')  # length 2n-1
    c = c_raw * dx
    
    # Get top M indices
    idx = np.argpartition(c, -top_m)[-top_m:]
    
    # For each active lag k (where k = m - (n-1)), compute dc_k/dh[i]
    grad = np.zeros(n)
    for m_idx in idx:
        k = m_idx - (n - 1)  # actual shift
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # dc_k/dh[i] = (1-h[i+k]) for i in [0, n-1-k]
            #              - h[i-k] for i in [k, n-1]
            for i in range(n - k):
                grad[i] += (1 - h[i + k])
            for i in range(k, n):
                grad[i] -= h[i - k]
        else:
            k_neg = -k
            # c_{-k_neg} = sum_{i=k_neg}^{n-1} h[i]*(1-h[i-k_neg])
            # dc/dh[i] = (1-h[i-k_neg]) for i in [k_neg, n-1]
            #           -h[i+k_neg] for i in [0, n-1-k_neg]
            for i in range(k_neg, n):
                grad[i] += (1 - h[i - k_neg])
            for i in range(n - k_neg):
                grad[i] -= h[i + k_neg]
    
    return grad * (1.0 / top_m)  # average gradient of top M

def projected_subgrad(h, n, deadline, n_iter=2000, step_scale=0.001):
    """Projected subgradient method on the top-M active set."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        grad = active_set_gradient(h, n, top_m=5)
        
        # Adaptive step size
        step = step_scale / (1 + it * 0.01)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step

            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= (diff * h[mask]) / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def active_set_grad(h, n, top_m=8):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    idx = np.argpartition(c, -top_m)[-top_m:]
    grad = np.zeros(n)
    for m_idx in idx:
        k = m_idx - (n - 1)
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n - k] += (1.0 - h[k:])
            if k > 0:
                grad[k:] -= h[:n - k]
        else:
            kn = -k
            if n - kn >= 0:
                grad[kn:] += (1.0 - h[:-kn])
            if kn > 0:
                grad[:n - kn] -= h[kn:]
    return grad / top_m

                    h3, c3 = optimize_n(h_back, n_init, min(deadline - 35, time.time() + 80))
                    if c3 < best_c:
                        best_c = c3
                        best_h = h3

    # Phase 3: Multi-restart perturbation (if time allows)
    if time.time() < deadline - 40:
        n_best = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 60))
        scales = [0.003, 0.01, 0.02, 0.04]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 25) * 0.3)
            if alloc < 20:
                break
            hr, cr = optimize_n(hp, n_best, time.time() + alloc)
            if cr < best_c:
                best_c = cr
                best_h = hr

    # Phase 4: Final ultra-fine polish
    if time.time() < deadline - 3:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = pair_search_full(
                best_h, n, deadline - 1,
                [0.0001, 0.00005, 0.00002, 0.00001, 0.000005], max_passes=500)
        else:
            best_h, best_c = pair_search_radius(
                best_h, n, deadline - 1,
                [0.0001, 0.00005, 0.00002], radius=70, max_passes=300)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
