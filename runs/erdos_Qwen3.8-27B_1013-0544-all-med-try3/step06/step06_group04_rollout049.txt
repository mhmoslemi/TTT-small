        for inner in range(n_inner):

def top_lag_slsqp(h, n, deadline, alpha=50000, top_m=5, maxiter=3000):
    """SLSQP on top m lags with smooth max."""
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        active = cc[top_idx]
        return smooth_max(active, alpha)
    
    def obj_grad(x):
        cc = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        active = cc[top_idx]
        w = smooth_max_grad(active, alpha)
        grad = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            if wi < 1e-12:
                continue
            lag = idx - (n - 1)
            grad += wi * _lag_gradient(x, n, lag)
        # chain rule for smooth max
        # Actually, the gradient of smooth_max(active, alpha) w.r.t. x is:
        # sum_i w_i * d(active_i)/dx * (1/alpha) ... wait no
        # smooth_max(c, alpha) = c.max() + log(sum(exp(alpha*(c-c.max()))))/alpha
        # d/dx = sum_i exp(alpha*(c_i - c.max()))/sum_j exp(alpha*(c_j-c.max())) * d(c_i)/dx
        # (the 1/alpha factor is absorbed because d c_i/dx is already computed)
        # Actually let me be more careful:
        # f(c) = m + (1/alpha) * log(sum exp(alpha*(c_i - m)))
        # df/dc_i = exp(alpha*(c_i-m)) / sum_j exp(alpha*(c_j-m))
        # So df/dx = sum_i (df/dc_i) * (dc_i/dx)
        # which is what I have above with w = softmax(alpha*(c-m))
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pairwise_search(h, n, deadline, top_m=3):
    """Pairwise coordinate descent focusing on top lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            # Focus on the single worst lag
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            
            # Try all pairs in the worst lag
            pairs_done = 0
            for i in range(n - m):
                if time.time() > deadline:
                    break
                pairs_done += 1
                if pairs_done > 200:  # limit per iteration
                    break
                j = i + m
                hi, hj = h[i], h[j]
                if hi <= hj:
                    continue
                # Move mass from i to j (reduce h[i], increase h[j])
                # This should reduce R(lag) since h[i] decreases and h[j] increases
                # R(lag) contains terms h[i]*h[j] (for positive lag)
                delta = min(d, (hi - hj) * 0.3, hi * 0.5, 0.5 * (1 - hj))
                if delta < 1e-9:
                    continue
                h[i] -= delta
                h[j] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[i] += delta
                    h[j] -= delta
    return h, best_c

def cvx_linearize(h, n, deadline, trust=0.1):
    """One step of outer approximation using CVXPY."""
    c = compute_all_lags(h, n)
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h, 2) <= trust]

        lag = k - (n - 1)
        grad = _lag_gradient(h, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return h.copy()
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    h_new = h_var.value
    if h_new is None:
        return h.copy()
    return project_feasible(h_new, n)

        cons.append(c[k] + dx * (grad @ (h_var - h)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    h_new = h_var.value
    if h_new is None:
        return h.copy()
    return project_feasible(h_new, n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 40:
        for _ in range(4):
            if time.time() > deadline - 40:
                break
            h_new = cvx_step(best_h, n, 0.08)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

    if time.time() < deadline - 30:
        ew_end = time.time() + min(50, (deadline - time.time()) * 0.35)
        h_ew, c_ew = exp_weight_optimize(best_h, n, ew_end, n_outer=12)
        if c_ew < best_c:
            best_c = c_ew
            best_h = h_ew

    if time.time() < deadline - 20:
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_top_lags(best_h, n, alpha, 5, 2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if time.time() < deadline - 15:
        h_new = slsqp_top_lags(best_h, n, 10000000, 3, 3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 5:
        h_new, c_new = pairwise_search(best_h, n, min(deadline - 2, time.time() + 35))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    n_candidates = []
    if h_init is not None:
        if n_init < 400:
            n_candidates.append(400)
        if n_init < 500:
            n_candidates.append(500)
        if n_init < 700:
            n_candidates.append(700)

    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 40
        if remaining < 70:
            break
        alloc = min(250, remaining * 0.3)
        if alloc < 50:
            break
        if best_h is not None:
            h_start = interpolate_h(best_h, len(best_h), n_try)
        elif h_init is not None:
            h_start = interpolate_h(h_init, n_init, n_try)
        else:
            h_start = np.full(n_try, 0.5)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 35
        n_restarts = max(1, int(remaining / 90))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.008 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(80, (deadline - time.time() - 30) * 0.5)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_top_lags(best_h, n, 10000000, 3, 5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            h_new, c_new = pairwise_search(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
