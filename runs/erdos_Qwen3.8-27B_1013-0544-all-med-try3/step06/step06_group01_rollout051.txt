def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert to binary with exactly n//2 ones, choosing the largest values."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def fast_local_search(h, n, deadline, max_sweeps=5000, n_restarts=3, seed=42, use_sa=False, t0_temp=1.0, t1_temp=0.001):

        part1 = np.where(valid1, 1.0 - h_curr[idx1], 0.0)
        part2 = np.where(valid2, h_curr[idx2], 0.0)
        return part1 - part2
    
    h = binarize(h, n)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP smooth optimization."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh = np.clip(sh, -500, 500)
        sm = sh.max()
        logsum = sm + np.log(np.exp(sh - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h
