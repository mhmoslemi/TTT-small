Actually, for the sum constraint, if I change h[j] by δ, I need to change some h[k] by -δ. The pair search does exactly this. The key improvement is to be smarter about which pair to choose.

Idea: Use the gradient to identify promising pairs.
- If ∇f[j] > 0 (increasing h[j] increases the objective), we want to decrease h[j]
- If ∇f[k] < 0, we want to increase h[k]
- So a good pair is (j, k) where ∇f[j] > 0 and ∇f[k] < 0
- The best such pair would minimize the change in the objective

This is essentially what gradient descent does, but with the sum constraint, we need paired moves.

Let me implement a "gradient-informed pair search":
1. Compute gradient ∇f
2. Find the j with largest positive gradient (worst to increase)
3. Find the k with largest negative gradient (best to increase)
4. Try moving δ from j to k (decrease h[j], increase h[k])
5. If it improves, accept; otherwise try next pair
6. This is O(n) to find the pairs, then O(n) to evaluate the new c

This should be much faster than the O(n²) pair search while being more effective.

        # We want grad[j] > 0 and grad[k] < 0
        # Best j: largest positive grad
        # Best k: most negative grad
        j = np.argmax(grad)
        k = np.argmin(grad)
        
        if grad[j] <= 0 or grad[k] >= 0:
            break  # No improving direction
        
        # Try several step sizes
        improved = False
        for delta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:

            h[j] -= delta
            h[k] += delta
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                improved = True
                break
            else:
                h[j] += delta
                h[k] -= delta
        
        if not improved:
            # Try next-best pair
            # Remove j and k from consideration
            grad2 = grad.copy()
            grad2[j] = -np.inf
            grad2[k] = np.inf
            j2 = np.argmax(grad2)
            k2 = np.argmin(grad2)
            if grad2[j2] <= 0 or grad2[k2] >= 0:
                break
            for delta in [0.05, 0.02, 0.01, 0.005]:
                if h[j2] - delta < 0 or h[k2] + delta > 1:

                h[j2] -= delta
                h[k2] += delta

                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[j2] += delta
                    h[k2] -= delta
            if not improved:

    return best_h, best_c

import numpy as np

from scipy.special import logsumexp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    f = float(logsumexp(alpha * c) / alpha)
    nfull = 2 * n - 1
    ac = alpha * c
    max_ac = ac.max()
    w = np.exp(ac - max_ac)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(nfull) / nfull
    else:
        w /= w_sum
    prefix = np.zeros(nfull + 1)
    np.cumsum(w, out=prefix[1:])
    S = prefix[n:2*n] - prefix[:n]
    corr_full = np.correlate(w, h, mode='full')
    T = corr_full[:n][::-1]
    grad = (1.0 - h) * S - T
    return f, grad

        f, _ = smooth_obj_grad(h, n, alpha)
        return f
    def jac(h):
        _, g = smooth_obj_grad(h, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16, 'maxls': 50})
        return project_feasible(res.x, n)

    if time.time() < deadline - 3:
        ht = slbfgs(h, n, 2000000, maxiter=300)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    if time.time() < deadline - 2:
        h, best_c = subgrad_local_search(h, n, deadline - 1)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def make_sinusoidal_init(n, rng, n_freqs=5):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(n_freqs):
        freq = 2 * (k + 1)
        amp = 0.3 / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * freq * x / 2 + phase)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def make_random_init(n, rng):
    h = rng.uniform(0.1, 0.9, n)
    h = project_feasible(h, n)
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = make_sinusoidal_init(250, rng)
        n_init = 250
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase 1: Optimize at initial resolution
    alloc1 = min(120, (deadline - t0) * 0.2)
    if alloc1 > 30:
        h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Try other resolutions
    for n_try in [200, 300, 400, 500, 350]:
        if time.time() > deadline - 60:
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        alloc = min(90, (deadline - time.time()) * 0.15)
        if alloc < 25:
            break
        h_opt, c_opt = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multi-restart with diverse initializations at best resolution
    n_best = len(best_h)
    remaining = deadline - time.time() - 40
    if remaining > 30:
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                h_r = best_h + rng.normal(0, 0.05, n_best)
            elif r == 1:
                h_r = best_h + rng.normal(0, 0.1, n_best)
            elif r == 2:
                h_r = make_sinusoidal_init(n_best, np.random.RandomState(seed + 1000 + r))
            elif r == 3:
                h_r = make_random_init(n_best, np.random.RandomState(seed + 2000 + r))
            else:
                h_r = best_h + rng.normal(0, 0.03 + 0.05 * r, n_best)
            h_r = project_feasible(h_r, n_best)
            alloc = min(45, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = full_optimize(h_r, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Refine at potentially better resolution
    if time.time() < deadline - 50:
        for n_try in [600, 750]:
            if time.time() > deadline - 40:
                break
            h_interp = interpolate_h(best_h, len(best_h), n_try)
            alloc = min(60, deadline - time.time() - 30)
            if alloc < 20:
                break
            h_opt, c_opt = full_optimize(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Final intensive local search
    if time.time() < deadline - 5:
        n = len(best_h)
        h, c = gradient_pair_search(best_h, n, min(deadline - 3, time.time() + 30))
        if c < best_c:
            best_c = c
            best_h = h
        
        if time.time() < deadline - 3:
            h, c = subgrad_local_search(best_h, n, deadline - 2)
            if c < best_c:
                best_c = c
                best_h = h
        
        if time.time() < deadline - 2:
            h, c = pair_search_fine(best_h, n, deadline - 1,
                                    [0.001, 0.0005, 0.0002, 0.0001])
            if c < best_c:
                best_c = c
                best_h = h
    
    # Final SLSQP-style polish with very high alpha
    if time.time() < deadline - 2:
        n = len(best_h)
        ht = slbfgs(best_h, n, 5000000, maxiter=500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
