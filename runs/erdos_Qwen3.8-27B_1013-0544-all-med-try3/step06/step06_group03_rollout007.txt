```python
def compute_smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500, 0)
    sw = np.exp(shifted)
    sw_sum = sw.sum()
    sw = sw / sw_sum  # softmax weights, length 2n-1
    
    # lags: for index m in [0, 2n-2], lag = m - (n-1)
    lags = np.arange(2*n-1) - (n-1)  # from -(n-1) to n-1
    
    # grad[j] = sum_m sw[m] * [(1-h[j+lag_m])*[0≤j+lag_m<n] - h[j-lag_m]*[0≤j-lag_m<n]]
    grad = np.zeros(n)
    for j in range(n):
        # First part: (1-h[j+lag_m]) where 0≤j+lag_m<n
        valid_plus = (j + lags) >= 0 & (j + lags) < n
        # Second part: h[j-lag_m] where 0≤j-lag_m<n
        valid_minus = (j - lags) >= 0 & (j - lags) < n
        
        grad[j] = np.sum(sw[valid_plus] * (1 - h[j + lags[valid_plus]])) - \
                  np.sum(sw[valid_minus] * h[j - lags[valid_minus]])
    
    grad *= dx
    return grad, c_max
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, n, alpha, penalty=100):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)) / len(c))
    f = c_max + log_sum / alpha
    s = h.sum() - n / 2.0
    return f + penalty * s * s

def smooth_gradient(h, n, alpha, penalty=100):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    
    # Vectorized computation
    # grad[j] = sum_m w[m] * [(1-h[j+lag_m])*[0<=j+lag_m<n] - h[j-lag_m]*[0<=j-lag_m<n]]
    
    for j in range(n):
        valid_plus = (j + lags) >= 0
        valid_plus &= (j + lags) < n
        valid_minus = (j - lags) >= 0
        valid_minus &= (j - lags) < n
        if valid_plus.any():
            grad[j] += np.dot(w[valid_plus], 1 - h[j + lags[valid_plus]])
        if valid_minus.any():
            grad[j] -= np.dot(w[valid_minus], h[j - lags[valid_minus]])
    
    grad *= dx
    s = h.sum() - n / 2.0
    grad += 2 * penalty * s * np.ones(n)
    return grad

        return smooth_gradient(x, n, alpha, penalty)

                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})

        return h_new

def vectorized_pair_search(h, n, end_time, deltas):
    """Vectorized pair search: for each delta, find the best pair to swap."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lags_all = np.arange(2 * n - 1) - (n - 1)

    for delta in deltas:
        sweep_improved = True
        while sweep_improved and time.time() < end_time:
            sweep_improved = False
            
            # Compute sensitivity matrix S[k, j] = dc[k]/dh[j] / dx
            # S[k, j] = (1-h[j+lag_k])*[valid] - h[j-lag_k])*[valid]
            # Vectorized:
            # For each k (lag = lags_all[k]):
            #   S[k, j] = (1-h[j+lags_all[k]]) * (0<=j+lags_all[k]<n) - h[j-lags_all[k]] * (0<=j-lags_all[k]<n)
            
            # This is a matrix of size (2n-1) x n
            # S[k, j] can be computed as:
            # term1[k,j] = (1-h[j+lag_k]) if 0<=j+lag_k<n else 0
            # term2[k,j] = h[j-lag_k] if 0<=j-lag_k<n else 0
            
            # Let's compute it using shifting:
            S = np.zeros((2*n-1, n))
            for k in range(2*n-1):
                lag = lags_all[k]
                # term1: (1-h[j+lag]) for 0<=j+lag<n
                if lag >= 0:
                    idx_src = np.arange(0, n-lag)
                    idx_dst = idx_src
                    S[k, idx_dst] = (1 - h[idx_src + lag]) 
                    # Wait, j+lag is the index in h, j is the column
                    # S[k,j] = (1-h[j+lag]) for valid j
                    # j+lag >= 0 always if lag >= 0 and j >= 0
                    # j+lag < n means j < n-lag
                    S[k, :n-lag] = (1 - h[lag:n])
                else:
                    # lag < 0, j+lag >= 0 means j >= -lag
                    S[k, -lag:] = (1 - h[:n+lag])
                
                # term2: h[j-lag] for 0<=j-lag<n
                if lag <= 0:
                    # j-lag >= 0 always if lag<=0 and j>=0
                    # j-lag < n means j < n+lag
                    S[k, :n+lag] -= h[:n+lag]  # h[j-lag] = h[j+|lag|] for j < n-|lag|... 
                    # Wait: j-lag = j+|lag|, and we need 0<=j+|lag|<n, so j<n-|lag|=n+lag
                    S[k, :n+lag] -= h[0:(n+lag)]
                else:
                    # j-lag >= 0 means j >= lag
                    S[k, lag:] -= h[0:(n-lag)]
            
            # For a pair (i,j) with i getting +delta, j getting -delta:
            # change in c[k] = delta * (S[k,i] - S[k,j]) * dx
            # We want to minimize max_k (c[k] + delta*(S[k,i]-S[k,j])*dx)
            
            # For small delta, the critical constraint is at the current max
            # Active lags: where c[k] is near max
            c_max = np.max(c)
            active = c >= c_max - 0.01

            # i.e., S[active, i] < S[active, j] (element-wise sum)
            
            # Score for giving +delta to i: sum of S[active, i] (want this small)
            # Score for giving -delta to j: sum of S[active, j] (want this large)
            
            active_idx = np.where(active)[0]
            score_i = S[active_idx, :].mean(axis=0)  # want to minimize (give +delta here)
            score_j = S[active_idx, :].mean(axis=0)  # want to maximize (give -delta here)
            
            # Best i: lowest score (and h[i]+delta <= 1)
            # Best j: highest score (and h[j]-delta >= 0)

            can_decrease = h - delta >= 0.0
            
            score_i_masked = np.where(can_increase, score_i, np.inf)
            score_j_masked = np.where(can_decrease, score_j, -np.inf)

                break

            h[j_best] -= delta
