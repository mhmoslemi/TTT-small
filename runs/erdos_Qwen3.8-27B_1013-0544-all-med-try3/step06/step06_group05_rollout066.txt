        w_top = np.ones(K) / K
    else:
        w_top = w_top / wsum
    grad = np.zeros(n)
    for i in range(K):
        idx = top_idx[i]
        s = idx - (n - 1)
        w = w_top[i] * dx
        t1 = max(0, -s)
        t2 = min(n, n - s)
        if t2 > t1:
            grad[t1:t2] += w * (1.0 - h[t1 + s:t2 + s])
        t3 = max(0, s)
        t4 = min(n, n + s)
        if t4 > t3:
            grad[t3:t4] += -w * h[t3 - s:t4 - s]
    target = n / 2.0
    grad += 2.0 * penalty * (h.sum() - target)
    return grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter, deadline):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(sg_func, h0, fprime=sg_grad,
                            args=(n, alpha, penalty),
                            bounds=bounds, maxiter=maxiter,
                            maxfun=maxiter * n // 10 + 100,
                            iprint=0, pgtol=1e-12, factr=1e8)
        h_new = project_feasible(res[0], n)
        return h_new
    except Exception:
        return h0

def coordinate_descent(h, n, deadline, max_rounds=200, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c_arr = compute_c_array(h, n)
    for step in step_sizes:
        for _round in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if j - i > 100:
                        continue
                    hi = h[i] + step
                    hj = h[j] - step
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i], h[j] = hi, hj
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i], h[j] = hi - step, hj + step
            if not improved:
                break
    return h, best_c

def random_binary_init(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, n // 2, replace=False)
    h[indices] = 1.0
    return project_feasible(h, n)

def structured_init(n, rng, freq=None):
    if freq is None:
        freq = 1.0
    x = np.linspace(0, 2, n)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h

def optimize_at_resolution(h_start, n, deadline, rng, n_alphas=5):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    penalties = [500, 2000, 10000, 50000]
    maxiters = [500, 400, 300, 200, 200, 150]
    for i in range(min(n_alphas, len(alphas))):
        if time.time() > deadline - 10:
            break
        alpha = alphas[i]
        pen = penalties[min(i, len(penalties) - 1)]
        maxit = maxiters[min(i, len(maxiters) - 1)]
        h_new = lbfgs_optimize(best_h, n, alpha, pen, maxit, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at native resolution
    if best_h is not None and time.time() < deadline - 300:
        alloc = min(150, (deadline - time.time()) * 0.25)
        h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + alloc, rng, n_alphas=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Try higher resolutions
    resolutions = [400, 600, 1000]
    if best_h is None:
        resolutions = [250, 500, 1000]

    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time()
        if remaining < 100:
            break
        alloc = min(200, remaining * 0.3)
        if alloc < 60:
            break

        if best_h is not None and len(best_h) != n_try:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif best_h is not None:
            h_start = best_h.copy()
        else:
            h_start = random_binary_init(n_try, rng)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, t0 + alloc, rng, n_alphas=4)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        # Also try a random/structured initialization at this resolution
        if time.time() < deadline - 60:
            h_rand = random_binary_init(n_try, rng)
            alloc2 = min(100, (deadline - time.time()) * 0.2)
            if alloc2 > 30:
                h_opt2, c_opt2 = optimize_at_resolution(h_rand, n_try, t0 + alloc2, rng, n_alphas=3)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 3: Try structured initializations at various frequencies
    if time.time() < deadline - 60:
        n_s = len(best_h) if best_h is not None else 250
        for freq in [2.0, 3.0, 4.0, 5.0]:
            if time.time() > deadline - 30:
                break
            h_struct = structured_init(n_s, rng, freq=freq)
            alloc = min(80, (deadline - time.time()) * 0.15)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(h_struct, n_s, t0 + alloc, rng, n_alphas=3)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Local coordinate descent on best solution
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        remaining = deadline - time.time() - 2
        if n <= 400:
            best_h, best_c = coordinate_descent(best_h, n, t0 + remaining,
                                                max_rounds=300,
                                                step_sizes=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            # For larger n, use a restricted local search
            h = project_feasible(best_h, n)
            best_c = compute_c5(h, n)
            steps = [0.01, 0.005, 0.002, 0.001]
            radius = min(80, n // 4)
            for step in steps:
                improved = True
                passes = 0
                while improved and passes < 50 and time.time() < deadline - 2:
                    improved = False
                    passes += 1
                    for i in range(n):
                        if time.time() > deadline - 2:
                            break
                        j_lo = max(0, i - radius)
                        j_hi = min(n, i + radius + 1)
                        for j in range(j_lo, j_hi):
                            if i == j:
                                continue
                            hi = h[i] + step
                            hj = h[j] - step
                            if 0 <= hi <= 1 and 0 <= hj <= 1:
                                h[i], h[j] = hi, hj
                                c = compute_c5(h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    improved = True
                                else:
                                    h[i], h[j] = hi - step, hj + step
            best_h = h

    # Phase 5: Final SLSQP-like refinement using one more L-BFGS with high alpha
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        h_final = lbfgs_optimize(best_h, n, 200000, 50000, 300, t0 + (deadline - 8))
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final projection
    if best_h is not None:
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
        return best_h, best_c, len(best_h)

    # Fallback
    n = n_init
    h = np.full(n, 0.5)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    return h, c, n
