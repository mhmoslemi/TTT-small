```python
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_correlation_fft(h):
    """Compute cross-correlation of h with 1-h using FFT."""
    n = len(h)
    v = 1.0 - h
    # Zero-pad to 2n-1 or next power of 2 for FFT
    m = 2 * n - 1
    # Power of 2 for FFT efficiency
    m_fft = 1
    while m_fft < m:
        m_fft *= 2
    H = np.zeros(m_fft, dtype=complex)
    V = np.zeros(m_fft, dtype=complex)
    H[:n] = h
    V[:n] = v
    # Cross-correlation via FFT: corr = ifft(conj(FFT(a)) * FFT(b))
    # np.correlate(a, v) = ifft(conj(FFT(a)) * FFT(v))
    corr = np.real(ifft(np.conj(fft(H, m_fft)) * fft(V, m_fft)))[:m]
    return corr

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
```

Let me also think about the step size. For the log-sum-exp objective with parameter α:

- Large α: close to the true max but steeper, harder to optimize

Strategy: start with small α, increase gradually.

- If f(h - η*grad) > f(h) + c*η*||grad||^2, reduce η and retry

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    """Compute np.correlate(h, 1-h, mode='full') - the full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_box_sum(h, target):
    """Project h onto {x: sum(x) = target, 0 <= x <= 1}."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_lse(h, n, alpha):
    """Compute gradient of log-sum-exp smoothed max of correlation."""
    dx = 2.0 / n
    g = compute_corr(h) * dx  # shape (2n-1,)
    # Softmax weights
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    w = exp_g / exp_g.sum()  # softmax weights, shape (2n-1,)
    
    # Gradient: dx * [conv(w, 1-h) - conv(flip(w), h)]
    # Using FFT for efficiency
    m = 2 * n - 1  # length of correlation
    # We need circular correlation/convolution
    # Actually, let's use direct computation for simplicity since n is moderate
    # grad[m] = dx * sum_k w[k] * (1-h[m-k_shift]) - dx * sum_k w[k] * h[m+k_shift]
    # where k_shift = k - (n-1)
    
    # Let's use FFT-based convolution
    n_fft = 1
    while n_fft < 2 * m:
        n_fft *= 2
    
    W = fft(w, n_fft)
    H = fft(h, n_fft)
    Hinv = fft(1.0 - h, n_fft)
    
    # conv(w, 1-h) = ifft(fft(w) * fft(1-h))
    conv1 = np.real(ifft(W * Hinv))
    # conv(flip(w), h) = ifft(fft(flip(w)) * fft(h))
    Wflip = fft(w[::-1], n_fft)
    conv2 = np.real(ifft(Wflip * H))
    
    grad = dx * (conv1[:n] - conv2[:n])
    return grad, g.max()

def projected_gradient(h, n, deadline, alpha_seq, eta0=0.005, max_iters_per_alpha=5000):
    """Projected gradient method with log-sum-exp smoothing."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        h_cur = h.copy()
        eta = eta0
        prev_f = None
        
        for it in range(max_iters_per_alpha):
            if time.time() > deadline:
                break
            
            grad, gmax = compute_gradient_lse(h_cur, n, alpha)
            
            # Backtracking line search
            for backtrack in range(20):
                h_try = project_box_sum(h_cur - eta * grad, n / 2.0)
                f_try = compute_c5(h_try, n)
                
                # Also compute the smooth objective for better line search
                # f_smooth = gmax + (1/alpha) * log(sum(exp(alpha*(g - gmax))))
                g = compute_corr(h_try) * dx
                gm = g.max()
                shifted = alpha * (g - gm)
                f_smooth_try = gm + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha
                
                if prev_f is None or f_smooth_try < prev_f - 1e-12:
                    break
                eta *= 0.5
            else:
                eta = max(eta, 1e-10)
                h_try = project_box_sum(h_cur - eta * grad, n / 2.0)
                f_smooth_try = None
            
            h_cur = h_try
            if f_smooth_try is not None:
                prev_f = f_smooth_try
            
            # Update best
            c_cur = compute_c5(h_cur, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        
        h = h_cur.copy()
    
    return best_h, best_c
```

```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    g = corr * dx
    g_max = g.max()
    exp_g = np.exp(alpha * (g - g_max))
    w = exp_g / exp_g.sum()
    
    grad = np.zeros(n)
    # w[k] corresponds to shift kp = k - (n-1)
    # grad[m] += w[k]*dx * [(1-h[m-kp]) if 0≤m-kp<n] - [h[m+kp] if 0≤m+kp<n]
    
    for k in range(2*n-1):
        kp = k - (n-1)
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        # Term 1: (1-h[m-kp]) for 0≤m-kp<n → m ∈ [kp, kp+n-1] ∩ [0,n-1]
        lo1 = max(0, kp)
        hi1 = min(n-1, kp+n-1)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * (1.0 - h[lo1-kp:hi1-kp+1])
        # Term 2: h[m+kp] for 0≤m+kp<n → m ∈ [-kp, n-1-kp] ∩ [0,n-1]
        lo2 = max(0, -kp)
        hi2 = min(n-1, n-1-kp)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * h[lo2+kp:hi2+kp+1]
    
    return grad, g_max
```

One more thought: the **p-norm** approximation might be better than log-sum-exp for this problem. The p-norm is:
f_p(h) = (max(0, min_g)^p + ... + max_n^p)^(1/p) for large p

Actually no, for minimizing the max, we want:
f_p(h) = (sum_k g_k(h)^p)^(1/p) → max_k g_k(h) as p → ∞

The gradient is:
∂f_p/∂h[m] = (1/p) * (sum g^p)^{1/p - 1} * p * sum_k g_k^{p-1} * ∂g_k/∂h[m]
= (sum g^p)^{1/p - 1} * sum_k g_k^{p-1} * ∂g_k/∂h[m]

The weights are w_k = g_k^{p-1} / sum_j g_j^{p-1}

This is similar to the softmax weights but with a different temperature. For large p, it concentrates on the max, similar to large alpha in LSE.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, target, tol=1e-12):
    """Project h onto {x: sum(x)=target, 0≤x≤1} via bisection on Lagrange multiplier."""
    h = np.clip(h, 0.0, 1.0)
    # Quick check
    if abs(h.sum() - target) < tol:
        return h
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return result

def compute_gradient(h, n, alpha):
    """Compute gradient of LSE-smoothed max of correlation * dx. O(n^2)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    # Handle numerical stability
    shifted_max = shifted.max()
    exp_s = np.exp(shifted - shifted_max)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        kp = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-16:

        hi2 = min(n - 1, n - 1 - kp)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * h[lo2+kp:hi2+kp+1]
    
    return grad, g_max

def lse_objective(h, n, alpha):
    """Compute the LSE-smoothed max of correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    sm = shifted.max()
    return g_max + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def projected_gradient_optimize(h_init, n, deadline, alpha_seq, eta0=0.01, max_inner=10000):

    dx = 2.0 / n

    h = h_init.copy()

        inner_iter = 0
        
        while inner_iter < max_inner:

                break
            inner_iter += 1

            # Backtracking line search
            accepted = False

                    prev_obj = obj_try

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()

        sm = shifted.max()

    def cons_fun(x):
        return x.sum() - n / 2.0
