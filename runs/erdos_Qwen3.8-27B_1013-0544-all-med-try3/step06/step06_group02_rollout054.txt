def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:

            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-15:

            h[mask] += (diff / tot) * h[mask]

    m = c.max()

    u[0] = w[n-1]

        u[l] = w[n-1-l] + w[n-1+l]

    # First term: (1-2h)*u[0]
    grad = (1 - 2*h) * u[0]

    # But (1-h) has length n, u has length n, so conv1 has length 2n-1
    # For i=0,...,n-1: conv1[i] = sum_{l=0}^{i} (1-h)[i-l]*u[l] (since i-l >= 0 requires l <= i)
    #   minus the l=0 term: sum_{l=1}^{i} (1-h)[i-l]*u[l] = conv1[i] - (1-h)[i]*u[0]
    
    second_term = conv1[:n] - (1-h) * u[0]
    
    # Third term: sum_{l=1}^{n-1-i} h[i+l]*u[l] for each i
    # Let j = i+l, so l = j-i, range of j: i+1 to n-1

    # This is a cross-correlation type sum.
    # Let's define: for each i, sum_{j=i+1}^{n-1} h[j]*u[j-i]
    # Let m = j-i (m = 1 to n-1-i): sum_{m=1}^{n-1-i} h[i+m]*u[m]

    # I can compute this as: for m=1,...,n-1, h[i+m]*u[m] summed over valid i
    # This is like a "truncated correlation"

    # For a fixed m, the term h[i+m]*u[m] contributes to grad[i] for i = 0, ..., n-1-m
    # So: third_term[i] = sum_{m=1}^{n-1-i} h[i+m]*u[m]

    # Actually, let's think of it as: third_term[i] = sum_{m=1}^{n-1} h[i+m]*u[m] * 1_{i+m < n}
    # = sum_{m=0}^{n-1} h[i+m]*u[m] * 1_{i+m < n} (the m=0 term is h[i]*u[0] which we'd subtract)
    # = (correlation of h with u, at position i) - h[i]*u[0]
    # where correlation[a, b][i] = sum_m a[i+m]*b[m] (one-sided)
    
    # Actually this IS the "valid" cross-correlation:
    # np.correlate(h, u, 'valid') doesn't quite work because both have length n
    # np.correlate(h, u, 'full')[n-1+i] = sum_j h[j]*(u[j-(n-1+i)] conjugate...) 
    # Hmm, numpy's correlate is: result[m] = sum_n a[n+m] * conj(v[n])
    # So correlate(h, u, 'full')[m] = sum_n h[n+m] * u[n] for valid n
    # For m = i: sum_n h[n+i] * u[n] for n = 0,...,n-1-i
    # = sum_{n=0}^{n-1-i} h[n+i]*u[n]
    # This is exactly what we want! (with n replaced by m)
    
    corr = np.correlate(h, u[:n], mode='full')  # length 2n-1
    # corr[i] for i=0,...,n-1: sum_{m=0}^{n-1-i} h[m+i]*u[m]
    # We want sum_{m=1}^{n-1-i} h[m+i]*u[m] = corr[i] - h[i]*u[0]
    
    third_term = corr[:n] - h * u[0]
    
    grad += second_term - third_term
    grad *= dx
    return grad
