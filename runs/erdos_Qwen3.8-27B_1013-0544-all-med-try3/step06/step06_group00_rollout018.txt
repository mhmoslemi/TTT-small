def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, alpha, lambda_pen):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    logsum = np.log(exp_shifted.sum())
    f = (m + logsum / alpha)
    p = exp_shifted / exp_shifted.sum()
    g = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(p[j:j+n], g)
    # Add penalty gradient
    sum_diff = h.sum() - n / 2.0
    f += lambda_pen * sum_diff ** 2
    grad += 2 * lambda_pen * sum_diff
    return f, grad
