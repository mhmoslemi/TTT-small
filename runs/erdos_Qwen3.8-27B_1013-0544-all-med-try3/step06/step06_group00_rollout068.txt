```python
    """Project x onto {h: 0≤h≤1, sum(h)=n/2} using water-filling."""

    # Binary search for lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_candidate = np.clip(x - mid, 0.0, 1.0)
        s = h_candidate.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_value_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    probs = exps / exps.sum()
    val = m + (sm + np.log(exps.sum())) / alpha
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        p = probs[m_idx] * dx
        if p < 1e-16:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[k:] += p * (1.0 - h[:n - k])
            grad[:n - k] -= p * h[k:]
        else:
            kp = -k
            grad[:n - kp] += p * (1.0 - h[kp:])
            grad[kp:] -= p * h[:n - kp]
    return val, grad

def active_grad(h, n, n_active=20):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    threshold = m - 0.02
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(c)[::-1][:n_active]
    else:
        active = active[:n_active]
    probs = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for m_idx in active:
        p = probs[list(active).index(m_idx)] * dx
        k = m_idx - (n - 1)
        if k >= 0:
            grad[k:] += p * (1.0 - h[:n - k])
            grad[:n - k] -= p * h[k:]
        else:
            kp = -k
            grad[:n - kp] += p * (1.0 - h[kp:])
            grad[kp:] -= p * h[:n - kp]
    return m, grad

def projected_gradient_opt(h_init, n, deadline, alpha_start=100, alpha_end=500000, max_iters=20000, use_active=False):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_active = 20
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        if use_active:
            val, grad = active_grad(h, n, n_active)
        else:
            val, grad = smooth_value_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (alpha * gnorm + 1)
        if step > 0.05:
            step = 0.05
        for _ls in range(20):
            h_trial = project_feasible(h - step * grad, n)
            if use_active:
                trial_val, _ = active_grad(h_trial, n, n_active)
            else:
                trial_val, _ = smooth_value_grad(h_trial, n, alpha)
            if trial_val <= val + 1e-12:
                h = h_trial
                break
            step *= 0.5
        else:
            h = project_feasible(h - step * grad, n)
        if it % 100 == 0:
            cur_c = compute_c5(h, n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            n_active = max(5, n_active - 1)
    cur_c = compute_c5(h, n)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
