def smooth_parts(h, beta):
    n = len(h); dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    v = beta*dx*corr
    m = v.max()
    ev = np.exp(v-m)
    W = ev/ev.sum()
    S = (m + np.log(ev.sum()))/beta
    G = np.zeros(n)
    nz = np.nonzero(W > 1e-9)[0]
    for idx in nz:
        k = idx-(n-1)
        G += W[idx]*dx*corr_grad_k(h,k)
    sharp = float(corr.max())*dx
    return S, G, sharp

def smooth_parts(h,beta):
    n=len(h); dx=2.0/n

    v=beta*dx*corr

def smooth_parts(h, beta):
    n=len(h); dx=2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    v = beta*dx*corr
    m = float(v.max()); ev = np.exp(v-m); s=float(ev.sum()); W = ev/s
    S = (m + mathlog(s))/beta
    idx0=n-1; w0=W[idx0]
    wpos=W[idx0+1: 2*n-1]  # length n-1
    wneg=W[0: idx0][::-1]   # length n-1

    pp = np.empty(n); pp[0]=0.0; pp[1:]=np.cumsum(wpos)
    pn = np.empty(n); pn[0]=0.0; pn[1:]=np.cumsum(wneg)

import numpy as np
import math
import time
from scipy.optimize import minimize as spminimize

# ------------------------------------------------------------------
# Module globals used by the (optional) SLSQP polish step
# ------------------------------------------------------------------
_POL_BETA = 1.0
_POL_TARGET = 0.0


def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_grad_k(h, k):
    """Exact gradient (w.r.t. h) of the lag-`k` correlation mass.
    k is the signed lag as used by np.correlate (np.correlate index - (n-1))."""
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    i = np.arange(n)
    l = abs(k)
    if k > 0:
        t1 = (i >= l).astype(float)
    else:
        t1 = (i < (n - l)).astype(float)
    t2 = np.zeros(n)
    t2[: max(0, n - l)] = h[l:n]
    t3 = np.zeros(n)
    t3[l:n] = h[0: (n - l) if (n - l) > 0 else 0]
    return t1 - t2 - t3


def smooth_parts(h, beta):
    """Return (soft-max value S, gradient G of S, sharp max-value)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = beta * dx * corr
    m = float(v.max())
    ev = np.exp(v - m)
    s = float(ev.sum())
    W = ev / s
    S = (m + math.log(s)) / beta
    nz = np.nonzero(W > 1e-7)[0]
    G = np.zeros(n)
    if int(nz.shape[0]) <= 80:
        for idx in nz:
            kk = int(idx) - (n - 1)
            G += W[idx] * dx * corr_grad_k(h, kk)
    else:
        idx0 = n - 1
        w0 = float(W[idx0])
        wpos = W[idx0 + 1: 2 * n - 1]
        wneg = W[0: idx0][::-1]
        cvec = wpos + wneg
        pp = np.empty(n); pp[0] = 0.0; pp[1:] = np.cumsum(wpos)
        pn = np.empty(n); pn[0] = 0.0; pn[1:] = np.cumsum(wneg)
        T1 = w0 + pp + pn[::-1]
        cfull = np.zeros(n); cfull[1:] = cvec
        termA = np.convolve(cfull, h, mode='full')[0:n]
        cb = np.correlate(h, cfull, mode='full')
        termB = cb[idx0: 2 * n - 1]
        G = dx * (T1 - termA - termB - 2.0 * w0 * h)
    if not np.all(np.isfinite(G)):
        G = np.zeros(n)
    sharp = float(corr.max()) * dx
    return S, G, sharp


def project_sum(p, c):
    """Euclidean projection onto {0<=h<=1, sum(h)=c}."""
    p = np.asarray(p, float)
    a = float(np.min(p)) - 2.0
    b = float(np.max(p)) + 1.0
    lo = a
    hi = b
    for _ in range(44):
        mid = 0.5 * (lo + hi)
        f = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if f > c:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    lam = 0.5 * (lo + hi)
    return np.clip(p - lam, 0.0, 1.0)


def resample_to_n(hold, n):
    hold = np.asarray(hold, float)
    m = len(hold)
    if m == n:
        return hold.copy()
    x0 = np.arange(m) * (2.0 / m)
    x1 = np.arange(n) * (2.0 / n)
    return np.interp(x1, x0, hold)

def topk_by_c5(scored, K):
    items = list(range(len(scored)))
    out = []
    K = min(K, len(items))
    for _ in range(K):
        bi = items[0]
        bc = scored[items[0]][0]
        for j in range(1, len(items)):
            if scored[items[j]][0] < bc:
                bc = scored[items[j]][0]
                bi = items[j]
        out.append(scored[bi][1])
        items.remove(bi)
    return out

        return [(30, 300), (150, 400), (700, 500), (3000, 600), (12000, 700)]

        return [(30, 250), (150, 350), (700, 450), (3000, 500), (12000, 600)]

def optimize_seed(h0, bsteps, dl, rng):

    h = project_sum(np.clip(h0.copy(), 0.0, 1.0), n / 2.0)
    _, _, c5 = smooth_parts(h, bsteps[0][0])
    best = h.copy()
    bestc5 = c5

    return best, bestc5


def _sp_obj(x):
    val, _, _ = smooth_parts(x, _POL_BETA)
    return val


def _sp_jac(x):
    _, g, _ = smooth_parts(x, _POL_BETA)
    return g


def _sp_con(x):
    return float(np.sum(x)) - _POL_TARGET


def _sp_con_jac(x):
    return np.ones(len(x))


def polish_slsq(h, n, pbeta):
    global _POL_BETA, _POL_TARGET
    _POL_BETA = pbeta
    _POL_TARGET = n / 2.0
    x0 = np.clip(h, 0.0, 1.0)
    x0 = project_sum(x0, _POL_TARGET)
    cons = [{'type': 'eq', 'fun': _sp_con, 'jac': _sp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    mx = 400 if n <= 300 else 250
    res = spminimize(_sp_obj, x0, method='SLSQP', bounds=bounds,
                     constraints=cons, jac=_sp_jac,
                     options={'maxiter': mx, 'ftol': 1e-14, 'disp': False})
    hr = project_sum(np.clip(res.x, 0.0, 1.0), _POL_TARGET)
    return hr, c5_of(hr)

    return (fin, fc5, len(fin))
