        best_h = h.copy()
    return best_h, best_c5


def multi_start_slsqp(h0, n, dx, target_sum, n_restarts=3, perturbation=0.01, time_limit_s=90.0):
    t_start = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_restarts):
        if time.time() - t_start > time_limit_s:
            break
        x0 = h0.copy()
        if r > 0:
            scale = perturbation / r
            x0 += np.random.randn(n) * scale
            x0 = project_feasible(x0, target_sum)
        h_new, c5 = slsqp_optimize(x0, n, dx, target_sum, [50, 200, 1000], maxiter_each=1500)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)

    n_parent = len(h_parent)
    parent_c5 = compute_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = parent_c5

    # Try multiple resolutions
    resolutions = []
    # Include parent resolution
    if n_parent not in resolutions:
        resolutions.append(n_parent)
    # Higher resolutions
    for nr in [100, 140, 200, 280, 350]:
        if nr > n_parent and time_left() > 100:
            resolutions.append(nr)
        elif nr < n_parent and nr > 30 and time_left() > 200:
            resolutions.append(nr)

    # Sort by priority: start with parent, then higher resolutions
    resolutions = [n_parent] + [r for r in sorted(set(resolutions) - {n_parent}, reverse=True) if time_left() > 60]

    for n in resolutions:
        if time_left() < 80:
            break
        if n < 30:
            continue

        dx = 2.0 / n
        target_sum = n / 2.0

        # Initialize from parent
        h_init = resample_h(h_parent, n)
        h_init = project_feasible(h_init, target_sum)

        # Also create a binarized version
        h_bin = np.where(h_init > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, target_sum)

        # Also create a smoothed version
        from scipy.ndimage import uniform_filter1d
        h_smooth = uniform_filter1d(h_init, size=5, mode='wrap')
        h_smooth = project_feasible(h_smooth, target_sum)

        candidates = [h_init, h_bin, h_smooth]

        for h_cand in candidates:
            if time_left() < 60:
                break

            # Phase 1: SLSQP with continuation
            h_opt, c5_opt = slsqp_optimize(h_cand, n, dx, target_sum,
                                            betas=[20, 80, 300, 1000, 3000],
                                            maxiter_each=2000, top_m=30)

            # Phase 2: Subgradient polish
            if time_left() > 40:
                h_sg, c5_sg = subgrad_polish(h_opt, n, dx, target_sum,
                                             n_iter=40000, time_limit_s=40.0)
                if c5_sg < c5_opt:
                    h_opt = h_sg
                    c5_opt = c5_sg

            # Phase 3: Coordinate descent polish
            if time_left() > 30:
                h_cd, c5_cd = coord_descent_polish(h_opt, target_sum,
                                                   n_iter=300, time_limit_s=25.0)
                if c5_cd < c5_opt:
                    h_opt = h_cd
                    c5_opt = c5_cd

            # Phase 4: Another SLSQP round from improved point
            if time_left() > 30:
                h_opt2, c5_opt2 = slsqp_optimize(h_opt, n, dx, target_sum,
                                                  betas=[500, 2000, 8000],
                                                  maxiter_each=1500, top_m=20)
                if c5_opt2 < c5_opt:
                    h_opt = h_opt2
                    c5_opt = c5_opt2

            # Phase 5: Final coordinate descent
            if time_left() > 15:
                h_cd2, c5_cd2 = coord_descent_polish(h_opt, target_sum,
                                                     n_iter=200, time_limit_s=15.0)
                if c5_cd2 < c5_opt:
                    h_opt = h_cd2
                    c5_opt = c5_cd2

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Multi-start at best resolution found so far
        if n == n_parent and time_left() > 120:
            h_ms, c5_ms = multi_start_slsqp(best_h, n, 2.0/n, n/2.0,
                                            n_restarts=4, perturbation=0.015,
                                            time_limit_s=100.0)
            if c5_ms < best_c5:
                best_h = h_ms.copy()
                best_c5 = c5_ms

            # Coordinate descent on the multi-start result
            if time_left() > 20:
                h_cd3, c5_cd3 = coord_descent_polish(best_h, n/2.0,
                                                     n_iter=200, time_limit_s=20.0)
                if c5_cd3 < best_c5:
                    best_h = h_cd3
                    best_c5 = c5_cd3

    # Final: try to improve at the best resolution with more aggressive optimization
    if time_left() > 30:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        ts_best = n_best / 2.0

        # Try SLSQP with very high smoothness
        h_final, c5_final = slsqp_optimize(best_h, n_best, dx_best, ts_best,
                                           betas=[5000, 15000, 40000],
                                           maxiter_each=3000, top_m=15)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

        # One more coordinate descent
        if time_left() > 15:
            h_cd_final, c5_cd_final = coord_descent_polish(best_h, ts_best,
                                                           n_iter=300, time_limit_s=15.0)
            if c5_cd_final < best_c5:
                best_h = h_cd_final
                best_c5 = c5_cd_final

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
