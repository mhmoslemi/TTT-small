import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    c_shift -= c_shift.max()
    weights = np.exp(c_shift)
    weights /= weights.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = weights[m] * dx
        if k >= 0:
            nv = n - k
            grad[k:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[k:n])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    return grad

def smooth_max_obj(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    else:

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:

    return float(val)

def penalty_grad(x, n, alpha, mu):

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200, alphas=None):

    bounds = [(0.0, 1.0)] * n

def coordinate_descent(h, n, deadline, max_rounds=50):

            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            # Try to optimize h[i] by golden section or simple search
            # The objective as function of h[i] is max of quadratics
            # Use a simple grid search around current value
            best_local = old_val
            best_local_c = best_c
            # Coarse search
            for frac in np.linspace(0.0, 1.0, 21):
                h[i] = frac
                c = compute_c5(h, n)
                if c < best_local_c - 1e-14:
                    best_local_c = c
                    best_local = frac
            # Fine search around best_local
            lo = max(0.0, best_local - 0.05)

                improved = True

    return h, best_c

            t_l = min(time.time() + 30, deadline - 10)
            h_l, c_l, n_l = try_resolution(best_h, best_n, n_try, t_l, rng)
            if c_l < best_c - 1e-12:
                best_c = c_l
                best_h = h_l
                best_n = n_l

    # Phase 8: Gradient-based pair refinement
    if time.time() < deadline - 20:
        t_g = min(time.time() + 40, deadline - 5)
        best_h, best_c = gradient_based_pair(best_h, best_n, t_g, alpha=5000, max_rounds=50)

    # Phase 9: Final coordinate descent polish
    if time.time() < deadline - 5:
        best_h, best_c = coordinate_descent(best_h, best_n, deadline - 2, max_rounds=20)

    # Phase 10: Ultra-fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = pair_coordinate_descent(best_h, best_n, deadline - 1,
                                                  step_sizes=[0.001, 0.0005, 0.0002, 0.0001])

    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
