def coordinate_minimize(a, b, lam):
    """
    Minimize f(x) = max_d (a[d]*x + b[d]) + lam*x subject to 0 <= x <= 1.
    a, b are arrays of length m (number of lags).
    Returns the optimal x.
    """
    m = len(a)
    # f(x) = max_d (a[d]*x + b[d]) + lam*x = max_d ((a[d]+lam)*x + b[d])
    a_eff = a + lam
    
    # Evaluate at boundaries
    f0 = np.max(b)
    f1 = np.max(a_eff + b)
    
    if f0 <= f1:
        # Minimum might be at 0 or where slope becomes 0
        # Check slope at 0: subdifferential is [min_{d in argmax} a_eff[d], max_{d in argmax} a_eff[d]]
        max_at_0 = f0
        tied = np.where(b == max_at_0)[0]
        slopes = a_eff[tied]
        if np.min(slopes) >= 0:

        # Need to go right
        # Find where the current max line is overtaken
        # Work iteratively
        x = 0.0
        for _ in range(m):  # at most m iterations

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 0.9999
            if mask.sum() == 0:
                mask = np.ones(len(h), dtype=bool)
        else:
            mask = h > 0.0001
            if mask.sum() == 0:
                mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def resample_h(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def min_1d_convex_pl(a, b):
    m = len(a)
    eps = 1e-12
    max0 = np.max(b)
    tied0 = np.where(b >= max0 - eps)[0]
    if len(tied0) == 0:
        tied0 = np.array([np.argmax(b)])
    if np.min(a[tied0]) >= -eps:
        return 0.0

    max1 = np.max(a + b)
    tied1 = np.where(a + b >= max1 - eps)[0]
    if len(tied1) == 0:
        tied1 = np.array([np.argmax(a + b)])
    if np.max(a[tied1]) <= eps:
        return 1.0

    lo = 0.0
    hi = 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        idx = np.argmax(vals)
        if a[idx] < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5


def coord_descent(h, n, lam, n_passes, target_sum, time_limit, t_start):
    nm1 = n - 1
    h = h.copy()
    for p in range(n_passes):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            old = h[i]
            a = np.zeros(2 * n - 1)

            m1_lo = max(0, nm1 - i)
            m1_hi = min(2 * nm1, nm1 + i)
            if m1_lo <= m1_hi:
                idx1 = (i - nm1) + np.arange(m1_lo, m1_hi + 1)
                a[m1_lo:m1_hi + 1] += (1.0 - h[idx1])

            m2_lo = max(0, i)
            m2_hi = min(2 * nm1, i + nm1)
            if m2_lo <= m2_hi:
                idx2 = (i + nm1) - np.arange(m2_lo, m2_hi + 1)
                a[m2_lo:m2_hi + 1] += -h[idx2]

            b = corr - a * old
            a_eff = a + lam
            x_opt = min_1d_convex_pl(a_eff, b)
            h[i] = x_opt

        h = project_feasible(h, target_sum)
        lam += 0.08 * (h.sum() - target_sum)
        lam = np.clip(lam, -5.0, 5.0)
    return h, lam


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_smooth_slsqp(h, dx, beta, target_sum):
    h_c = np.clip(h, 0, 1)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    pen = 100.0 * (h_c.sum() - target_sum) ** 2
    return smooth_max(corr, beta) * dx + pen


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    timeout = budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)

    best_h = None
    best_c5 = 1.0
    best_n = 0

    n_values = [100, 120, 80, 140]

    for n in n_values:
        if time.time() - t0 > timeout:
            break

        target_sum = n / 2.0
        if len(h_parent) != n:
            h_init = resample_h(h_parent, n)
        else:
            h_init = h_parent.copy()
        h_init = project_feasible(h_init, target_sum)

        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()
            best_n = n

        h_curr = h_init.copy()
        lam = 0.0
        passes_per_phase = [300, 500, 500, 300]
        lr_schedules = [0.08, 0.05, 0.03, 0.02]

        for phase in range(4):
            if time.time() - t0 > timeout:
                break
            nm1 = n - 1
            for p in range(passes_per_phase[phase]):
                if time.time() - t0 > timeout:
                    break
                corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
                for i in range(n):
                    old = h_curr[i]
                    a = np.zeros(2 * n - 1)

                    m1_lo = max(0, nm1 - i)
                    m1_hi = min(2 * nm1, nm1 + i)
                    if m1_lo <= m1_hi:
                        idx1 = (i - nm1) + np.arange(m1_lo, m1_hi + 1)
                        a[m1_lo:m1_hi + 1] += (1.0 - h_curr[idx1])

                    m2_lo = max(0, i)
                    m2_hi = min(2 * nm1, i + nm1)
                    if m2_lo <= m2_hi:
                        idx2 = (i + nm1) - np.arange(m2_lo, m2_hi + 1)
                        a[m2_lo:m2_hi + 1] += -h_curr[idx2]

                    b = corr - a * old
                    a_eff = a + lam
                    x_opt = min_1d_convex_pl(a_eff, b)
                    h_curr[i] = x_opt

                h_curr = project_feasible(h_curr, target_sum)
                lam += lr_schedules[phase] * (h_curr.sum() - target_sum)
                lam = np.clip(lam, -5.0, 5.0)

                if (p + 1) % 50 == 0:
                    c5_check = compute_c5(h_curr)
                    if c5_check < best_c5:
                        best_c5 = c5_check
                        best_h = h_curr.copy()
                        best_n = n

        c5_final = compute_c5(h_curr)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_curr.copy()
            best_n = n

    if time.time() - t0 < timeout:
        h_work = best_h.copy()
        n_work = best_n
        target = n_work / 2.0
        dx_w = 2.0 / n_work
        bounds = Bounds(np.zeros(n_work), np.ones(n_work))
        for beta in [500, 2000, 10000]:
            if time.time() - t0 > timeout:
                break
            try:
                res = minimize(obj_smooth_slsqp, h_work, args=(dx_w, beta, target),
                               method='SLSQP', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    if time.time() - t0 < timeout - 10:
        for restart in range(3):
            if time.time() - t0 > timeout:
                break
            n_r = best_n
            target_r = n_r / 2.0
            h_r = best_h.copy()
            h_r += np.random.randn(n_r) * 0.01
            h_r = project_feasible(h_r, target_r)
            lam_r = 0.0
            nm1_r = n_r - 1
            for p in range(400):
                if time.time() - t0 > timeout:
                    break
                corr_r = np.correlate(h_r, 1.0 - h_r, mode='full')
                for i in range(n_r):
                    old_r = h_r[i]
                    a_r = np.zeros(2 * n_r - 1)
                    m1_lo = max(0, nm1_r - i)
                    m1_hi = min(2 * nm1_r, nm1_r + i)
                    if m1_lo <= m1_hi:
                        idx1 = (i - nm1_r) + np.arange(m1_lo, m1_hi + 1)
                        a_r[m1_lo:m1_hi + 1] += (1.0 - h_r[idx1])
                    m2_lo = max(0, i)
                    m2_hi = min(2 * nm1_r, i + nm1_r)
                    if m2_lo <= m2_hi:
                        idx2 = (i + nm1_r) - np.arange(m2_lo, m2_hi + 1)
                        a_r[m2_lo:m2_hi + 1] += -h_r[idx2]
                    b_r = corr_r - a_r * old_r

                        best_c5 = c5_r
                        best_h = h_r.copy()

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
