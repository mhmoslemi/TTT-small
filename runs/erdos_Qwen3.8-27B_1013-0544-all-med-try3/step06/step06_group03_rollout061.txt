- Take step in negative gradient direction (with step size via backtracking)
- Project to feasible set
- Repeat

For the gradient-guided local search:
- Compute gradient
- For each point j, the "descent direction" is -grad[j]
- If grad[j] > 0, we want to decrease h[j]; if grad[j] < 0, increase h[j]
- Find pairs (i, j) where decreasing h[i] and increasing h[j] (or vice versa) gives the most improvement
- Actually, for a pair swap: move δ from i to j. The change in objective ≈ δ * (grad[j] - grad[i])
- We want to choose (i, j, δ) to minimize this, i.e., maximize grad[i] - grad[j]
- So: find i with largest positive gradient and j with largest negative gradient
- Move small δ from i to j

This is essentially a greedy approach based on the gradient.

        if not improved:
            stall_count += 1
            lr = max(lr * 0.5, 1e-10)
            if stall_count > 50:
                break
            continue
        else:
            lr = min(lr * 1.05, 0.1)
    
    return h, best_c

def grad_guided_search(h, n, alpha, deadline, max_rounds=200):
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        _, grad = obj_grad(h, n, alpha)
        g_sorted = np.argsort(grad)
        
        found = False
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            # Try top/bottom pairs
            max_pairs = min(50, n // 2)
            for a in range(max_pairs):
                if time.time() > deadline + 10:
                    return h, best_c
                for b in range(n - 1 - a, n - max_pairs + a, -1):
                    i = g_sorted[b]
                    j = g_sorted[a]
                    if grad[i] - grad[j] < 1e-10:
                        break
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            found = True
                        else:
                            h[i] += d
                            h[j] -= d
                        if found:
                            break
                if found:
                    break
            if found:
                break
            # Also try middle pairs
            mid = n // 2
            for a in range(mid - 10, mid + 10):
                if a < 0 or a >= n - 1:
                    continue
                for b in range(mid - 10, mid + 10):
                    if b < a + 1 or b >= n:
                        continue
                    i = g_sorted[b]
                    j = g_sorted[a]
                    if abs(int(i) - int(j)) < 2:
                        continue
                    if grad[i] - grad[j] < 1e-10:
                        continue
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            found = True
                        else:
                            h[i] += d
                            h[j] -= d
                        if found:
                            break
                if found:
                    break
        if not found:
            break
    
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def optimize_at_resolution(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    remaining = deadline - time.time()
    if remaining < 20:
        return h, best_c
    
    # SLSQP phase
    t_phase = time.time() + min(remaining * 0.25, 60)
    for alpha in [2000, 10000, 50000]:
        if time.time() > t_phase:
            break
        h_new = slsqp_step(h, n, alpha, maxiter=600)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
    
    # L-BFGS-B with penalty
    t_phase2 = time.time() + min(remaining * 0.2, 50)
    for penalty in [100, 1000, 10000]:
        if time.time() > t_phase2:
            break
        h_new = lbfgs_step(h, n, 20000, penalty, maxiter=1000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
    
    # Projected gradient descent
    t_phase3 = time.time() + min(remaining * 0.2, 40)
    for alpha in [10000, 50000, 100000]:
        if time.time() > t_phase3:
            break
        h_new, c_new = proj_grad_descent(h, n, alpha, lr_init=0.02,
                                         maxiter=5000, deadline=t_phase3)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
    
    # Gradient-guided local search
    t_phase4 = time.time() + min(remaining * 0.2, 40)
    h, best_c = grad_guided_search(h, n, 50000, t_phase4, max_rounds=200)
    
    # Near-pair search for fine tuning
    if time.time() < deadline - 5:
        h, best_c = near_pair_search(h, n, deadline - 2,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                     radius=40 if n > 200 else 100)
    
    # Final SLSQP polish
    if time.time() < deadline - 3:
        h_new = slsqp_step(h, n, 200000, maxiter=500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimize at initial resolution (250)
    alloc1 = min(200, (deadline - t0) * 0.22)
    h_opt, c_opt = optimize_at_resolution(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    
    # Phase 2: Try n=350
    if time.time() < deadline - 150:
        n_try = 350
        h_interp = upsample(best_h, len(best_h), n_try)
        alloc2 = min(150, (deadline - time.time()) * 0.25)
        if alloc2 > 30:
            h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc2, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    
    # Phase 3: Try n=500
    if time.time() < deadline - 120:
        n_try = 500
        h_interp = upsample(best_h, len(best_h), n_try)
        alloc3 = min(130, (deadline - time.time()) * 0.25)
        if alloc3 > 30:
            h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc3, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    
    # Phase 4: Random restarts at best resolution
    if time.time() < deadline - 60:
        n_best = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.06 * r / max(1, n_restarts)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 15) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    
    # Phase 5: Extended local search on best
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = grad_guided_search(best_h, n, 100000, deadline - 3, max_rounds=500)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3,
                                              [0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                              radius=50)
    
    # Phase 6: Final polish
    if time.time() < deadline - 3:
        n = len(best_h)
        h_new = slsqp_step(best_h, n, 500000, maxiter=500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
