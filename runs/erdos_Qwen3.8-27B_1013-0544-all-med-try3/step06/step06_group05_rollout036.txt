import numpy as np
from scipy.optimize import minimize
from scipy.special import softmax
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, pen_weight=1000.0):
    """Smooth max of correlation + penalty for sum constraint."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    sum_exp = exp_ac.sum()
    obj = (ac_max + np.log(sum_exp)) / alpha
    # Penalty for sum constraint
    s = x.sum() - n / 2.0
    obj += pen_weight * s * s
    return obj

def smooth_max_grad(x, n, alpha, pen_weight=1000.0):
    """Gradient of smooth max + penalty."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    sum_exp = exp_ac.sum()
    w = exp_ac / sum_exp
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wm = w[n-1+k]
        if wm < 1e-15:
            continue
        if k >= 0:
            if n - k > 0:
                grad[k:] += wm * (1.0 - x[:n-k])
                grad[:n-k] -= wm * x[k:]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - x[:n-kk])
            grad[:n-kk] -= wm * x[kk:]
    
    # Penalty gradient
    s = x.sum() - n / 2.0
    grad += 2.0 * pen_weight * s * np.ones(n)
    
    return grad

def combined_obj_grad(x, n, alpha, pen_weight=1000.0):
    return smooth_max_obj(x, n, alpha, pen_weight), smooth_max_grad(x, n, alpha, pen_weight)

Let me also add one more trick: after L-BFGS-B convergence, do a "steepest descent" with the exact (non-smooth) max. At a point where the max is achieved at lag k*, the steepest descent direction is -∂C_{k*}/∂h (the negative gradient of the worst lag). def steepest_on_worst(h, n, deadline, max_iter=10000):
    h = h.copy()
    best_c = compute_c5(h, n)
    step_size = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        all_c = compute_all_c(h, n)
        k_star = np.argmax(all_c)
        # Compute gradient of C_{k_star}
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            kk = -k
            grad[kk:] += (1.0 - h[:n-kk])
            grad[:n-kk] -= h[kk:]
        # Negative gradient descent (we want to minimize C_{k_star})
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            step_size *= 0.8
            if step_size < 1e-8:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, pen_w):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    s = exp_ac.sum()
    obj = (ac_max + np.log(s)) / alpha
    sv = x.sum() - n / 2.0
    obj += pen_w * sv * sv
    return float(obj)

def smax_grad(x, n, alpha, pen_w):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    s = exp_ac.sum()
    w = exp_ac / s
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wm = w[n - 1 + k]
        if wm < 1e-15:
            continue
        if k >= 0:
            grad[k:] += wm * (1.0 - x[:n - k])
            grad[:n - k] -= wm * x[k:]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - x[:n - kk])
            grad[:n - kk] -= wm * x[kk:]
    sv = x.sum() - n / 2.0
    grad += 2.0 * pen_w * sv
    return grad

def run_lbbox(h, n, alpha, pen_w, maxiter):
    bounds = [(0.0, 1.0)] * n
    result = minimize(smax_obj, h, jac=smax_grad,
                      args=(n, alpha, pen_w),
                      method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'iprint': -1})
    return project_feasible(result.x, n)

def pair_search(h, n, deadline, deltas, max_passes=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    if hi + d <= 1.0 and hj - d >= 0.0:
                        h[i] = hi + d
                        h[j] = hj - d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] = hi
                            h[j] = hj
                    if hi - d >= 0.0 and hj + d <= 1.0:
                        h[i] = hi - d
                        h[j] = hj + d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] = hi
                            h[j] = hj
            if not improved:
                break
    return h, best_c

def steepest_worst(h, n, deadline, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.005
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        all_c = compute_all_c(h, n)
        k_star = int(np.argmax(all_c))
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            kk = -k
            grad[kk:] += (1.0 - h[:n - kk])
            grad[:n - kk] -= h[kk:]
        h_try = h - step * grad
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
            stall = 0
        else:
            stall += 1
            step *= 0.7
            if step < 1e-10:
                break
            if stall > 50:
                break
    return best_h, best_c

def multi_restart_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    scales = [0.003, 0.008, 0.015, 0.025, 0.04]
    for r in range(len(scales)):
        if time.time() > deadline - 30:
            break
        scale = scales[r % len(scales)]
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        alloc_end = min(deadline - 25, time.time() + 80)
        for alpha in [50000, 200000]:
            if time.time() > alloc_end:
                break
            h_new = run_lbbox(h_pert, n, alpha, 100.0, maxiter=300)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h_pert = h_new.copy()
    return best_h, best_c

def coordinate_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    for _ in range(200):
        if time.time() > deadline:
            break
        all_c = compute_all_c(h, n)
        k_star = int(np.argmax(all_c))
        k = k_star - (n - 1)
        if k >= 0:
            contrib = h[:n - k] * (1.0 - h[k:])
            idx = int(np.argmax(contrib))
            candidates = [(idx, 'up'), (idx + k, 'down')]
        else:
            kk = -k
            contrib = h[kk:] * (1.0 - h[:n - kk])
            idx = int(np.argmax(contrib))
            candidates = [(idx + kk, 'up'), (idx, 'down')]
        found = False
        for pos, direction in candidates:
            if direction == 'up':
                h_try = h.copy()
                h_try[pos] += step
                h_try = project_feasible(h_try, n)
            else:
                h_try = h.copy()
                h_try[pos] -= step
                h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h = h_try
                found = True
                break
        if not found:
            step *= 0.7
            if step < 1e-8:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
