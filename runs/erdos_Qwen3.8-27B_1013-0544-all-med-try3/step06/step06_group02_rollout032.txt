def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def block_move_search(h, n, deadline, max_block=20):

                        return h, best_c

                    h_new = h.copy()
                    T *= cooling
                    continue
            else:
                T *= cooling
                continue

                continue
            temp = h_new[i:i+bs].copy()
            h_new[i:i+bs] = h_new[j:j+bs]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 1.0), (500, 5.0), (2000, 20.0),
        (10000, 100.0), (100000, 500.0), (500000, 2000.0),
    ]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        try:
            res = minimize(smooth_max_obj_pen, h, jac=smooth_max_grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-12, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)

            if c_new < best_c - 1e-15:

                h = h_new
        except Exception:

    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=1000):

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]

def pair_search_exhaustive(h, n, deadline, deltas, radius=None):
    """Exhaustive pair search: try all pairs within radius for each delta."""
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c

                hi = min(n, i + radius + 1) if radius else n

                        continue

def coordinate_descent(h, n, deadline):
    """1D coordinate descent: for each i, find best h[i] given others fixed."""
    best_c = compute_c5(h, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Binary search for optimal h[i]
            # The objective as a function of h[i] is piecewise linear (for the max)
            # or smooth (for smooth max). Try a few values.
            center = h[i]
            candidates = [center]
            for d in [0.01, 0.005, 0.002, 0.001]:
                if center + d <= 1:
                    candidates.append(center + d)
                if center - d >= 0:
                    candidates.append(center - d)
            
            best_val = center
            best_c_i = best_c
            for cv in candidates:
                if abs(cv - center) < 1e-10:
                    continue
                # Compute new sum
                new_sum = h.sum() - center + cv
                if abs(new_sum - n/2.0) > 0.01:
                    continue
                h[i] = cv
                c_i = compute_c5(h, n)
                if c_i < best_c_i - 1e-14:
                    best_c_i = c_i
                    best_val = cv
                h[i] = center
            
            if best_val != center:
                h[i] = best_val
                if best_c_i < best_c - 1e-14:
                    best_c = best_c_i
                    improved = True
                h[i] = best_val
    return h, best_c

            h_work[i] = new_i
            h_work[j] = new_j
        else:
            # Larger pair move
            d = rng.uniform(0.005, 0.015)
            if rng.random() < 0.5:
                new_i = old_i + d
                new_j = old_j - d
            else:
                new_i = old_i - d
                new_j = old_j + d
            if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                T *= cooling
                continue
            h_work[i] = new_i
            h_work[j] = new_j
        
        c_new = compute_c5(h_work, n)
        delta = c_new - best_c
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            if c_new < best_c:
                best_c = c_new
                best_h = h_work.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject - revert
            h_work[i] = old_i
            h_work[j] = old_j
            no_improve += 1
        
        T *= cooling
        
        # Reheat if stuck
        if no_improve > 500:
            T = max(T, 0.001)
            no_improve = 0
    
    return best_h, best_c

Let me now think about whether there's a fundamentally different approach that could help.

**Alternative: Direct search on the correlation vector**

Instead of optimizing h directly, what if we think about what the correlation vector c = correlate(h, 1-h) should look like? We want max(c) * dx to be small, subject to the constraint that c comes from some valid h.

The constraint is: c[m] = sum_i h[i] * (1-h[i+m-n+1]) for each lag m.

This is a non-linear constraint on h. But maybe we can think about it in terms of the "energy" distribution across lags.

For a ideal solution, we'd want c[m] to be as uniform as possible (all at the minimum possible value). But there are constraints:
- c[0] (or c[n-1] for the zero lag) = sum h[i]*(1-h[i]) which is bounded
- The sum of all c[m] has a fixed relationship to h

Actually, c[n-1] (zero lag) = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]² = n/2 - sum h[i]²

So sum h[i]² = n/2 - c[n-1]/dx... wait, let me be more careful.

c[n-1] = sum_i h[i] * (1-h[i]) = sum h[i] - sum h[i]² = n/2 - sum h[i]²

So sum h[i]² = n/2 - c[n-1]

And c[n-1] * dx = (n/2 - sum h[i]²) * dx = (n/2 - sum h[i]²) * 2/n = 1 - 2*sum(h[i]²)/n

So the zero-lag overlap is 1 - 2*sum(h²)/n. To minimize this, we want to maximize sum(h²), which means making h as "binary" as possible (values close to 0 or 1).

For a purely binary h with 150 ones and 150 zeros (n=300), sum(h²) = 150, so zero-lag overlap = 1 - 2*150/300 = 0. Good.

So the zero lag is not a constraint for binary h. The constraints come from other lags.

For binary h, c[m] * dx = (number of 1s in h that are followed by a 0 at distance k) * dx, where k = (m - n + 1) * dx.

This is essentially a combinatorial problem: arrange 150 ones and 150 zeros in a sequence of length 300 to minimize the maximum number of "10" pairs at any distance.

This is related to the concept of "balanced sequences" or "low-discrepancy sequences."

For a periodic binary sequence with period p, the overlap at lag k is related to the autocorrelation of the sequence.

Actually, for a sequence of 1s and 0s with exactly 150 of each, the overlap at lag k (for k > 0) is:
O(k) = number of i such that h[i]=1 and h[i+k]=0 (with appropriate boundary handling)

For a random arrangement, E[O(k)] ≈ 150 * 150/300 = 75 (for large n). So O(k)/n * n ≈ 0.25, meaning C₅ ≈ 0.25 * 2 = 0.5 for random. We need to do much better than random.

For the alternating pattern 1,0,1,0,...: at lag 1, O(1) = 150 (every 1 is followed by 0). At lag 2, O(2) = 0. So max is 150, giving C₅ ≈ 150*2/300 = 1. That's terrible.

The optimal binary arrangement is a hard combinatorial problem. But we're allowed continuous values, which gives more freedom.

I think the key insight is: **the optimal solution is not binary**. Having intermediate values (gradients) in the transition regions allows a more even distribution of overlaps across all lags.
