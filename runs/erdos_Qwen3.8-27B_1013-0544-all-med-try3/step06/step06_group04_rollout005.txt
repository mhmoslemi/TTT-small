    bounds = [(0.0, 1.0)] * n
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_equalization(h, n, deadline, top_m=8, n_rounds=50):
    """Try to equalize the top lags by adjusting h."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for rnd in range(n_rounds):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        target = c[top_idx].mean()
        
        # Find which lag is worst
        worst_in_top = top_idx[np.argmax(c[top_idx])]
        lag = worst_in_top - (n - 1)
        
        if lag == 0:
            break
        m = abs(lag)
        
        # Compute gradient of the worst lag
        grad = np.zeros(n)
        if lag > 0:
            grad[:n - m] -= h_curr[m:]
            grad[m:] += (1.0 - h_curr[:n - m])
        else:
            grad[:n - m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n - m]
        
        # Project gradient onto feasible direction (maintain sum constraint)
        grad = grad - grad.mean()
        
        # Take a step in the negative gradient direction (to decrease the worst lag)
        step_size = 0.01 / (np.linalg.norm(grad) + 1e-10)
        h_new = h_curr - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            # If no improvement, try smaller step
            for ss in [0.005, 0.001, 0.0005]:
                step_size = ss / (np.linalg.norm(grad) + 1e-10)
                h_try = project_feasible(h_curr - step_size * grad, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h_try.copy()
                    h_curr = h_try
                    break
    
    return best_h, best_c

def projected_subgradient(h, n, deadline, alpha_init=100, n_iters=500):
    """Project subgradient method for smooth max objective."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() > deadline - 3:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        
        if lag == 0:
            # Minimize sum h^2 (to reduce lag 0)
            grad = 2.0 * h_curr - c.max()  # approximate
            grad = 1.0 - 2.0 * h_curr
        else:
            m = abs(lag)
            grad = np.zeros(n)
            if lag > 0:
                grad[:n - m] -= h_curr[m:]
                grad[m:] += (1.0 - h_curr[:n - m])
            else:
                grad[:n - m] += (1.0 - h_curr[m:])
                grad[m:] -= h_curr[:n - m]
        
        # Adjust for multiple active lags
        if n_iters > 50:
            top_k = np.argsort(c)[-min(5, 2*n-1):]
            for tk in top_k:
                tl = tk - (n - 1)
                if tl == 0:
                    tg = 1.0 - 2.0 * h_curr
                else:
                    tm = abs(tl)
                    tg = np.zeros(n)
                    if tl > 0:
                        tg[:n - tm] -= h_curr[tm:]
                        tg[tm:] += (1.0 - h_curr[:n - tm])
                    else:
                        tg[:n - tm] += (1.0 - h_curr[tm:])
                        tg[tm:] -= h_curr[:n - tm]
                w = max(0, c[tk] - c[top_k[0]] + c[top_k[0]]) ** 2
                grad += w * tg
        
        grad = grad - grad.mean()
        
        lr = 1.0 / (alpha_init + it * 0.1)
        h_new = h_curr - lr * grad * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    
    return best_h, best_c

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    
    # Use top lags to limit problem size
    top_lags = np.argsort(c)[-min(20, 2*n-1):]
    
    for k_idx in top_lags:
        lag = k_idx - (n - 1)
        grad = np.zeros(n)
        m = abs(lag)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        elif lag > 0:
            grad[:n - m] -= h_curr[m:]
            grad[m:] += (1.0 - h_curr[:n - m])
        else:
            grad[:n - m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n - m]
        cons.append(c[k_idx] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    # Also include all lags with c > 0.9 * max
    threshold = 0.9 * c.max()
    all_lags = np.where(c >= threshold)[0]
    for k_idx in all_lags:
        if k_idx not in top_lags:
            lag = k_idx - (n - 1)
            grad = np.zeros(n)
            m = abs(lag)
            if m == 0:
                grad[:] = 1.0 - 2.0 * h_curr
            elif lag > 0:
                grad[:n - m] -= h_curr[m:]
                grad[m:] += (1.0 - h_curr[:n - m])
            else:
                grad[:n - m] += (1.0 - h_curr[m:])
                grad[m:] -= h_curr[:n - m]
            cons.append(c[k_idx] + dx * (grad @ (h_var - h_curr)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, max_time=30)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_loop(h, n, deadline, max_iters=60):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.12
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    
    return best_h, best_c

def fine_local_search(h, n, deadline):
    """Very fine local search targeting the maximum lag."""
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved:
            if time.time() > deadline - 1:
                break
            improved = False
            c = compute_all_lags(h_curr, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            
            # Try moves that reduce the worst lag
            candidates = []
            for i in range(n - m):
                j = i + m
                if abs(h_curr[i] - h_curr[j]) > 1e-6:
                    if h_curr[i] > h_curr[j]:
                        candidates.append((i, j, -1))  # decrease i, increase j
                    else:
                        candidates.append((i, j, 1))   # increase i, decrease j
            # Sort by potential improvement
            if lag > 0:
                candidates.sort(key=lambda x: (h_curr[x[0]] - h_curr[x[1]]) * x[2], reverse=True)
            else:
                candidates.sort(key=lambda x: (h_curr[x[0]] + h_curr[x[1]]) * x[2], reverse=True)
            
            for (i, j, sign) in candidates[:30]:  # Try top 30 candidates
                if time.time() > deadline - 1:
                    break
                if sign == -1:
                    delta = min(d, h_curr[i] * 0.8, (1.0 - h_curr[j]) * 0.8)
                else:
                    delta = min(d, h_curr[j] * 0.8, (1.0 - h_curr[i]) * 0.8)
                if delta < 1e-10:
                    continue
                h_curr[i] += sign * delta
                h_curr[j] -= sign * delta
                h_curr = np.clip(h_curr, 0, 1)
                c_new = compute_c5(h_curr, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h_curr[i] -= sign * delta
                    h_curr[j] += sign * delta
    
    return h_curr, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Outer approximation
    if time.time() < deadline - 40:
        oo_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_loop(best_h, n, oo_end, max_iters=50)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    
    # Phase 2: SLSQP with smooth max (multiple alphas)
    for alpha in [5000, 20000, 80000]:
        if time.time() > deadline - 25:
            break
        h_new = slsqp_smooth_max(best_h, n, alpha, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: Lag equalization
    if time.time() < deadline - 20:
        le_end = time.time() + min(30, (deadline - time.time()) * 0.3)
        h_le, c_le = lag_equalization(best_h, n, le_end, top_m=6, n_rounds=40)
        if c_le < best_c:
            best_c = c_le
            best_h = h_le
    
    # Phase 4: Another SLSQP round with high alpha
    if time.time() < deadline - 15:
        for alpha in [200000, 500000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_smooth_max(best_h, n, alpha, top_m=5, maxiter=2000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
    
    # Phase 5: Autocorr-based optimization
    if time.time() < deadline - 15:
        h_new = slsqp_autocorr_min(best_h, n, 50000, maxiter=2000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    
    # Phase 6: Projected subgradient
    if time.time() < deadline - 10:
        pg_end = time.time() + min(15, (deadline - time.time()) * 0.4)
        h_pg, c_pg = projected_subgradient(best_h, n, pg_end, alpha_init=50, n_iters=300)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    
    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 5:
        h_new = slsqp_smooth_max(best_h, n, 1000000, top_m=4, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 8: Fine local search
    if time.time() < deadline - 2:
        best_h, best_c = fine_local_search(best_h, n, deadline - 0.5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def structured_init(n, rng):
    """Create a structured initial guess."""
    x = np.linspace(0, 2, n)
    # Start with a perturbed constant
    h = np.full(n, 0.5)
    # Add some low-frequency structure
    h += 0.05 * np.cos(np.pi * x) * rng.uniform(0.5, 1.5)
    h += 0.03 * np.sin(2 * np.pi * x) * rng.uniform(0.5, 1.5)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase A: Optimize the parent at its native resolution
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase B: Try other resolutions
    n_candidates = []
    if h_init is not None:
        # Include nearby resolutions
        n_candidates = [n_init - 50, n_init, n_init + 50, n_init + 100, n_init + 200]
        n_candidates = [n for n in n_candidates if 100 <= n <= 600]
    else:
        n_candidates = [200, 250, 300, 350, 400]
    
    seen = set()
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if n_try in seen or n_try < 100:
            continue
        seen.add(n_try)
        
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = structured_init(n_try, rng)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase C: Multi-start from best solution
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, max(20, (deadline - time.time() - 25) * 0.35))
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase D: Final polishing on the best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Very high alpha SLSQP
        h_final = slsqp_smooth_max(best_h, n, 2000000, top_m=3, maxiter=5000, ftol=1e-16)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        # Fine local search with remaining time
        if time.time() < deadline - 2:
            best_h, best_c = fine_local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
