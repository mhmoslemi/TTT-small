def compute_all_lags(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, n, lag):
    """Compute gradient of the lag `lag` correlation w.r.t. h.
    lag is in terms of grid steps (can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return dx * grad
    if m >= n:

def compute_all_lags(h, n):

    c = compute_all_lags(h, n)

    t_var = cp.Variable()
    
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        h_var >= h - trust,
        h_var <= h + trust,
    ]

        lag = k - (n - 1)
        c_k = c[k]
        grad = np.zeros(n)
        if lag == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        elif lag > 0:
            s = lag
            grad[s:] += dx * (1.0 - h[:n-s])
            grad[:n-s] -= dx * h[s:]
        else:
            s = -lag
            grad[:n-s] += dx * (1.0 - h[s:])
            grad[s:] -= dx * h[:n-s]

            prob.solve(solver=cp.CBC)
        except Exception:
            return None

    val = h_var.value

        return None
    return val

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_feasible_bisect(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_proj.sum()
    if diff > 1e-14:
        room = 1.0 - h_proj
        tot = room.sum()
        if tot > 1e-14:
            h_proj += (diff / tot) * room
    elif diff < -1e-14:
        tot = h_proj.sum()
        if tot > 1e-14:
            h_proj += (diff / tot) * h_proj
    return np.clip(h_proj, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = shifted - shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            s = lag
            grad[s:] += wk * dx * (1.0 - h[:n-s])
            grad[:n-s] -= wk * dx * h[s:]
        else:
            s = -lag
            grad[:n-s] += wk * dx * (1.0 - h[s:])
            grad[s:] -= wk * dx * h[:n-s]
    return grad, m, w

def projected_gradient_step(h, n, alpha, lr):
    grad, m, _ = smooth_max_and_grad(h, n, alpha)
    h_new = h - lr * grad
    h_new = project_feasible_bisect(h_new, n)
    return h_new

def sca_lp_solve(h, n, active_indices, trust):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        h_var >= h - trust,
        h_var <= h + trust,
    ]
    for k in active_indices:
        lag = k - (n - 1)
        c_k = c[k]
        grad = np.zeros(n)
        if lag == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        elif lag > 0:
            s = lag
            grad[s:] += dx * (1.0 - h[:n-s])
            grad[:n-s] -= dx * h[s:]
        else:
            s = -lag
            grad[:n-s] += dx * (1.0 - h[s:])
            grad[s:] -= dx * h[:n-s]
        cons.append(c_k + grad @ (h_var - h) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                return None
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    val = h_var.value
    if val is None:
        return None
    return val

def sca_optimize(h, n, deadline, M=8, max_iters=200, init_trust=0.15):

        trust = max(0.0005, trust * 0.82)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-18):
    def obj(x):
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        shifted = shifted - shifted.max()
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    def obj_jac(x):
        c = compute_all_lags(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        shifted = shifted - shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += wk * dx * (1.0 - 2.0 * x)
            elif lag > 0:
                s = lag
                grad[s:] += wk * dx * (1.0 - x[:n-s])
                grad[:n-s] -= wk * dx * x[s:]
            else:
                s = -lag
                grad[:n-s] += wk * dx * (1.0 - x[s:])
                grad[s:] -= wk * dx * x[:n-s]
        return grad
    def sum_fun(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

            best_h = h_final
        # Local search
        h_ls, c_ls = local_pair_search(best_h, n, deadline - 2, max_rounds=30)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
