import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    target_sum = n / 2.0
    h = h.copy()

        if abs(diff) < 1e-13:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    sv = x.sum() - n / 2.0
    return float(val + penalty * sv * sv)

def optimize_lbfgsb(h, n, alpha, penalty, maxiter=300, deadline=None):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, penalty), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                                'maxfun': 100000, 'iprint': -1})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Subgradient method for the smooth max."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    step_size = 0.01
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        weights /= weights.sum()
        
        # Compute gradient
        r = n - 1
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-12:
                continue
            shift = k - r
            # Term 1: (1-h[m-shift]) for m in [max(0,shift), min(n, shift+n))
            lo1 = max(0, shift)
            hi1 = min(n, shift + n)
            if lo1 < hi1:
                grad[lo1:hi1] += wk * dx * (1.0 - h[lo1-shift:hi1-shift])
            # Term 2: -h[m+shift] for m in [max(0,-shift), min(n, n-shift))
            lo2 = max(0, -shift)
            hi2 = min(n, n - shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= wk * dx * h[lo2+shift:hi2+shift]
        
        # Step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.95
            if step_size < 1e-12:
                break
        
        # Periodically adjust step size
        if it % 100 == 0:
            step_size = max(step_size, 0.005 * (1 - it / n_iters) + 0.001)
    
    return best_h, best_c

                        improved = True
                    else:

            if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    sv = x.sum() - n / 2.0
    return float(val + penalty * sv * sv)

def smooth_obj_grad(x, n, alpha, penalty):
    """Analytical gradient of smooth_obj."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    weights = exp_shifted / exp_shifted.sum()
    
    # Gradient of the log-sum-exp part
    # d/dx [m + log(sum exp(alpha*(c-m)))/alpha]
    # = (1/alpha) * sum_k weights[k] * alpha * dc_k/dx
    # = sum_k weights[k] * dc_k/dx
    
    # c_k = dx * sum_t x[t] * (1 - x[t - (k-r)]) where r = n-1
    # dc_k/dx_m = dx * [(1-x[m-(k-r)]) * I(0<=m-(k-r)<n) - x[m+(k-r)] * I(0<=m+(k-r)<n)]
    
    r = n - 1
    grad_lse = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        shift = k - r
        
        # First part: (1-x[m-shift]) for valid m (i.e., 0<=m-shift<n => m in [shift, shift+n-1] ∩ [0,n-1])
        lo = max(0, shift)
        hi = min(n, shift + n)
        if lo < hi:
            grad_lse[lo:hi] += wk * dx * (1.0 - x[lo-shift:hi-shift])
        
        # Second part: -x[m+shift] for valid m (i.e., 0<=m+shift<n => m in [-shift, n-1-shift] ∩ [0,n-1])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad_lse[lo2:hi2] -= wk * dx * x[lo2+shift:hi2+shift]
    
    # Add penalty gradient: d/dx [penalty * (sum(x) - n/2)²] = 2*penalty*(sum(x)-n/2)
    sv = x.sum() - n / 2.0
    grad_lse += 2.0 * penalty * sv
    
    return grad_lse

def optimize_lbfgsb(h, n, alpha, penalty, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_obj_grad,
                       args=(n, alpha, penalty), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 50000, 'iprint': -1})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_refine(h, n, deadline, alpha=5000, n_iters=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx_val = 2.0 / n
    # Use adaptive step size
    step = 0.02
    patience = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx_val
        m = c.max()
        shifted = alpha * (c - m)
        mx_s = shifted.max()
        weights = np.exp(shifted - mx_s)
        weights /= weights.sum()
        
        # Gradient
        r = n - 1
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-15:
                continue
            shift = k - r
            lo = max(0, shift)
            hi = min(n, shift + n)
            if lo < hi:
                grad[lo:hi] += wk * dx_val * (1.0 - h[lo-shift:hi-shift])
            lo2 = max(0, -shift)
            hi2 = min(n, n - shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= wk * dx_val * h[lo2+shift:hi2+shift]
        
        # Normalize gradient to get a direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        h_new = h - step * grad / gnorm  # normalized step
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            patience = 0
            # Slightly increase step
            step = min(step * 1.02, 0.05)
        else:
            patience += 1
            step *= 0.9
            if step < 1e-8:
                break
            if patience > 200:
                break
    
    return best_h, best_c

def random_pair_search(h, n, deadline, deltas=None, n_seed=42):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(n_seed)
    
    for d in deltas:
        if time.time() > deadline:
            break
        n_passes = 300
        for _pass in range(n_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            # Number of random tries per pass
            n_tries = min(3000, n * 12)
            for _ in range(n_tries):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                sign = 1.0 if rng.rand() < 0.5 else -1.0
                hi = h[i] + sign * d
                hj = h[j] - sign * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = hi - sign * d, hj + sign * d
            if not improved:
                break
    return h, best_c
