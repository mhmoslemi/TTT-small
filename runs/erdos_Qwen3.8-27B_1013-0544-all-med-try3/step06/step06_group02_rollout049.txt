        if time.time() > deadline:
            break
        if itr % 100 == 0 and itr > 0:
            alpha = alpha_init * (1.0 + itr / 200.0)
        
        g = grad_smooth_min_R(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        improved = False
        step = lr
        for _ in range(30):
            h_new = project_feasible(h + step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
    
    return best_h, best_c

def pair_refine(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for d in deltas:
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for dr in [i - 8, i - 4, i - 2, i - 1, i + 1, i + 2, i + 4, i + 8]:
                    if dr < 0 or dr >= n or dr == i:
                        continue
                    if h[i] + d <= 1.0 and h[dr] - d >= 0.0:
                        h[i] += d
                        h[dr] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            progress = True
                        else:
                            h[i] -= d
                            h[dr] += d
                    if h[i] - d >= 0.0 and h[dr] + d <= 1.0:
                        h[i] -= d
                        h[dr] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            progress = True
                        else:
                            h[i] += d
                            h[dr] -= d
    
    return best_h, best_c

def lp_cutting_plane(h, n, deadline, max_iters=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for itr in range(max_iters):
        if time.time() > deadline:
            break
        R = compute_R(h, n)
        R_min = R.min()
        c5 = 1.0 - R_min
        
        # Find active set (bottom 20% of R values)
        threshold = np.percentile(R, 20)
        active = np.where(R <= threshold)[0]
        
        # Compute gradients for active shifts
        dx = 2.0 / n
        G = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                G[idx, :nv] += dx * h[k:]
                G[idx, k:] += dx * h[:nv]
            else:
                kk = -k
                nv = n - kk
                G[idx, kk:] += dx * h[:nv]
                G[idx, :nv] += dx * h[kk:]
        
        # LP: maximize z s.t. z <= R[m] + G[m] @ (x - h) for active m, 0<=x<=1, sum(x)dx=1
        x = cp.Variable(n)
        z = cp.Variable()
        constraints = [x >= 0, x <= 1, sum(x) * dx == 1.0]
        for idx in range(len(active)):
            constraints.append(z <= R[active[idx]] + G[idx] @ (x - h))
        
        prob = cp.Problem(cp.Maximize(z), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
            if x.value is not None and z.value is not None:
                h_new = project_feasible(np.array(x.value, dtype=float), n)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_c - 1e-12:
                    best_c = c5_new
                    best_h = h_new.copy()
                    h = h_new
                else:
                    h = project_feasible(h_new, n)
                    break
            else:
                break
        except Exception:
            break
    
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Coordinate descent on initial resolution
    t1 = min(t0 + 180, deadline - 400)
    if t1 > t0 + 30:
        h1, c1 = coord_descent(best_h, n_init, t1, max_sweeps=300, rng=rng)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Smooth gradient ascent polish
    t2 = min(time.time() + 100, deadline - 300)
    if t2 > time.time() + 10:
        h2, c2 = projected_grad_ascent(best_h, n_init, t2, alpha_init=5.0, max_iters=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: More coordinate descent
    t3 = min(time.time() + 120, deadline - 200)
    if t3 > time.time() + 10:
        h3, c3 = coord_descent(best_h, n_init, t3, max_sweeps=200, rng=rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: LP cutting plane
    t4 = min(time.time() + 120, deadline - 100)
    if t4 > time.time() + 20:
        h4, c4 = lp_cutting_plane(best_h, n_init, t4, max_iters=100)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Coordinate descent again after LP
    t5 = min(time.time() + 80, deadline - 60)
    if t5 > time.time() + 10:
        h5, c5 = coord_descent(best_h, n_init, t5, max_sweeps=200, rng=rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Try different resolutions
    for n_try in [256, 400, 512, 350]:
        if time.time() > deadline - 30:
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        alloc = min(60, deadline - time.time() - 20)
        if alloc < 30:
            break
        t_sub = time.time() + alloc * 0.5
        h_t, c_t = coord_descent(h_interp, n_try, t_sub, max_sweeps=150, rng=rng)
        if time.time() < deadline - 15:
            h_t, c_t = projected_grad_ascent(h_t, n_try, deadline - 15, alpha_init=20.0, max_iters=1000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    
    # Phase 7: Final coordinate descent polish on best
    if time.time() < deadline - 10:
        nf = len(best_h)
        h_f, c_f = coord_descent(best_h, nf, deadline - 5, max_sweeps=100, rng=rng)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    # Phase 8: Ultra-fine pair search
    if time.time() < deadline - 3:
        nf = len(best_h)
        h_f, c_f = pair_refine(best_h, nf, deadline - 1,
                                deltas=[0.002, 0.001, 0.0005, 0.0002])
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
