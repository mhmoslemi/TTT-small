def flip_delta(h, i0, delta, n):
    """Compute change in correlation when h[i0] changes by delta."""
    c = np.zeros(2*n - 1)
    nn1 = n - 1
    # a-factor: i = i0 - k + nn1, valid when 0 <= i0 - k + nn1 <= nn1
    # i.e., k <= i0 + nn1 and k >= nn1 - i0
    k_lo_a = max(0, nn1 - i0)
    k_hi_a = min(2*nn1, i0 + nn1)
    for k in range(k_lo_a, k_hi_a + 1):

        c[k] += delta * (1 - h[i])

    for k in range(k_lo_v, k_hi_v + 1):
