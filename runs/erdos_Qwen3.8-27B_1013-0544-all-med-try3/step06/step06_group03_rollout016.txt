```python
def compute_weighted_gradient(h, n, alpha):
    """Compute gradient of smooth max cross-correlation objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    w = w / w_sum  # These are the softmax weights
    
    # Weight of the max term: the smooth max gradient includes the max itself
    # d/dx [m + log(sum(exp(sh)))/alpha] = (1/alpha) * sum_k w_k * c_k'
    # where w_k = exp(alpha*(c_k - m)) / sum_j exp(alpha*(c_j - m))
    # and c_k' = dc_k/dx
    
    grad = np.zeros(n)
    
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-10:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            # ∂cross/∂h[j] = (1-h[j+L])*dx for j in [0, n-1-L]
            #                -h[j-L]*dx for j in [L, n-1]
            if n - L > 0:
                grad[:n-L] += w[k_idx] * (1 - h[L:n]) * dx
            if L < n:
                grad[L:] += w[k_idx] * (-h[:n-L]) * dx
        else:
            Lp = -L
            # ∂cross/∂h[j] = (1-h[j-Lp])*dx for j in [Lp, n-1]
            #                -h[j+Lp]*dx for j in [0, n-1-Lp]
            if n - Lp > 0:
                grad[Lp:] += w[k_idx] * (1 - h[:n-Lp]) * dx
            if Lp < n:
                grad[:n-Lp] += w[k_idx] * (-h[Lp:]) * dx
    
    return grad / alpha  # Chain rule with the 1/alpha factor
```

```python
def gradient_guided_pair_search(h, n, deadline, alpha=1000, n_steps=10000):
    """Pair search guided by gradient."""
    best_c = compute_c5(h, n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        grad = smooth_max_gradient(h, n, alpha)
        # Project gradient to zero-sum (to respect constraint)
        grad -= np.mean(grad)
        
        # Find the most negative gradient position (want to increase h there)
        # and the most positive gradient position (want to decrease h there)
        i_inc = np.argmin(grad)  # most negative → want to increase
        i_dec = np.argmax(grad)  # most positive → want to decrease
        
        if h[i_inc] >= 1.0 or h[i_dec] <= 0.0:
            break  # Can't make progress
        
        # Try a move: increase h[i_inc] by d, decrease h[i_dec] by d
        # Find the optimal d by line search
        d_max = min(1.0 - h[i_inc], h[i_dec])
        
        # Try several step sizes
        best_d = 0
        best_c_trial = best_c
        for d_frac in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
            d = d_max * d_frac
            if d < 1e-15:
                continue
            h[i_inc] += d
            h[i_dec] -= d
            c = compute_c5(h, n)
            if c < best_c_trial - 1e-15:
                best_c_trial = c
                best_d = d
            else:
                h[i_inc] -= d
                h[i_dec] += d
        
        if best_d > 0:
            best_c = best_c_trial
        else:
            # Try second-best pair
            # ... (simplified: just break)
            break
    
    return h, best_c
```
