    target = n / 2.0

def lfbfgs_optimize(h_init, n, deadline, alpha, penalty_iter=1e4):
    """L-BFGS-B optimization with log-sum-exp smoothing and penalty for sum constraint"""
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def objective_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        weights = exp_s / exp_s.sum()
        logsum = sm / alpha + np.log(exp_s.sum()) / alpha
        
        # Penalty
        sum_diff = x.sum() - target_sum
        pen = penalty_iter * sum_diff ** 2
        
        val = m + logsum + pen
        
        # Gradient
        # grad of m (subgradient at argmax)
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad_m = np.zeros(n)
        if k >= 0:
            grad_m[:n-k] += (1.0 - x[k:n])
            grad_m[k:] -= x[:n-k]
        else:
            grad_m[-k:] += (1.0 - x[:n+k])
            grad_m[:n+k] -= x[-k:]
        grad_m *= dx
        
        # grad of logsum: sum_k w_k * grad(c_k)
        grad_lse = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-12:
                continue
            w = weights[k_idx]
            k = k_idx - (n - 1)
            if k >= 0:
                grad_lse[:n-k] += w * (1.0 - x[k:n])
                grad_lse[k:] -= w * x[:n-k]
            else:
                grad_lse[-k:] += w * (1.0 - x[:n+k])
                grad_lse[:n+k] -= w * x[-k:]
        grad_lse *= dx
        
        # grad of penalty
        grad_pen = 2 * penalty_iter * sum_diff * np.ones(n)
        
        return val, grad_m + grad_lse + grad_pen
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective_and_grad, h_init, method='L-BFGS-B',
                   bounds=bounds, jac=True,
                   options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14,
                            'maxls': 100, 'maxfun': 100000})
    return project_feasible(res.x, n)
