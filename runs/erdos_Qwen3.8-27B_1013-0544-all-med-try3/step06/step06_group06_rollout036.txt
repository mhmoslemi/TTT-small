import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_run(h, n, alpha, maxiter=500):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        return m + np.log(np.exp(alpha * (g - m)).sum()) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_hill_climb(h, n, deadline, max_sweeps=500, best_improvement=False):

def simulated_annealing(h, n, deadline, t_start=0.005, t_end=0.0001, max_iter=500000, seed=42):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    rng = np.random.RandomState(seed)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Linear cooling
        t = t_start + (t_end - t_start) * (it / max_iter)
        
        # Random swap
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        
        h[i], h[j] = 1.0, 0.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            cur_c = new_c
            ones[jj] = i
            zeros[ii] = j
        else:
            h[i], h[j] = 0.0, 1.0
    
    return best_h, best_c

            else:
                h[a], h[b], h[c], h[d] = 0.0, 0.0, 1.0, 1.0
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_run(h, n, alpha, maxiter=500, ftol=1e-14):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        return m + np.log(np.exp(alpha * (g - m)).sum()) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_hill_climb(h, n, deadline, max_sweeps=500, seed=42):
    h = h.copy()
    if int(h.sum()) != n // 2:
        h = binary_from_continuous(h, n)

    zeros = np.where(h == 0)[0].tolist()

        if time.time() > deadline:
            break
        improved = False
        order_z = zeros[:]
        order_o = ones[:]
        rng.shuffle(order_z)
        rng.shuffle(order_o)

            if time.time() > deadline:
                break

                if j not in ones:

                h[i], h[j] = 1.0, 0.0

                    improved = True

                    zeros.remove(i)
                    zeros.append(j)
                    break
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def simulated_annealing(h, n, deadline, t_start=0.005, t_end=0.00005,
                        max_iter=500000, seed=42, reheat_interval=50000,
                        reheat_factor=3.0):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    rng = np.random.RandomState(seed)
    since_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max_iter)
        if since_improve > reheat_interval:
            t = max(t, t_start / reheat_factor)
            since_improve = 0
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        h[i], h[j] = 1.0, 0.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < -1e-15:
            accept = True
        elif delta < 1e-15:
            accept = True
        else:

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                since_improve = 0
            else:
                since_improve += 1
            cur_c = new_c
            ones[jj] = i
            zeros[ii] = j
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_c
