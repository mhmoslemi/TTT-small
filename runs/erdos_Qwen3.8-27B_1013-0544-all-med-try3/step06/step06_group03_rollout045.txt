`np.correlate(h, 1-h, mode='full')` at index `k` gives:
`sum_j h[j] * (1-h)[j + k - (n-1)]`

Let `g = 1 - h`. Then the output at index k is:
`sum_j h[j] * g[j + k - n + 1]`

For this to be valid, we need `0 <= j < n` and `0 <= j + k - n + 1 < n`.

Let `lag = k - (n-1)`. Then we need `0 <= j < n` and `0 <= j + lag < n`, i.e., `max(0, -lag) <= j < min(n, n-lag)`.

So `f_k = dx * sum_{j=max(0,-lag)}^{min(n, n-lag)-1} h[j] * g[j+lag]`

where `g[j] = 1 - h[j]`.

`df_k/dh_i`:
- From the h[j] term: contributes `dx * g[j+lag]` when `j = i` and `i` is in range
- From the g[j+lag] term: contributes `-dx * h[j]` when `j+lag = i`, i.e., `j = i-lag` and `i-lag` is in range

`= dx * ((1-h[i+lag]) * 1_{i+lag in [0,n)} - h[i-lag] * 1_{i-lag in [0,n)})`

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_full(h, n, alpha):
    corr = compute_all_corr(h, n)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def active_set_obj(x, n, active_idx, alpha):
    corr = compute_all_corr(x, n)
    vals = corr[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def find_active_set(h, n, top_m=15):
    corr = compute_all_corr(h, n)
    idx = np.argsort(corr)[::-1][:top_m]
    return idx, corr

def _sum_con_fun(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, active_idx, alpha, maxiter=3000):
    def obj(x):
        return active_set_obj(x, n, active_idx, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_smooth_full(h, n, alpha, maxiter=2000):
    def obj(x):
        return smooth_max_full(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_optimize(h, n, deadline, top_m=15, max_rounds=50):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        active_idx, _ = find_active_set(h, n, top_m)
        
        for alpha in [1000, 5000, 20000, 100000, 500000]:
            if time.time() > deadline - 3:

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c

def compute_grad_penalty(h, n, T):

    corr = compute_all_corr(h, n)
    violations = np.maximum(0.0, corr - T)
    
    if np.max(violations) < 1e-15:
        return np.zeros(n), 0.0
    
    grad = np.zeros(n)
    active = violations > 1e-13
    active_k = np.where(active)[0]

        lag = k_idx - (n - 1)
        w = 2.0 * dx * violations[k_idx]
        
        i_start = max(0, -lag)
        i_end = min(n, n - lag)

    T_hi = 0.45

    c_best = compute_c5(h_start, n)

            best_h, best_c = pair_search_local(best_h, n_best, deadline - 5,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = pair_search_local(best_h, n_best, deadline - 5,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], full_range=False)
    
    # Phase 7: Final SLSQP + pair polish
    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)
        h_fin = slsqp_smooth_full(best_h, n_best, 2000000, maxiter=5000)
        c_fin = compute_c5(h_fin, n_best)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
        if time.time() < deadline - 2:
            if n_best <= 200:
                best_h, best_c = pair_search_local(best_h, n_best, deadline - 1,
                    [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            else:
                best_h, best_c = pair_search_local(best_h, n_best, deadline - 1,
                    [0.0005, 0.0002, 0.0001, 0.00005], full_range=False)
    
    if best_h is None:
        n_best = n_init
        best_h = project_feasible(np.full(n_best, 0.5), n_best)
        best_c = compute_c5(best_h, n_best)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
