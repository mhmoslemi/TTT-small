    target = n / 2.0

        m = g.max()
        shifted = alpha * (g - m)
        # Numerical stability
        shifted -= shifted.max()
        log_sum = np.log(np.exp(shifted).sum())
        return m + log_sum / alpha
    
    def gradient(x):

        m = g.max()

        return x.sum() - target

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_analytic(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        return m + np.log(np.exp(shifted).sum()) / alpha
    
    def gradient(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        exp_g = np.exp(shifted)
        total = exp_g.sum()
        if total < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(g)] = 1.0
        else:
            p = exp_g / total
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            s = k - (n - 1)
            if s == 0:
                grad += p[k] * dx * (1 - 2 * x)
            elif s > 0:
                gs = np.zeros(n)
                gs[s:] += (1 - x[:n - s])
                gs[:n - s] -= x[s:]
                grad += p[k] * dx * gs
            else:
                t = -s
                gs = np.zeros(n)
                gs[:n - t] += (1 - x[t:])
                gs[t:] -= x[:n - t]
                grad += p[k] * dx * gs
        return grad
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},)
    
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_subgrad(h, n, deadline, m_active=3, max_iter=50000):
    """Subgradient method targeting top-m active lags with analytic gradient"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        
        # Get top m lags
        top_mask = g >= m - 1e-10
        top_idx = np.where(top_mask)[0]
        
        # Compute gradient as average over active lags
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s == 0:
                gs = (1 - 2 * h)
            elif s > 0:
                gs = np.zeros(n)
                gs[s:] += (1 - h[:n - s])
                gs[:n - s] -= h[s:]
            else:
                t = -s
                gs = np.zeros(n)
                gs[:n - t] += (1 - h[t:])
                gs[t:] -= h[:n - t]
            grad += dx * gs
        grad /= len(top_idx)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Backtracking
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.95
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    """Best-improvement 1-swap local search for binary patterns"""
    target = int(n // 2)
    h_bin = (h >= 0.5).astype(float)
    if h_bin.sum() != target:
        idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
    
    dx = 2.0 / n
    best_c = compute_c5(h_bin, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        g = compute_all_g(h_bin, n)
        m = g.max()
        k_star = int(np.argmax(g))
        s_star = k_star - (n - 1)
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        best_delta = 0.0
        best_i = -1
        best_j = -1
        
        for i in zeros:
            a_val = 0.0
            if 0 <= i - s_star < n:
                a_val += (1.0 - h_bin[i - s_star])
            if 0 <= i + s_star < n:
                a_val -= h_bin[i + s_star]
            if a_val == 0:
                continue
            for j in ones:
                b_val = 0.0
                if 0 <= j - s_star < n:
                    b_val -= (1.0 - h_bin[j - s_star])
                if 0 <= j + s_star < n:
                    b_val += h_bin[j + s_star]
                
                delta = a_val + b_val
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        h_bin[best_i] = 1.0
        h_bin[best_j] = 0.0
        new_c = compute_c5(h_bin, n)
        
        if new_c < best_c - 1e-15:
            best_c = new_c
        else:
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            break
    
    return h_bin, best_c

def binary_sa(h, n, deadline, max_iter=100000, t_start=0.01, t_end=0.0001):
    """Simulated annealing on binary patterns"""
    target = int(n // 2)
    h_bin = (h >= 0.5).astype(float)
    if h_bin.sum() != target:
        idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
    
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    curr_c = best_c
    
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    
    rng = np.random.RandomState(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Temperature (geometric cooling)
        t = t_start * (t_end / t_start) ** (it / max_iter)
        
        # Random swap
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        
        h_bin[i], h_bin[j] = 1.0, 0.0
        new_c = compute_c5(h_bin, n)
        
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            # Accept
            curr_c = new_c
            ones[ones.index(j)] = i
            zeros[zeros.index(i)] = j
            if curr_c < best_c:
                best_c = curr_c
                best_h = h_bin.copy()
        else:
            # Revert
            h_bin[i], h_bin[j] = 0.0, 1.0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_lag(h, n, s):
    dx = 2.0 / n
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    elif s > 0:
        gs = np.zeros(n)
        gs[s:] += (1.0 - h[:n - s])
        gs[:n - s] -= h[s:]
        return dx * gs
    else:
        t = -s
        gs = np.zeros(n)
        gs[:n - t] += (1.0 - h[t:])
        gs[t:] -= h[:n - t]
        return dx * gs

def slsqp_analytic(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0

    def objective(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        return m + np.log(np.exp(shifted).sum()) / alpha

    def gradient(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        exp_g = np.exp(shifted)
        total = exp_g.sum()
        if total < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(g)] = 1.0
        else:
            p = exp_g / total
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            s = k - (n - 1)
            grad += p[k] * compute_grad_lag(x, n, s)
        return grad

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},)
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_subgrad(h, n, deadline, m_active=3, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        top_mask = g >= m - 1e-10
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(g)])
        top_idx = np.sort(top_idx)[-m_active:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            grad += compute_grad_lag(h, n, s)
        grad /= len(top_idx)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.95
            if lr < 1e-13:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

        pos_i = zeros_list[i]

def pair_swap_search(h, n, deadline, max_iters=10000, seed=99):

    for it in range(max_iters):
        if time.time() > deadline:

        i1 = int(rng.randint(len(zeros_list)))

        ones_list_tmp = list(ones_set)

        p_i1 = zeros_list[i1]
        p_i2 = zeros_list[i2]
