import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_new = np.clip(h + mid, 0.0, 1.0)
        s_new = h_new.sum()
        if s_new < target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_out.sum()
    if abs(diff) > 1e-12:
        mask = (h_out > 1e-10) & (h_out < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_out[mask] += diff / mask.sum()
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def soft_max_grad(h, n, alpha):
    dx = 2.0 / n
    lags = np.correlate(h, 1 - h, mode='full') * dx
    m = lags.max()
    shifted = alpha * (lags - m)
    max_shift = shifted.max()
    if max_shift > 500:
        weights = np.exp(shifted - max_shift)
    else:
        weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        active = np.argmax(lags)
        weights = np.zeros(len(lags))
        weights[active] = 1.0
        wsum = 1.0
    weights = weights / wsum
    grad = np.zeros(n)
    L = len(lags)
    for k in range(L):
        if weights[k] < 1e-15:
            continue
        w = weights[k]
        shift = k - (n - 1)
        if shift >= 0:
            idx1 = np.arange(shift, n)
            grad[idx1] += w * (1.0 - h[idx1 - shift]) * dx
            idx2 = np.arange(n - shift)
            grad[idx2] -= w * h[idx2 + shift] * dx
        else:
            kp = -shift
            idx1 = np.arange(n - kp)
            grad[idx1] += w * (1.0 - h[idx1 + kp]) * dx
            idx2 = np.arange(kp, n)
            grad[idx2] -= w * h[idx2 - kp] * dx
    return grad, m

def subgradient_descent_soft(h, n, deadline, n_max_iters=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.85
    alpha = 0.1
    patience = 0
    patience_max = 5000
    
    for t in range(n_max_iters):
        if time.time() > deadline:
            break
        
        grad, m = soft_max_grad(h, n, 1000.0)
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        alpha = 0.5 / (1 + 0.05 * t)
        v = beta * v + alpha * grad
        h_new = project_feasible(h - v, n)
        
        c = compute_c5(h_new, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            patience = 0
            if patience < 2000:
                beta = min(0.95, beta + 0.001)
        else:
            h = h_new
            patience += 1
            if patience > patience_max:
                v = 0.0
                patience = 0
                alpha = max(0.01, alpha * 0.5)
                beta = 0.5
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            weights = np.exp(sh - mx)
        else:
            weights = np.exp(sh)
        wsum = weights.sum()
        weights = weights / wsum
        grad = np.zeros(n)
        L = len(c)
        for k in range(L):
            if weights[k] < 1e-12:
                continue
            w = alpha * weights[k]
            shift = k - (n - 1)
            if shift >= 0:
                idx1 = np.arange(shift, n)
                grad[idx1] += w * (1.0 - x[idx1 - shift]) * dx
                idx2 = np.arange(n - shift)
                grad[idx2] -= w * x[idx2 + shift] * dx
            else:
                kp = -shift
                idx1 = np.arange(n - kp)
                grad[idx1] += w * (1.0 - x[idx1 + kp]) * dx
                idx2 = np.arange(kp, n)
                grad[idx2] -= w * x[idx2 - kp] * dx
        return grad
    
    def sum_con(x):
        return x.sum() - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

    target = n // 2
    order = np.argsort(-h)

    h_bin[order[:target]] = 1.0

def fragment_search(h, n, deadline, max_sweeps=100):
    """Try moving small fragments (consecutive blocks) to different positions"""
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng_local = np.random.RandomState(sweep * 31337 + n)
        for t in range(2000):
            if time.time() > deadline:
                return h, best_c
            frag_len = rng_local.randint(2, min(8, n // 10))
            if frag_len >= n:
                continue
            start = rng_local.randint(0, n - frag_len)
            end = start + frag_len
            # Find a destination
            dest = rng_local.randint(0, n - frag_len)
            if abs(dest - start) < frag_len:
                continue
            if dest >= start:
                new_arr = np.concatenate([h[:dest], h[start:start+frag_len], h[dest:start], h[start+frag_len:dest+frag_len], h[dest+frag_len:]])
            else:
                new_arr = np.concatenate([h[:dest+frag_len], h[start:start+frag_len], h[dest+frag_len:start], h[start+frag_len:dest], h[dest:]])
            if len(new_arr) != n:
                continue
            if new_arr.sum() != h.sum():
                continue
            c = compute_c5(new_arr, n)
            if c < best_c - 1e-15:
                best_c = c
                h = new_arr
                improved = True
        if not improved:
            break
    return h, best_c

            if time.time() < deadline - 20:
                h_r = slsqp_optimize(h_r, n_best, 30000, maxiter=1000)
                c_r = compute_c5(h_r, n_best)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
    
    # Phase 5: Final extended subgradient + SLSQP
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        t_final = deadline - 10
        h_f, c_f = subgradient_descent_soft(best_h, n_best, min(time.time() + 30, t_final - 10), n_max_iters=100000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
        if time.time() < t_final:
            h_f = slsqp_optimize(best_h, n_best, 200000, maxiter=3000)
            c_f = compute_c5(h_f, n_best)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f.copy()
    
    # Phase 6: Fine local adjustment
    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)
        # Coordinate descent with small steps
        for d in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > deadline - 3:
                break
            improved = True
            while improved and time.time() < deadline - 3:
                improved = False
                for i in range(0, n_best, max(1, n_best // 500)):
                    if time.time() > deadline - 3:
                        break
                    for dd in [d, -d]:
                        if 0 <= best_h[i] + dd <= 1:
                            old = best_h[i]
                            best_h[i] = old + dd
                            # Adjust one other element to maintain sum
                            # Find element with most room
                            if dd > 0:
                                room = best_h < 1.0 - 1e-10
                                candidates = np.where(room)[0]
                                candidates = candidates[candidates != i]
                                if len(candidates) > 0:
                                    j = candidates[np.argmax(best_h[candidates])]
                                    best_h[j] -= dd
                                    c = compute_c5(best_h, n_best)
                                    if c < best_c - 1e-15:
                                        best_c = c
                                        improved = True
                                    else:
                                        best_h[i] = old
                                        best_h[j] += dd
                            else:
                                room = best_h > 1e-10
                                candidates = np.where(room)[0]
                                candidates = candidates[candidates != i]
                                if len(candidates) > 0:
                                    j = candidates[np.argmin(best_h[candidates])]
                                    best_h[j] -= dd
                                    c = compute_c5(best_h, n_best)
                                    if c < best_c - 1e-15:
                                        best_c = c
                                        improved = True
                                    else:
                                        best_h[i] = old
                                        best_h[j] += dd
    
    if best_h is None:
        n_best = 250
        best_h = project_feasible(np.full(n_best, 0.5), n_best)
        best_c = compute_c5(best_h, n_best)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
