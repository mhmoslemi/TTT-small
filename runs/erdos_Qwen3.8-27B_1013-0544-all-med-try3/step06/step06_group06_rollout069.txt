from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_g_k(h, n, k_idx):
    """Compute gradient of g[k_idx] w.r.t. h."""
    s = k_idx - (n - 1)  # lag
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # g = sum_{i=0}^{n-1-s} h[i+s](1-h[i])
        for i in range(n - s):
            grad[i + s] += (1.0 - h[i])
            grad[i] += (-h[i + s])
    else:
        s2 = -s
        for i in range(n - s2):
            grad[i] += (1.0 - h[i + s2])
            grad[i + s2] += (-h[i])
    return grad * dx

def grad_g_k_fast(h, n, k_idx):
    """Fast vectorized gradient of g[k_idx]."""
    s = k_idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # h[i+s] for i in [0, n-1-s] → indices [s, n-1]
        # h[i] for i in [0, n-1-s] → indices [0, n-1-s]
        grad[s:n] += (1.0 - h[0:n-s])
        grad[0:n-s] += (-h[s:n])
    else:
        s2 = -s
        grad[0:n-s2] += (1.0 - h[s2:n])
        grad[s2:n] += (-h[0:n-s2])
    return grad * dx

    """Solve QP: minimize g[k_idx] subject to sum(h)=n/2, 0<=h<=1."""
    def obj(x):
        return np.correlate(x, 1-x, mode='full')[k_idx] * (2.0/n)

        return grad_g_k_fast(x, n, k_idx)

                      constraints=cons, jac=grad,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
