```python
def project_polytope(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    lo, hi = h.min() - target_sum, h.max()  # τ range
    # Actually: sum(clip(h - tau, 0, 1)) is decreasing in tau
    # At tau = h.min() - 1: all h[i] - tau >= 1, so sum = n
    # At tau = h.max(): all h[i] - tau <= 0, so sum = 0
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)
```
