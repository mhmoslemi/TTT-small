def compute_all_c(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Add small epsilon to avoid log(0)
        shifted = (g - m + 1e-15) / alpha
        return float(m + np.log(np.exp(shifted).sum()) / alpha)
    
    def obj_grad(x):
        # Numerical gradient
        g = np.zeros(n)
        eps = 1e-7
        base = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - base) / eps
            x[i] -= eps
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    
    try:

            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = (g - m + 1e-15) / alpha
    return float(m + np.log(np.exp(shifted).sum()) / alpha)

def slsqp_step(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        return soft_max_obj(x, n, alpha)
    def constraint(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def soft_max_optimize(h, n, deadline, alphas, maxiter_each):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_new = slsqp_step(best_h, n, alpha, maxiter_each)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
    return best_h, best_c

def simulate_annealing(h_cont, n, deadline, max_iters=100000, T0=0.05, cool=0.9997):
    target = int(n / 2)
    idx = np.argsort(h_cont)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T0
    rng = np.random.RandomState(12345)
    
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if T < 1e-12:
            break
        r = rng.random()
        accepted = True
        if r < 0.45:
            # Pair swap
            if len(ones_set) == 0 or len(zeros_set) == 0:
                T *= cool
                continue
            i = list(ones_set)[rng.randint(len(ones_set))]
            j = list(zeros_set)[rng.randint(len(zeros_set))]
            h[i], h[j] = 0.0, 1.0
            ones_set.remove(i)
            ones_set.add(j)
            zeros_set.add(i)
            zeros_set.remove(j)
            c = compute_c5(h, n)
            delta = c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
                ones_set.add(i)
                ones_set.remove(j)
                zeros_set.remove(i)
                zeros_set.add(j)
                accepted = False
        elif r < 0.75:
            # Block move: flip a contiguous block of equal 0s and 1s
            start = rng.randint(n)
            bsz = rng.randint(2, max(3, n // 30))
            block = [(start + k) % n for k in range(bsz)]
            n1 = sum(1 for b in block if h[b] == 1)
            n0 = bsz - n1
            if n1 == n0:
                for b in block:
                    h[b] = 1.0 - h[b]
                for b in block:
                    if h[b] == 1:
                        ones_set.add(b)
                        zeros_set.discard(b)
                    else:
                        zeros_set.add(b)
                        ones_set.discard(b)
                c = compute_c5(h, n)
                delta = c - cur_c
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c = c
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                else:
                    for b in block:
                        h[b] = 1.0 - h[b]
                    for b in block:
                        if h[b] == 1:
                            ones_set.add(b)
                            zeros_set.discard(b)
                        else:
                            zeros_set.add(b)
                            ones_set.discard(b)
                    accepted = False
        else:
            # Triple swap (two 1s swap with two 0s plus one more)
            if len(ones_set) < 2 or len(zeros_set) < 2:
                T *= cool
                continue
            olist = list(ones_set)
            zlist = list(zeros_set)
            i1 = olist[rng.randint(len(olist))]
            i2 = olist[rng.randint(len(olist))]
            if i1 == i2:
                T *= cool
                continue
            j1 = zlist[rng.randint(len(zlist))]
            j2 = zlist[rng.randint(len(zlist))]
            if j1 == j2:
                T *= cool
                continue
            h[i1], h[i2] = 0.0, 0.0
            h[j1], h[j2] = 1.0, 1.0
            ones_set.discard(i1)
            ones_set.discard(i2)
            ones_set.add(j1)
            ones_set.add(j2)
            zeros_set.add(i1)
            zeros_set.add(i2)
            zeros_set.discard(j1)
            zeros_set.discard(j2)
            c = compute_c5(h, n)
            delta = c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i1], h[i2] = 1.0, 1.0
                h[j1], h[j2] = 0.0, 0.0
                ones_set.add(i1)
                ones_set.add(i2)
                ones_set.discard(j1)
                ones_set.discard(j2)
                zeros_set.discard(i1)
                zeros_set.discard(i2)
                zeros_set.add(j1)
                zeros_set.add(j2)
                accepted = False
        
        T *= cool
    
    return best_h, best_c

def greedy_binary_search(h, n, deadline, max_sweeps=500):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h_bin == 1)[0])
        zeros = list(np.where(h_bin == 0)[0])
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def boundary_relax(h_bin, n, deadline):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] < 0.05 or h[i] > 0.95:
                continue
            for d in [0.1, -0.1, 0.2, -0.2, 0.3, -0.3, 0.5, -0.5]:
                nv = h[i] + d
                if 0 <= nv <= 1:
                    h[i] = nv
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        break
                    else:
                        h = h_bin.copy()
        if not improved:
            break
    return h, best_c

def subgradient_descent(h, n, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_c(h, n)
        m = c.max()
        k_max = int(np.argmax(c))
        
        # Gradient of c[k_max] w.r.t. h
        # c[k_max] = sum_i h[i] * (1 - h[i+k_max-(n-1)]) * dx (approximately)
        # Numerical gradient
        g = np.zeros(n)
        eps = 1e-6
        base_val = np.correlate(h, 1 - h, mode='full')[k_max] * dx
        for i in range(n):
            h[i] += eps
            g[i] = (np.correlate(h, 1 - h, mode='full')[k_max] * dx - base_val) / eps
            h[i] -= eps
        
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / gn
            ht = project_feasible(h - step * g, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: SLSQP with aggressive alpha ramp (30% time)
    p1 = t0 + 0.30 * (budget_s - 5)
    h, c = soft_max_optimize(best_h.copy(), n, p1,
                             [200, 1000, 5000, 20000, 100000, 500000], 400)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 2: Subgradient on exact max (10% time)
    p2 = t0 + 0.40 * (budget_s - 5)
    h, c = subgradient_descent(best_h.copy(), n, p2, max_iter=2000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 3: Greedy binary + SA at current n (30% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    # First greedy to get a good binary start
    if time.time() < p3 - 30:
        hb, cb = greedy_binary_search(best_h, n, t0 + 0.50 * (budget_s - 5))
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Then SA to improve
    if time.time() < p3 - 5:
        hb, cb = simulate_annealing(best_h, n, p3 - 5,
                                     max_iters=80000, T0=0.03, cool=0.9998)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Boundary relaxation from best binary
    if time.time() < p3 and np.all((best_h == 0) | (best_h == 1)):
        h_rel, c_rel = boundary_relax(best_h, n, p3)
        if c_rel < best_c:
            best_c = c_rel
            best_h = h_rel.copy()
    
    # Phase 4: Multi-resolution SA (25% time)
    p4 = t0 + 0.95 * (budget_s - 5)
    for nh in [400, 500]:
        if time.time() > p4 - 60:
            break
        if nh <= n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        h_interp = project_feasible(np.interp(xn, xo, best_h), nh)
        
        # Quick SLSQP to get a good continuous start
        if time.time() < p4 - 30:
            h_interp, _ = soft_max_optimize(h_interp, nh, p4 - 20,
                                            [10000, 100000], 200)
        
        # Binary + SA
        if time.time() < p4 - 10:
            hb, cb = simulate_annealing(h_interp, nh, p4 - 5,
                                         max_iters=40000, T0=0.03, cool=0.9998)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                n = nh
    
    # Phase 5: Final SLSQP polish (5% time)
    if time.time() < dl - 5 and not np.all((best_h == 0) | (best_h == 1)):
        h, c = soft_max_optimize(best_h.copy(), n, dl - 2,
                                 [5000000, 20000000], 300)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
