    target_sum = n / 2.0

    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_logsumexp(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

                break
        
        v = momentum * v + (1 - momentum) * g
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        v = v / g_norm
        h = h - lr_init * v
        h = project_feasible(h, n)
        n_iter += 1
    
    return best_h, best_c

def mixed_local_search(h, n, deadline, delta_list=None):
    if delta_list is None:
        delta_list = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in delta_list:
        for _pass in range(300):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_local_search(h, n, deadline, radius=30, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.001, 0.0005, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(200):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def random_pair_search(h, n, deadline, n_trials=100000, delta_range=(0.001, 0.05)):
    rng = np.random.RandomState(12345)
    h = h.copy()
    best_c = compute_c5(h, n)
    t_end = deadline
    count = 0
    while time.time() < t_end and count < n_trials:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(delta_range[0], delta_range[1]) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
            else:
                h[i], h[j] = hi - d, hj + d
        count += 1
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def binary_round(h, n):
    target = int(n / 2)
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def local_binary_search(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(77)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if len(ones) * len(zeros) <= 50000:
            pairs = [(i, j) for i in ones for j in zeros]
        else:
            pairs = [(ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]) for _ in range(50000)]
        for i, j in pairs:
            if time.time() > deadline:
                return h, best_c
            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c

def targeted_lag_reduction(h, n, deadline, top_k=5, step_size=0.02):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _iter in range(200):
        if time.time() > deadline:
            return h, best_c
        c_all = compute_c5_vec(h, n)
        worst_idx = np.argsort(-c_all)[:top_k]
        improved = False
        for lag_idx in worst_idx:
            if time.time() > deadline:
                return h, best_c
            ell = int(lag_idx) - (n - 1)
            if ell == 0:
                continue
            n_pairs = max(1, abs(ell) if abs(ell) < n else 1)
            valid_range = (n - abs(ell))
            if valid_range <= 0:
                continue
            step_i = max(1, valid_range // 30)
            for i in range(max(0, ell), n - max(0, -ell), step_i):
                j = i - ell
                if 0 <= j < n and i != j:
                    if h[i] > h[j]:
                        d = min(step_size, (h[i] - h[j]) * 0.5)
                    else:
                        d = min(step_size, (h[j] - h[i]) * 0.5)
                    if d < 1e-8:
                        continue
                    orig_i, orig_j = h[i], h[j]
                    h[i] = orig_i - d if orig_i > orig_j else orig_i + d
                    h[j] = orig_j + d if orig_i > orig_j else orig_j - d
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[j] = orig_i, orig_j
                        h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    def try_update(h_new, c_new, n_new):
        nonlocal best_h, best_c, best_n
        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            best_n = n_new

    # Phase 1: Subgradient optimization at original resolution
    t_phase1_end = t0 + (budget_s - 5) * 0.25
    if time.time() < t_phase1_end:
        h_sg, c_sg = subgradient_optimize(best_h, best_n, t_phase1_end,
                                          lr_init=0.05, momentum=0.9, top_k=20, threshold_frac=0.95)
        try_update(h_sg, c_sg, best_n)

        # SLSQP polish
        if time.time() < deadline - 100:
            for alpha in [10000, 50000, 200000]:
                if time.time() > deadline - 100:
                    break
                h_new = slsqp_optimize(best_h, best_n, alpha, maxiter=5000)
                c_new = compute_c5(h_new, best_n)
                try_update(h_new, c_new, best_n)

        # Local search
        if time.time() < deadline - 80 and best_n <= 300:
            h_ls, c_ls = mixed_local_search(best_h, best_n, deadline - 80,
                                            [0.01, 0.005, 0.001, 0.0005, 0.0001])
            try_update(h_ls, c_ls, best_n)

    # Phase 2: Higher resolution optimization
    for n_try in [400, 600, 1000]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(remaining * 0.4, 250)
        t_end_phase = time.time() + alloc

        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)

        c_interp = compute_c5(h_interp, n_try)

        # Subgradient at new resolution
        t_sub_end = time.time() + min(alloc * 0.5, 100)
        h_sg, c_sg = subgradient_optimize(h_interp, n_try, t_sub_end,
                                          lr_init=0.03, momentum=0.9, top_k=30, threshold_frac=0.93)
        if c_sg < c_interp:
            h_interp = h_sg
            c_interp = c_sg

        # SLSQP
        if time.time() < t_end_phase - 20:
            for alpha in [20000, 100000, 500000]:
                if time.time() > t_end_phase - 20:
                    break
                h_new = slsqp_optimize(h_interp, n_try, alpha, maxiter=4000)
                c_new = compute_c5(h_new, n_try)
                if c_new < c_interp:
                    h_interp = h_new
                    c_interp = c_new

        # L-BFGS-B
        if time.time() < t_end_phase - 10:
            h_new = lbfgsb_optimize(h_interp, n_try, alpha=200000, maxiter=10000, penalty=100.0)
            c_new = compute_c5(h_new, n_try)
            if c_new < c_interp:
                h_interp = h_new
                c_interp = c_new

        # Local search (near-neighbor for larger n)
        if time.time() < deadline - 50:
            t_ls_end = min(deadline - 50, time.time() + 60)
            if n_try <= 400:
                h_ls, c_ls = mixed_local_search(h_interp, n_try, t_ls_end,
                                                [0.005, 0.001, 0.0005, 0.0001])
            else:
                h_ls, c_ls = near_local_search(h_interp, n_try, t_ls_end, radius=40,
                                               deltas=[0.005, 0.001, 0.0005])
            if c_ls < c_interp:
                h_interp = h_ls
                c_interp = c_ls

        try_update(h_interp, c_interp, n_try)

    # Phase 3: Binary solution exploration
    if time.time() < deadline - 60:
        n_bin = min(300, best_n)
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_bin = np.linspace(0, 2, n_bin, endpoint=False)
        h_bin_cont = np.interp(x_bin, x_old, best_h)
        h_bin = binary_round(h_bin_cont, n_bin)
        c_bin = compute_c5(h_bin, n_bin)

        if c_bin < best_c:
            h_bin, c_bin = local_binary_search(h_bin, n_bin, deadline - 30, max_rounds=100)
            try_update(h_bin, c_bin, n_bin)

        # Also try a different binary starting point (Thue-Morse inspired)
        if time.time() < deadline - 30:
            tm = np.zeros(n_bin)
            # Thue-Morse: t(n) = popcount(n) mod 2
            for i in range(n_bin):
                tm[i] = bin(i).count('1') % 2
            # Scale to have exactly n_bin/2 ones (adjust slightly)
            target_ones = n_bin // 2
            n_ones = int(tm.sum())
            if n_ones < target_ones:
                # Flip some 0s to 1s
                zeros = np.where(tm == 0)[0]
                rng.shuffle(zeros)
                tm[zeros[:target_ones - n_ones]] = 1.0
            elif n_ones > target_ones:
                ones = np.where(tm == 1)[0]
                rng.shuffle(ones)
                tm[ones[:n_ones - target_ones]] = 0.0
            c_tm = compute_c5(tm, n_bin)
            if c_tm < best_c:
                h_tm, c_tm2 = local_binary_search(tm.copy(), n_bin, deadline - 10, max_rounds=50)
                if c_tm2 < best_c:
                    try_update(h_tm, c_tm2, n_bin)

    # Phase 4: Final refinement on best solution
    if time.time() < deadline - 10:
        n = len(best_h)
        remaining = deadline - time.time() - 5

        # Subgradient final push
        if remaining > 20:
            t_sg_end = time.time() + min(remaining * 0.4, 80)
            h_sg, c_sg = subgradient_optimize(best_h, n, t_sg_end,
                                              lr_init=0.01, momentum=0.95, top_k=10, threshold_frac=0.99)
            try_update(h_sg, c_sg, n)

        # SLSQP final
        if time.time() < deadline - 5:
            h_new = slsqp_optimize(best_h, len(best_h), 10000000, maxiter=8000)
            c_new = compute_c5(h_new, len(best_h))
            try_update(h_new, c_new, len(best_h))

        # Very fine local search
        if time.time() < deadline - 2:
            n = len(best_h)
            if n <= 250:
                h_ls, c_ls = mixed_local_search(best_h, n, deadline - 1,
                                                [0.0001, 0.00005, 0.00002])
            else:
                h_ls, c_ls = near_local_search(best_h, n, deadline - 1, radius=50,
                                               deltas=[0.0005, 0.0002, 0.0001])
            try_update(h_ls, c_ls, n)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
