    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n_start = n_init
        alloc = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_resolution(h_init, n_start, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    n_candidates = [400, 500, 200]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.4)
        if alloc < 50:
            break
        if best_h is not None and len(best_h) != n_try:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 40) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_new = slsqp_full(best_h, n, 1000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = coordinate_descent(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
