def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_for_active(h, c_full, n, tau):
    dx = 2.0 / n
    n_full = 2 * n - 1
    shifted = c_full - c_full.max()
    weights = np.exp(shifted / tau)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    
    grad = np.zeros(n)
    for m in range(n_full):
        w = weights[m]
        if w < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            if kk < n:
                grad[kk:] += w * (1.0 - h[:n - kk])
                grad[:n - kk] -= w * h[kk:]
    return grad * dx

def slsqp_polish(h, n, alpha, maxiter=1500):
    def obj_smooth(x):
        dx_v = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx_v
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def sum_con(x):
        return float(x.sum() - n / 2.0)
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h.copy()
