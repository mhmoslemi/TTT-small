import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def coordinate_descent(h, target_sum, time_limit, t0, penalty=1e4, grid_size=30):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=np.float64).copy()
    h = project_feasible(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_grid = np.linspace(0.0, 1.0, grid_size)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while True:
        if time.time() - t0 > time_limit:
            break

        order = np.random.permutation(n)
        max_change = 0.0

        for j in order:
            old_val = h[j]

            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            delta = corr1 - corr0

            S_other = h.sum() - 1.0

            M = corr0[np.newaxis, :] + t_grid[:, np.newaxis] * delta[np.newaxis, :]
            f = M.max(axis=1) * dx + penalty * (S_other + t_grid - target_sum) ** 2
            best_idx = int(f.argmin())
            best_t = t_grid[best_idx]

            lo = t_grid[best_idx - 1] if best_idx > 0 else 0.0
            hi = t_grid[best_idx + 1] if best_idx < grid_size - 1 else 1.0

            if hi - lo > 1e-10:
                a1 = lo + (1.0 - gr) * (hi - lo)
                b1 = lo + gr * (hi - lo)
                for _ in range(40):
                    fa = (corr0 + a1 * delta).max() * dx + penalty * (S_other + a1 - target_sum) ** 2
                    fb = (corr0 + b1 * delta).max() * dx + penalty * (S_other + b1 - target_sum) ** 2
                    if fa < fb:
                        hi = b1
                        b1 = a1
                        a1 = lo + (1.0 - gr) * (hi - lo)
                    else:
                        lo = a1
                        a1 = b1
                        b1 = lo + gr * (hi - lo)
                best_t = (lo + hi) / 2.0

            h[j] = best_t
            max_change = max(max_change, abs(best_t - old_val))

        sweep += 1
        h = project_feasible(h, target_sum)
        if sweep % 5 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if max_change < 1e-13:
            break

    h = project_feasible(h, target_sum)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum):
    return h.sum() - target_sum


def con_jac(h, target_sum):
    return np.ones_like(h)


def slsqp_refine(h, target_sum, beta, maxiter=3000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h.copy(), compute_c5(h)
