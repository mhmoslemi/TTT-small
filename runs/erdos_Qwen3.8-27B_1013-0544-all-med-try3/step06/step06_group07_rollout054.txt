    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_step(h, k_worst, n):
    """Compute subgradient of the max objective at the point where k_worst is maximizing."""
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # corr[lag] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # d/dh[j] of corr[lag]:
        # If j appears as h[i+lag], then i = j-lag, and the derivative is (1-h[j-lag])
        # If j appears as h[i], then i = j, and the derivative is -h[j+lag]
        # So: d(corr)/dh[j] = (1-h[j-lag]) if j >= lag, and -h[j+lag] if j <= n-1-lag
        for j in range(n):
            if j >= lag:
                grad[j] += (1.0 - h[j - lag])
            if j <= n - 1 - lag:
                grad[j] -= h[j + lag]
    else:
        la = -lag
        # corr[-la] = sum_{i=0}^{n-1-la} h[i] * (1-h[i+la])
        # d/dh[j]:
        # If j appears as h[i], then i=j, derivative is (1-h[j+la]) if j <= n-1-la
        # If j appears as h[i+la], then i+la=j, i=j-la, derivative is -h[j-la] if j >= la
        for j in range(n):
            if j <= n - 1 - la:
                grad[j] += (1.0 - h[j + la])
            if j >= la:
                grad[j] -= h[j - la]
    return grad

    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:

        j_idx = np.arange(lag, n)
        grad[j_idx] += (1.0 - h[j_idx - lag])
        # For j <= n-1-lag: grad[j] -= h[j+lag]
        j_idx2 = np.arange(0, n - lag)
        grad[j_idx2] -= h[j_idx2 + lag]
    else:
        la = -lag
        # For j <= n-1-la: grad[j] += (1 - h[j+la])
        j_idx = np.arange(0, n - la)
        grad[j_idx] += (1.0 - h[j_idx + la])
        # For j >= la: grad[j] -= h[j-la]
        j_idx2 = np.arange(la, n)
        grad[j_idx2] -= h[j_idx2 - la]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    """Vectorized subgradient of corr_k with respect to h."""
    lag = k - (n - 1)
    grad = np.zeros(n)

        #          - h[j+lag] for j in [0, n-1-lag] (from (1-h[i]) term)

def smooth_max(x, beta):
    m = np.max(x)
