A better approach might be to use a "feasible direction" method:
1. Compute the gradient g
2. Find a feasible direction d (e.g., d = -g projected to the tangent space of the constraint)
3. Take a step in direction d

The tangent space of the constraints {0 ≤ h_i ≤ 1, sum(h) = n/2} at a point h is:
- If 0 < h_i < 1: d_i is free
- If h_i = 0: d_i ≥ 0
- If h_i = 1: d_i ≤ 0
- sum(d) = 0

A simple feasible direction: d = -g + mean(-g) (to ensure sum = 0), then clip to respect bounds.

Or even simpler: use the "pair move" interpretation. The gradient tells us which h_i should increase and which should decrease. We can make pair moves in the direction suggested by the gradient.

def smooth_obj_grad(h, n, alpha):

    m = c.max()

    # S1[j] = sum_{l=0}^{n-1} (1-h[l]) * w[j+n-1-l]
    # S2[j] = sum_{k=0}^{n-1} h[k] * w[k-j+n-1]
    
    # S1 = conv(1-h, w, 'full')[n-1 : 2n-1]
    one_minus_h = 1.0 - h
    conv1 = np.convolve(one_minus_h, w, mode='full')
    S1 = conv1[n-1 : 2*n-1]
    
    # For S2: S2[j] = sum_{k=0}^{n-1} h[k] * w[k-j+n-1]
    # Let's verify with convolution:
    # conv(h, w, 'full')[m] = sum_k h[k] * w[m-k]
    # We want sum_k h[k] * w[k-j+n-1] = sum_k h[k] * w[(n-1-j) + (k)]... 
    # Hmm, let me try: let u = w[::-1] (reversed w, length 2n-1)

    # We want w[k-j+n-1], so 2n-2-m+k = k-j+n-1 → 2n-2-m = -j+n-1 → m = n-1+j
    # So S2[j] = conv(h, w[::-1], 'full')[n-1+j]

    S2 = conv2[n-1 : 2*n-1]
    
    grad = (dx / alpha) * (S1 - S2)
    return obj, grad

                        improved = True

                        h[i] += d
                        h[j] -= d
                if h[j] - d >= -1e-12 and h[i] + d <= 1.0 + 1e-12:
                    h[j] -= d
                    h[i] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[j] += d
                        h[i] -= d
            h = best_h.copy()
            if not improved:
                break
    return best_h, best_c

def full_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if radius is None:
                    j_range = range(i + 1, n)
                else:
                    j_range = range(max(0, i - radius), min(n, i + radius + 1))
                for j in j_range:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 + 1e-12 and h[i] - d >= -1e-12:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] -= d
                            h[i] += d
            h = best_h.copy()
            if not improved:
                break
    return best_h, best_c

def optimize_from(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    alphas = [200, 500, 2000, 8000, 30000, 100000, 300000, 1000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_optimize(h, n, a, maxiter=800)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    if time.time() < deadline - 20:
        pg_deadline = min(time.time() + min(60, (deadline - time.time()) * 0.3), deadline - 15)
        h_pg, c_pg = projected_gradient_descent(h, n, pg_deadline,
                                                 alpha_start=5000, alpha_mult=1.5)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg.copy()
            h = h_pg

    if time.time() < deadline - 5:
        gps_deadline = min(time.time() + min(40, (deadline - time.time()) * 0.4), deadline - 3)
        h_gps, c_gps = guided_pair_search(h, n, gps_deadline, alpha=200000, top_k=100)
        if c_gps < best_c:
            best_c = c_gps
            best_h = h_gps.copy()
            h = h_gps

    if time.time() < deadline - 3:
        fps_deadline = deadline - 2
        if n <= 300:
            h_fps, c_fps = full_pair_search(h, n, fps_deadline, [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            h_fps, c_fps = full_pair_search(h, n, fps_deadline, [0.005, 0.002, 0.001, 0.0005], radius=60)
        if c_fps < best_c:
            best_c = c_fps
            best_h = h_fps.copy()

    h = project_feasible(best_h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize the initial construction
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_from(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try different resolutions
    x_init = np.linspace(0, 2, n_init)
    for n_try in [150, 250, 350]:
        if time.time() > deadline - 200:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, (deadline - time.time()) * 0.2)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_from(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with diverse perturbations
    if best_h is not None and time.time() < deadline - 120:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            if r % 3 == 0:
                scale = 0.03 + 0.05 * r / max(1, n_restarts)
                hp = best_h + rng.normal(0, scale, n)
            elif r % 3 == 1:
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                hp = best_h + pert
            else:
                mask = rng.random(n) < 0.1
                scale = np.where(mask, rng.choice([-1, 1]) * 0.08, 0)
                hp = best_h + scale
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 40) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_from(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended guided + full pair search on best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        gps_deadline = min(time.time() + 60, deadline - 30)
        h_gps, c_gps = guided_pair_search(best_h, n, gps_deadline, alpha=500000, top_k=120)
        if c_gps < best_c:
            best_c = c_gps
            best_h = h_gps

    # Phase 5: Final full pair search (very small steps)
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = full_pair_search(best_h, n, deadline - 10,
                                              [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = full_pair_search(best_h, n, deadline - 10,
                                              [0.002, 0.001, 0.0005, 0.0002], radius=50)

    # Phase 6: Final SLSQP ultra-high alpha
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_final = slsqp_optimize(best_h, n, 2000000, maxiter=3000)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        if time.time() < deadline - 5:
            best_h, best_c = full_pair_search(best_h, n, deadline - 3,
                                              [0.0005, 0.0002, 0.0001])

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
