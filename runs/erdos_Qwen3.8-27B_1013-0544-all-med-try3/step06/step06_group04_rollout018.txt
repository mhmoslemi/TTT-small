def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    # Use iterative water-filling / bisection
    # We want: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_out = np.clip(h - lam, 0.0, 1.0)
    # Fix residual
    diff = target - h_out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h_out
            tot = room.sum()
            if tot > 1e-14:
                h_out += (diff / tot) * room
        else:
            tot = h_out.sum()
            if tot > 1e-14:
                h_out += (diff / tot) * h_out
    return np.clip(h_out, 0.0, 1.0)

    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

def smooth_max_and_grad(h, n, alpha, top_k=30):
    """Compute smooth max and its gradient using top-k lags."""
    c = compute_all_lags(h, n)
    # Use top-k lags for efficiency
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[-k:]
    active_c = c[top_idx]

            lr = 0.05  # Reset learning rate
            if stall_count > 20:
                break
        else:
            stall_count = 0
