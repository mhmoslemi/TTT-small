import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def dcorr_dh_vec(h, n, idx):
    """Compute dc_k/dh[idx] for all k, vectorized."""
    n_corr = 2 * n - 1
    offset = n - 1
    L = np.arange(n_corr) - offset
    idx_plus_L = idx + L
    mask1 = (idx_plus_L >= 0) & (idx_plus_L < n)
    first = np.zeros(n_corr)
    first[mask1] = 1.0 - h[idx_plus_L[mask1]]
    idx_minus_L = idx - L
    mask2 = (idx_minus_L >= 0) & (idx_minus_L < n)
    second = np.zeros(n_corr)
    second[mask2] = -h[idx_minus_L[mask2]]
    return first + second

def pair_coord_descent(h, max_sweeps=30, n_pairs_per_sweep=3000, seed=42):
    """Paired coordinate descent maintaining sum constraint."""
    n = len(h)
    target_sum = h.sum()
    h = h.copy()
    rng = np.random.RandomState(seed)
    n_corr = 2 * n - 1
    
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for sweep in range(max_sweeps):
        if n_pairs_per_sweep >= len(all_pairs):
            pairs = all_pairs[:]
            rng.shuffle(pairs)
        else:
            pairs = [all_pairs[rng.randint(len(all_pairs))] for _ in range(n_pairs_per_sweep)]
        
        for (j, ip) in pairs:
            s = h[j] + h[ip]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            dj = dcorr_dh_vec(h, n, j)
            dip = dcorr_dh_vec(h, n, ip)
            
            alpha = dj - dip
            beta = corr - alpha * h[j]
            
            # Evaluate on a grid
            t_grid = np.linspace(lo, hi, 40)
            vals = np.max(alpha[:, None] * t_grid[None, :] + beta[:, None], axis=0)
            best_idx = np.argmin(vals)
            t_best = t_grid[best_idx]
            
            # Refine
            if 0 < best_idx < 39:
                t_lo = t_grid[max(0, best_idx - 1)]
                t_hi = t_grid[min(39, best_idx + 1)]

                vals_fine = np.max(alpha[:, None] * t_fine[None, :] + beta[:, None], axis=0)
                best_fidx = np.argmin(vals_fine)
                t_best = t_fine[best_fidx]
            
            if abs(t_best - h[j]) > 1e-10:
                h[j] = t_best
                h[ip] = s - t_best

        h = project_feasible(h, target_sum)
    
    return h

def subgrad_refine(h, max_iters=50000, base_lr=0.003, seed=42):
    """Subgradient method with decaying step size."""
    n = len(h)
    target_sum = h.sum()
    h = h.copy()
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        
        # Exact subgradient of c_{k_worst}
        offset = n - 1
        L = k_worst - offset
        grad = np.zeros(n)
        if L >= 0:
            for i in range(n - L):
                grad[i] += (1.0 - h[i + L])
                grad[i + L] -= h[i]
        else:
            la = -L
            for i in range(la, n):
                grad[i] += (1.0 - h[i - la])
                grad[i - la] -= h[i]
        
        lr = base_lr / (1.0 + it * 0.0003)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
    
    return h
