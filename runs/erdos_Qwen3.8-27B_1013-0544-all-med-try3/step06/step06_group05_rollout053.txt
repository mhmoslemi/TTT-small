import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    
    # Bisection for projection onto {0<=x<=1, sum(x)=target}
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 0) & (result < 1)
        nm = mask.sum()
        if nm > 0:
            result[mask] += diff / nm
            result = np.clip(result, 0.0, 1.0)
        diff = target - result.sum()
        if abs(diff) > 1e-15:
            if diff > 0:
                idx = np.argmin(result)
            else:
                idx = np.argmax(result)
            result[idx] += diff
            result = np.clip(result, 0.0, 1.0)
    
    return result

    """Log-sum-exp approximation of max correlation"""
    c = compute_correlation_vector(h, n)
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / alpha)

    c = compute_correlation_vector(h, n)
    m = c.max()
    # p_m = softmax(alpha * c)
    shifted = alpha * (c - m)
    p = np.exp(shifted - shifted.max())
    p = p / p.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n_half = n - 1  # index of zero lag
    
    for j in range(n):
        g = 0.0
        for m_idx in range(2 * n - 1):
            k = m_idx - n_half

        grad[j] = g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        nm = mask.sum()
        if nm > 0:
            result[mask] += diff / nm
            result = np.clip(result, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = (result < 1.0 - 1e-10)
        else:
            mask = (result > 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def lbfgsb_optimize(h, n, deadline, alphas=None, mu=500.0, maxiter=1000):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 500000]
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline:
            break
        dx = 2.0 / n

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            m = c.max()
            f = m + float(np.log(np.sum(np.exp(c - m))) / alpha)
            pen = mu * (x.sum() - n / 2.0) ** 2
            return f + pen

        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            mx = shifted.max()
            p = np.exp(shifted - mx)
            p /= p.sum()
            grad = np.zeros(n)
            n_half = n - 1
            for m_idx in range(2 * n - 1):
                pm = p[m_idx]
                if pm < 1e-13:
                    continue
                k = m_idx - n_half
                js = max(0, k)
                je = min(n - 1, n - 1 + k)
                if js <= je:
                    grad[js:je + 1] += pm * (1.0 - x[js - k:je - k + 1])
                js2 = max(0, -k)
                je2 = min(n - 1, n - 1 - k)
                if js2 <= je2:
                    grad[js2:je2 + 1] -= pm * x[js2 + k:je2 + k + 1]
            grad *= dx
            grad += 2.0 * mu * (x.sum() - n / 2.0)
            return grad

        try:
            res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        return m + float(np.log(np.sum(np.exp(c - m))) / alpha)

    def sum_eq(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def near_pair_search(h, n, deadline, deltas, radius=60, max_passes=200):
    h = h.copy()
    best_c = compute_c5(h, n)

    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi_val = h[i] + d
                    hj_val = h[j] - d
                    if 0.0 <= hi_val <= 1.0 and 0.0 <= hj_val <= 1.0:
                        h[i], h[j] = hi_val, hj_val
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi_val - d, hj_val + d
            if not improved:
                break
    return h, best_c

def coordinate_descent(h, n, deadline, steps=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0

    for _ in range(steps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            remain_sum = target - (h.sum() - old_val)
            lo = max(0.0, remain_sum - (n - 1) * 1.0)
            hi = min(1.0, remain_sum)
            if lo >= hi:
                continue
            best_val = old_val
            best_c_val = compute_c5(h, n)
            for trial_val in np.linspace(lo, hi, 20):
                h[i] = trial_val
                c_val = compute_c5(h, n)
                if c_val < best_c_val:
                    best_c_val = c_val
                    best_val = trial_val
            h[i] = best_val
            if best_val != old_val:
                improved = True
            if compute_c5(h, n) < best_c - 1e-15:
                best_c = compute_c5(h, n)
        if not improved:
            break
    return h, best_c
