def compute_gradient_matrix(h, n):
    """Compute g[k, p] = dc[k]/dh[p] for all k, p."""
    # g[k, p] = (1 - 2*h[k-p]) if low <= p <= high, else 0
    # where low = max(0, k-n+1), high = min(n-1, k)
    g = np.zeros((2*n-1, n))

        low = max(0, k - n + 1)
        high = min(n - 1, k)
        if low <= high:
            indices = np.arange(low, high + 1)
            g[k, indices] = 1.0 - 2.0 * h[k - indices]
    return g

def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()

            h += (diff / tot) * room
        else:

            if tot < 1e-15:

            h += (diff / tot) * h

def subgradient_ascent(h, n, deadline):
    """Subgradient method for minimizing max correlation."""
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(c.max())
    
    # Learning rate schedule
    lr = 0.01
    decay = 0.99
    
    for iteration in range(100000):
        if time.time() > deadline:
            break
        
        k_star = int(np.argmax(c))
        # Compute subgradient at k_star
        # g[k, p] = (1 - 2*h[k-p]) * I[valid]
        low = max(0, k_star - n + 1)
        high = min(n - 1, k_star)
        indices = np.arange(low, high + 1)
        grad = np.zeros(n)
        grad[indices] = (1.0 - 2.0 * h[k_star - indices]) * dx
        
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c_val = float(c_new.max())

            h = h_new
            c = c_new
        else:
            # Try smaller step
            lr *= 0.5
            if lr < 1e-8:
                lr = 0.01
                decay *= 0.95
        
        lr = max(lr * decay, 1e-8)
    
    return h, best_c

            if tot < 1e-15:
