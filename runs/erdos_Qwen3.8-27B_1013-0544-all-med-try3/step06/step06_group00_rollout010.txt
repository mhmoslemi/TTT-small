def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad(h, n, weights):
    """Compute gradient of weighted sum of correlations.
    weights[n-1 + k] corresponds to shift k (from -(n-1) to n-1)."""
    dx = 2.0 / n

            t = -k

                grad[t:] += -w * h[:n-t]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad_analytical(h, n, weights):
    """
    Compute gradient of sum_k weights[k] * c_k where c_k = correlate_k * dx.
    weights is indexed by correlate index (0 to 2n-2), corresponding to shift k = idx - (n-1).
    Returns gradient vector of length n.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)  # shift
        if k >= 0:
            # c_k = sum_{i=k}^{n-1} h[i]*(1-h[i-k])*dx
            # d/dh[j]: (1-h[j-k])*dx for j in [k, n-1]; -h[j+k]*dx for j in [0, n-1-k]
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h) * dx
            else:
                grad[k:] += w * (1.0 - h[:n-k]) * dx
                grad[:n-k] += -w * h[k:] * dx
        else:
            t = -k
            # c_k = sum_{i=0}^{n-1-t} h[i]*(1-h[i+t])*dx
            # d/dh[j]: (1-h[j+t])*dx for j in [0, n-1-t]; -h[j-t]*dx for j in [t, n-1]
            grad[:n-t] += w * (1.0 - h[t:]) * dx
            grad[t:] += -w * h[:n-t] * dx
    return grad

def smooth_gd(h, n, deadline, alpha_list, lr_init=0.01, momentum=0.9):
    """Gradient descent with log-sum-exp smoothing."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        lr = lr_init / max(1, np.log10(alpha))
        mu = momentum
        stall_count = 0
        sub_deadline = min(deadline - 2, time.time() + 0.08 * (deadline - time.time()))
        
        while time.time() < sub_deadline:
            c = np.correlate(h, 1 - h, mode='full') * dx
            m = c.max()
            shifted = alpha * (c - m)
            # Guard against overflow
            shifted = shifted - shifted.max()
            exp_s = np.exp(shifted)
            weights = exp_s / exp_s.sum()
            
            grad = compute_grad_analytical(h, n, weights)
            
            v = mu * v - lr * grad
            h_new = project_feasible(h + v, n)
            
            # Adaptive step: if objective increases too much, reduce step
            c_new = compute_c5(h_new, n)
            if c_new > c + 0.001:
                # Reject step
                v = v * 0.5
                lr *= 0.7
                stall_count += 1
            else:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                stall_count = 0
            
            if stall_count > 50:
                break
        
        v = np.zeros(n)
        mu = min(0.95, momentum + 0.01)
    
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    """Greedy binary local search."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(int(j))
                    zeros_list.remove(int(i))
                    ones_list.append(int(i))
                    zeros_list.append(int(j))
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.9999
    
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (int(j) - int(i))
        delta[k1] += 1.0
        k2 = n - 1 + (int(i) - int(j))
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(5000.0 * (cur_c - nc) / T))
        if accept:
            h[int(i)], h[int(j)] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    
    return best_h, best_c
