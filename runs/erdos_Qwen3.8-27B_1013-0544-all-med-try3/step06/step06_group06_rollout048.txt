import numpy as np
from scipy.fft import fft, ifft
import time

def compute_auto_fft(h, n):
    """Compute linear autocorrelation of binary h using FFT."""
    # Pad to length 2n-1 for linear correlation
    size = 2 * n - 1
    H = fft(h, size=size)
    auto = np.real(ifft(H * np.conj(H)))[:n]  # lags 0 to n-1
    # Full autocorrelation (lags -(n-1) to n-1)
    auto_full = np.concatenate([auto[:0:-1], auto])
    return auto_full  # length 2n-1, index n-1 is lag 0

def compute_c5_from_auto(auto_full, n):
    """Compute C5 from autocorrelation."""
    dx = 2.0 / n
    g = n/2.0 - auto_full  # This is the overlap at each shift
    return float(np.max(g) * dx)

    """Generate a golden ratio based binary pattern."""
    phi = (1 + np.sqrt(5)) / 2
    h = np.zeros(n)
    for i in range(n):
