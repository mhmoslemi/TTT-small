        # Step 2a: Subgradient with momentum
        sg_time = 0.15 * budget_s
        max_it = min(50000, int(sg_time * 200))
        h_work, c5_work = subgradient_optimize(h_work, n, ts, max_it, expired)
        if c5_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c5_work
        
        if expired():
            break
        
        # Step 2b: SLSQP with increasing beta
        for beta in [50, 300, 1500]:
            if expired():
                break
            h_work, c5_work = slsqp_optimize(h_work, n, dx, ts, beta, maxiter=2000)
            if c5_work < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_work
        
        if expired():
            break
        
        # Step 2c: Active set refinement
        h_work, c5_work = active_set_refine(h_work, n, dx, ts, max_rounds=10, time_limit_check=expired)
        if c5_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c5_work
    
    # Phase 3: Intensive refinement of the best solution
    if time_left() > 30:
        n_best = len(best_h)
        ts_best = n_best / 2.0
        dx_best = 2.0 / n_best
        h_work = best_h.copy()
        
        # More aggressive SLSQP
        for beta in [1000, 5000, 20000]:
            if expired():
                break
            h_new, c5_new = slsqp_optimize(h_work, n_best, dx_best, ts_best, beta, maxiter=5000)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new.copy()
        
        # Active set with more rounds
        if time_left() > 20:
            h_new, c5_new = active_set_refine(h_work, n_best, dx_best, ts_best,
                                              max_rounds=20, time_limit_check=expired)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Phase 4: Try fine-tuning resolution around the best
    if time_left() > 50:
        n_ref = len(best_h)
        for dn in [-2, -1, 1, 2, 3, 4, 5, -4, -3]:
            if time_left() < 25:
                break
            n_try = n_ref + dn
            if n_try < 30:
                continue
            ts_t = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            c5_t = compute_c5(h_t)
            
            if c5_t < best_c5 * 1.005:  # Only if close
                # Quick SLSQP
                h_t, c5_t = slsqp_optimize(h_t, n_try, dx_t, ts_t, 500, maxiter=1500)
                if c5_t < best_c5:
                    best_h = h_t.copy()
                    best_c5 = c5_t
                # Active set
                if time_left() > 15:
                    h_t, c5_t = active_set_refine(h_t, n_try, dx_t, ts_t,
                                                  max_rounds=8, time_limit_check=expired)
                    if c5_t < best_c5:
                        best_h = h_t.copy()
                        best_c5 = c5_t
    
    # Phase 5: Final subgradient polish on the best
    if time_left() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_work = best_h.copy()
        h_work, c5_work = subgradient_optimize(h_work, n_f, ts_f, 20000, expired)
        if c5_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c5_work
    
    # Final SLSQP polish
    if time_left() > 8:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        h_work, c5_work = slsqp_optimize(best_h, n_f, dx_f, ts_f, 10000, maxiter=3000)
        if c5_work < best_c5:
            best_h = h_work.copy()
            best_c5 = c5_work
    
    return best_h, best_c5, len(best_h)
