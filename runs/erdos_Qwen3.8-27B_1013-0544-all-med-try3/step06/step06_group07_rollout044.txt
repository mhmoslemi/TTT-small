    # Phase 3: Subgradient refinement
    if time_left() > 80:
        h_sg, c5_sg = subgradient_method(best_h, n, dx, target_sum, max_iters=50000, time_fn=time_left)
        if c5_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c5_sg

    # Phase 4: High-beta SLSQP polish
    if time_left() > 40:
        for beta in [50000, 100000, 200000]:
            if time_left() < 15:
                break
            h_new = slsqp_run(best_h, n, dx, target_sum, beta, maxiter=10000, ftol=1e-16, maxls=100)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    # Phase 5: Try different resolutions
    if time_left() > 60:
        n_base = len(best_h)
        resolutions = []
        for delta in [10, 20, 30, 50, -10, -20, 5, 15, 40]:
            n_try = n_base + delta
            if n_try >= 30 and n_try not in resolutions:
                resolutions.append(n_try)

        for n_try in resolutions:
            if time_left() < 30:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)

            for beta in [500, 3000, 10000, 30000]:
                if time_left() < 15:
                    break
                h_new = slsqp_run(h_t, n_try, dx_t, ts_t, beta, maxiter=4000, ftol=1e-15, maxls=50)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_try
                h_t = h_new.copy()

    # Phase 6: Coordinate descent polish at best resolution
    if time_left() > 20:
        n_cd = len(best_h)
        dx_cd = 2.0 / n_cd
        ts_cd = n_cd / 2.0
        h_cd, c5_cd = coordinate_descent(best_h, n_cd, dx_cd, ts_cd, max_iters=20000, time_fn=time_left)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 7: Final SLSQP with ultra-high beta
    if time_left() > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_new = slsqp_run(best_h, n_f, dx_f, ts_f, 500000, maxiter=10000, ftol=1e-16, maxls=100)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

    # Final verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
