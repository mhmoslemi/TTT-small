import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def swap_delta(h, a, b, n, ks):
    delta = np.zeros(2 * n - 1)
    ak = a - ks
    m = (ak >= 0) & (ak < n)
    delta[m] += -(1.0 - h[ak[m]])
    bk = b - ks
    m = (bk >= 0) & (bk < n)
    delta[m] += 1.0 - h[bk[m]]
    ak2 = a + ks
    m = (ak2 >= 0) & (ak2 < n)
    delta[m] += h[ak2[m]]
    bk2 = b + ks
    m = (bk2 >= 0) & (bk2 < n)
    delta[m] -= h[bk2[m]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (a - b)] += 1.0
    delta[n - 1 + (b - a)] += 1.0

def sa_optimize(h_start, n, deadline, seed=42):
    """Run simulated annealing on binary solutions."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    ks = np.arange(-(n-1), n)
    
    h = to_binary(h_start, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = 3.0
    T_min = 0.05
    cool_rate = 0.9998
    no_improve = 0
    improve_threshold = 50000
    
    while time.time() < deadline:
        a = rng.randint(len(ones))
        b = rng.randint(len(zeros))
        ai = ones[a]
        bi = zeros[b]
        
        delta = swap_delta(h, ai, bi, n, ks)
        new_corr = corr + delta
        new_max = new_corr.max()

            h[ai], h[bi] = 0.0, 1.0
            corr = new_corr
            old_cur = cur_max
            cur_max = new_max
            # Update ones and zeros
            ones[a] = bi
            zeros[b] = ai
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Accept with probability exp(-delta/T)
            d = new_max - cur_max
            if d < T * 10:  # Only compute exp if reasonable

                    corr = new_corr
                    cur_max = new_max

                    zeros[b] = ai

def greedy_polish(h, n, deadline):
    """Greedy binary search."""
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i_idx in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[i_idx]
            for j_idx in range(len(ones)):
                j = ones[j_idx]
                delta = swap_delta(h, j, i, n, ks)

                    if cur_max < best_max:

                        best_h = h.copy()
                    break
            if improved:

    return best_h, best_max * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def swap_delta(h, a, b, n, ks):
    delta = np.zeros(2 * n - 1)
    ak = a - ks
    m = (ak >= 0) & (ak < n)
    delta[m] += -(1.0 - h[ak[m]])
    bk = b - ks
    m = (bk >= 0) & (bk < n)
    delta[m] += 1.0 - h[bk[m]]
    ak2 = a + ks
    m = (ak2 >= 0) & (ak2 < n)
    delta[m] += h[ak2[m]]
    bk2 = b + ks
    m = (bk2 >= 0) & (bk2 < n)
    delta[m] -= h[bk2[m]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (a - b)] += 1.0
    delta[n - 1 + (b - a)] += 1.0
    return delta

def greedy_polish(h, n, deadline):
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    max_sweeps = 50
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i_idx in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[i_idx]
            for j_idx in range(len(ones)):
                if time.time() > deadline:
                    break
                j = ones[j_idx]
                delta = swap_delta(h, j, i, n, ks)
                new_max = (corr + delta).max()
                if new_max < cur_max - 1e-15:
                    h[j], h[i] = 0.0, 1.0
                    corr = corr + delta
                    cur_max = new_max
                    ones[j_idx] = i
                    zeros[i_idx] = j
                    improved = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                    break
            if improved:
                break
        if not improved:
            break

    return best_h, best_max * dx
