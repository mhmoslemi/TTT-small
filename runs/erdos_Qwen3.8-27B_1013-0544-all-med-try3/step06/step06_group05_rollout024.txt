import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    omx = 1.0 - x
    c = np.correlate(x, omx, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    obj = c_max + log_sum / alpha
    p = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        pk = p[idx]
        if pk < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            jend = n - k
            grad[0:jend] += pk * omx[k:k+jend]
            grad[k:n] -= pk * x[0:n-k]
        else:
            kk = -k
            grad[kk:n] += pk * omx[0:n-kk]
            grad[0:n-kk] -= pk * x[kk:n]
    grad *= dx
    return obj, grad

def _slsqp_obj(x, n, alpha):
    val, _ = _smooth_obj_grad(x, n, alpha)
    return val

def _slsqp_jac(x, n, alpha):
    _, g = _smooth_obj_grad(x, n, alpha)
    return g

def _sum_con_fun(x):
    return float(np.sum(x) - x.size / 2.0)

def _sum_con_jac(x):
    return np.ones(x.size)

def slsqp_optimize(h, n, alpha, deadline, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_coord_descent(h, n, deadline, rng, max_iter=None):
    if max_iter is None:
        max_iter = 500000
    best_c = compute_c5(h, n)
    h = h.copy()
    dx = 2.0 / n
    iter_count = 0
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        iter_count += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_min >= d_max - 1e-15:
            continue
        lo = d_min
        hi = d_max
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            h[i] += m1
            h[j] -= m1
            c1 = compute_c5(h, n)
            h[i] += m2 - m1
            h[j] -= m2 - m1
            c2 = compute_c5(h, n)
            if c1 < c2:
                hi = m2
            else:
                lo = m1
            if hi - lo < 1e-12:
                break
        h[i] += lo
        h[j] -= lo
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:

                hp = project_feasible(hp, n)
                cp = compute_c5(hp, n)
                if cp < best_c + 0.001:
                    h = hp
                stagnant = 0
    return h, best_c
