    return np.correlate(h, 1 - h, mode='full') * dx

def compute_grad_matrix(h, n):
    """Compute grad[k, p] = d(corr[k]*dx)/d(h[p]) for all k, p"""
    dx = 2.0 / n
    # corr[k + n - 1] = sum_i h[i+k]*(1-h[i]) * dx
    # grad[k+n-1, p] = dx * ((1-h[p-k])_valid - h[p+k]_valid)
    grad = np.zeros((2*n-1, n))
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)  # actual lag
        for p in range(n):
            g = 0.0
            # (1-h[p-k]) if 0 <= p-k < n
            if 0 <= p - k < n:
                g += (1.0 - h[p-k])
            # -h[p+k] if 0 <= p+k < n
            if 0 <= p + k < n:
                g -= h[p+k]
            grad[k_idx, p] = dx * g
    return grad

        # (1-h[p-k]) for valid p: p-k in [0,n) means p in [k, k+n)
        # intersected with [0,n)
        p_start = max(0, k)
        p_end = min(n, k + n)

        grad[k_idx, idx] += (1.0 - h[idx - k])

        # intersected with [0,n)

        p_end2 = min(n, n - k)
        idx2 = np.arange(p_start2, p_end2)
        grad[k_idx, idx2] -= h[idx2 + k]
    grad *= dx
    return grad

Actually, the key insight is: for a given pair (i, j), the change in the max is:
max_k (c[k] + δ * (G[k,i] - G[k,j]))

Let d_k = G[k,i] - G[k,j]. Then the new max is:
max_k (c[k] + δ * d_k)

For small δ, the new max is approximately:
c[max] + δ * d[max] (if the current max lag has the most negative d)

But more precisely, for a given (i,j), the optimal δ that minimizes the max is found by:
min_δ max_k (c[k] + δ * d_k)

This is a 1D piecewise-linear optimization. The minimum occurs at a vertex, which is where two of the lines c[k] + δ*d_k intersect.

For efficiency, I'll just try a few δ values: {0.001, 0.005, 0.01, 0.05, 0.1, 0.5} and pick the best.

        i_scores = G[k_worst, one_positions]
        j_scores = G[k_worst, zero_positions]

        n_try_j = min(50, len(j_ranked))

                    best_c = c.max()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h_bin, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_grad_matrix(h, n):
    dx = 2.0 / n
    grad = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        k = idx - (n-1)
        p_lo = max(0, k)
        p_hi = min(n, k + n)
        if p_lo < p_hi:
            ps = np.arange(p_lo, p_hi)
            grad[idx, ps] += (1.0 - h[ps - k])
        p_lo2 = max(0, -k)
        p_hi2 = min(n, n - k)
        if p_lo2 < p_hi2:
            ps2 = np.arange(p_lo2, p_hi2)
            grad[idx, ps2] -= h[ps2 + k]
    grad *= dx
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def continuous_local_search(h, n, deadline, max_sweeps=200):
    h = h.copy()
    c = compute_all_lags(h, n)
    best_c = c.max()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        G = compute_grad_matrix(h, n)
        k_worst = np.argmax(c)
        
        # Find positions where increasing h helps (G[k_worst, p] < 0)
        inc_pos = np.where(G[k_worst] < -1e-8)[0]
        dec_pos = np.where(G[k_worst] > 1e-8)[0]
        
        if len(inc_pos) == 0 or len(dec_pos) == 0:
            break
        
        # Rank by effect on worst lag
        inc_pos = inc_pos[np.argsort(G[k_worst, inc_pos])]  # most negative first
        dec_pos = dec_pos[np.argsort(-G[k_worst, dec_pos])]  # most positive first
        
        improved = False
        n_try = min(40, len(inc_pos))
        n_try_d = min(40, len(dec_pos))
        
        for ii in range(n_try):
            if time.time() > deadline:
                break
            i = inc_pos[ii]
            if h[i] >= 1.0 - 1e-10:
                continue
            for jj in range(n_try_d):
                j = dec_pos[jj]
                if h[j] <= 1e-10:
                    continue
                if i == j:
                    continue
                
                delta_max = min(h[j], 1.0 - h[i])
                if delta_max < 1e-8:
                    continue
                
                # Try several delta values
                for frac in [0.5, 0.2, 0.1, 0.05, 0.01]:
                    delta = delta_max * frac
                    if delta < 1e-6:
                        continue
                    new_c = c + delta * (G[:, i] - G[:, j])
                    new_max = new_c.max()
                    if new_max < best_c - 1e-12:
                        h[i] += delta
                        h[j] -= delta
                        c = new_c
                        best_c = new_max
                        improved = True
                        break
        
        # Also try reverse direction (decrease at inc_pos, increase at dec_pos)
        # This handles the case where the gradient-based direction is wrong
        # (due to the non-convexity)
        if not improved:
            for ii in range(min(20, len(inc_pos))):
                if time.time() > deadline:
                    break
                i = dec_pos[min(ii, len(dec_pos)-1)]  # actually swap the roles
                j = inc_pos[min(ii, len(inc_pos)-1)]
                if i == j or h[j] >= 1.0 or h[i] <= 0.0:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                for frac in [0.5, 0.2, 0.1]:
                    delta = delta_max * frac
                    if delta < 1e-6:
                        continue
                    new_c = c + delta * (G[:, i] - G[:, j])
                    new_max = new_c.max()
                    if new_max < best_c - 1e-12:
                        h[i] += delta
                        h[j] -= delta
                        c = new_c
                        best_c = new_max
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    c = compute_all_lags(h, n)
    return h, float(c.max())

    h_bin = (h >= 0.5).astype(float)

        one_idx = np.where(h_bin == 1)[0]
        to_remove = one_idx[np.argsort(h[one_idx])[:n_ones - target]]
        h_bin[to_remove] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        to_add = zero_idx[np.argsort(-h[zero_idx])[:target - n_ones]]
        h_bin[to_add] = 1
    
    c = compute_all_lags(h_bin, n)

        if time.time() > deadline:
            break
        
        G = compute_grad_matrix(h_bin, n)
        k_worst = np.argmax(c)
        
        one_pos = np.where(h_bin == 1)[0]
        zero_pos = np.where(h_bin == 0)[0]
        
        if len(one_pos) == 0 or len(zero_pos) == 0:

        # Δc = G[:, j] - G[:, i]
        # Want G[k_worst, j] - G[k_worst, i] < 0, i.e., G[k_worst, i] > G[k_worst, j]

            if improved:
                break

            n_try2 = min(60, len(i_ranked))
