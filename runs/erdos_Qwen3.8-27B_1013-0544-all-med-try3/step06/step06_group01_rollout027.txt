import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary_search(h_init, n, deadline, seed=42, max_sweeps=100000):
    """
    Full greedy best-improvement binary local search.
    Tries ALL possible swaps each sweep until no improvement is found.
    """
    target = n // 2
    dx = 2.0 / n
    
    # Initialize binary
    sorted_idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    
    best_max = cur_max
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    m = np.arange(-(n-1), n)  # lag values
    ncorr = 2*n - 1  # length of correlation
    
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        best_swap_improvement = cur_max  # we want to minimize, so track best (lowest) new max
        best_i = -1
        best_j = -1
        best_delta = None
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Try all pairs
        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                
                # Compute delta
                ip = i + m
                im = i - m
                jp = j + m
                jm = j - m
                
                delta = np.zeros(ncorr)
                msk = (ip >= 0) & (ip < n)
                delta[msk] += (1.0 - h[ip[msk]])
                msk = (jm >= 0) & (jm < n)
                delta[msk] += (1.0 - h[jm[msk]])
                msk = (im >= 0) & (im < n)
                delta[msk] -= (1.0 - h[im[msk]])
                msk = (jp >= 0) & (jp < n)
                delta[msk] -= (1.0 - h[jp[msk]])
                delta[n-1+i-j] += 1.0
                delta[n-1+j-i] += 1.0
                
                new_max = (corr + delta).max()
                
                if new_max < best_swap_improvement - 1e-15:
                    best_swap_improvement = new_max
                    best_i = i
                    best_j = j
                    best_delta = delta
        
        if best_i >= 0:
            # Apply the best swap
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = corr + best_delta
            cur_max = best_swap_improvement
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break  # No improvement possible
    
    return best_h, best_max * dx
