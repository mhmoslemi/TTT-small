```python
def find_optimal_coord(C, slope, n_points):
    """Find x in [0,1] minimizing max_k (C_k + x*slope_k)."""
    def slope_range(x):
        vals = C + x * slope
        m = vals.max()
        active = vals >= m - 1e-14
        return slope[active].min(), slope[active].max()
    
    # Check boundaries
    s_lo, s_hi = slope_range(0.0)
    if s_lo > 1e-14:
        return 0.0
    
    s_lo, s_hi = slope_range(1.0)
    if s_hi < -1e-14:
        return 1.0
    
    # Binary search
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_lo, s_hi = slope_range(mid)
        if s_hi < -1e-14:
            lo = mid  # still decreasing, minimum is to the right
        elif s_lo > 1e-14:
            hi = mid  # already increasing, minimum is to the left
        else:
            lo = hi = mid  # 0 is in subdifferential
            break
    
    return (lo + hi) / 2
```
