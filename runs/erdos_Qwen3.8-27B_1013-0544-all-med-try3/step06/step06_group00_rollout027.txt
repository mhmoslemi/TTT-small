import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_simplex_box(x, target_sum):
    """Project x onto {y: 0<=y_i<=1, sum(y)=target_sum}"""
    if x.sum() == target_sum:
        return np.clip(x, 0.0, 1.0)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_lse_grad_and_value(h, n, alpha):
    """Compute gradient and value of (1/alpha)*log(sum exp(alpha*corr_k*dx))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Numerically stable log-sum-exp
    m = corr.max()
    shifted = alpha * (corr - m)
    log_sum = m + np.log(np.exp(shifted).sum()) / alpha
    # Weights (softmax)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient
    # grad[j] = dx * sum_s w_s * (1-h[j+s])_{valid} - h[j-s]_{valid}
    # = dx * (1 - sum_s w_s * h[j+s]_{valid} - sum_s w_s * h[j-s]_{valid}) ... wait
    # Actually: grad[j] = dx * sum_s w_s * [(1-h[j+s])*1_{j+s in [0,n)} - h[j-s]*1_{j-s in [0,n)}]
    
    grad = np.zeros(n)
    n1 = n - 1
    # For each shift s = c - (n-1) where c is corr index:
    # contribution to grad from shift s:
    #   w_s * (1-h[j+s]) for j in [max(0,-s), min(n, n-s))  [where j+s is valid]
    #   -w_s * h[j-s] for j in [max(0,s), min(n, n+s))      [where j-s is valid]
    
    for c in range(2 * n - 1):
        w = weights[c]
        if w < 1e-14:
            continue
        s = c - n1
        if s >= 0:
            # j+s valid: j in [0, n-1-s], i.e., h[j+s] for j=0..n-1-s
            # j-s valid: j in [s, n-1], i.e., h[j-s] for j=s..n-1
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            # j+s = j-ss valid: j in [ss, n-1], i.e., h[j-ss] for j=ss..n-1
            # j-s = j+ss valid: j in [0, n-1-ss], i.e., h[j+ss] for j=0..n-1-ss
            grad[ss:] += w * (1.0 - h[:n-ss])
            grad[:n-ss] -= w * h[ss:]
    
    grad *= dx
    return grad, log_sum

def pgd_optimize(h, n, alpha, deadline, lr_scale=1.0, n_inner=200):
    """Project gradient descent with momentum on LSE objective"""
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    momentum = np.zeros(n)
    beta = 0.9
    # Initial step size
    grad, _ = compute_lse_grad_and_value(h, n, alpha)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return best_h, best_c
    eta = lr_scale / (gnorm * np.sqrt(n))  # adaptive initial step
    
    for it in range(n_inner):
        if time.time() > deadline:
            break
        grad, _ = compute_lse_grad_and_value(h, n, alpha)
        momentum = beta * momentum + grad
        h_new = project_simplex_box(h - eta * momentum, target)
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        eta *= 0.9995
        if eta < 1e-10:
            break
    
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def binary_local_search(h_init, n, deadline, max_swaps=100000):
    """Fast binary local search using incremental correlation"""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    m_range = np.arange(-(n-1), n)
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    swaps = 0
    no_improve = 0
    while swaps < max_swaps and no_improve < 50:
        if time.time() > deadline:
            break
        # Random swap
        i = zeros_pos[np.random.randint(len(zeros_pos))]
        j = ones_pos[np.random.randint(len(ones_pos))]
        if i == j:
            continue
        
        # Compute delta
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        delta[k1] += 1.0
        k2 = n-1 + (i-j)
        delta[k2] += 1.0
        
        nc = (corr + delta).max() * dx
        swaps += 1
        if nc < cur_c - 1e-15:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            # Update position lists
            np.delete(zeros_pos, np.where(zeros_pos == i)[0][0])
            np.append(zeros_pos, j)
            np.delete(ones_pos, np.where(ones_pos == j)[0][0])
            np.append(ones_pos, i)
        else:
            no_improve += 1
    
    return best_h, best_c

    #   h[j+s] is valid for j in [max(0,-s), min(n, n-s))
    #   h[j-s] is valid for j in [max(0,s), min(n, n+s))

Actually, one more idea: instead of pure gradient descent, what if I use coordinate descent? At each step, I pick a random coordinate j and optimize over it (given all other coordinates fixed).

For coordinate j, the objective is:
max_k corr_k(h) where only h[j] varies.

corr_k as a function of h[j]:
corr_k = sum_i h[i]*(1-h[i+k])
The terms involving h[j] are:
- h[j]*(1-h[j+k]) if j+k is valid
- h[j-k]*(1-h[j]) if j-k is valid

So corr_k = (terms not involving j) + h[j]*(1-h[j+k]) + h[j-k]*(1-h[j])
= C_k + h[j] - h[j]*h[j+k] + h[j-k] - h[j-k]*h[j]
= C_k + h[j] - h[j]*(h[j+k] + h[j-k])
= C_k + h[j]*(1 - h[j+k] - h[j-k])

This is linear in h[j]! So for each k, corr_k is linear in h[j], and the max of linear functions is convex in h[j]. I can minimize this 1D convex function in closed form.

The optimal h[j] given all other coordinates:
min_{0≤x≤1, sum constraint} max_k (C_k + x * a_k)

where a_k = 1 - h[j+k] - h[j-k] (with appropriate boundary handling).

This is a 1D problem: min_x max_k (C_k + x*a_k) subject to 0≤x≤1 and x = n/2 - sum_{i≠j} h[i] + h[j]_old... 

Wait, the constraint is sum_i h[i] = n/2. If I change h[j], I need to adjust something else. Unless I do it jointly.

Actually, coordinate descent with the equality constraint is tricky. I'd need to either:
(a) Drop the constraint and use projection
(b) Do 2D updates (change two coordinates simultaneously)

Option (b) is better: at each step, pick coordinates (i, j) and optimize over (h[i], h[j]) subject to h[i] + h[j] = c (the sum constraint) and 0≤h[i],h[j]≤1.

Given h[i] + h[j] = c, let h[i] = t, h[j] = c - t. The feasible range for t is [max(0, c-1), min(1, c)].

The objective becomes max_k corr_k(t) which is max of linear functions of t (since each corr_k is linear in h[i] and h[j] separately, hence linear in t).

So I'm minimizing a piecewise-linear convex function of t over an interval. The minimum is at one of the breakpoints or at an endpoint.

The breakpoints occur where two of the (C_k + t * B_k) functions cross, which happens at t = (C_k' - C_k) / (B_k - B_k') for pairs (k, k').

There are O(n^2) breakpoints but typically only a few are active. A practical approach:
- Evaluate the max at a few candidate t values (endpoints, and where the top-2 active constraints cross)
- Take the best one

This is actually quite efficient! Each 2D update costs O(n) (to compute all corr_k as functions of t) + O(n) (to find the best t). With O(n^2) possible (i,j) pairs, a full sweep is O(n^3). But I can do random pairs for O(1) sweeps.

Actually, let me simplify: for each random pair (i, j), compute the optimal t in O(n) time. Do this for many random pairs. This gives O(n) per update vs O(n^2) for the full gradient. Much faster!
