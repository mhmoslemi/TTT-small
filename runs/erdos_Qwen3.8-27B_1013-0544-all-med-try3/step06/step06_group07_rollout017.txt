def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_arr(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_full_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_arr(corr, beta) * dx


def obj_top_lags(h, dx, beta, top_indices):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sub = corr[top_indices]
    m = np.max(sub)
    shifted = np.clip(beta * (sub - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx


def sum_eq_constr(h, target_sum):

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()

        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0

def binary_local_search(h_bin, n, target_sum, max_iters, rng, best_c5_tracker):
    """Local search on binary patterns with exactly n/2 ones"""
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_local_c5 = current_c5
    best_local_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        move_type = rng.randint(3)
        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            # Swap one 1 with one 0
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            ones[ones == i] = j
            zeros[zeros == j] = i
        elif move_type == 1 and len(ones) >= 2 and len(zeros) >= 2:
            # Swap two 1s with two 0s
            i1, i2 = rng.choice(len(ones), size=2, replace=False)
            j1, j2 = rng.choice(len(zeros), size=2, replace=False)
            a, b = ones[i1], ones[i2]
            c, d = zeros[j1], zeros[j2]
            h[a] = 0.0
            h[b] = 0.0
            h[c] = 1.0
            h[d] = 1.0
            ones[i1] = c
            ones[i2] = d
            zeros[j1] = a
            zeros[j2] = b
        else:
            continue
        
        c5_new = compute_c5(h)
        if c5_new < current_c5 - 1e-15:
            current_c5 = c5_new
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local_h = h.copy()
                best_c5_tracker[0] = c5_new
        else:

                h[i] = 1.0
                h[j] = 0.0
                ones[ones == j] = i
                zeros[zeros == i] = j
            elif move_type == 1:
                h[a] = 1.0
                h[b] = 1.0
                h[c] = 0.0
                h[d] = 0.0
                ones[i1] = a
                ones[i2] = b
                zeros[j1] = c
                zeros[j2] = d
    
    return best_local_h, best_local_c5

                break
            corr = compute_corr(best_h)
            top = np.argsort(corr)[-n_top:]
            max_val = corr[top[0]]
            # Also include lags within 5% of max
            near_top = np.where(corr > 0.95 * max_val)[0]
            top = np.unique(np.concatenate([top, near_top]))
            top = top[:10]  # Cap at 10

            for beta in [200, 1000, 5000]:
                if rem() < 20:
                    break
                try:
                    cons = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                             'args': (target_sum,)}]
                    res = minimize(obj_top_lags, best_h, args=(dx, beta, top),
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 3: Stochastic local search (continuous)
    if rem() > 150:
        h_sa = best_h.copy()
        c5_sa = best_c5
        n_sa = best_n
        target_sa = n_sa / 2.0
        T = 0.0005
        T_min = 0.000005
        step = 0.03
        time_limit = rem() * 0.7

        it = 0
        sa_start = time.time()
        while time.time() - sa_start < time_limit and rem() > 15:
            if it % 500 == 499:
                T = max(T_min, T * 0.97)
                step = max(0.003, step * 0.99)

            n_pert = rng.randint(2, 7)
            idx = rng.choice(n_sa, size=n_pert, replace=False)
            h_pert = h_sa.copy()
            h_pert[idx] += rng.randn(n_pert) * step
            h_pert = project_feasible(h_pert, target_sa)
            c5_pert = compute_c5(h_pert)
            delta = c5_pert - c5_sa

            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                h_sa = h_pert
                c5_sa = c5_pert
                if c5_sa < best_c5:
                    best_h = h_sa.copy()
                    best_c5 = c5_sa
            it += 1

    # Phase 4: Binary local search
    if rem() > 100:
        n_cur = best_n
        target_cur = n_cur / 2.0
        # Round current best to binary
        h_bin = (best_h > 0.5).astype(float)
        # Fix sum
        ones_count = int(h_bin.sum())
        target_ones = n_cur // 2
        if ones_count > target_ones:
            # Remove some ones (those with lowest h values)
            one_idx = np.where(h_bin == 1.0)[0]
            remove = np.argsort(best_h[one_idx])[:ones_count - target_ones]
            h_bin[one_idx[remove]] = 0.0
        elif ones_count < target_ones:
            zero_idx = np.where(h_bin == 0.0)[0]
            add = np.argsort(-best_h[zero_idx])[:target_ones - ones_count]
            h_bin[zero_idx[add]] = 1.0

        best_c5_tracker = [best_c5]
        h_bin_best, c5_bin_best = binary_local_search(
            h_bin, n_cur, target_cur, 20000, rng, best_c5_tracker)

        if c5_bin_best < best_c5:
            best_h = h_bin_best.copy()
            best_c5 = c5_bin_best

        # Relax from binary with SLSQP
        if c5_bin_best <= best_c5 + 0.005 and rem() > 40:
            h_relax = h_bin_best.copy()
            h_relax = project_feasible(h_relax, target_cur)
            dx_cur = 2.0 / n_cur
            for beta in [200, 1000]:
                if rem() < 15:
                    break
                try:
                    cons = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                             'args': (target_cur,)}]
                    res = minimize(obj_full_smooth, h_relax, args=(dx_cur, beta),
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_cur)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        h_relax = h_new.copy()
                except Exception:
                    pass

        # Try another random binary start
        if rem() > 60:
            h_bin2 = random_binary(n_cur, rng)
            best_c5_tracker2 = [best_c5]
            h_bin2_best, c5_bin2_best = binary_local_search(
                h_bin2, n_cur, target_cur, 15000, rng, best_c5_tracker2)
            if c5_bin2_best < best_c5:
                best_h = h_bin2_best.copy()
                best_c5 = c5_bin2_best

    # Phase 5: Try higher resolutions
    for n_try in [best_n + 10, best_n + 20, best_n + 40, best_n + 60]:
        if rem() < 60:
            break
        if n_try % 2 != 0:
            n_try += 1
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, target_t)
        c5_t = compute_c5(h_t)
        bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))

        for beta in [100, 500, 2000]:
            if rem() < 20:
                break
            try:
                cons_t = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                           'args': (target_t,)}]
                res = minimize(obj_full_smooth, h_t, args=(dx_t, beta),
                              method='SLSQP', bounds=bounds_t, constraints=cons_t,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_t)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_try
                    h_t = h_new.copy()
            except Exception:
                pass

        # Active-set at new resolution
        if rem() > 30:
            corr_t = compute_corr(h_t)
            top_t = np.argsort(corr_t)[-5:]
            try:
                cons_t = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                           'args': (target_t,)}]
                res = minimize(obj_top_lags, h_t, args=(dx_t, 2000, top_t),
                              method='SLSQP', bounds=bounds_t, constraints=cons_t,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_t)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_try
            except Exception:
                pass

    # Phase 6: Try lower resolution too
    for n_try in [best_n - 10, best_n - 20]:
        if rem() < 50 or n_try < 30:
            break
        if n_try % 2 != 0:
            n_try -= 1
        if n_try < 30:
            continue
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, target_t)
        c5_t = compute_c5(h_t)
        bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))

        for beta in [200, 1000]:
            if rem() < 15:
                break
            try:
                cons_t = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                           'args': (target_t,)}]
                res = minimize(obj_full_smooth, h_t, args=(dx_t, beta),
                              method='SLSQP', bounds=bounds_t, constraints=cons_t,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_t)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_try
            except Exception:
                pass

    # Phase 7: Final polish with very high beta on top lags
    if rem() > 20:
        n_cur = best_n
        dx_cur = 2.0 / n_cur
        target_cur = n_cur / 2.0
        bounds_cur = Bounds(np.zeros(n_cur), np.ones(n_cur))
        corr = compute_corr(best_h)
        top = np.argsort(corr)[-3:]

        for beta in [5000, 20000, 50000]:
            if rem() < 8:
                break
            try:
                cons = [{'type': 'eq', 'fun': sum_eq_constr, 'jac': sum_eq_jac,
                         'args': (target_cur,)}]
                res = minimize(obj_top_lags, best_h, args=(dx_cur, beta, top),
                              method='SLSQP', bounds=bounds_cur, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_cur)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 8: One more stochastic refinement round
    if rem() > 40:
        h_sa = best_h.copy()
        c5_sa = best_c5
        n_sa = best_n
        target_sa = n_sa / 2.0
        T = 0.0002
        step = 0.01
        sa_end = time.time() + rem() * 0.8
        it = 0
        while time.time() < sa_end and rem() > 5:
            if it % 300 == 299:
                T = max(0.00001, T * 0.95)
                step = max(0.002, step * 0.98)
            n_pert = rng.randint(1, 5)
            idx = rng.choice(n_sa, size=n_pert, replace=False)
            h_pert = h_sa.copy()
            h_pert[idx] += rng.randn(n_pert) * step
            h_pert = project_feasible(h_pert, target_sa)
            c5_pert = compute_c5(h_pert)
            if c5_pert < c5_sa:
                h_sa = h_pert
                c5_sa = c5_pert
                if c5_sa < best_c5:
                    best_h = h_sa.copy()
                    best_c5 = c5_sa
            it += 1

    return best_h, best_c5, len(best_h)
