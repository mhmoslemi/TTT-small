```python
def frank_wolfe_step(h, n, alpha_weights, step_size):
    """Take one Frank-Wolfe step."""
    dx = 2.0/n
    # Compute gradients of active correlations
    # For shift s (0-indexed in full correlation, offset by n-1):
    # dR_s/dh[j] = dx * [(1-h[j+s']) if valid - h[j-s''] if valid]
    # where s' = s - (n-1) is the actual shift
    
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Get top shifts
    top_idx = np.argsort(corr)[-10:]
    
    # Compute negative gradient (direction of steepest descent of the weighted sum)
    grad = np.zeros(n)
    for s in top_idx:
        shift = s - (n-1)
        w = alpha_weights.get(s, 1.0)  # weight for this shift

    grad -= grad.mean()
```

```python
def obj_lbfgsb(x, n, alpha):
    """Objective for L-BFGS-B. x has n-1 elements, h[n-1] computed implicitly."""
    h = np.empty(n)
    h[:-1] = x
    h[n-1] = n/2.0 - x.sum()
    if h[n-1] < 0 or h[n-1] > 1:
        return 1e10  # infeasible
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full') * dx
    m = corr.max()
    # Smooth max
    shifted = alpha * (corr - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_grad_lbfgsb(x, n, alpha):
    """Gradient for L-BFGS-B."""
    h = np.empty(n)
    h[:-1] = x
    h[n-1] = n/2.0 - x.sum()
    if h[n-1] < 0 or h[n-1] > 1:
        return np.ones(n-1) * 1e10
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient of smooth max w.r.t. h
    grad_h = np.zeros(n)
    n_full = 2*n - 1
    ncs = n - 1  # center of full correlation
    for s in range(n_full):
        if p[s] < 1e-15:
            continue
        shift = s - ncs
        for j in range(n):
            jp = j + shift
            if 0 <= jp < n:
                grad_h[j] += p[s] * dx * (1 - h[jp])
            jm = j - shift
            if 0 <= jm < n:
                grad_h[j] += p[s] * dx * (-h[jm])
    
    # Chain rule: h[i] = x[i] for i < n-1, h[n-1] = n/2 - sum(x)
    # dh[i]/dx[j] = 1 if i=j, -1 if i=n-1
    grad = grad_h[:-1] - grad_h[n-1]
    return grad
```

This is p[s] * dx * (-h[0:n-shift]) shifted by `shift`.

```python
def exact_pair_step(h, n, i, j):
    """Find optimal transfer from j to i. Returns best t and new C."""
    dx = 2.0/n
    # Compute current correlations and gradients
    corr = compute_corr(h, n)  # shape (2n-1,)
    # For each shift s, compute g_s[i] - g_s[j]
    # g_s[k] = dx * [(1-h[k+shift])*1_{valid} - h[k-shift]*1_{valid}]
    ncs = n-1
    a = np.zeros(2*n-1)  # a[s] = g_s[i] - g_s[j]
    for s in range(2*n-1):
        shift = s - ncs
        # g_s[i]:
        vi = 0.0
        ip = i + shift
        if 0 <= ip < n:
            vi += dx * (1 - h[ip])
        im = i - shift
        if 0 <= im < n:
            vi += dx * (-h[im])
        # g_s[j]:
        vj = 0.0
        jp = j + shift
        if 0 <= jp < n:
            vj += dx * (1 - h[jp])
        jm = j - shift
        if 0 <= jm < n:
            vj += dx * (-h[jm])
        a[s] = vi - vj
    
    # C(t) = max_s (corr[s] + t * a[s])
    # Feasible range: h[i]+t ∈ [0,1] and h[j]-t ∈ [0,1]
    t_min = max(-h[i], -(1-h[j]))  # h[i]+t>=0 → t>=-h[i]; h[j]-t<=1 → t>=h[j]-1
    t_max = min(1-h[i], h[j])        # h[i]+t<=1 → t<=1-h[i]; h[j]-t>=0 → t<=h[j]

    # Find minimum by checking: boundaries and where the max "switches"
    # For convex PL function, min is where subgradient contains 0
    # i.e., where max_s (a[s] for s achieving max) changes sign
    
    # Simple approach: ternary search (works for convex functions)
    # Or: find the t where the sorted lines cross

    # Since it's convex PL, the minimum is at a kink or boundary

        if c1 < c2:

        else:
            lo = m1
        if hi - lo < 1e-10:
            break
    t_star = (lo + hi) / 2
    c_star = np.max(corr + t_star * a)
    
    return t_star, c_star
```

```python

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    """Compute G[s, j] = dR_s/dh[j] for all s, j."""
    dx = 2.0 / n
    n_full = 2 * n - 1
    ncs = n - 1
    G = np.zeros((n_full, n))
    for s in range(n_full):
        shift = s - ncs
        if shift == 0:
            G[s, :] = dx * (1 - 2*h)
        elif shift > 0:
            G[s, :n-shift] += dx * (1 - h[shift:])
            G[s, shift:] += dx * (-h[:n-shift])
        else:
            neg = -shift
            G[s, neg:] += dx * (1 - h[:n-neg])
            G[s, :n-neg] += dx * (-h[neg:])
    return G

def exact_pair_search(h, n, corr, G, deadline, max_sweeps=50):
    """Exact pair search with optimal step size finding."""
    best_c = float(np.max(corr))
    n_full = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c, corr, G
            for j in range(i+1, n):
                # Compute a[s] = G[s,i] - G[s,j]
                a = G[:, i] - G[:, j]
                # Check if there's a possible improvement
                # C(t) = max_s(corr[s] + t*a[s])
                # At t=0, C(0) = max(corr) = best_c
                # Moving in direction d: C changes by t*(a[s*]) where s* is argmax
                # If all a[s] >= 0 for s achieving max, moving in -direction might help
                # If all a[s] <= 0 for s achieving max, moving in +direction might help
                
                # Feasible range
                t_min = max(-h[i], h[j] - 1.0)
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-12:
                    continue
                
                # Quick check: evaluate C at boundaries
```

```python
    dx = 2.0 / n

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    dx = 2.0 / n
    n_full = 2 * n - 1
    ncs = n - 1
    G = np.zeros((n_full, n))
    for s in range(n_full):
        shift = s - ncs
        if shift == 0:
            G[s, :] = dx * (1.0 - 2.0 * h)
        elif shift > 0:
            G[s, :n-shift] = dx * (1.0 - h[shift:])
            G[s, shift:] = G[s, shift:] + dx * (-h[:n-shift])
        else:
            neg = -shift
            G[s, neg:] = G[s, neg:] + dx * (1.0 - h[:n-neg])
            G[s, :n-neg] = G[s, :n-neg] + dx * (-h[neg:])
    return G

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    return float(m + (mx_sh + np.log(np.sum(np.exp(sh - mx_sh)))) / alpha)

def obj_grad_smooth(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)

    exp_sh = np.exp(sh - mx_sh)

    G = compute_grad_matrix(x, n)
    return G.T @ p  # shape (n,)

    return np.ones(n)
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    dx = 2.0 / n
    n_full = 2 * n - 1
    ncs = n - 1
    G = np.zeros((n_full, n))
    for s in range(n_full):
        shift = s - ncs
        if shift == 0:
            G[s, :] = dx * (1.0 - 2.0 * h)
        elif shift > 0:
            G[s, :n-shift] += dx * (1.0 - h[shift:])
            G[s, shift:] += dx * (-h[:n-shift])
        else:
            neg = -shift
            G[s, neg:] += dx * (1.0 - h[:n-neg])
            G[s, :n-neg] += dx * (-h[neg:])
    return G

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    return float(m + (mx_sh + np.log(np.sum(np.exp(sh - mx_sh)))) / alpha)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, deadline=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    if deadline is not None:
        remaining = deadline - time.time()
        if remaining < 5:
            return h.copy()
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=None,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def exact_pair_search(h, n, deadline, max_sweeps=100):
    corr = compute_corr(h, n)
    G = compute_grad_matrix(h, n)
    best_c = float(np.max(corr))
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        n_imp = 0
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                t_min = max(-h[i], h[j] - 1.0)
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-12:
                    continue
                
                a = G[:, i] - G[:, j]
                
                # Fast path
                if a.max() <= 1e-15:
                    t_star = t_max
                elif a.min() >= -1e-15:
                    t_star = t_min
                else:
                    lo, hi = t_min, t_max
                    for _ in range(35):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        c1 = np.max(corr + m1 * a)
                        c2 = np.max(corr + m2 * a)
                        if c1 < c2:
                            hi = m2
                        else:
                            lo = m1
                        if hi - lo < 1e-11:
                            break
                    t_star = (lo + hi) / 2.0
                
                c_star = float(np.max(corr + t_star * a))
                
                if c_star < best_c - 1e-13:
                    h[i] += t_star
                    h[j] -= t_star
                    h[i] = min(max(h[i], 0.0), 1.0)
                    h[j] = min(max(h[j], 0.0), 1.0)
                    best_c = c_star
                    corr = compute_corr(h, n)
                    G = compute_grad_matrix(h, n)
                    improved = True
                    n_imp += 1
        
        if not improved:
            break
    
    return h, best_c
```

Actually, for a convex piecewise linear function, the derivative (where it exists) is the slope of the active line at that point. The minimum is where the slope changes from negative to positive.

I can binary search on t:
- Compute the "subgradient" of C(t): it's the interval [min{a[s] : s ∈ S_max(t)}, max{a[s] : s ∈ S_max(t)}] where S_max(t) = {s : corr[s] + t*a[s] = C(t)}
- If the entire subgradient is > 0, the minimum is to the left
- If the entire subgradient is < 0, the minimum is to the right
- If 0 is in the subgradient, we've found the minimum

But computing S_max(t) requires finding all s that achieve the max, which is O(K). And this is the same cost as evaluating C(t). So binary search doesn't save asymptotically, but saves a constant factor (binary search converges faster than ternary for piecewise linear functions).
