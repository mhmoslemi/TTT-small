import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    g = compute_g_full(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    ms = shifted.max()
    return m + (ms + np.log(np.exp(shifted - ms).sum())) / alpha

def smooth_max_and_grad(h, n, alpha):
    g = compute_g_full(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    ms = shifted.max()
    exp_vals = np.exp(shifted - ms)
    w = exp_vals / exp_vals.sum()
    val = m + (ms + np.log(exp_vals.sum())) / alpha
    
    # Compute gradient analytically
    nn = n
    k = np.arange(2*nn-1) - (nn-1)
    grad = np.zeros(nn)
    # For efficiency, use the formula:
    # grad[j] = (2/n) * [sum_p w[p]*(1-h[j+k_p])*I(0<=j+k_p<n) - sum_p w[p]*h[j-k_p]*I(0<=j-k_p<n)]
    
    # Precompute valid indices
    for j in range(nn):
        idx1 = (j + k)
        valid1 = (idx1 >= 0) & (idx1 < nn)
        idx2 = (j - k)
        valid2 = (idx2 >= 0) & (idx2 < nn)
        grad[j] = (2.0/nn) * (np.dot(w[valid1], 1-h[idx1[valid1]]) - np.dot(w[valid2], h[idx2[valid2]]))
    
    return val, grad

def slsqp_optimize(h, n, alpha, maxiter, ftol, deadline):
    def obj(x):
        return smooth_max_value(x, n, alpha)
    
    def obj_grad(x):
        _, g = smooth_max_and_grad(x, n, alpha)
        return g
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    
    try:
        res = minimize(obj_grad, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_gradient(h, n, alpha, deadline, max_iter=10000, lr_init=0.01):
    """Projected gradient with backtracking line search on the smooth max."""
    best_h = h.copy()
    best_val = smooth_max_value(h, n, alpha)
    lr = lr_init
    t = np.zeros(n)  # momentum
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, grad = smooth_max_and_grad(h, n, alpha)
        
        # Backtracking line search
        step = lr
        found = False
        for _ in range(50):
            h_new = project_feasible(h - step * grad, n)
            new_val = smooth_max_value(h_new, n, alpha)
            if new_val < val - 1e-12 or new_val < best_val - 1e-12:
                h = h_new
                found = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        if not found:
            lr *= 0.8
            if lr < 1e-15:
                break
            continue
        
        t = compute_c5(h, n)
        if t < best_val:
            best_val = t
            best_h = h.copy()
    
    return best_h, best_val
