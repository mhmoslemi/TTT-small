def compute_sensitivities(h, n):
    """Compute s[i, p] = d(c_p)/d(h_i) for all i, p."""
    m = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    m = m[None, :]  # shape (1, 2n-1)
    idx = np.arange(n)  # shape (n,)
    
    # i - m_p: shape (n, 2n-1)
    im = idx[:, None] - m  # shape (n, 2n-1)
    ip = idx[:, None] + m  # shape (n, 2n-1)
    
    s = np.zeros((n, 2*n-1))

    s[valid_im] = 1 - h[np.clip(im, 0, n-1)[valid_im]]
    
    # For valid_ip: s[i, p] += -h[i + m_p]
    s[valid_ip] -= h[np.clip(ip, 0, n-1)[valid_ip]]

Let me also add a "two-coordinate" optimization: for a given pair (i,j), find the optimal d (not just try fixed d values). This is a 1D optimization: minimize max_k (c_all[k] + d*(s[i,k]-s[j,k])) subject to 0 ≤ h[i]+d ≤ 1 and 0 ≤ h[j]-d ≤ 1.

The function g(d) = max_k (c_all[k] + d*(s[i,k]-s[j,k])) is the max of linear functions of d, so it's convex in d. We can find its minimum using ternary search or by finding where the subgradient crosses zero.

Actually, since g(d) is piecewise linear and convex, its minimum is at a point where the slope changes sign, which is at a "kink" point. The kink points are where two of the linear functions c_all[k] + d*(s[i,k]-s[j,k]) are equal.

For a 1D convex piecewise-linear function, we can use a simple bisection on the subgradient:
- g'(d) = (s[i,k*] - s[j,k*]) where k* is an argmax
- If g'(d) < 0, increase d; if g'(d) > 0, decrease d

Or even simpler: since the domain of d is bounded (by the box constraints), and g is convex, we can just try a few values of d (e.g., 20 values in the feasible range) and pick the best.

One more thought: the sensitivity-based search only gives the EXACT change in the correlation if we change only h[i] and h[j] (and the sum constraint is maintained). But if we also need to project to maintain the sum constraint, the analysis breaks down.

Solution: Instead of two-coordinate transfers, use the transfer (h[i] += d, h[j] -= d) which automatically preserves the sum. Then we don't need projection (as long as 0 ≤ h[i]+d ≤ 1 and 0 ≤ h[j]-d ≤ 1).

This is the key: transfers preserve feasibility (box + sum) without projection!

So the algorithm is:
1. Start with feasible h
2. Transfer search: for pairs (i,j) and step d, if h[i]+d ∈ [0,1] and h[j]-d ∈ [0,1], compute new correlation and accept if it improves
3. No projection needed!

This is clean and fast.

For the SLSQP part, I still need projection (since SLSQP might violate constraints). But for the local search, no projection is needed.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def obj_fn(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new, n) < compute_c5(h, n):
                return h_new
    except Exception:
        pass
    return h.copy()

def compute_sensitivities(h, n):
    """Compute s[i, p] = d(c_p)/d(h_i) for all i in [0,n), p in [0, 2n-2)."""
    m = np.arange(2 * n - 1) - (n - 1)  # lags, shape (2n-1,)
    idx = np.arange(n)
    
    im = idx[:, None] - m[None, :]  # shape (n, 2n-1): i - m_p
    ip = idx[:, None] + m[None, :]  # shape (n, 2n-1): i + m_p
    
    s = np.zeros((n, 2 * n - 1))
    
    valid_im = (im >= 0) & (im < n)
    valid_ip = (ip >= 0) & (ip < n)
    
    s[valid_im] = 1.0 - h[np.maximum(np.minimum(im, n-1), 0)[valid_im]]
    s[valid_ip] -= h[np.maximum(np.minimum(ip, n-1), 0)[valid_ip]]
    
    return s

def fast_local_search(h, n, deadline, max_passes=100, radius=None):
    """Fast local search using precomputed sensitivities.
    Uses transfers (h[i]+=d, h[j]-=d) which preserve feasibility."""
    if radius is None:
        radius = min(n // 2, 100)
    
    h = h.copy()
    c_all = compute_c5_vec(h, n)
    best_c = float(np.max(c_all))
    
    # Sensitivity matrix
    s = compute_sensitivities(h, n)
    
    # Deltas to try (adaptive: start large, go small)
    base_deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d0 in base_deltas:
        if time.time() > deadline:
            break
        
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                
                # Determine valid j range
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    
                    # Direction 1: h[i] += d, h[j] -= d
                    d = d0
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        new_c = c_all + d * (s[i] - s[j])
                        new_max = float(np.max(new_c))
                        if new_max < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c_all = new_c
                            s = compute_sensitivities(h, n)
                            best_c = new_max
                            improved = True
                            continue
                    
                    # Direction 2: h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        new_c = c_all - d * (s[i] - s[j])
                        new_max = float(np.max(new_c))
                        if new_max < best_c - 1e-14:
                            h[i] -= d
                            h[j] += d
                            c_all = new_c
                            s = compute_sensitivities(h, n)
                            best_c = new_max
                            improved = True
            
            if not improved:
                break
    
    return h, best_c
