    no_improve = 0
    best_no_improve_reset = 0

    while time.time() < deadline and it < max_iter:
        progress = min(1.0, it / 4000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        c_all = compute_all_overlaps(h, n)
        m = c_all.max()
        exp_c = np.exp(alpha * (c_all - m))
        p = exp_c / exp_c.sum()

        grad = compute_gradient(h, n, p)
        grad_centered = grad - grad.mean()

        h_new = h - lr * grad_centered
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h_best = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            if no_improve > 500:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-5:
                    lr = lr_init * 0.3
                    # Perturb to escape
                    h = project_feasible(h + 0.01 * np.random.randn(n), n)
                    no_improve = 0

        it += 1

    return h_best, best_c

def slsqp_polish(h, n, alpha, maxiter=1000):
    def obj(x):
        c = compute_all_overlaps(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    def sum_constraint(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def local_pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try moving mass from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def structured_init(n, rng):
    """Create structured initializations."""
    choices = []
    # Sine wave
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 5]:
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        choices.append(h.copy())
    # Square wave
    for period in [4, 6, 8, 12]:
        h = np.zeros(n)
        for i in range(n):
            if (i % period) < period // 2:
                h[i] = 1.0
        h = project_feasible(h, n)
        choices.append(h.copy())
    # Random
    h = rng.uniform(0, 1, n)
    h = project_feasible(h, n)
    choices.append(h)
    return choices

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        best_h = project_feasible(h_init.copy(), n_init)
        best_c = compute_c5(best_h, n_init)
    except Exception:
        best_h = np.full(200, 0.5)
        best_c = compute_c5(best_h, 200)
        n_init = 200

    # Phase 1: PGD on initial resolution
    alloc1 = min(80, (deadline - t0) * 0.15)
    if alloc1 > 20:
        h1, c1 = pgd_optimize(h_init, n_init, t0 + alloc1)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # SLSQP polish on initial
    if time.time() < deadline - 200:
        h_s = slsqp_polish(best_h, len(best_h), alpha=100000, maxiter=500)
        c_s = compute_c5(h_s, len(h_s))
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 2: Try higher resolutions
    resolutions = []
    n_best = len(best_h)
    for n_try in [n_best, 350, 500, 700]:
        if n_try not in resolutions and n_try != n_best:
            resolutions.append(n_try)
    resolutions = list(dict.fromkeys(resolutions))  # deduplicate, keep order

    for n_new in resolutions:
        if time.time() > deadline - 100:
            break
        if n_new <= n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)

        alloc = min(100, (deadline - time.time() - 50) * 0.25)
        if alloc < 30:
            break

        h_new, c_new = pgd_optimize(h_interp, n_new, time.time() + alloc,
                                     alpha_start=100.0, alpha_end=150000.0,
                                     lr_init=0.3)
        if c_new < best_c:
            best_c, best_h = c_new, h_new

        # SLSQP polish at this resolution
        if time.time() < deadline - 60:
            h_s = slsqp_polish(best_h, len(best_h), alpha=50000, maxiter=300)
            c_s = compute_c5(h_s, len(h_s))
            if c_s < best_c:
                best_c, best_h = c_s, h_s

    # Phase 3: Multi-restart with different initializations
    n_best = len(best_h)
    remaining = deadline - time.time() - 40
    if remaining > 60:
        n_restarts = max(3, int(remaining / 35))
        n_restarts = min(n_restarts, 8)
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                # Perturb current best
                hp = best_h + rng.normal(0, 0.03, n_best)
            elif r == 1:
                hp = best_h + rng.normal(0, 0.06, n_best)
            else:
                inits = structured_init(n_best, rng)
                hp = inits[r % len(inits)]
            hp = project_feasible(hp, n_best)
            alloc = min(35, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            hr, cr = pgd_optimize(hp, n_best, time.time() + alloc,
                                   alpha_start=50.0, alpha_end=100000.0)
            if cr < best_c:
                best_c, best_h = cr, hr

    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 15:
        n = len(best_h)
        for a in [200000, 500000]:
            if time.time() > deadline - 8:
                break
            h_s = slsqp_polish(best_h, n, alpha=a, maxiter=500)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c, best_h = c_s, h_s

    # Phase 5: Local pair search polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 300:
            best_h, best_c = local_pair_search(best_h, n, deadline - 2,
                                                [0.005, 0.002, 0.001, 0.0005],
                                                radius=80)
        else:
            best_h, best_c = local_pair_search(best_h, n, deadline - 2,
                                                [0.005, 0.002, 0.001],
                                                radius=50)

    # Final PGD with very high alpha for last-mile improvement
    if time.time() < deadline - 3:
        n = len(best_h)
        h_final, c_final = pgd_optimize(best_h, n, deadline - 1,
                                         alpha_start=50000.0, alpha_end=500000.0,
                                         lr_init=0.1)
        if c_final < best_c:
            best_c, best_h = c_final, h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
