            h_new = _build_h_sym(v_new, n)
            c_new = _true_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_v = v_new
        except Exception:
            pass
    return best_v, best_c

def _slsqp_sym(v, n, deadline, alpha=100000, maxiter=2000):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(_obj_sym, v, jac=_grad_sym, args=(n, alpha),
                      method='SLSQP', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        v_new = np.clip(res.x, 0.0, 1.0)
        h_new = _build_h_sym(v_new, n)
        c_new = _true_c5(h_new, n)
        return v_new, c_new
    except Exception:
        return v.copy(), _true_c5(_build_h_sym(v, n), n)

def _local_search_sym(v, n, deadline, steps=None):
    if steps is None:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    m = n // 2
    best_v = v.copy()
    best_c = _true_c5(_build_h_sym(best_v, n), n)
    for s in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(m):
                if time.time() > deadline:
                    break
                if best_v[i] + s <= 1.0:
                    old = best_v[i]
                    best_v[i] = old + s
                    c = _true_c5(_build_h_sym(best_v, n), n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_v[i] = old
                if best_v[i] - s >= 0.0:
                    old = best_v[i]
                    best_v[i] = old - s
                    c = _true_c5(_build_h_sym(best_v, n), n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_v[i] = old
    return best_v, best_c

def _local_search_pairs(v, n, deadline, steps=None, radius=50):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001]
    m = n // 2
    best_v = v.copy()
    best_c = _true_c5(_build_h_sym(best_v, n), n)
    for s in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(m):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(m, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_v[i] + s <= 1.0 and best_v[j] - s >= 0.0:
                        oi, oj = best_v[i], best_v[j]
                        best_v[i] = oi + s
                        best_v[j] = oj - s
                        c = _true_c5(_build_h_sym(best_v, n), n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_v[i], best_v[j] = oi, oj
                    if best_v[i] - s >= 0.0 and best_v[j] + s <= 1.0:
                        oi, oj = best_v[i], best_v[j]
                        best_v[i] = oi - s
                        best_v[j] = oj + s
                        c = _true_c5(_build_h_sym(best_v, n), n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_v[i], best_v[j] = oi, oj
    return best_v, best_c

def _sym_from_h(h, n):
    m = n // 2
    v = np.zeros(m)
    for i in range(m):
        v[i] = (h[i] + (1.0 - h[n - 1 - i])) / 2.0
    return np.clip(v, 0.0, 1.0)

def _interpolate_sym(h, n_old, n_new):
    m_old = n_old // 2
    v_old = _sym_from_h(h, n_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    m_new = n_new // 2
    v_new = np.zeros(m_new)
    for i in range(m_new):
        v_new[i] = (h_interp[i] + (1.0 - h_interp[n_new - 1 - i])) / 2.0
    return np.clip(v_new, 0.0, 1.0)

def _run_nonsymmetric(h, n, deadline, alphas):
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = _true_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 3:
            break
        max_it = max(30, int(min(remaining * 20, 3000)))
        try:
            res = minimize(_obj_nonsym, best_h, jac=_grad_nonsym, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_it, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_it * 5, 'maxls': 30})
            h_new = _project(res.x, n)
            c_new = _true_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def _obj_nonsym(x, n, alpha):
    c = _correlate_full(x, n)
    val = _smooth_max_val(c, alpha)
    val += 1e-4 * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_nonsym(x, n, alpha):
    c = _correlate_full(x, n)
    w = _smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        lag = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if lag >= 0:
            nv = n - lag
            grad[:nv] += wm * (1.0 - x[lag:])
            grad[lag:] += wm * (-x[:nv])
        else:
            la = -lag
            nv = n - la
            grad[la:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[la:])
    grad += 2e-4 * (x.sum() - n / 2.0)
    return grad

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < target:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h += (target - s) / rs * room
        else:
            ps = h.sum()
            if ps < 1e-15:
                break
            h -= (s - target) / ps * h
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(300, 0.5)
        n_parent = 300

    best_h = h_parent.copy()
    best_c = _true_c5(best_h, n_parent)
    best_n = n_parent

    # Phase 1: Symmetric optimization at n=300 (from parent)
    n1 = n_parent if n_parent % 2 == 0 else n_parent + 1
    if n1 != n_parent:
        x_old = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n1)
        h_interp = np.interp(x_new, x_old, h_parent)
    else:
        h_interp = h_parent
    v_sym = _sym_from_h(h_interp, n1)
    t1 = min(t0 + 150, deadline - 500)
    v_sym, c1 = _lbfgs_sym(v_sym, n1, t1, [500, 2000, 10000, 50000])
    if time.time() < deadline - 400:
        v_sym, c1 = _slsqp_sym(v_sym, n1, min(time.time() + 40, deadline - 400), alpha=200000)
    h1 = _build_h_sym(v_sym, n1)
    if c1 < best_c:
        best_c = c1
        best_h = h1
        best_n = n1

    # Phase 2: Symmetric optimization at n=400
    n2 = 400
    if time.time() < deadline - 300:
        v2 = _interpolate_sym(best_h, best_n, n2)
        t2 = min(time.time() + 180, deadline - 250)
        v2, c2 = _lbfgs_sym(v2, n2, t2, [500, 2000, 10000, 50000, 200000])
        if time.time() < deadline - 200:
            v2, c2 = _slsqp_sym(v2, n2, min(time.time() + 50, deadline - 200), alpha=500000)
        h2 = _build_h_sym(v2, n2)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2

    # Phase 3: Symmetric optimization at n=500
    n3 = 500
    if time.time() < deadline - 200:
        v3 = _interpolate_sym(best_h, best_n, n3)
        t3 = min(time.time() + 200, deadline - 150)
        v3, c3 = _lbfgs_sym(v3, n3, t3, [1000, 5000, 20000, 100000])
        if time.time() < deadline - 100:
            v3, c3 = _slsqp_sym(v3, n3, min(time.time() + 60, deadline - 100), alpha=500000)
        h3 = _build_h_sym(v3, n3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n3

    # Phase 4: Local search on best symmetric solution
    if best_n % 2 == 0 and time.time() < deadline - 60:
        m_best = best_n // 2
        v_best = _sym_from_h(best_h, best_n)
        if time.time() < deadline - 30:
            v_best, c_ls = _local_search_sym(v_best, best_n, min(time.time() + 40, deadline - 30),
                                             steps=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
            if c_ls < best_c:
                best_c = c_ls
                best_h = _build_h_sym(v_best, best_n)
        if time.time() < deadline - 10:
            v_best = _sym_from_h(best_h, best_n)
            v_best, c_ls2 = _local_search_pairs(v_best, best_n, deadline - 3,
                                                steps=[0.005, 0.002, 0.001], radius=30)
            if c_ls2 < best_c:
                best_c = c_ls2
                best_h = _build_h_sym(v_best, best_n)

    # Phase 5: Try non-symmetric refinement
    if time.time() < deadline - 30:
        h_ns = _project(best_h, best_n)
        t5 = min(time.time() + 40, deadline - 10)
        h_ns, c_ns = _run_nonsymmetric(h_ns, best_n, t5, [50000, 200000])
        if c_ns < best_c:
            best_c = c_ns
            best_h = h_ns

    # Phase 6: Try sine-based initialization at different frequencies
    if time.time() < deadline - 40:
        for freq in [1, 2, 3]:
            if time.time() > deadline - 20:
                break
            n_try = 300
            x = np.linspace(0, 2, n_try)
            h_sine = 0.5 + 0.25 * np.sin(np.pi * freq * x)
            h_sine = np.clip(h_sine, 0.0, 1.0)
            h_sine = _project(h_sine, n_try)
            v_sine = _sym_from_h(h_sine, n_try)
            alloc = min(20, deadline - time.time() - 10)
            if alloc > 5:
                v_sine, c_sine = _lbfgs_sym(v_sine, n_try, time.time() + alloc, [2000, 10000, 50000])
                h_sine_full = _build_h_sym(v_sine, n_try)
                if c_sine < best_c:
                    best_c = c_sine
                    best_h = h_sine_full
                    best_n = n_try

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_n = len(best_h)
    if best_n % 2 == 0:
        # Optionally re-verify symmetry is still good
        pass
    best_c = _true_c5(best_h, best_n)

    return best_h, best_c, best_n
