    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)

        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_corr(h, k_idx, n):
    """Gradient of corr[k_idx] w.r.t. h. Vectorized."""
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    
    # i_min = max(0, shift), i_max = min(n-1, n-1-shift)
    i_min = max(0, shift)
    i_max = min(n - 1, n - 1 - shift)
    
    if i_min > i_max:
        return grad
    
    # Term 1: (1-h[j-shift]) for j in [i_min, i_max]
    # j = i, so for j in [i_min, i_max]: grad[j] += (1-h[j-shift])
    j_range = np.arange(i_min, i_max + 1)
    idx1 = j_range - shift
    # idx1 should be in [0, n-1] by construction
    grad[j_range] += (1.0 - h[idx1])
    
    # Term 2: -h[j+shift] for j+shift in [i_min, i_max]
    # j = i - shift, so j+shift = i, i in [i_min, i_max]
    # j = i - shift, range: [i_min-shift, i_max-shift]
    # But we need j in [0, n-1]
    j2_min = max(0, i_min - shift)
    j2_max = min(n - 1, i_max - shift)
    if j2_min <= j2_max:
        j2_range = np.arange(j2_min, j2_max + 1)
        grad[j2_range] -= h[j2_range + shift]

import numpy as np
from scipy.optimize import minimize, Bounds
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_corr(h, k_idx, n):
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    i_min = max(0, shift)
    i_max = min(n - 1, n - 1 - shift)
    if i_min > i_max:
        return grad
    j_range = np.arange(i_min, i_max + 1)
    idx1 = j_range - shift
    grad[j_range] += (1.0 - h[idx1])
    j2_min = max(0, i_min - shift)
    j2_max = min(n - 1, i_max - shift)
    if j2_min <= j2_max:
        j2_range = np.arange(j2_min, j2_max + 1)
        grad[j2_range] -= h[j2_range + shift]
    return grad

    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx


def obj_smooth_grad_fn(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-12:
            grad += weights[k] * compute_grad_corr(x, k, n)
    return dx * grad

    return x.sum() - target_sum


def con_eq_jac(x, target_sum):
    return np.ones(len(x))

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def do_slsqp(h0, n, target_sum, beta, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq_fn, 'jac': con_eq_jac, 'args': (target_sum,)}]

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0

        grad[j2_range] -= h[j2_range + shift]
    return grad


def obj_smooth_fn(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx


def obj_smooth_grad_fn(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w > 1e-12:
            grad += w * compute_grad_corr(x, k, n)
    return dx * grad


def con_eq_fn(x, target_sum):
    return x.sum() - target_sum


def con_eq_jac(x, target_sum):
    return np.ones(len(x))


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def do_slsqp(h0, n, target_sum, beta, maxiter=5000, ftol=1e-14):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq_fn, 'jac': con_eq_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth_fn, h0, args=(dx, beta),
                       jac=obj_smooth_grad_fn,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0


def do_cutting_plane(h_star, n, target_sum):
    corr = compute_corr(h_star)
    n_corr = len(corr)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for k in range(n_corr):
        grad = compute_grad_corr(h_star, k, n)
        constraints.append(t_var >= corr[k] + grad @ (h_var - h_star))
    problem = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            problem.solve(solver=solver)
            if h_var.value is not None:
                h_val = np.array(h_var.value).flatten()
                return project_feasible(h_val, target_sum)
        except Exception:
            continue
    return None


def subgradient_polish(h, n, target_sum, max_iters, time_fn):
    h = h.copy()
    for it in range(max_iters):
        if time_fn() < 10:
            break
        corr = compute_corr(h)
        k_worst = int(np.argmax(corr))
        grad = compute_grad_corr(h, k_worst, n)
        lr = 0.004 / (1.0 + it * 0.0003)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    # Initial values
    if 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float).copy()
        except (NameError, TypeError, ValueError):
            h_init = np.full(70, 0.5)
    else:
        h_init = np.full(70, 0.5)

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Main optimization at initial resolution
    n = n_init
    target_sum = n / 2.0
    h_cur = project_feasible(h_init.copy(), target_sum)
    c5_cur = compute_c5(h_cur)

    # Phase 1: SLSQP continuation with analytical gradient
    betas = [5, 20, 50, 100, 300, 1000, 3000, 10000]
    for beta in betas:
        if tl() < 30:
            break
        for trial in range(3):
            if tl() < 15:
                break
            x0 = h_cur.copy()
            if trial > 0:
                scale = 0.015 / trial
                x0 = x0 + np.random.randn(n) * scale
                x0 = project_feasible(x0, target_sum)
            h_new = do_slsqp(x0, n, target_sum, beta, maxiter=3000, ftol=1e-14)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-13:
                h_cur = h_new
                c5_cur = c5_new

    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Phase 2: Cutting plane refinement
    if tl() > 50:
        h_cp = h_cur.copy()
        stagnant = 0
        for it in range(200):
            if tl() < 15:
                break
            h_new = do_cutting_plane(h_cp, n, target_sum)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_cp) - 1e-11:
                    h_cp = h_new
                    stagnant = 0
                else:
                    stagnant += 1
                    if stagnant >= 8:
                        h_cp = h_cp + np.random.randn(n) * 0.008
                        h_cp = project_feasible(h_cp, target_sum)
                        stagnant = 0
        c5_cp = compute_c5(h_cp)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()

    # Phase 3: Subgradient polish
    if tl() > 30:
        h_sg = best_h.copy()
        n_sg = len(h_sg)
        h_sg = subgradient_polish(h_sg, n_sg, n_sg / 2.0, 80000, tl)
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h
