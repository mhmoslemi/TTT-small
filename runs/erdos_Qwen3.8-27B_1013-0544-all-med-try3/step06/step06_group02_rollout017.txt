    """Find optimal value of h[i] with other coordinates fixed."""
    # c_k = sum_j h[j]*(1-h[j+k])
    # For coordinate i, c_k = a_k * h[i] + b_k
    # where a_k = coefficient of h[i] in c_k
    # and b_k = c_k - a_k * h[i] (with current h[i])
    
    # Compute current c_k values
    c = np.correlate(h, 1-h, mode='full')
    
    # For each shift k, find the coefficient of h[i]
    # c_k = sum_j h[j] - h[j]*h[j+k]
    # dh[i] appears in:
    #   - h[j] term: when j = i, contributes +1 to c_k for all valid k
    #   - h[j]*h[j+k] term: when j = i (contributes -h[i+k]) or j+k = i (contributes -h[j])
    
    n_full = 2 * n - 1
    a = np.zeros(n_full)
    
    for m in range(n_full):
        k = m - (n - 1)
        if k >= 0:
            # c_k = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
            # h[i] appears if:
            #   i in [0, n-1-k]: contributes (1-h[i+k])
            #   i-k in [0, n-1-k], i.e., i in [k, n-1]: contributes -h[i-k]
            coeff = 0.0
            if 0 <= i <= n-1-k:
                coeff += (1 - h[i+k])
            if k <= i <= n-1:
                coeff -= h[i-k]
            a[m] = coeff
        else:
            kk = -k
            # c_k = sum_{j=kk}^{n-1} h[j]*(1-h[j-kk])
            # h[i] appears if:
            #   i in [kk, n-1]: contributes (1-h[i-kk])
            #   i+kk in [kk, n-1], i.e., i in [0, n-1-kk]: contributes -h[i+kk]
            coeff = 0.0
            if kk <= i <= n-1:
                coeff += (1 - h[i-kk])
            if 0 <= i <= n-1-kk:
                coeff -= h[i+kk]
            a[m] = coeff
    
    # b[m] = c[m] - a[m] * h[i]
    b = c - a * h[i]

    f0 = np.max(b)  # at t=0

    # Find where the max changes

    # Minimum where 0 is in the subgradient

        if min_grad > 0:

            hi = mid
        elif max_grad < 0:
            # f is decreasing at mid, go right
            lo = mid
        else:

            lo = hi = mid
            break

from scipy.optimize import minimize
import time

def compute_c5_values(h, n):
    """Return the full correlation vector (multiplied by dx)."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_values(h, n)))

def project_feasible(h, n):
    """Project onto {x: 0<=x<=1, sum(x)=n/2} using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max objective."""
    c = compute_c5_values(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w = w / w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n_full = 2 * n - 1
    
    for m in range(n_full):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    alphas_mus = [
        (100, 1.0), (500, 5.0), (2000, 20.0),
        (10000, 100.0), (50000, 500.0), (200000, 2000.0),
        (1000000, 10000.0), (5000000, 50000.0)
    ]
    
    for alpha, mu in alphas_mus:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 2:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxfun': max_iters * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_values(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_values(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for m in range(n_full):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_values(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=300):
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas_mus = [
        (50, 0.5), (200, 2.0), (1000, 10.0), (5000, 50.0),
        (20000, 200.0), (100000, 1000.0), (500000, 5000.0),
        (2000000, 20000.0)
    ]
    for alpha, mu in alphas_mus:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 3:
            break
        iters = min(max_iters, int(remaining * 3))
        if iters < 10:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': iters, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxfun': iters * 20,
                                  'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_h = h_new
                best_c = c_new
        except Exception:
            pass
    return best_h, best_c

def compute_pair_coeffs(h, n, i, j):
    n_full = 2 * n - 1
    a_i = np.zeros(n_full)
    a_j = np.zeros(n_full)
    for m in range(n_full):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            if i < nv:
                a_i[m] += (1.0 - h[i + k])
            if i >= k:
                a_i[m] -= h[i - k]
            if j < nv:
                a_j[m] += (1.0 - h[j + k])
            if j >= k:
                a_j[m] -= h[j - k]
        else:
            kk = -k
            nv = n - kk
            if i >= kk:
                a_i[m] += (1.0 - h[i - kk])
            if i < nv:
                a_i[m] -= h[i + kk]
            if j >= kk:
                a_j[m] += (1.0 - h[j - kk])
            if j < nv:
                a_j[m] -= h[j + kk]
    return a_i - a_j

def find_optimal_delta(a_diff, c_base, lo, hi):
    if lo >= hi - 1e-15:
        return 0.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a_diff * mid + c_base
        mx = vals.max()
        tol = mx * 1e-12 + 1e-14
        active = np.abs(vals - mx) < tol
        if active.any():
            sg_min = a_diff[active].min()
            sg_max = a_diff[active].max()
            if sg_min > 1e-12:
                hi = mid
            elif sg_max < -1e-12:
                lo = mid
            else:
                break
        else:
            idx = np.argmax(vals)
            if a_diff[idx] > 0:
                hi = mid
            else:
                lo = mid
    return (lo + hi) / 2.0

def exact_pair_move(h, n, i, j):
    c_base = np.correlate(h, 1.0 - h, mode='full')
    a_diff = compute_pair_coeffs(h, n, i, j)
    lo = max(-h[i], -h[j]) + 1e-15
    hi = min(1.0 - h[i], h[j]) - 1e-15
    if lo >= hi:
        return h, compute_c5(h, n)
    delta = find_optimal_delta(a_diff, c_base, lo, hi)
    if abs(delta) < 1e-15:
        return h, compute_c5(h, n)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    c_new = compute_c5(h_new, n)
    return h_new, c_new

def gradient_guided_pair_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        g = grad_smooth_max(best_h, n, alpha=5000)
        order = np.argsort(-np.abs(g))
        improved = False
        n_pairs = min(n, 40)
        for idx in range(n_pairs):
            if time.time() > deadline:
                return best_h, best_c
            i = order[idx]
            partner_candidates = []
            for p in range(min(6, n)):
                j = order[(idx + p * 7 + 3) % n]
                if j != i:
                    partner_candidates.append(j)
            j = (i + n // 2) % n
            if j not in partner_candidates:
                partner_candidates.append(j)
            for j in partner_candidates:
                if time.time() > deadline:
                    return best_h, best_c
                h_try, c_try = exact_pair_move(best_h, n, i, j)
                if c_try < best_c - 1e-14:
                    best_h = h_try
                    best_c = c_try
                    improved = True
        if not improved:
            break
    return best_h, best_c

def full_pair_sweep(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_c
                h_try, c_try = exact_pair_move(best_h, n, i, j)
                if c_try < best_c - 1e-14:
                    best_h = h_try
                    best_c = c_try
                    improved = True
        if not improved:
            break
    return best_h, best_c

def random_perturb_reoptimize(h, n, rng, deadline, n_perturb=5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for _ in range(n_perturb):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.randn(n) * 0.02
        h_pert = project_feasible(h_pert, n)
        h_opt, c_opt = lbfgs_optimize(h_pert, n, time.time() + 30, max_iters=100)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)
