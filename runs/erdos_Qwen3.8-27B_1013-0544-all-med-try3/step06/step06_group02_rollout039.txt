    target = n / 2.0

        if abs(diff) < 1e-13:
            break

            room = 1.0 - h

        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h

    c = np.correlate(h, 1 - h, mode='full')

        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def active_set_descent(h, n, deadline, K=12):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.005
    for _ in range(20000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        c_scaled = c * (2.0 / n)
        topk_idx = np.argsort(c)[-K:]
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in topk_idx:
            s = m - (n - 1)
            if s >= 0:
                nv = n - s
                grad[:nv] += (1.0 - h[s:s+nv]) * dx
                grad[s:s+nv] += (-h[:nv]) * dx
            else:
                t = -s
                nv = n - t
                grad[t:t+nv] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[t:t+nv]) * dx
        grad /= K
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad = grad / g_norm
        step = lr
        improved = False
        for _ in range(60):
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            if lr < 1e-12:
                break
    return best_h, best_c

def targeted_pair_search(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        passes = 0
        while time.time() < deadline:
            passes += 1
            if passes > 80:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def grad_guided_pairs(h, n, deadline, step=0.003, radius=15):
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000]
    for alpha in alphas:
        passes = 0
        while time.time() < deadline:
            passes += 1
            if passes > 30:
                break
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    move = -step * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c

def try_complement(h, n):
    h_comp = 1.0 - h
    h_comp = project_feasible(h_comp, n)
    c_h = compute_c5(h, n)
    c_comp = compute_c5(h_comp, n)
    if c_comp < c_h:
        return h_comp, c_comp
    return h, c_h

def perturb_and_optimize(h, n, deadline, rng, sigma=0.005):
    h_pert = h + rng.normal(0, sigma, n)
    h_pert = project_feasible(h_pert, n)
    c_pert = compute_c5(h_pert, n)
    t_short = min(time.time() + 30, deadline)
    h_opt, c_opt = lbfgs_optimize(h_pert, n, t_short, max_per_alpha=100)
    if time.time() < deadline - 5:
        h_opt, c_opt = active_set_descent(h_opt, n, min(time.time() + 15, deadline), K=10)
    return h_opt, c_opt
