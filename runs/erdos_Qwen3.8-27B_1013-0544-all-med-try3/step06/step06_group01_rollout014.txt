def project_box_sum(h, n, target):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - proj
            s = room.sum()
            if s > 1e-14:
                proj = proj + diff / s * room
        else:
            s = proj.sum()
            if s > 1e-14:
                proj = proj + diff / s * proj
    return np.clip(proj, 0.0, 1.0)

def smooth_max_grad(f_k, alpha):
    """Compute the value and gradient of the log-sum-exp smooth max."""
    f_max = f_k.max()
    shifted = alpha * (f_k - f_max)
    exp_shifted = np.exp(shifted)
    log_sum = f_max + np.log(exp_shifted.sum()) / alpha
    weights = exp_shifted / exp_shifted.sum()
    return log_sum, weights

def compute_gradient_smooth(h, n, alpha):
    """Compute gradient of smooth max of f_k."""
    dx = 2.0 / n
    f_k = np.correlate(h, 1 - h, mode='full') * dx
    _, weights = smooth_max_grad(f_k, alpha)
    
    # Gradient of f_k w.r.t. h[j]:
    # f_k = dx * sum_i h[i](1-h[i+k]) for valid i
    # d f_k / d h[j] = dx * [(1-h[j+k]) * 1_{valid} - h[j-k] * 1_{valid}]
    
    # For the full correlation, the lag index m ranges from 0 to 2n-2,
    # corresponding to shift k = m - (n-1)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        k = m - (n - 1)  # shift
        # For this shift k, f_k = dx * sum_i h[i](1-h[i+k]) where 0<=i<n and 0<=i+k<n
        # d/dh[j] = dx * [(1-h[j+k]) if 0<=j+k<n - h[j-k] if 0<=j-k<n]
        # = dx * [(1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}]
        
        # Vectorized:
        j_range = np.arange(n)
        jp = j_range + k  # j + k
        jm = j_range - k  # j - k
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        
        contrib = np.zeros(n)
        contrib[mask_p] += (1 - h[jp[mask_p]])
        contrib[mask_m] -= h[jm[mask_m]]
        
        grad += w * dx * contrib
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full').astype(np.float64) * dx

def project_box_sum(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - proj
            s = room.sum()
            if s > 1e-14:
                proj = proj + (diff / s) * room
        else:
            s = proj.sum()
            if s > 1e-14:
                proj = proj + (diff / s) * proj
    return np.clip(proj, 0.0, 1.0)

def smooth_max_value_and_weights(f, alpha):
    fmax = f.max()
    shifted = alpha * (f - fmax)
    max_shift = shifted.max()
    exp_s = np.exp(shifted - max_shift)
    log_sum = fmax + (max_shift + np.log(exp_s.sum())) / alpha
    weights = exp_s / exp_s.sum()
    return log_sum, weights

def smooth_max_objective(h, n, alpha):
    f = compute_corr(h, n)
    val, _ = smooth_max_value_and_weights(f, alpha)
    return val

def smooth_max_gradient(h, n, alpha):
    f = compute_corr(h, n)
    _, weights = smooth_max_value_and_weights(f, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    nw = 2 * n - 1
    for m in range(nw):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        j = np.arange(n)
        jp = j + k
        jm = j - k
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        contrib = np.zeros(n)
        contrib[mp] += (1.0 - h[jp[mp]])
        contrib[mm] -= h[jm[mm]]
        grad += w * dx * contrib
    return grad

def slsqp_optimize(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        f = compute_corr(x, n)
        val, _ = smooth_max_value_and_weights(f, alpha)
        return val
    def obj_grad(x):
        return smooth_max_gradient(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
        return project_box_sum(res.x, n)
    except Exception:
        return h_init

def subgradient_optimize(h_init, n, deadline, n_iter=5000):
    h = h_init.copy().astype(np.float64)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    f = compute_corr(h, n)
    eta = 0.02
    for it in range(n_iter):
        if time.time() > deadline:
            break
        f = compute_corr(h, n)
        c_val = f.max()
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        k_star = np.argmax(f)
        k = k_star - (n - 1)
        j = np.arange(n)
        jp = j + k
        jm = j - k
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        grad = np.zeros(n)
        grad[mp] += (1.0 - h[jp[mp]])
        grad[mm] -= h[jm[mm]]
        grad *= dx
        h_new = h - eta * grad
        h_new = project_box_sum(h_new, n)
        if np.sum((h_new - h) ** 2) < 1e-20:
            eta *= 0.95
        h = h_new
        eta *= 0.9997
    return best_h, best_c

def vectorized_binary_search(h_init, n, deadline, max_sweeps=2000, n_restarts=4, seed=42):
    target = n // 2
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    best_c = np.inf
    best_h = np.zeros(n)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            h = np.zeros(n)
            idx = np.argsort(h_init)[::-1]
            h[idx[:target]] = 1.0
        else:
            h = best_h.copy()
            for _ in range(rng.randint(2, 8)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
        
        h = h.copy()
        corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            improved = False
            rng.shuffle(zeros)
            for i in zeros:
                if time.time() > deadline:
                    break
                i0 = i
                # Precompute base contribution from i0 (flipping 0->1)
                base_i0 = np.zeros(2 * n - 1)
                for m in range(2 * n - 1):
                    k = m - (n - 1)
                    ip = i0 + k
                    im = i0 - k
                    val = 0.0
                    if 0 <= ip < n:
                        val += (1.0 - h[ip])
                    if 0 <= im < n:
                        val -= h[im]
                    base_i0[m] = val
                
                # For each j in ones, compute delta
                j_arr = np.array(ones, dtype=np.int64)
                # base_j0 for all j simultaneously
                # For lag m, k = m-(n-1):
                # term from j as "i": h[j+k]-1 if j+k valid and j+k != i0, else -1 if j+k==i0
                # term from j as "i+k": h[j-k] if j-k valid and j-k != i0, else 1 if j-k==i0
                # Cross with i0: at lag where j+k==i0 (k=i0-j) and j-k==i0 (k=j-i0)
                
                full_idx = np.arange(2 * n - 1)
                k_arr = full_idx - (n - 1)
                
                # Shape: (len(j_arr), 2n-1)
                jp_all = j_arr[:, None] + k_arr[None, :]
                jm_all = j_arr[:, None] - k_arr[None, :]
                mp_all = (jp_all >= 0) & (jp_all < n)
                mm_all = (jm_all >= 0) & (jm_all < n)
                
                # Term from j as "i": -(1-h_new[j+k]) = h_new[j+k] - 1
                # h_new[j+k] = h[j+k] if j+k != i0, else 1
                term_j_as_i = np.where(mp_all, 
                                       np.where(jp_all == i0, 1.0, h[jp_all].clip(0, n-1)[np.minimum(jp_all, n-1)] * 0 + np.where(jp_all == i0, 1.0, np.where(mp_all, h[np.minimum(jp_all, n-1)], 0.0))),
                                       0.0)
                # Simpler: 
                h_jp = np.where(mp_all, h[np.clip(jp_all, 0, n-1)], 0.0)
                h_jp = np.where(jp_all == i0, 1.0, h_jp)
                term_j_as_i = np.where(mp_all, h_jp - 1.0, 0.0)
                
                # Term from j as "i+k": h_new[j-k]
                h_jm = np.where(mm_all, h[np.clip(jm_all, 0, n-1)], 0.0)
                h_jm = np.where(jm_all == i0, 1.0, h_jm)
                term_j_as_ik = np.where(mm_all, h_jm, 0.0)
                
                delta_j = term_j_as_i + term_j_as_ik  # shape (len(j_arr), 2n-1)
                delta_total = delta_j + base_i0[None, :]  # broadcast
                
                new_corr = corr[None, :] + delta_total
                new_max = new_corr.max(axis=1) * dx
                best_j_idx = np.argmin(new_max)
                
                if new_max[best_j_idx] < cur_c - 1e-14:
                    j0 = ones[best_j_idx]
                    # Execute the flip
                    h[i0] = 1.0
                    h[j0] = 0.0
                    corr = corr + delta_total[best_j_idx]
                    cur_c = new_max[best_j_idx]
                    ones.remove(j0)
                    zeros.remove(i0)
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if not improved:
                break
    
    return best_h, best_c

def multi_resolution_optimize(h_best, n_best, c_best, deadline, seed=42):

    resolutions = [100, 150, 200, 250, 300, 350, 400, 500]

            break
        if nh == n_best:
            continue

        if nh > n_best:

            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = np.interp(xn, xo, h_best)
        else:
            xo = np.linspace(0, 2, n_best, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = np.interp(xn, xo, h_best)
        hh = project_box_sum(hh, nh)

        time_left = deadline - time.time()

        for alpha in [200, 1000, 5000, 20000, 100000]:
