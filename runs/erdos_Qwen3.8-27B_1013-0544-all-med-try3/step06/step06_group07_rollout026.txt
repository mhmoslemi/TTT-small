```python
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta
```

```python
    # h[j] appears when i+lag=j, i.e., i=j-lag (as the "(1-h[i+lag])" factor): contributes -h[j-lag]
    
    # So ∂f/∂h[j] = (1-h[j+lag]) [if j+lag in [0,n-1]] + (-h[j-lag]) [if j-lag in [0,n-1]]
    
    # For the LP:
    # Variables: h[0], ..., h[n-1], t  (total n+1 variables)
    # Minimize: t
    # Subject to:
    #   For each m in [0, 2n-2]: corr[m] + grad_m^T (h - h_star) ≤ t
    #     i.e., grad_m^T h - t ≤ corr[m] - grad_m^T h_star
    #   0 ≤ h[i] ≤ 1
    #   sum(h) = target_sum
    
    # Let me set this up as a linprog problem.
    # min c^T x subject to A_ub x ≤ b_ub, A_eq x = b_eq, bounds
    
    # x = [h_0, ..., h_{n-1}, t]
    # c = [0, ..., 0, 1]
    
    # For each lag m:
    # grad_m^T h - t ≤ corr[m] - grad_m^T h_star
    # i.e., [grad_m, -1] · x ≤ corr[m] - grad_m · h_star
    
    # A_eq: [1, 1, ..., 1, 0] · x = target_sum
    
    # bounds: 0 ≤ h_i ≤ 1, t free (or t ≥ 0)
    
    # Build the constraint matrix
    # This could be large: (2n-1) rows for the correlation constraints + 1 for the sum
    # With n=70: 139 correlation constraints + 1 sum constraint = 140 rows
    # Variables: 71 (70 h's + t)
    
    # Compute all gradients
    # For each m, grad[j] = (1-h[j+lag]) * [j+lag valid] + (-h[j-lag]) * [j-lag valid]
    
    A_ub = np.zeros((2*n, n+1))  # 2n-1 corr constraints + 1 slack (we'll use 2n-1)
    b_ub = np.zeros(2*n-1)
    
    for m in range(2*n-1):
        lag = m - (n-1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + lag <= n-1:
                grad[j] += (1.0 - h_star[j + lag])
            if 0 <= j - lag <= n-1:
                grad[j] += (-h_star[j - lag])
        A_ub[m, :n] = grad
        A_ub[m, n] = -1.0
        b_ub[m] = corr[m] - grad @ h_star
    
    # Equality constraint: sum(h) = target_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds_lp = [(0, 1)] * n + [(None, None)]  # h bounded, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new, res
    else:
        return None, None, None
```

```python
        # j ranges from max(0, lag) to min(n-1, n-1+lag)

        j_end2 = min(n-1, n-1+lag) + 1
```

```python
                h_cur = project_feasible(h_cur, target_sum)
                c5_cur = compute_c5(h_cur)
                stall_count = 0
            elif stall_count > 20:
                h_cur += np.random.randn(n) * 0.001
                h_cur = project_feasible(h_cur, target_sum)
                c5_cur = compute_c5(h_cur)

        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur

    # Phase 2: Try higher resolutions with cutting plane
    resolutions_to_try = [80, 100, 120, 140, 160, 200]
    for n_try in resolutions_to_try:
        if time_left() < 80:
            break

        n_t = n_try
        dx_t = 2.0 / n_t
        ts_t = n_t / 2.0
        h_t = resample_h(best_h, n_t)
        h_t = project_feasible(h_t, ts_t)
        c5_t = compute_c5(h_t)

        if c5_t > best_c5 + 0.002:
            continue

        stall_t = 0
        for it in range(8000):
            if time_left() < 40:
                break
            h_proposed, _ = cutting_plane_step(h_t, n_t, ts_t)
            if h_proposed is None:
                break

            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                h_try = (1.0 - alpha) * h_t + alpha * h_proposed
                h_try = project_feasible(h_try, ts_t)
                c5_try = compute_c5(h_try)
                if c5_try < compute_c5(h_t) - 1e-12:
                    h_t = h_try
                    improved = True
                    stall_t = 0
                    break

            if not improved:
                stall_t += 1
                if stall_t > 60:
                    h_t += np.random.randn(n_t) * 0.002
                    h_t = project_feasible(h_t, ts_t)
                    stall_t = 0

            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t.copy()
                best_c5 = c5_t

    # Phase 3: SLSQP polish at best resolution
    if time_left() > 40:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        ts_best = n_best / 2.0
        bounds_best = Bounds(np.zeros(n_best), np.ones(n_best))

        for beta in [500, 2000, 10000, 50000]:
            if time_left() < 20:
                break
            cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian,
                    'args': (ts_best,)}]
            try:
                res = minimize(obj_smooth_func, best_h, args=(dx_best, beta),
                               method='SLSQP', bounds=bounds_best,
                               constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts_best)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Subgradient refinement at best resolution
    if time_left() > 60:
        n_sg = len(best_h)
        ts_sg = n_sg / 2.0
        h_sg = best_h.copy()
        c5_sg = compute_c5(h_sg)
        lr_base = 0.002

        for it in range(30000):
            if time_left() < 20:
                break
            corr = np.correlate(h_sg, 1.0 - h_sg, mode='full')
            k_worst = np.argmax(corr)
            grad = subgrad_dir(h_sg, k_worst, n_sg)
            lr = lr_base / (1.0 + it * 0.0003)
            h_sg = h_sg - lr * grad
            h_sg = project_feasible(h_sg, ts_sg)

            if it % 3000 == 2999:
                c5_sg = compute_c5(h_sg)
                if c5_sg < best_c5:
                    best_h = h_sg.copy()
                    best_c5 = c5_sg

        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c5_sg

    # Phase 5: Final cutting plane polish
    if time_left() > 30:
        n_fp = len(best_h)
        ts_fp = n_fp / 2.0
        h_fp = best_h.copy()
        c5_fp = compute_c5(h_fp)

        for it in range(5000):
            if time_left() < 10:
                break
            h_proposed, _ = cutting_plane_step(h_fp, n_fp, ts_fp)
            if h_proposed is None:
                break
            for alpha in [1.0, 0.5, 0.25]:
                h_try = (1.0 - alpha) * h_fp + alpha * h_proposed
                h_try = project_feasible(h_try, ts_fp)
                c5_try = compute_c5(h_try)
                if c5_try < c5_fp - 1e-13:
                    h_fp = h_try
                    c5_fp = c5_try
                    break
                else:
                    if alpha == 0.25:
                        break

        if c5_fp < best_c5:
            best_h = h_fp.copy()
            best_c5 = c5_fp

    # Phase 6: Try a few random restarts if time remains
    if time_left() > 80:
        n_rr = len(best_h)
        ts_rr = n_rr / 2.0
        dx_rr = 2.0 / n_rr
        for restart in range(5):
            if time_left() < 40:
                break
            h_rr = best_h + np.random.randn(n_rr) * 0.05
            h_rr = project_feasible(h_rr, ts_rr)

            for it in range(3000):
                if time_left() < 20:
                    break
                h_proposed, _ = cutting_plane_step(h_rr, n_rr, ts_rr)
                if h_proposed is None:
                    break
                for alpha in [1.0, 0.5, 0.25]:
                    h_try = (1.0 - alpha) * h_rr + alpha * h_proposed
                    h_try = project_feasible(h_try, ts_rr)
                    c5_try = compute_c5(h_try)
                    if c5_try < compute_c5(h_rr) - 1e-12:
                        h_rr = h_try
                        break

            c5_rr = compute_c5(h_rr)
            if c5_rr < best_c5:
                best_h = h_rr.copy()
                best_c5 = c5_rr

    # Final verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
