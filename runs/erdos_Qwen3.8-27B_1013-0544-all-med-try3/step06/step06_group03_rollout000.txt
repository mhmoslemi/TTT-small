from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (abs(diff) / tot) * h
    h = np.clip(h, 0.0, 1.0)
    final_diff = target - h.sum()
    if abs(final_diff) > 1e-14:
        idx = int(np.argmax(h))
        h[idx] -= final_diff
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lse(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c)
    mx = sh.max()
    if mx > 500:
        return float((mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(np.log(np.sum(np.exp(sh))) / alpha)

def _sc(x, n):
    return x.sum() - n / 2.0

def _sj(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _sj, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_steps(h, n, deadline, n_iter=5000):
    """Subgradient method with projection for minimizing max correlation"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)
    
    eta = 0.01
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = c.max()
        if cur_max < best_c - 1e-15:
            best_c = cur_max
            best_h = h.copy()
        
        # Find all active lags (within epsilon of max)
        active_mask = c >= cur_max - 1e-6
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask = (c == c.max())
        
        # Subgradient: average of gradients of active lags
        # grad of c_k w.r.t. h: for lag k, d(c_k)/d(h[p]) 
        # c_k = sum_i h[i+k]*(1-h[i]) for k >= 0 (in our indexing)
        # Using the full correlation formulation:
        # output[m] where m = k + (n-1), k is the lag
        # d(output[m])/d(h[p]) = (1-h[p-k])*I(0<=p-k<n) - h[p+k]*I(0<=p+k<n)
        
        grad = np.zeros(n)
        active_indices = np.where(active_mask)[0]
        for m in active_indices:
            k = k_vals[m]
            for p in range(n):
                i1 = p - k
                if 0 <= i1 < n:
                    grad[p] += (1.0 - h[i1])
                i2 = p + k
                if 0 <= i2 < n:
                    grad[p] -= h[i2]
        grad /= n_active
        
        # Take step
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        # Check if improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        # Decrease step size
        eta *= 0.999
        if eta < 1e-8:
            eta = 1e-8
    
    return best_h, best_c
