    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()

            if s < 1e-14:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(x, n, alpha):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 700:

    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        h = np.clip(x, 0, 1)
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def best_improvement_binary(h, n, deadline, max_sweeps=100):
    """Best-improvement binary local search."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1 - h_bin, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_bin.copy()
    
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        best_swap_c = cur_c
        best_i = -1
        best_j = -1
        best_delta = None
        
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2*n-1)
                
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_bin[im[msk]]
                
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_bin[imn[msk]]
                
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_bin[jm[msk]]
                
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_bin[jmn[msk]]
                
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                new_max = (corr + delta).max() * dx
                if new_max < best_swap_c - 1e-15:
                    best_swap_c = new_max
                    best_i = i
                    best_j = j
                    best_delta = delta
        
        
        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            corr = corr + best_delta
            cur_c = best_swap_c
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        
        if not improved:
            break
    
    return best_h, best_c

def annealing_binary(h, n, deadline, seed=42):
    """Simulated annealing on binary representation."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_sa = np.zeros(n)
    h_sa[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h_sa, 1 - h_sa, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_sa.copy()
    
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    # Temperature schedule
    T_start = 0.002  # in C5 units
    T_end = 1e-5
    n_steps = 200000
    cooldown = (T_end / T_start) ** (1.0 / n_steps)
    T = T_start
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T *= cooldown
        
        i = rng.randint(n)
        j = rng.randint(n)
        if h_sa[i] == h_sa[j]:
            continue
        
        # Ensure i is the 0 and j is the 1
        if h_sa[i] == 1:
            i, j = j, i
        
        # Compute delta
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h_sa[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h_sa[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h_sa[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h_sa[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        
        if dc <= 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            h_sa[i] = 1.0
            h_sa[j] = 0.0
            corr = new_corr
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_sa.copy()
    
    return best_h, best_c
