def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        else:

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

            h = np.clip(h, 0.0, 1.0)

    """Find optimal h[i] given all other h fixed, maintaining sum constraint."""
    # Compute alpha_k for all lags k
    # For the full correlation array of length 2n-1:
    # correlate(h, 1-h)[k] where k = 0..2n-2, actual lag = k - (n-1)
    
    # For coordinate i, compute its contribution to each lag
    # lag = k - (n-1)
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    alpha = np.zeros(2*n-1)
    for idx, lag in enumerate(lags):
        if lag >= 0:
            # h[i] appears as:
            # - h[i+lag]*(1-h[i]) if i+lag < n: coeff of h[i] is -h[i+lag]
            # - h[i]*(1-h[i-lag]) if i-lag >= 0: coeff of h[i] is (1-h[i-lag])
            coeff = 0.0
            if i + lag < n:
                coeff += -h[i + lag]
            if i - lag >= 0:
                coeff += (1.0 - h[i - lag])
            alpha[idx] = coeff
        else:
            m = -lag  # m > 0
            # correlate[h, 1-h][k] = sum_j h[j]*(1-h[j+m]) for j=0..n-1-m
            # h[i] appears as:
            # - h[i]*(1-h[i+m]) if i+m < n: coeff is (1-h[i+m])
            # - h[i-m]*(1-h[i]) if i-m >= 0: coeff is -h[i-m]

            if i - m >= 0:
                coeff += -h[i - m]
            alpha[idx] = coeff

    # Minimize f(v) = max_k (base[k] + alpha[k] * v) over v in [0, 1]

        return np.max(base + alpha * v)
    
    res = minimize_scalar(f_1d, bounds=[0.0, 1.0], method='bounded',
                          options={'xatol': 1e-12})
    v_opt = res.x
    return v_opt

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break

        else:

        if cnt == 0:

def compute_alpha(h, i, n):
    """Compute alpha vector for coordinate i."""
    lags = np.arange(2*n-1) - (n-1)
    alpha = np.zeros(2*n-1)
    for idx, l in enumerate(lags):

                c += 1.0 - h[i - l]

        else:

            c = 0.0
            if i + m < n:
                c += 1.0 - h[i + m]
            if i - m >= 0:
                c += -h[i - m]
            alpha[idx] = c
    return alpha

def find_opt_v(base, alpha):
    """Find v in [0,1] minimizing max_k (base[k] + alpha[k]*v)."""
    # Quick grid
    vs = np.linspace(0, 1, 20)
    fs = np.array([np.max(base + alpha * v) for v in vs])
    i_best = np.argmin(fs)

        return 0.0
    if i_best == len(vs) - 1:
        return 1.0

    for _ in range(50):

        if avg_alpha > 0:
            hi = mid
        elif avg_alpha < 0:

        else:

def coord_descent(h, n, target_sum, max_sweeps=500, t0=None, budget_s=None):
    """Coordinate descent with incremental corr updates."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = np.max(corr) * dx
    h_work = h.copy()
    
    for sweep in range(max_sweeps):
        if t0 is not None:
            elapsed = time.time() - t0
            if budget_s - elapsed < 10:
                break
        
        for i in range(n):
            alpha = compute_alpha(h_work, i, n)
            base = corr - alpha * h_work[i]
            v_opt = find_opt_v(base, alpha)
            delta = v_opt - h_work[i]
            if abs(delta) > 1e-15:
                corr += alpha * delta
                h_work[i] = v_opt
        
        h_work = project_feasible(h_work, target_sum)
        # Recompute corr after projection
        corr = np.correlate(h_work, 1.0 - h_work, mode='full').astype(float)
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
    
    return h_work, best_c5
