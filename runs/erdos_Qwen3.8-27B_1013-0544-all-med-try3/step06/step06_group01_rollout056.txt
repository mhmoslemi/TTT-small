def simulated_annealing(h, n, t_budget, seed):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    # Compute initial correlation
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    # Temperature schedule
    t_start = time.time()
    t_end = t_start + t_budget
    T0 = 0.5  # Initial temperature
    T_end = 0.01

    zeros = np.where(h == 0)[0]
    
    while time.time() < t_end:
        # Pick random swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta for correlation
        # ... (update corr in O(n))

        # Accept/reject
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = new_c
            ones = np.setdiff1d(ones, [j])
            ones = np.union1d(ones, [i])
            zeros = np.setdiff1d(zeros, [i])
            zeros = np.union1d(zeros, [j])
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        # Cool down
        progress = (time.time() - t_start) / t_budget
        T = T0 * (T_end / T0) ** progress
    
    return best_h, best_c
