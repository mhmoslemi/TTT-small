```python
def logsumexp_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gm = g.max()
    a = alpha * (g - gm)
    ea = np.exp(a)
    se = ea.sum()
    obj = gm + np.log(se) / alpha
    p = ea / se  # shape (2n-1,)
    
    grad = np.zeros(n)
    # For each lag k (indexed 0 to 2n-2), compute contribution to gradient
    # lag = idx - (n-1)
    # For lag >= 0: d g_lag / d h[m] = (1-h[m-lag])*[lag<=m<=n-1] - h[m+lag]*[0<=m<=n-1-lag]
    # For lag < 0 (lag = -l): d g_lag / d h[m] = (1-h[m+l])*[0<=m<=n-1-l] - h[m-l]*[l<=m<=n-1]
    
    for idx in range(len(p)):
        if p[idx] < 1e-15:
            continue
        lag = idx - (n - 1)
        w = p[idx] * dx
        if lag >= 0:
            # (1-h[m-lag]) for m in [lag, n-1]
            if lag < n:
                grad[lag:] += w * (1 - h[:n-lag])
            # -h[m+lag] for m in [0, n-1-lag]
            if lag < n:
                grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            # (1-h[m+l]) for m in [0, n-1-l]
            grad[:n-l] += w * (1 - h[l:])
            # -h[m-l] for m in [l, n-1]
            grad[l:] -= w * h[:n-l]
    
    return obj, grad
```

This is O(n²) in the worst case but with early termination for small p[idx]. In practice, for large alpha, only a few lags have significant weight, so it's much faster.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def logsumexp_obj_and_grad(h, n, alpha, mu=0.0):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gm = g.max()
    a = alpha * (g - gm)
    ea = np.exp(a)
    se = ea.sum()
    
    # Objective: gm + log(se)/alpha + mu * (sum(h) - n/2)^2
    sum_h = h.sum()
    sum_pen = mu * (sum_h - n/2.0)**2
    obj = gm + np.log(se) / alpha + sum_pen
    
    p = ea / se  # shape (2n-1,)
    
    grad = np.zeros(n)
    for idx in range(len(p)):
        pi = p[idx]
        if pi < 1e-14:
            continue
        lag = idx - (n - 1)
        w = pi * dx
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n-lag])
                grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n-l]
    
    # Add sum constraint gradient
    grad += 2.0 * mu * (sum_h - n/2.0)
    
    return obj, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, n, alpha, mu, deadline, maxiter=5000):
    def obj_fun(x):
        val, _ = logsumexp_obj_and_grad(x, n, alpha, mu)
        return val
    
    def grad_fun(x):
        _, g = logsumexp_obj_and_grad(x, n, alpha, mu)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def focus_optimize(h, n, deadline, max_iter=2000):
    """Focus on the worst lag and try to reduce it."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        
        # Compute gradient of g[k_star]
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n-lag]) * dx
            grad[:n-lag] -= h[lag:] * dx
        else:
            l = -lag
            grad[:n-l] += (1.0 - h[l:]) * dx
            grad[l:] -= h[:n-l] * dx
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Try decreasing step sizes
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = factor / ng
            ht = h - step * grad
            ht = project_feasible(ht, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                best_h = h.copy()
                best_c = ct
                found = True
                break
        
        if not found:
            # Try a small random perturbation
            h = project_feasible(h + 0.01 * np.random.randn(n), n)
            c = compute_c5(h, n)
            if c < best_c:
                best_h = h.copy()
                best_c = c
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500):
    """Local search on binary solutions."""
    target = n // 2
    # Initialize binary from h
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all single swaps
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                # Try swap i and j
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(j)
                    ones.add(i)
                    zeros.remove(i)
                    zeros.add(j)
                    improved = True
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    
    return h_bin, best_c

def multi_lag_focus(h, n, deadline, top_k=3, max_iter=1000):
    """Focus on top-k worst lags simultaneously."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()

        for idx in top_idx:

            else:
                l = -lag
                grad[:n-l] += (1.0 - h[l:]) * dx
                grad[l:] -= h[:n-l] * dx
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:

        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            step = factor / (top_k * ng)
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                best_h = h.copy()
                best_c = ct
                found = True
                break
        
        if not found:
            break
    
    return best_h, best_c
```

```python
def logsumexp_obj_and_grad(h, n, alpha, mu=0.0):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gm = g.max()
    a = alpha * (g - gm)
```

Wait, np.correlate is O(n²) for the full output. So each evaluation is O(n²). For n=250, that's 62500 operations per evaluation. At 10^8 ops/sec, that's 0.6ms per evaluation. So 100000 evaluations would take 60 seconds. That's feasible!

But I need to be smarter. Instead of computing the full correlation each time, I can compute just the change in the correlation due to a single swap.

When I swap h[i] and h[j] (h[i] goes from 0 to 1, h[j] goes from 1 to 0):
- For each lag k, g_k changes by:
  Δg_k = contribution of h[i] going 0→1 minus contribution of h[j] going 1→0

The contribution of h[m] to g_k (for lag k ≥ 0) is:
- If m appears as h[i+k] (i.e., i = m-k): contributes (1-h[m-k])
- If m appears as h[i] (i = m): contributes -h[m+k]

So Δg_k from h[i] going 0→1:
= (1-h[i-k]) * [i-k valid] - h[i+k] * [i+k valid] (for the specific lag k)

This is still O(n) per lag, and there are O(n) lags, so O(n²) per swap evaluation. Same as before.

But I can precompute the current g values and then compute the change quickly:
Δg_k = (value of new terms) - (value of old terms)

Actually, for a single swap (i: 0→1, j: 1→0), the change in g_k is:
Δg_k = [contribution of position i when h[i]=1] - [contribution of position i when h[i]=0]
      - [contribution of position j when h[j]=0] + [contribution of position j when h[j]=1]

The contribution of h[m] to g_k (treating h[m] as a variable):
For lag k ≥ 0:
∂g_k/∂h[m] = (1-h[m-k]) * [k≤m≤n-1] - h[m+k] * [0≤m≤n-1-k]

But this depends on the other h values. When h[m] changes from a to b, the change in g_k is:
(b-a) * [(1-h[m-k]) * [k≤m≤n-1] - h[m+k] * [0≤m≤n-1-k]]

Wait, that's only true if the other terms don't change. But when h[m] changes, the terms that involve h[m] in the "other position" also change!

Specifically, g_k = sum_i h[i+k](1-h[i]). If I change h[m]:
- The term where m appears in h[i+k]: i = m-k, term is h[m](1-h[m-k]), changes by (b-a)(1-h[m-k])
- The term where m appears in h[i]: i = m, term is h[m+k](1-h[m]), changes by -(b-a)h[m+k]

So Δg_k = (b-a) * [(1-h[m-k]) * [valid] - h[m+k] * [valid]]

This is exactly the partial derivative times (b-a). And since a=0, b=1 (for i) and a=1, b=0 (for j):
Δg_k = 1 * [(1-h[i-k]) * [valid_i] - h[i+k] * [valid_i]] + (-1) * [(1-h[j-k]) * [valid_j] - h[j+k] * [valid_j]]

So for each swap, I can compute Δg_k for all k in O(n) time (since the valid conditions are simple ranges). Then I can update the max in O(n) time.

Total: O(n) per swap evaluation! That's much better than O(n²).

```python
def swap_delta_g(h, n, i_new, j_old):
    """Compute the change in g_k for all k when h[i_new] goes 0→1 and h[j_old] goes 1→0.
    Returns array of shape (2n-1,) with the deltas."""
    dx = 2.0 / n
    delta = np.zeros(2*n-1)
    
    # Contribution of position i_new going from 0 to 1
    for idx in range(2*n-1):
        lag = idx - (n-1)
        d = 0.0
        if lag >= 0:
            m = i_new
            if lag <= m <= n-1:
                d += (1.0 - h[m-lag])
            if 0 <= m <= n-1-lag:
                d -= h[m+lag]
        else:
            l = -lag
            m = i_new
            if 0 <= m <= n-1-l:
                d += (1.0 - h[m+l])
            if l <= m <= n-1:
                d -= h[m-l]
        delta[idx] += d * dx
    
    # Contribution of position j_old going from 1 to 0 (negative)
    for idx in range(2*n-1):
        lag = idx - (n-1)
        d = 0.0
        if lag >= 0:
            m = j_old
            if lag <= m <= n-1:
                d += (1.0 - h[m-lag])
            if 0 <= m <= n-1-lag:
                d -= h[m+lag]
        else:
            l = -lag
            m = j_old
            if 0 <= m <= n-1-l:
                d += (1.0 - h[m+l])
            if l <= m <= n-1:
                d -= h[m-l]
        delta[idx] -= d * dx
    
    return delta
```

This is O(n) per position, so O(n) total for two positions. But it's a Python loop over n elements, which is slow. Let me vectorize:

For position m, f_m(k) for k ≥ 0:
- (1-h[m-k]) when k ≤ m ≤ n-1, i.e., k ∈ [0, m] (and also need m-k ≤ n-1-k which is m ≤ n-1, already satisfied)
  Also need m-k ≥ 0, i.e., k ≤ m. And m ≤ n-1 (always true).
  So: (1-h[m-k]) for k ∈ [0, min(m, n-1)] but also need the index m-k to be valid in the sum, i.e., the original sum index i = m-k must satisfy i ∈ [0, n-1-k]. So m-k ≤ n-1-k, i.e., m ≤ n-1. ✓
  
- -h[m+k] when 0 ≤ m ≤ n-1-k, i.e., k ≤ n-1-m. And m+k < n (which follows from k ≤ n-1-m).
  So: -h[m+k] for k ∈ [0, min(n-1-m, n-1)]

For k < 0, let l = -k > 0:
- (1-h[m+l]) when 0 ≤ m ≤ n-1-l, i.e., l ≤ n-1-m. So l ∈ [1, n-1-m].
- -h[m-l] when l ≤ m ≤ n-1, i.e., l ∈ [1, m].

So f_m(k) for k ≥ 0:
= (1-h[m-k]) * [0 ≤ k ≤ m] - h[m+k] * [0 ≤ k ≤ n-1-m]

And f_m(k) for k = -l (l > 0):
= (1-h[m+l]) * [1 ≤ l ≤ n-1-m] - h[m-l] * [1
