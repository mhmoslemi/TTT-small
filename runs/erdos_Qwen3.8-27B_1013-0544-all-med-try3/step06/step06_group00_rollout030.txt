import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum_box(x, target):
    """Project x onto {h: sum(h)=target, 0<=h<=1} via binary search on lambda."""
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(x - lam, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[: n // 2]] = 1.0
    return out

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max value and gradient of correlation objective."""
    m = 2 * n - 1
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    cm = c.max()
    ec = np.exp(alpha * (c - cm))
    p = ec / ec.sum()
    val = cm + np.log(ec.sum()) / alpha

    # Fast gradient via convolution
    v = 1.0 - h
    # A[j] = sum_k p[k]*(1-h)[j+s_k] with boundary
    A = np.correlate(v, p, mode='full')
    # B[j] = sum_k p[k]*h[j-s_k] with boundary
    B = np.correlate(h, p[::-1], mode='full')
    # Both A and B have length 2n-1, take the [n-1] centered part
    grad = A - B

    return val, grad

def slsqp_smooth(h, n, alpha, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = c.max()
        ec = np.exp(alpha * (c - cm))
        return cm + np.log(ec.sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum_box(res.x, n / 2.0)
    except Exception:
        return project_sum_box(h, n / 2.0)

    for alpha in [5000, 50000, 500000, 5e6, 5e7, 5e8]:
        if time.time() > p3 - 3:
            break
        h = slsqp_smooth(h, n, alpha, maxiter=4000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Resolution exploration (~180s)
    p4 = t0 + 0.74 * total
    for nh in [384, 420, 440, 360]:
        if time.time() > p4 - 15:
            break
        hh = resample_h(best_h, n, nh)
        # Quick binary improvement
        hh, ch = greed_binary(hh, nh, min(time.time() + 8, p4 - 12), max_sweeps=300)
        for alpha in [50000, 5e6, 5e8]:
            if time.time() > p4 - 8:
                break
            hh = slsqp_smooth(hh, nh, alpha, maxiter=2000, ftol=1e-18)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh

    # Phase 5: Noise restart + high-alpha SLSQP polish (~120s)
    p5 = dl - 3
    for trial in range(8):
        if time.time() > p5 - 1.0:
            break
        noise_scale = 0.002 + 0.003 * (trial % 3)
        h = best_h + noise_scale * rng.randn(n)
        h = project_sum_box(h, n / 2.0)
        for alpha in [5e7, 5e8, 5e9]:
            if time.time() > p5 - 0.5:
                break
            h = slsqp_smooth(h, n, alpha, maxiter=6000, ftol=1e-19)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Final: short gradient polish on the best solution
    if time.time() < dl - 2:
        h_fin, c_fin = nesterov_gradient(best_h, n, dl - 1,
                                         alpha_init=1e6, alpha_final=1e10, n_iter=200000)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    # Validation
    best_h = project_sum_box(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
