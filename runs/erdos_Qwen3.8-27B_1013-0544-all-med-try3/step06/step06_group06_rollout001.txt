def delta_one_matrix(h, positions, n):
    """Compute delta_one for multiple positions at once.
    positions: array of positions
    Returns: (len(positions), 2n-1) matrix
    """
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # positions: shape (m,)
    # ip: shape (m, 2n-1), ip[p, k] = positions[p] + s[k]
    # im: shape (m, 2n-1), im[p, k] = positions[p] - s[k]
    
    pos = positions[:, np.newaxis]  # (m, 1)
    ip = pos + s[np.newaxis, :]  # (m, 2n-1)
    im = pos - s[np.newaxis, :]  # (m, 2n-1)
    
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    
    delta = np.zeros((len(positions), 2*n-1))
    delta[mask_p] = 1 - h[ip[mask_p]]
    delta[mask_m] -= h[im[mask_m]]
    return delta
