def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()

            s = h.sum()
            if s < 1e-14:

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break

def slsqp_optimize(h, n, alpha, maxiter, deadline):

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

        return x.sum() - n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta(h, n, i, j, m_range, L):
    """Compute the change in correlation when h[i] goes 0->1 and h[j] goes 1->0."""
    delta = np.zeros(L)
    L2 = 2 * n - 1
    
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]

    delta[idx] -= 2.0

    if 0 <= k1 < L2:
        delta[k1] += 1.0
    k2 = idx + (i - j)
    if 0 <= k2 < L2:
        delta[k2] += 1.0
    
    return delta

def greedy_polish(h_init, n, deadline, max_sweeps=3000):
    """Greedy binary local search."""
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    L2 = 2 * n - 1
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set([i for i in range(n) if i not in ones_set])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = sorted(zeros_set)
        o_list = sorted(ones_set)
        
        for i in z_list:
            if time.time() > deadline:
                break
            for j in o_list:
                delta = compute_delta(h, n, i, j, m_range, L2)
                new_corr = corr + delta
                nc = new_corr.max()

                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones_set.discard(j)
                    ones_set.add(i)
                    zeros_set.discard(i)
                    zeros_set.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    ones_set = set(ones_list)

    rng = np.random.RandomState(seed)
    t0 = time.time()
    total_time = deadline - t0
    
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - t0
        progress = min(1.0, elapsed / max(total_time, 0.01))
        t = t_start * (t_end / t_start) ** progress
        
        if it % 1000 == 0:
            # Periodic greedy polish
            pass  # skip for now
        
        # Decide: biased (70%) or random (30%)
        if rng.random() < 0.7:
            # Biased: target the max lag
            max_lag_idx = corr.argmax()
            k_star = max_lag_idx - (n - 1)
            
            # Find j: h[j]=1 and h[j+k_star]=0
            bad_js = []
            for j in ones_list:
                jk = j + k_star
                if 0 <= jk < n and h[jk] == 0:
                    bad_js.append(j)
            
            # Find i: h[i]=0 and h[i+k_star]=1
            good_is = []
            for i in zeros_list:
                ik = i + k_star
                if 0 <= ik < n and h[ik] == 1:
                    good_is.append(i)
            
            if bad_js and good_is:
                j = bad_js[rng.randint(len(bad_js))]
                i = good_is[rng.randint(len(good_is))]
            elif bad_js:
                j = bad_js[rng.randint(len(bad_js))]
                i = zeros_list[rng.randint(len(zeros_list))]
            else:
                # Fall back to random
                i = zeros_list[rng.randint(len(zeros_list))]
                j = ones_list[rng.randint(len(ones_list))]
        else:
            i = zeros_list[rng.randint(len(zeros_list))]
            j = ones_list[rng.randint(len(ones_list))]
        
        if i == j:
            continue
        
        delta = compute_delta(h, n, i, j, m_range, L2)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or (t > 1e-10 and rng.random() < np.exp(-delta_c / t)):
            h[i], h[j] = 1.0, 0.0

            cur_c = nc
            ones_set.discard(j)
            ones_set.add(i)
            zeros_set.discard(i)
            zeros_set.add(j)
            # Update lists
            ones_list = list(ones_set)
            zeros_list = list(zeros_set)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c
