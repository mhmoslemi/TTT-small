        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def projected_subgradient(h, n, deadline, alpha=1000.0, n_iter=50000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr_init = 1.0 / np.sqrt(n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 2000 == 0:
            alpha = min(alpha * 1.3, 10000.0)
        g = smooth_max_grad(h, n, alpha)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lr = lr_init / np.sqrt(it + 1)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
            # Pick up improvements
            h = best_h.copy()
    return best_h, best_c

def gradient_local_search(h, n, deadline, step=0.003):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            h = best_h.copy()
            if not improved:
                break
    return best_h, best_c

def make_binary_init(n, pattern='alternate', p_count=None, rng=None):
    if p_count is None:
        p_count = n // 2
    h = np.zeros(n)
    if pattern == 'alternate':
        h[::2] = 1.0
        total = h.sum()
        if total > p_count:
            excess = int(total - p_count)
            idx = np.where(h == 1.0)[0]
            h[idx[:excess]] = 0.0
        elif total < p_count:
            deficit = int(p_count - total)
            idx = np.where(h == 0.0)[0]
            h[idx[:deficit]] = 1.0
    elif pattern == 'cluster':
        h[:p_count] = 1.0
    elif pattern == 'spread':
        if n >= p_count:
            h[np.arange(0, n, n // p_count)[:p_count]] = 1.0
    elif pattern == 'random':
        if rng is not None:
            idx = rng.choice(n, size=p_count, replace=False)
            h[idx] = 1.0
    return project_feasible(h, n)

def make_cosine_init(n, freq=1.0, amp=0.5):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.cos(np.pi * freq * x)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Refine parent at its native resolution
    if time.time() < deadline - 400:
        t_phase1 = min(t0 + 150, deadline - 400)
        best_h, best_c = lbfgs_optimize(best_h, best_n, t_phase1, max_iters=200)
        if time.time() < deadline - 350:
            t_s = min(time.time() + 60, deadline - 350)
            h_s, c_s = slsqp_optimize(best_h, best_n, t_s, alpha=500000, maxiter=1500)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
        if time.time() < deadline - 300:
            t_p = min(time.time() + 40, deadline - 300)
            best_h, best_c = pair_search(best_h, best_n, t_p,
                                         deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002],
                                         radius=50)

    # Phase 2: Multi-resolution optimization
    resolutions = [400, 500, 600, 800]
    for n_try in resolutions:
        if time.time() > deadline - 100:
            break
        # Interpolate current best to new resolution
        h_try = interpolate_to(best_h, best_n, n_try)
        alloc = min(120, deadline - time.time() - 80)
        if alloc < 40:
            break

        # L-BFGS at this resolution
        t_lb = min(time.time() + alloc * 0.5, deadline - 80)
        h_opt, c_opt = lbfgs_optimize(h_try, n_try, t_lb, max_iters=200)

        # SLSQP refinement
        if time.time() < deadline - 60:
            t_sl = min(time.time() + 40, deadline - 60)
            h_opt, c_opt = slsqp_optimize(h_opt, n_try, t_sl, alpha=1000000, maxiter=1000)

        # Local search
        if time.time() < deadline - 40 and c_opt < best_c + 0.001:
            t_ls = min(time.time() + 30, deadline - 40)
            h_opt, c_opt = pair_search(h_opt, n_try, t_ls,
                                       deltas=[0.002, 0.001, 0.0005],
                                       radius=40)

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            best_n = n_try

    # Phase 3: Try diverse initializations at high resolution
    diverse_n = 500
    if time.time() < deadline - 120:
        inits = []
        inits.append(interpolate_to(best_h, best_n, diverse_n))
        inits.append(make_binary_init(diverse_n, 'alternate', rng=rng))
        inits.append(make_binary_init(diverse_n, 'spread', rng=rng))
        inits.append(make_cosine_init(diverse_n, freq=2.0, amp=0.5))
        inits.append(make_cosine_init(diverse_n, freq=3.0, amp=0.5))
        inits.append(make_binary_init(diverse_n, 'random', rng=rng))

        time_per_init = max(15, (deadline - time.time() - 60) / len(inits))
        for h_d in inits:
            if time.time() > deadline - 60:
                break
            t_d = min(time.time() + time_per_init, deadline - 60)
            h_d, c_d = lbfgs_optimize(h_d, diverse_n, t_d, max_iters=150)
            if c_d < best_c:
                best_c = c_d
                best_h = h_d
                best_n = diverse_n

    # Phase 4: Final refinement at best resolution
    if time.time() < deadline - 20:
        # L-BFGS final
        t_f1 = min(time.time() + 50, deadline - 15)
        best_h, best_c = lbfgs_optimize(best_h, best_n, t_f1, max_iters=200)

    if time.time() < deadline - 10:
        # SLSQP final
        t_f2 = min(time.time() + 30, deadline - 10)
        h_f, c_f = slsqp_optimize(best_h, best_n, t_f2, alpha=2000000, maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    if time.time() < deadline - 5:
        # Final pair search
        best_h, best_c = pair_search(best_h, best_n, deadline - 2,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001],
                                     radius=40)

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
