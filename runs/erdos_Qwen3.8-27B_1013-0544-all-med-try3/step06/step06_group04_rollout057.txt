And for all other lags m: c(m) + Δc(m) ≤ c(m*)  (or ≤ current_max)

Since Δc(m) ∈ {-2dx, -dx, 0, dx, 2dx}, and current_max - c(m) for "non-max" lags is typically much larger than 2dx, the constraint is usually only binding for lags near the maximum.

1. Find m* (worst lag) and the set S of "critical" lags (those with c(m) > current_max - 2dx)
2. For the swap to be valid: Δc(m*) < 0 AND for all m ∈ S, c(m) + Δc(m) ≤ current_max
3. Find the swap that minimizes c(m*) + Δc(m*) (i.e., maximizes the reduction at the worst lag) subject to the validity constraint

For step 3, we want to minimize h(i+m*) + h(i-m*) - h(j+m*) - h(j-m*), i.e., minimize (h(i+m*) + h(i-m*)) and maximize (h(j+m*) + h(j-m*)).

Since h is binary, h(i+m*) + h(i-m*) ∈ {0, 1, 2} and similarly for j.

So the best case is: h(i+m*) + h(i-m*) = 0 and h(j+m*) + h(j-m*) = 2, giving Δc(m*) = -2dx.

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary while maintaining sum = n/2"""
    h_sorted = np.sort(h)
    threshold = h_sorted[n // 2 - 1] if n % 2 == 0 else h_sorted[n // 2]
    h_bin = (h > threshold).astype(float)
    # Fix sum
    current_sum = h_bin.sum()
    target = n / 2.0
    if current_sum > target:
        # Need to flip some 1s to 0s
        excess = int(current_sum - target)
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones])
        h_bin[ones[order[:excess]]] = 0
    elif current_sum < target:
        deficit = int(target - current_sum)
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:deficit]]] = 1
    return h_bin

def binary_local_search(h, n, deadline, rng, top_k=30, max_time=60):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full') * dx
    start_time = time.time()
    
    while time.time() < deadline and (time.time() - start_time) < max_time:
        current_max = c.max()
        max_idx = int(np.argmax(c))
        m_star = max_idx - (n - 1)
        
        # Find critical lags
        tol = 2.5 * dx
        critical = np.where(c >= current_max - tol)[0]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute scores at distance m_star
        m_abs = abs(m_star)
        s1 = np.zeros(len(ones))
        for idx, i in enumerate(ones):
            val = 0.0
            p1 = i + m_star
            p2 = i - m_star
            if 0 <= p1 < n:
                val += h[p1]
            if 0 <= p2 < n:
                val += h[p2]
            s1[idx] = val
        
        s0 = np.zeros(len(zeros))
        for idx, j in enumerate(zeros):
            val = 0.0
            p1 = j + m_star
            p2 = j - m_star
            if 0 <= p1 < n:
                val += h[p1]
            if 0 <= p2 < n:
                val += h[p2]
            s0[idx] = val
        
        # Get top candidates
        i_cand_idx = np.argsort(s1)[:min(top_k, len(ones))]
        j_cand_idx = np.argsort(-s0)[:min(top_k, len(zeros))]
        
        found = False
        for ii in i_cand_idx:
            i = ones[ii]
            for jj in j_cand_idx:
                j = zeros[jj]
                s_i = s1[ii]
                s_j = s0[jj]
                if s_i >= s_j:
                    continue
                # Check critical lags
                valid = True
                for cm in critical:
                    m = int(cm) - (n - 1)
                    if m == m_star:
                        continue
                    delta = 0.0
                    for p in [i + m, i - m, j + m, j - m]:
                        if 0 <= p < n:
                            if p == i + m or p == i - m:
                                delta += h[p]
                            else:
                                delta -= h[p]
                    if c[cm] + dx * delta > current_max + 1e-9:
                        valid = False
                        break
                if valid:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = np.correlate(h, 1 - h, mode='full') * dx
                    found = True
                    break
            if found:
                break
        
        if not found:
            # Try random moves to escape local optimum
            for attempt in range(10):
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                old_c_max = c.max()
                h[i] = 0.0
                h[j] = 1.0
                new_c = np.correlate(h, 1 - h, mode='full') * dx
                new_max = new_c.max()
                if new_max < old_c_max:
                    c = new_c
                    found = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if not found:
                break
    
    return h, c.max()

def smooth_max_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def smooth_max_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_constraint_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_all, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_refine(h, n, top_m=10, alpha=100000, maxiter=3000):

    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def continuous_refine(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Try different alphas
    for alpha in [5000, 20000, 50000, 100000, 200000]:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_refine(best_h, n, alpha, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    
    # Active set refinement
    for top_m in [5, 10, 15]:
        if time.time() > deadline - 5:

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    """Full optimization pipeline at a given resolution"""
    # Phase 1: Binarize and do binary local search
    h_bin = binarize(h_start, n)
    h_bin, c_bin = binary_local_search(h_bin, n, min(deadline - 30, time.time() + 80), rng)
    
    # Phase 2: Continuous refinement starting from binary
    h_cont, c_cont = continuous_refine(h_bin.astype(float), n, deadline)
    
    # Phase 3: Also try continuous refinement from the original start
    if time.time() < deadline - 30:
        h_cont2, c_cont2 = continuous_refine(h_start, n, deadline)
        if c_cont2 < c_cont:
            h_cont, c_cont = h_cont2, c_cont2
    
    return h_cont, c_cont
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = (h >= 0.5).astype(float)
    current_sum = h_bin.sum()
    target = n / 2.0
    if current_sum > target:
        excess = int(current_sum - target)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order[:excess]]] = 0
    elif current_sum < target:
        deficit = int(target - current_sum)
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:deficit]]] = 1
    return h_bin

def _lag_delta(h, i, j, m, n):
    dx_local = 2.0 / n
    delta = 0.0
    p1 = i + m
    p2 = i - m
    p3 = j + m
    p4 = j - m
    if 0 <= p1 < n:
        delta += h[p1]
    if 0 <= p2 < n:
        delta += h[p2]
    if 0 <= p3 < n:
        delta -= h[p3]
    if 0 <= p4 < n:
        delta -= h[p4]
    return delta * dx_local

        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

                val += h[p1]
            if 0 <= p2 < n:
                val += h[p2]
            s1[idx] = val

        s0 = np.zeros(len(zeros))
        for idx, j in enumerate(zeros):
            val = 0.0
            p1 = j + m_star
            p2 = j - m_star
            if 0 <= p1 < n:
                val += h[p1]
            if 0 <= p2 < n:
                val += h[p2]
            s0[idx] = val

def _smooth_obj_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

    dx = 2.0 / n

    shifted = alpha * (active - m)

    if mx > 500:
```
