        if 0 <= i + lag < n:
            term_i -= h[i + lag]
        # dc/dh[j]:

        if 0 <= j + lag < n:

    # h[j]-δ <= 1 → δ >= h[j]-1

    # So: δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]

def compute_slopes(h, n, i, j):
    """Compute s[k] = dc[k]/dh[i] - dc[k]/dh[j] for all k."""
    dx = 2.0/n
    
    def dc_dp(p):
        """Compute dc[lag]/dh[p] for all lags."""
        result = np.zeros(2*n-1)
        # term1: (1-h[p-lag]) for valid lag
        # p-lag in [0, n-1] → lag in [p-(n-1), p]
        lo1 = max(0, p-(n-1) + (n-1))  # index in result array = lag + (n-1)
        hi1 = min(2*n-1, p + (n-1) + 1)
        # Actually let me think in terms of the result array index
        # result[k] corresponds to lag = k-(n-1)
        # We want: for lag where 0<=p-lag<n: result[lag+(n-1)] += (1-h[p-lag])
        # lag range: p-(n-1) to p, clipped to [-(n-1), n-1]
        lag_lo = max(p-(n-1), -(n-1))
        lag_hi = min(p, n-1)
        if lag_lo <= lag_hi:
            lags = np.arange(lag_lo, lag_hi+1)
            result[lags + (n-1)] += (1 - h[p - lags])
        
        # term2: -h[p+lag] for valid lag
        # p+lag in [0, n-1] → lag in [-p, n-1-p]
        lag_lo2 = max(-p, -(n-1))
        lag_hi2 = min(n-1-p, n-1)
        if lag_lo2 <= lag_hi2:
            lags = np.arange(lag_lo2, lag_hi2+1)
            result[lags + (n-1)] -= h[p + lags]
        
        return result
    
    return dx * (dc_dp(i) - dc_dp(j))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def sum_fun(x):
        return x.sum() - target
    def sum_jac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def efficient_pair_search(h, n, deadline, num_sweeps=5):
    """Efficient pair search using gradient information."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(num_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        improved = False
        
        # Identify top lags
        top_lags = np.argsort(c)[-10:]
        
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            
            # Compute dc/dh[i] for all lags
            di = np.zeros(2*n-1)
            for k in range(2*n-1):
                lag = k - (n-1)
                if 0 <= i - lag < n:
                    di[k] += (1 - h[i - lag])
                if 0 <= i + lag < n:
                    di[k] -= h[i + lag]
            di *= dx
            
            # For each j in top_lags neighborhood, find optimal delta
            # Actually, let's just try j = (i+1)%n, (i+10)%n, (i+50)%n, i//2, etc.
            candidates_j = [(i + d) % n for d in [1, 2, 5, 10, 25, 50, 100, n//2]]
            
            best_delta = 0.0
            best_val = c.max()
            best_j = -1
            
            for j in set(candidates_j):
                if j == i:
                    continue
                # Compute dc/dh[j]
                dj = np.zeros(2*n-1)
                for k in range(2*n-1):
                    lag = k - (n-1)
                    if 0 <= j - lag < n:
                        dj[k] += (1 - h[j - lag])
                    if 0 <= j + lag < n:
                        dj[k] -= h[j + lag]
                dj *= dx
                
                s = di - dj  # slope of c[k](δ)
                
                # Feasible range for δ
                d_lo = max(-h[i], h[j] - 1)
                d_hi = min(1 - h[i], h[j])
                if d_lo >= d_hi:
                    continue
                
                # Find optimal δ: minimize max_k(c[k] + δ*s[k]) over δ in [d_lo, d_hi]
                # Quick approach: check a few candidates
                cands = [0.0, d_lo, d_hi]
                # Add kink points for top lags
                top = np.argsort(c)[::-1][:8]
                for a in top:
                    for b in top:
                        if a == b or abs(s[a] - s[b]) < 1e-12:
                            continue
                        dk = (c[b] - c[a]) / (s[a] - s[b])
                        if d_lo < dk < d_hi:
                            cands.append(dk)
                
                for dk in cands:
                    val = (c + dk * s).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = dk
                        best_j = j
            
            if best_j >= 0 and abs(best_delta) > 1e-10:
                h[i] += best_delta
                h[best_j] -= best_delta
                h = np.clip(h, 0, 1)
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    improved = True
                else:
                    # Revert
                    h[i] -= best_delta
                    h[best_j] += best_delta
                    h = project_feasible(h, n)
        
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def _sf(x):
        return x.sum() - target
    def _sj(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_gradient_cmax(h, n, top_k=20):
    """Compute subgradient of max correlation using top-k lags."""
    c = compute_c5_all(h, n)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:top_k]
    gap = c.max() - c[idx[-1]]
    if gap < 1e-10:
        w = np.ones(top_k) / top_k
    else:
        weights = np.exp(200 * (c[idx] - c.max()))
        weights /= weights.sum()
        w = weights
    grad = np.zeros(n)
    for ii, ki in enumerate(idx):
        lag = ki - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        m = np.arange(lo, hi)
        grad[m + lag] += w[ii] * dx * (1 - h[m])
        grad[m] += w[ii] * dx * (-h[m + lag])
    return grad, c

def grad_based_local_search(h, n, deadline, steps_per_sweep=5):
    """Local search using gradient to guide coordinate changes."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(steps_per_sweep * 20):
        if time.time() > deadline:
            break
        grad, c = compute_gradient_cmax(h, n, top_k=15)
        improved = False
        
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if grad[i] < -1e-10:
                # Want to increase h[i]; find j with grad[j] > 0 to decrease
                j_cands = np.argsort(-grad)[:10]
                for j in j_cands:
                    if j == i:
                        continue
                    # Find optimal delta for h[i]+=d, h[j]-=d
                    d_lo = max(-h[i], h[j] - 1)
                    d_hi = min(1 - h[i], h[j])
                    if d_lo >= d_hi - 1e-10:
                        continue
                    # Try a few candidate deltas
                    cands_d = [min(d_hi, 0.01), min(d_hi, 0.005), min(d_hi, 0.002),
                               min(d_hi, 0.001), (d_lo + d_hi) / 2]
                    best_d = 0.0
                    best_val = c.max()
                    for d in cands_d:
                        if d_lo <= d <= d_hi:
                            hi_new = h[i] + d
                            hj_new = h[j] - d
                            # Quick estimate using linear approximation
                            est = c.max() + d * grad[i] + d * grad[j]
                            if est < best_val - 1e-12:
                                # Verify with exact computation
                                h[i] = hi_new
                                h[j] = hj_new
                                hc = compute_c5(h, n)
                                if hc < best_val - 1e-12:
                                    best_val = hc
                                    best_d = d
                                else:
                                    h[i] -= d
                                    h[j] += d
                            else:
                                h[i] -= d
                                h[j] += d
                    if best_d > 1e-10:
                        h[i] += best_d
                        h[j] -= best_d
                        h = np.clip(h, 0, 1)
                        h = project_feasible(h, n)
                        new_c = compute_c5(h, n)
                        if new_c < best_c - 1e-14:
                            best_c = new_c
                            improved = True
                        else:
                            h[i] -= best_d
                            h[j] += best_d
                            h = project_feasible(h, n)

    return h, best_c

    best_h = h.copy()
    best_c = compute_c5(h, n)
    remaining = deadline - time.time() - 10

        return best_h, best_c
