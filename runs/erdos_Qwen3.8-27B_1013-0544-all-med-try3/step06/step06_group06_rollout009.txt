def weighted_sum_optimize(h, n, deadline, max_iter=5000):
    """
    Optimize weighted sum of g values, updating weights to focus on worst lags.
    """
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    L = 2 * n - 1
    
    # Initialize weights
    beta = 1.0
    g = compute_all_g(h, n)
    m = g.max()
    w = np.exp(beta * g / (m + 1e-10))
    w /= w.sum()
    
    step = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute gradient of weighted sum
        grad = np.zeros(n)
        for k in range(L):
            if w[k] < 1e-10:
                continue
            grad += w[k] * grad_at_lag(h, n, k)
        
        # Take step with projection
        h_try = project_box_sum(h - step * grad, target)
        c_try = compute_c5(h_try, n)
        
        if c_try < best_c:
            h = h_try
            best_c = c_try
            best_h = h.copy()
            step = min(step * 1.02, 0.5)
        else:
            # Try smaller step
            h_try2 = project_box_sum(h - step * 0.5 * grad, target)
            c_try2 = compute_c5(h_try2, n)
            if c_try2 < best_c:
                h = h_try2
                best_c = c_try2
                best_h = h.copy()
            else:
                step *= 0.95
        
        # Update weights every 100 iterations
        if (it + 1) % 100 == 0:
            beta *= 2.0
            beta = min(beta, 1000.0)
            g = compute_all_g(h, n)
            m = g.max()
            if m > 1e-15:
                w = np.exp(beta * g / m)
                w /= w.sum()
    
    return best_h, best_c

def targeted_local_search(h, n, deadline, max_sweeps=100):
    """
    Local search that targets the worst lag.
    For the worst lag k*, try adjusting pairs (i, i+k*) to reduce overlap.
    """
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        g = compute_all_g(h, n)
        k_star = int(np.argmax(g))
        j = k_star - (n - 1)  # actual lag
        
        improved = False
        if j > 0:
            # g(j) = sum_{i=0}^{n-1-j} h[i](1-h[i+j]) dx
            # To reduce this, we want to make h[i] small or h[i+j] large
            # Try: decrease h[i] and increase h[i+j] for i where h[i] is large and h[i+j] is small
            for i in range(0, n - j):
                if time.time() > deadline:
                    break
                if h[i] > 0.01 and h[i+j] < 0.99:
                    # Try to move mass from h[i] to h[i+j]
                    delta = min(h[i], 1 - h[i+j], 0.05)
                    h[i] -= delta
                    h[i+j] += delta
                    h = project_box_sum(h, target)
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h = best_h.copy()  # revert
        elif j < 0:
            jp = -j
            for i in range(0, n - jp):
                if time.time() > deadline:
                    break
                if h[i] > 0.01 and h[i+jp] < 0.99:
                    delta = min(h[i], 1 - h[i+jp], 0.05)
                    h[i] -= delta
                    h[i+jp] += delta
                    h = project_box_sum(h, target)
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h = best_h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
