    ac = alpha * c

    w = e / e.sum()

    obj = mc + np.log(e.sum()) / alpha
    return c, w, obj

The derivative f'(t) = a_k*(t) + 2λ(t+μ) where k*(t) is the active index (the one achieving the max).

def coord_desc(h, n, lam, deadline, max_sweeps=2000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c_cur = np.correlate(h, 1-h, mode='full')
        S_total = h.sum()
        top_k = np.argsort(c_cur)[::-1][:30]
        
        for i in range(n):
            hi = h[i]
            # Compute a (gradient of c w.r.t. h[i])
            a = np.empty(2*n-1)
            for idx in range(2*n-1):
                l = lags[idx]
                al = 1.0
                if 0 <= i+l < n:
                    al -= h[i+l]
                if 0 <= i-l < n:
                    al -= h[i-l]
                a[idx] = al
            
            b = c_cur - a * hi
            S_rest = S_total - hi
            
            # Candidates
            cands = [0.0, 1.0]
            for k in top_k:
                t_star = (n/2.0 - S_rest - a[k]/(2.0*lam))
                if 0.0 < t_star < 1.0:
                    cands.append(t_star)
            # Breakpoints between top lags
            for k1 in top_k:
                for k2 in top_k:
                    if abs(a[k1] - a[k2]) > 1e-12:
                        t_bp = (b[k2] - b[k1]) / (a[k1] - a[k2])
                        if 0.0 < t_bp < 1.0:
                            cands.append(t_bp)
            
            best_t = hi
            best_val = np.inf
            for t in cands:
                t = float(np.clip(t, 0.0, 1.0))
                c_test = c_cur + a * (t - hi)
                val = c_test.max() + lam * (S_rest + t - n/2.0)**2
                if val < best_val:
                    best_val = val
                    best_t = t
            
            h[i] = best_t
            c_cur = c_cur + a * (best_t - hi)
            S_total = S_rest + best_t
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
