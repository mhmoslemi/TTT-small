import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def penalty_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    return smooth_max_val(x, n, alpha)

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, best_c):
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 5.0, 300),
        (500, 10.0, 500),
        (2000, 50.0, 500),
        (10000, 200.0, 1000),
        (50000, 1000.0, 2000),
        (200000, 5000.0, 3000),
        (1000000, 20000.0, 5000),
    ]
    for alpha, mu, maxiter in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-17,
                                   'gtol': 1e-14, 'maxfun': maxiter * 20,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def subgradient_descent(h, n, deadline, best_c, alpha=100.0):
    """Use top-k gradient with backtracking line search."""
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    best_h = h.copy()
    k_top = 3
    for iteration in range(5000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-k_top:]
        # Average gradient of top k lags
        g = np.zeros(n)
        for m in top_idx:
            kk = m - (n - 1)
            if kk >= 0:
                nv = n - kk
                g[0:nv] += (1.0 - h[kk:n])
                g[kk:n] += (-h[0:nv])
            else:
                kk2 = -kk
                nv = n - kk2
                g[kk2:n] += (1.0 - h[0:nv])
                g[0:nv] += (-h[kk2:n])
        g /= k_top
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        # Backtracking line search
        lr = 0.05
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            # Try slightly perturbed direction
            for _ in range(5):
                pert = np.random.randn(n) * 0.01
                g2 = g + 0.1 * pert / np.linalg.norm(pert + 1e-30)
                h_new = project_feasible(h - 0.01 * g2, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
                    break
            else:
                if lr < 1e-10:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, best_c, radius=None):
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline - 5:
            break
        improved_in_pass = True
        while improved_in_pass and time.time() < deadline - 5:
            improved_in_pass = False
            for i in range(n):
                if time.time() > deadline - 5:
                    return best_h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved_in_pass = True
                    else:
                        h[i] -= d
                        h[j] += d
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved_in_pass = True
                    else:
                        h[i] += d
                        h[j] -= d
            if time.time() > deadline - 5:
                break
        # Reset h to best found
        h = best_h.copy()
    return best_h, best_c

def gradient_pair_search(h, n, deadline, best_c, alpha=5000.0, n_inner=3):
    """For each pair, use gradient to determine direction and do a small 1D search."""
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    best_h = h.copy()
    radius = max(30, n // 5)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > deadline - 5:
            break
        improved_any = True
        while improved_any and time.time() < deadline - 5:
            improved_any = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline - 5:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Determine direction based on gradient
                    if g[i] > g[j]:
                        direction = -1  # h[i] -= step, h[j] += step
                    else:
                        direction = 1   # h[i] += step, h[j] -= step
                    if direction == -1:
                        if h[i] - step < 0 or h[j] + step > 1:
                            continue
                        h[i] -= step
                        h[j] += step
                    else:
                        if h[i] + step > 1 or h[j] - step < 0:
                            continue
                        h[i] += step
                        h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved_any = True
                    else:
                        if direction == -1:
                            h[i] += step
                            h[j] -= step
                        else:
                            h[i] -= step
                            h[j] += step
            h = best_h.copy()
            alpha *= 2.0
    return best_h, best_c
