    def update_corr(corr, h, i, n, old_v, new_v):
        """Incrementally update correlation after changing h[i]."""
        d_nonzero, non_zero = precompute_coord_data(h, i, n)
        # Note: precompute uses current h, but for k≠0, h[i] doesn't appear in d_k
        # So this is correct regardless of whether h[i] has been updated.
        delta = new_v - old_v
        corr[non_zero] += delta * d_nonzero
        corr[n-1] += new_v * (1.0 - new_v) - old_v * (1.0 - old_v)

    def coord_descent(h, n, max_sweeps, t0, time_limit):
        """Run coordinate descent. Returns (best_h, best_c5)."""
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        best_c5 = float(np.max(corr) * dx)
        best_h = h.copy()

            if time.time() - t0 > time_limit:
                break

            any_improved = False

                if time.time() - t0 > time_limit:
                    break

                    h[i] = new_v

                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                    any_improved = True
            
            if not any_improved:
                break
        
        return best_h, best_c5
