```python
def compute_lse_grad(h, n, alpha):
    dx = 2.0 / n
    v = 1.0 - h
    g = np.correlate(h, v, mode='full') * dx  # length 2n-1
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()  # length 2n-1
    
    # S1[j] = sum_k w[k] * v[j+k-(n-1)] for valid indices
    # S2[j] = sum_k w[k] * h[j-k+(n-1)] for valid indices
    # Vectorize: create index matrices
    j = np.arange(n)  # length n
    k = np.arange(2*n-1)  # length 2n-1
    J, K = np.meshgrid(j, k, indexing='ij')  # both n x (2n-1)
    
    v_idx = J + K - (n-1)  # index into v
    h_idx = J - K + (n-1)  # index into h
    
    v_valid = (v_idx >= 0) & (v_idx < n)
    h_valid = (h_idx >= 0) & (h_idx < n)
    
    # S1[j] = sum_k w[k] * v[v_idx] * v_valid
    S1 = np.sum(w[None, :] * np.where(v_valid, v[np.clip(v_idx, 0, n-1)], 0), axis=1)
    S2 = np.sum(w[None, :] * np.where(h_valid, h[np.clip(h_idx, 0, n-1)], 0), axis=1)
    
    grad = dx * (S1 - S2)
    return grad
```

```python
    dx = 2.0 / n
    v = 1.0 - h
    g = np.correlate(h, v, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    
    # S1[j] = sum_k w[k] * v[j+k-(n-1)] for valid v indices
    # S2[j] = sum_k w[k] * h[j-k+(n-1)] for valid h indices
    
    # Use broadcasting
    j_idx = np.arange(n)[:, None]  # n x 1
    k_idx = np.arange(2*n-1)[None, :]  # 1 x (2n-1)
    
    v_idx = j_idx + k_idx - (n - 1)  # n x (2n-1)
    h_idx = j_idx - k_idx + (n - 1)  # n x (2n-1)
    
    v_mask = (v_idx >= 0) & (v_idx < n)
    h_mask = (h_idx >= 0) & (h_idx < n)
    
    v_vals = np.where(v_mask, v[np.clip(v_idx, 0, n-1)], 0.0)
    h_vals = np.where(h_mask, h[np.clip(h_idx, 0, n-1)], 0.0)
    
    S1 = (w[None, :] * v_vals).sum(axis=1)
    S2 = (w[None, :] * h_vals).sum(axis=1)
    
    return dx * (S1 - S2)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Force feasibility
        if diff > 0:
            idx = np.argsort(1.0 - h)
            need = diff
            for i in idx:
                add = min(need, 1.0 - h[i])
                h[i] += add
                need -= add
                if need < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            need = -diff
            for i in idx:
                rem = min(need, h[i])
                h[i] -= rem
                need -= rem
                if need < 1e-14:
                    break
    return h

def lse_obj_and_grad(h, n, alpha):
    """Compute LSE smooth max objective and its gradient."""
    dx = 2.0 / n
    v = 1.0 - h
    g = np.correlate(h, v, mode='full') * dx  # length 2n-1
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_sh = np.exp(shifted - sm)
    Z = exp_sh.sum()
    obj = m + (sm + np.log(Z)) / alpha
    w = exp_sh / Z  # softmax weights
    
    # Gradient computation (vectorized O(n^2))
    j_idx = np.arange(n)[:, None]  # n x 1
    k_idx = np.arange(2 * n - 1)[None, :]  # 1 x (2n-1)
    v_idx = j_idx + k_idx - (n - 1)  # n x (2n-1)
    h_idx = j_idx - k_idx + (n - 1)  # n x (2n-1)
    
    v_mask = (v_idx >= 0) & (v_idx < n)
    h_mask = (h_idx >= 0) & (h_idx < n)
    
    v_vals = np.where(v_mask, v[np.clip(v_idx, 0, n - 1)], 0.0)
    h_vals = np.where(h_mask, h[np.clip(h_idx, 0, n - 1)], 0.0)
    
    S1 = (w[None, :] * v_vals).sum(axis=1)
    S2 = (w[None, :] * h_vals).sum(axis=1)
    
    grad = dx * (S1 - S2)
    return obj, grad

def lbfgsb_optimize(h0, n, alpha, penalty, maxiter, deadline):
    """L-BFGS-B optimization with LSE + penalty for sum constraint."""
    def obj_and_grad(x):
        obj, grad = lse_obj_and_grad(x, n, alpha)
        # Add penalty for sum constraint
        sum_v = x.sum() - n / 2.0
        obj += penalty * sum_v ** 2
        grad += 2.0 * penalty * sum_v
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                      bounds=bounds, jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        result = project_feasible(res.x, n)
        return result
    except Exception:
        return project_feasible(h0, n)

def nesterov_subgrad(h0, n, deadline, max_iter=20000):
    """Nesterov-accelerated subgradient method on exact max."""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    lr = 0.5  # initial learning rate
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        g = compute_all_g(y, n)
        m = g.max()
        ks = int(np.argmax(g))
        
        # Subgradient at the maximizing shift
        dx = 2.0 / n
        v = 1.0 - y
        # grad[j] = dx * (v[j+k-(n-1)] - y[j-k+(n-1)]) for the maximizing k
        k = ks
        grad = np.zeros(n)
        # v[j+k-(n-1)] for valid j
        v_start = max(0, n - 1 - k)  # first j where j+k-(n-1) >= 0
        v_end = min(n, k + 1)  # last j where j+k-(n-1) < n, so j < n-k+n-1... 
        # Actually: 0 <= j+k-(n-1) < n => n-1-k <= j < n+k-1, and 0 <= j < n
        v_lo = max(0, n - 1 - k)
        v_hi = min(n, n + k - 1)
        if v_hi > v_lo:
            jj = np.arange(v_lo, v_hi)
            grad[jj] += dx * v[jj + k - (n - 1)]
        
        h_start = max(0, 1 - k + n - 1)  # 0 <= j-k+(n-1) < n => k-(n-1) <= j < k+n-1, and 0<=j<n
        h_lo = max(0, k - (n - 1))
        h_hi = min(n, k + n - 1)
        if h_hi > h_lo:
            jj = np.arange(h_lo, h_hi)
            grad[jj] -= dx * y[jj - k + (n - 1)]
        
        # Nesterov update
        h_new = project_feasible(y - lr * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t - 1.0) / t_new * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        # Decay learning rate
        lr *= (1.0 + it) ** (-0.5) if it > 0 else 1.0
        lr = max(lr, 1e-6)
        
        # Track best
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=100):

    target = n // 2
    # Start from the thresholded version

    zeros = set(range(n)) - ones

        if time.time() > deadline:
            break
        improved = False
        # Try moving mass from a 0 to a 1 (swap)
        zero_list = list(zeros)
        one_list = list(ones)
        for i in zero_list:
            if time.time() > deadline:
                break
            best_j = None
            for j in one_list:
                h_bin[i] = 1.0

                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    improved = True
                    one_list = list(ones)
                    zero_list = list(zeros)
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    
    return h_bin, best_c

def pair_refine(h, n, deadline, deltas):
    """Try small perturbations of pairs of values."""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(0, n, 2):  # stride 2 for speed
                if time.time() > deadline:
                    break
                for j in range(n - 1, max(i - 1, -1), -2):
                    if i == j:
                        continue
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c
```
