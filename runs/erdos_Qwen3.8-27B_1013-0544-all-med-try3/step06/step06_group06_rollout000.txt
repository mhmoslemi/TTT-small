            if s < 1e-12:

            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0

    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_best_improvement(h, n, deadline, max_sweeps=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improvement = 0
        best_swap = None
        
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c = compute_c5(h, n)
                h[i] = 0.0
                h[j] = 1.0
                improvement = current_c - c
                if improvement > best_improvement:

                    best_swap = (i, j)
        
        if best_swap is None:

        i, j = best_swap

        h[j] = 0.0

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_best_improvement(h, n, deadline, max_sweeps=5000):
    h = h.copy()
    best_c = compute_c5(h, n)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        current_c = best_c
        best_improvement = 0
        best_swap = None

        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c = compute_c5(h, n)
                h[i] = 0.0
                h[j] = 1.0
                improvement = current_c - c
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)

        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 1.0
        h[j] = 0.0
        best_c = compute_c5(h, n)

    return h, best_c

def sa_binary(h, n, deadline, rng, max_iter=1000000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    T = 0.04
    cool_every = 10000
    cool_factor = 0.92

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % cool_every == 0:
            T *= cool_factor

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]

        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h, n)

        delta = new_c - current_c
        if delta <= 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0

    return best_h, best_c

def multi_resolution_search(best_h, best_c, n_current, deadline, rng):
    """Try multiple resolutions and return the best solution found."""
    for nh in [200, 250, 300, 350, 400, 500]:
        if time.time() > deadline:
            break
        if nh == n_current:
            continue
        candidates = []

        # Candidate 1: interpolate from current best
        xo = np.linspace(0, 2, n_current)
        xn = np.linspace(0, 2, nh)
        h_interp = np.interp(xn, xo, best_h)
        candidates.append(to_binary(h_interp, nh))

        # Candidate 2: random
        h_rand = (rng.rand(nh) < 0.5).astype(float)
        candidates.append(to_binary(h_rand, nh))

        # Candidate 3: structured (balanced spread)
        h_struct = np.zeros(nh)
        for k in range(nh // 2):
            h_struct[(2 * k + 1) % nh] = 1.0
        candidates.append(h_struct)

        for cand in candidates:
            if time.time() > deadline - 5:
                break
            sub_dl = min(deadline - 3, time.time() + 45)
            h_test, c_test = binary_best_improvement(cand, nh, sub_dl, max_sweeps=2000)
            if c_test < best_c:
                best_c = c_test
                best_h = h_test.copy()
                n_current = nh

    return best_h, best_c, n_current
