def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):

        if abs(diff) < 1e-14:

            s = room.sum()
            if s < 1e-14:

            if s < 1e-14:

    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n

        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]

            b = ones_list[rng.randint(len(ones_list))]

                        cur_c = nc
                        zeros_list = np.where(h == 0)[0]
                        ones_list = np.where(h == 1)[0]
            
            T *= cooling
        
    return best_h, best_c
