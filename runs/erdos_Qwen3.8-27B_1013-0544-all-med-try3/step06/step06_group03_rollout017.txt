    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            if tot < 1e-15:

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # c has length 2n-1, indexed from -(n-1) to (n-1)
    # c[n-1] corresponds to lag 0
    # c[n-1+k] corresponds to lag k
    idx = np.argsort(c)[-top_m:]  # top m indices
    grad = np.zeros(n)
    for k_idx in idx:

        # ∇g_k(h)[j] = dx * (1-h[j+k]) * [0≤j+k<n] - dx * h[j-k] * [0≤j-k<n]

                g += (1.0 - h[j + k])
            if 0 <= j - k < n:
                g -= h[j - k]
            grad[j] += g
    grad *= dx / len(idx)
    return grad

For k < 0 (let k = -m, m > 0):

  - Second term: -h[j+m] for j = 0, ..., n-1-m; 0 for j = n-m, ..., n-1

def subgradient_of_max(h, n, top_m=5):

Let me also consider: **bundle methods** or **cutting plane methods** for the minimax problem. These are more sophisticated but might converge faster near the optimum.

For a cutting plane approach:
- Maintain a set of "cuts" (affine underestimators of the max function)
- At each iteration, solve a QP using the cuts
- Evaluate the true function at the solution
- Add a new cut
- Repeat

The cuts are: for each previously evaluated point h_t, we have g_{k_t}(h_t) as a lower bound on max g_k, and ∇g_{k_t}(h_t) as the normal to the cut.
Cut: f(h) ≥ f(h_t) + ∇g_{k_t}(h_t)^T (h - h_t)

The QP is:
min C
s.t. C ≥ f(h_t) + ∇g_{k_t}(h_t)^T (h - h_t) for all cuts t
0 ≤ h ≤ 1
sum h = n/2

This is a linear program in (h, C)! With 2n+1 variables and |cuts| + n + 1 constraints. If |cuts| is, say, 50, we have 251 variables and 301 constraints. This can be solved with scipy's linprog or CVXOPT.

But wait, the cuts are only valid lower bounds if g_{k_t} is a valid lower bound on max g_k, which is true (since g_{k_t} ≤ max g_k). So this is a valid cutting plane method.

The issue is that with only linearized cuts, we might not converge to the exact optimum (the cuts approximate the max from below, and the approximation can be poor far from the evaluation points). But near the optimum, it should work well.

Let me implement a simple cutting plane method:

```python
def cutting_plane(h0, n, deadline, max_iter=500):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    cuts = []  # list of (c_val, grad_vec, h_point)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Solve LP: min C s.t. C ≥ c_t + g_t^T (h - h_t) for all cuts, 0≤h≤1, sum h = n/2
        # Variables: h[0],...,h[n-1], C (total n+1 variables)
        # Objective: minimize C
        # Constraints:
        #   C - g_t^T h ≥ c_t - g_t^T h_t for each cut t (rewrite as -C + g_t^T h ≤ g_t^T h_t - c_t)
        #   -h[i] ≤ 0 for all i
        #   h[i] ≤ 1 for all i
        #   sum h = n/2
        
        from scipy.optimize import linprog
        
        m_cuts = len(cuts)
        # c_obj = [0,0,...,0, 1] (minimize C, which is the last variable)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A_ub @ x ≤ b_ub
        # C - g_t^T h ≥ c_t - g_t^T h_t  =>  -C + g_t^T h ≤ g_t^T h_t - c_t
        A_ub = np.zeros((m_cuts + 2*n, n + 1))
        b_ub = np.zeros(m_cuts + 2*n)
        
        for t, (c_val, g_vec, h_pt) in enumerate(cuts):
            A_ub[t, :n] = g_vec
            A_ub[t, n] = -1.0
            b_ub[t] = g_vec @ h_pt - c_val
        
        # -h[i] ≤ 0
        A_ub[m_cuts:m_cuts+n, :n] = -np.eye(n)
        b_ub[m_cuts:m_cuts+n] = 0
        
        # h[i] ≤ 1
        A_ub[m_cuts+n:m_cuts+2*n, :n] = np.eye(n)
        b_ub[m_cuts+n:m_cuts+2*n] = 1.0
        
        # Equality: sum h = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0,1)]*n + [(0, None)], method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        
        # Evaluate true objective
        c_true = compute_c5(h_new, n)
        
        if c_true < best_c - 1e-14:
            best_c = c_true
            best_h = h_new.copy()
        
        # Find the active lag at h_new
        c_vals = np.correlate(h_new, 1 - h_new, mode='full') * (2.0/n)
        k_active = np.argmax(c_vals)
        k = k_active - (n - 1)
        g_active = compute_grad_k(h_new, n, k)
        
        # Add cut
        cuts.append((c_true, g_active, h_new.copy()))
        
        # Limit number of cuts
        if len(cuts) > 100:
            cuts = cuts[-100:]
        
        # Check convergence
        if abs(res.x[n] - c_true) < 1e-6:
            # The LP relaxation is tight, we're at the optimum
            break
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
    else:
        m = -k
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad * dx

def subgradient_of_max(h, n, top_m=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)
        grad += compute_grad_k(h, n, k)
    return grad / top_m

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, deadline, top_m=5, max_iter=200000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t0 = time.time()
    for t in range(max_iter):
        if time.time() > deadline:
            break
        eta = 0.005 / (1.0 + 0.005 * t)
        grad = subgradient_of_max(h, n, top_m)
        h = h - eta * grad
        h = project_feasible(h, n)
        if t % 200 == 0:
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def sliding_window_optimize(h, n, B, deadline, max_passes=50):
    best_c = compute_c5(h, n)
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = 1
        for p in range(0, n - B + 1, step):
            if time.time() > deadline:
                return h, best_c
            block = h[p:p+B].copy()
            target_sum = block.sum()
            
            def win_obj(x):
                h_temp = h.copy()
                np.clip(x, 0.0, 1.0, out=x)
                h_temp[p:p+B] = x
                return obj_smooth(h_temp, n, 50000)
            
            def win_sum(x):
                return float(x.sum() - target_sum)
            
            def win_sum_jac(x):
                return np.ones(B)
            
            try:
                r = minimize(win_obj, block, method='SLSQP',
                            bounds=[(0.0, 1.0)] * B,
                            constraints=[{'type': 'eq', 'fun': win_sum, 'jac': win_sum_jac}],
                            options={'maxiter': 50, 'ftol': 1e-13})
                x_new = project_feasible(r.x, B) if B >= 2 else np.clip(r.x, 0, 1)
                # Re-project to maintain exact sum
                x_new = np.clip(r.x, 0.0, 1.0)
                diff = target_sum - x_new.sum()
                if abs(diff) > 1e-14:
                    if diff > 0:
                        room = 1.0 - x_new
                        tot = room.sum()
                        if tot > 1e-15:
                            x_new += (diff / tot) * room
                    else:
                        tot = x_new.sum()
                        if tot > 1e-15:
                            x_new += (diff / tot) * x_new
                    x_new = np.clip(x_new, 0.0, 1.0)
                
                h_new = h.copy()
                h_new[p:p+B] = x_new
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    h = h_new
                    best_c = c_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, best_c

def pair_search_local(h, n, deadline, deltas, radius=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                            improved = True
                        else:

            if not improved:

    # Phase 2: Try other resolutions (30%)
    for n_try in [200, 300, 400]:
        if time.time() > deadline - 250:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.15)
        if alloc < 20:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multi-restart at best resolution (30%)
    n_best = len(best_h)
    if time.time() < deadline - 100:
        remaining = deadline - time.time() - 30
        n_restarts = max(4, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Sliding window polish (10%)
    if time.time() < deadline - 30:
        n = len(best_h)
        B = min(8, n)
        h_opt, c_opt = sliding_window_optimize(best_h, n, B,
                                               min(time.time() + 100, deadline - 10))
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Final aggressive SLSQP + local search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 10000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            best_h, best_c = pair_search_local(best_h, n, deadline - 0.5,
                                               [0.001, 0.0005, 0.0002, 0.0001],
                                               radius=40)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
