    n = len(h)
    if shuffle_order is None:
        order = list(range(n))
    else:
        order = shuffle_order

        old = h[i]
        h[i] = 0.0
        base = np.correlate(h, 1.0 - h, mode='full').astype(float)
        h[i] = 1.0
        one = np.correlate(h, 1.0 - h, mode='full').astype(float)
        h[i] = old

        # Compute weighted gradient

                g[idx + la] -= h[idx]
