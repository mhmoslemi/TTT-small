We want to minimize max_k ∫ h(x)(1-h(x+k)) dx = max_k (∫h - ∫h(x)h(x+k)dx) = max_k (1 - ∫h(x)h(x+k)dx)

(since ∫₀² h(x)dx = 1)

So we want to maximize min_k ∫h(x)h(x+k)dx.

This is the minimum of the autocorrelation of h. We want h to have high minimum autocorrelation.

For a function with mean 1/2 (since ∫₀² h = 1 over interval of length 2), the autocorrelation at lag 0 is ∫h(x)² dx. By Cauchy-Schwarz, this is ≥ (∫h)²/2 = 1/2, with equality when h is constant (h=1/2).

At other lags, the autocorrelation can be positive or negative. We want it to be as high as possible at all lags.

A function that is "spread out" (like a triangle or Gaussian) will have high autocorrelation at small lags but low at large lags. A periodic function will have high autocorrelation at multiples of the period but low elsewhere.

The optimal solution likely has a trade-off: some structure that balances autocorrelation across all lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    a = shifted.max()
    return float(m + (a + np.log(np.sum(np.exp(shifted - a)))) / alpha)

def smooth_max_probs(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    a = shifted.max()
    logsum = a + np.log(np.sum(np.exp(shifted - a)))
    p = np.exp(shifted - logsum)
    return p, c

def compute_gradient(x, n, alpha):
    dx = 2.0 / n
    p, c = smooth_max_probs(x, n, alpha)
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(len(c)):
        pk = p[k_idx]
        if pk < 1e-14:
            continue
        s = k_idx - n1
        if s == 0:
            grad += pk * dx * (1.0 - 2.0 * x)
        elif s > 0:
            grad[s:] += pk * dx * (1.0 - 2.0 * x[:n - s])
            grad[:n - s] += pk * dx * (-x[s:])
        else:
            sa = -s
            grad[:n - sa] += pk * dx * (1.0 - 2.0 * x[sa:])
            grad[sa:] += pk * dx * (-x[:n - sa])
    return grad / alpha

def gradient_descent(h, n, alpha, deadline, lr_init=0.02):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stall_count = 0
    
    while time.time() < deadline:
        grad = compute_gradient(h, n, alpha)
        grad_centered = grad - grad.mean()
        
        h_new = h - lr * grad_centered
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall_count = 0
            lr = min(lr * 1.02, 0.08)
        else:
            h = h_new
            stall_count += 1
            lr *= 0.95
            if stall_count > 50 or lr < 1e-10:
                break
    return best_h, best_c

        return smooth_max_value(x, n, alpha)
    
    def obj_grad(x):
        return compute_gradient(x, n, alpha)

        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    current_obj = smooth_max_value(h, n, alpha)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        ht = project_feasible(res.x, n)
        ct = compute_c5(ht, n)
        if ct < current_obj * 1.0001:
            return ht
    except Exception:
        pass
    return h.copy()

def gradient_guided_pair_search(h, n, deadline, max_passes=300):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    alpha_guide = 5000
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            grad = compute_gradient(h, n, alpha_guide)
            order_desc = np.argsort(-grad)
            order_asc = np.argsort(grad)
            
            improved = False
            tried = 0
            max_tries = 800
            
            for pos_i in range(n):
                if tried >= max_tries or time.time() > deadline:
                    break
                i = order_desc[pos_i]
                for pos_j in range(n - 1, max(n - 1 - 80, -1), -1):
                    if tried >= max_tries or time.time() > deadline:
                        break
                    j = order_asc[pos_j]
                    if i == j:
                        continue
                    tried += 1
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def full_pair_search(h, n, deadline, max_passes=100):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def optimize_single_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Gradient descent with increasing alpha
    alphas = [200, 1000, 5000, 20000, 100000, 400000]
    for a in alphas:
        if time.time() > deadline - 25:
            break
        t_start = time.time()
        max_time = min(12, (deadline - time.time()) - 25)
        if max_time < 2:
            break
        ht, ct = gradient_descent(best_h, n, a, t_start + max_time)
        if ct < best_c - 1e-14:
            best_c = ct
            best_h = ht.copy()
    
    # Phase 2: SLSQP refinement
    for a in [50000, 200000, 800000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    # Phase 3: Local search
    if time.time() < deadline - 5:
        if n <= 150:
            best_h, best_c = full_pair_search(best_h, n, deadline - 2, max_passes=80)
        else:
            best_h, best_c = gradient_guided_pair_search(best_h, n, deadline - 2, max_passes=200)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        # Optimize at parent resolution
        alloc1 = min(350, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_single_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
        
        # Try other resolutions
        x_init = np.linspace(0, 2, n_init)
        candidates = []
        for nc in [100, 150, 200, 300, 400, 500]:
            if nc != n_init:
                candidates.append(nc)
        candidates.sort(key=lambda x: abs(x - n_init))
        
        for n_try in candidates:
            if time.time() > deadline - 50:
                break
            remaining = deadline - time.time()
            alloc = min(100, remaining * 0.2)
            if alloc < 25:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_single_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                x_init = x_new
    
    # If no parent, start from scratch
    if best_h is None:
        n = 250
        h_start = 0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, n))
        h_start = project_feasible(h_start, n)
        best_h, best_c = optimize_single_n(h_start, n, deadline - 5, rng)
    
    # Multi-restart phase
    if time.time() < deadline - 35:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(2, int(remaining / 55))
        scales = [0.003, 0.008, 0.015, 0.03, 0.05, 0.08, 0.12]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.3)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_single_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final fine polishing
    if time.time() < deadline - 3:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = full_pair_search(best_h, n, deadline - 1, max_passes=50)
        else:
            best_h, best_c = gradient_guided_pair_search(best_h, n, deadline - 1, max_passes=200)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
