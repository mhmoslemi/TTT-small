            h_half = project_feasible(h_half, n)

            if c_half < c_current:
                h = h_half
            # else keep h as is

            h = h_new

Let me refine the approach:

1. Compute subgradient g (average of active shift gradients)
2. Project g onto the tangent space of the constraints:
   - Remove the component along the "sum" direction (to satisfy sum constraint)
   - The box constraints are handled by projection
3. If ||projected g|| < threshold, stop
4. Otherwise, take a step in the -projected_g direction with line search

The tangent space for the sum constraint is {v : sum(v) = 0}. To project g onto this space:
g_proj = g - (sum(g)/n) * ones(n)

This removes the component that would change the sum.

Then the step direction is -g_proj, and I project onto [0,1]^n after taking the step.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlaps_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    ns = 2 * n - 1
    for idx in range(ns):
        w = weights[idx]
        if w < 1e-15:
            continue
        m = idx - (n - 1)
        if m >= 0:
            ne = n - m
            if ne > 0:
                grad[0:ne] += w * (1.0 - h[m:n])
                grad[m:n] += w * (-h[0:ne])
        else:
            mp = -m
            ne = n - mp
            if ne > 0:
                grad[mp:n] += w * (1.0 - h[0:ne])
                grad[0:ne] += w * (-h[mp:n])
    grad *= dx
    return grad

def smooth_max_val_grad(h, n, alpha, pen_mult):
    dx = 2.0 / n
    overlaps = compute_overlaps(h, n)
    m = overlaps.max()
    shifted = np.minimum(alpha * (overlaps - m), 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    weights = exp_vals / exp_sum
    val = m + np.log(exp_sum) / alpha
    pen = pen_mult * (h.sum() - n / 2.0) ** 2
    val += pen
    grad = overlaps_grad(h, n, weights)
    grad += 2.0 * pen_mult * (h.sum() - n / 2.0)
    return val, grad

def _obj_func(x, n, alpha, pen_mult):
    v, _ = smooth_max_val_grad(x, n, alpha, pen_mult)
    return v

def _jac_func(x, n, alpha, pen_mult):
    _, g = smooth_max_val_grad(x, n, alpha, pen_mult)
    return g

def lbfgsb_optimize(h, n, deadline):
    h = project_feasible(h.copy(), n)
    alphas = [200, 1000, 5000, 20000, 100000, 500000]
    pen_mult = 5e4
    for a in alphas:
        if time.time() > deadline - 3:
            break
        bounds = [(0.0, 1.0)] * n
        maxiter = 3000 if a < 10000 else 1500
        try:
            res = minimize(_obj_func, h, jac=_jac_func, args=(n, a, pen_mult),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def subgradient_polish(h, n, deadline, max_iter=15000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_cur = best_c
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h, n)
        c_cur = overlaps.max()
        if c_cur < best_c - 1e-15:

            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        nw = active.sum()
        weights = np.zeros(2 * n - 1)
        weights[active] = 1.0 / nw
        grad = overlaps_grad(h, n, weights)
        g_proj = grad - np.mean(grad)
        gn = np.linalg.norm(g_proj)
        if gn < 1e-13:

        direction = -g_proj / gn
        improved = False
        for s in [0.03, 0.015, 0.007, 0.003, 0.001, 0.0004, 0.0001]:
            h_try = h + s * direction
            h_try = project_feasible(h_try, n)
