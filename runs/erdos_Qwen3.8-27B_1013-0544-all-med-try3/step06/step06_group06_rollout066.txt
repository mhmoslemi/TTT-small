                    delta -= (1 - h[idx3])

                if 0 <= idx4 < n:
                    delta += h[idx4]

                    best_delta = delta

            g = np.correlate(h, 1 - h, mode='full')
            if g.max() < best_c:
                best_c = g.max() * (2.0 / n)
        else:
            break
    
    return h, best_c

def sa_binary_search(h, n, deadline, max_iter=500000, seed=42):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]

    T_min = 0.005
    cool_rate = 0.9998

            break

        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        
        # Perform swap
        h[i], h[j] = 1.0, 0.0
        new_c = compute_c5(h, n)

            cur_c = new_c
            # Update index arrays

            zeros_idx = np.where(h == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        
        T = max(T * cool_rate, T_min)
    
    return best_h, best_c
