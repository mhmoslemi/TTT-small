```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h_curr, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return dx * (1.0 - 2.0 * h_curr)
    if lag > 0:
        grad[m:] += dx * (1.0 - h_curr[:n-m])
        grad[:n-m] += -dx * h_curr[m:]
    else:
        grad[:n-m] += dx * (1.0 - h_curr[m:])
        grad[m:] += -dx * h_curr[:n-m]
    return grad

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_idx, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_accumulated(h_curr, n, deadline, max_iters=50):
    """Outer approximation with accumulated cuts."""
    h = h_curr.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Accumulated cuts: list of (h_point, c_values, gradients)
    cuts = []
    trust = 0.3
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h, n)
        if c.max() < best_c - 1e-14:
            best_c = c.max()
            best_h = h.copy()
        
        # Add new cut at current point
        cut_grads = []
        cut_consts = []
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            g = lag_gradient(h, n, lag)
            cut_grads.append(g)
            cut_consts.append(c[k])
        cuts.append((h.copy(), cut_consts, cut_grads))
        
        # Keep only last 10 cuts to avoid too many constraints
        if len(cuts) > 10:
            cuts = cuts[-10:]
        
        # Solve LP with all accumulated cuts
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        cons.append(cp.norm(h_var - h, 2) <= trust)  # trust region
        
        for (h_pt, c_vals, g_list) in cuts:
            for k in range(2 * n - 1):
                cons.append(c_vals[k] + g_list[k] @ (h_var - h_pt) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        
        step = np.max(np.abs(h_new - h))
        if step < 1e-10:
            break
        
        h = h_new
        trust = max(0.01, trust * 0.85)
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def subgradient_method(h, n, deadline, steps=500):
    """Subgradient method for min-max problem."""
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for step in range(steps):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_gradient(h_curr, n, lag)
        
        # Step size: decrease over time
        step_size = 0.1 / (1.0 + 0.01 * step)
        
        h_new = h_curr - step_size * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        
        h_curr = h_new
    
    return best_h, best_c

def local_search_gradient(h, n, deadline):
    """Local search using gradient-guided pair moves."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            
            # Compute gradient of the active lag
            g = lag_gradient(h, n, lag)
            
            # Try moves that decrease the active lag
            for i in range(n):
                if time.time() > deadline:
                    break
                if g[i] > 0:
                    # Decreasing h[i] would help
                    # Find j to increase (g[j] < 0)
                    for j in range(i+1, n, max(1, n // 50)):
                        if g[j] < 0:
                            delta = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                            if delta < 1e-9:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                            break  # Only try one j per i
                elif g[i] < 0:
                    for j in range(i+1, n, max(1, n // 50)):
                        if g[j] > 0:
                            delta = min(d, (1.0 - h[i]) * 0.3, h[j] * 0.3)
                            if delta < 1e-9:
                                continue
                            h[i] += delta
                            h[j] -= delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] -= delta
                                h[j] += delta
                            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h_curr, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return dx * (1.0 - 2.0 * h_curr)
    if lag > 0:
        grad[m:] += dx * (1.0 - h_curr[:n - m])
        grad[:n - m] += -dx * h_curr[m:]
    else:
        grad[:n - m] += dx * (1.0 - h_curr[m:])
        grad[m:] += -dx * h_curr[:n - m]
    return grad

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_idx, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_accumulated(h_curr, n, deadline, max_iters=40):
    h = h_curr.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cuts = []
    trust = 0.25
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        if c.max() < best_c - 1e-14:
            best_c = c.max()
            best_h = h.copy()
        cut_grads = []
        cut_consts = []
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            g = lag_gradient(h, n, lag)
            cut_grads.append(g)
            cut_consts.append(c[k])
        cuts.append((h.copy(), cut_consts, cut_grads))
        if len(cuts) > 8:
            cuts = cuts[-8:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        cons.append(cp.norm_cp(h_var - h) <= trust)
        for (h_pt, c_vals, g_list) in cuts:
            for k in range(2 * n - 1):
                cons.append(c_vals[k] + g_list[k] @ (h_var - h_pt) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        step = np.max(np.abs(h_new - h))
        if step < 1e-10:
            break
        h = h_new
        if c.max() > 0.35:
            trust = max(0.05, trust * 0.9)
        else:
            trust = max(0.02, trust * 0.82)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()

    step_size = 0.08
    for step in range(steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-m:]
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            g += lag_gradient(h_curr, n, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        g = g / g_norm
        h_new = h_curr - step_size * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            step_size = min(0.1, step_size * 1.05)
        else:
            step_size *= 0.7
            step_size = max(step_size, 0.001)
        h_curr = h_new
    return best_h, best_c

def local_search_pairs(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            g = lag_gradient(h, n, lag)
            for i in range(n):
                if time.time() > deadline:
                    break
                if abs(g[i]) < 1e-10:
                    continue
                for j in range(i + 1, n, max(1, n // 30)):
                    if g[i] * g[j] < 0:
                        if g[i] > 0:
                            delta = min(d, h[i] * 0.4, (1.0 - h[j]) * 0.4)
                            if delta < 1e-9:
                                continue
                            h[i] -= delta
                            h[j] += delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += delta
                                h[j] -= delta
                        else:
                            delta = min(d, (1.0 - h[i]) * 0.4, h[j] * 0.4)
                            if delta < 1e-9:
                                continue
                            h[i] += delta
                            h[j] -= delta
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] -= delta
                                h[j] += delta
                        break
    return h, best_c

def bisection_lp_method(h_curr, n, deadline, max_iters=30):
    h = h_curr.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        if c.max() < best_c - 1e-14:
            best_c = c.max()
            best_h = h.copy()
        target = c.max() - 0.002 * (1.0 - it / max_iters)
        h_var = cp.Variable(n)
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        violated = c > target - 1e-12
        n_violated = np.sum(violated)
        if n_violated == 0:
            break
        idx_violated = np.where(violated)[0][:20]
        for k in idx_violated:
            lag = k - (n - 1)
            g = lag_gradient(h, n, lag)
            cons.append(c[k] + g @ (h_var - h) <= target)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), cons)
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        if np.max(np.abs(h_new - h)) < 1e-9:
            break
        h = h_new
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 15:
        h_new = slsqp_optimize(best_h, n, 5000, maxiter=1500, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 30:
        oo_dl = time.time() + min(40, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx_accumulated(best_h, n, oo_dl, max_iters=25)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    if time.time() < deadline - 15:
        c_all = compute_all_lags(best_h, n)
        top_m = 5
        top_idx = np.argsort(c_all)[-top_m:]
        for alpha in [50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_active(best_h, n, top_idx, alpha, maxiter=2500, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                c_all = compute_all_lags(best_h, n)
                top_idx = np.argsort(c_all)[-top_m:]
    if time.time() < deadline - 10:
        sg_dl = time.time() + min(15, (deadline - time.time()) * 0.2)
        h_sg, c_sg = multi_gradient_descent(best_h, n, sg_dl, m=4, steps=200)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 50000, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 8:
        bisection_dl = time.time() + min(20, (deadline - time.time()) * 0.3)
        h_b, c_b = bisection_lp_method(best_h, n, bisection_dl, max_iters=15)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b
    if time.time() < deadline - 5:
        ls_dl = min(deadline - 2, time.time() + 25)
        h_ls, c_ls = local_search_pairs(best_h, n, ls_dl)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    n_candidates = [150, 200, 250, 300, 350, 400, 450, 500, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 70:
            break
        alloc = min(120, remaining * 0.22)
        if alloc < 40:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        h_opt_sym = project_feasible(1.0 - h_opt, n_try)
        c_opt_sym = compute_c5(h_opt_sym, n_try)
        if c_opt_sym < c_opt:
            h_opt, c_opt = h_opt_sym, c_opt_sym
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r % 3 == 0:
                scale = 0.012 * (1.0 + 0.3 * r)
                hp = best_h + rng.normal(0, scale, n)
            elif r % 3 == 1:
                hp = 1.0 - best_h + rng.normal(0, 0.008, n)
            else:
                hp = best_h + rng.normal(0, 0.005, n)
                hp[::2] += rng.normal(0, 0.01, n // 2 + 1)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 40) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 200000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        c_all = compute_all_lags(best_h, n)
        top_idx = np.argsort(c_all)[-3:]
        h_new = slsqp_active(best_h, n, top_idx, 1000000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 5:
            sg_dl = time.time() + 10
            h_sg, c_sg = multi_gradient_descent(best_h, n, sg_dl, m=3, steps=200)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
        if time.time() < deadline - 3:
            best_h, best_c = local_search_pairs(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
```
