import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def objective_and_grad(h, n, alpha, penalty):
    c = compute_correlation(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    log_sum_exp = shifted.max() + np.log(w_sum)
    obj = (m + log_sum_exp) / alpha
    # Add penalty for sum constraint
    sum_dev = h.sum() - n / 2.0
    obj += penalty * sum_dev ** 2
    
    # Gradient
    w_norm = w / w_sum
    grad = np.zeros(n)
    dx = 2.0 / n
    N = len(c)  # 2n-1
    for ki in range(N):
        wi = w_norm[ki]
        if wi < 1e-15:
            continue
        s = ki - (n - 1)
        if s >= 0:
            ns = n - s
            grad[:ns] += wi * (1 - h[s:]) * dx
            grad[s:] -= wi * h[:ns] * dx
        else:
            t = -s
            nt = n - t
            grad[t:] += wi * (1 - h[:nt]) * dx
            grad[:nt] -= wi * h[t:] * dx
    # Penalty gradient
    grad += 2 * penalty * sum_dev * np.ones(n)
    return obj, grad

def lbbgss_optimize(h, n, alpha, penalty, maxiter=2000, gtol=1e-12):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def local_pair_search(h, n, deadline, steps=None):
    """Focused local search using small pair swaps."""
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    n = len(h)
    
    for step in steps:
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try a few nearby j values
                for j in [0, 1, 2, n//4, n//2, 3*n//4, n-3, n-2, n-1]:
                    if j == i or j is None:
                        continue
                    if j < 0 or j >= n:
                        continue
                    # Try moving mass from i to j and vice versa
                    for di, dj in [(step, -step), (-step, step)]:
                        hi, hj = h[i] + di, h[j] + dj
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h[i], h[j] = hi, hj
                            h = project_feasible(h, n)
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - di, hj - dj
                                h = project_feasible(h, n)
            if not improved:
                break
    return h, best_c

def multi_alpha_optimize(h, n, deadline, rng):

    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]

        c = compute_c5(h, n)
        if c < best_c - 1e-14:

            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_exp_obj_grad(h, n, alpha, penalty, top_k=None):
    c = compute_correlation(h, n)
    N = len(c)
    if top_k is not None and top_k < N:
        idx = np.argsort(c)[-top_k:]
        c_f = c[idx]
        m = c_f.max()
        shifted = alpha * (c_f - m)
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        obj = (m + shifted.max() + np.log(w_sum)) / alpha
        w_norm = w / w_sum
    else:
        m = c.max()
        shifted = alpha * (c - m)
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        obj = (m + shifted.max() + np.log(w_sum)) / alpha
        w_norm = w / w_sum
        idx = np.arange(N)
    sum_dev = h.sum() - n / 2.0
    obj += penalty * sum_dev * sum_dev
    grad = np.zeros(n)
    dx = 2.0 / n
    for ki_pos, ki in enumerate(idx):
        wi = w_norm[ki_pos]
        if wi < 1e-15:
            continue
        s = ki - (n - 1)
        if s >= 0:
            ns = n - s
            grad[:ns] += wi * (1.0 - h[s:]) * dx
            grad[s:] -= wi * h[:ns] * dx
        else:
            t = -s
            nt = n - t
            grad[t:] += wi * (1.0 - h[:nt]) * dx
            grad[:nt] -= wi * h[t:] * dx
    grad += 2.0 * penalty * sum_dev
    return obj, grad

def lbbgss_opt(h, n, alpha, penalty, maxiter=1500, gtol=1e-10, top_k=None):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sum_exp_obj_grad, h, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
