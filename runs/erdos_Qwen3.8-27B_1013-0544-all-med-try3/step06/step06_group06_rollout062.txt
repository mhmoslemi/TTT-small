```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, n, alpha):
    """Compute logsumexp(alpha*g)/alpha and its gradient"""
    g = compute_all_g(h, n)
    ag = alpha * g
    m = ag.max()
    exp_ag = np.exp(ag - m)
    lse = (m + np.log(exp_ag.sum())) / alpha
    p = exp_ag / exp_ag.sum()
    
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for s in shifts:
        a = abs(s)
        if a >= n:
            continue
        j_range = np.arange(a, n - a)
        if len(j_range) == 0:
            continue
        pi = p[s + n - 1]
        if pi < 1e-15:
            continue
        grad[j_range] += pi * dx * ((1 - h[j_range - s]) - h[j_range + s])
    
    return lse, grad

def lbfgs_optimize(h_start, n, deadline, alphas, maxiter_each):
    """L-BFGS-B optimization with penalty for sum constraint"""
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def fun(x):
            lse, _ = lse_objective_and_grad(x, n, alpha)
            # Add penalty for sum constraint
            penalty = 1000.0 * (x.sum() - n/2.0)**2
            return lse + penalty
        
        def grad(x):
            _, g = lse_objective_and_grad(x, n, alpha)
            g = g + 2000.0 * (x.sum() - n/2.0)
            return g
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, h_start, jac=grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter_each, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_start = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def projected_gradient(h_start, n, deadline, alpha, max_iter=10000):
    """Projected gradient method with log-sum-exp"""
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n)
    h = h_start.copy()
    lr = 0.1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        _, grad = lse_objective_and_grad(h, n, alpha)
        
        # Find a good step size
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c or c_new < compute_c5(h, n):
                h = h_new
                found = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        # Else keep lr or increase slightly
    
    return best_h, best_c

def binary_sls(h_start, n, deadline, max_sweeps=100):
    """Binary local search using swap moves"""
    target = n // 2
    # Start from h_start, threshold to binary
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g_current = compute_all_g(h, n)
        current_max = g_current.max()
        
        for zi, z in enumerate(zeros):
            if time.time() > deadline:
                break
            # Try swapping z with each one
            # Precompute delta for z going from 0 to 1
            shifts = np.arange(-(n-1), n)
            delta_z = np.zeros(2*n-1)
            for si, s in enumerate(shifts):
                a = abs(s)
                if a >= n:
                    continue
                if z - a <= z <= n-1-a:  # z is in valid range for shift s
                    # Actually: 0 <= z-s < n and 0 <= z+s < n
                    if 0 <= z - s < n and 0 <= z + s < n:
                        delta_z[si] = (1 - h[z - s]) - h[z + s]
            
            for oi, o in enumerate(ones):
                if time.time() > deadline:
                    break
                # Delta for o going from 1 to 0
                delta_o = np.zeros(2*n-1)
                for si, s in enumerate(shifts):
                    if 0 <= o - s < n and 0 <= o + s < n:
                        delta_o[si] = h[o + s] - (1 - h[o - s])
                
                new_g = g_current + (delta_z + delta_o) * (2.0/n)
                new_max = new_g.max()
                if new_max < current_max - 1e-12:
                    # Accept the swap
                    h[z] = 1.0
                    h[o] = 0.0
                    ones[oi] = z
                    zeros[zi] = o
                    g_current = new_g
                    current_max = new_max
                    best_c = float(current_max)
                    improved = True
                    break
        
        if not improved:
            break
    
    return h, best_c
```

```python
            delta_z = np.zeros(2*n-1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha):
    """Log-sum-exp objective and gradient for the C5 approximation."""
    g = compute_all_g(h, n)
    ag = alpha * g
    m = ag.max()
    exp_ag = np.exp(ag - m)
    lse = (m + np.log(exp_ag.sum())) / alpha
    p = exp_ag / exp_ag.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    
    for si, s in enumerate(shifts):
        pi = p[si]
        if pi < 1e-15:
            continue
        a = abs(int(s))
        if a >= n:
            continue
        j_min = a
        j_max = n - a
        if j_min >= j_max:
            continue
        j_range = np.arange(j_min, j_max)
        grad[j_range] += pi * dx * ((1 - h[j_range - s]) - h[j_range + s])
    
    return lse, grad

def lbfgs_phase(h_start, n, deadline, alphas):
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def fun(x):
            lse, _ = lse_obj_grad(x, n, alpha)
            pen = 500.0 * (x.sum() - n/2.0)**2
            return lse + pen
        
        def grad(x):
            _, g = lse_obj_grad(x, n, alpha)
            g = g + 1000.0 * (x.sum() - n/2.0) * np.ones(n)
            return g
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, h_start, jac=grad, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 2000, 'gtol': 1e-10, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_start = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def subgrad_phase(h_start, n, deadline, alpha=10000, max_iter=20000):
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n)
    h = h_start.copy()
    lr = 0.05
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        _, grad = lse_obj_grad(h, n, alpha)
        
        # Find step size
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                patience = 0
                found = True
                break
            elif c_new < compute_c5(h, n) + 1e-12:
                h = h_new
                found = True
                break
        
        if not found:
            lr *= 0.9
            patience += 1
            if patience > 100 or lr < 1e-14:
                break
    
    return best_h, best_c

def binary_local_search(h_start, n, deadline, max_sweeps=5000):
    """Vectorized binary local search."""
    target = n // 2
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    ones_set = set(np.where(h == 1)[0].tolist())
    shifts = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g_cur = compute_all_g(h, n)
        cur_max = g_cur.max()
        
        zeros = [i for i in range(n) if i not in ones_set]
        
        for z in zeros:
            if time.time() > deadline:
                break
            
            # Delta for z: 0 -> 1
            valid_z = (z - shifts >= 0) & (z - shifts < n) & (z + shifts >= 0) & (z + shifts < n)
            if not valid_z.any():
                continue
            delta_z = np.zeros(2*n-1)
            delta_z[valid_z] = (1 - h[z - shifts[valid_z]]) - h[z + shifts[valid_z]]
            
            # For all ones, compute delta simultaneously
            ones_arr = np.array(sorted(ones_set))
            if len(ones_arr) == 0:
                break
            
            # Vectorize: for each one o, delta_o[m] = h[o+s] - (1-h[o-s]) where s=shifts[m]
            ones_col = ones_arr[:, None]  # (n_ones, 1)
            shifts_row = shifts[None, :]  # (1, 2n-1)
            
            pos_idx = ones_col + shifts_row  # o + s
            neg_idx = ones_col - shifts_row  # o - s
            
            valid_o = (pos_idx >= 0) & (pos_idx < n) & (neg_idx >= 0) & (neg_idx < n)
            # valid_o shape: (n_ones, 2n-1)
            
            delta_o = np.zeros((len(ones_arr), 2*n-1))
            delta_o[valid_o] = h[pos_idx[valid_o]] - (1 - h[neg_idx[valid_o]])
            
            # Check all swaps
            new_g = g_cur[None, :] + (delta_z[None, :] + delta_o) * dx
            new_max = new_g.max(axis=1)
            
            bi = np.argmin(new_max)
            if new_max[bi] < cur_max - 1e-13:
                o = ones_arr[bi]
                h[z] = 1.0
                h[o] = 0.0
                ones_set.remove(o)
                ones_set.add(z)
                g_cur = new_g[bi]
                cur_max = new_max[bi]
                best_c = float(cur_max)
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c
```
