```python
def apply_swap_to_corr(corr, h, i, j, n, n_fft):
    """Apply swap (i becomes 1, j becomes 0) to correlation array.
    Returns None (modifies corr in place) and updates h."""
    # Before swap: h[i]=0, h[j]=1
    # After swap: h[i]=1, h[j]=0
    # Change in corr at lag k:
    # From h[i] going 0->1: adds (1-h[i+k]) at lag k for valid i+k
    #                       subtracts h[i-k] at lag k for valid i-k (from the (1-h) factor)
    # From h[j] going 1->0: removes (1-h[j+k]) at lag k for valid j+k
    #                       adds h[j-k] at lag k for valid j-k
    
    # In FFT indexing, lag k corresponds to:
    # k >= 0: index k
    # k < 0: index n_fft + k
    
    # For h[i] 0->1:
    #   For each m where h[m] matters:
    #   corr[k] = sum_m h[m]*(1-h[m+k])
    #   The change from h[i]:0->1:
    #     For the term where m=i: h[i] changes, so delta at lag k = (1-h[i+k]) for k where 0<=i+k<n
    #     For the term where m+k=i (i.e., m=i-k): (1-h[m]) changes (from 1 to 0), so delta = -h[m] = -1 for m where m is a 1-position, i.e., delta = -h[i-k] for k where 0<=i-k<n
    #   So: delta1[k] = (1-h[i+k])*1[0<=i+k<n] - h[i-k]*1[0<=i-k<n]
    
    # For h[j] 1->0:
    #   delta2[k] = -(1-h[j+k])*1[0<=j+k<n] + h[j-k]*1[0<=j-k<n]
    
    # Total: delta[k] = delta1[k] + delta2[k]
    
    # In FFT index:
    # lag k >= 0 -> index k
    # lag k < 0  -> index n_fft + k
    
    # Let me compute this vectorized:
    # delta1 at lag k: (1-h[i+k]) if 0<=i+k<n, else 0, minus h[i-k] if 0<=i-k<n, else 0
    
    # For the first part (1-h[i+k]) for 0<=i+k<n:
    # k ranges from -i to n-1-i
    # In FFT index: -i to n-1-i maps to n_fft-i to n_fft+n-1-i for negative k, and 0 to n-1-i for non-negative k
    
    # This is getting complex. Let me just compute it directly using numpy slicing.
    
    # Create a temporary delta array
    delta = np.zeros(n_fft)
    
    # Part 1: (1-h[i+k]) for 0 <= i+k < n, i.e., -i <= k < n-i
    k_max = n - i - 1  # max k for i+k <= n-1
    k_min = -i  # min k for i+k >= 0
    # For k from k_min to k_max:
    # h_index = i + k
    # delta contribution: (1 - h[i+k]) at lag k
    idx = np.arange(max(k_min, -(n-1)), min(k_max, n-1) + 1)
    h_idx = i + idx
    # These are valid since k_min <= k and k <= k_max ensures 0 <= i+k < n
    # FFT index for lag k: k if k>=0, n_fft+k if k<0
    fft_idx = np.where(idx >= 0, idx, n_fft + idx)
    delta[fft_idx] += (1.0 - h[h_idx])
    
    # Part 2: -h[i-k] for 0 <= i-k < n, i.e., i-n+1 <= k <= i
    k_max2 = i  # max k for i-k >= 0
    k_min2 = i - n + 1  # min k for i-k <= n-1
    idx2 = np.arange(max(k_min2, -(n-1)), min(k_max2, n-1) + 1)
    h_idx2 = i - idx2
    fft_idx2 = np.where(idx2 >= 0, idx2, n_fft + idx2)
    delta[fft_idx2] -= h[h_idx2]
    
    # Part 3: -(1-h[j+k]) for 0 <= j+k < n
    k_max3 = n - j - 1
    k_min3 = -j
    idx3 = np.arange(max(k_min3, -(n-1)), min(k_max3, n-1) + 1)
    h_idx3 = j + idx3
    fft_idx3 = np.where(idx3 >= 0, idx3, n_fft + idx3)
    delta[fft_idx3] -= (1.0 - h[h_idx3])
    
    # Part 4: +h[j-k] for 0 <= j-k < n
    k_max4 = j
    k_min4 = j - n + 1
    idx4 = np.arange(max(k_min4, -(n-1)), min(k_max4, n-1) + 1)
    h_idx4 = j - idx4
    fft_idx4 = np.where(idx4 >= 0, idx4, n_fft + idx4)
    delta[fft_idx4] += h[h_idx4]
    
    corr += delta
    h[i] = 1.0
    h[j] = 0.0
```
