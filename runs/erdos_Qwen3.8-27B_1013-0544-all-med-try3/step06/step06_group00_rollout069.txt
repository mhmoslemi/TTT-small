    """Project x onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on tau
    lo, hi = -1.0, 2.0  # tau range
    for _ in range(50):
        tau = (lo + hi) / 2.0
        h = np.clip(x - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau  # need to reduce h, increase tau
        else:
            hi = tau
    h = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    return h

    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    e = np.exp(alpha * (c - c_max))
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        s = k - (n-1)
        # term1: (1-h[j-s]) for valid j
        j_start = max(0, s)
        j_end = min(n, n+s)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += w[k] * (1.0 - h[j_start-s:j_end-s])
        # term2: -h[j+s] for valid j
        j_start2 = max(0, -s)
        j_end2 = min(n, n-s)  # exclusive
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w[k] * h[j_start2+s:j_end2+s]
    
    return grad

def compute_gradient_fast(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    e = np.exp(alpha * (c - c_max))
    w = e / e.sum()
    
    grad = np.zeros(n)
    # For each shift s, the contribution to grad is:
    # w[s] * (shifted (1-h) - shifted h)
    # where the shifts are by -s and +s respectively
    
    # Vectorized: for shift s, 
    # term1 adds w[s]*(1-h[i-s]) to grad[i] for valid i
    # term2 subtracts w[s]*h[i+s] from grad[i] for valid i
    
    # Alternatively: 
    # grad = sum_s w[s+n-1] * ((1-h) shifted right by s) - sum_s w[s+n-1] * (h shifted left by s)
    # where "shifted right by s" means: (shifted)[i] = (1-h)[i-s] if valid
    
    # This is equivalent to:
    # grad = correlate(w, 1-h) - correlate(w, h) ... no that's not right either.
    
    # Let me just do the loop but with numpy vectorization for each shift:
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        s = k - (n-1)
        # term1: (1-h[i-s]) for i where 0<=i-s<n
        i_lo = max(0, s)
        i_hi = min(n, n+s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w[k] * (1.0 - h[i_lo-s:i_hi-s])
        # term2: -h[i+s] for i where 0<=i+s<n
        i_lo2 = max(0, -s)

        if i_hi2 > i_lo2:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    """Project x onto {h: 0<=h_i<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        h = np.clip(x - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_and_grad(h, n, alpha, beta=10000.0):
    """Compute smooth objective and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    e = np.exp(alpha * (c - c_max))
    se = e.sum()
    w = e / se
    obj = c_max + np.log(se) / alpha
    # Add penalty
    sum_h = h.sum()
    penalty = beta * (sum_h - n / 2.0) ** 2
    obj += penalty
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w[k] * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= w[k] * h[i_lo2 + s:i_hi2 + s]
    # Penalty gradient
    grad += 2.0 * beta * (sum_h - n / 2.0)
    return obj, grad

def lbgfsb_smooth(h, n, alpha, beta, maxiter=500):
    """Run L-BFGS-B with smooth objective."""
    res = minimize(
        smooth_obj_and_grad, h, method='L-BFGS-B', jac=True,
        args=(n, alpha, beta),
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12}
    )
    return project_feasible(res.x, n)

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    """Greedy binary swap local search."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_set = set(int(x) for x in np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        best_delta = None
        best_gain = 0
        best_i = -1
        best_j = -1
        
        for i in zeros_set:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones_set:
                # Compute the change in max correlation
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                gain = cur_c - nc
                if gain > best_gain:
                    best_gain = gain
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_gain > 1e-15:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr += best_delta
            cur_c = corr.max() * dx
            ones_set.remove(best_j)
            zeros_set.remove(best_i)
            ones_set.add(best_i)
            zeros_set.add(best_j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        h = np.clip(x - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_and_grad(x, n, alpha, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    e = np.exp(alpha * (c - c_max))
    se = e.sum()
    w = e / se
    obj = c_max + np.log(se) / alpha
    sum_h = x.sum()
    obj += beta * (sum_h - n / 2.0) ** 2
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w[k] * (1.0 - x[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= w[k] * x[i_lo2 + s:i_hi2 + s]
    grad += 2.0 * beta * (sum_h - n / 2.0)
    return obj, grad

def lbgfsb_smooth(h, n, alpha, beta, maxiter=300):
    res = minimize(
        smooth_obj_and_grad, h, method='L-BFGS-B', jac=True,
        args=(n, alpha, beta),
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14}
    )
    return project_feasible(res.x, n)

def greedy_binary_fast(h_init, n, deadline):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(int(x) for x in np.where(h == 1)[0])
    zeros_list = list(int(x) for x in np.where(h == 0)[0])
    n_elite = min(60, len(zeros_list))
    for _ in range(3000):
        if time.time() > deadline:
            break
        improved = False
        pool_z = zeros_list[:n_elite] if len(zeros_list) > n_elite else zeros_list[:]
        import random
        random.shuffle(pool_z)
        for i in pool_z:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)

            if improved:
