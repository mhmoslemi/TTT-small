                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def greedy_local_search(h, n, deadline, max_sweeps=500, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h <= 0.5)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        rng.shuffle(z_list)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = list(ones)
            rng.shuffle(o_list)
            for j in o_list:
                delta = compute_delta(h, i, j, n, m_range)
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def greedy_with_restarts(h_init, n, deadline, max_sweeps=300, n_restarts=5, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    current_h = h.copy()
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_kicks = rng.randint(2, 8)
            for _ in range(n_kicks):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        sw = max(20, max_sweeps // (restart + 1))
        h, c = greedy_local_search(h, n, deadline, max_sweeps=sw, seed=seed + restart)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

def iterated_local_search(h_init, n, deadline, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    h, c = greedy_local_search(h.copy(), n, deadline, max_sweeps=200, seed=seed)
    best_c = c
    best_h = h.copy()
    
    iteration = 0
    while time.time() < deadline:
        kick_strategy = rng.randint(0, 4)
        h = best_h.copy()
        
        if kick_strategy == 0:
            n_kicks = rng.randint(1, 6)
            for _ in range(n_kicks):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = rng.choice(oi)
                b = rng.choice(zi)
                h[a], h[b] = 0.0, 1.0
        elif kick_strategy == 1:
            block_len = rng.randint(2, 8)
            start = rng.randint(0, n - block_len)
            end = start + block_len
            h[start:end] = 1.0 - h[start:end]
            diff = h.sum() - target
            if diff > 0:
                zi = np.where(h == 1)[0]
                for _ in range(int(np.ceil(diff))):
                    b = rng.choice(zi)
                    h[b] = 0.0
                    zi = np.where(h == 1)[0]
                    if len(zi) == 0:
                        break
            elif diff < 0:
                oi = np.where(h == 0)[0]
                for _ in range(int(np.ceil(-diff))):
                    a = rng.choice(oi)
                    h[a] = 1.0
                    oi = np.where(h == 0)[0]
                    if len(oi) == 0:
                        break
        elif kick_strategy == 2:
            n_flip = rng.randint(3, 12)
            flip_idx = rng.choice(n, size=n_flip, replace=False)
            h[flip_idx] = 1.0 - h[flip_idx]
            diff = h.sum() - target
            if diff > 0:
                oi = np.where(h == 1)[0]
                for _ in range(int(np.ceil(diff))):
                    if len(oi) == 0:
                        break
                    b = rng.choice(oi)
                    h[b] = 0.0
                    oi = np.where(h == 1)[0]
            elif diff < 0:
                oi = np.where(h == 0)[0]
                for _ in range(int(np.ceil(-diff))):
                    if len(oi) == 0:
                        break
                    a = rng.choice(oi)
                    h[a] = 1.0
                    oi = np.where(h == 0)[0]
        else:
            noise = 0.3 * rng.randn(n)
            h_continuous = best_h + noise
            h_continuous = np.clip(h_continuous, 0, 1)
            h_continuous = project_feasible(h_continuous, n)
            top_idx = np.argsort(h_continuous)[::-1][:target]
            h = np.zeros(n)
            h[top_idx] = 1.0
        
        ls_deadline = min(deadline, time.time() + 10)
        h, c = greedy_local_search(h.copy(), n, ls_deadline, max_sweeps=100, seed=seed + iteration)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        iteration += 1
    
    return best_h, best_c

def simulated_annealing(h_init, n, deadline, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    T_init = 0.015
    T_min = 5e-5
    total_budget = deadline - time.time()
    est_iters = int(total_budget * 80000)
    est_iters = max(est_iters, 10000)
    
    one_idx = list(np.where(h == 1)[0])
    zero_idx = list(np.where(h == 0)[0])
    
    iteration = 0
    last_greedy = 0
    
    while time.time() < deadline:
        batch = 2000
        end_iter = min(iteration + batch, est_iters)
        
        for _ in range(end_iter - iteration):
            if time.time() > deadline:
                break
            
            frac = iteration / max(est_iters, 1)
            T = T_init * (T_min / T_init) ** frac
            T = max(T, T_min)
            
            i = zero_idx[rng.randint(len(zero_idx))]
            j = one_idx[rng.randint(len(one_idx))]
            
            if i == j:
                iteration += 1
                continue
            
            delta = compute_delta(h, i, j, n, m_range)
            new_corr = corr + delta
            new_c = new_corr.max() * dx
            dc = new_c - cur_c
            
            if dc < 0 or (T > 1e-15 and rng.random() < np.exp(-dc / T)):
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = new_c
                zero_idx[zero_idx.index(i)] = j
                one_idx[one_idx.index(j)] = i
                
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            
            iteration += 1
        
        if iteration - last_greedy > 30000 and time.time() < deadline:
            gd = min(deadline, time.time() + 3)
            gh, gc = greedy_local_search(h.copy(), n, gd, max_sweeps=30, seed=seed + iteration)
            if gc < best_c:
                best_c = gc
                best_h = gh.copy()
                h = gh.copy()
                corr = np.correlate(h, 1 - h, mode='full').copy()
                cur_c = gc
                one_idx = list(np.where(h == 1)[0])
                zero_idx = list(np.where(h == 0)[0])
            last_greedy = iteration
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: SLSQP smooth optimization
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [1000, 5000, 50000, 500000, 5000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, 800, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Greedy with multiple restarts
    p2 = t0 + 0.22 * (budget_s - 5)
    hb, cb = greedy_with_restarts(best_h, n, p2, max_sweeps=300, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 3: Iterated local search (main exploration)
    p3 = t0 + 0.62 * (budget_s - 5)
    hb, cb = iterated_local_search(best_h, n, p3, seed=seed + 1000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 4: Simulated annealing
    p4 = t0 + 0.82 * (budget_s - 5)
    hb, cb = simulated_annealing(best_h, n, p4, seed=seed + 2000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 5: Polish - SLSQP then greedy
    p5 = dl - 3
    if time.time() < p5 - 20:
        hh = project_feasible(best_h + 0.02 * rng.randn(n), n)
        hh = slsqp_optimize(hh, n, 2000000, 1000, min(p5 - 15, time.time() + 30))
        ch = compute_c5(hh, n)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
        hb, cb = greedy_with_restarts(best_h, n, min(p5 - 5, time.time() + 30),
                                       max_sweeps=200, n_restarts=3, seed=seed + 3000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 6: Try a different resolution
    if time.time() < dl - 10:
        nh = n + 50
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh = slsqp_optimize(hh, nh, 100000, 400, min(dl - 8, time.time() + 25))
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
        else:
            hb, cb = greedy_with_restarts(hh, nh, min(dl - 5, time.time() + 20),
                                           max_sweeps=100, n_restarts=2, seed=seed + 4000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                n = nh
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
