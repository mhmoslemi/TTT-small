def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + np.log(exp_shifted.sum()) / alpha
    p = exp_shifted / exp_shifted.sum()
    sum_viol = h.sum() - n / 2.0
    val = log_sum + penalty * sum_viol ** 2
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m] * dx
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo > hi:
            continue
        # First term: (1-h[m-i]) for i in [lo, hi]
        idx1 = np.arange(lo, hi + 1)
        m_minus_i = m - idx1
        grad[idx1] += pm * (1.0 - h[m_minus_i])
        # Second term: -h[m-i] for m-i in [lo, hi], i.e., i in [m-hi, m-lo]
        i_lo = max(0, m - hi)
        i_hi = min(n - 1, m - lo)
        if i_lo <= i_hi:
            idx2 = np.arange(i_lo, i_hi + 1)
            m_minus_i2 = m - idx2
            grad[idx2] -= pm * h[m_minus_i2]
    grad += 2.0 * penalty * sum_viol
    return val, grad

def lbfgsb_optimize(h_init, n, alpha, penalty, maxiter, deadline):
    h = h_init.copy()

    def obj(x):
        return smooth_obj_and_grad(x, n, alpha, penalty)[0]

        return smooth_obj_and_grad(x, n, alpha, penalty)

                       jac=True, options={'maxiter': maxiter, 'gtol': 1e-10,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = c_max + np.log(exp_s.sum()) / alpha
    p = exp_s / exp_s.sum()
    sv = h.sum() - n / 2.0
    val = log_sum + penalty * sv * sv
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m] * dx
        if pm < 1e-15:
            continue
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo > hi:
            continue
        idx1 = np.arange(lo, hi + 1)
        mi = m - idx1
        grad[idx1] += pm * (1.0 - h[mi])
        i_lo = max(0, m - hi)
        i_hi = min(n - 1, m - lo)
        if i_lo <= i_hi:
            idx2 = np.arange(i_lo, i_hi + 1)
            mi2 = m - idx2
            grad[idx2] -= pm * h[mi2]
    grad += 2.0 * penalty * sv
    return val, grad

def lbfgsb_run(h_init, n, alpha, penalty, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_grad, h_init, method='L-BFGS-B',
                       args=(n, alpha, penalty), jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 100})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def make_binary(h, n, rng):
    h = h.copy()
    target = n // 2
    idx = np.argsort(-h)  # largest first
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def swap_delta(h, n, i, j):
    n_total = 2 * n - 1
    m_vals = np.arange(n_total)
    delta = np.zeros(n_total)
    valid_i = (m_vals >= i) & (m_vals <= i + n - 1) & (m_vals != 2 * i)
    if valid_i.any():
        mi = m_vals[valid_i] - i
        delta[valid_i] = 1.0 - 2.0 * h[mi]
    valid_j = (m_vals >= j) & (m_vals <= j + n - 1) & (m_vals != 2 * j)
    if valid_j.any():
        mj = m_vals[valid_j] - j
        delta[valid_j] -= (1.0 - 2.0 * h[mj])
    return delta

def binary_sa(h_init, n, deadline, seed=42, n_steps=500000):
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, n, rng)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    T = 0.05
    T_min = 0.0005
    cool_rate = (T_min / T) ** (1.0 / n_steps)
    
    t0 = time.time()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = max(T * cool_rate, T_min)
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        d = swap_delta(h, n, int(i), int(j))
        new_corr = corr + d
        new_c = float(new_corr.max()) * dx
        delta_c = new_c - cur_c
        
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / max(T, 1e-10)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            # Update ones and zeros lists
            ones = np.delete(ones, np.where(ones == j)[0][0])
            ones = np.append(ones, i)
            zeros = np.delete(zeros, np.where(zeros == i)[0][0])
            zeros = np.append(zeros, j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c

    h_bin, c_bin = binary_sa(best_h, n, p2, seed=seed, n_steps=600000)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    # Phase 3: Binary first-improvement local search (15% time)
    p3 = t0 + 0.50 * (budget_s - 3)
    h_bi, c_bi = binary_first_improve(best_h, n, p3, max_sweeps=500, seed=seed + 1, n_restarts=3)
    if c_bi < best_c:
        best_c = c_bi
        best_h = h_bi.copy()

    # Phase 4: Round-trip: L-BFGS-B from best binary solution (15% time)
    p4 = t0 + 0.65 * (budget_s - 3)
    if time.time() < p4:
        h_cont = best_h.copy()
        for alpha in [10000, 50000, 200000, 1000000]:
            if time.time() > p4 - 10:
                break
            h_cont = lbfgsb_run(h_cont, n, alpha, 500.0, 400, p4 - 5)
            c = compute_c5(h_cont, n)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()

    # Phase 5: Multi-resolution exploration (20% time)
    p5 = t0 + 0.85 * (budget_s - 3)
    for nh in [200, 300, 400]:
        if time.time() > p5 - 30:
            break
        if nh == n:
            continue
        xn = np.linspace(0, 2, nh, endpoint=False)
        xo = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(xn, xo, best_h)
        h_interp = project_feasible(h_interp, nh)
        c_interp = compute_c5(h_interp, nh)
        # L-BFGS-B at new resolution
        h_new = h_interp.copy()
        for alpha in [5000, 50000, 200000]:
            if time.time() > p5 - 20:
                break
            h_new = lbfgsb_run(h_new, nh, alpha, 300.0, 300, p5 - 15)
        c_new = compute_c5(h_new, nh)
        # Binary SA at new resolution
        if time.time() < p5 - 10:
            h_b, c_b = binary_sa(h_new, nh, p5 - 5, seed=seed + 10 + nh, n_steps=200000)
            c_new = min(c_new, c_b)
            if c_b < c_new or c_b < compute_c5(h_new, nh):
                h_new = h_b
        final_c = min(c_new, c_interp)
        if final_c < best_c:
            best_c = final_c
            best_h = h_new.copy()
            n = nh

    # Phase 6: Final polish (last 15%)
    p6 = deadline - 2
    if time.time() < p6 - 30:
        h = best_h + 0.02 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [500000, 2000000, 5000000]:
            if time.time() > p6 - 15:
                break
            h = lbfgsb_run(h, n, alpha, 1000.0, 500, p6 - 10)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    if time.time() < p6 - 5:
        h_bi, c_bi = binary_first_improve(best_h, n, p6, max_sweeps=300, seed=seed + 99, n_restarts=2)
        if c_bi < best_c:
            best_c = c_bi
            best_h = h_bi.copy()
    if time.time() < p6 - 1:
        h_bin, c_bin = binary_sa(best_h, n, p6, seed=seed + 200, n_steps=200000)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
