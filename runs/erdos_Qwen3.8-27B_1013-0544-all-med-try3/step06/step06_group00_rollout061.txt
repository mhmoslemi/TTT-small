```python
def subgradient_step(h, n, step_size):
    """One subgradient step for min max_k c_k(h)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    c_vals = corr * dx
    max_val = c_vals.max()
    # Find active lags
    tol = 1e-6
    active = np.where(c_vals > max_val - tol)[0]
    # Subgradient: for each active lag k, compute grad of c_k
    # c_k = sum_i h_i * (1 - h_{i+k}) for valid i
    # dc_k/dh_i = (1 - h_{i+k}) - h_{i-k} (with boundary handling)
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # convert to lag
        # c_k = sum_i h_i (1-h_{i+k}) where i and i+k are in [0,n-1]
        # For i in [max(0,-k), min(n, n-k)]:
        #   dc_k/dh_i = (1 - h_{i+k}) if i+k in range, plus -h_{i-k} if i-k in range
        for i in range(n):

                val += (1.0 - h[i+k])

            grad[i] += val
    # Projected subgradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
            if s < 1e-15:
```

```python
    # For h[j_old] going 1→0 (equivalently, h[j_old] "decreases"):
    # For each k ≠ j_old: E[k][n-1+j_old-k] += 1, E[k][n-1+k-j_old] += 1
```

```python
def binary_steepest(h, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    # Compute E
    E = compute_E(h, n)  # (n, 2n-1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = cur_max
        best_i, best_j = -1, -1
        
        # Batch search for best swap
        batch_size = 16
        for bz in range(0, len(zeros), batch_size):
            zb = zeros[bz:bz+batch_size]
            bases = corr[None, :] + E[zb]  # (batch, 2n-1)
            diffs = bases[:, None, :] - E[ones][None, :, :]  # (batch, n_ones, 2n-1)
            new_maxes = diffs.max(axis=2)  # (batch, n_ones)
            flat_min = np.argmin(new_maxes)
            bi = flat_min // len(ones)
            bj = flat_min % len(ones)
            if new_maxes[bi, bj] < best_improvement:
                best_improvement = new_maxes[bi, bj]
                best_i = zb[bi]
                best_j = ones[bj]
        
        if best_i == -1:
            break  # Local optimum
        
        # Apply swap
        i, j = best_i, best_j
        h[i], h[j] = 1.0, 0.0
        corr = corr + E[i] - E[j]
        cur_max = corr.max()
        
        # Update E
        update_E(E, h, n, i, j)
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max * dx
```

    mask_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    first = np.where(mask_fwd, 1.0 - h[np.clip(idx_fwd, 0, n-1)], 0.0)
    idx_bwd = P - L
    mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    second = np.where(mask_bwd, -h[np.clip(idx_bwd, 0, n-1)], 0.0)
    E = first + second
    E[:, n-1] = 0.0
    return E

def recompute_E_row(E, h, n, i):
    lags = np.arange(-(n-1), n)
    idx_fwd = i + lags
    mask_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    first = np.where(mask_fwd, 1.0 - h[np.clip(idx_fwd, 0, n-1)], 0.0)
    idx_bwd = i - lags
    mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    second = np.where(mask_bwd, -h[np.clip(idx_bwd, 0, n-1)], 0.0)
    E[i] = first + second
    E[i, n-1] = 0.0

def update_E(E, h, n, i, j):
    """Update E after swapping h[i] (0→1) and h[j] (1→0)."""
    # h was already modified in the caller
    # We need to account for both changes
    # 
    # Before the swap: h[i]=0, h[j]=1
    # After the swap: h[i]=1, h[j]=0
    #
    # The change in E due to h[i] going 0→1:
    #   For each k≠i: E[k][n-1+i-k] decreases by 1 (first term: 1-h[i] went 1→0)
    #                  E[k][n-1+k-i] decreases by 1 (second term: -h[i] went 0→-1)
    # The change in E due to h[j] going 1→0:
    #   For each k≠j: E[k][n-1+j-k] increases by 1 (first term: 1-h[j] went 0→1)
    #                  E[k][n-1+k-j] increases by 1 (second term: -h[j] went -1→0)
    
    J = np.arange(n)
    # h[i] 0→1: decrease
    m1 = n - 1 + i - J
    m2 = n - 1 + J - i
    valid1 = (m1 >= 0) & (m1 < 2*n-1) & (J != i)
    valid2 = (m2 >= 0) & (m2 < 2*n-1) & (J != i)
    rows1 = J[valid1]
    rows2 = J[valid2]
    E[rows1, m1[valid1]] -= 1.0
    E[rows2, m2[valid2]] -= 1.0
    
    # h[j] 1→0: increase
    m1 = n - 1 + j - J
    m2 = n - 1 + J - j
    valid1 = (m1 >= 0) & (m1 < 2*n-1) & (J != j)
