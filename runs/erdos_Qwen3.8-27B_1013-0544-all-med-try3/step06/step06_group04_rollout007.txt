import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def euclidean_project(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h_proj

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj_all(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _compute_lag_grad(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    if trust is not None:
        cons.append(cp.norm(h_var - h_curr, 1) <= trust)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.PDLP, verbose=False)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx_method(h, n, deadline, max_iters=80):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 1.0 / n * 10
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(2.0 / n, trust * 0.85)
    return best_h, best_c

def slsqp_run(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_all, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_subgradient(h, n, max_iters, deadline):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    step_sizes = np.linspace(0.02, 0.002, max_iters)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        max_val = c.max()
        if max_val < best_c:
            best_c = max_val
            best_h = h_curr.copy()
        
        # Find active lags (within epsilon of max)
        eps = 1e-6
        active = np.where(c >= max_val - eps)[0]
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            grad += _compute_lag_grad(h_curr, n, lag)
        grad /= len(active)
        
        # Projected step
        step = step_sizes[min(it, len(step_sizes) - 1)]
        h_new = h_curr - step * grad
        h_new = euclidean_project(h_new, n)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        
        # Check improvement, backtrack if needed
        c_new = compute_c5(h_new, n)
        if c_new > best_c * 1.01:
            h_new = h_curr
        h_curr = h_new
    
    return best_h, best_c

        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h.copy()
            noise = rng.normal(0, scale, n)
            hp += noise
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Structured initializations
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        # Try a "block" structure
        struct_starts = []
        # Alternating blocks
        blk = n // 8
        h_block = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_block[i:i+blk] = 1.0
        h_block = project_feasible(h_block, n)
        struct_starts.append(h_block)
        
        # Ramp
        h_ramp = np.linspace(0, 1, n)
        h_ramp = project_feasible(h_ramp, n)
        struct_starts.append(h_ramp)
        
        # Sinusoidal
        x = np.linspace(0, 2, n, endpoint=False)
        h_sin = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x)
        h_sin = project_feasible(h_sin, n)
        struct_starts.append(h_sin)
        
        for hs in struct_starts:
            if time.time() > deadline - 50:
                break
            alloc = min(40, (deadline - time.time() - 30) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_full(hs, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_run(best_h, n, 2000000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # One more round of targeted search
        if time.time() < deadline - 2:
            h_lt, c_lt = lag_targeted_search(best_h, n, deadline - 1, max_rounds=20)
            if c_lt < best_c:
                best_c = c_lt
                best_h = h_lt
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
