    h = np.clip(x, 0, 1)
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
            if not mask.any():
                break
            # Distribute increase proportionally to room
            room = 1.0 - h
            room[~mask] = 0
            s = room.sum()
            if s < 1e-14:

            if not mask.any():

            s = h[mask].sum()

            h[mask] += diff / s * h[mask]

def project_fast(x, n):
    """Project x onto {h : 0 ≤ h_i ≤ 1, sum(h) = n/2}."""
    target = n / 2.0
    h = np.clip(x, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for numerical precision
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute correction to non-boundary elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():

def project_fast(x, n):

    h = np.clip(x, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h, n):

def grad_logsumexp(h, n, alpha, corr):

def slsqp_refine(h, n, alpha, maxiter, dept):

    def obj(x):

        m = g_corr.max()

        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_fast(res.x, n)
    except Exception:
        return project_fast(h, n)

    best_h = project_fast(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
