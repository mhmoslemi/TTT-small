def smooth_gradient(h, alpha, dx, n):
    g = np.correlate(h, 1-h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Compute A[j] and B[j] using prefix sums of w
    # d ranges from -(n-1) to n-1, indexed as d + (n-1) in [0, 2n-2]
    cumw = np.zeros(2*n-1+1)
    cumw[1:] = np.cumsum(w)

    # d index: d + (n-1), so range is [n-1-j, n-1+n-j-1] = [n-1-j, 2n-2-j]

    # B[j] = sum of w for d in [j-n+1, j]
    # d index range: [j-n+1+n-1, j+n-1] = [j, j+n-1]
    # B[j] = cumw[j+n-1+1] - cumw[j]
    
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):

        a_end = 2*n-2-j  # d = n-1-j → index 2n-2-j
