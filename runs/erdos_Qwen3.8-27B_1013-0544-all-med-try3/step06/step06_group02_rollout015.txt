def project_feasible(h, n):

            if tot < 1e-15:

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp_matrix(h, n):
    dx = 2.0 / n
    max_k = 2 * n - 1
    A = np.zeros((max_k, n + 1))
    for m in range(max_k):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            A[m, 0:nv] = (1.0 - h[k:n]) * dx
        else:
            kk = -k
            nv = n - kk
            A[m, kk:n] = (1.0 - h[0:nv]) * dx
        A[m, n] = -1.0
    return A

def solve_lp_for_h(h_fixed, n, top_k=None):
    dx = 2.0 / n
    target = n / 2.0
    max_k = 2 * n - 1
    A = build_lp_matrix(h_fixed, n)
    if top_k is not None and top_k < max_k:
        c_vals = compute_c5_vec(h_fixed, n)
        indices = np.argsort(c_vals)[::-1][:top_k]
        A = A[indices]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    b_ub = np.zeros(A.shape[0])
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

def lp_alternate_v2(h, n, deadline, max_iters=10000, use_top_k=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_lp = solve_lp_for_h(h, n, top_k=use_top_k)
        if h_lp is None:
            break
        h_cand = project_feasible(h_lp, n)
        c_cand = compute_c5(h_cand, n)
        if c_cand < best_c - 1e-12:
            best_c = c_cand
            best_h = h_cand.copy()
            h = h_cand
            stagnant = 0
        else:
            improved = False
            for alpha in [0.5, 0.3, 0.1]:
                h_d = project_feasible(alpha * h_lp + (1.0 - alpha) * h, n)
                c_d = compute_c5(h_d, n)
                if c_d < c_cand:
                    h = h_d
                    if c_d < best_c - 1e-12:
                        best_c = c_d
                        best_h = h_d.copy()
                    improved = True
                    break
            if improved:
                stagnant = 0
            else:
                h = h_cand
                stagnant += 1
                if stagnant > 30:
                    break
    return best_h, best_c

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 3:
            break
        def obj_fn(x, alpha=a):
            return _slsqp_obj(x, n, alpha)
        def grad_fn(x, alpha=a):
            return smooth_max_grad(x, n, alpha)
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def _lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lbfgs_phase(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 200), (1000, 300), (5000, 500), (50000, 1000), (500000, 2000)]
    for alpha, mi in configs:
        if time.time() > deadline - 5:
            break
        def obj_fn(x, a=alpha):
            return _lbfgs_obj(x, n, a)
        def grad_fn(x, a=alpha):
            return smooth_max_grad(x, n, a) + 0.1 * (x.sum() - n/2.0) * np.ones(n)
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    if radius is None:
        radius = min(n, 100)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    h_ext = np.concatenate([h_old, h_old])
    x_old_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h_old, h_old, h_old])
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old_ext, h_ext)
    return project_feasible(h_new, n_new)
