    v = np.zeros(n)  # second moment

    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

        # Get current alpha
        alpha = alpha_schedule(t, n_steps)
        
        # Compute gradient
        g = grad_smooth_max(h_curr, n, alpha)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_curr = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
    
    return best_h, best_c

            grad[:n - m] += wk * dx * (-h[m:])
    return grad

def adam_phase(h, n, deadline, alpha, lr, beta1=0.85, beta2=0.999, eps=1e-8, init_m=None, init_v=None):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    if init_m is not None:
        m_adam = init_m.copy()
        v_adam = init_v.copy()
    else:
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
    t = 0
    while time.time() < deadline:
        t += 1
        g = grad_smooth_max(h_curr, n, alpha)
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * (g * g)
        m_hat = m_adam / (1.0 - beta1 ** t)
        v_hat = v_adam / (1.0 - beta2 ** t)
        h_curr = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_curr = project_feasible(h_curr, n)
        if t % 20 == 0:
            c_curr = compute_c5(h_curr, n)
            if c_curr < best_c - 1e-13:
                best_c = c_curr
                best_h = h_curr.copy()
    c_curr = compute_c5(h_curr, n)
    if c_curr < best_c:
        best_c = c_curr
        best_h = h_curr.copy()
    return best_h, best_c, m_adam, v_adam

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -500, 500)
        ev = np.exp(shifted)
        return float(m + np.log(ev.sum()) / alpha)
    def obj_grad(x):
        g = grad_smooth_max(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_local_search(h, n, deadline, alpha=10000, lr=0.005):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        g = grad_smooth_max(h_curr, n, alpha)
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * (g * g)
        m_hat = m_adam / (1.0 - beta1 ** t)
        v_hat = v_adam / (1.0 - beta2 ** t)
        step = lr / (1.0 + 0.01 * t)
        h_curr = h_curr - step * m_hat / (np.sqrt(v_hat) + eps)
        h_curr = project_feasible(h_curr, n)
        if t % 10 == 0:
            c_curr = compute_c5(h_curr, n)
            if c_curr < best_c - 1e-14:
                best_c = c_curr
                best_h = h_curr.copy()
    c_curr = compute_c5(h_curr, n)
    if c_curr < best_c:
        best_c = c_curr
        best_h = h_curr.copy()
    return best_h, best_c

def targeted_local_search(h, n, deadline, max_rounds=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0:
            h = np.where(h > 0.5, np.minimum(1.0, h + 0.002), np.maximum(0.0, h - 0.002))
            h = project_feasible(h, n)
        else:
            if lag > 0:
                diffs = h[:n - m] - h[m:]
                i_best = int(np.argmax(diffs))
                if diffs[i_best] < 1e-7:
                    break
                delta = min(0.01, 0.4 * diffs[i_best])
                h[i_best] -= delta
                h[i_best + m] += delta
            else:
                diffs = h[m:] - h[:n - m]
                i_best = int(np.argmax(diffs))
                if diffs[i_best] < 1e-7:
                    break
                delta = min(0.01, 0.4 * diffs[i_best])
                h[i_best + m] -= delta
                h[i_best] += delta
            h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
        elif rnd > 500:
            c_all = compute_all_lags(h, n)
            if c_all[k_star] < best_c / (2.0 / n) - 1e-10:
                pass
            else:
                if rnd > 2000:
                    break
    return h, best_c

def multi_pair_search(h, n, deadline, max_rounds=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_k = 3
        top_idxs = np.argsort(c)[-top_k:]
        improved = False
        for k_idx in top_idxs:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            m = abs(lag)
            if m == 0:
                continue
            if lag > 0:
                diffs = h[:n - m] - h[m:]
            else:
                diffs = h[m:] - h[:n - m]
            i_best = int(np.argmax(diffs))
            if diffs[i_best] < 1e-7:
                continue
            delta = min(0.005, 0.3 * diffs[i_best])
            if lag > 0:
                h[i_best] -= delta
                h[i_best + m] += delta
            else:
                h[i_best + m] -= delta
                h[i_best] += delta
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
                break
            else:
                if lag > 0:
                    h[i_best] += delta
                    h[i_best + m] -= delta
                else:
                    h[i_best + m] += delta
                    h[i_best] -= delta
                h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c

def slsqp_direct(h, n, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        alpha = 50000.0
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -500, 500)
        ev = np.exp(shifted)
        return float(m + np.log(ev.sum()) / alpha)
    def obj_grad(x):
        return grad_smooth_max(x, n, 50000.0)
    bounds = [(0.0, 1.0)] * n
    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        n_work = n_init
        h_work = project_feasible(h_init.copy(), n_work)
        best_h = h_work.copy()
        best_c = compute_c5(best_h, n_work)

        # Phase 1: Adam with alpha annealing (45% of budget)
        phase1_end = t0 + (deadline - t0) * 0.45
        alphas = [5, 20, 100, 500, 2000, 8000]
        lrs = [0.03, 0.025, 0.02, 0.015, 0.01, 0.005]
        m_adam = np.zeros(n_work)
        v_adam = np.zeros(n_work)
        n_phases = len(alphas)
        phase_dur = (phase1_end - t0) / n_phases
        for i in range(n_phases):
            phase_end = min(phase1_end, t0 + (i + 1) * phase_dur)
            if time.time() > phase_end - 1:
                break
            h_work, c_work, m_adam, v_adam = adam_phase(
                h_work, n_work, phase_end, alphas[i], lrs[i],
                init_m=m_adam, init_v=v_adam
            )
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

        # Phase 2: SLSQP polish (15% of budget)
        if time.time() < deadline - 60:
            phase2_end = min(deadline - 60, time.time() + (deadline - t0) * 0.15)
            for alpha in [20000, 80000]:
                if time.time() > phase2_end - 5:
                    break
                h_slsqp = slsqp_smooth(best_h, n_work, alpha, maxiter=2000, ftol=1e-15)
                c_slsqp = compute_c5(h_slsqp, n_work)
                if c_slsqp < best_c:
                    best_c = c_slsqp
                    best_h = h_slsqp.copy()

        # Phase 3: Gradient local search (20% of budget)
        if time.time() < deadline - 40:
            phase3_end = min(deadline - 40, time.time() + (deadline - t0) * 0.2)
            h_work, c_work = gradient_local_search(best_h, n_work, phase3_end, alpha=15000, lr=0.003)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

        # Phase 4: Targeted local search (10% of budget)
        if time.time() < deadline - 20:
            phase4_end = min(deadline - 20, time.time() + (deadline - t0) * 0.1)
            h_work, c_work = targeted_local_search(best_h, n_work, phase4_end)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

        # Phase 5: Multi-pair search (5% of budget)
        if time.time() < deadline - 10:
            phase5_end = min(deadline - 10, time.time() + (deadline - t0) * 0.05)
            h_work, c_work = multi_pair_search(best_h, n_work, phase5_end)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()

        # Phase 6: Final SLSQP + local search
        if time.time() < deadline - 5:
            h_work = slsqp_direct(best_h, n_work, maxiter=3000, ftol=1e-16)
            c_work = compute_c5(h_work, n_work)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
            h_work, c_work = targeted_local_search(best_h, n_work, deadline - 1)
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
            best_h = project_feasible(best_h, n_work)
            best_c = compute_c5(best_h, n_work)

    # Try higher resolution if time permits
    if best_h is not None and time.time() < deadline - 80:
        n_high = max(n_init + 100, int(n_init * 1.5))
        if n_high > 2 * n_init:
            n_high = int(n_init * 1.4)
        remaining = deadline - time.time() - 40
        if remaining > 60:
            h_high = upsample_h(best_h, len(best_h), n_high)
            alloc = min(remaining * 0.6, 100)
            h_high, c_high, _, _ = adam_phase(h_high, n_high, time.time() + alloc, 500, 0.015)
            if time.time() < deadline - 30:
                h_high, c_high2 = gradient_local_search(h_high, n_high, deadline - 30, alpha=10000, lr=0.003)
                c_high = min(c_high, c_high2)
            if c_high < best_c:
                best_c = c_high
                best_h = h_high.copy()

    # Final attempt: random perturbation + quick optimize
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        scale = 0.01
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(15, deadline - time.time() - 3)
        if alloc > 5:
            h_pert, c_pert, _, _ = adam_phase(hp, n, time.time() + alloc, 1000, 0.01)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
