            h_test[j_s] += delta
            h_test = project_feasible(h_test, n)
            c_test = compute_c5(h_test, n)
            if c_test < best_c - 1e-15:
                h = h_test
                best_c = c_test
                break
    
    return h, best_c

def swap_local_search(h, n, deadline):
    """Local search that swaps mass between pairs for the critical lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m >= n or lag == 0:
                if lag == 0:
                    g = _lag_grad(h, n, 0)
                    i = int(np.argmax(g))
                    j = int(np.argmin(g))
                    if h[i] > d and h[j] < 1 - d:
                        h_test = h.copy()
                        h_test[i] -= d
                        h_test[j] += d
                        h_test = project_feasible(h_test, n)
                        if compute_c5(h_test, n) < best_c:
                            h = h_test
                            best_c = compute_c5(h, n)
                            improved = True
                break
            
            if lag > 0:
                for i in range(0, n - m, max(1, (n - m) // 200)):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1 - h[j]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h_test = h.copy()
                        h_test[i] -= delta
                        h_test[j] += delta
                        h_test = project_feasible(h_test, n)
                        if compute_c5(h_test, n) < best_c - 1e-15:
                            h = h_test
                            best_c = compute_c5(h, n)
                            improved = True
                    elif h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1 - h[i]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h_test = h.copy()
                        h_test[i] += delta
                        h_test[j] -= delta
                        h_test = project_feasible(h_test, n)
                        if compute_c5(h_test, n) < best_c - 1e-15:
                            h = h_test
                            best_c = compute_c5(h, n)
                            improved = True
            else:
                for i in range(0, n - m, max(1, (n - m) // 200)):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1 - h[i]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h_test = h.copy()
                        h_test[i] += delta
                        h_test[j] -= delta
                        h_test = project_feasible(h_test, n)
                        if compute_c5(h_test, n) < best_c - 1e-15:
                            h = h_test
                            best_c = compute_c5(h, n)
                            improved = True
                    elif h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1 - h[j]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h_test = h.copy()
                        h_test[i] -= delta
                        h_test[j] += delta
                        h_test = project_feasible(h_test, n)
                        if compute_c5(h_test, n) < best_c - 1e-15:
                            h = h_test
                            best_c = compute_c5(h, n)
                            improved = True
    
    return h, best_c

def optimize_at_resolution(h_start, n, deadline):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: LP outer approximation with trust region
    if time.time() < deadline - 20:
        lp_end = min(time.time() + 60, deadline - 20)
        h_lp, c_lp = lp_outer_approx(best_h, n, lp_end, max_iters=30, trust_init=0.2)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp
    
    # Phase 2: LP without trust region (fewer iterations)
    if time.time() < deadline - 15:
        h_lp2, c_lp2 = lp_no_trust(best_h, n, min(time.time() + 30, deadline - 15), max_iters=8)
        if c_lp2 < best_c:
            best_c = c_lp2
            best_h = h_lp2
    
    # Phase 3: SLSQP with increasing alpha
    for alpha in [20000, 100000, 500000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Top-k SLSQP
    for top_m in [3, 5, 8]:
        if time.time() > deadline - 10:
            break
        h_new = topk_slsqp(best_h, n, 100000, top_m, maxiter=2000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Swap local search
    if time.time() < deadline - 5:
        best_h, best_c = swap_local_search(best_h, n, min(deadline - 2, time.time() + 30))
    
    # Phase 6: Fine local search
    if time.time() < deadline - 1:
        best_h, best_c = fine_local_search(best_h, n, deadline - 0.5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Optimize at initial resolution
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    
    # Try other resolutions
    n_candidates = [200, 300, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        # Very sharp SLSQP
        if time.time() < deadline - 10:
            h_new = slsqp_optimize(best_h, n, 2000000, maxiter=5000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Final local search
        if time.time() < deadline - 1:
            best_h, best_c = fine_local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
