- For (b): C_k^0 + s_k * x = C_m^0 + s_m * x → x = (C_m^0 - C_k^0)/(s_k - s_m) (if s_k ≠ s_m)

So there are O(n) candidates from (a) and O(n²) from (b). But we only need to check the ones where the relevant curve is actually on top.

In practice, we can:
1. Evaluate f at the current h[i]
2. Find which lags are "active" (close to the max)
3. Only consider breakpoints involving active lags
4. This reduces to O(n) candidates per coordinate

This should give a very efficient coordinate descent.

**Phase 3: Block coordinate descent or pair updates**
- Update pairs of coordinates (i, j) together for better convergence
- This handles the interaction between nearby coordinates

Let me also think about the **block structure**. If we group coordinates into blocks of size b, we can do a 1D (or b-D) optimization on each block. For b=1, it's coordinate descent. For b=2, it's pair descent but more efficient.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_corr(h, n):
    """Compute all correlation values (unscaled)."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto the feasible set [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            active = h > 1e-15
            tot = h[active].sum()
            if tot < 1e-15:
                break
            h[active] += (diff / tot) * h[active]
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    """Smooth max of correlations using log-sum-exp."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if m > 50:
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    sh = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_grad(x, n, alpha):
    """Gradient of smooth objective."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    # For numerical stability
    mx = sh.max()
    exp_sh = np.exp(sh - mx)
    z = np.sum(exp_sh)
    weights = exp_sh / z  # softmax weights
    
    # c = corr * dx, where corr[k] = sum_valid h[i] * (1-h[i+k])
    # dc/dh[i] for lag k:
    # For lag k (indexed as m = n-1+k in the full output):
    #   if k > 0: dc_k/dh[i] = dx * [(1-h[i-k]) * I(k<=i<=n-1) - h[i+k] * I(i<=n-1-k)]
    #   if k = 0: dc_k/dh[i] = dx * (1 - 2*h[i])
    #   if k < 0: dc_k/dh[i] = dx * [(-h[i-k]) * I(-k<=i) - (1-h[i+k]) * I(i+k<=n-1)]
    #     = dx * [-h[i+k] * I(|k|<=i) - (1-h[i-k]) * I(i-|k|<=n-1)]

    # corr_full[m] = sum_j h[j + (m-n+1)] * (1-h[j]) for valid j
    # Let lag = m - (n-1)
    # For lag > 0: corr = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
    #   d/dh[i]: (1-h[i-lag])*I(lag<=i<=n-1) + (-h[i+lag])*I(0<=i<=n-1-lag)
    # For lag = 0: corr = sum h[j]*(1-h[j]), d/dh[i] = 1-2*h[i]
    # For lag < 0 (let k = -lag > 0): corr = sum_{j=k}^{n-1} h[j-k]*(1-h[j])
    #   d/dh[i]: (1-h[i+k])*I(i+k<=n-1) + (-h[i-k])*I(i>=k)
    #     Wait, let me redo this.
    #   corr = sum_{j=k}^{n-1} h[j-k]*(1-h[j])
    #   h[i] appears as h[j-k] when j-k=i, i.e., j=i+k, valid if k<=i+k<=n-1, i.e., 0<=i<=n-1-k
    #     contribution: (1-h[i+k])
    #   h[i] appears as (1-h[j]) when j=i, valid if k<=i<=n-1
    #     contribution: -h[i-k]
    #   So d/dh[i] = (1-h[i+k])*I(0<=i<=n-1-k) - h[i-k]*I(k<=i<=n-1)
    
    # Let me compute gradient directly
    grad = np.zeros(2*n-1)  # gradient w.r.t. correlation values (before * dx)
    # Actually I need d(corr)/dh[i] for each correlation index m
    
    # Let me build the gradient w.r.t. x (the h vector)
    # g[i] = sum_m weights[m] * (dc_m/dx[i]) * dx
    
    # dc_m/dx[i] where m = n-1+lag:
    # For lag >= 0 (m = n-1+lag):
    #   if lag = 0: dc/dx[i] = 1 - 2*x[i]
    #   if lag > 0: dc/dx[i] = (1-x[i-lag])*I(lag<=i<=n-1) + (-x[i+lag])*I(0<=i<=n-1-lag)
    # For lag < 0, let k = -lag (m = n-1-k):
    #   dc/dx[i] = (1-x[i+k])*I(0<=i<=n-1-k) + (-x[i-k])*I(k<=i<=n-1)
    
    g = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-12:
            continue
        lag = m - (n-1)
        w = weights[m] * dx
        if lag == 0:
            g += w * (1 - 2*x)
        elif lag > 0:
            # (1-x[i-lag]) for lag <= i <= n-1
            if lag < n:

            # -x[i+lag] for 0 <= i <= n-1-lag
            if lag < n:
                g[:n-lag] -= w * x[lag:]
        else:
            k = -lag
            # (1-x[i+k]) for 0 <= i <= n-1-k
            if k < n:
                g[:n-k] += w * (1 - x[k:])
            # -x[i-k] for k <= i <= n-1
            if k < n:
                g[k:] -= w * x[:n-k]
    
    return g

def lbfgs_optimize(h0, n, alpha, maxiter=500, tol=1e-10):
    """Optimize using L-BFGS-B."""
    res = minimize(smooth_obj, h0, args=(n, alpha),
                   jac=smooth_grad, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def coord_descent_step(h, i, n):
    """Find optimal h[i] given all other coordinates fixed.
    
    The objective is max_k C_k where:
    C_0 = (n/2 - sum_{j≠i} h[j]²) - h[i]²  [quadratic in h[i]]
    C_k = C_k^0 + s_k * h[i] for k ≠ 0  [linear in h[i]]
    
    We find the minimum of max of these curves over h[i] ∈ [0,1].
    """
    # Precompute correlation values and coefficients
    # C_k^0 and s_k for each lag k
    
    x = h[i]
    # Save old value
    h_old = h[i]
    
    # Compute current correlations
    dx = 2.0 / n
    
    # For each lag m (0 to 2n-2), compute:
    # C_m^0 (value without h[i]) and s_m (coefficient of h[i])
    
    # lag = m - (n-1)
    # For lag = 0: C_0 = sum h[j](1-h[j]) = n/2 - sum h[j]²
    #   C_0 = (n/2 - sum_{j≠i} h[j]²) - h[i]²
    #   So "C_0^0" = n/2 - sum_{j≠i} h[j]², and the term is -h[i]²
    #   In the scaled version: C_0 * dx = dx*(n/2 - sum_{j≠i} h[j]²) - dx*h[i]²
    
    # For lag ≠ 0: C_m = C_m^0 + s_m * h[i] (linear)
    #   C_m * dx = dx*C_m^0 + dx*s_m * h[i]
    
    # Let me compute the unscaled values and scale at the end
    
    # For efficiency, compute the correlation vector once (with current h[i])
    # and then adjust for the change in h[i]
    
    # Actually, let's just do a 1D optimization directly
    # Evaluate f(x) = max_m (C_m(x)) where C_m(x) is the m-th correlation with h[i]=x
    
    # We need to compute C_m(x) for arbitrary x efficiently
    # Current C_m (with h[i] = h_old):
    corr_current = np.correlate(h, 1-h, mode='full')  # 2n-1 values
    
    # When we change h[i] from h_old to x, the change in each C_m is:
    # ΔC_m = (x - h_old) * s_m for linear terms
    # For m = n-1 (lag 0): ΔC_m = -2*h_old*(x-h_old) - (x-h_old)² = h_old² - x²
    #   (since C_0 = ... - h[i]², so C_0(x) = C_0(h_old) + h_old² - x²)
    
    h[i] = x  # temporarily set to x? No, let me compute analytically.
    
    # Let me compute s_m for each m:
    # s_m = dC_m/dh[i] (for m ≠ n-1)
    # For m = n-1 (lag 0): C_0(x) = C_0(h_old) - 2*h_old*(x-h_old) - (x-h_old)²
    #   Actually C_0 = sum h[j](1-h[j]) = sum h[j] - sum h[j]² = n/2 - sum h[j]²
    #   C_0(x) = n/2 - (sum_{j≠i} h[j]² + x²) = (n/2 - sum_{j≠i} h[j]²) - x²
    #   C_0(h_old) = (n/2 - sum_{j≠i} h[j]²) - h_old²
    #   So C_0(x) = C_0(h_old) + h_old² - x²
    
    s = np.zeros(2*n-1)
    for m in range(2*n-1):
        if m == n-1:
            continue  # handle separately
        lag = m - (n-1)
        if lag > 0:
            # C_m = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
            # h[i] appears as h[j+lag] when j+lag=i, i.e., j=i-lag: contributes (1-h[i-lag])
            #   valid if 0 <= i-lag <= n-1-lag, i.e., lag <= i <= n-1
            # h[i] appears as (1-h[j]) when j=i: contributes -h[i+lag]
            #   valid if 0 <= i <= n-1-lag
            if lag <= i <= n-1:
                s[m] += (1 - h[i-lag])
            if 0 <= i <= n-1-lag:
                s[m] -= h[i+lag]
        elif lag < 0:
            k = -lag
            # C_m = sum_{j=k}^{n-1} h[j-k]*(1-h[j])
            # h[i] appears as h[j-k] when j-k=i, i.e., j=i+k: contributes (1-h[i+k])
            #   valid if k <= i+k <= n-1, i.e., 0 <= i <= n-1-k
            # h[i] appears as (1-h[j]) when j=i: contributes -h[i-k]
            #   valid if k <= i <= n-1
            if 0 <= i <= n-1-k:
                s[m] += (1 - h[i+k])
            if k <= i <= n-1:
                s[m] -= h[i-k]
    
    # Now the objective as a function of x:
    # f(x) = max( C_0(h_old) + h_old² - x², max_{m≠n-1} (C_m(h_old) + s[m]*(x - h_old)) )
    #      = max( A - x², max_{m≠n-1} (B_m + s_m * x) )
    # where A = C_0(h_old) + h_old² = n/2 - sum_{j≠i} h[j]²
    #       B_m = C_m(h_old) - s_m * h_old
    
    A = corr_current[n-1] + h_old**2  # = n/2 - sum_{j≠i} h[j]²
    B = corr_current - s * h_old  # B[m] = C_m(h_old) - s[m]*h_old
    B[n-1] = 0  # we'll handle m=n-1 separately
    
    # f(x) = max(A - x², max_{m≠n-1} (B[m] + s[m]*x)) * dx
    # We want to minimize f(x) over x ∈ [0, 1]
    # Since dx > 0, we can minimize the unscaled version
    
    # The function g(x) = max(A - x², max_{m≠n-1}(B[m] + s[m]*x))
    # is the upper envelope of a concave parabola and linear functions.
    
    # Strategy: evaluate g(x) at candidate points and pick the best
    # Candidate points:
    # 1. x = 0, x = 1
    # 2. Intersections of parabola with lines: A - x² = B[m] + s[m]*x
    #    → x² + s[m]*x + (B[m] - A) = 0
    #    → x = (-s[m] ± sqrt(s[m]² - 4*(B[m]-A))) / 2
    # 3. Intersections of pairs of lines: B[m] + s[m]*x = B[p] + s[p]*x
    #    → x = (B[p] - B[m]) / (s[m] - s[p]) (if s[m] ≠ s[p])
    
    # For efficiency, only consider "relevant" lines (those that could be on top)
    # A line B[m] + s[m]*x is relevant if at some x ∈ [0,1], it exceeds A - x²
    # and is among the top few.
    
    # Simple approach: just evaluate at x = 0, 1, and at the parabola-line intersections
    # that fall in [0,1]. There are at most O(n) such points.
    
    candidates = [0.0, 1.0, h_old]
    
    for m in range(2*n-1):
        if m == n-1:
            continue
        # Solve A - x² = B[m] + s[m]*x
        # x² + s[m]*x + (B[m] - A) = 0
        a_coef = 1.0
        b_coef = s[m]
        c_coef = B[m] - A
        disc = b_coef**2 - 4*a_coef*c_coef
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            x1 = (-b_coef + sqrt_disc) / (2*a_coef)
            x2 = (-b_coef - sqrt_disc) / (2*a_coef)
            for xc in [x1, x2]:
                if -1e-9 <= xc <= 1 + 1e-9:
                    candidates.append(max(0.0, min(1.0, xc)))
    
    # Also add some intermediate points for robustness
    candidates = list(set([round(c, 10) for c in candidates]))
    
    best_x = h_old
    best_f = float('inf')
    for xc in candidates:
        xc = max(0.0, min(1.0, xc))
        # Compute g(xc)
        par_val = A - xc**2
        line_vals = B + s * xc  # shape (2n-1,)
        line_vals[n-1] = par_val  # replace with parabola
        g_val = line_vals.max()
        if g_val < best_f:
            best_f = g_val
            best_x = xc
    
    h[i] = best_x
    return best_x

def coordinate_descent(h, n, max_sweeps=100, tol=1e-12, deadline=None):
    """Perform coordinate descent to minimize C_5."""
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                return h, best_c
            old_x = h[i]
            new_x = coord_descent_step(h, i, n)
            if abs(new_x - old_x) > 1e-12:
                improved = True
        # Check convergence
        if not improved:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        elif best_c - c > 1e-10:
            # shouldn't happen, but safety
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
