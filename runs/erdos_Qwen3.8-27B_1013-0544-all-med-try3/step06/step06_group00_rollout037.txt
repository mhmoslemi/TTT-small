def logsumexp_obj_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    shifted -= sm

def max_subgrad_step(h, n, lr):
    dx = 2.0 / n
    g = np.correlate(h, 1-h, mode='full') * dx
    k_max = np.argmax(g)
    # Subgradient of g[k_max] with respect to h
    l = k_max - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        # ∂g[k_max]/∂h[i] = (1-h[i+l])·[i+l valid] - h[i-l]·[i-l valid]
        if 0 <= i+l < n:
            grad[i] += dx * (1 - h[i+l])
        if 0 <= i-l < n:
            grad[i] -= dx * h[i-l]
    # Average over all maximizing indices
    k_all = np.where(g >= g.max() - 1e-10)[0]
    grad = np.zeros(n)
    for k in k_all:
        l = k - (n-1)
        for i in range(n):
            if 0 <= i+l < n:
                grad[i] += dx * (1 - h[i+l])
            if 0 <= i-l < n:
                grad[i] -= dx * h[i-l]
    grad /= len(k_all)
    h_new = h - lr * grad
    return project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_obj_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    shifted -= sm
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    weights = exp_shifted / w_sum
    obj = m + (sm + np.log(w_sum)) / alpha
    C = np.correlate(h, weights, 'full')[n-1:2*n-1]
    B = np.convolve(h, weights, 'full')[n-1:2*n-1]
    grad = dx * (1.0 - C - B)
    return obj, grad

def optimize_slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    ones = np.ones(n)
    def fun(x):
        return logsumexp_obj_grad(x, n, alpha)[0]
    def jac(x):
        return logsumexp_obj_grad(x, n, alpha)[1]
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return ones
    cons = {'type': 'eq', 'fun': cfun, 'jac': cjac}
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def optimize_lbfgs(h, n, alpha, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    ones = np.ones(n)
    def fun(x):
        val, _ = logsumexp_obj_grad(x, n, alpha)
        diff = x.sum() - target
        return val + penalty * diff * diff
    def jac(x):
        _, g = logsumexp_obj_grad(x, n, alpha)
        diff = x.sum() - target
        return g + 2.0 * penalty * diff * ones
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_refine(h, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
