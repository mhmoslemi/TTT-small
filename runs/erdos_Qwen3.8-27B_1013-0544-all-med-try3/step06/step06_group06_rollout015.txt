def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    h = cp.Variable(n)
    t = cp.Variable()

    pairs = []
    for a in range(n):
        for b in range(a):

    w = cp.Variable(n_pairs)

        pair_idx[(a, b)] = i

        h >= 0,

    for k in range(1, n):

            quad_pos += w[pair_idx[(i + k, i)]]
        constraints.append(lin_pos - quad_pos <= t_over_dx)

        lin_neg = 0

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def solve_lp_erdos(n, timeout=120):
    """Solve LP relaxation of Erdős overlap problem."""
    t_start = time.time()
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Pairs (a, b) with a > b
    pairs_a = []
    pairs_b = []
    pair_idx = {}
    idx = 0
    for a in range(n):
        for b in range(a):
            pairs_a.append(a)
            pairs_b.append(b)
            pair_idx[(a, b)] = idx
            idx += 1
    n_pairs = idx
    w = cp.Variable(n_pairs)
    
    dx = 2.0 / n
    t_over_dx = t / dx
    
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    
    # Bilinear linearization (vectorized)
    ha = h[pairs_a]
    hb = h[pairs_b]
    constraints.append(w <= ha)
    constraints.append(w <= hb)
    constraints.append(w >= ha + hb - 1)
    constraints.append(w >= 0)
    
    # Shift constraints
    # For shift k (positive): sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    # = sum_{i=0}^{n-1-k} h[i+k] - sum_{i=0}^{n-1-k} w[i+k, i]
    for k in range(1, n):
        # Positive shift
        lin_pos = cp.sum(h[k:])  # h[k], h[k+1], ..., h[n-1] → sum of h[i+k] for i=0..n-1-k
        # quad: w[pair_idx[(i+k, i)]] for i=0..n-1-k
        # pair_idx[(i+k, i)] = index of (i+k, i) in pairs
        quad_indices_pos = [pair_idx[(i + k, i)] for i in range(n - k)]
        quad_pos = cp.sum(w[quad_indices_pos])
        constraints.append(lin_pos - quad_pos <= t_over_dx)
        
        # Negative shift: sum_{i=k}^{n-1} h[i-k]*(1-h[i])
        # = sum_{j=0}^{n-1-k} h[j] - sum_{j=0}^{n-1-k} w[j+k, j]
        lin_neg = cp.sum(h[:n-k])  # h[0], h[1], ..., h[n-1-k]
        # quad: w[pair_idx[(j+k, j)]] for j=0..n-1-k
        # This is the same as quad_pos!
        # Because pair_idx[(j+k, j)] for j=0..n-1-k is the same set as pair_idx[(i+k, i)] for i=0..n-1-k
        # So quad_neg = quad_pos
        constraints.append(lin_neg - quad_pos <= t_over_dx)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in [cp.GLOP, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if h.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                h_val = np.array(h.value).flatten()
                t_val = float(t.value)
                return h_val, t_val
        except Exception:
            continue
    
    return None, None

def slsqp_smooth(h, n, alpha, maxiter=1000):

        m = g.max()
        shifted = g - m
        amax = shifted.max()
        logsum = amax + np.log(np.exp(shifted - amax).sum())

        # grad of logsum/alpha: (1/alpha) * sum_k weights_k * grad_k
        # grad of g_k w.r.t. h[i]:
        # g_k = sum_j h[j+k](1-h[j]) * dx (for the appropriate range)
        # This is complex to compute analytically; let's use finite differences
        # Actually, let's just let SLSQP compute gradients numerically
        return None  # Let SLSQP use numerical gradients
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

    best_patience = 0

            break

        # Hmm, this is getting complicated. Let me use a different approach.

        # g = np.correlate(h, 1-h, mode='full') * dx
        # This is a bilinear form: g_k = dx * sum_i A_k[i] * h[i] * (1-h[i+k_shift])

        # where the correlation index ranges from 0 to 2n-2
        # Let shift = k - (n-1), so shift ranges from -(n-1) to (n-1)
        # g[shift] = sum_{i valid} h[i + shift] * (1 - h[i])

        # - If j appears as i+j=shift (i = j-shift): term is (1-h[j-shift]) [if valid]
        #   wait no. If h[j] = h[i+shift] where i = j-shift, then d/d(h[j]) of h[i+shift]*(1-h[i]) = (1-h[i]) = (1-h[j-shift])

        grad = np.zeros(n)
        for shift_idx in top_indices:
            shift = shift_idx - (n - 1)  # actual shift
            wt = dx / top_k

        # Project gradient to satisfy sum constraint
        grad -= grad.mean()  # project onto sum=0 subspace

        found = False

        else:

            patience += 1

def binary_search_and_polish(h, n, deadline):
    """Try to improve by binary flipping and local search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = int(n / 2)
    dx = 2.0 / n
    
    # Get current "on" set (where h is closest to 1)
    thr = 0.5
    on = set(np.where(h >= thr)[0])
    off = set(np.where(h < thr)[0])
    
    # Adjust to get exactly target elements in "on"
    if len(on) > target:
        # Remove ones with highest h values that are least critical
        on_list = sorted(on, key=lambda i: h[i], reverse=True)
        for i in on_list[target:]:
            on.remove(i)
            off.add(i)
    elif len(on) < target:
        off_list = sorted(off, key=lambda i: h[i], reverse=True)
        for i in off_list[:target - len(on)]:
            off.remove(i)
            on.add(i)
    
    h_bin = np.zeros(n)
    h_bin[list(on)] = 1.0
    c_bin = compute_c5(h_bin, n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Pair swap local search
    on_list = list(on)
    off_list = list(off)
    improved = True
    max_iters = 500
    it = 0
    while improved and it < max_iters and time.time() < deadline:
        improved = False
        it += 1
        # Try moving one element from on to off and one from off to on
        for i in range(len(off_list)):
            if time.time() > deadline:
                return best_h, best_c
            io = off_list[i]
            for j in range(len(on_list)):
                jo = on_list[j]
                h_bin[io] = 1.0
                h_bin[jo] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h_bin.copy()
                    on_list[j] = io
                    off_list[i] = jo
                    improved = True
                    break
                else:
                    h_bin[io] = 0.0
                    h_bin[jo] = 1.0
            if improved:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def solve_lp_erdos(n, timeout=120):
    t_start = time.time()
    h = cp.Variable(n)
    t = cp.Variable()

    pairs_a = []
    pairs_b = []
    pair_idx = {}
    idx = 0
    for a in range(n):
        for b in range(a):
            pairs_a.append(a)
            pairs_b.append(b)
            pair_idx[(a, b)] = idx
            idx += 1
    n_pairs = idx
    w = cp.Variable(n_pairs)

    dx = 2.0 / n
    t_over_dx = t / dx

    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]

    ha = h[pairs_a]
    hb = h[pairs_b]
    constraints.append(w <= ha)
    constraints.append(w <= hb)
    constraints.append(w >= ha + hb - 1)
    constraints.append(w >= 0)

    for k in range(1, n):
        quad_indices = np.array([pair_idx[(i + k, i)] for i in range(n - k)])
        quad_sum = cp.sum(w[quad_indices])
        lin_pos = cp.sum(h[k:])
        lin_neg = cp.sum(h[:n - k])
        constraints.append(lin_pos - quad_sum <= t_over_dx)
        constraints.append(lin_neg - quad_sum <= t_over_dx)

    prob = cp.Problem(cp.Minimize(t), constraints)

    for solver in [cp.GLOP, cp.CBC, cp.SCIP]:
        try:
            prob.solve(solver=solver)
            if h.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                h_val = np.array(h.value).flatten()
                t_val = float(t.value)
                return h_val, t_val
        except Exception:
            continue
    return None, None

def slsqp_smooth(h, n, alpha, maxiter=800):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = g - m
        amax = shifted.max()
        logsum = amax + np.log(np.exp(shifted - amax).sum())
        return m + logsum / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_polish(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.05
    patience = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        top_k = min(5, len(g))
        top_indices = np.argsort(g)[-top_k:]
        grad = np.zeros(n)
        for shift_idx in top_indices:
            shift = shift_idx - (n - 1)
            wt = dx / top_k
            for j in range(n):
                i = j - shift
                if 0 <= i < n:
                    grad[j] += wt * (1.0 - h[i])
                if 0 <= j + shift < n:
                    grad[j] -= wt * h[j + shift]

        grad -= grad.mean()
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break

        found = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = lr * factor / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break

        if found:
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            lr *= 0.92
            patience += 1

        if patience > 300:
            lr = max(lr, 0.01)
            patience = 0

        if lr < 1e-10:
            break

    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=300):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros = [x for x in range(n) if x not in ones]
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def pair_refine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
    return h, best_c

def multi_alpha_slsqp(h, n, deadline, alpha_list, maxiter_each=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_new = slsqp_smooth(best_h, n, alpha, maxiter_each)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
