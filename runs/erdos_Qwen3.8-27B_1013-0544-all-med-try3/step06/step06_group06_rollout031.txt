import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    """Return the raw correlation array (before multiplying by dx)."""
    return np.correlate(h, 1 - h, mode='full')

def compute_max_corr(h, n):
    """Return the max raw correlation (before *dx)."""
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n):
    """Round h to binary with exactly n/2 ones."""
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_swap_search(h, n, deadline, max_sweeps=500):
    """Greedy local search: try all swaps, accept best improvement each sweep."""
    n ONES = int(n // 2)
    best_c = compute_c5(h, n)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_val = float(corr.max())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_delta = 0
        best_i = -1
        best_j = -1
        for i_pos in ones:
            for j_pos in zeros:
                # Try swap
                h[i_pos], h[j_pos] = 0.0, 1.0
                new_corr = np.correlate(h, 1-h, mode='full')
                new_max = float(new_corr.max())
                if new_max < max_val - 1e-15:
                    best_delta = new_max - max_val  # negative
                    best_i = i_pos
                    best_j = j_pos
                    max_val = new_max
                    corr = new_corr
                h[i_pos], h[j_pos] = 1.0, 0.0
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            max_val = max_val + best_delta
            best_c = max_val * dx
        else:
