def compute_g(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:
            break
        if diff > 0:

        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h

    return float(np.max(c) * dx)

def project_feasible(h, n):

            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:

                    best_c = new_c
                    g = new_g
                    any_imp = True
                    break
                else:
                    h_bin[p] = old
            if not any_imp:
                break
    
    return h_bin, best_c

def swap_optimize(h, n, deadline, max_sweeps=50):
    """Try swapping pairs to reduce max correlation."""
    g = compute_g(h, n)
    best_c = g.max()
    best_h = h.copy()
    npoints = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(n - 1, i, -1):
                if time.time() > deadline:
                    break
                # Try transferring from i to j
                di = h[i]
                dj = h[j]
                if di <= 0 or dj <= 0:
                    continue
                
                # Max transfer amount
                max_t = min(di, dj)
                # Try a few transfer amounts
                for frac in [0.5, 1.0, 0.25]:
                    t = max_t * frac
                    hi = h[i] - t
                    hj = h[j] + t
                    if hi < 0 or hj > 1:
                        continue
                    h[i] = hi
                    h[j] = hj
                    new_g = compute_g(h, n)
                    new_c = new_g.max()
                    if new_c < best_c - 1e-13:
                        best_c = new_c
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] = di
                        h[j] = dj
        
        if not improved:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: Coordinate descent at original resolution (30% budget)
    p1 = t0 + 0.28 * budget_s
    h = project_feasible(best_h.copy(), n0)
    h, c = coordinate_descent(h, n0, p1, max_sweeps=400)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 2: SLSQP with increasing alpha (18% budget)
    p2 = t0 + 0.46 * budget_s
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p2 - 10:
            break
        h = slsqp_step(best_h.copy(), best_n, alpha, 800, p2 - 10)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Subgradient polish (8% budget)
    p3 = t0 + 0.54 * budget_s
    h, c = subgrad_polish(best_h.copy(), best_n, p3, max_iter=2000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 4: Multiple resolutions (25% budget)
    p4 = t0 + 0.79 * budget_s
    res_list = [200, 300, 400, 150, 350]
    for nh in res_list:
        if time.time() > p4 - 15:
            break
        if nh != best_n:
            xo = np.linspace(0, 2, best_n)
            xn = np.linspace(0, 2, nh)
            hh = np.interp(xn, xo, best_h)
            hh = project_feasible(hh, nh)
        else:
            hh = best_h.copy()
        
        sub_dl = min(p4 - 10, time.time() + 0.15 * budget_s)
        hh, c = coordinate_descent(hh, nh, sub_dl, max_sweeps=200)
        
        if time.time() < p4 - 8:
            hh = slsqp_step(hh, nh, 50000, 500, p4 - 8)
            c = compute_c5(hh, nh)
        
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            best_n = nh
    
    # Phase 5: Binary local search (8% budget)
    p5 = t0 + 0.87 * budget_s
    h_bin, c_bin = binary_flip_search(project_feasible(best_h.copy(), best_n), best_n, p5, max_iters=15000)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Also try binary from perturbed continuous
    hp = project_feasible(best_h + 0.15 * rng.randn(best_n), best_n)
    h_bin2, c_bin2 = binary_flip_search(hp, best_n, p5, max_iters=8000)
    if c_bin2 < best_c:
        best_c = c_bin2
        best_h = h_bin2.copy()
    
    # Phase 6: Final polish (remaining time)
    if time.time() < dl - 10:
        h, c = coordinate_descent(project_feasible(best_h.copy(), best_n), best_n, dl - 5, max_sweeps=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if time.time() < dl - 3:
        h = slsqp_step(best_h.copy(), best_n, 1000000, 500, dl - 2)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
