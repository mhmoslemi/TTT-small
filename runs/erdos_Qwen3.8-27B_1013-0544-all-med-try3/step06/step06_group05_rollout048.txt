            if n - llag > 0:
                grad[0:n - llag] += wm * (1.0 - x[llag:n])
                grad[llag:n] += wm * (-x[0:n - llag])
    return obj, grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_and_grad, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=20000, mu=50.0):
    def obj_pen(x):
        obj, grad = smooth_obj_and_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        obj += mu * s * s
        grad = grad + 2.0 * mu * s
        return obj, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': maxiter * 5})
        if time.time() > deadline:
            return h.copy()
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, alpha, deadline, max_iter=100000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.05
    stagnation = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = smooth_obj_and_grad(h, n, alpha)
        grad_c = grad - grad.mean()
        c_cur = compute_c5(h, n)
        found = False
        for _bt in range(25):
            h_try = project_feasible(h - eta * grad_c, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_cur - 1e-15 or _bt == 24:
                found = True
                break
            eta *= 0.5
        if found:
            h = h_try
            c_try = compute_c5(h, n)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 300:
                eta *= 0.7
                stagnation = 0
        else:
            break
        if it % 3000 == 0 and alpha < 1000000:
            alpha = int(alpha * 1.5)
    return best_h, best_c

def targeted_refine(h, n, deadline, rng, max_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c_array(h, n)
        k_max = np.argmax(c)
        lag_max = k_max - (n - 1)
        i = rng.randint(n)
        if rng.rand() < 0.3:
            j = (i + lag_max) % n
        else:
            j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.0005, 0.01) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i], h[j] = hi, hj
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c

def optimize_multi(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    alphas_slsqp = [1000, 5000, 20000, 100000, 500000, 2000000]
    for a in alphas_slsqp:
        if time.time() > deadline - 30:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = ht.copy()
    
    alphas_lbfgs = [50000, 200000, 1000000, 5000000]
    for a in alphas_lbfgs:
        if time.time() > deadline - 20:
            break
        ht = lbfgsb_optimize(best_h, n, a, deadline - 20, maxiter=10000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = ht.copy()
    
    if time.time() < deadline - 10:
        ht, ct = subgradient_descent(best_h, n, 200000, deadline - 5)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    
    if time.time() < deadline - 3:
        ht, ct = targeted_refine(best_h, n, deadline - 1, rng, max_iter=20000)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        best_h = h_init.copy()
        best_c = compute_c5(h_init, n_init)
        
        # Phase 1: Optimize at native resolution
        alloc1 = min(300, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_multi(h_init, n_init, t0 + alloc1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        
        # Phase 2: Try higher resolutions
        resolutions = [400, 500, 600]
        for n_try in resolutions:
            if time.time() > deadline - 60:
                break
            remaining = deadline - time.time() - 40
            if remaining < 30:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            
            # Add small structured perturbation
            freqs = [1.0, 2.0, 3.0, 5.0, 7.0]
            perturb = np.zeros(n_try)
            for f in freqs:
                perturb += rng.uniform(0.001, 0.005) * np.sin(2 * np.pi * f * x_new / 2.0 + rng.uniform(0, 2*np.pi))
            h_interp = h_interp + perturb
            h_interp = project_feasible(h_interp, n_try)
            
            alloc = min(remaining * 0.6, remaining)
            h_opt, c_opt = optimize_multi(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
        
        # Phase 3: Random restarts at best resolution
        n_best = len(best_h)
        if time.time() < deadline - 80:
            remaining = deadline - time.time() - 40
            n_restarts = max(2, int(remaining / 60))
            scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
            for r in range(n_restarts):
                if time.time() > deadline - 30:
                    break
                scale = scales[r % len(scales)]
                hp = best_h + rng.normal(0, scale, n_best)
                hp = project_feasible(hp, n_best)
                alloc = min(50, (deadline - time.time() - 20) * 0.4)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_multi(hp, n_best, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
        
        # Phase 4: Final aggressive refinement
        if time.time() < deadline - 5:
            n_final = len(best_h)
            # L-BFGS with very high alpha
            for a in [5000000, 20000000, 50000000]:
                if time.time() > deadline - 3:
                    break
                ht = lbfgsb_optimize(best_h, n_final, a, deadline - 3, maxiter=20000)
                c = compute_c5(ht, n_final)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = ht.copy()
            
            if time.time() < deadline - 2:
                ht, ct = subgradient_descent(best_h, n_final, 1000000, deadline - 2, max_iter=50000)
                if ct < best_c:
                    best_c = ct
                    best_h = ht.copy()
    
    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
