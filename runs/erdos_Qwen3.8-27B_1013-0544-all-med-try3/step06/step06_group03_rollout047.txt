def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max objective and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    max_sh = shifted.max()
    log_sum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    obj = cmax + log_sum / alpha
    
    # Weights for gradient
    weights = np.exp(shifted - max_sh)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = dx * weights[s + n - 1]
        if w < 1e-15:
            continue
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            sa = -s
            grad[sa:] += w * (1.0 - h[:n-sa])
            grad[:n-sa] -= w * h[sa:]
    
    return obj, grad

    obj, _ = smooth_max_and_grad(h, n, alpha)
    return obj

def objective_grad_func(h, n, alpha):
    return smooth_max_and_grad(h, n, alpha)

def _sum_j(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-15):
    """Run SLSQP with smooth max objective."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    
    def fun(x):
        return objective_func(x, n, alpha)
    
    def jac(x):
        _, g = objective_grad_func(x, n, alpha)
        return g
    
    try:
        res = minimize(fun, h, jac=jac, args=(), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Projected gradient descent on smooth max."""
    best_h = h.copy()
    best_obj = smooth_max_and_grad(h, n, alpha)[0]
    
    # Adaptive step size
    eta = 0.01
    for it in range(maxiter):

            break

            eta *= 0.7  # decrease step

                            best_c = c
                            improved = True
                        else:

                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_rnd = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_rnd = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            if not improved_rnd:
                break
        if not improved and d == deltas[-1]:
            break
    return best_h, best_c

def optimize_pipeline(h_start, n, deadline, alphas):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_optimize(h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 8:
        h, c = gradient_local_search(h, n, alphas[-1], deadline - 5)
        if c < best_c:
            best_c = c
    if time.time() < deadline - 5:
        h, c = informed_local_search(h, n, deadline - 3, alpha=alphas[-1])
        if c < best_c:
            best_c = c
    if time.time() < deadline - 2:
        radius = n if n <= 200 else max(40, n // 8)
        h, c = pair_search_general(h, n, deadline - 1,
                                   [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                   radius=radius)
        if c < best_c:
            best_c = c
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at initial resolution with full alpha schedule
    alloc1 = min(300, (deadline - t0) * 0.35)
    alphas_full = [200, 800, 3000, 10000, 40000, 150000]
    h_opt, c_opt = optimize_pipeline(best_h, n_init, t0 + alloc1, alphas_full)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try nearby resolutions
    x_best = np.linspace(0, 2, len(best_h))
    resolutions = [200, 250, 300, 350, 400, 500]
    resolutions = [r for r in resolutions if r != n_init]
    remaining = deadline - time.time()
    per_res = max(40, remaining / (len(resolutions) + 2))

    for n_try in sorted(resolutions, key=lambda x: abs(x - n_init)):
        if time.time() > deadline - 50:
            break
        alloc = min(per_res, (deadline - time.time()) * 0.35)
        if alloc < 30:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alphas_short = [500, 3000, 15000, 60000, 200000]
        h_opt, c_opt = optimize_pipeline(h_interp, n_try, time.time() + alloc, alphas_short)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            x_best = np.linspace(0, 2, len(best_h))

    # Phase 3: Multi-restart
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.04, n_best)
            elif r % 4 == 1:
                hp = best_h + rng.normal(0, 0.08, n_best)
            elif r % 4 == 2:
                shift = rng.randint(2, max(3, n_best // 20))
                hp = np.roll(best_h, shift)
                hp = project_feasible(hp, n_best)
            else:
                lo = rng.randint(0, max(1, n_best - 40))
                hi = min(n_best, lo + rng.randint(15, 40))
                hp = best_h.copy()
                hp[lo:hi] = rng.uniform(0, 1, hi - lo)
            hp = project_feasible(hp, n_best)
            alloc = min(45, (deadline - time.time() - 25) * 0.45)
            if alloc < 15:
                break
            alphas_r = [1000, 5000, 25000, 100000]
            h_opt, c_opt = optimize_pipeline(hp, n_best, time.time() + alloc, alphas_r)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final intensive polish
    if time.time() < deadline - 10:
        n = len(best_h)
        for a in [300000, 500000, 1000000, 2000000]:
            if time.time() > deadline - 6:
                break
            ht = slsqp_optimize(best_h, n, a, maxiter=5000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = ht
        if time.time() < deadline - 6:
            best_h, c = gradient_local_search(best_h, n, 1000000, deadline - 4)
            if c < best_c:
                best_c = c
        if time.time() < deadline - 3:
            best_h, c = informed_local_search(best_h, n, deadline - 2, alpha=500000)
            if c < best_c:
                best_c = c
        if time.time() < deadline - 1.5:
            radius = n if n <= 200 else max(30, n // 10)
            best_h, c = pair_search_general(best_h, n, deadline - 0.5,
                                            [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                            radius=radius)
            if c < best_c:
                best_c = c

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
