    target = n / 2.0

        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_local(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def random_binary_init(n, rng):
    h = rng.choice([0.0, 1.0], size=n)
    target = n / 2.0
    h_sum = h.sum()
    if h_sum != target:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if h_sum > target:
            n_swap = int(h_sum - target)
            if n_swap <= len(ones):
                idx = rng.choice(ones, size=min(n_swap, len(ones)), replace=False)
                h[idx] = 0.0
        else:
            n_swap = int(target - h_sum)
            if n_swap <= len(zeros):
                idx = rng.choice(zeros, size=min(n_swap, len(zeros)), replace=False)
                h[idx] = 1.0
    return project_feasible(h, n)

def golden_ratio_init(n):
    h = np.zeros(n)
    golden = (1 + np.sqrt(5)) / 2
    for i in range(n):
        x = (i / n * golden) % 1.0
        if x < 0.5:
            h[i] = 1.0
    return project_feasible(h, n)

def active_set_pgd(h, n, k_active, alpha, iters, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    v = np.zeros(n)
    lr = 2.0 / alpha

    for it in range(iters):
        if time.time() > deadline:
            break
        c_full = np.correlate(h, 1 - h, mode='full') * dx
        c_act = c_full[k_active]
        m = c_act.max()
        shifted = np.clip(alpha * (c_act - m), -500, 500)
        e = np.exp(shifted)
        p = e / e.sum()
        obj = m + np.log(e.sum()) / alpha

        grad = np.zeros(n)
        for idx, k in enumerate(k_active):
            if p[idx] < 1e-8:
                continue
            idx1 = np.arange(n) + k - n + 1
            idx2 = np.arange(n) - k + n - 1
            term1 = np.where((idx1 >= 0) & (idx1 < n), 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
            term2 = np.where((idx2 >= 0) & (idx2 < n), h[np.clip(idx2, 0, n - 1)], 0.0)
            grad += p[idx] * (term1 - term2)
        grad *= dx

        v = 0.9 * v - lr * grad
        h_try = project_feasible(h + v, n)

        c_full_try = np.correlate(h_try, 1 - h_try, mode='full') * dx
        c_act_try = c_full_try[k_active]
        m_try = c_act_try.max()
        shifted_try = np.clip(alpha * (c_act_try - m_try), -500, 500)
        e_try = np.exp(shifted_try)
        obj_try = m_try + np.log(e_try.sum()) / alpha

        if obj_try < obj + 1e-14:
            h = h_try
        else:
            v = 0.5 * v
            h_try = project_feasible(h + v, n)
            c_full_try = np.correlate(h_try, 1 - h_try, mode='full') * dx
            c_act_try = c_full_try[k_active]
            m_try = c_act_try.max()
            shifted_try = np.clip(alpha * (c_act_try - m_try), -500, 500)
            e_try = np.exp(shifted_try)
            obj_try = m_try + np.log(e_try.sum()) / alpha
            if obj_try < obj + 1e-14:
                h = h_try
                v = 0.5 * v
            else:
                v = np.zeros(n)

        if it % 100 == 0:
            c_act_v = compute_c5(h, n)
            if c_act_v < best_c:
                best_c = c_act_v
                best_h = h.copy()

    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Projected gradient with momentum over multiple temperatures
    alphas_phase1 = [30, 75, 150, 300, 600, 1200, 2500]
    h1, c1 = projected_gradient(h0, n, alphas_phase1, 800, 0.9, time.time() + 250, lr_scale=1.5)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set refinement
    if time.time() < deadline - 100:
        c_full = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)
        max_c = c_full.max()
        threshold = max_c - 0.02 * max_c
        active_idx = np.where(c_full >= threshold)[0]
        if len(active_idx) > 0 and len(active_idx) < 50:
            for a_act in [500, 2000, 10000]:
                if time.time() > deadline - 60:
                    break
                h2, c2 = active_set_pgd(best_h, n, active_idx, a_act, 500, time.time() + 60)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                c_full = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)
                max_c = c_full.max()
                threshold = max_c - 0.02 * max_c
                active_idx = np.where(c_full >= threshold)[0]
                if len(active_idx) > 50:
                    active_idx = np.argsort(c_full)[-30:]

    # Phase 3: SLSQP polish
    if time.time() < deadline - 50:
        for alpha_sls in [10000, 50000]:
            if time.time() > deadline - 25:
                break
            h3 = slsqp_optimize(best_h, n, alpha_sls, maxiter=3000)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Local pair search
    if time.time() < deadline - 30:
        best_h, best_c = pair_search_local(best_h, n, time.time() + 150,
                                           [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                           radius=100)

    # Phase 5: Multi-restart from different initializations
    if time.time() < deadline - 80:
        inits = []
        inits.append(golden_ratio_init(n))
        inits.append(random_binary_init(n, rng))
        inits.append(best_h + rng.normal(0, 0.05, n))
        inits.append(best_h + rng.normal(0, 0.1, n))

        for init_h in inits:
            if time.time() > deadline - 30:
                break
            init_h = project_feasible(init_h, n)
            c_init = compute_c5(init_h, n)
            if c_init < best_c:
                best_c = c_init
                best_h = init_h.copy()
            alphas_r = [100, 500, 2000]
            h_r, c_r = projected_gradient(init_h, n, alphas_r, 600, 0.9,
                                           time.time() + 50, lr_scale=1.2)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 6: Final pair search on best solution
    if time.time() < deadline - 10:
        best_h, best_c = pair_search_local(best_h, n, deadline - 3,
                                           [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                           radius=80)

    # Phase 7: Final SLSQP + pair search
    if time.time() < deadline - 5:
        h_fin = slsqp_optimize(best_h, n, 100000, maxiter=2000)
        c_fin = compute_c5(h_fin, n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
        if time.time() < deadline - 2:
            best_h, best_c = pair_search_local(best_h, n, deadline - 1,
                                               [0.0005, 0.0002, 0.0001, 0.00005],
                                               radius=60)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
