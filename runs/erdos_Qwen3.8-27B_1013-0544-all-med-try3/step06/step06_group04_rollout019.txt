def fast_binary_local_search(h, n, deadline):
    h = h.copy()
    idx = np.argsort(h)[::-1]
    h[idx[:n//2]] = 1.0
    h[idx[n//2:]] = 0.0
    
    best_c = compute_c5(h, n)
    
    # Precompute D and E
    # D[i, k_idx]: change in overlap at lag k when h[i] goes 1->0
    # k ranges from -(n-1) to n-1, indexed as k + (n-1)
    
    D = np.zeros((n, 2*n-1))
    E = np.zeros((n, 2*n-1))
    
    for i in range(n):
        # Term 1: -(1-h[i+k]) for k >= 0, i+k < n
        for k in range(0, n - i):
            D[i, k + n - 1] -= (1 - h[i+k])
        # Term 2: h[i-k] for k < 0, i-k >= 0
        for k in range(-(i), 0):
            D[i, k + n - 1] += h[i-k]
    
    for j in range(n):
        # Term 3: +(1-h[j+k]) for k >= 0, j+k < n
        for k in range(0, n - j):
            E[j, k + n - 1] += (1 - h[j+k])
        # Term 4: -h[j-k] for k < 0, j-k >= 0
        for k in range(-j, 0):
            E[j, k + n - 1] -= h[j-k]
    
    c = compute_all_lags(h, n)  # shape (2n-1,)
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    while True:
        if time.time() > deadline:
            break
        
        best_improvement = -1e-10
        best_swap = None
        
        for i in ones:
            # For this i, compute new_max for all j in zeros
            # new_overlaps[j, k] = c[k] + D[i, k] + E[j, k]
            # new_max[j] = max_k new_overlaps[j, k]
            
            base = c + D[i]  # shape (2n-1,)
            # For each j: max(base + E[j])
            # Vectorized: (base[None, :] + E[zeros_array, :]).max(axis=1)
            new_maxs = (base[None, :] + E[zeros, :]).max(axis=1)  # shape (len(zeros),)
            
            improvements = best_c - new_maxs
            best_j_idx = np.argmax(improvements)
            if improvements[best_j_idx] > best_improvement:
                best_improvement = improvements[best_j_idx]
                best_swap = (i, zeros[best_j_idx])
        
        if best_swap is None or best_improvement < 1e-12:
            break
        
        i, j = best_swap
        old_hi, old_hj = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        
        # Update best_c
        best_c = compute_c5(h, n)
        c = compute_all_lags(h, n)
        
        # Update D and E for the changed positions
        # D[i] and D[j] and E[i] and E[j] change
        for pos in [i, j]:
            D[pos, :] = 0
            if h[pos] == 1:  # position is now 1, so D represents 1->0 change
                for k in range(0, n - pos):
                    D[pos, k + n - 1] -= (1 - h[pos+k])
                for k in range(-pos, 0):
                    D[pos, k + n - 1] += h[pos-k]
            E[pos, :] = 0
            if h[pos] == 0:  # position is now 0, so E represents 0->1 change
                for k in range(0, n - pos):
                    E[pos, k + n - 1] += (1 - h[pos+k])
                for k in range(-pos, 0):
                    E[pos, k + n - 1] -= h[pos-k]
        
        # Update ones and zeros lists
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        
        # Also need to update D and E for ALL positions since h changed
        # Actually, D[i'] for i' ≠ i,j depends on h[i']±k which might include h[i] or h[j]
        # So we need to recompute D and E entirely
        # This makes it O(n²) per iteration for the precomputation
        
        # Let's just recompute everything
        for pos in range(n):
            D[pos, :] = 0
            for k in range(0, n - pos):
                D[pos, k + n - 1] -= (1 - h[pos+k])
            for k in range(-pos, 0):
                D[pos, k + n - 1] += h[pos-k]
        
        for pos in range(n):
            E[pos, :] = 0
            for k in range(0, n - pos):
                E[pos, k + n - 1] += (1 - h[pos+k])
            for k in range(-pos, 0):
                E[pos, k + n - 1] -= h[pos-k]
    
    return h, best_c

    """Compute D and E matrices."""
    D = np.zeros((n, 2*n-1))  # D[i, k_idx]: change in f_k when h[i]: 1->0
    E = np.zeros((n, 2*n-1))  # E[j, k_idx]: change in f_k when h[j]: 0->1
    
    for i in range(n):
        # D[i]: h[i] goes from 1 to 0
        # a=i term: -(1-h[i+k]) for valid k
        for k in range(max(-(n-1), -i), min(n-1, n-1-i) + 1):

            D[i, idx] = -(1 - h[i + k])
        # a=i-k term: +h[i-k] for valid k, k≠0
        for k in range(max(-(n-1), i-n+1), min(n-1, i) + 1):
            if k == 0:

            D[i, idx] += h[i - k]
    
    for j in range(n):

        # a=j term: +(1-h[j+k]) for valid k
        for k in range(max(-(n-1), -j), min(n-1, n-1-j) + 1):

        # a=j-k term: -h[j-k] for valid k, k≠0

            if k == 0:

            idx = k + n - 1
            E[j, idx] += -h[j - k]
    
    return D, E

                       base_i[j-i] + E[j, j-i] + 1,
                       base_i[i-j] + E[j, i-j] + 1 )

But (base_i + E[j]).max() already includes the values at k=j-i and k=i-j (without the +1). So:
new_max(j) = max( (base_i + E[j]).max(),
                   (base_i + E[j])[j-i] + 1,
                   (base_i + E[j])[i-j] + 1 )

Which simplifies to:
new_max(j) = max( (base_i + E[j]).max(),
                   (base_i + E[j])[j-i+n-1] + 1,
                   (base_i + E[j])[i-j+n-1] + 1 )

This is great! For a fixed i, I can compute base_i + E[j] for all j simultaneously (matrix operation), take the max, and then also check the two correction positions.

Let me vectorize:
- base_i has shape (2n-1,)
- E[zeros] has shape (|zeros|, 2n-1)
- M = base_i[None,:] + E[zeros,:] has shape (|zeros|, 2n-1)
- max_M = M.max(axis=1) has shape (|zeros|,)
- For corrections: for each j in zeros, the correction positions are j-i and i-j.
  - M[:, j-i+n-1] + 1 for the first correction (but j varies, so the column index varies)
  - This is a diagonal-like operation

For the first correction (k = j-i): the column index is (j-i)+(n-1) = j - i + n - 1.
For each j in zeros array, the column is zeros[m] - i + n - 1 where m is the index in the zeros array.
So: corr1 = M[np.arange(len(zeros)), zeros - i + n - 1] + 1

For the second correction (k = i-j): the column index is (i-j)+(n-1) = i - j + n - 1.
corr2 = M[np.arange(len(zeros)), i - zeros + n - 1] + 1

Then: new_maxs = np.maximum(np.maximum(max_M, corr1), corr2)

This is very efficient! For each i, the cost is O(|zeros| * (2n-1)) for the matrix M, plus O(|zeros|) for the max and corrections.

Total for all i: O(|ones| * |zeros| * (2n-1)) = O(n³/2) for balanced n.

For n=300: 150 * 150 * 599 ≈ 13.5M. In numpy, this should take about 50-100ms.
