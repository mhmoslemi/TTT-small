```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        if time.time() > deadline:
            break
        improved = False
        
        for i in range(n):
            # Compute alpha (coefficient of h[i] in each c_full[j])
            alpha = np.zeros(n_lags)
            idx_a = i + np.arange(n_lags) - n + 1  # = i + j - n + 1 for j=0,...,n_lags-1
            mask_a = (idx_a >= 0) & (idx_a < n)
            alpha[mask_a] += (1.0 - h[idx_a[mask_a]])
            
            idx_b = i - np.arange(n_lags) + n - 1  # = i - j + n - 1
            mask_b = (idx_b >= 0) & (idx_b < n)
            alpha[mask_b] += -h[idx_b[mask_b]]
            
            beta = c_full - h[i] * alpha
            
            # Find best t in [0,1] minimizing max(alpha*t + beta)
            # Dense sampling
            t_cands = np.linspace(0, 1, 80)
            vals = alpha[:, None] * t_cands[None, :] + beta[:, None]
            obj = vals.max(axis=0)
            best_idx = np.argmin(obj)
            best_t = t_cands[best_idx]
            
            # Refine
            lo = max(0.0, best_t - 1.0/80)
            hi = min(1.0, best_t + 1.0/80)
            t_fine = np.linspace(lo, hi, 40)
            vals_f = alpha[:, None] * t_fine[None, :] + beta[:, None]
            obj_f = vals_f.max(axis=0)
            best_t = t_fine[np.argmin(obj_f)]
            obj_best = obj_f.min()
            
            # Current max at h[i]
            obj_cur = (alpha * h[i] + beta).max()
            
            if obj_best < obj_cur - 1e-14:
                h[i] = best_t
                c_full = alpha * best_t + beta
                improved = True
        
        new_c = c_full.max() * dx
        if new_c < best_c:
            best_c = new_c
        
        if not improved:
            break
    
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is not None:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                else:
                    lo, hi = i + 1, n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def subgradient_loop(h, n, deadline, initial_step=0.05, decay=0.92, max_no_improve=300):
    """Chase-the-max subgradient method."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step
    no_improve = 0
    
    while time.time() < deadline:
        dx = 2.0 / n
        c_full = np.correlate(h, 1 - h, mode='full')
        k_max = int(np.argmax(c_full))
        s = k_max - (n - 1)
        
        # Gradient of c_full[k_max] w.r.t. h
        # d c_full[n-1+s] / d h[m] = (1-h[m+s])*[0<=m+s<n] - h[m-s]*[0<=m-s<n]
        grad = np.zeros(n)
        m = np.arange(n)
        idx_s = m + s
        mask_s = (idx_s >= 0) & (idx_s < n)
        grad[mask_s] += (1.0 - h[idx_s[mask_s]])
        idx_ms = m - s
        mask_ms = (idx_ms >= 0) & (idx_ms < n)
        grad[mask_ms] += -h[idx_ms[mask_ms]]
        grad *= dx
        
        # Step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-12:
            best_c = c_new
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step *= decay
                no_improve = 0
                if step < 1e-9:
                    break
            elif no_improve > 50:
                step *= 0.95
    
    return h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    for a in [500, 5000, 50000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    
    if time.time() < deadline - 15:
        alloc_cd = min(80, (deadline - time.time()) * 0.4)
        h_cd, c_cd = coord_descent(h, n, time.time() + alloc_cd)
        if c_cd < best_c:
            best_c = c_cd
            h = h_cd
    
    if time.time() < deadline - 8:
        if n <= 150:
            h, best_c = pair_search(h, n, deadline - 5,
                                    [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            h, best_c = pair_search(h, n, deadline - 5,
                                    [0.01, 0.005, 0.002, 0.001], radius=50)
    
    if time.time() < deadline - 5:
        ht = slsqp_opt(h, n, 200000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    if time.time() < deadline - 2:
        h_cd, c_cd = coord_descent(h, n, deadline - 0.5, randomized=True, rng=rng)
        if c_cd < best_c:
            best_c = c_cd
            h = h_cd
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    def remaining():
        return deadline - time.time()
    
    # Phase 1: Optimize at initial resolution
    alloc1 = min(200, remaining() * 0.22)
    if alloc1 > 40:
        h_opt, c_opt = optimize_resolution(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Higher resolution
    for n_try in [350, 500]:
        if remaining() < 130:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(200, remaining() * 0.3)
        if alloc < 60:
            break
        h_opt, c_opt = optimize_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Other resolutions for diversity
    for n_try in [128, 192, 300]:
        if remaining() < 100:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, remaining() * 0.2)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_resolution(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Multi-restart at best resolution
    if best_h is not None and remaining() > 80:
        n = len(best_h)
        n_restarts = max(3, int(remaining() / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.10 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 30) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Intensive local search
    if best_h is not None and remaining() > 20:
        n = len(best_h)
        h_cd, c_cd = coord_descent(best_h, n, deadline - 10, max_sweeps=500)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
        if remaining() > 10:
            if n <= 150:
                best_h, best_c = pair_search(best_h, n, deadline - 5,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 5,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=60)
    
    # Phase 6: Final polish
    if best_h is not None and remaining() > 8:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 1000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        h_cd, c_cd = coord_descent(best_h, n, deadline - 1, max_sweeps=300)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
        if n <= 150 and remaining() > 4:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                         [0.0005, 0.0002, 0.0001, 0.00005])
        elif remaining() > 4:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                         [0.0005, 0.0002, 0.0001], radius=40)
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
