    ```python
    def compute_swap_delta(h, i, j, n):
        offset = n - 1
        delta = np.zeros(2*n - 1)
        s = np.arange(n)
        p = np.arange(n)
        
        delta[offset + s - i] += (1.0 - h[s])  # h[i]: 0->1, first factor
        delta[offset + i - p] -= h[p]           # 1-h[i]: 1->0, second factor
        delta[offset + s - j] -= (1.0 - h[s])  # h[j]: 1->0, first factor
        delta[offset + j - p] += h[p]           # 1-h[j]: 0->1, second factor
        
        # Corrections for double-counting at special positions
        delta[offset] -= 2.0
        delta[offset + (j - i)] += 1.0
        delta[offset + (i - j)] += 1.0
        
        return delta
    ```
