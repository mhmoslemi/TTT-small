def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    if shifted.max() > 500:
        return m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def adaptive_coord_descent(h, n, deadline, neighborhood=100):

    h = h.copy()
    best_c = compute_c5(h, n)
    
    steps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - neighborhood)
                j_hi = min(n, i + neighborhood + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    # Try h[i] += step, h[j] -= step
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)

                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:

def active_lag_search(h, n, deadline):
    """Search focused on reducing active (max) lags."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for iteration in range(1000):
        if time.time() > deadline:
            break
        lags = compute_all_lags(h, n)
        c_max = lags.max()
        # Active lags (within 1% of max)
        active = np.where(lags > c_max * 0.999)[0]
        
        # For each active lag, find the pair of indices that most increase it
        # and try to decrease it
        improved = False
        for lag_idx in active:
            if time.time() > deadline:
                break
            k = lag_idx - (n - 1)  # The actual lag
            # The overlap at this lag is sum_i h[i] * (1-h[i+k]) for valid i
            # To decrease it, we want to either decrease h[i] where (1-h[i+k]) is large
            # or increase h[i+k] where h[i] is large
            
            # Find the index i that contributes most to this lag
            if k >= 0:
                i_range = range(n - k)
                contrib = h[:n-k] * (1 - h[k:n])
            else:
                i_range = range(-k, n)
                contrib = h[-k:] * (1 - h[:n+k])
            
            i_star = np.argmax(contrib)
            # Move: decrease h at the "source" and increase at the "target"
            if k >= 0:
                src = i_star
                dst = i_star + k
            else:
                src = i_star - k
                dst = i_star
            
            d = 0.001
            if h[src] >= d and h[dst] <= 1.0 - d:
                h[src] -= d
                h[dst] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    improved = True
                else:
                    h[src] += d
                    h[dst] -= d
        
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

_DEADLINE = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            rem = diff
            for idx in order:
                add = min(rem, room[idx])
                h[idx] += add
                rem -= add
                if rem < 1e-15:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for idx in order:
                sub = min(rem, h[idx])
                h[idx] -= sub
                rem -= sub
                if rem < 1e-15:
                    break
    return h

def _soft_max(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _con_eq(x, n):
    return float(x.sum() - n * 0.5)

def _slsqp_opt(h, n, alpha, maxiter, deadline):
    if time.time() > deadline - 5:
        return h.copy()
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'args': (n,)}]
    best = h.copy()
    best_c = compute_c5(best, n)
    for _ in range(2):
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_soft_max, best, args=(n, alpha), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best = h_new
                best_c = c_new
        except Exception:
            break
    return best

def _coord_search(h, n, deadline, nbrhood, steps, bidirectional=True):
    h = h.copy()
    best_c = compute_c5(h, n)
    for step in steps:
        converged = False
        while not converged and time.time() < deadline:
            converged = True
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - nbrhood)
                j_hi = min(n, i + nbrhood + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            converged = False
                        else:
                            h[i] -= step
                            h[j] += step
                    if bidirectional and h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            converged = False
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

def _random_restart_search(h, n, deadline, rng, n_restarts=5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alloc = (deadline - time.time()) / max(1, n_restarts)
    for r in range(n_restarts):
        end = time.time() + alloc
        scale = 0.02 + 0.03 * r
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        c0 = compute_c5(hp, n)
        if c0 < best_c:
            best_c = c0
            best_h = hp.copy()
        hp, c1 = _coord_search(hp, n, end, nbrhood=30,
                               steps=[0.01, 0.005, 0.002, 0.001, 0.0005])
        if c1 < best_c:
            best_c = c1
            best_h = hp.copy()
    return best_h, best_c
