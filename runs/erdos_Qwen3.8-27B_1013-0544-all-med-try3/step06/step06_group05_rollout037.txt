from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx  # This is the vector of overlap integrals for each shift

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_box_hyperplane(y, target_sum):
    """Project y onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1.0 - 1
    hi = y.max() + 1.0 + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_descent(h, n, deadline, lr_0=0.01, momentum=0.9):
    """Subgradient method with projection for the max-of-quadratics objective."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    dx = 2.0 / n
    
    it = 0
    while time.time() < deadline:
        # Compute correlation and find active shift
        c_vec = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c_vec.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        
        # Find active shifts (within 1% of max)
        active_mask = c_vec >= c_max * 0.99 - 1e-6
        active_indices = np.where(active_mask)[0]
        
        # Compute subgradient (average of gradients of active constraints)
        grad = np.zeros(n)
        for k_idx in active_indices:
            # k_idx corresponds to shift k = k_idx - (n-1)
            k = k_idx - (n - 1)
            # Gradient of c_k with respect to h[j]
            # For shift k: c_k = sum_{i: valid} h[i] * (1 - h[i+k])

            # j+k valid means 0 <= j+k < n

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_box_hyperplane(y, target_sum):
    n = len(y)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > target_sum:

        else:

        else:
            tot = h.sum()
            if tot > 1e-13:

def compute_subgradient(h, n, active_indices):
    """Compute average subgradient of the max function at active indices."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in active_indices:
        shift = k_idx - (n - 1)
        g = np.zeros(n)
        # First term: (1 - h[j+shift]) for valid j
        if shift >= 0:
            if n - shift > 0:
                g[:n-shift] += (1.0 - h[shift:n])
        else:
            s = -shift
            if n - s > 0:

                g[shift:] += -h[:n-shift]
        else:
            s = -shift
            if n - s > 0:
                g[:n-s] += -h[s:]
        grad += g
    return grad / len(active_indices) * dx

    h = h.copy()

    v = np.zeros(n)

    while time.time() < deadline:
        c_vec = compute_c5_vector(h, n)
        c_max = c_vec.max()
        
        if c_max < best_c - 1e-10:
            best_c = c_max
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Find active indices (within 2% of max or within 1e-4)
        threshold = c_max - 0.002
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        grad = compute_subgradient(h, n, active)
        
        # Update

        if no_improve_count > 5000:

            no_improve_count = 0
        
        if lr < 1e-8:

    return best_h, best_c

def pair_descent(h, n, deadline, grid_size=200):
    """Pair coordinate descent with 1D grid search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    n_rounds = 0
    while time.time() < deadline:
        improved = False
        # Try random pairs
        for _ in range(500):
            if time.time() > deadline:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            
            S = h[i] + h[j]
            t_lo = max(0.0, S - 1.0)
            t_hi = min(1.0, S)
            if t_hi - t_lo < 1e-10:

            # Grid search for optimal t = h[i]
            ts = np.linspace(t_lo, t_hi, grid_size)
            best_t = t_lo + (t_hi - t_lo) * h[i] / S if S > 0 else t_lo
            best_val = float('inf')
            
            for t in ts:
                h[i] = t
                h[j] = S - t
                c = compute_c5(h, n)
                if c < best_val:
                    best_val = c
                    best_t = t
            
            h[i] = best_t
            h[j] = S - best_t
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                improved = True
        
        n_rounds += 1
        if not improved and n_rounds > 10:
            break
    
    return h, best_c

    log_sum = c_max + np.log(exp_shifted.sum()) / alpha
    # Softmax weights
    p = exp_shifted / exp_shifted.sum()
    # Gradient: dx * sum_k p_k * grad(c_k)
    # But c_k already includes dx, so grad(c_k * dx) = dx * grad(c_k)
    # Actually c_vec[k] = dx * raw_corr[k], so grad(c_vec[k]) = dx * grad(raw_corr[k])
    # And grad of log_sum_exp = sum_k p_k * grad(c_vec[k])
    # = sum_k p_k * dx * grad(raw_corr[k])
    # = sum_k p_k * grad(α * c_vec[k]) / α ... 
    
    # Let me redo: f = (1/alpha) * log(sum exp(alpha * c_vec))
    # df/dh[j] = sum_k (alpha * c_vec[k])' * exp(alpha * c_vec[k]) / sum exp(alpha * c_vec[k])
    # = sum_k alpha * (c_vec[k])' * p_k
    # where (c_vec[k])' = d/dh[j] of (dx * raw_corr_k) = dx * d(raw_corr_k)/dh[j]
    
    # So df/dh[j] = alpha * dx * sum_k p_k * d(raw_corr_k)/dh[j]
    
    grad = np.zeros(n)
    for k_idx in range(len(c_vec)):
        if p[k_idx] < 1e-15:
            continue
        shift = k_idx - (n - 1)
        g_k = grad_cc(h, n, shift)
        grad += p[k_idx] * g_k
    grad *= alpha * dx
    
    return log_sum, grad

def slsqp_smooth(h, n, alpha, maxiter=2000):
    x0 = h.copy()
    
    def obj(x):
        val, _ = smooth_obj_and_grad(x, n, alpha)
        return val
    
    def jac(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_method(h, n, deadline, rng, lr_init=0.003):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr_init
    momentum = 0.8
    patience = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_all(h, n)
        c_max = c_vec.max()
        
        if c_max < best_c - 1e-10:
            best_c = c_max
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Active indices
        threshold = c_max - 0.003
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        # Limit active set size
        if len(active) > 50:
            # Keep only the top 50
            top_idx = np.argsort(c_vec[active])[::-1][:50]
            active = active[top_idx]
        
        # Compute subgradient
        grad = np.zeros(n)
        dx = 2.0 / n
        for k_idx in active:
            shift = k_idx - (n - 1)
            grad += grad_cc(h, n, shift)
        grad /= len(active)
        grad *= dx
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = project_box_hyperplane(h + v, n / 2.0)
        
        # Adaptive lr
        c_new = compute_c5(h_new, n)
        if c_new > best_c + 0.02:
            lr *= 0.7
            v *= 0.5
        elif patience > 2000:
            lr *= 0.9
            patience = 0
        
        if lr < 1e-7:
            break
        
        h = h_new
    
    return best_h, best_c

def pair_descent_grid(h, n, deadline, rng, grid_size=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        improved = False

            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            S = h[i] + h[j]
            t_lo = max(0.0, S - 1.0)
            t_hi = min(1.0, S)
            if t_hi - t_lo < 1e-10:
                continue
            
            # Current t value
            t_cur = h[i]
            
            # Grid search
            ts = np.linspace(t_lo, t_hi, grid_size)
            best_t = t_cur
            best_val = float('inf')
            
            for t in ts:
                h[i] = t
                h[j] = S - t
                c = compute_c5(h, n)
                if c < best_val:
                    best_val = c
                    best_t = t
            
            h[i] = best_t
            h[j] = S - best_t
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                improved = True
        
        if not improved:
            # Try more random pairs or break
            if time.time() > deadline - 5:
                break
            # Continue with more pairs
    
    return h, best_c

def compute_c5(h, n):

            if tot < 1e-15:

            h += (diff / tot) * h

        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-13:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_raw(h, n, shift):
    """Gradient of raw correlation at given shift."""
    grad = np.zeros(n)
    if shift >= 0:
        if n - shift > 0:
            grad[:n - shift] += (1.0 - h[shift:n])
            grad[shift:] += -h[:n - shift]
    else:
        s = -shift
        if n - s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += -h[s:]
    return grad

def smooth_val_grad(h, n, alpha):
    """Compute smooth objective value and gradient."""
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode='full')
    c = raw * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    val = c_max + np.log(exp_s.sum()) / alpha
    p = exp_s / exp_s.sum()
    # Gradient: sum_k p_k * dx * grad_raw_k
    grad = np.zeros(n)
    for k in range(len(c)):
        if p[k] < 1e-10:
            continue
        shift = k - (n - 1)
        grad += p[k] * grad_raw(h, n, shift)
    grad *= dx
    return val, grad

def slsqp_opt(h, n, alpha, maxiter=3000):
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_val_grad(x, n, alpha)[0]
    
    def jac(x):
        return smooth_val_grad(x, n, alpha)[1]
    
    def con(x):
        return np.sum(x) - n / 2.0
    
    def con_j(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        result = project_feasible(res.x, n)
        return result
    except Exception:
        return h.copy()

def subgrad_opt(h, n, deadline, rng):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.002
    mom = 0.8
    dx = 2.0 / n
    stall = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_all(h, n)
        c_max = c_vec.max()
        if c_max < best_c - 1e-10:
            best_c = c_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Active constraints
        thresh = c_max - 0.005
        active = np.where(c_vec >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 30:
            top = np.argsort(c_vec)[::-1][:30]
            active = top
        
        grad = np.zeros(n)
        for ki in active:
            shift = ki - (n - 1)
            grad += grad_raw(h, n, shift)
        grad /= len(active)
        grad *= dx
        
        v = mom * v - lr * grad
        h_new = project_box_hyperplane(h + v, n / 2.0)
        c_new = compute_c5(h_new, n)
        
        if c_new > best_c + 0.05:
            lr *= 0.5
            v *= 0.3
        elif stall > 3000:
            lr *= 0.95
            stall = 0
        
        h = h_new
        if lr < 1e-8:
            break
    
    return best_h, best_c

def pair_grid_opt(h, n, deadline, rng, n_grid=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        improved = False
        for _ in range(300):
            if time.time() > deadline:
                return h, best_c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            S = h[i] + h[j]
            t_lo = max(0.0, S - 1.0)
            t_hi = min(1.0, S)
            if t_hi - t_lo < 1e-8:
                continue
            
            ts = np.linspace(t_lo, t_hi, n_grid)
            best_t = h[i]
            best_v = float('inf')
            for t in ts:
                h[i] = t
                h[j] = S - t
                c = compute_c5(h, n)
                if c < best_v:
                    best_v = c
                    best_t = t
            
            h[i] = best_t
            h[j] = S - best_t
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-13:
                best_c = c
                improved = True
        
        if not improved:
            break
    
    return h, best_c
