import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def compute_c5_array(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_to_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, n)
    except Exception:
        return h.copy()

def get_lag_coefficients(h, n, lag):
    """Get the coefficient vector for the correlation at a specific lag.
    lag is the index in the full correlation array (0 to 2n-2).
    Returns coeff such that c[lag] = sum of h[j]*coeff[j] + const (approximately).
    Actually, c[lag] is bilinear in h, so this isn't quite right.
    Instead, for a small perturbation h -> h + eps*p (with sum(p)=0),
    dc[lag] = eps * (sum p[j]*h[j+k] + sum h[j+k]*p[j]) ... this is getting complicated.
    
    Let me just return the "linearized" coefficient:
    c[lag] = sum_j h[j] * g[j] where g depends on h (it's not linear).
    
    For the swap h[a] += d, h[b] -= d:
    delta_c = d * (g[a] - g[b]) approximately, where g[j] is the partial derivative.
    """
    k = lag - (n - 1)  # shift value
    
    g = np.zeros(n)
    if k >= 0:
        # c[n-1+k] = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
        # partial wrt h[p]:
        #   (1-h[p-k]) if p >= k (p appears as h[j+k] with j=p-k)
        #   -h[p+k] if p <= n-1-k (p appears in (1-h[j]) with j=p)
        for p in range(k, n):
            g[p] += (1.0 - h[p - k])
        for p in range(0, n - k):
            g[p] -= h[p + k]

        for j in range(l, n):

    return g
