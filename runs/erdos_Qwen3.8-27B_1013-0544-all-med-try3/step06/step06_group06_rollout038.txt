def binary_swap_delta(h, c_current, a, b, n):
    """Compute new correlation after swapping h[a]: 1->0, h[b]: 0->1.
    Returns the new full correlation array (scaled by dx).
    """
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    one_minus_h_rev = one_minus_h[::-1]
    
    delta = np.zeros(2*n-1)
    
    # -correlate(e_a, 1-h): -(1-h)[::-1] at positions [a, a+n)
    delta[a:a+n] -= one_minus_h_rev
    # +correlate(e_b, 1-h): +(1-h)[::-1] at positions [b, b+n)
    delta[b:b+n] += one_minus_h_rev
    # +correlate(h, e_a): +h at positions [n-1-a, 2n-1-a)
    delta[n-1-a:2*n-1-a] += h
    # -correlate(h, e_b): -h at positions [n-1-b, 2n-1-b)
    delta[n-1-b:2*n-1-b] -= h
    # Self-correlation terms
    delta[n-1+(a-b)] += 1
    delta[n-1+(b-a)] += 1
    delta[n-1] -= 2
    
    c_new = c_current + delta * dx
    return c_new

def binary_first_improvement(h, n, deadline, max_steps=20000):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    current_max = float(c.max())
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    delta = np.zeros(2 * n - 1)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        omh_rev = (1.0 - h)[::-1]
        improved = False
        for a in ones:
            if time.time() > deadline:
                break
            for b in zeros:
                delta[:] = 0.0
                delta[n-1-a : n-1-a+n] += h
                delta[n-1-b : n-1-b+n] -= h
                delta[a : a+n] -= omh_rev
                delta[b : b+n] += omh_rev
                ap = n - 1 + (a - b)
                bp = n - 1 + (b - a)
                delta[ap] += 1.0
                delta[bp] += 1.0
                delta[n-1] -= 2.0
                new_max = (c + delta * dx).max()
                if new_max < current_max - 1e-14:
                    h[a], h[b] = 0.0, 1.0
                    c = c + delta * dx
                    current_max = float(new_max)
                    ones.remove(a)
                    ones.append(b)
                    zeros.remove(b)
                    zeros.append(a)
                    omh_rev = (1.0 - h)[::-1]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if step % 200 == 0:
            c = np.correlate(h, 1 - h, mode='full')
            current_max = float((c * dx).max())
            c = c * dx
    return h, current_max
