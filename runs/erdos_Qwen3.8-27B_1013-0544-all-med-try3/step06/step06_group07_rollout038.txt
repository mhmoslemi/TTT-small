def active_set_lp(h_curr, target_sum, n, n_active, t_max):
    """Solve the linearized active set problem."""
    dx = 2.0 / n
    corr = compute_all_corr(h_curr)
    
    # Find active set
    thresh = t_max / dx - 0.5 * dx  # Within half a pixel of the max
    active = np.where(corr >= thresh)[0]
    if len(active) > n_active:
        # Keep only the top n_active
        top_idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[top_idx]
    
    # Compute gradients for active lags
    A_rows = []
    b_const = []
    for p in active:
        g = compute_corr_gradient(h_curr, p, n)
        A_rows.append(dx * g)
        b_const.append(dx * corr[p])
    
    A = np.array(A_rows)  # shape (m, n) where m = len(active)
    
    # LP: min t_new
    # s.t. dx * corr[p]_0 + dx * grad[p] · (h - h_0) <= t_new  for p in active
    #      0 <= h[i] <= 1
    #      sum(h) = target_sum
    
    # Variables: [h_0, h_1, ..., h_{n-1}, t_new]
    # Objective: minimize t_new => c = [0, 0, ..., 0, 1]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A*h - t <= b  (i.e., A*h - t <= 0 after moving)
    # dx*corr[p] + dx*grad[p]·h - dx*grad[p]·h_0 - t <= 0
    # (A + dx*grad[p]*h_0... )
    
    # Let me rewrite: dx*corr[p] + dx * grad[p]·(x - h_0) <= t
    # dx * grad[p]·x - t <= dx*grad[p]·h_0 - dx*corr[p]
    # In standard form: A_ub @ vars <= b_ub
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i, p in enumerate(active):
        g = compute_corr_gradient(h_curr, p, n)
        A_ub[i, :n] = dx * g
        A_ub[i, n] = -1.0
        b_ub[i] = dx * g @ h_curr - dx * corr[p]
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: 0 <= h[i] <= 1, t unbounded below (but >= 0 makes sense)
    bounds = list(zip(np.zeros(n), np.ones(n))) + [(0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None

    corr = compute_all_corr(h_curr)
    n_corr = 2 * n - 1

    c_vec[-1] = 1.0

    b_ub = np.zeros(n_corr)
    for p in range(n_corr):
        k = p - (n - 1)
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        g = np.zeros(n)
        # From h[i+k] term
        j1_lo = max(0, lo + k)
        j1_hi = min(n - 1, hi + k)
        if j1_lo <= j1_hi:
            idx = np.arange(j1_lo, j1_hi + 1)
            g[idx] += (1.0 - h_curr[idx - k])
        # From (1-h[i]) term
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] -= h_curr[idx + k]
        
        A_ub[p, :n] = dx * g
        A_ub[p, n] = -1.0
        b_ub[p] = dx * g @ h_curr - dx * corr[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = list(zip(np.zeros(n), np.ones(n))) + [(0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None
