                active = a[vals >= mx - 1e-15]

            if tot < 1e-15:
                break

    h = x  # assume x is already in [0,1]

                opt_d = hi_d
            else:
                dlo, dhi = lo_d, hi_d
                for _ in range(40):
                    dmid = (dlo + dhi) * 0.5
                    sm_mid = np.max(C + dmid * a)
                    vals = C + dmid * a
                    mx_v = vals.max()
                    active = a[vals >= mx_v - 1e-14]
                    if len(active) == 0:
                        break
                    gmin = active.min()
                    gmax = active.max()
                    if gmin > 0:
                        dhi = dmid
                    elif gmax < 0:
                        dlo = dmid
                    else:
                        break
                opt_d = (dlo + dhi) * 0.5
            new_val = best_h[i] + opt_d
            new_c = np.max(C + opt_d * a)
            if new_c < best_c - 1e-16:
                C += opt_d * a
                best_h[i] = new_val
                best_c = new_c
                improved = True
        if not improved:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_h = h.copy()
    best_c = compute_c5(h)
    for d in deltas:
        while time.time() < deadline - 2:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return best_h, best_c
                if radius is None:
                    lo_j, hi_j = 0, n
                else:
                    lo_j = max(0, i - radius)
                    hi_j = min(n, i + radius + 1)
                for j in range(lo_j, hi_j):
                    if i == j:
                        continue
                    if best_h[i] >= d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[j] >= d and best_h[i] + d <= 1.0:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
            if not improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)
    h = project_feasible(h_init, n_init)
    best_c = compute_c5(h)
    # Phase 1: L-BFGS-B with increasing alpha
    alpha_seq1 = [50, 200, 1000, 5000, 20000, 100000, 500000]
    mu_seq1 = [500, 2000, 10000, 50000, 100000, 200000, 500000]
    t1_end = min(t0 + 280, deadline - 350)
    h, best_c = run_lbfgs(h, n_init, t1_end, alpha_seq1, mu_seq1, maxiter_per=250)
    # Phase 2: Coordinate descent
    if time.time() < deadline - 350:
        t2_end = min(time.time() + 120, deadline - 350)
        h2, c2 = coord_descent(h, n_init, t2_end)
        if c2 < best_c:
            best_c = c2
            h = h2
    # Phase 3: Active constraint polish
    if time.time() < deadline - 300:
        t3_end = min(time.time() + 80, deadline - 300)
        h3, c3 = active_polish(h, n_init, t3_end)
        if c3 < best_c:
            best_c = c3
            h = h3
    # Phase 4: Try higher resolutions
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 250:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, h)
        h_interp = project_feasible(h_interp, n_try)
        alpha_seq2 = [1000, 10000, 100000, 500000]
        mu_seq2 = [10000, 100000, 200000, 500000]
        t4_end = min(time.time() + 150, deadline - 200)
        h_t, c_t = run_lbfgs(h_interp, n_try, t4_end, alpha_seq2, mu_seq2, maxiter_per=200)
        if c_t < best_c:
            best_c = c_t
            h = h_t
            n_init = n_try
    # Phase 5: Final L-BFGS at best resolution
    if time.time() < deadline - 100:
        alpha_seq3 = [50000, 200000, 1000000]
        mu_seq3 = [100000, 200000, 500000]
        t5_end = min(time.time() + 80, deadline - 100)
        h, best_c = run_lbfgs(h, n_init, t5_end, alpha_seq3, mu_seq3, maxiter_per=200)
    # Phase 6: Coordinate descent
    if time.time() < deadline - 60:
        t6_end = min(time.time() + 40, deadline - 60)
        h6, c6 = coord_descent(h, n_init, t6_end)
        if c6 < best_c:
            best_c = c6
            h = h6
    # Phase 7: Active polish
    if time.time() < deadline - 40:
        t7_end = min(time.time() + 30, deadline - 40)
        h7, c7 = active_polish(h, n_init, t7_end)
        if c7 < best_c:
            best_c = c7
            h = h7
    # Phase 8: Pair search for final refinement
    if time.time() < deadline - 10:
        t8_end = deadline - 5
        h, best_c = pair_search(h, n_init, t8_end,
                                [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                radius=50)
    # Final feasibility check
    h = project_feasible(h, n_init)
    best_c = compute_c5(h)
    return h, best_c, n_init
