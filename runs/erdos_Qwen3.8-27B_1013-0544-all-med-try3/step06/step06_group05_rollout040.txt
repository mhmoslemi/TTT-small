def project_box_sum(y, n):
    y = np.asarray(y, dtype=float)
    S = 0.5*n
    h = np.clip(y, 0.0, 1.0)
    for _ in range(12):
        diff = S - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot <= 1e-14:
                break
            h = h + (diff / tot) * h
    h = np.clip(h,0.0,1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        # bisection
        lo = float(np.min(y)) - 1.0
        hi = float(np.max(y))
        th = float(np.mean(y)) - 0.5
        if th < lo: th = lo
        if th > hi: th = hi
        for _ in range(30):
            v = np.clip(y - th, 0.0, 1.0)
            s = v.sum()
            if abs(s - S) < 1e-13:
                break
            if s > S:
                lo = th
            else:
                hi = th
            th = 0.5*(lo+hi)
        h = np.clip(y - th, 0.0, 1.0)
    # exact final single correction
    diff = S - h.sum()
    if diff > 0:
        idxs = np.where(h < 1.0 - 1e-15)[0]
        if idxs.size:
            # choose with largest room
            idx = idxs[np.argmax(1.0 - h[idxs])]
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            # if still diff, distribute among all room
            diff2 = S - h.sum()
            if abs(diff2) > 1e-14:
                idxs = np.where(h < 1.0)[0]
                if idxs.size:
                    room = 1.0 - h
                    room_h = np.zeros_like(h); room_h[idxs]=room[idxs]
                    tot=room_h.sum()
                    if tot>1e-14:
                        h = h + (diff2/tot)*room_h
    elif diff < 0:
        # similar
    h = np.clip(h,0.0,1.0)
    # one last tiny fix if needed: adjust a non-bound element with smallest impact
    diff = S - h.sum()
    if diff != 0.0:
        idxs = np.where((h > 1e-12) & (h < 1.0-1e-12))[0]
        if idxs.size:
            h[int(idxs[0])] += diff
            h = np.clip(h,0.0,1.0)
    return h

def compute_c_raw(h, n):

def smooth_value(c, alpha):
    if alpha <= 0.0:
        return float(np.max(c))
    m = float(np.max(c))
    x = alpha * (c - m)
    x = np.minimum(x, 0.0)
    z = np.exp(x)
    s = float(np.sum(z))
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / alpha


def logsum_stats(c, alpha):
    if alpha <= 0.0:
        m = float(np.max(c))
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        return m, w
    m = float(np.max(c))
    x = alpha * (c - m)
    x = np.minimum(x, 0.0)

        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        val = m
    else:

def sym_project(v, n):

    h = 0.5 * (v + v[::-1])

def resample_step(h_src, n_src, n_dst):
    n_src = int(n_src)
    n_dst = int(n_dst)
    if n_dst == n_src:
        return project_box_sum(np.asarray(h_src, dtype=float).copy(), n_dst)
    dx_s = 2.0 / n_src
    dx_d = 2.0 / n_dst

    x_d = (np.arange(n_dst) + 0.5) * dx_d
    v = np.interp(x_d, x_s, np.asarray(h_src, dtype=float))
    return project_box_sum(v, n_dst)

def convolution_mid(h, ws, n):
    n = int(n)
    if n <= 900:
        return np.convolve(h, ws, mode='full')[n - 1:2 * n - 1]
    need = 3 * n - 2
    m = 1
    while m < need:
        m <<= 1
    fa = np.fft.rfft(h, m)
    fw = np.fft.rfft(ws, m)
    out = np.fft.irfft(fa * fw, m)[:need]
    return out[n - 1:2 * n - 1]


def weighted_grad(w, h, n):
    n = int(n)
    ws = w + w[::-1]
    prod = convolution_mid(h, ws, n)

    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    p = np.arange(n)
    starts = n - 1 - p
    ends = 2 * n - 1 - p
    A = pref[ends] - pref[starts]
    return A - prod


def zero_mean_noise(h, n, rng, sigma):
    if n <= 1:
        return np.asarray(h, dtype=float).copy()
    noise = rng.normal(0.0, sigma, n)
    noise -= noise.mean()
    return np.asarray(h, dtype=float) + noise


def adam_refine(h, n, stage_deadline, rng, force_sym=False):
    n = int(n)
    h = project_box_sum(np.asarray(h, dtype=float).copy(), n)
    if force_sym:
        h = sym_project(h, n)

    dx = 2.0 / n
    c = compute_c_raw(h, n)
    best_c = float(np.max(c) * dx)
    best_h = h.copy()

    alphas = (20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0)
    for a in alphas:
        if time.time() > stage_deadline - 2.0:
            break
        alpha_raw = a * dx
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.02
        stale = 0

        while time.time() < stage_deadline - 2.0:
            max_raw = float(np.max(c))
            cur_c = max_raw * dx
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()

            val_old, w = logsum_stats(c, alpha_raw)
            g = weighted_grad(w, h, n)
            g -= g.mean()

            if float(np.max(np.abs(g))) < 1e-15:
                break

            t += 1
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            step = (m / (1.0 - 0.9 ** t)) / (np.sqrt(v / (1.0 - 0.999 ** t)) + 1e-8)
            step = np.clip(step, -1.0, 1.0)

            y = h - lr * step
            hn = project_box_sum(y, n)
            if force_sym:
                hn = sym_project(hn, n)

            cn = compute_c_raw(hn, n)
            maxn = float(np.max(cn))
            new_c = maxn * dx
            val_new = smooth_value(cn, alpha_raw) * dx
            val_old_s = val_old * dx

            accept = False
            if new_c < cur_c - 1e-13:
                accept = True
            elif new_c <= cur_c + 1e-9 and val_new < val_old_s - 1e-12 and new_c < best_c + 1e-7:
                accept = True

            if accept:
                h = hn
                c = cn
                lr = min(0.07, lr * 1.04)
                stale = 0
            else:
                lr *= 0.72
                stale += 1
                if lr < 1e-6:
                    h = project_box_sum(zero_mean_noise(h, n, rng, 0.015), n)
                    if force_sym:
                        h = sym_project(h, n)
                    c = compute_c_raw(h, n)
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 0
                    lr = 0.04
                if stale > 150:
                    break

    h = project_box_sum(h, n)
    if force_sym:
        h = sym_project(h, n)
    c = compute_c_raw(h, n)
    cur_c = float(np.max(c) * dx)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h.copy(), best_c


def pair_polish(h, n, stage_deadline, rng):
    n = int(n)
    if n < 2:
        h = project_box_sum(h, n)
        c = compute_c_raw(h, n)
        return h, float(np.max(c))

    h = project_box_sum(h, n)
    c = compute_c_raw(h, n)
    dx = 2.0 / n
    best_c = float(np.max(c) * dx)
    best_raw = best_c / dx
    best_h = h.copy()

    alpha_list = (20000.0, 8000.0, 3000.0, 1200.0, 500.0, 200.0)
    alpha_idx = 0
    alpha = alpha_list[0] * dx
    low = []
    high = []
    gvals = None
    attempts = 0
    no_improve = 0
    max_attempts = int(max(500, min(8000, 2200000 // max(1, n))))

    while time.time() < stage_deadline and attempts < max_attempts:
        if no_improve > 180 and alpha_idx < len(alpha_list) - 1:
            alpha_idx += 1
            alpha = alpha_list[alpha_idx] * dx
            no_improve = 0

        if not low or attempts % 40 == 0:
            _, w = logsum_stats(c, alpha)
            gvals = weighted_grad(w, h, n)
            gvals -= gvals.mean()
            order = np.argsort(gvals)
            sz = max(6, min(n, n // 15))
            low = order[:sz].tolist()
            high = order[-sz:].tolist()

        if low and high and rng.rand() < 0.75:
            i = int(low[int(rng.randint(len(low)))])
            j = int(high[int(rng.randint(len(high)))])
            if i == j:
                i = (i + 1) % n
            if gvals is not None and gvals[i] > gvals[j]:
                i, j = j, i
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            while i == j:
                j = int(rng.randint(n))
            if h[j] < h[i] - 1e-12 or (abs(h[j] - h[i]) <= 1e-12 and rng.rand() < 0.5):
                i, j = j, i

        max_step = min(1.0 - h[i], h[j])
        if max_step <= 1e-13:
            attempts += 1
            continue

        base = min(max_step, 0.05)
        ds = []
        for f in (1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002):
            d = base * f
            if d > max_step:
                d = max_step
            if d > 1e-8:
                ds.append(float(d))
        for f in (0.01, 0.003, 0.001, 0.0003, 0.0001, 3e-5, 1e-5, 3e-6, 1e-6, 3e-7):
            if f <= max_step:
                ds.append(float(f))

        ds = sorted(set(ds), reverse=True)
        if len(ds) > 16:
            ds = ds[:16]

        old_i = h[i]
        old_j = h[j]
        best_mv = best_raw
        best_hi = None
        best_hj = None

        for d in ds:
            hi = old_i + d
            hj = old_j - d
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                continue
            h[i] = hi
            h[j] = hj
            mv = float(np.max(compute_c_raw(h, n)))
            if mv < best_mv:
                best_mv = mv
                best_hi = hi
                best_hj = hj

        if best_hi is not None and (best_mv * dx) < best_c - 1e-14:
            h[i] = best_hi
            h[j] = best_hj
            c = compute_c_raw(h, n)
            best_c = best_mv * dx
            best_raw = best_mv
            best_h = h.copy()
            no_improve = 0
            attempts = 0
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
            attempts += 1

    h = project_box_sum(h, n)
    c = compute_c_raw(h, n)
    cur = float(np.max(c) * dx)
    if cur < best_c:
        best_c = cur
        best_h = h.copy()
    return best_h.copy(), best_c / dx


def make_restart(base_h, base_n, target_n, scale, rng, force_sym=False):
    target_n = int(target_n)
    if target_n == int(base_n):
        h = np.asarray(base_h, dtype=float).copy()
    else:
        h = resample_step(base_h, base_n, target_n)
    if scale > 0.0:
        h = zero_mean_noise(h, target_n, rng, scale)
    h = project_box_sum(h, target_n)
    if force_sym:
        h = sym_project(h, target_n)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    if deadline < t0 + 20.0:
        deadline = t0 + max(20.0, float(budget_s))
    rng = np.random.RandomState(seed)

    try:
        arr = np.asarray(initial_h_values, dtype=float)
    except Exception:
        arr = None

    if arr is None or arr.size == 0 or not np.all(np.isfinite(arr)):
        nf = 300
        x = (np.arange(nf) + 0.5) * (2.0 / nf)
        arr = np.clip(0.5 + 0.25 * np.sin(2.0 * np.pi * x) + rng.normal(0.0, 0.1, nf), 0.0, 1.0)

    if arr.size > 1000:
        h1, c1 = evaluate_candidate(arr, arr.size)
        h2, c2 = evaluate_candidate(resample_step(arr, arr.size, 800), 800)
        if c2 < c1:
            best_h, best_n, best_c = h2, 800, c2
        else:
            best_h, best_n, best_c = h1, int(arr.size), c1
    else:
        best_n = int(arr.size)
        best_h, best_c = evaluate_candidate(arr, best_n)

    base = best_h
    flips = [
        base.copy(),
        base[::-1].copy(),
        1.0 - base,
        0.5 * (base + base[::-1]),
    ]
    if best_n > 2:
        for off in (1, max(1, best_n // 4), max(1, best_n // 3), best_n // 2, (3 * best_n) // 4):
            flips.append(np.roll(base, off))
    else:
        flips.append(np.roll(base, 0))

    for cand in flips:
        hf, cf = evaluate_candidate(cand, best_n)
        if cf < best_c - 1e-15:
            best_h, best_c = hf, cf

    full_end = t0 + 0.25 * (deadline - t0)
    if time.time() < full_end - 15:
        ho, co = adam_refine(best_h, best_n, full_end - 10.0, rng, force_sym=False)
        if co < best_c - 1e-15:
            best_h, best_c = ho, co

    sym_end = t0 + 0.40 * (deadline - t0)
    if time.time() < sym_end - 20:
        hsym = sym_project(best_h, best_n)
        ho, co = adam_refine(hsym, best_n, sym_end - 10.0, rng, force_sym=True)
        if co < best_c - 1e-15:
            best_h, best_c = ho, co

    high_end = t0 + 0.80 * (deadline - t0)
    high_targets = []
    raw_high = [
        int(round(best_n * 1.5)),
        int(round(best_n * 2.5)),
        1000,
    ]
    for t in raw_high:
        t = max(150, min(1000, int(t)))
        if t > best_n * 0.85 and t not in high_targets:
            high_targets.append(t)
    high_targets.sort()
    high_targets = high_targets[:3]

    for tn in high_targets:
        if time.time() > high_end - 45:
            break
        ht = resample_step(best_h, best_n, tn)
        ht, ct = evaluate_candidate(ht, tn)
        if ct < best_c - 1e-15:
            best_h, best_n, best_c = ht, tn, ct

        alloc = min(75.0, high_end - time.time() - 25.0)
        if alloc > 15.0:
            ho, co = adam_refine(ht, tn, time.time() + alloc, rng, force_sym=False)
            if co < best_c - 1e-15:
                best_h, best_n, best_c = ho, tn, co
            if time.time() < high_end - 35:
                hs = sym_project(ho, tn)
                ho2, co2 = adam_refine(hs, tn, min(high_end - 20.0, time.time() + 0.45 * alloc), rng, force_sym=True)
                if co2 < best_c - 1e-15:
                    best_h, best_n, best_c = ho2, tn, co2

    restart_scales = (0.010, 0.020, 0.040, 0.070, 0.110)
    r = 0
    while time.time() < high_end - 35 and r < 6:
        nr = best_n
        if r % 3 == 1 and best_n > 800:
            nr = min(best_n, 600)
        scale = restart_scales[r % len(restart_scales)]
        hp = make_restart(best_h, best_n, nr, scale, rng, force_sym=(r % 4 == 3))
        hp, cp = evaluate_candidate(hp, nr)
        if cp < best_c - 1e-15:
            best_h, best_n, best_c = hp, nr, cp

        rem = high_end - time.time() - 25.0
        if rem < 15.0:
            break
        alloc = min(45.0, rem * 0.55)
        ho, co = adam_refine(hp, nr, time.time() + alloc, rng, force_sym=(r % 4 == 3))
        if co < best_c - 1e-15:
            best_h, best_n, best_c = ho, nr, co
        r += 1

    final_ns = []
    if best_n <= 900:
        final_ns.append(int(best_n))
    final_ns.extend([500, 300])
    cleaned = []
    for x in final_ns:
        x = int(x)
        if x >= 100 and x not in cleaned:
            cleaned.append(x)
    final_ns = cleaned[:2]

    for tn in final_ns:
        rem = deadline - time.time() - 10.0
        if rem < 25.0:
            break
        hp = resample_step(best_h, best_n, tn)
        hp, cp = evaluate_candidate(hp, tn)
        if cp < best_c - 1e-15:
            best_h, best_n, best_c = hp, tn, cp

        alloc = min(220.0, rem * 0.45)
        adam_alloc = min(70.0, max(12.0, alloc * 0.35))
        adam_deadline = time.time() + adam_alloc
        ho, co = adam_refine(hp, tn, adam_deadline, rng, force_sym=False)
        if co < best_c - 1e-15:
            best_h, best_n, best_c = ho, tn, co

        if time.time() < deadline - 8.0:
            pair_deadline = min(deadline - 4.0, time.time() + max(10.0, alloc - adam_alloc))
            po, pc = pair_polish(ho, tn, pair_deadline, rng)
            if pc < best_c - 1e-15:
                best_h, best_n, best_c = po, tn, pc

    if best_c > 0.38085 and best_n <= 900 and time.time() < deadline - 5.0:
        po, pc = pair_polish(best_h, best_n, deadline - 3.0, rng)
        if pc < best_c - 1e-15:
            best_h, best_n, best_c = po, best_n, pc

    best_h = project_box_sum(np.asarray(best_h, dtype=float), int(best_n))
    c_raw = compute_c_raw(best_h, best_n)
    best_c = float(np.max(c_raw) * (2.0 / best_n))
    return best_h, best_c, int(best_n)
