    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    """Return full correlation array (not just max)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = h.copy()
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h_proj
            tot = room.sum()
            if tot > 1e-14:
                h_proj += (diff / tot) * room
        else:
            tot = h_proj.sum()
            if tot > 1e-14:
                h_proj += (diff / tot) * h_proj
    return np.clip(h_proj, 0.0, 1.0)

def c5_gradient(h, n, k_star):
    """Compute gradient of f_{k_star} = corr[k_star] * dx."""
    dx = 2.0 / n
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n-lag] += (1.0 - h[lag:])
        if lag > 0:
            grad[lag:] += -h[:n-lag]
    else:
        p = -lag
        grad[p:] += (1.0 - h[:n-p])
        if p > 0:
            grad[:n-p] += -h[p:]
    return grad * dx

        grad = c5_gradient(h, n, k_star)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

        c1 = compute_c5(h1, n_init)
        best_h, best_c = h1.copy(), c1

        if alloc1 > 10:
            for alpha in [2000, 20000, 200000]:
                if time.time() > t0 + alloc1:
                    break
                ht = slsqp_optimize(best_h, n_init, alpha, maxiter=2000)
                ct = compute_c5(ht, n_init)
                if ct < best_c - 1e-14:
                    best_c = ct
                    best_h = ht

        # Phase 2: Upsample to n=500 and optimize
        n_target = 500
        if n_init < n_target:
            h_up = upsample_h(best_h, n_init, n_target)
        else:
            h_up = best_h.copy()
            n_target = n_init

        c_up = compute_c5(h_up, n_target)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()

        # SLSQP warm start at higher resolution
        alloc2 = min(120, (deadline - time.time()) * 0.18)
        if alloc2 > 20 and time.time() < deadline - 200:
            for alpha in [5000, 50000, 500000, 5000000]:
                if time.time() > time.time() and time.time() > t0 + alloc1 + alloc2:
                    break
                if time.time() > deadline - 600:
                    break
                ht = slsqp_optimize(best_h, n_target, alpha, maxiter=3000)
                ct = compute_c5(ht, n_target)
                if ct < best_c - 1e-14:
                    best_c = ct
                    best_h = ht

        # Phase 3: Subgradient optimization at n=500
        alloc3 = min(400, (deadline - time.time()) * 0.45)
        if time.time() < deadline - 200:
            h_sg, c_sg = subgradient_optimize(best_h, n_target, time.time() + alloc3, rng,
                                             lambda_reg=0.003)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg

            # Second round with different regularization
            if time.time() < deadline - 150:
                h_sg2, c_sg2 = subgradient_optimize(best_h, n_target,
                                                    time.time() + min(150, deadline - time.time() - 100),
                                                    rng, lambda_reg=0.01)
                if c_sg2 < best_c:
                    best_c = c_sg2
                    best_h = h_sg2

        # Phase 4: Local refinement
        if time.time() < deadline - 60:
            n_cur = len(best_h)
            if n_cur <= 250:
                best_h, best_c = pair_search_focused(best_h, n_cur,
                         time.time() + min(80, deadline - time.time() - 30),
                         range(n_cur), [0.01, 0.005, 0.002, 0.001, 0.0005])
            else:
                best_h, best_c = near_pair_search(best_h, n_cur,
                         time.time() + min(120, deadline - time.time() - 30),
                         [0.01, 0.005, 0.002, 0.001], radius=60)

        # Phase 5: Final SLSQP polish
        if time.time() < deadline - 30:
            n_cur = len(best_h)
            ht = slsqp_optimize(best_h, n_cur, 10000000, maxiter=4000)
            ct = compute_c5(ht, n_cur)
            if ct < best_c - 1e-14:
                best_c = ct
                best_h = ht

        # Phase 6: Final micro pair search
        if time.time() < deadline - 5:
            n_cur = len(best_h)
            if n_cur <= 250:
                best_h, best_c = pair_search_focused(best_h, n_cur,
                         deadline - 2, range(n_cur),
                         [0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = near_pair_search(best_h, n_cur,
                         deadline - 2, [0.001, 0.0005, 0.0001], radius=40)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
