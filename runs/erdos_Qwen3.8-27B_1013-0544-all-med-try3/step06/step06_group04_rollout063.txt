def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Use diminishing step sizes
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        M = c.max()

            best_c = M
            best_h = h.copy()

        threshold = M - 0.005

                best_step_h = h_try

def _outer_approx_step(h_curr, n, trust=0.1):

    c = compute_all_lags(h_curr, n)

    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for idx in threshold_idx:
        lag = idx - (n - 1)
        grad = _lag_gradient(h_curr, n, lag)
        cons.append(c[idx] + dx * (grad / dx) @ (h_var - h_curr) <= t_var)
    # Wait, c[idx] already includes dx. And grad already includes dx.
    # c[idx] + (grad @ (h_var - h_curr)) <= t_var
    # Let me fix:
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]

        grad = _lag_gradient(h_curr, n, lag)

    prob = cp.Problem(cp.Minimize(t_var), cons)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
        return grad
    m = abs(lag)
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] += -h[:n - m] * dx
    else:
        grad[m:] += (1.0 - h[:n - m]) * dx
        grad[:n - m] += -h[m:] * dx
    return grad

def _sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_eq_jac(x, n):
    return np.ones(n)

def _slsqp_full(h, n, alpha, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]

    def _f(x, n, alpha, dx):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            sh = shifted - mx
            w = np.exp(sh)
            ws = w.sum()
            return m + (mx + np.log(ws)) / alpha
        w = np.exp(shifted)
        ws = w.sum()
        return m + np.log(ws) / alpha

    def _g(x, n, alpha, dx):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            sh = shifted - mx
            w = np.exp(sh)
        else:
            w = np.exp(shifted)
        ws = w.sum()
        nw = w / ws
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = nw[k]
            if wk > 1e-12:
                lag = k - (n - 1)
                if lag == 0:
                    grad += wk * (1.0 - 2.0 * x) * dx
                elif 0 < lag < n:
                    grad[:n - lag] += wk * (1.0 - x[lag:]) * dx
                    grad[lag:] += wk * (-x[:n - lag]) * dx
                elif -n < lag < 0:
                    m2 = -lag
                    grad[m2:] += wk * (1.0 - x[:n - m2]) * dx
                    grad[:n - m2] += wk * (-x[m2:]) * dx
        return grad

    try:
        res = minimize(_f, h, args=(n, alpha, dx), jac=_g,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _slsqp_top(h, n, top_idx, alpha, maxiter=3000, ftol=1e-14):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]

    def _f2(x, n, top_idx, alpha, dx):
        c = np.correlate(x, 1 - x, mode='full') * dx
        active = c[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            sh = shifted - mx
            w = np.exp(sh)
            ws = w.sum()
            return m + (mx + np.log(ws)) / alpha
        w = np.exp(shifted)
        ws = w.sum()
        return m + np.log(ws) / alpha

    def _g2(x, n, top_idx, alpha, dx):
        c = np.correlate(x, 1 - x, mode='full') * dx
        active = c[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        if mx > 500:
            sh = shifted - mx
            w = np.exp(sh)
        else:
            w = np.exp(shifted)
        ws = w.sum()
        nw = w / ws
        grad = np.zeros(n)
        for j in range(len(top_idx)):
            wk = nw[j]
            if wk > 1e-12:
                lag = int(top_idx[j]) - (n - 1)
                if lag == 0:
                    grad += wk * (1.0 - 2.0 * x) * dx
                elif 0 < lag < n:
                    grad[:n - lag] += wk * (1.0 - x[lag:]) * dx
                    grad[lag:] += wk * (-x[:n - lag]) * dx
                elif -n < lag < 0:
                    m2 = -lag
                    grad[m2:] += wk * (1.0 - x[:n - m2]) * dx
                    grad[:n - m2] += wk * (-x[m2:]) * dx
        return grad

    try:
        res = minimize(_f2, h, args=(n, top_idx, alpha, dx), jac=_g2,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _outer_approx(h_curr, n, trust=0.1):
    c = compute_all_lags(h_curr, n)
    n_lags = 2 * n - 1
    top_half = np.argsort(c)[n_lags // 2:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for idx in top_half:
        lag = int(idx) - (n - 1)
        grad = _lag_gradient(h_curr, n, lag)
        cons.append(c[idx] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solved = False
    for solver in [cp.GLOP, cp.CBC, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved or prob.status not in ['optimal', 'optimal_inaccurate'] or h_var.value is None:
        return h_curr.copy()
    return project_feasible(np.clip(h_var.value, 0, 1), n)

def _subgradient_polish(h, n, deadline, n_iters=2000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = 0.0
    beta = 0.9
    base_alpha = 0.02
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        M = float(c.max())
        if M < best_c - 1e-14:
            best_c = M
            best_h = h.copy()
        top_k = 15
        top_pos = np.argsort(c)[-top_k:]
        active_pos = top_pos[c[top_pos] >= M - 0.002]
        if len(active_pos) == 0:
            active_pos = top_pos[:5]
        g = np.zeros(n)
        for pos in active_pos:
            lag = int(pos) - (n - 1)
            g += _lag_gradient(h, n, lag)
        g /= len(active_pos)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        momentum = beta * momentum - base_alpha * g
        h_try = project_feasible(h + momentum, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            h = h_try
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        else:
            h_try2 = project_feasible(h - 0.5 * base_alpha * g, n)
            c_try2 = compute_c5(h_try2, n)
            if c_try2 < best_c - 1e-14:
                h = h_try2
                if c_try2 < best_c:
                    best_c = c_try2
                    best_h = h_try2.copy()
                momentum = 0.0
            else:
                break
    return best_h, best_c

def _subgradient_plain(h, n, deadline, n_iters=1000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        M = float(c.max())
        if M < best_c - 1e-14:
            best_c = M
            best_h = h.copy()
        top_k = 10
        top_pos = np.argsort(c)[-top_k:]
        active_pos = top_pos[c[top_pos] >= M - 0.003]
        if len(active_pos) == 0:
            active_pos = top_pos[:3]
        g = np.zeros(n)
        for pos in active_pos:
            lag = int(pos) - (n - 1)
            g += _lag_gradient(h, n, lag)
        g /= len(active_pos)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        improved = False
        for scale in [0.05, 0.02, 0.01, 0.005, 0.002]:
            h_try = project_feasible(h - scale * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h_try.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c
