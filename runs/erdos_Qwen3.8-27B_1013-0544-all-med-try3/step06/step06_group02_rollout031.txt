                            h[i] -= step; h[j] += step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_improvement = best_c - c
                                best_move = (i, j, -step, step)
                                best_c = c
                            else:
                                h[i] += step; h[j] -= step
                    else:
                        if h[i] + step <= 1 and h[j] - step >= 0:
                            h[i] += step; h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_improvement = best_c - c
                                best_move = (i, j, step, -step)
                                best_c = c
                            else:
                                h[i] -= step; h[j] += step
        
        if best_move is None:
            base_step *= 0.5
            if base_step < 1e-6:
                break
    
    return h, best_c

def multiresolution_refine(h, n, deadline):
    """Go to coarser grid, optimize, come back."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 30:
            continue
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        # Quick L-BFGS at coarse level
        t_c = min(30, (deadline - time.time()) * 0.25)
        if t_c > 15:
            h_c, c_c = run_lbfgs_full(h_c, n_coarse, time.time() + t_c,
                                       configs=[(200, 50.0), (1000, 500.0), (3000, 5000.0)])
        
        # Upsample
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Initial L-BFGS at original resolution
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = run_lbfgs_full(best_h, n_orig, t1,
        configs=[(100, 10.0), (500, 100.0), (2000, 1000.0), (5000, 10000.0)])

    # Phase 2: Upsample to n=500 for finer resolution
    n = 500
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_new_res = np.interp(x_new, x_old, best_h)
    h_new_res = project_feasible(h_new_res, n)
    c_new_res = compute_c5(h_new_res, n)
    if c_new_res < best_c:
        best_h = h_new_res
        best_c = c_new_res
    else:
        # Keep trying at higher res anyway
        best_h = h_new_res
        best_c = c_new_res

    # Phase 3: L-BFGS at n=500
    if time.time() < deadline - 350:
        t3 = min(time.time() + 180, deadline - 350)
        best_h, best_c = run_lbfgs_full(best_h, n, t3,
            configs=[(200, 50.0), (1000, 500.0), (3000, 5000.0),
                     (5000, 50000.0), (10000, 200000.0)])

    # Phase 4: Top-shifts focused optimization
    if time.time() < deadline - 250:
        t4 = min(time.time() + 120, deadline - 250)
        best_h, best_c = run_top_shifts_opt(best_h, n, t4, top_k=5)

    # Phase 5: SLSQP polish
    if time.time() < deadline - 180:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 160:
                break
            t5 = min(time.time() + 50, deadline - 160)
            h5, c5 = run_slqp(best_h, n, t5, alpha=alpha, maxiter=3000)
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Another top-shifts round
    if time.time() < deadline - 120:
        t6 = min(time.time() + 60, deadline - 120)
        best_h, best_c = run_top_shifts_opt(best_h, n, t6, top_k=3)

    # Phase 7: Gradient-guided search
    if time.time() < deadline - 60:
        t7 = min(time.time() + 50, deadline - 60)
        best_h, best_c = gradient_guided_search(best_h, n, t7, base_step=0.002)

    # Phase 8: Fine pair search
    if time.time() < deadline - 20:
        t8 = min(time.time() + 30, deadline - 20)
        best_h, best_c = fine_pair_search(best_h, n, t8, center_radius=25)

    # Phase 9: Final SLSQP
    if time.time() < deadline - 5:
        h9, c9 = run_slqp(best_h, n, deadline - 2, alpha=500000, maxiter=2000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: also try at original resolution
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
