from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_autocorr_full(h):
    return np.correlate(h, h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_min_obj_and_grad(h, alpha):
    """Returns (value, gradient) of the soft-min of autocorrelation."""
    n = len(h)
    R = compute_autocorr_full(h)
    shifted = -alpha * R
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    p /= p_sum
    value = -((shifted.max() + np.log(p_sum)) / alpha)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[:n-d] += pk * h[d:]
            grad[d:] += pk * h[:n-d]
        else:
            dd = -d
            grad[dd:] += pk * h[:n-dd]
            grad[:n-dd] += pk * h[dd:]
    return value, grad

def projected_gradient(h, n, deadline, alphas, max_iter_per_alpha=3000, lr_base=0.05):
    """Projected gradient ascent on soft-min autocorrelation."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = h.copy()
        lr = lr_base
        f_cur = soft_min_obj_and_grad(h_cur, alpha)[0]
        
        stall_count = 0
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            f_cur, grad = soft_min_obj_and_grad(h_cur, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-14:
                break
            
            # Normalize gradient for stable step size
            grad_dir = grad / grad_norm
            
            # Try step
            step = lr
            h_try = project_feasible(h_cur + step * grad_dir * grad_norm, n)
            f_try = soft_min_obj_and_grad(h_try, alpha)[0]
            
            if f_try > f_cur:
                h_cur = h_try
                f_cur = f_try
                stall_count = 0
                # Occasionally increase step
                if it % 200 == 0:
                    lr = min(0.5, lr * 1.05)
            else:
                stall_count += 1
                lr *= 0.7
                if stall_count > 50:
                    break
        
        h = h_cur
        c_cur = compute_c5(h, n)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_passes=200):
    """Local pair search for final polish."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj

def slsqp_optimize(h, n, alpha, maxiter=3000):
    """SLSQP optimization with smooth max objective."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        return float(m + np.log(np.sum(np.exp(shifted - shifted.max())) - np.log(len(shifted))) / alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        shifted -= shifted.max()
        p = np.exp(shifted)
        p /= p.sum()
        # Gradient of max c[k] w.r.t. x
        # c[k] = sum_i x[i] * (1-x[i+k-(n-1)]) * dx
        # dc[k]/dx[j] = (1-x[j+k-(n-1)]) - x[j-k+(n-1)] (with boundary handling) * dx
        grad = np.zeros(n)
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-18:
                continue
            d = k - (n-1)
            if d >= 0:
                # (1-x[j+d]) for j=0..n-1-d
                grad[:n-d] += pk * (1 - x[d:]) * dx
                # -x[j-d] for j=d..n-1
                grad[d:] -= pk * x[:n-d] * dx
            else:
                dd = -d
                # (1-x[j-dd]) for j=dd..n-1
                grad[dd:] += pk * (1 - x[:n-dd]) * dx
                # -x[j+dd] for j=0..n-1-dd
                grad[:n-dd] -= pk * x[dd:] * dx
        return grad
    
    def sum_con(x):
        return float(np.sum(x) - n/2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Optimize at a given resolution with multiple methods."""
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()

            c_s = compute_c5(h_s, n)

                best_c = c_s
                best_h = h_s

    best_h = project_feasible(best_h, n)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_autocorr_full(h):
    return np.correlate(h, h, mode='full')

def project_feasible_fast(v, n):
    target = n / 2.0
    vmin = v.min()
    vmax = v.max()
    lo = -vmax - 0.1
    hi = 1.0 - vmin + 0.1

        if s < target:

        else:
            hi = mid

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    shifted -= shifted.max()

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    log_sum = shifted.max() + np.log(np.sum(np.exp(shifted)) - 1.0 + 1e-30)
    return float(m + (log_sum - np.log(len(shifted))) / alpha)

def _slsqp_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[:n - d] += pk * (1 - x[d:]) * dx
            grad[d:] -= pk * x[:n - d] * dx
        else:
            dd = -d
            grad[dd:] += pk * (1 - x[:n - dd]) * dx
            grad[:n - dd] -= pk * x[dd:] * dx
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), jac=_slsqp_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimize_at_resolution(h_init, n, deadline, rng):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Projected gradient with alpha continuation
    alphas = [50, 200, 500, 2000, 5000]
    alloc1 = (deadline - time.time()) * 0.45
    if alloc1 > 5:
        h_pg, c_pg = projected_gradient(h, n, time.time() + alloc1, alphas)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg

    # Phase 2: Worst-lag gradient
    if time.time() < deadline - 20:
        alloc2 = (deadline - time.time()) * 0.35
        if alloc2 > 5:
            h_wl, c_wl = worst_lag_gradient(best_h, n, time.time() + alloc2)
            if c_wl < best_c:
                best_c = c_wl
                best_h = h_wl

    # Phase 3: SLSQP polish
    if time.time() < deadline - 25:
        for a in [10000, 100000]:
            if time.time() > deadline - 25:
                break
            h_s = slsqp_optimize(best_h, n, a, maxiter=2000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 4: Pair search polish
    if time.time() < deadline - 5:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005], max_passes=150)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.01, 0.005, 0.002, 0.001], radius=40)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 250

    best_h = None
    best_c = float('inf')

    # Strategy 1: Optimize parent at its native resolution
    if h_parent is not None:
        alloc = (deadline - t0) * 0.30
        h_opt, c_opt = optimize_at_resolution(h_parent, n_parent, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Strategy 2: Binary initialization
    if h_parent is not None and time.time() < deadline - 60:
        h_bin = (h_parent > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n_parent)
        alloc = min(70, (deadline - time.time()) * 0.20)
        if alloc > 15:
            h_opt, c_opt = optimize_at_resolution(h_bin, n_parent, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Strategy 3: Different resolutions
    if h_parent is not None and time.time() < deadline - 70:
        x_parent = np.linspace(0, 2, n_parent)
        resolutions = [120, 180, 300, 400]
        for n_try in resolutions:
            if n_try == n_parent:
                continue
            remaining = deadline - time.time()
            alloc = min(55, remaining * 0.15)
            if alloc < 20:
                break
            h_interp = np.interp(np.linspace(0, 2, n_try), x_parent, h_parent)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Strategy 4: Random restarts for diversity
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(2, int(remaining / 35))
        scales = [0.03, 0.06, 0.1, 0.15, 0.02]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(30, (deadline - time.time() - 25) * 0.4)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Strategy 5: Structured initializations (periodic patterns)
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        for period in [10, 15, 20, 25, 30]:
            if time.time() > deadline - 35:
                break
            x = np.linspace(0, 2, n)
            h_periodic = 0.5 + 0.5 * np.sin(2.0 * np.pi * x / (2.0 / period))
            h_periodic = project_feasible(np.clip(h_periodic, 0, 1), n)
            alloc = min(20, (deadline - time.time() - 30) * 0.3)
            if alloc < 8:
                break
            h_opt, c_opt = optimize_at_resolution(h_periodic, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], max_passes=300)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001], radius=60)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_parent
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
