def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_objective(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def constraint(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h

    current_c = best_c

    for step in range(10000000):
        if time.time() > deadline:
            break
        # Annealing schedule
        progress = step / 10000000.0  # won't reach here but for schedule

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-13:

            if s < 1e-13:

def sa_binary(h_bin, n, deadline, seed=42, t_start=0.02, t_end=1e-7, n_steps=5000000):
    """Simulated annealing for binary solutions"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_bin.copy()
    target = n // 2
    
    # Ensure we have exactly target ones
    if h.sum() != target:
        idx = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[idx[:target]] = 1.0
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    corr = np.correlate(h, 1.0 - h, 'full')
    current_c = float(corr.max() * dx)
    best_h = h.copy()
    best_c = current_c
    no_improve = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        # Geometric cooling
        T = t_start * (t_end / t_start) ** (float(step) / 500000.0)
        if T < t_end:
            T = t_end
        
        # Random swap
        ii = rng.randint(n_zeros)
        jj = rng.randint(n_ones)
        i = zeros[ii]
        j = ones[jj]
        
        # Swap
        h[i], h[j] = 1.0, 0.0
        new_corr = np.correlate(h, 1.0 - h, 'full')
        new_c = float(new_corr.max() * dx)
        
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c = new_c
            corr = new_corr
            zeros[ii] = j
            ones[jj] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = 0.0, 1.0
            no_improve += 1
        
        # Early stopping if stuck for too long at low temperature
        if T < 1e-5 and no_improve > 50000:
            break
    
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def con(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

Actually, I just realized something important. The `compute_c5` function computes the maximum of the full correlation array. The full correlation has 2n-1 elements, corresponding to shifts from -(n-1) to (n-1). But in the original problem, the shift k ranges over [0, 2] (or rather, the integral is over the valid range). Wait, let me re-read the problem. The integral is ∫ h(x)(1-h(x+k)) dx, and k can be any real number. But h is defined on [0,2], so for the integral to make sense, we need x ∈ [0,2] and x+k ∈ [0,2], which means x ∈ [0, 2] ∩ [-k, 2-k].

In the discrete version, np.correlate(h, 1-h, mode='full') gives the sum of h[i]*(1-h[j]) where j-i = lag. The "full" mode includes all possible lags from -(n-1) to (n-1). The maximum of this (times dx) is C5.

So the problem is: find h to minimize the maximum of the cross-correlation between h and 1-h over all lags. This is what the code does.

Now, one subtle point: the "full" correlation includes lags where the overlap is partial (near the edges). These might give smaller values than the central lags (where the full overlap is available). So the maximum is usually achieved at a central lag.
