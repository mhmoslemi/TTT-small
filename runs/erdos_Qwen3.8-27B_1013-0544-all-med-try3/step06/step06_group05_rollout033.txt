import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_full_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(x, target_sum):
    n = len(x)
    lo = -x.max() - 2.0
    hi = 1.0 - x.min() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def grad_smooth_max(h, c, alpha, dx):
    n = len(h)
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    pn = p.sum()
    if pn < 1e-300:
        return np.zeros(n)
    p /= pn
    omh = 1.0 - h
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = p[ki]
        if wk < 1e-14:
            continue
        k = ki - (n - 1)
        if k >= 0:
            ln = n - k
            if ln > 0:
                grad[k:] += wk * omh[:ln]
                grad[:ln] -= wk * h[k:]
        else:
            kk = -k
            ln = n - kk
            if ln > 0:
                grad[:ln] += wk * omh[kk:]
                grad[kk:] -= wk * h[:ln]
    return grad * dx

def pgd_optimize(h, n, deadline, target):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 400000]
    remaining = deadline - time.time()
    time_per = remaining / len(alphas)
    for alpha in alphas:
        t_end = time.time() + time_per
        if time.time() > t_end:
            continue
        c = compute_full_corr(h, n)
        f = smooth_max(c, alpha)
        lr = 0.02
        v = np.zeros(n)
        momentum = 0.8
        stall = 0
        for it in range(200000):
            if time.time() > t_end:
                break
            g = grad_smooth_max(h, c, alpha, dx)
            v = momentum * v - lr * g
            step = v
            if np.max(np.abs(step)) > 0.1:
                step = step * 0.1 / np.max(np.abs(step))
            h_new = project_box_sum(h + step, target)
            c_new = compute_full_corr(h_new, n)
            f_new = smooth_max(c_new, alpha)
            if f_new <= f + 1e-10:
                if f_new < f - 1e-14:
                    stall = 0
                    lr = min(lr * 1.05, 0.5)
                else:
                    stall += 1
                    if stall > 50:
                        lr *= 0.8
                        stall = 0
                h = h_new
                c = c_new
                f = f_new
                cv = float(c_new.max())
                if cv < best_c - 1e-14:
                    best_c = cv
                    best_h = h.copy()
            else:
                lr = max(lr * 0.5, 1e-7)
                v = momentum * 0.5 * v
    return best_h, best_c

def grad_guided_search(h, n, deadline, target):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    top_k = 20
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            c = compute_full_corr(h, n)
            alpha = 50000
            g = grad_smooth_max(h, c, alpha, dx)
            dec_idx = np.argsort(g)[::-1][:top_k]
            inc_idx = np.argsort(g)[:top_k]
            improved = False
            for i in dec_idx:
                if time.time() > deadline:
                    break
                for j in inc_idx:
                    if i == j:
                        continue
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[j] -= d
                        h[i] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not improved:
                break
        h = project_box_sum(h, target)
    return h, best_c

def full_pair_search(h, n, deadline, target, deltas, max_passes=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    hi = h[i] - d
                    hj = h[j] + d
                    if 0.0 <= hi and hj <= 1.0:
                        h[i], h[j] = hi, hj
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i], h[j] = hi + d, hj - d
                    hi = h[i] + d
                    hj = h[j] - d
                    if hi <= 1.0 and 0.0 <= hj:
                        h[i], h[j] = hi, hj
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, target, deltas, radius=60, max_passes=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi and hi <= 1.0 and 0.0 <= hj:
                        h[i], h[j] = hi, hj
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def steepest_descent_pass(h, n, target, top_k=10):
    c = compute_full_corr(h, n)
    alpha = 100000
    dx = 2.0 / n
    g = grad_smooth_max(h, c, alpha, dx)
    dec_idx = np.argsort(g)[::-1][:top_k]
    inc_idx = np.argsort(g)[:top_k]

    for d in [0.05, 0.03, 0.01, 0.005, 0.002]:

def multi_restart(h, n, deadline, target, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    remaining = deadline - time.time()
    n_restarts = max(3, int(remaining / 40))
    scales = [0.003, 0.008, 0.015, 0.025, 0.04, 0.06]
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        scale = scales[r % len(scales)]
        hp = best_h + rng.normal(0, scale, n)
        hp = project_box_sum(hp, target)
        alloc = min(35, (deadline - time.time() - 10) * 0.4)

            break
        ht, ct = pgd_optimize(hp, n, time.time() + alloc, target)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
