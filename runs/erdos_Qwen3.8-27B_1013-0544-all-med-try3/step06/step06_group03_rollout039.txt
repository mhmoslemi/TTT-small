For k≠0: c_k = ∫₀^{2-|k|} 0.5 * 0.5 dx = 0.25 * (2-|k|) (in continuous case)

So the max is at k=0 and equals 0.5. This is the "trivial" solution with C₅=0.5.

The optimal balance is what we're trying to find.

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def corr_and_weights(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    if shifted.max() > 500:
        shifted -= shifted.max()
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    w = exp_s / Z
    f = (max_c + np.log(Z)) / alpha
    return f, corr, w

def gradient_from_weights(h, n, w):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        i_lo = max(0, j - nm1)
        i_hi = min(n - 1, j + nm1)
        k_lo = i_lo - j + nm1
        k_hi = i_hi - j + nm1
        s1 = np.dot(w[k_lo:k_hi+1], one_minus_h[i_lo:i_hi+1])
        k_lo2 = max(0, j)
        k_hi2 = min(2*n-2, j + nm1)
        i_vals = j - np.arange(k_lo2, k_hi2+1) + nm1
        s2 = np.dot(w[k_lo2:k_hi2+1], h[i_vals])
        grad[j] = dx * (s1 - s2)
    return grad

def projected_gradient_opt(h, n, deadline, alphas, max_iters=3000, momentum=0.85):
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        v = np.zeros(n)
        f, corr, w = corr_and_weights(h, n, alpha)
        grad = gradient_from_weights(h, n, w)
        v = grad.copy()
        no_improve = 0
        f_best_iter = f
        
        for it in range(max_iters):
            if time.time() > deadline - 10:
                break
            h_new = project_sum_box(h - 0.5 * v, target)
            f_new, corr_new, w_new = corr_and_weights(h_new, n, alpha)
            
            if f_new > f:
                step = 0.25
                for _ in range(30):
                    h_try = project_sum_box(h - step * v, target)
                    f_try, _, _ = corr_and_weights(h_try, n, alpha)
                    if f_try < f:
                        h_new = h_try
                        f_new = f_try
                        break
                    step *= 0.5
                else:
                    v *= 0.5
                    no_improve += 1
                    if no_improve > 50:
                        v = grad.copy()
                        no_improve = 0
                    continue
            
            h = h_new
            f = f_new
            f, corr, w = corr_and_weights(h, n, alpha)
            grad = gradient_from_weights(h, n, w)
            v = momentum * v + grad
            
            if f < f_best_iter - 1e-10:
                f_best_iter = f
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 200:
                break
            
            c_act = float(np.max(corr))
            if c_act < best_c - 1e-12:
                best_c = c_act
                best_h = h.copy()
    
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        sh -= sh.max()
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    if shifted.max() > 500:
        shifted -= shifted.max()
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    return gradient_from_weights(x, n, w)

def slsqp_constraint_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_constraint_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint_fun, 'jac': slsqp_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), jac=lambda x, nn, aa: slsqp_grad(x, nn, aa),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_pol = project_sum_box(res.x, n / 2.0)
            if compute_c5(h_pol, n) < compute_c5(h, n) + 1e-10:
                return h_pol
    except Exception:
        pass
    return h.copy()

def active_set_refine(h, n, deadline, top_k=5):
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    
    for outer in range(30):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        active = np.argsort(corr)[-top_k:]
        c_active = corr[active]
        c_mean = c_active.mean()
        c_max = corr.max()
        
        if c_max - c_active.min() < 1e-6:
            break
        
        # Minimize sum of squares of active lags via projected gradient
        a_idx = active
        n_step = 0
        for it in range(200):
            if time.time() > deadline:
                return best_h, best_c
            # Gradient of sum of c_k^2 for active k
            corr_raw = np.correlate(h, 1 - h, mode='full')
            grad = np.zeros(n)
            for k in a_idx:
                tau = k - (n - 1)
                c_k = corr_raw[k] * dx
                for j in range(n):
                    val = 0.0
                    if 0 <= j + tau < n:
                        val += dx * (1 - h[j + tau])
                    if 0 <= j - tau < n:
                        val += -dx * h[j - tau]
                    grad[j] += 2.0 * c_k * val
            step_size = 0.01
            for _ in range(20):
                h_new = project_sum_box(h - step_size * grad, target)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    break
                step_size *= 0.5
            else:
                break
    
    best_h = project_sum_box(h, target)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def pair_search_local(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    target = n / 2.0
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def make_structured_start(n, rng, type_idx):
    x = np.linspace(0, 2, n)
    if type_idx == 1:
        m = int(rng.randint(2, 12))
        A = rng.uniform(0.15, 0.45)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + A * np.sin(2 * np.pi * m * x / 2 + phase)
    elif type_idx == 2:
        h = np.full(n, 0.5)
        for _ in range(3):
            m = int(rng.randint(2, 10))
            A = rng.uniform(0.05, 0.18)
            phase = rng.uniform(0, 2 * np.pi)
            h += A * np.sin(2 * np.pi * m * x / 2 + phase)
    elif type_idx == 3:
        k = int(rng.randint(3, 9))
        breaks = np.sort(rng.uniform(0.1, 1.9, k - 1))
        h = np.zeros(n)
        for i in range(k):
            lo = breaks[i-1] if i > 0 else 0.0
            hi = breaks[i] if i < k - 1 else 2.0
            vals = (x >= lo) & (x <= hi)
            h[vals] = rng.uniform(0, 1)
    elif type_idx == 4:
        center = rng.uniform(0.5, 1.5)
        width = rng.uniform(0.3, 0.9)
        h = np.exp(-0.5 * ((x - center) / width) ** 2)
        h = h / h.sum() * (n / 2)
    elif type_idx == 5:
        c1 = rng.uniform(0.3, 0.8)
        c2 = rng.uniform(1.2, 1.7)
        w1 = rng.uniform(0.2, 0.5)
        w2 = rng.uniform(0.2, 0.5)
        h = np.exp(-0.5 * ((x - c1) / w1) ** 2) + np.exp(-0.5 * ((x - c2) / w2) ** 2)
        h = h / h.sum() * (n / 2)
    elif type_idx == 6:
        m = int(rng.randint(3, 8))
        h = 0.5 + 0.35 * np.cos(np.pi * m * x) * np.sin(2 * np.pi * x / 2)
    else:
        h = rng.uniform(0, 1, n)
    
    h = np.clip(h, 0, 1)
    h = project_sum_box(h, n / 2.0)
    return h
