def lp_perturb(b, n, delta_max=0.3):
    """Solve LP to find optimal perturbation of binary solution b."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    m_total = 2 * n - 1
    
    # Build the matrix A where delta_corr[m] = A[m] @ eps
    A = np.zeros((m_total, n))
    for m in range(m_total):
        s = m - (n - 1)  # shift
        for i in range(n):
            # Term 1: (1-b[i]) * eps[i+s] (if i+s is valid)
            j = i + s
            if 0 <= j < n:
                A[m, j] += (1 - b[i])
            # Term 2: -b[i+s] * eps[i] (if i+s is valid)
            if 0 <= j < n:
                A[m, i] -= b[i+s]
    
    # Current correlation values
    corr = np.correlate(b, 1 - b, mode='full')
    
    # LP: minimize t
    # subject to: A @ eps + corr <= t (element-wise), i.e., A @ eps - t * ones <= -corr
    #             sum(eps) = 0
    #             -delta_max * b <= eps <= delta_max * (1-b)  (or just |eps| <= delta_max)
    
    # Variables: [eps_0, ..., eps_{n-1}, t]
    # n + 1 variables
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1  # coefficient of t
    
    # Inequality constraints: A @ eps <= t - corr  =>  A @ eps - t <= -corr
    A_ub = np.zeros((m_total, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1
    b_ub = -corr
    
    # Equality constraints: sum(eps) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -b[i]*delta_max <= eps[i] <= (1-b[i])*delta_max, t >= 0
    bounds = []
    for i in range(n):
        bounds.append((-delta_max * b[i], delta_max * (1 - b[i])))
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        eps = res.x[:n]
        h = b + eps
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        return h
    return b
