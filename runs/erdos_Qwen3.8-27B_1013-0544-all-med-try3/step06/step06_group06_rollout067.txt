import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    """Convert to binary: top n/2 values become 1, rest become 0."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, max_sweeps=10000):
    """Efficient binary local search with single swaps."""
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0])
    zeros = set(np.where(h_bin == 0)[0])
