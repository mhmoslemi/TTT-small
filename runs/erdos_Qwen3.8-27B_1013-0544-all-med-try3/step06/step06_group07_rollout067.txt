def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_val_grad(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    e = np.exp(shifted)
    s = e.sum()
    w = e / s
    val = m + np.log(s) / beta
    return val, w

def corr_grad(h, weights, n):
    """Compute gradient of weighted sum of correlations w.r.t. h.
    weights[k] corresponds to lag = k - (n-1) in np.correlate output.
    """
    grad = np.zeros(n)

            grad[lag:] += w * (-h[:n-lag])

            grad[:n-la] += w * (-h[la:])
