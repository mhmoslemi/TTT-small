    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            if tot < 1e-14:

    m = c.max()
    sh = alpha * (c - m)
    # sh <= 0 for all entries, sh.max() = 0
    # soft_max = m + log(sum(exp(sh))) / alpha
    log_sum = np.log(np.sum(np.exp(sh)))
    return m + log_sum / alpha
