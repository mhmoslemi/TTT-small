import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    """Compute smooth max of array c."""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    """Gradient of smooth max of correlation values."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m in range(n_lags):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=500, alpha_schedule=None, mu_schedule=None):
    """L-BFGS-B optimization with adaptive alpha."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    if alpha_schedule is None:
        alpha_schedule = [50, 200, 1000, 5000, 20000, 100000, 500000]
    if mu_schedule is None:
        mu_schedule = [1.0, 10.0, 100.0, 1000.0, 10000.0]
    
    for i, (alpha, mu) in enumerate(zip(alpha_schedule, [mu] * len(alpha_schedule) if len(mu_schedule)==1 else mu_schedule[:len(alpha_schedule)])):
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def top_k_focus(h, n, top_k=20):
    """Identify the top-k worst lags and return their indices."""
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    return idx, c[idx]

def focused_grad(h, n, alpha, top_k):
    """Gradient focusing only on top-k lags."""
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    c_shift = alpha * (c_top - c_top.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, m in enumerate(idx):
        if w[i] < 1e-14:

        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def pair_search(h, n, deadline, deltas, radius=None, max_pairs=None):
    """Local pair search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for m in range(n_lags):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def objective_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_value(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def objective_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def eq_constraint(x, n):
    return float(x.sum() - n / 2.0)

def eq_jacobian(x, n):
    return np.ones(n)

def pair_search(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            h = np.clip(h, 0.0, 1.0)
            if not improved:
                break
    return h, best_c

def multi_start_lbfgs(h, n, deadline, bounds, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng_local = np.random.RandomState(int(time.time()) % 10000)
    
    alpha_list = [30, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    iters_per_alpha = [300, 300, 250, 250, 200, 200, 150, 150, 100, 100]
    
    for start in range(n_starts):
        if time.time() > deadline - 30:
            break
        if start > 0:
            h_try = best_h + rng_local.randn(n) * 0.01
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try > best_c + 0.01:
                continue
        
        h_cur = h_try if start > 0 else best_h
        c_cur = compute_c5(h_cur, n)
        
        for i, (alpha, max_it) in enumerate(zip(alpha_list, iters_per_alpha)):
            if time.time() > deadline - 20:
                break
            mu = 100.0 * (1.0 + i * 0.5)
            try:
                res = minimize(objective_func, h_cur, jac=objective_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': max_it, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': max_it * 15,
                                       'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < c_cur - 1e-15:
                    c_cur = c_new
                    h_cur = h_new
                if c_cur < best_c - 1e-15:
                    best_c = c_cur
                    best_h = h_cur.copy()
            except Exception:
                pass
    return best_h, best_c

def slsqp_run(h, n, deadline, bounds, alpha_list=None):
    if alpha_list is None:
        alpha_list = [5000, 20000, 100000, 500000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian, 'args': (n,)}]
    for alpha in alpha_list:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(objective_func, best_h,
                          args=(n, alpha, 0.0), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
