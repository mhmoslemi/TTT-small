def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:

        target_t = n_try / 2.0
        h_t = project_feasible(resample_h(best_h, n_try), target_t)
        bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
        cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_t, n_try)}]

        for beta in [200, 2000, 20000]:
            if budget_s - (time.time() - t0) < 25:
                break
            for trial in range(2):
                if budget_s - (time.time() - t0) < 12:
                    break
                x0 = h_t.copy()
                if trial > 0:
                    x0 += np.random.randn(n_try) * 0.01
                    x0 = project_feasible(x0, target_t)
                try:
                    res = minimize(obj_smooth, x0, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 6000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, target_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new
                        best_c5 = c5
                        h_t = h_new.copy()
                except Exception:
                    pass

        # Quick gradient descent at this resolution
        if budget_s - (time.time() - t0) > 30:
            h_g = (best_h.copy() if len(best_h) == n_try
                   else project_feasible(h_t, target_t))
            beta_g = 5000
            lr_g = 0.003
            for it in range(20000):
                if budget_s - (time.time() - t0) < 15:
                    break
                grad = compute_softmax_grad(h_g, beta_g)
                lr = lr_g / (1.0 + it * 0.002)
                h_g = h_g - lr * grad
                h_g = project_feasible(h_g, target_t)
                if it % 5000 == 4999:
                    c5 = compute_c5(h_g)
                    if c5 < best_c5:
                        best_h = h_g.copy()
                        best_c5 = c5
            c5 = compute_c5(h_g)
            if c5 < best_c5:
                best_h = h_g.copy()
                best_c5 = c5

    # Phase 6: Final polish at best resolution
    if budget_s - (time.time() - t0) > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_f, n_f)}]
        for beta in [50000, 200000]:
            if budget_s - (time.time() - t0) < 12:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx_f, beta), method='SLSQP',
                               bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 15000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, target_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: L-BFGS-B with penalty as final attempt
    if budget_s - (time.time() - t0) > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        bnds_f = [(0.0, 1.0)] * n_f
        try:
            res = minimize(obj_lbfgs, best_h, args=(dx_f, target_f, 20000),
                           method='L-BFGS-B', bounds=bnds_f,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target_f)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
