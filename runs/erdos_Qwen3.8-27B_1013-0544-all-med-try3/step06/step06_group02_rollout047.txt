def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Gradient of smooth max approximation of max(c)"""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n

        k = m - (n - 1)

        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

def gradient_descent_with_sa(h, n, deadline, alpha=5000, lr_init=0.005, 
                              sa_temp=0.001, sa_cool=0.99):
    """Gradient descent with simulated annealing acceptance."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    temp = sa_temp
    rng = np.random.RandomState(42)
    
    for iteration in range(5000):
        if time.time() > deadline:
            break
        g = get_gradient(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Try multiple step sizes
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            elif c_new < best_c + temp:
                # Accept worse solution with probability exp(-(c_new - best_c)/temp)
                if rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-10)):
                    h = h_new
                    improved = True
                    break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.2
            temp *= sa_cool
        if lr < 1e-13:
            break
    
    return best_h, best_c

def smart_pair_search(h, n, deadline, steps=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        while time.time() < deadline:
            improved = False
            g = get_gradient(h, n, 10000)
            # Identify coordinates to decrease (positive gradient) and increase (negative gradient)
            g_thresh = 0.05 * (np.max(np.abs(g)) + 1e-15)
            dec_idx = list(np.where(g > g_thresh)[0])
            inc_idx = list(np.where(g < -g_thresh)[0])
            
            for i in dec_idx:
                if time.time() > deadline:
                    return h, best_c
                for j in inc_idx:
                    if i == j:
                        continue
                    move = min(step, h[i], 1.0 - h[j])
                    if move < 1e-14:
                        continue
                    h[i] -= move
                    h[j] += move
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] += move
                        h[j] -= move
            if not improved:
                break
    return h, best_c
