def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def get_gradient(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        if l == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
    return grad

def project_feasible(h, n):

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < m_val - 1e-14:
            h = h_new
            stuck_count = 0
            if alpha < 0.1:
                alpha *= 1.1
        else:
            h_restore = project_feasible(h - velocity, n)
            velocity = 0.5 * velocity
            stuck_count += 1
            if stuck_count % 50 == 0:
                alpha *= 0.7
                if alpha < 1e-7:
                    break
            h = np.clip(h - velocity, 0, 1)
            h = project_feasible(h, n)

    return best_h, best_c

def top_m_gradient_optimize(h, n, deadline, top_m=3):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.01
    vel = np.zeros(n)
    no_improve = 0

    while time.time() < deadline:
        lags = compute_all_lags(h, n)
        m_val = lags.max()

        if m_val < best_c - 1e-15:
            best_c = m_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            if top_m < 7:
                top_m += 1
            alpha *= 0.5
            no_improve = 0
            vel = np.zeros(n)
            if alpha < 1e-7:
                break

        sorted_idx = np.argsort(lags)[::-1]
        top_k = sorted_idx[:top_m]
        weights = np.exp((lags[top_k] - m_val) / 0.002)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, ki in zip(weights, top_k):
            lag = ki - (n - 1)
            grad += w * get_gradient(h, n, lag)
        grad -= grad.mean()

        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break

        vel = 0.5 * vel - alpha * grad / ng
        h_new = project_feasible(h + vel, n)
        c_new = compute_c5(h_new, n)

        if c_new < m_val - 1e-14:
            h = h_new
            if alpha < 0.05:
                alpha *= 1.05
        else:
            alpha *= 0.8
            vel = 0.3 * vel
            if alpha < 1e-8:
                break

    return best_h, best_c

def binary_local_search_smart(h_init, n, deadline, max_sweeps=500):
    h = (h_init > 0.5).astype(float).copy()
    target = n // 2
    ones_count = int(h.sum())
    if ones_count > target:
        ones_idx = list(np.where(h == 1)[0])
        for i in range(ones_count - target):
            h[ones_idx[i]] = 0
    elif ones_count < target:
        zeros_idx = list(np.where(h == 0)[0])
        for i in range(target - ones_count):
            h[zeros_idx[i]] = 1
    h = project_feasible(h, n)

    best_c = compute_c5(h, n)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        lags = compute_all_lags(h, n)
        m_val = lags.max()
        k_star = int(np.argmax(lags))
        lag_star = k_star - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = list(np.where(h == 0)[0])
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_flip = None
        best_flip_c = best_c

        if lag_star >= 0:
            candidates_i = [i for i in ones if i + lag_star < n and h[i + lag_star] == 0]
        else:
            l = -lag_star
            candidates_i = [i for i in ones if i >= l and h[i - l] == 1]

        rng = np.random.RandomState(sweep + 42)
        if len(candidates_i) > 100:
            candidates_i = list(rng.choice(candidates_i, size=100, replace=False))

        for i in candidates_i:
            if time.time() > deadline:
                break
            for j_sample in rng.choice(len(zeros), size=min(30, len(zeros)), replace=False):
                j = zeros[j_sample]
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h, n)
                if c < best_flip_c - 1e-14:
                    best_flip_c = c
                    best_flip = (i, j)
                    h[i], h[j] = 1.0, 0.0
                else:
                    h[i], h[j] = 1.0, 0.0

        if best_flip is not None:
            i, j = best_flip
            h[i], h[j] = 0.0, 1.0
            best_c = best_flip_c
            best_h = h.copy()
        else:
            break

    return best_h, best_c

def binary_full_search(h_init, n, deadline, max_sweeps=200):
    h = (h_init > 0.5).astype(float).copy()
    target = n // 2
    if int(h.sum()) != target:
        h = project_feasible(h, n)
        h = (h > 0.5).astype(float)

    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break

    return best_h, best_c

def pair_refine_fast(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for d in deltas:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            lags = compute_all_lags(h, n)
            k_star = int(np.argmax(lags))
            lag_star = k_star - (n - 1)
            grad = get_gradient(h, n, lag_star)
            grad -= grad.mean()

            if grad.max() - grad.min() < 1e-10:
                break

            i = int(np.argmin(grad))
            j = int(np.argmax(grad))
            if i == j:
                break

            h[i] -= d
            h[j] += d
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved_any = True
            else:
                h[i] += d
                h[j] -= d

    return h, best_c

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    return h

def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block = rng.randint(3, 15)
    for i in range(0, n, 2 * block):
        h[i:i + block] = 1.0
    return project_feasible(h, n)

def multires_optimize(h_start, n_target, deadline, phase_budget_s):
    phase_end = time.time() + phase_budget_s

    n = n_target
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if time.time() < phase_end - 10:
        t = min(30.0, (phase_end - time.time()) * 0.4)
        h_top1, c_top1 = top_m_gradient_optimize(h, n, time.time() + t, top_m=1)
        if c_top1 < best_c:
            best_c = c_top1
            best_h = h_top1.copy()

    if time.time() < phase_end - 10:
        t = min(40.0, (phase_end - time.time()) * 0.4)
        h_top3, c_top3 = top_m_gradient_optimize(best_h, n, time.time() + t, top_m=3)
        if c_top3 < best_c:
            best_c = c_top3
            best_h = h_top3.copy()

    if time.time() < phase_end - 5:
        h_pair, c_pair = pair_refine_fast(best_h, n, phase_end)
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair.copy()

    h_result = project_feasible(best_h, n)
    c_result = compute_c5(h_result, n)
    return h_result, c_result, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_deadline = t0 + budget_s - 15

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 200

    rng = np.random.RandomState(seed)

    n_main = 500
    best_h = None
    best_c = float('inf')

    starts = []

    if h0 is not None:
        xo = np.linspace(0, 2, n0)
        xn = np.linspace(0, 2, n_main)
        h_interp = np.interp(xn, xo, h0)
        starts.append(project_feasible(h_interp, n_main))

    starts.append(random_binary_init(n_main, seed))
    starts.append(random_binary_init(n_main, seed + 1))
    starts.append(structured_init(n_main, seed + 2))

    n_starts = len(starts)
    per_start_budget = (total_deadline - t0 - 300) / n_starts

    for si, h_start in enumerate(starts):
        if time.time() > total_deadline - 300:
            break
        phase_dl = min(t0 + 300 + si * 0 + per_start_budget * (si + 1),
                       total_deadline - 250)
        phase_dl = min(phase_dl, time.time() + per_start_budget + 20)
        h_opt, c_opt, n_opt = multires_optimize(h_start, n_main, phase_dl, per_start_budget)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    if best_h is None:
        best_h = random_binary_init(n_main, seed)
        best_c = compute_c5(best_h, n_main)

    if time.time() < total_deadline - 80:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n_main)
        h_bin, c_bin = binary_local_search_smart(h_bin, n_main, total_deadline - 80, max_sweeps=200)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    if time.time() < total_deadline - 60 and len(best_h) < 700:
        for n_high in [750, 1000]:
            if time.time() > total_deadline - 50:
                break
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_high)
            h_high = project_feasible(np.interp(xn, xo, best_h), n_high)
            phase_dl = min(time.time() + 120, total_deadline - 20)
            h_opt, c_opt, n_opt = multires_optimize(h_high, n_high, phase_dl, 120)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    if time.time() < total_deadline - 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, len(h_bin2))
        h_bin2, c_bin2 = binary_local_search_smart(h_bin2, len(h_bin2), total_deadline - 10, max_sweeps=100)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
