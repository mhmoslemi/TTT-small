            # Find all near-argmax lags and average subgradients
            max_val = corr.max()
            near_max = np.where(corr > max_val - 1e-4 * n)[0]
            grad = np.zeros(n)
            for kw in near_max:
                grad += subgrad_for_lag(h_sg, int(kw), n)
            grad /= len(near_max)

            lr = 0.0005 / (1.0 + it * 0.0005)
            h_sg = h_sg - lr * grad
            h_sg = project_feasible(h_sg, target_sum)

            if it % 2000 == 1999:
                c5 = compute_c5(h_sg)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sg.copy()

        c5 = compute_c5(h_sg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg.copy()

    # Phase 4: SLSQP polish at very high beta from the best point
    if tl() > 20:
        n = best_n
        target_sum = n / 2.0
        dx = 2.0 / n

        for beta in [50000, 200000]:
            if tl() < 10:
                break
            h_new = slsqp_optimize(best_h, n, target_sum, dx, beta, maxiter=5000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Try a nearby resolution with a quick optimization
    if tl() > 40:
        for n_try in [best_n + 20, best_n - 20, best_n + 40]:
            if n_try < 40 or tl() < 20:
                continue
            target_sum_t = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, target_sum_t)

            for beta in [1000, 10000]:
                if tl() < 10:
                    break
                h_new = slsqp_optimize(h_t, n_try, target_sum_t, dx_t, beta, maxiter=2000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    best_n = n_try
                    break

    # Phase 6: Final Nelder-Mead with penalty
    if tl() > 20:
        n = best_n
        target_sum = n / 2.0
        dx = 2.0 / n

        def obj_nm_pen(hv, dxv, betav, tsv):
            hv_c = np.clip(hv, 0, 1)
            pen = 100.0 * (hv_c.sum() - tsv) ** 2
            corr = np.correlate(hv_c, 1.0 - hv_c, mode='full')
            return smooth_max(corr, betav) * dxv + pen

        try:
            res = minimize(obj_nm_pen, best_h, args=(dx, 20000, target_sum),
                           method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14,
                                    'adaptive': True})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
