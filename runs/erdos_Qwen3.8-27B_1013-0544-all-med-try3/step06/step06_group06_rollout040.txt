```python
    """Compute all correlation values g(k) = sum_i h[i]*(1-h[i+k]) for the 'full' mode."""
    return np.correlate(h, 1-h, mode='full') * (2.0/n)

    """Compute C5 = max_k g(k)."""
    return float(np.max(compute_g(h, n)))

def exact_subgradient(h, n, k_offset):
    """
    Compute exact subgradient of g(k_offset) with respect to h.
    k_offset is the index in the 'full' correlation output (0 to 2n-2).
    The lag is lag = k_offset - (n-1).
    g(k) = sum_j h[j] * (1 - h[j+lag]) for valid j.
    ∂g/∂h[i] = (1-h[i+lag]) - h[i-lag] (with out-of-range terms = 0)
    """
    lag = k_offset - (n-1)
    grad = np.zeros(n)
    # First term: (1-h[i+lag]) for valid i+lag
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        valid_i = slice(0, n-lag)
        grad[valid_i] += (1 - h[lag:])
    else:
        # i ranges from -lag to n-1
        valid_i = slice(-lag, n)
        grad[valid_i] += (1 - h[:n+lag])
    
    # Second term: -h[i-lag] for valid i-lag
    if lag >= 0:
        # i-lag ranges from 0 to n-1, so i ranges from lag to n-1
        valid_i = slice(lag, n)
        grad[valid_i] -= h[:n-lag]
    else:
        # i-lag = i+|lag| ranges from 0 to n-1, so i ranges from 0 to n-1-|lag|
        valid_i = slice(0, n+lag)
        grad[valid_i] -= h[-lag:]
    
    return grad

    dx = 2.0 / n
```
```python
    return np.correlate(h, 1 - h, mode='full') * dx
```

```python
    return float(np.max(compute_g(h, n)))

def exact_subgradient(h, n, pos):
    k = pos - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        if n > k:
            grad[:n-k] -= h[k:]
    else:
        nk = -k
        grad[:n-k] += (1.0 - h[nk:])
        grad[nk:] -= h[:n-k]
    return grad

def project_feasible(h, n):
    target = n / 2.0

            s = room.sum()

            if s < 1e-15:
```

```python
def multi_lag_subgradient(h, n, top_m=5, alpha=100):
    g = compute_g(h, n)
    # Get top M lags
    top_indices = np.argsort(g)[::-1][:top_m]
    top_g = g[top_indices]
    # Softmax weights
    w = np.exp(alpha * (top_g - top_g.max()))
    w /= w.sum()
    # Weighted subgradient
    grad = np.zeros(n)
    for idx, wt in zip(top_indices, w):
        grad += wt * exact_subgradient(h, n, idx)
    return grad, g[top_indices].max()
```

This is O(M * n) per iteration instead of O(n), but M is small (5-10). The benefit is a smoother descent direction.
