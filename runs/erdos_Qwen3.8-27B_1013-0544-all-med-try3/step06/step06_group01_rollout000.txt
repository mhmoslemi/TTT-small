def efficient_binary_search(h_init, n, deadline, max_sweeps=1000, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)  # lags k
    
    best_c = np.inf
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            # Perturb
            for _ in range(rng.randint(5, 15)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 1-h[a], 1-h[b]
        
        corr = np.correlate(h, 1-h, mode='full').copy()
        cur_c = corr.max() * dx
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

            for i in zeros:

                ik = i + m_range  # positions i+k for each lag k
                jk_candidates = None  # we'll compute per j
                
                # For lag k (indexed as k_idx in 0..2n-2, actual lag = k_idx - (n-1)):

                pos_i_plus = i + m_range  # i + k for each k
                pos_i_minus = i - m_range  # i - k for each k
                
                A = np.zeros(2*n-1)
                m1 = (pos_i_plus >= 0) & (pos_i_plus < n)
                A[m1] += 1.0 - h[pos_i_plus[m1]]
                m2 = (pos_i_minus >= 0) & (pos_i_minus < n)
                A[m2] += h[pos_i_minus[m2]]
                # Note: for positions where both m1 and m2 are true, we add both
                
                # For each j in ones:
                # B_j[k] = 1 - h[j+k] + h[j-k]
                # We want to check if max(corr + A - B_j) < cur_c for any j

                # pos_j_plus[j, k] = j + m_range[k]
                # pos_j_minus[j, k] = j - m_range[k]

                m2j = (j_minus >= 0) & (j_minus < n)
                B[m2j] += h[j_minus[m2j]]
                
                # new_corr[j, k] = corr[k] + A[k] - B[j, k]
                new_max = (corr[None, :] + A[None, :] - B).max(axis=1) * dx
                best_j_idx = new_max.argmin()
                
                if new_max[best_j_idx] < cur_c - 1e-15:

                    h[i], h[j] = 1.0, 0.0
                    corr = corr + A - B[best_j_idx]
                    cur_c = new_max[best_j_idx]

                    break
            
            if not improved:

            best_c = cur_c
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary by taking the top n/2 values."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def vec_binary_search(h_start, n, deadline, max_sweeps=10000, seed=42):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    m_range = np.arange(-(n-1), n)
    
    h = h_start.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    # Padded h for fast access: h_ext[n + x] = h[x] if 0<=x<n else 0
    h_ext = np.zeros(3 * n)
    h_ext[n:2*n] = h
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute j_plus, j_minus for all ones
        # j_plus[j, k] = ones[j] + m_range[k]
        # h[ones[j] + m_range[k]] = h_ext[n + ones[j] + m_range[k]]
        j_plus_idx = n + ones[:, None] + m_range[None, :]  # (n_ones, n_corr)
        j_minus_idx = n + ones[:, None] - m_range[None, :]  # (n_ones, n_corr)
        
        # Clip to valid range of h_ext
        j_plus_idx = np.clip(j_plus_idx, 0, 3*n-1)
        j_minus_idx = np.clip(j_minus_idx, 0, 3*n-1)
        
        # h values at these positions
        h_j_plus = h_ext[j_plus_idx]  # (n_ones, n_corr)
        h_j_minus = h_ext[j_minus_idx]  # (n_ones, n_corr)
        
        # B[j, k] = (1 - h[j+k]) + h[j-k]
        B = (1.0 - h_j_plus) + h_j_minus  # (n_ones, n_corr)
        
        for i in zeros:
            pos_i_plus = i + m_range
            pos_i_minus = i - m_range
            h_i_plus = h_ext[np.clip(n + pos_i_plus, 0, 3*n-1)]
            h_i_minus = h_ext[np.clip(n + pos_i_minus, 0, 3*n-1)]
            
            A = (1.0 - h_i_plus) + h_i_minus  # (n_corr,)
            base = corr + A  # (n_corr,)
            
            # new_max[j] = max_k (base[k] - B[j, k])
            new_corr = base[None, :] - B  # (n_ones, n_corr)
            new_max = new_corr.max(axis=1) * dx  # (n_ones,)
            
            best_j_idx = new_max.argmin()

                h[i] = 1.0

                h_ext[n+i] = 1.0

                cur_c = corr.max() * dx

                break
        
        if not improved:
            break
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    
    return best_h, best_c
