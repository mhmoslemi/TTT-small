So per pair evaluation:
- Compute Δc: O(n)
- Update h: O(1)
- Compute new max: O(n)
- Total: O(n)

For n=300 and n² pairs: 300 * 90000 = 27M operations. At 1ns/op: 27ms per full pass. That's fast!

With 7 deltas: 189ms per full set. Very fast!

def pair_search_vectorized(h, n, deadline, deltas):
    dx = 2.0 / n
    best_c = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    lags = np.arange(2 * n - 1) - (n - 1)  # lags[m] = m - (n-1)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Vectorized computation of Δc for all m
                # Δc[m] = d * [(1-h[i+k_m])*1{i+k_m<n} - h[i-k_m]*1{i-k_m>=0}
                #             - (1-h[j+k_m])*1{j+k_m<n} + h[j-k_m]*1{j-k_m>=0}]
                
                # Precompute for position i
                # term1_i[m] = (1-h[i+k_m]) * 1{i+k_m < n}
                # term2_i[m] = h[i-k_m] * 1{i-k_m >= 0}
                
                # For position i, k_m ranges from -(n-1) to (n-1)
                # i + k_m is in [i-(n-1), i+(n-1)]
                # We need i + k_m in [0, n-1], so k_m in [-i, n-1-i]
                # And i - k_m in [0, n-1], so k_m in [i-(n-1), i]
                
                # This is getting complex. Let me just use np.correlate for simplicity.
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c_new = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            c_all = np.correlate(h, 1 - h, mode='full') * dx
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c_new = float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            c_all = np.correlate(h, 1 - h, mode='full') * dx
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

For n=300, each np.correlate call is about 0.1ms. Per pair: 2 calls (for two directions) = 0.2ms. Per full pass (n² pairs): 90000 * 0.2ms = 18 seconds. With 7 deltas: 126 seconds per full set. That's a lot.
