def slsqp_optimize(h, n, deadline, alpha=100000, maxiter=500):
    """SLSQP with log-sum-exp smoothing of the max."""
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        if alpha > 1e6:
            return m
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
