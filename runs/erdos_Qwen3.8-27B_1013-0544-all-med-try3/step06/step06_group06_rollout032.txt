import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return correlate(h, 1 - h, mode='full') * dx

def soft_max_value(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    return m + np.log(np.exp(sh).sum()) / alpha

def soft_max_weights(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    e = np.exp(sh)
    return e / e.sum()

    Compute gradient of soft_max(g, alpha) w.r.t. h.
    p: softmax weights, length 2n-1, p[k] corresponds to lag s = k-(n-1)

```python
def compute_gradient_direct(h, n, p):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    
    for k in range(n_corr):
        s = k - (n - 1)
        pk = p[k]
        if abs(pk) < 1e-18:
            continue
        # term1: p[k] * (1-h[i+s]) for i where 0 <= i <= n-1 and 0 <= i+s <= n-1
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += pk * (1.0 - h[i_lo+s:i_hi+s+1])
        # term2: -p[k] * h[i-s] for i where 0 <= i <= n-1 and 0 <= i-s <= n-1
        i_lo2 = max(0, s)
        i_hi2 = min(n-1, n-1+s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= pk * h[i_lo2-s:i_hi2-s+1]
    
    return grad * dx
```
