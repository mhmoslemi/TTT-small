def adam_optimize(h, n, deadline, alpha=1000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    while time.time() < deadline:
        t += 1
        g = grad_smooth_max(h, n, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Try the step
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h + step, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h + factor * step, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
                    break
            else:
                # No improvement, reduce learning rate
                lr *= 0.5
                if lr < 1e-15:
                    break
    
    return h, best_c
```

Wait, that's not quite right either. The projection onto the intersection of the box [0,1]^n and the hyperplane sum = target is NOT simply h + lambda. It's the Euclidean projection, which has the form:

P(x) = clip(x + lambda, 0, 1) for the right lambda.

This is the standard algorithm for projecting onto the "feasible set" of a resource allocation problem. The bisection finds the right lambda.

Let me verify: if x is already feasible (0 ≤ x[i] ≤ 1 and sum = target), then lambda = 0 and P(x) = x. ✓

If x has sum > target (too much mass), we need to reduce it. The projection will decrease some coordinates (those that can be decreased without hitting 0) by equal amounts, until the sum equals target. This is exactly what clip(x + lambda, 0, 1) does for the right lambda < 0.

If x has sum < target, we increase similarly.

The bisection is:
- f(lambda) = sum(clip(x + lambda, 0, 1)) - target
- f is continuous and non-decreasing in lambda
- We want f(lambda) = 0
- f(-x.max()) = 0 - target < 0 (all clipped to 0)
- f(1 - x.min()) = n - target > 0 (all clipped to 1)

Since f is non-decreasing, we can use:

- if f(mid) < 0: hi = mid (need to increase lambda)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on lambda
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        ak = abs(k)
        nv = n - ak
        if k >= 0:
            grad[0:nv] += wm * (1.0 - h[ak:n])
            grad[ak:n] += wm * (-h[0:nv])
        else:
            grad[ak:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[ak:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    return grad_smooth_max(h, n, alpha)

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha)
        def grad(x):
            return smooth_max_grad(x, n, alpha)
        def constraint(x):
            return x.sum() - n / 2.0
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def momentum_gd(h, n, deadline, alpha=1000, lr_init=0.05, beta=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stale = 0
    
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        v = beta * v + g
        step = -lr * v
        h_new = project_feasible(h + step, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stale = 0
        else:
            h = h_new  # still move, but don't update best
            stale += 1
            if stale % 10 == 0:
                lr *= 0.8
                beta = min(beta + 0.01, 0.99)
            if lr < 1e-12:
                break
    return best_h, best_c

def coordinate_refine(h, n, deadline, steps=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]

    alpha = 5000
    g = grad_smooth_max(h, n, alpha)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = -h.max() - 0.01
    hi = 1.0 - h.min() + 0.01
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            # First factor: h[i] as h[n+k], i in [k, n-1], contrib (1-h[i-k])
            grad[k:n] += wm * (1.0 - h[0:n-k])
            # Second factor: h[i] in (1-h[n]), i in [0, n-1-k], contrib -h[i+k]
            grad[0:n-k] += wm * (-h[k:n])
        else:
            kk = -k
            # First factor: h[i] as h[n-kk], i in [0, n-1-kk], contrib (1-h[i+kk])
            grad[0:n-kk] += wm * (1.0 - h[kk:n])
            # Second factor: h[i] in (1-h[n]), i in [kk, n-1], contrib -h[i-kk]
            grad[kk:n] += wm * (-h[0:n-kk])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        def obj(x):
            return smooth_max_val(x, n, alpha)
        def gr(x):
            return smooth_max_grad(x, n, alpha)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def momentum_gd(h, n, deadline, alpha=2000, lr_init=0.05, beta=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        v = beta * v + g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.7
                if lr < 1e-12:
                    break
    return best_h, best_c

def greedy_coord_search(h, n, deadline, top_k=5, deltas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    alpha = 5000
    
    for _ in range(50):  # max passes
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Determine direction
            if g[i] > 0:
                # Want to decrease h[i]
                direction = -1
            elif g[i] < 0:
                direction = 1
            else:
                continue
            
            # Find top-k candidates for j
            if direction == -1:
                # h[i] decreases, so we need h[j] to increase
                # Best j: most negative gradient (increasing h[j] helps most)
                scores = g.copy()
                scores[i] = np.inf
                scores[h >= 1.0] = np.inf  # can't increase these
                top_indices = np.argsort(scores)[:top_k]
            else:
                # h[i] increases, so we need h[j] to decrease
                scores = -g.copy()
                scores[i] = -np.inf
                scores[h <= 0.0] = -np.inf  # can't decrease these
                top_indices = np.argsort(scores)[:top_k]
            
            for j in top_indices:
                if j == i or j < 0:
                    continue
                for d in deltas:
                    di = direction * d
                    if h[i] + di < 0 or h[i] + di > 1:
                        continue
                    if h[j] - di < 0 or h[j] - di > 1:
                        continue
                    h[i] += di
                    h[j] -= di
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        g = smooth_max_grad(h, n, alpha)
                        break
                    else:
                        h[i] -= di
                        h[j] += di
                if c_new < best_c:  # if we improved, break j loop
                    break
        
        if not improved:
            break
    return h, best_c
