```python
        mask = (idx >= 0) & (idx < len(h))
        out[mask] = h[idx[mask]]

        # Subtract quadratic terms
        if 0 <= cross_idx1 < 2*n-1:

        if 0 <= cross_idx2 < 2*n-1:
```

```python
def coordinate_descent(h, n, alpha, n_sweeps, deadline, rng):
    """Coordinate descent with log-sum-exp smoothing."""
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    m_vals = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        # Random permutation of indices for pair selection
        perm = rng.permutation(n)
        # Pair up: (perm[0], perm[1]), (perm[2], perm[3]), etc.
        for p in range(n // 2):
            i = perm[2*p]
            j = perm[2*p+1]
            if h[i] <= 0 or h[j] >= 1:
                continue  # Skip if no room to move
            
            # Compute A
            A = safe_index(h, j + m_vals) + safe_index(h, j - m_vals) - safe_index(h, i + m_vals) - safe_index(h, i - m_vals)
            
            # Cross terms
            cross1 = (j - i) + (n - 1)
            cross2 = (i - j) + (n - 1)
            
            # Feasible range
            dx_lo = max(-h[i], h[j] - 1.0) + 1e-12
            dx_hi = min(1.0 - h[i], h[j]) - 1e-12
            if dx_hi - dx_lo < 1e-12:
                continue
            
            # Bisection on f'(δ) = 0
            lo, hi = dx_lo, dx_hi
            for _ in range(40):
                mid = (lo + hi) / 2
                O_delta = corr + A * mid
                if 0 <= cross1 < 2*n-1:
                    O_delta[cross1] -= mid * mid
                if 0 <= cross2 < 2*n-1:
                    O_delta[cross2] -= mid * mid
                m_val = O_delta.max()
                shifted = alpha * (O_delta - m_val)
                sm = shifted.max()
                p_w = np.exp(shifted - sm)
                p_sum = p_w.sum()
                if p_sum < 1e-30:
                    break
                p_w /= p_sum
                dO = A.copy()
                if 0 <= cross1 < 2*n-1:
                    dO[cross1] -= 2.0 * mid
                if 0 <= cross2 < 2*n-1:
                    dO[cross2] -= 2.0 * mid
                fp = p_w @ dO
                if fp > 0:
                    hi = mid
                else:
                    lo = mid
            
            delta = (lo + hi) / 2
            if abs(delta) > 1e-12:
                h[i] += delta
                h[j] -= delta
                corr += A * delta
                if 0 <= cross1 < 2*n-1:
                    corr[cross1] -= delta * delta
                if 0 <= cross2 < 2*n-1:
                    corr[cross2] -= delta * delta
    
    return h, corr
```

```python
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def coordinate_descent(h, n, alpha, n_sweeps, deadline, rng):
    """Fast coordinate descent with log-sum-exp smoothing."""
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_vals = np.arange(-(n - 1), n)
    two_n_minus_1 = 2 * n - 1
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        perm = rng.permutation(n)
        for p in range(n // 2):
            i = perm[2 * p]
            j = perm[2 * p + 1]
            hi_val = h[i]
            hj_val = h[j]
            
            # Feasible range for delta
            dx_lo = max(-hi_val, hj_val - 1.0)
            dx_hi = min(1.0 - hi_val, hj_val)
            if dx_hi - dx_lo < 1e-14:
                continue
            
            # Compute A = h[j+m] + h[j-m] - h[i+m] - h[i-m]
            A = (safe_index(h, j + m_vals) + safe_index(h, j - m_vals) - 
                 safe_index(h, i + m_vals) - safe_index(h, i - m_vals))
            
            # Cross-term indices
            cross1 = (j - i) + (n - 1)
            cross2 = (i - j) + (n - 1)
            has_cross1 = 0 <= cross1 < two_n_minus_1
            has_cross2 = 0 <= cross2 < two_n_minus_1 and cross2 != cross1
            
            # Bisection on f'(delta) = 0
            lo = dx_lo + 1e-14
            hi = dx_hi - 1e-14
            delta = 0.0
            for _ in range(35):
                mid = (lo + hi) * 0.5
                O_delta = corr + A * mid
                if has_cross1:
                    O_delta[cross1] -= mid * mid
                if has_cross2:
                    O_delta[cross2] -= mid * mid
                m_val = O_delta.max()
                shifted = alpha * (O_delta - m_val)
                sm = shifted.max()
                p_w = np.exp(shifted - sm)
                p_sum = p_w.sum()
                if p_sum < 1e-300:
                    delta = mid
                    break
                p_w /= p_sum
                dO = A.copy()
                if has_cross1:
                    dO[cross1] -= 2.0 * mid
                if has_cross2:
                    dO[cross2] -= 2.0 * mid
                fp = p_w @ dO
                if fp > 0:
                    hi = mid
                else:
                    lo = mid
                delta = (lo + hi) * 0.5
            
            if abs(delta) > 1e-12:
                h[i] += delta
                h[j] -= delta
                corr += A * delta
                if has_cross1:
                    corr[cross1] -= delta * delta
                if has_cross2:
                    corr[cross2] -= delta * delta
    
    # Final projection to fix any numerical drift
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        p_w = np.exp(shifted - sm)
        p_w /= p_w.sum()
        # grad of (m + logsum/alpha) w.r.t. x
        # = (1/alpha) * sum_k p_k * grad c_k
        # grad c_k w.r.t. x[i] = (1 - x[i+k]) - x[i-k] (with boundary handling)
        # Actually let's just use the subgradient of the smoothed function
        # The gradient of c_k = sum_i x[i](1-x[i+k]) w.r.t. x[i] is:
        # (1 - x[i+k]) - x[i-k] (summing over valid k)
        # This is getting complex, let's skip the gradient and let SLSQP approximate it
        return None
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
