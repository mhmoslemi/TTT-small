```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def subgradient_method(h, n, deadline, max_iters=10000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    alpha = 0.02
    stagnation = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = compute_lag_gradient(h_curr, n, lag)
        
        # Step
        h_new = h_curr - alpha * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            stagnation = 0
            # Slightly increase alpha
            alpha = min(alpha * 1.01, 0.05)
        else:
            h_curr = project_feasible(h_curr, n)  # stay
            stagnation += 1
            if stagnation > 50:
                alpha *= 0.9
                if alpha < 0.001:
                    break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
```

This means the outer approximation approach is NOT valid for bilinear functions! The previous code's approach of adding linearization constraints is only a heuristic, not a valid relaxation.

So the binary search + LP approach won't give a valid certificate. It can still be used as a heuristic (find a feasible point, then verify), but it won't guarantee optimality.

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def gradient_descent(h, n, deadline, alpha_start=5000, alpha_end=50000, max_steps=20000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    alpha = alpha_start
    step = 0.005
    patience = 0
    n_lags = 2 * n - 1
    
    for it in range(max_steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        temp = alpha * (c - c_max)
        temp -= temp.max()
        p = np.exp(temp)
        p /= p.sum()
        
        g = np.zeros(n)
        for k in range(n_lags):
            if p[k] < 1e-12:
                continue
            lag = k - (n - 1)
            g += p[k] * compute_lag_gradient(h_curr, n, lag)
        
        h_new = h_curr - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            patience = 0
            step = min(step * 1.02, 0.02)
            alpha = min(alpha * 1.01, alpha_end)
        else:
            patience += 1
            if patience > 200:
                step *= 0.8
                if step < 1e-5:
                    break
                patience = 0
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_set_optimize(h, n, deadline, top_m=10, max_rounds=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        for alpha in [10000, 50000, 200000]:
            if time.time() > deadline - 5:
                break
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
            try:
                res = minimize(active_set_smooth_obj, h_curr,
                              args=(n, top_idx, alpha),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_curr = h_new
            except Exception:
                pass
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        stalling = 0
        while stalling < 500:
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            g = compute_lag_gradient(h, n, lag)
            best_gain = 0.0
            best_i = -1
            if lag > 0:
                for i in range(n - m):
                    j = i + m
                    gain = g[i] - g[j]
                    if gain > best_gain:
                        best_gain = gain
                        best_i = i
            else:
                for i in range(n - m):
                    j = i + m
                    gain = g[j] - g[i]
                    if gain > best_gain:
                        best_gain = gain
                        best_i = i
            if best_i < 0 or best_gain < 1e-12:
                break
            i = best_i
            j = i + m
            h_save = h.copy()
            if lag > 0:
                delta = min(d, h[j] * 0.5, 1.0 - h[i])
                if delta < 1e-12:
                    break
                h[i] += delta
                h[j] -= delta
            else:
                delta = min(d, h[i] * 0.5, 1.0 - h[j])
                if delta < 1e-12:
                    break
                h[i] -= delta
                h[j] += delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                stalling = 0
            else:
                h[:] = h_save
                stalling += 1
    return h, best_c

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.85)
    return best_h, best_c

    best_h = h.copy()

        if c_gd < best_c:
```
