def project_feasible(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    mid = 0.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - mid, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - result
            tot = room.sum()
            if tot > 1e-12:
                result += (diff / tot) * room
        else:
            tot = result.sum()
            if tot > 1e-12:
                result += (diff / tot) * result
    return np.clip(result, 0.0, 1.0)

def grad_lag(h, n, lag):
    grad = np.zeros(n)
    if 0 <= lag < n:
        Np = n - lag
        grad[:Np] += (1.0 - h[lag:lag+Np])
        grad[lag:n] -= h[:Np]
    elif -n < lag < 0:
        L = -lag
        Np = n - L
        grad[L:n] += (1.0 - h[:Np])
        grad[:Np] -= h[L:L+Np]
    return grad

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
