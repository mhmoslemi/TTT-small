import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    w = np.exp(c_shift)
    w_sum = w.sum()
    w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    mx = sh.max()
    if mx > 500:
        val = c_max + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = c_max + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    mx = sh.max()
    if mx > 500:
        val = c_max + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = c_max + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [100, 500, 1000, 5000, 20000, 100000, 500000]

                          args=(n, alphas[i], mus[i]), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 60})
            h_new = project_feasible(res.x, n)

            if c_new < best_c - 1e-15:

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: L-BFGS-B with increasing alpha from parent
    t1 = min(t0 + 180, deadline - 600)
    best_h, best_c = lbfgs_optimize(best_h, n0, t1, max_iters=400)

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 500:
        t2 = min(time.time() + 80, deadline - 500)
        for a in [200000, 1000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, n0, min(time.time() + 40, t2), alpha=a, maxiter=1500)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Subgradient descent
    if time.time() < deadline - 400:
        t3 = min(time.time() + 60, deadline - 400)
        h3, c3 = subgradient_descent(best_h, n0, t3, lr_init=0.003, alpha_init=200)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Top-k descent (focus on active correlations)
    if time.time() < deadline - 350:
        t4 = min(time.time() + 50, deadline - 350)
        h4, c4 = topk_descent(best_h, n0, t4, k_top=3, lr_init=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Perturbation restarts (escape local optima)
    if time.time() < deadline - 250:
        t5 = min(time.time() + 80, deadline - 250)
        h5, c5 = perturbation_restart(best_h, n0, t5, n_restarts=6, perturbation_scale=0.03)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Multi-resolution search
    if time.time() < deadline - 200:
        best_h, best_c = multi_resolution_search(best_h, best_c, deadline - 80)

    # Phase 7: L-BFGS polish on best
    if time.time() < deadline - 80:
        nf = len(best_h)
        t7 = min(time.time() + 60, deadline - 40)
        best_h, best_c = lbfgs_optimize(best_h, nf, t7, max_iters=300)

    # Phase 8: Subgradient + topk polish
    if time.time() < deadline - 50:
        nf = len(best_h)
        t8a = min(time.time() + 25, deadline - 30)
        h8, c8 = subgradient_descent(best_h, nf, t8a, lr_init=0.002, alpha_init=500)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        if time.time() < deadline - 20:
            t8b = min(time.time() + 20, deadline - 15)
            h8b, c8b = topk_descent(best_h, nf, t8b, k_top=3, lr_init=0.001)
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    # Phase 9: Coordinate descent
    if time.time() < deadline - 30:
        nf = len(best_h)
        t9 = min(time.time() + 25, deadline - 10)
        h9, c9 = coordinate_descent(best_h, nf, t9, max_sweeps=30)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final pair search
    if time.time() < deadline - 3:
        nf = len(best_h)
        t10 = deadline - 2
        best_h, best_c = pair_search(best_h, nf, t10,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=50)

    # Final check
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    final_c = compute_c5(best_h, nf)
    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, nf
