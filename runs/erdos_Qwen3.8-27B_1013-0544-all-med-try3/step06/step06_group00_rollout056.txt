```python
def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * c * dx
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    p = exp_ac / exp_ac.sum()
    obj = ac_max / alpha + np.log(exp_ac.sum()) / alpha
    # Gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        if abs(p[n-1+d]) < 1e-15:
            continue
        pd = p[n-1+d]
        if d == 0:
            grad += pd * (1.0 - 2.0 * h)
        elif d > 0:
            # (1-h[j-d]) * [d≤j≤n-1] - h[j+d] * [0≤j≤n-1-d]
            j1 = np.arange(d, n)
            grad[j1] += pd * (1.0 - h[j1 - d])
            j2 = np.arange(0, n-d)
            grad[j2] -= pd * h[j2 + d]
        else:
            d_abs = -d
            # (1-h[j+d_abs]) * [0≤j≤n-1-d_abs] - h[j-d_abs] * [d_abs≤j≤n-1]
            j1 = np.arange(0, n-d_abs)
            grad[j1] += pd * (1.0 - h[j1 + d_abs])
            j2 = np.arange(d_abs, n)
            grad[j2] -= pd * h[j2 - d_abs]
    grad *= dx
    return obj, grad
```

```python
def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            s = room.sum()

            if s < 1e-14:
```
