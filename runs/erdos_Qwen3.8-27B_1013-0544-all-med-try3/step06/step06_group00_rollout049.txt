```python
def compute_grad(h, w, n):
    """Compute gradient of smooth max objective.
    w: softmax weights of length 2n-1
    Returns gradient of length n.
    """
    m = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    i_idx = np.arange(n)  # shape (n,)
    
    # Forward: A[i] = sum_k w[k] * h[i+m[k]] (valid indices only)
    idx_f = i_idx[:, None] + m[None, :]  # shape (n, 2n-1)
    mask_f = (idx_f >= 0) & (idx_f < n)
    h_f = np.where(mask_f, h[np.clip(idx_f, 0, n-1)], 0.0)
    A = h_f @ w  # shape (n,)
    
    # Backward: B[i] = sum_k w[k] * h[i-m[k]] (valid indices only)
    idx_b = i_idx[:, None] - m[None, :]
    mask_b = (idx_b >= 0) & (idx_b < n)
    h_b = np.where(mask_b, h[np.clip(idx_b, 0, n-1)], 0.0)
    B = h_b @ w
    
    return 1.0 - A - B
```

{d : sum(d) = 0, d[i] ≠ 0 only where 0 < h[i] < 1}

So the projected gradient is:

This is the Euclidean projection of grad onto the tangent space.

```python
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def smooth_obj_and_grad(x, n, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = corr.max()
    shifted = np.clip(alpha * (corr - M), -700, 700)

    w = np.exp(shifted - log_sum)
    m = np.arange(2*n-1) - (n-1)
    i_idx = np.arange(n)
    idx_f = i_idx[:, None] + m[None, :]

    dx = 2.0 / n
    h = project_feasible(h, n)
    def fun(x):
        x_p = project_feasible(x, n)
        val, _ = smooth_obj_and_grad(x_p, n, alpha, dx)
        return val

        x_p = project_feasible(x, n)
        _, g = smooth_obj_and_grad(x_p, n, alpha, dx)
        return g

    try:

                                'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta(h, n, i, j, m_range, corr):
    """Compute change in correlation when swapping h[i] and h[j] (i was 0, j was 1)."""
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

        j = int(ones_arr[bi])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_and_grad(x, n, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = corr.max()
    shifted = np.clip(alpha * (corr - M), -700, 700)
    log_sum = np.log(np.exp(shifted).sum())
    obj = (M + log_sum / alpha) * dx
    w = np.exp(shifted - log_sum)
    m = np.arange(2 * n - 1) - (n - 1)
    i_idx = np.arange(n)
    idx_f = i_idx[:, None] + m[None, :]
    mask_f = (idx_f >= 0) & (idx_f < n)
    h_f = np.where(mask_f, x[np.clip(idx_f, 0, n - 1)], 0.0)
    A = h_f @ w
    idx_b = i_idx[:, None] - m[None, :]
    mask_b = (idx_b >= 0) & (idx_b < n)
    h_b = np.where(mask_b, x[np.clip(idx_b, 0, n - 1)], 0.0)
    B = h_b @ w
    raw_grad = (1.0 - A - B) * dx
    free = (x > 1e-10) & (x < 1.0 - 1e-10)
    if free.sum() > 0:
        raw_grad[free] -= raw_grad[free].mean()
    return obj, raw_grad

def lbfgsb_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    h = project_feasible(h, n)

    def fun(x):
        x_p = project_feasible(x, n)
        val, _ = smooth_obj_and_grad(x_p, n, alpha, dx)
        return val

    def jac(x):
        x_p = project_feasible(x, n)
        _, g = smooth_obj_and_grad(x_p, n, alpha, dx)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 5})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_delta_vec(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(n_zeros):
            if time.time() > deadline:
                break
            i = int(zeros_arr[zi])
            best_j = -1
            best_nc = cur_c
            for oi in range(n_ones):
                j = int(ones_arr[oi])
                delta = compute_delta_vec(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j_idx = int(np.where(ones_arr == best_j)[0][0])
                h[i] = 1.0
                h[best_j] = 0.0
                delta = compute_delta_vec(h, n, i, best_j, m_range)
                # Recompute delta since h changed
                h[i] = 0.0
                h[best_j] = 1.0
                delta = compute_delta_vec(h, n, i, best_j, m_range)
                h[i] = 1.0
                h[best_j] = 0.0
                corr += delta
                cur_c = best_nc
                zeros_arr[zi] = best_j
                ones_arr[j_idx] = i
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def binary_anneal(h_init, n, deadline, seed=42, max_iters=5000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    T = 0.02
    T_min = 1e-9
    cool = (T_min / T) ** (1.0 / max(max_iters, 1))
    scale = 1.0 / dx
    stuck_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 50000 == 0 and it > 0 and T > T_min:
            if stuck_count > 100000:
                T = max(T * 3.0, 0.005)
                stuck_count = 0
        ai = rng.randint(n_zeros)
        bi = rng.randint(n_ones)
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = compute_delta_vec(h, n, i, j, m_range)
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc <= 0 or (T > T_min and rng.random() < np.exp(-scale * dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            stuck_count += 1
        T *= cool
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
```
