from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, eta, alpha=1.0):
    """One subgradient step: reduce the worst correlation"""
    c = compute_correlations(h, n)
    k_worst = np.argmax(c)
    k_lag = k_worst - (n - 1)
    
    # Gradient of the worst correlation
    # d/dc[k_lag]/dh[j] = dx * (1-h[j+k_lag]) - dx * h[j-k_lag]
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Term 1: dx * (1 - h[j+k_lag]) for valid j+k_lag
    if k_lag >= 0:
        # j ranges from 0 to n-1-k_lag
        valid_j = slice(0, n - k_lag)
        shifted = slice(k_lag, n)
        grad[valid_j] += dx * (1.0 - h[shifted])
    else:
        # j ranges from -k_lag to n-1
        valid_j = slice(-k_lag, n)
        shifted = slice(0, n + k_lag)
        grad[valid_j] += dx * (1.0 - h[shifted])
    
    # Term 2: -dx * h[j-k_lag] for valid j-k_lag
    if k_lag >= 0:
        # j-k_lag >= 0 => j >= k_lag
        valid_j = slice(k_lag, n)
        shifted = slice(0, n - k_lag)
        grad[valid_j] -= dx * h[shifted]
    else:
        # j-k_lag = j+|k_lag| < n => j < n-|k_lag|
        valid_j = slice(0, n + k_lag)
        shifted = slice(-k_lag, n)
        grad[valid_j] -= dx * h[shifted]
    
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new, k_worst

    best_c = compute_c5(h, n)
    best_h = h.copy()
    eta = initial_eta
    no_improve = 0
    
    while time.time() < deadline:
        h_new, k_worst = subgradient_step(h, n, eta)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            no_improve += 1

                eta *= 0.5
                no_improve = 0
            elif no_improve > 100:
                h = h_new  # Accept even if not better (exploration)
        
        # Gradually reduce eta
        eta = max(0.0001, eta * decay)
    
    return best_h, best_c

def smooth_logsumexp(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

This is equivalent to maximizing ∑_i h[i]h[i+k*] (since ∑h[i] is fixed).

The gradient of ∑_i h[i]h[i+k*] w.r.t. h[j] is:

So the direction to increase the autocorrelation at lag k* is h[j+k*] + h[j-k*].

To MINIMIZE the overlap at lag k*, we want to DECREASE h where h[j+k*] + h[j-k*] is large.

So the steepest descent direction for the worst lag is: -sign of (h[j+k*] + h[j-k*]) at each j.

This gives a very structured update: for each j, if h[j+k*] + h[j-k*] > 0 (which is always true), we should decrease h[j]. But we can't decrease all h[j] (that violates the sum constraint). So we need to decrease h[j] where h[j+k*]+h[j-k*] is largest, and increase it where it's smallest.

This is essentially a "water-filling" approach for each lag.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
