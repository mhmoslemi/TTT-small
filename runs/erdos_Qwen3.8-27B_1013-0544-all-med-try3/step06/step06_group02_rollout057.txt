    best_h = h.copy()

    zeros = list(np.where(h == 0)[0])

        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h[i] = 0; h[j] = 1
        new_raw_c = np.correlate(h, 1-h, mode='full')
        new_c = new_raw_c.max()
        delta = new_c - current_c
        
        if delta <= 0 or np.random.random() < np.exp(-delta / T):

            ones[i_idx] = j
            zeros[j_idx] = i

                best_c = current_c
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
        
        T *= cooling
    
    return best_h, best_c * dx

One more thought: the "effect" of a swap on the correlation can be computed very efficiently. Let me define:

For position p and lag k (where k = m - (n-1), ranging from -(n-1) to (n-1)):
- If p is a 1 being removed: Δ = - (1-h[p+k]) + h[p-k] = h[p-k] - 1 + h[p+k] (for valid indices)
- If p is a 0 being added: Δ = + (1-h[p+k]) - h[p-k] = -h[p-k] + 1 - h[p+k] = -(h[p-k] - 1 + h[p+k])

So the effect of adding at p is the negative of the effect of removing at p. Let's call e[p, k] = h[p-k] - 1 + h[p+k] (with out-of-range = 0).

For a swap (remove i, add j):
Δc[m] = e[i, k] - e[j, k] where k = m-(n-1)

Now, e[p, k] = h[p-k] + h[p+k] - 1 (for valid indices, treating out-of-range as 0)

For a binary h, e[p,k] ∈ {-1, 0, 1} (since h[p-k] and h[p+k] are each 0 or 1, so their sum is 0, 1, or 2, minus 1 gives -1, 0, or 1).

So |Δc[m]| ≤ 2 for any swap. That's useful!

The new c[m] = old c[m] + e[i,k] - e[j,k]. Since |e| ≤ 1, the new c[m] ∈ {old c[m] - 2, old c[m] - 1, old c[m], old c[m]+1, old c[m]+2}.

For the swap to be an improvement, we need max_m (c[m] + e[i,k_m] - e[j,k_m]) < old_max.

Since each c[m] can change by at most 2, the new max is at most old_max + 2. For it to be less than old_max, the positions that were at old_max must decrease, and no position can increase by 2 above old_max.

        #      Since |Δc[m]| <= 2 and raw_c[m] <= old_max - 2 for non-critical:
        #      raw_c[m] + Δc[m] <= old_max - 2 + 2 = old_max. 
        #      So we need the non-critical positions not to reach old_max.
        #      i.e., for m not in critical with raw_c[m] = old_max - 1: already in critical.
        #      For m not in critical with raw_c[m] <= old_max - 2: raw_c[m] + Δc[m] <= old_max.
        #      We need strict: raw_c[m] + Δc[m] < old_max, i.e., Δc[m] <= old_max - 1 - raw_c[m]
        #      For raw_c[m] = old_max - 2 (but not in critical, which means old_max - 2 < old_max - 1, contradiction)
        #      Wait, critical = {m : raw_c[m] >= old_max - 1}. So non-critical has raw_c[m] <= old_max - 2.
        #      For non-critical m: raw_c[m] + Δc[m] <= old_max - 2 + 2 = old_max. 
        #      We need < old_max, so we need Δc[m] <= old_max - 1 - raw_c[m] >= 1.
        #      Since Δc[m] <= 2, we need to check: if raw_c[m] = old_max - 1, it's in critical.
        #      If raw_c[m] <= old_max - 2, then even Δc[m] = 2 gives raw_c[m] + 2 <= old_max.
        #      We need < old_max, so if raw_c[m] = old_max - 2 and Δc[m] = 2, we get old_max (not < old_max).
        #      So we also need to check positions where raw_c[m] = old_max - 2.
        
        # Let me expand critical to include raw_c[m] >= old_max - 2:
        critical2 = np.where(raw_c >= old_max - 2)[0]
        
        # For the improvement, we need:
        # max_m (raw_c[m] + e[i,k_m] - e[j,k_m]) < old_max
        # For m in critical (raw_c = old_max): need e[i,k_m] - e[j,k_m] < 0, i.e., e[i,k_m] <= e[j,k_m] - 1
        # For m with raw_c = old_max - 1: need e[i,k_m] - e[j,k_m] <= 0, i.e., e[i,k_m] <= e[j,k_m]
        # For m with raw_c = old_max - 2: need e[i,k_m] - e[j,k_m] <= 1
        # For m with raw_c <= old_max - 3: always OK (Δc[m] >= -2, so raw_c[m]+Δc[m] <= old_max-1 < old_max)
        
        # Since e values are in {-1, 0, 1}, the constraints are:
        # At max positions: e[i,k] <= e[j,k] - 1 (so e[j,k] - e[i,k] >= 1)
        # At old_max-1 positions: e[i,k] <= e[j,k]
        # At old_max-2 positions: e[i,k] <= e[j,k] + 1
        
        # This is a set of inequality constraints on (e[i,k] - e[j,k]) for specific k values.
        
        # For efficiency, let's precompute for each position p the vector e[p, k] for k in the critical lags.
        
        n_critical = len(critical2)
        if n_critical == 0:
            break
        
        # e_matrix[p, t] = e[p, k_critical2[t]] = h[p - k_critical2[t]] + h[p + k_critical2[t]] - 1
        # This is an (n × n_critical) matrix
        
        # For ones: e_ones[ia, t] for ia in range(len(ones))
        # For zeros: e_zeros[ja, t] for ja in range(len(zeros))
        
        # We want: for all t where raw_c[critical2[t]] == old_max:
        #   e_ones[ia, t] - e_zeros[ja, t] <= -1
        # For all t where raw_c[critical2[t]] == old_max - 1:
        #   e_ones[ia, t] - e_zeros[ja, t] <= 0
        # For all t where raw_c[critical2[t]] == old_max - 2:
        #   e_ones[ia, t] - e_zeros[ja, t] <= 1
        
        # Define the "tolerance" vector: tol[t] = old_max - raw_c[critical2[t]]
        # Constraint: e_ones[ia, t] - e_zeros[ja, t] <= tol[t] - 1 for all t
        # (since we need raw_c[critical2[t]] + e_ones[ia,t] - e_zeros[ja,t] < old_max)
        # i.e., e_ones[ia,t] - e_zeros[ja,t] < old_max - raw_c[critical2[t]]
        # i.e., e_ones[ia,t] - e_zeros[ja,t] <= old_max - raw_c[critical2[t]] - 1 (for integers)
        
        tol = (old_max - raw_c[critical2]).astype(int) - 1
        # tol[t] = 0 for max positions, 1 for old_max-1 positions, 2 for old_max-2 positions
        
        # We need: max_t (e_ones[ia,t] - e_zeros[ja,t] - tol[t]) <= 0
        
        # For each ia, ja: check if max_t(e_ones[ia,t] - e_zeros[ja,t] - tol[t]) <= 0
        # This is O(len(ones) * len(zeros) * n_critical) = O(n³/4) for balanced ones/zeros
        
        # For n=300: 150 * 150 * (n_critical) operations
        # If n_critical is small (say 10-50), this is 150*150*50 = 1.1M, very fast!
        # If n_critical is large (say 100+), it's 2.25M, still fast.
        
        # Compute e matrices
        # e_ones[ia, t] = h[ones[ia] - k_c2[t]] + h[ones[ia] + k_c2[t]] - 1
        # With bounds checking
        
        # Vectorized:
        o_plus = ones[:,None] + critical2[None,:] - (n-1)  # ones[ia] + k_c2[t]
        o_minus = ones[:,None] - critical2[None,:] + (n-1)  # ones[ia] - k_c2[t]
        
        # Wait, k_c2[t] = critical2[t] - (n-1)
        # e[p, k_c2[t]] = h[p - k_c2[t]] + h[p + k_c2[t]] - 1
        # = h[p - (critical2[t]-(n-1))] + h[p + (critical2[t]-(n-1))] - 1
        # = h[p + (n-1) - critical2[t]] + h[p - (n-1) + critical2[t]] - 1
        
        idx_plus = ones[:,None] - (n-1) + critical2[None,:]  # p + k where k = c2 - (n-1), so p+k = p + c2 - (n-1)
        idx_minus = ones[:,None] + (n-1) - critical2[None,:]  # p - k = p - c2 + (n-1)
        
        e_ones = np.zeros((len(ones), len(critical2)))
        valid_plus = (idx_plus >= 0) & (idx_plus < n)
        valid_minus = (idx_minus >= 0) & (idx_minus < n)
        e_ones[valid_plus] += h[np.clip(idx_plus[valid_plus], 0, n-1)]
        e_ones[valid_minus] += h[np.clip(idx_minus[valid_minus], 0, n-1)]
        e_ones -= 1
        
        # Similarly for zeros
        idx_plus_z = zeros[:,None] - (n-1) + critical2[None,:]
        idx_minus_z = zeros[:,None] + (n-1) - critical2[None,:]
        
        e_zeros = np.zeros((len(zeros), len(critical2)))
        valid_plus_z = (idx_plus_z >= 0) & (idx_plus_z < n)
        valid_minus_z = (idx_minus_z >= 0) & (idx_minus_z < n)
        e_zeros[valid_plus_z] += h[np.clip(idx_plus_z[valid_plus_z], 0, n-1)]
        e_zeros[valid_minus_z] += h[np.clip(idx_minus_z[valid_minus_z], 0, n-1)]
        e_zeros -= 1
        
        # Now find the best (ia, ja) pair:
        # We need: max_t (e_ones[ia,t] - e_zeros[ja,t] - tol[t]) <= 0
        # And we want to maximize the "margin" = -max_t (e_ones[ia,t] - e_zeros[ja,t] - tol[t])
        
        # diff[ia, ja, t] = e_ones[ia,t] - e_zeros[ja,t] - tol[t]
        # This is (n_ones × n_zeros × n_critical) which might be too large.
        
        # Alternative: for each ia, compute the "constraint vector" v_ia = e_ones[ia,:] 
        # For each ja, the constraint is: max_t (v_ia[t] - e_zeros[ja,t] - tol[t]) <= 0
        # i.e., v_ia[t] - tol[t] <= e_zeros[ja,t] for all t
        # i.e., (v_ia - tol) is element-wise <= e_zeros[ja,:]
        
        # So for each ia: find ja such that e_zeros[ja,:] >= (e_ones[ia,:] - tol) element-wise.
        # Among valid (ia, ja), pick the one that maximizes the "improvement."
        
        # The improvement at the max positions is: -min_t (e_ones[ia,t] - e_zeros[ja,t]) for t where tol[t]=0
        # = min_t (e_zeros[ja,t] - e_ones[ia,t]) for max positions
        
        # This is still a lot of computation. Let me simplify.
        
        # For each ia:
        #   lower_bound[ia, t] = e_ones[ia, t] - tol[t]
        #   Find ja such that e_zeros[ja, t] >= lower_bound[ia, t] for all t
        #   Among valid ja, pick the one that gives the best improvement
        
        # For n=300 with 150 ones and 150 zeros, and say 20 critical positions:
        # For each ia (150): check 150 ja values, each check is O(20). Total: 150*150*20 = 450K. Fine!
        
        found = False
        for ia in range(len(ones)):
            lb = e_ones[ia, :] - tol
            # Find ja where e_zeros[ja, :] >= lb (element-wise)
            valid_ja = np.where(np.all(e_zeros >= lb[None, :], axis=1))[0]
            if len(valid_ja) == 0:
                continue
            # Among valid_ja, pick the one that gives the best improvement
            # Improvement = reduction in max = -max_t(e_ones[ia,t] - e_zeros[ja,t] - tol[t])
            # For the max positions (tol=0): improvement = -(e_ones[ia,t] - e_zeros[ja,t]) = e_zeros[ja,t] - e_ones[ia,t]
            # Best ja maximizes min_t (e_zeros[ja,t] - e_ones[ia,t]) for t where tol[t]=0
            # Or more simply: minimizes max_t (e_ones[ia,t] - e_zeros[ja,t]) for t where tol[t]=0
            
            max_positions = (tol == 0)
            if np.sum(max_positions) > 0:
                scores = -np.max(e_ones[ia, max_positions][None,:] - e_zeros[valid_ja, max_positions], axis=1)
            else:
                scores = np.zeros(len(valid_ja))
            
            best_ja = valid_ja[np.argmax(scores)]
            
            # Accept the swap
            i_pos = ones[ia]
            j_pos = zeros[best_ja]
            h[i_pos] = 0
            h[j_pos] = 1
            raw_c = np.correlate(h, 1-h, mode='full')
            new_max = raw_c.max()
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
            found = True
            break  # restart the search with the new configuration
        
        if not found:
            break
    
    return best_h, best_c * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_binary(h, n):
    """Project h to a binary vector with exactly n//2 ones."""
    target = n // 2  # number of ones
    h_copy = h.copy()
    # Threshold at 0.5
    binary = (h_copy >= 0.5).astype(float)
    n_ones = binary.sum()
    if n_ones == target:
        return binary
    if n_ones > target:
        # Need to turn off some ones: pick the ones with lowest h values
        one_idx = np.where(binary == 1)[0]
        order = np.argsort(h_copy[one_idx])
        for i in order[:int(n_ones - target)]:
            binary[one_idx[i]] = 0
    else:
        # Need to turn on some zeros: pick the zeros with highest h values
        zero_idx = np.where(binary == 0)[0]
        order = np.argsort(-h_copy[zero_idx])
        for i in order[:int(target - n_ones)]:
            binary[zero_idx[i]] = 1
    return binary

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def target_local_search(h, n, deadline):
    """Local search targeting the maximum positions."""
    dx = 2.0 / n
    raw_c = np.correlate(h, 1-h, mode='full')
    best_c = raw_c.max()
    best_h = h.copy()
    
    while time.time() < deadline:
        old_max = raw_c.max()
        if old_max <= best_c and False:
            break
        
        # Critical positions: where raw_c[m] >= old_max - 1
        critical = np.where(raw_c >= old_max - 1)[0]
        k_c = critical - (n-1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute e_ones[ia, t] and e_zeros[ja, t]
        # e[p, k] = h[p-k] + h[p+k] - 1 (out-of-range = 0)
        # k = k_c[t] = critical[t] - (n-1)
        # p - k = ones[ia] - (critical[t] - (n-1)) = ones[ia] + (n-1) - critical[t]
        # p + k = ones[ia] + (critical[t] - (n-1)) = ones[ia] - (n-1) + critical[t]
        
        idx_minus_o = ones[:, None] + (n-1) - critical[None, :]
        idx_plus_o = ones[:, None] - (n-1) + critical[None, :]
        
        e_ones = np.zeros((len(ones), len(critical)), dtype=np.int8)
        vpm = (idx_minus_o >= 0) & (idx_minus_o < n)
        vpo = (idx_plus_o >= 0) & (idx_plus_o < n)
        e_ones[vpm] += h[np.minimum(np.maximum(idx_minus_o[vpm], 0), n-1)].astype(np.int8)
        e_ones[vpo] += h[np.minimum(np.maximum(idx_plus_o[vpo], 0), n-1)].astype(np.int8)
        e_ones -= 1
        
        idx_minus_z = zeros[:, None] + (n-1) - critical[None, :]
        idx_plus_z = zeros[:, None] - (n-1) + critical[None, :]
        
        e_zeros = np.zeros((len(zeros), len(critical)), dtype=np.int8)
        vpm_z = (idx_minus_z >= 0) & (idx_minus_z < n)
        vpo_z = (idx_plus_z >= 0) & (idx_plus_z < n)
        e_zeros[vpm_z] += h[np.minimum(np.maximum(idx_minus_z[vpm_z], 0), n-1)].astype(np.int8)
        e_zeros[vpo_z] += h[np.minimum(np.maximum(idx_plus_z[vpo_z], 0), n-1)].astype(np.int8)
        e_zeros -= 1
        
        # Tolerance: we need raw_c[critical[t]] + e_i[t] - e_j[t] < old_max
        # i.e., e_i[t] - e_j[t] < old_max - raw_c[critical[t]]
        # i.e., e_i[t] - e_j[t] <= old_max - raw_c[critical[t]] - 1 (integer)
        tol = (old_max - raw_c[critical]).astype(int) - 1
        
        # For each ia, find ja such that e_ones[ia,t] - e_zeros[ja,t] <= tol[t] for all t
        # i.e., e_zeros[ja,t] >= e_ones[ia,t] - tol[t] for all t
        
        found = False
        for ia in range(len(ones)):
            lb = e_ones[ia, :] - tol
            # Need: e_zeros[ja, :] >= lb for all t
            valid = np.all(e_zeros >= lb[None, :], axis=1)
            valid_idx = np.where(valid)[0]
            if len(valid_idx) == 0:
                continue
            
            # Score: how much do we reduce the max?
            # At max positions (tol=0): reduction = e_zeros[ja,t] - e_ones[ia,t]
            # We want to maximize the minimum reduction at max positions
            max_pos = (tol == 0)
            if np.any(max_pos):
                reductions = e_zeros[valid_idx, max_pos] - e_ones[ia, max_pos][None, :]
                scores = np.min(reductions, axis=1)
            else:
                scores = np.zeros(len(valid_idx))
            
            best_ja = valid_idx[np.argmax(scores)]
            
            # Execute the swap
            i_pos = ones[ia]
            j_pos = zeros[best_ja]
            h[i_pos] = 0
            h[j_pos] = 1
            raw_c = np.correlate(h, 1-h, mode='full')
            new_max = raw_c.max()
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
            found = True
            break
        
        if not found:
            break
    
    return best_h, best_c * dx

def simulate_anneal(h, n, deadline, seed=42):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    raw_c = np.correlate(h, 1-h, mode='full')
    best_c = raw_c.max()
    best_h = h.copy()
    current_c = raw_c.max()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    T = 0.3  # initial temperature (in units of the raw correlation, which is ~n/4 for random)
    # Actually, let's think about this. The raw correlation values are in [0, n/2].
    # A "good" swap reduces the max by 1. A "bad" swap increases it by 1 or 2.
    # T should be on the order of the typical |delta| we want to accept.
    # Let's use T in units of the max value, so T=0.3 means we're willing to accept
    # increases of up to ~0.3*old_max.
    
    while time.time() < deadline:
        # Adaptive cooling
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        h[i] = 0
        h[j] = 1
        new_raw_c = np.correlate(h, 1-h, mode='full')
        new_c = new_raw_c.max()
        delta = new_c - current_c
        
        if delta <= 0 or rng.random() < np.exp(-float(delta) / T):
            raw_c = new_raw_c
            current_c = new_c
            ones[i_idx] = j
            zeros[j_idx] = i
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T *= 0.99999
    
    return best_h, best_c * dx

def continuous_refine(h, n, deadline):
    """Refine using L-BFGS-B with penalty method."""
    from scipy.optimize import minimize as sp_min
    
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        alpha = 5000
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += 1e6 * (x.sum() - n/2.0)**2
        return float(val)
    
    def grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        alpha = 5000
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = np.zeros(n)
        for m_idx in range(2*n-1):
            if w[m_idx] < 1e-13:
                continue
            k = m_idx - (n-1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        g += 2e6 * (x.sum() - n/2.0)
        return g
    
    try:
        res = sp_min(obj, h.copy(), jac=grad, method='L-BFGS-B',
                     bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except:
        return h.copy(), compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
