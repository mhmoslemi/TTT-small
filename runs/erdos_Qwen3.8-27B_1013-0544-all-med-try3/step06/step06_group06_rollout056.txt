import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _coeffs_for_j(h, n, j):
    """Compute affected k-range and coefficients for coordinate j."""
    s_lo = max(j - (n - 1), -j)
    s_hi = min(j, n - 1 - j)
    if s_lo > s_hi:
        return None
    k_lo = s_lo + (n - 1)
    k_hi = s_hi + (n - 1)
    ss = np.arange(k_lo, k_hi + 1, dtype=np.int64) - (n - 1)
    va = (j - ss >= 0) & (j - ss < n)
    vb = (j + ss >= 0) & (j + ss < n)
    coeffs = np.zeros(k_hi - k_lo + 1)
    if np.any(va):
        coeffs[va] += 1.0 - h[j - ss[va]]
    if np.any(vb):
        coeffs[vb] -= h[j + ss[vb]]
    return k_lo, k_hi, coeffs

def _solve_1d_convex(c_aff, coeffs, c_unmax, t_lo, t_hi):
    """Minimize max(c_aff + t*coeffs, c_unmax) over t in [t_lo, t_hi]."""
    if t_lo >= t_hi:
        return 0.0, max(np.max(c_aff), c_unmax)
    
    f0 = max(np.max(c_aff), c_unmax)
    
    # Check if f0 is already at c_unmax (flat)
    if f0 <= c_unmax + 1e-14:
        return 0.0, f0
    
    best_t = 0.0
    best_f = f0
    lo, hi = t_lo, t_hi
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = c_aff + mid * coeffs
        f_mid = max(np.max(vals), c_unmax)
        
        if f_mid < best_f:
            best_f = f_mid
            best_t = mid
        
        if f_mid <= c_unmax + 1e-14:
            break
        
        vm = np.max(vals)
        if vm <= c_unmax + 1e-14:
            break
        
        active = vals >= vm - 1e-10
        if not np.any(active):
            break
        min_c = np.min(coeffs[active])
        max_c = np.max(coeffs[active])
        
        if min_c <= 1e-14 and max_c >= -1e-14:
            break
        elif max_c < -1e-14:
            lo = mid
        else:
            hi = mid
    
    return best_t, best_f

def coord_descent(h_in, n, deadline, max_sweeps=100, seed=42):
    """Single-coordinate descent with exact 1D optimization."""
    rng = np.random.RandomState(seed)
    h = h_in.astype(np.float64).copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    best_c = np.max(c) * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.time() > deadline:
                break
            
            res = _coeffs_for_j(h, n, j)
            if res is None:
                continue
            k_lo, k_hi, coeffs = res
            m = k_hi - k_lo + 1
            c_aff = c[k_lo:k_hi + 1]
            
            if k_lo > 0:
                c_unmax_left = np.max(c[:k_lo])
            else:
                c_unmax_left = -np.inf
            if k_hi < 2 * n - 2:
                c_unmax_right = np.max(c[k_hi + 1:])
            else:
                c_unmax_right = -np.inf
            c_unmax = max(c_unmax_left, c_unmax_right)
            
            t_lo = -h[j]
            t_hi = 1.0 - h[j]
            if t_lo >= t_hi:
                continue
            
            best_t, best_f = _solve_1d_convex(c_aff, coeffs, c_unmax, t_lo, t_hi)
            
            if best_f < np.max(c) - 1e-11:
                h[j] += best_t
                c[k_lo:k_hi + 1] = c_aff + best_t * coeffs
                improved = True
        
        # Project to maintain feasibility
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
        c_new = np.max(c) * dx
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        
        if not improved and sweep > 5:
            # Try a small shake to escape local optima
            if sweep % 20 == 0 and time.time() < deadline - 10:
                pert = 0.05 * rng.randn(n)
                h_try = project_feasible(h + pert, n)
                c_try = np.max(np.correlate(h_try, 1 - h_try, mode='full')) * dx
                if c_try < best_c:
                    h = h_try
                    c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def simulated_annealing(h_in, n, deadline, max_steps=200000, T_init=0.5, T_min=0.001):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(12345)
    # Binarize
    idx = np.argsort(h_in)[::-1]
    target = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    h[idx[target:]] = 0.0
    
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    current_max = np.max(c)
    
    best_h = h.copy()
    best_max = current_max
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    n_ones = len(ones)
    T = T_init
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if step % 1000 == 0 and step > 0:
            T = max(T_min, T_init * (T_min / T_init) ** (step / max_steps))
        
        i = rng.randint(len(zeros))
        j = rng.randint(len(ones))
        zi = zeros[i]
        oj = ones[j]
        
        # Perform swap: h[zi] 0->1, h[oj] 1->0
        # Coefficients for zi (was 0, becoming 1): Δ=+1
        res_i = _coeffs_for_j(h, n, zi)
        res_j = _coeffs_for_j(h, n, oj)
        
        if res_i is None or res_j is None:
            continue
        k_lo_i, k_hi_i, coeff_i = res_i
        k_lo_j, k_hi_j, coeff_j = res_j
        
        # Combined affected range
        k_lo_min = min(k_lo_i, k_lo_j)
        k_hi_max = max(k_hi_i, k_hi_j)
        
        # Save old values
        old_c = c[k_lo_min:k_hi_max + 1].copy()
        old_max_in_range = np.max(old_c)
        old_global_max = current_max
        
        # Apply swap
        h[zi] = 1.0
        h[oj] = 0.0
        
        # Compute new c for affected range
        # Δc[k] = coeff_i[k] (for k in i's range) - coeff_j[k] (for k in j's range)
        c_new = c.copy()
        c_new[k_lo_i:k_hi_i + 1] += coeff_i
        c_new[k_lo_j:k_hi_j + 1] -= coeff_j
        
        new_max = np.max(c_new)
        
        if new_max <= current_max or (T > T_min and rng.random() < np.exp(-(new_max - current_max) / (T * 1e-3))):
            c = c_new
            current_max = new_max
            zeros[i] = oj
            ones[j] = zi
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Revert
            c[k_lo_min:k_hi_max + 1] = old_c
            h[zi] = 0.0
            h[oj] = 1.0
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def pair_coord_descent(h_in, n, deadline, max_sweeps=50, seed=99):
    """Pair coordinate descent maintaining sum constraint exactly."""
    rng = np.random.RandomState(seed)
    h = h_in.astype(np.float64).copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    best_c = np.max(c) * dx
    best_h = h.copy()
    
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(pairs)
        improved = False
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            
            # Feasible range for t: h[i]+t in [0,1], h[j]-t in [0,1]
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_lo >= t_hi - 1e-14:
                continue
            
            res_i = _coeffs_for_j(h, n, i)
            res_j = _coeffs_for_j(h, n, j)
            if res_i is None or res_j is None:
                continue
            k_lo_i, k_hi_i, coeff_i = res_i
            k_lo_j, k_hi_j, coeff_j = res_j
            
            k_lo_min = min(k_lo_i, k_lo_j)
            k_hi_max = max(k_hi_i, k_hi_j)
            m = k_hi_max - k_lo_min + 1
            
            # Combined coefficient: t*(coeff_i) + (-t)*(coeff_j) = t*(coeff_i - coeff_j)
            combined = np.zeros(m)
            # Add coeff_i at its offset
            off_i = k_lo_i - k_lo_min
            combined[off_i:off_i + (k_hi_i - k_lo_i + 1)] += coeff_i
            # Subtract coeff_j at its offset
            off_j = k_lo_j - k_lo_min
            combined[off_j:off_j + (k_hi_j - k_lo_j + 1)] -= coeff_j
            
            c_aff = c[k_lo_min:k_hi_max + 1]
            
            if k_lo_min > 0:
                c_unmax_left = np.max(c[:k_lo_min])
            else:
                c_unmax_left = -np.inf
            if k_hi_max < 2 * n - 2:
                c_unmax_right = np.max(c[k_hi_max + 1:])
            else:
                c_unmax_right = -np.inf
            c_unmax = max(c_unmax_left, c_unmax_right)
            
            best_t, best_f = _solve_1d_convex(c_aff, combined, c_unmax, t_lo, t_hi)
            
            if best_f < np.max(c) - 1e-11:
                h[i] += best_t
                h[j] -= best_t
                c[k_lo_min:k_hi_max + 1] = c_aff + best_t * combined
                improved = True
        
        c_curr = np.max(c) * dx
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    return best_h, best_c

def slsqp_polish(h, n, deadline, alphas):
    """Multiple SLSQP steps with increasing alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:

        c_new = compute_c5(h_new, n)
        if c_new < best_c:

            best_h = h_new.copy()
    return best_h, best_c
