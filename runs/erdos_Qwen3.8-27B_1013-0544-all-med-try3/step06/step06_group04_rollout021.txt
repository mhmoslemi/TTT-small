        obj = m + log_sum / alpha
    else:

        log_sum = np.log(exp_vals.sum())

    grad = np.zeros(n)

    for idx in significant:

        grad += weights[idx] * lag_gradient(h, n, lag)
    
    return obj, grad

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n-m] += dx * (1.0 - h[m:])
        grad[m:] += -dx * h[:n-m]
    else:
        grad[m:] += dx * (1.0 - h[:n-m])
        grad[:n-m] += -dx * h[m:]
    return grad

def smooth_max_obj_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    log_sum = mx + np.log(exp_vals.sum())
    obj = m + log_sum / alpha
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    sig = np.where(weights > 1e-12)[0]
    for idx in sig:
        lag = int(idx) - (n - 1)
        grad += weights[idx] * lag_gradient(h, n, lag)
    return obj, grad

def outer_approx_step(h_curr, n, trust):

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] += dx * (1.0 - h[m:])
        grad[m:] += -dx * h[:n - m]
    else:
        grad[m:] += dx * (1.0 - h[:n - m])
        grad[:n - m] += -dx * h[m:]
    return grad


def smooth_max_obj_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    log_sum = mx + np.log(exp_vals.sum())
    obj = m + log_sum / alpha
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    sig = np.where(weights > 1e-12)[0]
    for idx in sig:
        lag = int(idx) - (n - 1)
        grad += weights[idx] * lag_gradient(h, n, lag)
    return obj, grad


def _smooth_obj_wrap(x, n, alpha):
    o, _ = smooth_max_obj_grad(x, n, alpha)
    return o


def _smooth_jac_wrap(x, n, alpha):
    _, g = smooth_max_obj_grad(x, n, alpha)
    return g


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def slsqp_analytical(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj_wrap, h, jac=_smooth_jac_wrap,
                      args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def subgradient_descent(h, n, max_iters=50000, deadline=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, max_iters + 1):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = lag_gradient(h, n, lag)
        step = 0.5 / np.sqrt(t)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c


def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)


def outer_approx(h, n, deadline, max_iters=60):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.85)
    return best_h, best_c
