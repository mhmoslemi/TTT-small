def exact_pair_step(h, n, i, j):
    """Find optimal v to move from h[j] to h[i], maintaining sum."""
    # h_new[i] = h[i] + v, h_new[j] = h[j] - v
    # Valid range: max(0, -h[i]) <= v <= min(1-h[i], h[j])
    # Objective: max_k correlate(h_new, 1-h_new, k) * dx

        h_new = base.copy()
        h_new[i] += v
        h_new[j] -= v

    lo = max(0.0, -h[i])
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return h, compute_c5(h, n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:

    m = c.max()

At each iteration:
1. Compute c = correlate(h, 1-h) * dx
2. Find k* = argmax c
3. Compute the subgradient with respect to h for the active lag k*
4. Take a step in the negative subgradient direction
5. Project back to feasible set

def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_star = np.argmax(c)
    k = k_star - (n - 1)
    
    # Subgradient of c_k with respect to h[i]:
    # ∂c_k/∂h[i] = (1-h[i+k])*dx if valid - h[i-k]*dx if valid
    sg = np.zeros(n)
    if k >= 0:
        nv = n - k
        sg[0:nv] += (1 - h[k:n]) * dx
        sg[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        sg[kk:n] += (1 - h[0:nv]) * dx
        sg[0:nv] += (-h[kk:n]) * dx
    
    h_new = h - step_size * sg
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_penalty(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_penalty(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search_v2(h, n, deadline):

    h = h.copy()
    c = compute_correlation(h, n)
    best_c = float(c.max())
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    total_lags = 2 * n - 1
    
    passes = 0
    while time.time() < deadline - 10:
        if passes > 200:
            break
        passes += 1
        improved = False
        n_improvements = 0
        
        for i in range(n):
            if time.time() > deadline - 10:
                break

                if time.time() > deadline - 10:
                    break
                for (si, sj) in [(i, j), (j, i)]:
                    lo = max(0.0, -h[si])
                    hi = min(1.0 - h[si], h[sj])
                    if hi - lo < 1e-12:
                        continue
                    
                    # Compute d vector
                    d = np.zeros(total_lags)
                    m1 = (sj + ks >= 0) & (sj + ks < n)
                    if m1.any():
                        d[m1] += h[sj + ks[m1]]
                    m2 = (si + ks >= 0) & (si + ks < n)
                    if m2.any():
                        d[m2] -= h[si + ks[m2]]
                    m3 = (si - ks >= 0) & (si - ks < n)
                    if m3.any():
                        d[m3] += h[si - ks[m3]]
                    m4 = (sj - ks >= 0) & (sj - ks < n)
                    if m4.any():
                        d[m4] -= h[sj - ks[m4]]
                    d *= dx

                    if max_decrease < 1e-12:
                        continue
                    
                    def obj1d(v):
                        return float(np.max(c + v * d))

                                         options={'xatol': 1e-13})

                        h[sj] = np.clip(h[sj] - res.x, 0, 1)
                        c = c + res.x * d
                        best_c = float(c.max())
                        improved = True
                        n_improvements += 1
        
        if not improved:

    return h, best_c
