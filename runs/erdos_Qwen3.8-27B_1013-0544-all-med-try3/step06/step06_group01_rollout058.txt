```python
def project_to_box_and_sum(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # clip(h - lambda, 0, 1) = max(0, min(1, h - lambda))
    # This is a monotonically decreasing function of lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any floating point issues
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - result
            s = room.sum()
            if s > 1e-14:
                result += diff / s * room
        else:
            s = result.sum()
            if s > 1e-14:
                result += diff / s * result
    return np.clip(result, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    # Bisection projection
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - result
            s = room.sum()
            if s > 1e-14: result += diff/s*room
        else:
            s = result.sum()
            if s > 1e-14: result += diff/s*result
    return np.clip(result, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    # Work with corr * dx for the objective
    c = corr * dx
    max_c = c.max()
    sh = alpha * (c - max_c)
    sh_max = sh.max()
    exp_sh = np.exp(sh - sh_max)
    log_sum = np.log(exp_sh.sum())
    obj = max_c + (sh_max + log_sum) / alpha
    p = exp_sh / exp_sh.sum()
    # Gradient using FFT
    A = fftconvolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    B_full = fftconvolve(p, h[::-1].copy(), mode='full')
    B = B_full[n-1:2*n-1][::-1]
    grad = dx * (A - B)
    return obj, grad

def projected_gradient(h, n, alpha, deadline, max_iter=5000):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 2.0 / alpha  # initial step size
    obj, grad = smooth_obj_and_grad(h, n, alpha)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Try step
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            h = h_new
            step = min(step * 1.5, 10.0 / alpha)
            obj, grad = smooth_obj_and_grad(h, n, alpha)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c

def binary_sa(h_init, n, deadline, seed=42):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    target = n // 2
    # Start from the top n/2 values of h_init
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    T = 0.02  # Initial temperature
    cooling = 0.999
    ones = set(np.where(h == 1)[0].tolist())
    zeros = list(np.where(h == 0)[0].tolist())
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Pick random swap
        i = zeros[rng.randint(len(zeros))]
        j = list(ones)[rng.randint(len(ones))]
        
        # Compute new correlation
        h[i], h[j] = 1.0, 0.0
        new_corr = np.correlate(h, 1-h, mode='full')
        new_c = new_corr.max() * dx
        delta = new_c - cur_c
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            corr = new_corr
            cur_c = new_c
            ones.discard(j); ones.add(i)
            zeros.remove(i); zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Reject
            h[i], h[j] = 0.0, 1.0
        
        T *= cooling
        if T < 1e-6:
            T = 1e-6
    
    return best_h, best_c
```

```python
    T_start = 0.05
    T_min = 1e-7
    it_budget = n_iterations if n_iterations else int(max(1, (deadline - time.time()) * 500))
    cooling = (T_min / T_start) ** (1.0 / max(1, it_budget))
    T = T_start
    best_iter = 0
    patience = 0
    max_patience = int(it_budget * 0.05)

    for it in range(it_budget):
        if time.time() > deadline:
            break
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = zi[rng.randint(len(zi))]
        j = oi[rng.randint(len(oi))]
        h[i], h[j] = 1.0, 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c = new_corr.max() * dx
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_iter = it
                patience = 0
            else:
                patience += 1
        else:
            h[i], h[j] = 0.0, 1.0
            patience += 1
        T = max(T * cooling, T_min)
        if patience > max_patience:
            break
    return best_h, best_c

def multi_slsqp_polish(h, n, deadline):
    rng = np.random.RandomState(12345)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [10000, 50000, 200000, 1000000, 5000000]:
        if time.time() > deadline:
            break
        for attempt in range(2):
            if time.time() > deadline:
                break
            h0 = best_h + 0.03 * rng.randn(n)
            h0 = project_feasible(h0, n)
            h_new = slsqp_smooth(h0, n, alpha, 1000, deadline)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Quick SLSQP from parent
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 20000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 800, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Projected gradient with FFT-based gradients
    p2 = t0 + 0.30 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > p2:
            break
        h, c = projected_grad_descent(h, n, alpha, p2, max_iter=10000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Binary local search
    p3 = t0 + 0.45 * (budget_s - 5)
    hb, cb = binary_local_search(best_h, n, p3, max_sweeps=2000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 4: Binary simulated annealing
    p4 = t0 + 0.62 * (budget_s - 5)
    if time.time() < p4 - 10:
        hb, cb = binary_simulated_annealing(best_h, n, p4, seed=seed + 10)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Run a quick local search after SA
        if time.time() < p4 - 2:
            hb2, cb2 = binary_local_search(best_h, n, p4, max_sweeps=500, seed=seed + 11)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()

    # Phase 5: Multi-resolution exploration
    p5 = t0 + 0.78 * (budget_s - 5)
    for nh in [200, 300, 375, 500]:
        if time.time() > p5 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h, kind='linear'), nh)
        # Quick projected gradient
        hh, ch = projected_grad_descent(hh, nh, 50000, p5 - 15, max_iter=3000)
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
        # Quick binary search
        if time.time() < p5 - 5:
            hb3, cb3 = binary_local_search(best_h if n == nh else hh, 
                                            nh if n != nh else nh, 
                                            p5 - 3, max_sweeps=200, seed=seed + nh)
            if cb3 < best_c:
                best_c = cb3
                best_h = hb3.copy()
                n = nh

    # Phase 6: SLSQP polish on best solution
    p6 = t0 + 0.88 * (budget_s - 5)
    h_best, c_best = multi_slsqp_polish(best_h, n, p6)
    if c_best < best_c:
        best_c = c_best
        best_h = h_best.copy()

    # Phase 7: Final binary polish
    if time.time() < dl - 3:
        hb, cb = binary_local_search(best_h, n, dl, max_sweeps=1000, seed=seed + 20)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
