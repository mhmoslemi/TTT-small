import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        exp_s = np.exp(shifted - mx)
    else:
        exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    n_lags = len(c)
    for k_idx in range(n_lags):
        pk = p[k_idx]
        if pk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        mm = abs(lag)
        if mm >= n:
            continue
        if mm == 0:
            grad += pk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n-mm] += pk * (-x[mm:])
            grad[mm:] += pk * (1.0 - x[:n-mm])
        else:
            grad[:n-mm] += pk * (1.0 - x[mm:])
            grad[mm:] += pk * (-x[:n-mm])
    if mx > 500:
        obj = m + (mx + np.log(np.sum(exp_s))) / alpha
    else:
        obj = m + np.log(np.sum(exp_s)) / alpha
    return obj, grad

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth_obj_grad(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        exp_s = np.exp(shifted - mx)
    else:
        exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for ii, k_idx in enumerate(top_idx):
        pk = p[ii]
        if pk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        mm = abs(lag)
        if mm >= n:
            continue
        if mm == 0:
            grad += pk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n-mm] += pk * (-x[mm:])
            grad[mm:] += pk * (1.0 - x[:n-mm])
        else:
            grad[:n-mm] += pk * (1.0 - x[mm:])
            grad[mm:] += pk * (-x[:n-mm])
    if mx > 500:
        obj = m + (mx + np.log(np.sum(exp_s))) / alpha
    else:
        obj = m + np.log(np.sum(exp_s)) / alpha
    return obj, grad

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj_func(x):
        return _smooth_obj(x, n, alpha)
    def grad_func(x):
        _, g = _smooth_obj_grad(x, n, alpha)
        return g
    try:
        res = minimize(obj_func, h, jac=grad_func, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_idx, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj_func(x):
        return _active_smooth_obj(x, n, top_idx, alpha)
    def grad_func(x):
        _, g = _active_smooth_obj_grad(x, n, top_idx, alpha)
        return g
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n or m == 0:
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust, active_lags):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in active_lags:
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=200):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        tol = max(0.002, 0.05 * cmax)
        active_lags = np.where(c >= cmax - tol)[0]
        if len(active_lags) > 30:
            active_lags = np.argsort(c)[-30:]
        h_new = outer_approx_step(h_curr, n, trust, active_lags)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            trust = max(0.001, trust * 0.7)
        else:
            trust = max(0.001, trust * 0.85)
        h_curr = h_new
    return best_h, best_c

def local_search_pair_swap(h, n, deadline, max_passes=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for pswap in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cmax = c.max()
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        improved = False
        step_size = 0.001
        for i in range(n - m):
            if time.time() > deadline:
                break
            j = i + m
            if h[i] <= h[j]:
                continue
            delta = min(step_size, (h[i] - h[j]) * 0.4)
            if delta < 1e-9:
                continue
            h[i] -= delta
            h[j] += delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                h[i] += delta
                h[j] -= delta
        if not improved:
            step_size *= 0.5
            if step_size < 1e-6:
                break
    return h, best_c

def local_search_top_lags(h, n, deadline, top_m=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    for pass_num in range(100):
        if time.time() > deadline:
            break
        improved = False
        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] <= h[j]:
                    continue
                delta = min(0.0005, (h[i] - h[j]) * 0.3)
                if delta < 1e-9:
                    continue
                h[i] -= delta
                h[j] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i] += delta
                    h[j] -= delta
        if not improved:
            break
    return h, best_c

def subgradient_method(h, n, deadline, iters=5000):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    c_upper = best_c + 0.01
    x_avg = h_curr.copy()
    for it in range(iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h_curr.copy()
        tol = max(0.003, 0.1 * cmax)
        active = c >= cmax - tol
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])
        grad = np.zeros(n)
        for k_idx in active_idx:
            lag = k_idx - (n - 1)
            grad += _compute_lag_gradient(h_curr, n, lag)
        grad /= len(active_idx)
        c_lower = cmax
        step = (c_upper - c_lower) / (np.dot(grad, grad) + 1e-20)
        step = min(step, 0.1)
        h_new = h_curr - step * grad
        h_new = project_feasible(h_new, n)
        x_avg = (it / (it + 1.0)) * x_avg + (1.0 / (it + 1.0)) * h_new
        h_curr = h_new
        if cmax < c_upper:
            c_upper = cmax + 0.005
        if (it + 1) % 200 == 0:
            c_avg = compute_c5(x_avg, n)
            if c_avg < best_c:
                best_c = c_avg
                best_h = x_avg.copy()
    c_avg = compute_c5(x_avg, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = x_avg.copy()
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Outer approximation
    if time.time() < deadline - 40:
        oo_end = time.time() + min(100, (deadline - time.time()) * 0.30)
        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    # Phase 2: SLSQP with increasing alpha
    if time.time() < deadline - 30:
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 25:
        c = compute_all_lags(best_h, n)
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 20:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-top_m:]
            for alpha in [20000, 100000]:
                if time.time() > deadline - 18:
                    break
                h_new = slsqp_active(best_h, n, top_idx, alpha, maxiter=1500, ftol=1e-15)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    # Phase 4: Subgradient with averaging
    if time.time() < deadline - 30:
        sg_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_sg, c_sg = subgradient_method(best_h, n, sg_end, iters=3000)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 15:
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 12:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 6: Local search
    if time.time() < deadline - 5:
        best_h, best_c = local_search_pair_swap(best_h, n, min(deadline - 1, time.time() + 20))
    if time.time() < deadline - 2:
        best_h, best_c = local_search_top_lags(best_h, n, deadline - 1, top_m=3)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        c_init = compute_c5(h_init, n_init)
        best_h = h_init.copy()
        best_c = c_init
        # Optimize at initial resolution
        alloc = min(300, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Try different resolutions
    n_candidates = [n_init, 400, 500]
    if h_init is not None:
        n_candidates = [n_init] + [n for n in [400, 500] if n != n_init]

    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 40
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Random restarts from best solution
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining / 60))
        for r in range(min(n_restarts, 8)):
            if time.time() > deadline - 30:
                break
            scale = 0.008 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 1000000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = local_search_pair_swap(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
