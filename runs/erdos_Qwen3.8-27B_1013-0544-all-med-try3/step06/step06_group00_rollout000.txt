import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init

                if 0 <= i - k < n:

        grad /= len(top_idx)

        h = project_feasible(h - lr * grad, n)
        
        # Decay learning rate
        lr *= 0.9999
    
    return best_h, best_c

            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 0)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def subgradient_polish(h, n, deadline, n_active=5, lr_init=0.005, decay=0.99995):
    """Polish using subgradient on active shifts only."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stuck_count = 0
    
    for it in range(5000000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max() * dx
        
        if cur_max < best_c - 1e-12:
            best_c = cur_max
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        if stuck_count > 50000:
            break
        
        # Find top n_active shifts
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        # Compute negative subgradient (direction to decrease)
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = int(k_idx) - (n - 1)
            # grad[m] += (1-h[m+k])*dx for valid m+k
            # grad[m] -= h[m-k]*dx for valid m-k
            if k >= 0:
                m_start, m_end = 0, n - k
                grad[m_start:m_end] += (1 - h[m_start + k:m_end + k]) * dx
                m_start2, m_end2 = k, n
                grad[m_start2:m_end2] -= h[0:m_end2 - k] * dx
            else:
                neg_k = -k
                m_start, m_end = neg_k, n
                grad[m_start:m_end] += (1 - h[0:m_end - neg_k]) * dx
                m_start2, m_end2 = 0, n - neg_k
                grad[m_start2:m_end2] -= h[m_start2 + neg_k:m_end2 + neg_k] * dx
        grad /= len(top_idx)
        
        h = project_feasible(h - lr * grad, n)
        lr *= decay
    
    return best_h, best_c

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    """Greedy binary local search with pairwise swaps."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, T_init=0.05, n_iters=5000000):
    """Simulated annealing with better schedule."""
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T_init
    cool = 0.99999
    scale = 5000.0
    best_iter = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-15:
            delta_e = scale * (cur_c - nc) / T
            if delta_e > -700 and rng.random() < np.exp(delta_e):
                accept = True
        
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_iter = it
        
        T *= cool
    
    return best_h, best_c
