        x0 = np.concatenate([h, [t_init]])

        try:
            res = minimize(as_obj, x0, args=(n,), jac=as_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': max_inner_iter, 'ftol': 1e-14})
            if time.time() - t_start > time_limit:
                break

            h_new = project_feasible(res.x[:n], target_sum)
            c5_new = compute_c5(h_new)

            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
            t_val = res.x[n]

            new_violated = []
            for idx in range(2 * n - 1):
                lag = idx - (n - 1)
                if lag not in active_lags and corr_new[idx] > t_val + 1e-9:
                    new_violated.append((lag, corr_new[idx]))

            if len(new_violated) == 0:
                break

            new_violated.sort(key=lambda x: -x[1])
            added = 0
            for lag, _ in new_violated:
                if len(active_lags) >= 25:
                    break
                if lag not in active_lags:
                    active_lags.append(lag)
                    added += 1
            if added == 0:
                break

            h = h_new
            c5 = c5_new

        except Exception:
            break

    return best_h, best_c5


def smooth_slsqp_step(h_init, n, beta, max_iter=5000):
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_wrapped(x, dx2, beta2, ts):
        corr = np.correlate(x, 1.0 - x, mode='full')
        pen = 50.0 * (x.sum() - ts) ** 2
        return smooth_max(corr, beta2) * dx2 + pen

    def con_eq(x, ts):
        return x.sum() - ts

    def con_eq_jac(x, ts):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_wrapped, h_init, args=(dx, beta, target_sum),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)


def comb_initialization(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return project_feasible(h, n / 2.0)


def block_initialization(n, nblocks=4):
    h = np.zeros(n)
    block_size = n // nblocks
    for i in range(0, nblocks, 2):
        start = i * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    total = h.sum()
    if total > 0:
        h = h * (n / 2.0) / total
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = project_feasible(h_init, n0 / 2.0)
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: Quick smooth optimization at original resolution
    if remaining() > 80:
        for beta in [50, 300, 1500]:
            if remaining() < 30:
                break
            h_new, c5_new = smooth_slsqp_step(best_h, best_n, beta, max_iter=3000)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 2: Active-set SLSQP at original resolution
    if remaining() > 60:
        h_new, c5_new = active_set_slsqp(best_h, best_n, max_outer=20, max_inner_iter=5000, time_limit=min(remaining() - 30, 120))
        if c5_new < best_c5:
            best_h, best_c5, best_n = h_new, c5_new, best_n

    # Phase 3: Resolution expansion
    resolutions = [90, 120, 160, 220, 300, 400]
    for n_try in resolutions:
        if remaining() < 50:
            break
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, n_try / 2.0)

        # Quick smooth step first
        if remaining() > 80:
            h_t, c5_t = smooth_slsqp_step(h_t, n_try, 200, max_iter=2000)
            if c5_t < best_c5:
                best_h, best_c5, best_n = h_t, c5_t, n_try

        # Active-set refinement
        if remaining() > 40:
            tl = min(remaining() - 20, 100)
            h_new, c5_new = active_set_slsqp(best_h if best_n == n_try else h_t,
                                             n_try, max_outer=20, max_inner_iter=4000,
                                             time_limit=tl)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, n_try

    # Phase 4: Try alternative initializations at the best resolution
    if remaining() > 80:
        for init_func in [comb_initialization, block_initialization]:
            if remaining() < 40:
                break
            h_alt = init_func(best_n)
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5 + 0.05:
                h_alt, c5_alt = smooth_slsqp_step(h_alt, best_n, 100, max_iter=2000)
                if remaining() > 30:
                    h_alt, c5_alt = active_set_slsqp(h_alt, best_n, max_outer=10,
                                                     max_inner_iter=3000, time_limit=30)
                if c5_alt < best_c5:
                    best_h, best_c5 = h_alt, c5_alt

    # Phase 5: Random perturbations at best resolution
    if remaining() > 60:
        for attempt in range(8):
            if remaining() < 25:
                break
            scale = [0.08, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005][min(attempt, 7)]
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n / 2.0)
            h_new, c5_new = active_set_slsqp(h_pert, best_n, max_outer=15,
                                             max_inner_iter=3000, time_limit=min(remaining() - 15, 50))
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Final high-precision polish
    if remaining() > 25:
        h_new, c5_new = active_set_slsqp(best_h, best_n, max_outer=30,
                                         max_inner_iter=10000, time_limit=min(remaining() - 10, 80))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final feasibility check
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-15:
        mask = (best_h < 1.0) if diff > 0 else (best_h > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
