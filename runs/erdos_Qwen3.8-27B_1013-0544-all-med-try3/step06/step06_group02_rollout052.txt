    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return val

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            h = y_new.copy()
        else:
            lr *= 0.7
            h = y.copy()  # revert
            y = y.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

Now, for the pair move h_i += δ, h_j -= δ:
Δc_m = δ * (∂c_m/∂h_i - ∂c_m/∂h_j)

Let me define:
φ_j(m) = ∂c_m/∂h_j / dx = [1-h[j+shift_m]] * I(0 ≤ j+shift_m ≤ n-1) - h[j-shift_m] * I(0 ≤ j-shift_m ≤ n-1)

where shift_m = m - (n-1).

Then Δc_m = δ * dx * (φ_i(m) - φ_j(m))

So c_m(δ) = c_m(0) + δ * dx * (φ_i(m) - φ_j(m))

And the objective is: f(δ) = max_m c_m(δ) = max_m (c_m(0) + δ * dx * (φ_i(m) - φ_j(m)))

This is indeed convex in δ (max of linear functions).

To find the minimum of f(δ) over the feasible range:
- Feasible: h_i + δ ∈ [0,1] and h_j - δ ∈ [0,1]
  So δ ∈ [-h_i, 1-h_i] ∩ [h_j-1, h_j] = [max(-h_i, h_j-1), min(1-h_i, h_j)]

For the minimization, since f is convex piecewise linear:
- If f is monotonically decreasing on [lo, hi], minimum is at hi
- If f is monotonically increasing, minimum is at lo
- Otherwise, minimum is at a breakpoint

I'll use a simple approach: evaluate f at lo, hi, and a few intermediate points, then refine.

Actually, the simplest correct approach for a convex piecewise linear function: the minimum is where the "effective slope" changes sign. The slope of f at δ is:
slope(δ) = dx * sum_{m active} w_m * (φ_i(m) - φ_j(m))

where "active" means c_m(δ) = max, and w_m are the barycentric coordinates.

For a practical implementation, I'll just do a ternary search (50 iterations) which will give the minimum to high precision.

Wait, ternary search works for unimodal functions. Since f is convex, it is unimodal, so ternary search works!

Let me implement this:

```python
def find_optimal_delta(c, d, lo, hi, n_iter=50):
    """Minimize max(c + delta * d) over delta in [lo, hi].
    c and d are arrays of length 2n-1.
    Returns the optimal delta.
    """
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c + m1 * d)
        f2 = np.max(c + m2 * d)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

shift_m = m - (n-1), which ranges from -(n-1) to (n-1).

φ_j(m) = [1-h[j+shift_m]] * I(0 ≤ j+shift_m ≤ n-1) - h[j-shift_m] * I(0 ≤ j-shift_m ≤ n-1)

Let s = shift_m.
Term 1: (1-h)[j+s] if 0 ≤ j+s ≤ n-1, i.e., -s ≤ j ≤ n-1-s

```python
def compute_phi_matrix(h, n):
    """Compute phi[m, j] = d(correlate[m]) / d(h[j]) / dx for all m, j."""
    # correlate[m] = sum_i h[i] * (1-h[i + m - (n-1)])
    # d(correlate[m])/d(h[j]) = (1-h[j+s])*I(0≤j+s≤n-1) - h[j-s]*I(0≤j-s≤n-1)
    # where s = m - (n-1)
    
    phi = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        s = m - (n - 1)
        # Term 1: (1-h[j+s]) for j where 0 ≤ j+s ≤ n-1
        if s >= 0:
            # j+s ≥ 0 always (j ≥ 0), need j+s ≤ n-1, i.e., j ≤ n-1-s
            phi[m, 0:n-s] += (1.0 - h[s:n])
        else:
            # j+s ≥ 0 means j ≥ -s
            phi[m, -s:n] += (1.0 - h[0:n+s])
        
        # Term 2: -h[j-s] for j where 0 ≤ j-s ≤ n-1
        if s >= 0:
            # j-s ≥ 0 means j ≥ s, and j-s ≤ n-1 means j ≤ n-1+s (always since j ≤ n-1)
            phi[m, s:n] -= h[0:n-s]
        else:
            # j-s ≤ n-1 always (j ≤ n-1, -s > 0), and j-s ≥ 0 means j ≥ s (but s < 0, so always)
            phi[m, 0:n] -= h[-s:n]
    
    return phi
```
