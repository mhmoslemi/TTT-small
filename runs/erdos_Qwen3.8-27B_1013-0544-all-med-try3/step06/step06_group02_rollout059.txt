            break
        improved = False
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            old_val = h[i]
            
            # Compute a_p for all p
            a = np.zeros(2*n - 1)
            for idx, p in enumerate(p_vals):
                if p == 0:
                    continue
                a_p = 0.0
                if 0 <= i - p < n:
                    a_p += (1.0 - h[i - p])
                if 0 <= i + p < n:
                    a_p -= h[i + p]
                a[idx] = a_p

            # For p = 0: c_0(t) = d0 + t - t^2 where d0 = c_full[n-1] - h[i] + h[i]^2
            idx0 = n - 1
            d0 = c_full[idx0] - old_val + old_val**2
            
            def obj_1d(t):
                val0 = d0 + t - t*t
                vals = a * t + (c_full - a * old_val)
                vals[idx0] = val0
                return np.max(vals)
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                   options={'xatol': 1e-10})
            new_val = res.x
            
            if abs(new_val - old_val) > 1e-12:
                delta = new_val - old_val
                h[i] = new_val
                # Update c_full
                for idx, p in enumerate(p_vals):
                    if p == 0:
                        c_full[idx] = d0 + new_val - new_val**2
                    else:
                        c_full[idx] += a[idx] * delta
                
                new_c = np.max(c_full) * (2.0/n)
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    best_h = h.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c
```

def compute_a(h, n, i):
    """Compute a_p = ∂c_p/∂h[i] for all p"""
    a = np.zeros(2*n - 1)
    # p ranges from -(n-1) to n-1, indexed 0 to 2n-2
    # a[idx] corresponds to p = idx - (n-1)
    
    # First term: (1 - h[i-p]) for valid i-p, i.e., 0 ≤ i-p < n
    # p = i - j where j = i-p, j ∈ [0, n-1]
    # So p ranges from i-(n-1) to i
    # In index terms: idx = p + (n-1), so idx ranges from i-(n-1)+(n-1)=i to i+(n-1)
    j_start = 0
    j_end = n
    p_start = i - (j_end - 1)  # = i - n + 1
    p_end = i - j_start  # = i
    # Clip to valid p range [-(n-1), n-1]
    p_start = max(p_start, -(n-1))
    p_end = min(p_end, n-1)
    if p_start <= p_end:
        idx_start = p_start + (n-1)
        idx_end = p_end + (n-1) + 1
        j_vals = i - np.arange(p_start, p_end + 1)
        a[idx_start:idx_end] += (1.0 - h[j_vals])
    
    # Second term: -h[i+p] for valid i+p, i.e., 0 ≤ i+p < n
    # p ranges from -i to n-1-i
    p_start2 = max(-i, -(n-1))
    p_end2 = min(n-1-i, n-1)
    if p_start2 <= p_end2:
        idx_start2 = p_start2 + (n-1)
        idx_end2 = p_end2 + (n-1) + 1
        j_vals2 = i + np.arange(p_start2, p_end2 + 1)
        a[idx_start2:idx_end2] -= h[j_vals2]
    
    # Set a for p=0 (idx = n-1) to 0 (handled separately)
    a[n-1] = 0.0
    return a
```

The 1D problem is: minimize max(f_0(t), f_1(t), ..., f_{2n-2}(t)) over t ∈ [0,1]
where f_0(t) = d0 + t - t^2 (concave) and f_p(t) = a_p*t + b_p (linear/affine).

The maximum of these functions is a piecewise function. Its minimum occurs at:
- A boundary (t=0 or t=1), or
- A point where two or more of the functions are equal and the derivative of the max changes from negative to positive

For the linear functions, the upper envelope is a convex piecewise linear function. Adding the concave quadratic makes it more complex, but the minimum of the overall max should still occur at a "vertex" of the upper envelope or at a point where the quadratic is tangent to a line.

A practical approach: use `scipy.optimize.minimize_scalar` with `method='bounded'` which uses Brent's method. This should be very efficient for 1D problems.

- Adding the concave quadratic: the min of max(quadratic, lines) is where either the quadratic is maximal (and we minimize it subject to being ≥ all lines) or one of the lines is maximal (and we minimize the max of lines subject to being ≥ the quadratic)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_weight(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weight(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):

            continue

            grad[0:n - p] += wm * (-h[p:n])

            grad[kk:n] += wm * (-h[0:n - kk])

    val = smooth_max_val(c * (2.0 / n), alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    g += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return g

    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                for j in range(max(0, i-80), min(n, i+81)):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d; h[j] += d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            h = project_feasible(h, n)
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_cur = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_cur = 300
    
    best_h = project_feasible(h_init, n_cur)
    best_c = compute_c5(best_h, n_cur)
    
    # Phase 1: L-BFGS-B at current resolution
    if time.time() < deadline - 300:
        t1 = min(time.time() + 180, deadline - 300)
        best_h, best_c = lbfgs_phase(best_h, n_cur, t1, max_iters=200)
    
    # Phase 2: Coordinate descent at current resolution
    if time.time() < deadline - 200:
        t2 = min(time.time() + 200, deadline - 200)
        h_co, c_co = fast_coordinate_descent(best_h, n_cur, t2, max_passes=80)
        if c_co < best_c:
            best_c = c_co
            best_h = h_co
    
    # Phase 3: L-BFGS-B again (from coordinate descent result)
    if time.time() < deadline - 150:
        t3 = min(time.time() + 100, deadline - 150)
        h_l, c_l = lbfgs_phase(best_h, n_cur, t3, max_iters=150)
        if c_l < best_c:
            best_c = c_l
            best_h = h_l
    
    # Phase 4: Coordinate descent again
    if time.time() < deadline - 100:
        t4 = min(time.time() + 120, deadline - 100)
        h_co2, c_co2 = fast_coordinate_descent(best_h, n_cur, t4, max_passes=60)
        if c_co2 < best_c:
            best_c = c_co2
            best_h = h_co2
    
    # Phase 5: Try higher resolution
    for n_try in [400, 500, 600]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        alloc = min(120, deadline - time.time() - 60)
        if alloc < 40:
            break
        
        # L-BFGS at new resolution
        t5 = min(time.time() + alloc * 0.4, deadline - 60)
        h_t, c_t = lbfgs_phase(h_interp, n_try, t5, max_iters=100)
        
        # Coordinate descent at new resolution
        if time.time() < deadline - 40:
            t6 = min(time.time() + alloc * 0.35, deadline - 40)
            h_t2, c_t2 = fast_coordinate_descent(h_t, n_try, t6, max_passes=40)
            if c_t2 < c_t:
                c_t = c_t2
                h_t = h_t2
        
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            n_cur = n_try
    
    # Phase 6: SLSQP fine-tuning
    if time.time() < deadline - 30:
        t7 = min(time.time() + 60, deadline - 30)
        h_s, c_s = slsqp_phase(best_h, n_cur, t7, alpha=1000000, maxiter=800)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    
    # Phase 7: Final coordinate descent + pair refine
    if time.time() < deadline - 15:
        t8 = min(time.time() + 30, deadline - 15)
        h_f, c_f = fast_coordinate_descent(best_h, n_cur, t8, max_passes=30)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    if time.time() < deadline - 5:
        t9 = min(time.time() + 15, deadline - 5)
        h_p, c_p = pair_refine(best_h, n_cur, t9)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    
    return best_h, best_c, n_cur
