For a fixed h[i], the correlation at lag k involves terms like h[i]*(1-h[i+k]) or h[i+k]*(1-h[i]). So changing h[i] affects correlations at lags k = 0, ±k₁, ±k₂, ... where k₁, k₂, ... are the shifts that involve index i.

The optimal h[i] (for fixed others) would minimize the maximum of these affected correlations. This is a 1D minimax problem that can be solved by bisection or by finding the intersection points.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_arr(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _slsqp_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_arr(c, alpha)

def _slsqp_obj_top(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_arr(c[top_idx], alpha)

def slsqp_opt(h, n, alpha, top_m=None, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        if top_m is not None:
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]
            res = minimize(_slsqp_obj_top, h, args=(n, alpha, top_idx),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
        else:
            res = minimize(_slsqp_obj_full, h, args=(n, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def outer_approx_fast(h_curr, n, trust, num_lags=30):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-num_lags:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in top_idx:
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_single(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Mid-game: more outer approx iterations on best
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        h_curr = best_h.copy()
        trust = 0.05
        for it in range(30):
            if time.time() > deadline - 40:
                break
            h_new = outer_approx_fast(h_curr, n, trust, num_lags=20)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            step = np.max(np.abs(h_new - h_curr))
            if step < 1e-9:
                break
            h_curr = h_new
            trust = max(0.002, trust * 0.8)
    
    # Restart phase: perturb and re-optimize
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.008 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 30) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final intensive refinement
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # SLSQP with very high alpha
        if time.time() < deadline - 8:
            h_new = slsqp_opt(best_h, n, 500000, top_m=3, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Very fine local search
        if time.time() < deadline - 3:
            h_new, c_new = fine_local_search(best_h, n, deadline - 1, top_m=3)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
