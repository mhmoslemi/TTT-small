def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

OK so the problem is genuinely non-trivial. Let me focus on the algorithm.

Given the small gap to the target (0.380866 vs 0.3808), I think the key improvements are:
1. Better binary search (faster, allowing more sweeps)
2. Possibly trying different resolutions
3. Better continuous optimization

Let me write a clean, efficient implementation.

One important optimization: in the vectorized binary search, instead of computing the full (nz, no, 2n-1) array, I can note that:

cand_max[a,b] = max_k (A[a,k] + B[b,k] + corr[k])

This can be rewritten as: for each k, define C_k = corr[k]. Then:
cand_max[a,b] = max_k (A[a,k] + B[b,k] + C_k)

If I define D_k = B[b,k] + C_k for each b, then cand_max[a,b] = max_k (A[a,k] + D_k[b]).

This is still O(nz * no * (2n-1)) in the worst case, but I can potentially use the fact that A and B are "sparse" (many zeros) to speed things up.

Actually, for binary h, A[a,k] is non-zero only for k in a range of length at most 2*min(i, n-1-i) + 1 (the range of shifts where i±m is in bounds). Similarly for B. So most entries are 0.

But with numpy, the vectorized approach should be fast enough. Let me just go with it.

Let me also add a "simulated annealing" component: with small probability, accept a swap that doesn't improve (or even worsens) the objective, to escape local minima.

import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

            s = h.sum()

def slsqp_smooth(h, n, alpha, maxiter, deadline):

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def build_A(zeros_idx, h, n, m_range):
    nz = len(zeros_idx)
    A = np.zeros((nz, 2 * n - 1))
    for a in range(nz):
        i = int(zeros_idx[a])
        lo = max(-i, -(n - 1))
        hi = min(n - 1 - i, n - 1)
        if lo <= hi:
            A[a, lo + n - 1:hi + n] = 1.0 - h[i + m_range[lo:hi + 1]]
        lo = max(i - (n - 1), -(n - 1))
        hi = min(i, n - 1)
        if lo <= hi:
            A[a, lo + n - 1:hi + n] -= h[i - m_range[lo:hi + 1]]
    return A

def build_B(ones_idx, h, n, m_range):
    no = len(ones_idx)
    B = np.zeros((no, 2 * n - 1))
    for b in range(no):
        j = int(ones_idx[b])
        lo = max(-j, -(n - 1))
        hi = min(n - 1 - j, n - 1)
        if lo <= hi:
            B[b, lo + n - 1:hi + n] -= (1.0 - h[j + m_range[lo:hi + 1]])
        lo = max(j - (n - 1), -(n - 1))
        hi = min(j, n - 1)
        if lo <= hi:
            B[b, lo + n - 1:hi + n] += h[j - m_range[lo:hi + 1]]
    return B

def vec_binary_search(h_bin, n, deadline, max_sweeps=5000, n_restarts=5, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.copy()
    best_c = np.inf
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    if cur_max * dx < best_c:
        best_c = cur_max * dx
        best_h = h.copy()
    rng = np.random.RandomState(seed)

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            h = best_h.copy()
            n_pert = rng.randint(3, 8) + restart * 2
            for _ in range(n_pert):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = corr.max()
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break

            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            nz, no = len(zeros_idx), len(ones_idx)
            if nz == 0 or no == 0:
                break

            A = build_A(zeros_idx, h, n, m_range)
            B = build_B(ones_idx, h, n, m_range)

            D = A[:, None, :] + B[None, :, :] + corr[None, None, :]
            cand_max = D.max(axis=2)
            flat_idx = cand_max.argmin()
            a_best, b_best = np.unravel_index(flat_idx, cand_max.shape)
            i_best = int(zeros_idx[a_best])
            j_best = int(ones_idx[b_best])

            h[i_best] = 1.0
            h[j_best] = 0.0
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = new_corr.max()

            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                if cur_max * dx < best_c:
                    best_c = cur_max * dx
                    best_h = h.copy()
            else:
                h[i_best] = 0.0
                h[j_best] = 1.0
                break

    return best_h, best_c
