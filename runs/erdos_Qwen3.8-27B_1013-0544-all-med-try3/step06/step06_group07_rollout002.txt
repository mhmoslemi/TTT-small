```
def grad_weights_f(f, w):
    m=f.size; N=2*m; h=np.empty(N); g=np.empty(N)
    h[:m]=f; h[m:]=1-f; g=1-h
    grad_h=np.zeros(N)
    wm = w[N-1]
    if wm !=0: grad_h += wm*(g-h)
    for t in range(1,N):
       a = w[N-1-t]
       b = w[N-1+t]
       if a==0 and b==0: continue
       if a:
          grad_h[:N-t] += a*g[t:]
          grad_h[t:] -= a*h[:N-t]
       if b:
          grad_h[t:] += b*g[:N-t]
          grad_h[:N-t] -= b*h[t:]
    return (grad_h[:m] - grad_h[m:]) / m
```

```
def binary_hill_climb(f, deadline, iters=20000):
    m = f.size
    best = (f > 0.5).astype(float)
    best_c = c5_from_f(best)
    cur = best.copy(); cur_c=best_c
    temp=0.02
    for it in range(iters):
       if time.time()>deadline: break
       i=int(np.random.randint(m))
       trial=cur.copy(); trial[i]=1.0-trial[i]
       c=c5_from_f(trial)
       if c < cur_c:
          cur=trial; cur_c=c
          if c < best_c: best=trial.copy(); best_c=c
       else:
          if temp > 1e-6 and np.random.rand() < np.exp(-(c-cur_c)/temp):
             cur=trial; cur_c=c
       if it % 500 == 499: temp *= 0.97
    # local one-flip pass multiple times
    for pass in range(3):
       improved=False
       for i in range(m):
          if time.time()>deadline: break
          trial=best.copy(); trial[i]=1-trial[i]
          c=c5_from_f(trial)
          if c < best_c - 1e-15:
             best=trial; best_c=c; improved=True
       if not improved: break
    return best,best_c
```

```python
            idxs = np.where(h < 1.0)[0]
            for j in idxs:
                add = min(1.0 - h[j], target - float(h.sum()))
                if add <= 0.0:
                    break
                h[j] += add
        else:
            idxs = np.where(h > 0.0)[0]
            for j in idxs:
                sub = min(h[j], float(h.sum()) - target)
                if sub <= 0.0:
                    break
                h[j] -= sub
    return h


def make_hg_f(f):
    f = np.asarray(f, dtype=float)
    m = int(f.size)
    N = 2 * m
    h = np.empty(N, dtype=float)
    g = np.empty(N, dtype=float)
    h[:m] = f
    h[m:] = 1.0 - f
    g[:m] = 1.0 - f
    g[m:] = f
    return h, g


def f_to_h(f):
    f = np.asarray(f, dtype=float)
    h = np.empty(2 * f.size, dtype=float)
    h[:f.size] = f
    h[f.size:] = 1.0 - f
    return h


def c5_from_f(f):
    f = np.asarray(f, dtype=float)
    m = int(f.size)
    if m == 0:
        return float('inf')
    h, g = make_hg_f(f)
    corr = np.correlate(h, g, mode='full')
    val = float(np.max(corr) / float(m))
    if not np.isfinite(val):
        return float('inf')
    return val


def softmax_weights_c(corr, dx, beta):
    corr = np.asarray(corr, dtype=float)
    c = corr * float(dx)
    M = float(np.max(c))
    if not np.isfinite(M):
        c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
        M = float(np.max(c)) if c.size else 0.0
    beta = float(beta)
    if beta <= 0.0:
        w = np.zeros_like(cor)
        if corr.size:
            w[int(np.argmax(corr))] = 1.0
        return w, M
    shift = beta * (c - M)
    shift = np.clip(shift, -700.0, 0.0)
    e = np.exp(shift)
    S = float(np.sum(e))
    if S <= 0.0 or not np.isfinite(S):
        e[int(np.argmax(c))] = 1.0
        S = 1.0
    w = e / S
    val = M + float(np.log(S)) / beta
    return w, val


def grad_weights_h(h, g, w):
    h = np.asarray(h, dtype=float)
    g = np.asarray(g, dtype=float)
    w = np.asarray(w, dtype=float)
    N = int(h.size)
    grad = np.zeros(N, dtype=float)
    if N == 0:
        return grad
    center = N - 1
    c0 = float(w[center])
    if c0 != 0.0:
        grad += c0 * (g - h)

    if _CORR_NEG_FIRST:
        for t in range(1, N):
            a = float(w[center - t])
            b = float(w[center + t])
            if a == 0.0 and b == 0.0:
                continue
            Lt = N - t
            if a != 0.0:
                grad[:Lt] += a * g[t:]
                grad[t:] -= a * h[:Lt]
            if b != 0.0:
                grad[t:] += b * g[:Lt]
                grad[:Lt] -= b * h[t:]
    else:
        for t in range(1, N):
            a = float(w[center + t])
            b = float(w[center - t])
            if a == 0.0 and b == 0.0:
                continue
            Lt = N - t
            if a != 0.0:
                grad[:Lt] += a * g[t:]
                grad[t:] -= a * h[:Lt]
            if b != 0.0:
                grad[t:] += b * g[:Lt]
                grad[:Lt] -= b * h[t:]
    return grad


def smooth_f_obj(f, beta):
    f = np.asarray(f, dtype=float)
    m = int(f.size)
    if m == 0:
        return float('inf')
    h, g = make_hg_f(f)
    corr = np.correlate(h, g, mode='full')
    _, val = softmax_weights_c(corr, 1.0 / float(m), beta)
    return float(val)


def smooth_f_jac(f, beta):
    f = np.asarray(f, dtype=float)
    m = int(f.size)
    if m == 0:
        return np.zeros_like(f)
    h, g = make_hg_f(f)
    corr = np.correlate(h, g, mode='full')
    w, _ = softmax_weights_c(corr, 1.0 / float(m), beta)
    grad_h = grad_weights_h(h, g, w)
    return (grad_h[:m] - grad_h[m:]) / float(m)


def full_smooth_obj(h, beta):
    h = np.asarray(h, dtype=float)
    N = int(h.size)
    if N == 0:
        return float('inf')
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    _, val = softmax_weights_c(corr, 2.0 / float(N), beta)
    return float(val)


def full_smooth_jac(h, beta):
    h = np.asarray(h, dtype=float)
    N = int(h.size)
    if N == 0:
        return np.zeros_like(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    w, _ = softmax_weights_c(corr, 2.0 / float(N), beta)
    grad = grad_weights_h(h, g, w)
    return (2.0 / float(N)) * grad


def sum_constraint(h, target):
    return float(np.sum(h) - target)


def sum_constraint_jac(h, target):
    return np.ones_like(np.asarray(h, dtype=float))


def resample_f(f, m_new):
    f = np.asarray(f, dtype=float)
    m_old = int(f.size)
    m_new = int(m_new)
    if m_new <= 0:
        return np.array([], dtype=float)
    if m_old == 0:
        return np.full(m_new, 0.5, dtype=float)
    if m_old == m_new:
        return f.copy()
    if m_old == 1:
        return np.full(m_new, float(f[0]), dtype=float)
    x_old = np.linspace(0.0, 1.0, m_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, m_new, endpoint=False)
    y = np.interp(x_new, x_old, f, left=float(f[0]), right=float(f[-1]))
    return np.clip(y, 0.0, 1.0)


def default_f(m):
    m = max(2, int(m))
    x = np.linspace(0.0, 1.0, m, endpoint=False)
    f = 0.5 + 0.45 * np.sin(2.0 * np.pi * x)
    return np.clip(f, 0.0, 1.0)


def anti_project_to_f(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n < 2:
        return np.clip(h, 0.0, 1.0)
    m = n // 2
    if n % 2 == 0:
        f = 0.5 * (h[:m] + (1.0 - h[m:2 * m]))
    else:
        f = f if False else h[:m]
    return np.clip(f, 0.0, 1.0)


def active_grad_polish(f, deadline, lr0=0.012, eps=1e-5, max_iter=None):
    f = np.clip(np.asarray(f, dtype=float), 0.0, 1.0).copy()
    m = int(f.size)
    if m <= 0:
        return f, float('inf')
    if max_iter is None:
        if m <= 60:
            max_iter = 30000
        elif m <= 160:
            max_iter = 10000
        else:
            max_iter = 4000

    best_f = f.copy()
    best_c = c5_from_f(f)
    lr = float(lr0)
    stall = 0

    for _ in range(int(max_iter)):
        if time.time() > float(deadline):
            break

        h, g = make_hg_f(f)
        corr = np.correlate(h, g, mode='full')
        C = corr / float(m)
        cmax = float(np.max(C))
        if cmax < best_c:
            best_c = cmax
            best_f = f.copy()

        active = C > (cmax - float(eps))
        if not np.any(active):
            idx = int(np.argmax(C))
            active[idx] = True
        cnt = float(np.count_nonzero(active))
        w = np.zeros_like(C, dtype=float)
        w[active] = 1.0 / cnt

        grad_h = grad_weights_h(h, g, w)
        grad_f = (grad_h[:m] - grad_h[m:]) / float(m)

        trial = np.clip(f - lr * grad_f, 0.0, 1.0)
        c_trial = c5_from_f(trial)

        if c_trial < cmax - 1e-15:
            f = trial
            lr = min(0.05, lr * 1.08)
            stall = 0
        else:
            lr *= 0.6
            stall += 1
            if stall > 30 or lr < 1e-7:
                f = np.clip(best_f + 0.003 * np.random.randn(m), 0.0, 1.0)
                lr = float(lr0)
                stall = 0

    return best_f, best_c


def optimize_f(f, deadline):
    f = np.clip(np.asarray(f, dtype=float), 0.0, 1.0).copy()
    m = int(f.size)
    if m <= 0:
        return f, float('inf')

    best_f = f.copy()
    best_c = c5_from_f(f)
    bounds = Bounds(np.zeros(m, dtype=float), np.ones(m, dtype=float))

    if m <= 80:
        betas = [100, 300, 1000, 3000, 10000]
        scales = [0.0, 0.02, 0.012, 0.006, 0.003]
    elif m <= 160:
        betas = [100, 300, 1000, 3000]
        scales = [0.0, 0.012, 0.006]
    else:
        betas = [100, 300, 1000]
        scales = [0.0, 0.008, 0.004]

    if m <= 40:
        base_iter = 500
    elif m <= 80:
        base_iter = 300
    elif m <= 160:
        base_iter = 180
    else:
        base_iter = 90

    for beta in betas:
        if time.time() > float(deadline):
            break
        for scale in scales:
            if time.time() > float(deadline):
                break
            x0 = best_f.copy()
            if scale > 0.0:
                x0 = x0 + float(scale) * np.random.randn(m)
                np.clip(x0, 0.0, 1.0, out=x0)

            maxiter = base_iter
            if time.time() > float(deadline) - 30.0:
                maxiter = min(maxiter, 40)

            try:
                res = minimize(
                    smooth_f_obj,
                    x0,
                    jac=smooth_f_jac,
                    args=(float(beta),),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={
                        'maxiter': int(maxiter),
                        'ftol': 1e-12,
                        'gtol': 1e-6,
                        'maxls': 50,
                        'disp': False
                    }
                )
                x1 = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
                c1 = c5_from_f(x1)
                if np.isfinite(c1) and c1 < best_c:
                    best_f = x1
                    best_c = c1
            except Exception:
                pass

    if time.time() + 5.0 < float(deadline):
        if m <= 80:
            pol_dead = min(float(deadline), time.time() + 60.0)
        else:
            pol_dead = min(float(deadline), time.time() + 30.0)
        f_pol, c_pol = active_grad_polish(best_f, pol_dead)
        if np.isfinite(c_pol) and c_pol < best_c:
            best_f = f_pol
            best_c = c_pol

    for thr in (0.30, 0.40, 0.50, 0.60, 0.70):
        fthr = (best_f > float(thr)).astype(float)
        if not np.any(fthr) and np.all(fthr == 0.0):
            pass
        cthr = c5_from_f(fthr)
        if np.isfinite(cthr) and cthr < best_c:
            best_f = fthr
            best_c = cthr

    return best_f, best_c


def binary_hill_climb(f, deadline, iters=20000):
    f = np.asarray(f, dtype=float)
    m = int(f.size)
    if m <= 0:
        return np.array([], dtype=float), float('inf')

    cur = (f > 0.5).astype(float)
    best_f = cur.copy()
    cur_c = c5_from_f(cur)
    best_c = cur_c
    temp = 0.02

    for it in range(int(iters)):
        if time.time() > float(deadline):
            break
        i = int(np.random.randint(m))
        trial = cur.copy()
        trial[i] = 1.0 - trial[i]
        c = c5_from_f(trial)

        if c < cur_c:
            cur = trial
            cur_c = c
            if c < best_c:
                best_f = trial.copy()
                best_c = c
        else:
            if temp > 1e-8:
                try:
                    p = float(np.exp(-(c - cur_c) / temp))
                except Exception:
                    p = 0.0
                if np.random.rand() < p:
                    cur = trial
                    cur_c = c

        if it % 500 == 499:
            temp *= 0.97

    for _ in range(3):
        improved = False
        for i in range(m):
            if time.time() > float(deadline):
                break
            trial = best_f.copy()
            trial[i] = 1.0 - trial[i]
            c = c5_from_f(trial)
            if c < best_c - 1e-15:
                best_f = trial
                best_c = c
                improved = True
        if not improved:
            break

    return best_f, best_c


def refine_full_slsqp(h0, deadline):
    h0 = np.asarray(h0, dtype=float).copy()
    N = int(h0.size)
    if N <= 2:
        h0 = project_feasible(h0, N / 2.0)
        return h0, compute_c5(h0)

    target = float(N) / 2.0
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_c = compute_c5(h)

    if N > 120:
        return best_h, best_c

    bounds = Bounds(np.zeros(N, dtype=float), np.ones(N, dtype=float))
    cons = [
        {
            'type': 'eq',
            'fun': sum_constraint,
            'jac': sum_constraint_jac,
            'args': (target,)
        }
    ]

    if N <= 80:
        betas = [100, 300, 1000, 3000, 10000]
        scales = [0.0, 0.01, 0.005]
        maxiter = 400
    else:
        betas = [100, 300, 1000, 3000]
        scales = [0.0, 0.008, 0.004]
        maxiter = 160

    for beta in betas:
        if time.time() > float(deadline):
            break
        for scale in scales:
            if time.time() > float(deadline):
                break
            x0 = best_h.copy()
            if scale > 0.0:
                x0 = x0 + float(scale) * np.random.randn(N)
                x0 = project_feasible(x0, target)

            mi = maxiter
            if time.time() > float(deadline) - 30.0:
                mi = min(mi, 50)

            try:
                res = minimize(
                    full_smooth_obj,
                    x0,
                    jac=full_smooth_jac,
                    args=(float(beta),),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={
                        'maxiter': int(mi),
                        'ftol': 1e-12,
                        'gtol': 1e-7
                    }
                )
                x1 = project_feasible(np.asarray(res.x, dtype=float), target)
                c1 = compute_c5(x1)
                if np.isfinite(c1) and c1 < best_c:
                    best_h = x1
                    best_c = c1
            except Exception:
                pass

    best_h = project_feasible(best_h, best_h.size / 2.0)
    best_c = compute_c5(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(int(seed))
    calibrate_corr_order()

    t0 = time.time()
    total = float(budget_s)
    if total < 20.0:
        total = 20.0
    deadline = t0 + max(5.0, total - 5.0)

    best_h = None
    best_c = float('inf')
    best_f = None
    best_f_c = float('inf')
    parent_h = None

    if 'initial_h_values' in globals():
        iv = initial_h_values
        if iv is not None:
            parent_h = np.asarray(iv, dtype=float).copy()
            if parent_h.size > 0:
                cp = compute_c5(parent_h)
                if np.isfinite(cp) and cp < best_c:
                    best_h = parent_h.copy()
                    best_c = cp

    if best_h is None:
        f_def = default_f(35)
        h_def = f_to_h(f_def)
        best_h = h_def
        best_c = compute_c5(h_def)
        best_f = f_def.copy()
        best_f_c = float(c5_from_f(f_def))

    if parent_h is not None:
        f_start = anti_project_to_f(parent_h)
    elif best_f is not None:
        f_start = best_f.copy()
    else:
        f_start = default_f(35)

    if f_start.size < 8:
        f_start = default_f(35)

    c_start = c5_from_f(f_start)
    if np.isfinite(c_start):
        if c_start < best_c:
            best_h = f_to_h(f_start)
            best_c = c_start
        if best_f is None or c_start < best_f_c:
            best_f = f_start.copy()
            best_f_c = c_start
    else:
        f_start = default_f(35)
        c_start = c5_from_f(f_start)
        best_f = f_start.copy()
        best_f_c = c_start
        best_h = f_to_h(f_start)
        best_c = c_start

    m0 = int(f_start.size)

    if m0 >= 15:
        targets = [m0]
        for mult in (2, 3, 4, 6, 8):
            t = int(round(float(m0) * float(mult)))
            if t > targets[-1] and 0 < t <= 256:
                targets.append(t)
        if targets[-1] < 256:
            targets.append(256)
    else:
        targets = []
        for t in (32, 64, 128, 192, 256):
            if t > (targets[-1] if targets else 0):
                targets.append(t)
        if not targets:
            targets = [35]

    if total > 200.0:
        opt_budget = max(120.0, total - 150.0)
    else:
        opt_budget = total * 0.70
    n_targets = max(1, len(targets))
    sec_each = max(20.0, opt_budget / float(n_targets))

    shape_f = best_f if best_f is not None else f_start
    shape_c = best_f_c if best_f_c < float('inf') else c_start

    for mt in targets:
        if time.time() > deadline:
            break
        mt = int(mt)
        if mt <= 0:
            continue

        if shape_f is None or shape_f.size == 0:
            base_f = best_f
        elif shape_f.size < mt:
            base_f = shape_f
        else:
            base_f = best_f

        if base_f is None:
            base_f = best_f
        if base_f is None or base_f.size == 0:
            base_f = default_f(mt)

        if base_f.size == mt:
            start_f = base_f.copy()
        else:
            start_f = resample_f(base_f, mt)

        stage_dead = min(deadline, time.time() + sec_each)
        f_opt, c_opt = optimize_f(start_f, stage_dead)

        if np.isfinite(c_opt):
            if c_opt < best_f_c:
                best_f = f_opt.copy()
                best_f_c = c_opt
            h_opt = f_to_h(f_opt)
            c_h = c_opt
            if c_h < best_c:
                best_h = h_opt
                best_c = c_h

            if c_opt <= shape_c + 2e-5:
                shape_f = f_opt.copy()
                shape_c = c_opt

    if best_f is not None and best_f_c < best_c:
        best_h = f_to_h(best_f)
        best_c = best_f_c

    if best_f is not None and best_f.size <= 80 and time.time() + 30.0 <= deadline:
        bin_dead = min(deadline, time.time() + max(30.0, 0.05 * total))
        f_bin, c_bin = binary_hill_climb(best_f, bin_dead)
        if np.isfinite(c_bin):
            if c_bin < best_c:
                best_h = f_to_h(f_bin)
                best_c = c_bin
            if c_bin < best_f_c:
                best_f = f_bin
                best_f_c = c_bin

    if best_h is not None and best_h.size % 2 == 0 and best_h.size > 2:
        f_from_best = anti_project_to_f(best_h)
        c_from_best = c5_from_f(f_from_best)
        if np.isfinite(c_from_best):
            if c_from_best < best_c:
                best_h = f_to_h(f_from_best)
                best_c = c_from_best
            if best_f is None or c_from_best < best_f_c:
                best_f = f_from_best
                best_f_c = c_from_best

    full_candidates = []
    if parent_h is not None:
        full_candidates.append(parent_h)
    if best_f is not None and 2 * int(best_f.size) <= 120:
        full_candidates.append(f_to_h(best_f))
    if best_h is not None and best_h.size <= 120:
        full_candidates.append(best_h)
    if not full_candidates and best_f is not None:
        m_small = max(2, min(45, int(best_f.size)))
        full_candidates.append(f_to_h(resample_f(best_f, m_small)))

    used_full = []
    for h0 in full_candidates:
        if time.time() + 20.0 > deadline:
            break
        if h0 is None or h0.size < 4:
            continue
        seen = False
        for u in used_full:
            if u.size == h0.size and np.allclose(u, h0, rtol=1e-12, atol=1e-12):
                seen = True
                break
        if seen:
            continue
        used_full.append(np.asarray(h0, dtype=float).copy())

        full_dead = min(deadline, time.time() + max(45.0, 0.07 * total))
        h_ref, c_ref = refine_full_slsqp(h0, full_dead)
        if np.isfinite(c_ref) and c_ref < best_c:
            best_h = h_ref
            best_c = c_ref

        if best_h is not None and best_h.size % 2 == 0 and best_h.size > 2:
            f_new = anti_project_to_f(best_h)
            c_new = c5_from_f(f_new)
            if np.isfinite(c_new):
                if c_new < best_c:
                    best_h = f_to_h(f_new)
                    best_c = c_new
                if best_f is None or c_new < best_f_c:
                    best_f = f_new
                    best_f_c = c_new

    if best_f is not None and time.time() + 15.0 <= deadline:
        f_fin, c_fin = optimize_f(best_f, deadline)
        if np.isfinite(c_fin):
            if c_fin < best_c:
                best_h = f_to_h(f_fin)
                best_c = c_fin
            if c_fin < best_f_c:
                best_f = f_fin
                best_f_c = c_fin

    if best_f is not None and best_f_c < best_c:
        best_h = f_to_h(best_f)
        best_c = best_f_c

    if best_h is None or not np.isfinite(best_c):
        f_def = default_f(35)
        best_h = f_to_h(f_def)
        best_c = compute_c5(best_h)

    best_h = np.asarray(best_h, dtype=float)
    target_sum = float(best_h.size) / 2.0
    if abs(float(best_h.sum()) - target_sum) > 1e-8:
        best_h = project_feasible(best_h, target_sum)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c = compute_c5(best_h)
    if not np.isfinite(best_c):
        f_def = default_f(35)
        best_h = f_to_h(f_def)
        best_c = compute_c5(best_h)

    return best_h, float(best_c), int(best_h.size)
```
