   import numpy as np
   from scipy.optimize import minimize, linprog
   import time

   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)

   def compute_all_lags(h, n):
       dx = 2.0 / n
       return np.correlate(h, 1 - h, mode='full') * dx

   def project_feasible(h, n):
       target_sum = n / 2.0
       h = np.clip(h.copy(), 0.0, 1.0)
       for _ in range(500):
           diff = target_sum - h.sum()
           if abs(diff) < 1e-15:
               break
           if diff > 0:
               room = 1.0 - h
               tot = room.sum()
               if tot < 1e-14:
                   break
               h += (diff / tot) * room
           else:
               tot = h.sum()
               if tot < 1e-14:
                   break
               h += (diff / tot) * h
       return np.clip(h, 0.0, 1.0)

   def lag_gradient(h, n, lag):
       grad = np.zeros(n)
       m = abs(lag)
       if m == 0:
           grad[:] = 1.0 - 2.0 * h
       elif m >= n:
           return grad
       elif lag > 0:
           grad[m:] += (1.0 - h[:n-m])
           grad[:n-m] -= h[m:]
       else:
           grad[:n-m] += (1.0 - h[m:])
           grad[m:] -= h[:n-m]
       return grad

   def smooth_max_obj(h, n, alpha):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full') * dx
       m = c.max()
       shifted = alpha * (c - m)
       mx = shifted.max()
       if mx > 500:
           return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
       return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

   def active_max_obj(h, n, top_idx, alpha):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full') * dx
       active = c[top_idx]
       m = active.max()
       shifted = alpha * (active - m)
       mx = shifted.max()
       if mx > 500:
           return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
       return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

   def sum_constraint(h, n):
       return float(np.sum(h) - n / 2.0)

   def sum_constraint_jac(h, n):
       return np.ones(n)

   def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-15):
       bounds = [(0.0, 1.0)] * n
       cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
       try:
           res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
           return project_feasible(res.x, n)
       except:
           return h.copy()

   def active_set_slsqp(h, n, top_idx, alpha, maxiter=3000):
       bounds = [(0.0, 1.0)] * n
       cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
       try:
           res = minimize(active_max_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
           return project_feasible(res.x, n)
       except:
           return h.copy()

   def lp_direction(h, n, active_lags, deadline):
       """Find a descent direction using LP over active lags."""
       M = len(active_lags)
       G = np.array([lag_gradient(h, n, lag) for lag in active_lags])
       # Normalize rows
       row_norms = np.linalg.norm(G, axis=1)
       row_norms[row_norms < 1e-10] = 1.0
       G_norm = G / row_norms[:, None]
       
       # LP: min t s.t. G_norm @ d <= t, -1 <= d <= 1, sum(d) = 0
       # Variables: d_0, ..., d_{n-1}, t
       # c = [0, ..., 0, 1] (minimize t)
       c_vec = np.zeros(n + 1)
       c_vec[-1] = 1.0
       
       # Inequality constraints: G_norm @ d - t <= 0 for each row
       # And -1 <= d_i <= 1
       A_ub = np.zeros((M + 2 * n, n + 1))
       b_ub = np.zeros(M + 2 * n)
       
       for i in range(M):
           A_ub[i, :n] = G_norm[i]
           A_ub[i, n] = -1.0
           b_ub[i] = 0.0
       
       for i in range(n):
           A_ub[M + i, i] = 1.0
           b_ub[M + i] = 1.0
           A_ub[M + n + i, i] = -1.0
           b_ub[M + n + i] = 1.0  # -d_i <= 1 i.e. d_i >= -1
       
       # Equality: sum(d) = 0
       A_eq = np.zeros((1, n + 1))
       A_eq[0, :n] = 1.0
       b_eq = np.array([0.0])
       
       try:
           result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           method='highs')
           if result.status == 0 and result.fun < 0:
               return result.x[:n]
       except:
           pass
       return None

   def lp_search(h, n, deadline, max_rounds=500, top_m=5):
       """Iterative LP-based local search."""
       h_curr = h.copy()
       best_c = compute_c5(h_curr, n)
       
       for rnd in range(max_rounds):
           if time.time() > deadline:
               break
           
           c = compute_all_lags(h_curr, n)
           top_idx = np.argsort(c)[-top_m:]
           active_lags = top_idx - (n - 1)
           
           d = lp_direction(h_curr, n, active_lags, deadline)
           if d is None:
               break
           
           # Line search: find best alpha
           best_alpha = 0
           best_val = compute_c5(h_curr, n)
           for alpha in [0.01, 0.005, 0.001, 0.0005]:
               h_try = h_curr + alpha * d
               h_try = project_feasible(h_try, n)
               val = compute_c5(h_try, n)
               if val < best_val:
                   best_val = val
                   best_alpha = alpha
           
           if best_alpha > 0:
               h_curr = project_feasible(h_curr + best_alpha * d, n)
               if h_curr.sum() != n / 2:
                   h_curr = project_feasible(h_curr, n)
               if best_val < best_c - 1e-14:
                   best_c = best_val
           else:
               # Try smaller top_m
               top_idx2 = np.argsort(c)[-1]  # just the worst
               active_lags2 = [int(top_idx2) - (n - 1)]
               d2 = lp_direction(h_curr, n, active_lags2, deadline)
               if d2 is not None:
                   for alpha in [0.01, 0.005, 0.001]:
                       h_try = project_feasible(h_curr + alpha * d2, n)
                       val = compute_c5(h_try, n)
                       if val < best_c - 1e-14:
                           h_curr = h_try
                           best_c = val
                           break
               if time.time() > deadline:
                   break
       
       return h_curr, best_c
