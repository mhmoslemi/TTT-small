def projected_subgradient(h, n, alpha, steps, deadline):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = h.copy()
    
    for step in range(steps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)  # actual lag value
        
        # Compute subgradient
        grad = np.zeros(n)
        # h[m] appears in c_k as:
        # h[m] * (1-h[m-k]) for the term where n = m-k (i.e., h[n+k] = h[m]) → contributes (1-h[m-k])*dx
        # h[m] appears as: (1-h[n]) where n=m, h[n+k]=h[m+k] → contributes -h[m+k]*dx
        # So: ∂c_k/∂h[m] = (1-h[m-k])*dx - h[m+k]*dx (when indices valid)
        
        # For term 1: h[m] appears as h[n+k] where n+k=m, so n=m-k
        if m-k >= 0:
            idx1 = np.arange(0, n)
            valid1 = (idx1 - lag >= 0) & (idx1 - lag < n)
            grad[valid1] += (1 - h[idx1[valid1] - lag]) * dx
        
        # For term 2: h[m] appears in (1-h[n]) where n=m

        grad[valid2] -= h[np.arange(0, n)[valid2] + lag] * dx
        
        h = project_feasible(h - alpha * grad, n)
    
    return h

def compute_c5(h, n):
    dx = 2.0 / n

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + np.log(np.exp(sh - sm).sum() + 1e-30) / alpha
