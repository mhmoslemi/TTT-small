    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s <= 1e-15:

    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(float(1.0 - h[idx]), float(diff))
            if add <= 0.0:
                room = 1.0 - h
                tot = float(room.sum())
                if tot <= 1e-15:
                    break
                h += (diff / tot) * room
                h = np.clip(h, 0.0, 1.0)
            else:
                h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(float(h[idx]), float(-diff))
            if sub <= 0.0:
                break
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)


def evaluate_base(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c, float(np.max(c)), dx


def build_indicators(n):
    L = 2 * n - 1
    ind = np.zeros((L, n), dtype=np.int8)
    for m in range(L):
        k = m - (n - 1)
        if k >= 0:
            start = k
            end = n - 1
        else:
            start = 0
            end = n - 1 + k
        if start <= end:
            ind[m, start:end + 1] = 1
    return ind


def refresh_active(base, current_c, tol, L, indF):
    active = np.where(base >= current_c - tol)[0]
    active_list = active.tolist()
    indA = indF[active]
    active_pos = np.full(L, -1, dtype=np.int32)
    if active.size:
        active_pos[active] = np.arange(active.size, dtype=np.int32)
    return active_list, indA, active_pos


def pair_direction(active_list, ind, indA, active_pos, i, j, pi, pj, n):
    A = len(active_list)
    if A == 0:
        return 0
    u = float(pi) - float(pj)
    d = i - j
    if d < 0:
        d = -d
    mp = n - 1 + d
    mn = n - 1 - d

    if A <= 64:
        mind = 1e30
        maxd = -1e30
        for m in active_list:
            b = int(ind[m, i]) - int(ind[m, j])
            dv = float(b)
            if m == mp or m == mn:
                dv += u
            if dv < mind:
                mind = dv
            if dv > maxd:
                maxd = dv
    else:
        bA = indA[:, i] - indA[:, j]
        p1 = int(active_pos[mp])
        if p1 >= 0:
            bA[p1] += u
        p2 = int(active_pos[mn])
        if p2 >= 0 and p2 != p1:
            bA[p2] += u
        mind = float(np.min(bA))
        maxd = float(np.max(bA))

    tol_dir = 1e-10
    if maxd < -tol_dir:
        return 1
    if mind > tol_dir:
        return -1
    return 0


def pair_vec(t, base, b, dx, mp, mn, u):
    v = base + (dx * float(t)) * b
    a = dx * (u * t + t * t)
    v[mp] += a
    v[mn] += a
    return v


def pair_max(t, base, b, dx, mp, mn, u):
    v = pair_vec(t, base, b, dx, mp, mn, u)
    return float(np.max(v))


def line_search_pair(base, b, dx, mp, mn, u, lb, ub, current_c):
    if ub - lb < 1e-14:
        return None, None, None

    best_t = 0.0
    best_f = float(current_c)

    t_end = ub if ub != 0.0 else lb
    if abs(t_end) > 1e-14:
        f_end = pair_max(t_end, base, b, dx, mp, mn, u)
        if f_end < best_f - 1e-14:
            best_f = f_end
            best_t = float(t_end)

    length = ub - lb
    if length < 1e-10:
        if best_f < current_c - 1e-13:
            vec = pair_vec(best_t, base, b, dx, mp, mn, u)
            return float(best_t), float(np.max(vec)), vec
        return None, None, None

    gr = (3.0 - 5.0 ** 0.5) / 2.0
    left = float(lb)
    right = float(ub)

    gold_c = right - gr * (right - left)
    gold_d = left + gr * (right - left)
    fc = pair_max(gold_c, base, b, dx, mp, mn, u)
    fd = pair_max(gold_d, base, b, dx, mp, mn, u)

    if fc < best_f - 1e-14:
        best_f = fc
        best_t = gold_c
    if fd < best_f - 1e-14:
        best_f = fd
        best_t = gold_d

    for _ in range(38):
        if right - left < 1e-8:
            break
        if fc < fd:
            right = gold_d
            gold_d = gold_c
            fd = fc
            gold_c = right - gr * (right - left)
            fc = pair_max(gold_c, base, b, dx, mp, mn, u)
        else:
            left = gold_c
            gold_c = gold_d
            fc = fd
            gold_d = left + gr * (right - left)
            fd = pair_max(gold_d, base, b, dx, mp, mn, u)

        if fc < best_f - 1e-14:
            best_f = fc
            best_t = gold_c
        if fd < best_f - 1e-14:
            best_f = fd
            best_t = gold_d

    t_mid = 0.5 * (left + right)
    fm = pair_max(t_mid, base, b, dx, mp, mn, u)
    if fm < best_f - 1e-14:
        best_f = fm
        best_t = t_mid

    if best_f < current_c - 1e-13:
        vec = pair_vec(best_t, base, b, dx, mp, mn, u)
        return float(best_t), float(np.max(vec)), vec
    return None, None, None


def pair_descent(h, base, ind, indF, dx, deadline, max_passes=2000):
    n = int(len(h))
    L = int(len(base))
    active_tol = 1e-12

    h = np.clip(h, 0.0, 1.0).copy()
    base, current_c, _ = evaluate_base(h, n)

    active_list, indA, active_pos = refresh_active(base, current_c, active_tol, L, indF)
    any_impr_global = False

    for _ in range(max_passes):
        if time.time() > deadline:
            break

        h = np.clip(h, 0.0, 1.0)
        if abs(float(h.sum()) - 0.5 * n) > 1e-9:
            h = project_feasible(h, n)
        base, current_c, _ = evaluate_base(h, n)
        active_list, indA, active_pos = refresh_active(base, current_c, active_tol, L, indF)

        any_impr = False

        for i in range(n):
            if time.time() > deadline:
                return h, base, current_c
            pi = float(h[i])

            for j in range(i + 1, n):
                pj = float(h[j])

                if pi >= 1.0 - 1e-14 and pj <= 1e-14:
                    continue
                if pi <= 1e-14 and pj >= 1.0 - 1e-14:
                    continue

                dirv = pair_direction(active_list, ind, indA, active_pos, i, j, pi, pj, n)
                if dirv == 0:
                    continue

                if dirv > 0:
                    ub = min(1.0 - pi, pj)
                    if ub <= 1e-13:
                        continue
                    lb = 0.0
                else:
                    mag = min(pi, 1.0 - pj)
                    if mag <= 1e-13:
                        continue
                    lb = -mag
                    ub = 0.0

                d = i - j
                if d < 0:
                    d = -d
                mp = n - 1 + d
                mn = n - 1 - d
                u = pi - pj

                bfull = indF[:, i] - indF[:, j]
                t_new, c_new, vec_new = line_search_pair(
                    base, bfull, dx, mp, mn, u, lb, ub, current_c
                )

                if t_new is not None:
                    new_pi = pi + t_new
                    new_pj = pj - t_new
                    clipped = False
                    if new_pi < 0.0:
                        new_pi = 0.0
                        clipped = True
                    elif new_pi > 1.0:
                        new_pi = 1.0
                        clipped = True
                    if new_pj < 0.0:
                        new_pj = 0.0
                        clipped = True
                    elif new_pj > 1.0:
                        new_pj = 1.0
                        clipped = True

                    h[i] = new_pi
                    h[j] = new_pj

                    if clipped:
                        base, current_c, _ = evaluate_base(h, n)
                    else:
                        base = vec_new
                        current_c = c_new

                    any_impr = True
                    any_impr_global = True
                    pi = float(h[i])
                    active_list, indA, active_pos = refresh_active(
                        base, current_c, active_tol, L, indF
                    )

        if not any_impr:
            h = np.clip(h, 0.0, 1.0)
            base, current_c, _ = evaluate_base(h, n)
            return h, base, current_c

    h = np.clip(h, 0.0, 1.0)
    base, current_c, _ = evaluate_base(h, n)
    return h, base, current_c


def interpolate_to(h, n_old, n_new):
    if n_old == n_new:
        return np.asarray(h, dtype=np.float64).copy()
    x_old = np.linspace(0.0, 2.0, int(n_old))
    x_new = np.linspace(0.0, 2.0, int(n_new))
    return np.interp(x_new, x_old, np.asarray(h, dtype=np.float64))


def default_initial(n, rng):
    n = int(n)
    if n % 2 == 0:
        h = np.zeros(n, dtype=np.float64)
        if n > 0:
            cols = rng.choice(n, size=n // 2, replace=False)
            h[cols] = 1.0
    else:
        h = (rng.rand(n) < 0.5).astype(np.float64)

    x = np.linspace(0.0, 2.0, n)
    s = 0.5 + 0.35 * np.sin(np.pi * x) + 0.15 * np.sin(2.0 * np.pi * x + 0.7)
    s = np.clip(s, 0.0, 1.0)
    h = 0.55 * h + 0.45 * s
    return project_feasible(h, n)


def optimize_candidate(h, n, deadline_limit, max_passes=3):
    n = int(n)
    h = np.asarray(h, dtype=np.float64).copy()
    h = project_feasible(h, n)
    base, c, dx = evaluate_base(h, n)
    if not np.isfinite(c):
        return h, np.inf, n
    if time.time() >= deadline_limit:
        return h, float(c), n

    ind = build_indicators(n)
    indF = ind.astype(np.float64)
    h, base, c = pair_descent(h, base, ind, indF, dx, deadline_limit, max_passes)
    h = np.clip(h, 0.0, 1.0)
    base, c, dx = evaluate_base(h, n)
    return h, float(c), n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = float(budget_s)
    deadline = t0 + max(10.0, total_budget - 6.0)
    rng = np.random.RandomState(int(seed))

    fallback_h = None
    fallback_c = np.inf
    fallback_n = 300

    try:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
        n0 = int(len(h_init))
        if n0 < 20:
            raise ValueError
    except Exception:
        n0 = 300
        h_init = default_initial(n0, rng)

    h0 = project_feasible(h_init, n0)
    base0, c0, dx0 = evaluate_base(h0, n0)

    best_h = h0.copy()
    best_c = float(c0)
    best_n = int(n0)
    best_dx = float(dx0)

    if np.isfinite(best_c):
        fallback_h = best_h.copy()
        fallback_c = best_c
        fallback_n = best_n

    remaining_total = deadline - t0
    if remaining_total > 600.0:
        main_reserve = 220.0
    elif remaining_total > 250.0:
        main_reserve = 90.0
    else:
        main_reserve = 12.0
    main_deadline = deadline - main_reserve

    if time.time() < main_deadline:
        h_main, c_main, n_main = optimize_candidate(
            h0, n0, main_deadline, max_passes=10000
        )
        if np.isfinite(c_main) and c_main < best_c:
            if n_main != best_n or True:
                best_h = h_main.copy()
                best_c = float(c_main)
                best_n = int(n_main)
                best_dx = 2.0 / float(best_n)
                fallback_h = best_h.copy()
                fallback_c = best_c
                fallback_n = best_n

    if time.time() < deadline - 120.0:
        lim = min(deadline - 100.0, time.time() + 70.0)
        cand = 0.5 * (best_h + best_h[::-1])
        h_c, c_c, n_c = optimize_candidate(cand, best_n, lim, max_passes=4)
        if np.isfinite(c_c) and c_c < best_c:
            best_h = h_c.copy()
            best_c = float(c_c)
            best_n = int(n_c)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    if time.time() < deadline - 100.0:
        lim = min(deadline - 80.0, time.time() + 55.0)
        cand = 0.5 * (best_h + (1.0 - best_h[::-1]))
        h_c, c_c, n_c = optimize_candidate(cand, best_n, lim, max_passes=4)
        if np.isfinite(c_c) and c_c < best_c:
            best_h = h_c.copy()
            best_c = float(c_c)
            best_n = int(n_c)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    if time.time() < deadline - 80.0:
        lim = min(deadline - 60.0, time.time() + 50.0)
        cand = np.where(best_h > 0.5, 1.0, 0.0)
        h_c, c_c, n_c = optimize_candidate(cand, best_n, lim, max_passes=3)
        if np.isfinite(c_c) and c_c < best_c:
            best_h = h_c.copy()
            best_c = float(c_c)
            best_n = int(n_c)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    perturb_scales = (0.003, 0.006, 0.012, 0.025, 0.06, 0.12)
    for scale in perturb_scales:
        if time.time() > deadline - 50.0:
            break
        h_p = best_h + rng.normal(0.0, float(scale), best_n)
        h_p = project_feasible(h_p, best_n)
        _, c_p, _ = evaluate_base(h_p, best_n)
        if not np.isfinite(c_p):
            continue
        if np.isfinite(best_c) and c_p > best_c + 0.08:
            continue
        lim = min(deadline - 40.0, time.time() + 35.0)
        h_c, c_c, n_c = optimize_candidate(h_p, best_n, lim, max_passes=3)
        if np.isfinite(c_c) and c_c < best_c:
            best_h = h_c.copy()
            best_c = float(c_c)
            best_n = int(n_c)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    original_n = best_n
    for nt in (original_n - 1, original_n + 1, original_n + 2, original_n + 4, 2 * original_n):
        if nt <= 40 or nt > 1200:
            continue
        if nt == best_n:
            continue
        if time.time() > deadline - 50.0:
            break
        h_t = interpolate_to(best_h, best_n, nt)
        h_t = project_feasible(h_t, nt)
        _, c_t, _ = evaluate_base(h_t, nt)
        if not np.isfinite(c_t):
            continue
        if np.isfinite(best_c) and c_t > best_c + 0.004:
            continue

        if nt <= 500:
            lim = min(deadline - 25.0, time.time() + 55.0)
        else:
            lim = min(deadline - 20.0, time.time() + 25.0)
        h_c, c_c, n_c = optimize_candidate(h_t, int(nt), lim, max_passes=3)
        if np.isfinite(c_c) and c_c < best_c:
            best_h = h_c.copy()
            best_c = float(c_c)
            best_n = int(n_c)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    if time.time() < deadline - 1.5:
        h_f = project_feasible(best_h, best_n)
        h_f, c_f, n_f = optimize_candidate(h_f, best_n, deadline - 0.5, max_passes=2)
        if np.isfinite(c_f) and c_f < best_c:
            best_h = h_f.copy()
            best_c = float(c_f)
            best_n = int(n_f)
            best_dx = 2.0 / float(best_n)
            fallback_h = best_h.copy()
            fallback_c = best_c
            fallback_n = best_n

    # Make the returned vector feasible.  Prefer the lighter fix if the fuller projection
    # slightly perturbs an already very good point.
    h_fix = minimal_sum_fix(best_h, best_n)
    base_fix, c_fix, dx_fix = evaluate_base(h_fix, best_n)
    h_proj = project_feasible(best_h, best_n)
    base_proj, c_proj, dx_proj = evaluate_base(h_proj, best_n)

    if np.isfinite(c_fix) and c_fix <= (np.inf if not np.isfinite(c_proj) else c_proj):
        final_h = h_fix
        final_c = float(c_fix)
    else:
        final_h = h_proj
        final_c = float(c_proj)

    final_n = int(best_n)

    if not np.isfinite(final_c) and fallback_h is not None:
        final_h = minimal_sum_fix(fallback_h, fallback_n)
        _, final_c, _ = evaluate_base(final_h, fallback_n)
        final_n = int(fallback_n)

    if not np.isfinite(final_c):
        final_h = project_feasible(np.full(300, 0.5, dtype=np.float64), 300)
        _, final_c, _ = evaluate_base(final_h, 300)
        final_n = 300

    return final_h, float(final_c), final_n
