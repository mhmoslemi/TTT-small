def subgradient_method(h, n, deadline, initial_step=0.05):

    best_h = h.copy()

        weights /= weights.sum()

            m_abs = abs(m)

                    grad[m_abs:] += w * (1 - h[:n-m_abs]) * dx
                else:
                    grad[:n-m_abs] += w * (1 - h[m_abs:]) * dx
                    grad[m_abs:] += w * (-h[:n-m_abs]) * dx

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        step *= 0.99

            break

        c_new = float(c_all.max())
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        step *= 0.992
        if step < 1e-7:
            break
    return best_h, best_c
