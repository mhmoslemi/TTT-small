def compute_smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Smooth max
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_vals = np.exp(shifted)
    log_sum = c_max + np.log(exp_vals.sum()) / alpha
    weights = exp_vals / exp_vals.sum()
    # Gradient
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-10:
            continue
        k = p - (n - 1)
        if k >= 0:
            # First term: j in [0, n-1-k], adds w*(1-h[j+k])
            end1 = n - k
            if end1 > 0:
                grad[:end1] += w * (1.0 - h[k:end1+k])  # h[j+k] for j in [0, end1-1]
            # Second term: j in [k, n-1], subtracts w*h[j-k]
            start2 = k
            if start2 < n:
                grad[start2:] -= w * h[:n-start2]  # h[j-k] for j in [k, n-1]
        else:
            m = -k
            # Third term: j in [m, n-1], adds w*(1-h[j-m]) = w*(1-h[j+k])

            end4 = n - m
            if end4 > 0:
                grad[:end4] -= w * h[m:m+end4]  # h[j+m] for j in [0, end4-1]
    return log_sum, grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    target = n / 2.0
    # Bisection to find lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    # Fine correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full')
    c = c_full * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # Handle potential overflow
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total == 0 or not np.isfinite(total):
        # Fallback: uniform weights
        weights = np.ones(2*n-1) / (2*n-1)
        obj = c_max
    else:
        weights = exp_vals / total
        obj = c_max + np.log(total) / alpha
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-14:
            continue
        k = p - (n - 1)
        if k >= 0:
            end1 = n - k
            if end1 > 0:
                grad[:end1] += w * (1.0 - h[k:k + end1])
            if k < n:
                grad[k:] -= w * h[:end1]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m])
            if m < n:
                grad[:n - m] -= w * h[m:]
    return obj, grad

def proj_gradient_optimize(h, n, alphas, iters_per_alpha, base_lr, deadline):
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    for alpha in alphas:
        h_cur = best_h.copy()
        obj_cur = smooth_max_and_grad(h_cur, n, alpha)[0]
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_obj
            _, grad = smooth_max_and_grad(h_cur, n, alpha)
            lr = base_lr / (1.0 + it * 0.01)
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, n)
            obj_new = smooth_max_and_grad(h_new, n, alpha)[0]
            if obj_new < obj_cur:
                h_cur = h_new
                obj_cur = obj_new
            # Check actual c5
            c5 = compute_c5(h_cur, n)
            if c5 < best_obj:

                best_h = h_cur.copy()
    return best_h, best_obj

    idx = np.argsort(h)[::-1]

def binary_local_search(h_init, n, deadline, max_sweeps=200):

    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)

    best_h = h.copy()

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full')
    c = c_full * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if not np.isfinite(total) or total < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
        obj = c_max
    else:
        weights = exp_vals / total
        obj = c_max + np.log(total) / alpha
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-14:
            continue
        k = p - (n - 1)
        if k >= 0:
            end1 = n - k
            if end1 > 0:
                grad[:end1] += w * (1.0 - h[k:k + end1])
            if k < n:
                grad[k:] -= w * h[:end1]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m])
            if m < n:
                grad[:n - m] -= w * h[m:]
    return obj, grad

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def binary_greedy(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = binary_swap_delta(h, i, j, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                i_idx = zeros.index(i)
                j_idx = ones.index(best_j)
                h[i], h[best_j] = 1.0, 0.0
                corr += binary_swap_delta(h, i, best_j, n, m_range)
                # Recompute delta correctly since h changed
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                zeros[i_idx] = best_j
                ones[j_idx] = i
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    cool = 0.99995
    improve_count = 0
    while time.time() < deadline:
        if T < 1e-10:
            break
        i = int(rng.randint(len(zeros)))
        j = int(rng.randint(len(ones)))
        ii = zeros[i]
        jj = ones[j]

            if nc < best_c:

            else:

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: Projected gradient on original resolution (35% time)
    p1 = t0 + 0.35 * total_budget
    alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    iters_per_alpha = [800, 600, 600, 500, 400, 300, 200]
    lr_base = 0.008
    h_cur = best_h.copy()
    for idx_a, (alpha, niter) in enumerate(zip(alphas, iters_per_alpha)):
        if time.time() > p1 - 2:
            break
        phase_dl = min(p1, t0 + total_budget * (0.35 * (idx_a + 1) / len(alphas)))
        h_cur, c5 = proj_grad_loop(h_cur, n, alpha, niter, lr_base, phase_dl)
        if c5 < best_c:
            best_c = c5
            best_h = h_cur.copy()

    # Phase 2: Binary local search (15% time)
    p2 = t0 + 0.50 * total_budget
    hb, cb = binary_greedy(best_h, n, p2, max_sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Re-optimize from improved binary
    if cb < best_c + 0.0001:
        h_cur = to_binary(best_h, n).astype(float)
        for alpha in [100, 1000, 10000]:
            if time.time() > p2 - 3:
                break
            h_cur, c5 = proj_grad_loop(h_cur, n, alpha, 300, lr_base, p2 - 3)
            if c5 < best_c:
                best_c = c5
                best_h = h_cur.copy()

    # Phase 3: Higher resolution attempts (30% time)
    p3 = t0 + 0.80 * total_budget
    resolutions = [600, 800, 1000]
    for nh in resolutions:
        if time.time() > p3 - 40:
            break
        h_new = interpolate_h(best_h, best_n, nh)
        c5_new = compute_c5(h_new, nh)
        if c5_new < best_c:
            best_c = c5_new
            best_h = h_new.copy()
            best_n = nh
        alphas3 = [50, 500, 5000, 50000]
        for alpha in alphas3:
            if time.time() > p3 - 30:
                break
            h_new, c5 = proj_grad_loop(h_new, nh, alpha, 400, lr_base * (n / nh), p3 - 30)
            if c5 < best_c:
                best_c = c5
                best_h = h_new.copy()
                best_n = nh
                n = nh

    # Phase 4: Simulated annealing on binary (15% time)
    p4 = t0 + 0.92 * total_budget
    ha, ca = binary_anneal(best_h, best_n, p4, seed=seed)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
        best_n = len(ha)
    # Polish from annealed binary
    if time.time() < p4 - 5:
        h_cur = to_binary(best_h, best_n).astype(float)
        for alpha in [1000, 10000, 100000]:
            if time.time() > p4 - 5:
                break
            h_cur, c5 = proj_grad_loop(h_cur, best_n, alpha, 200, lr_base, p4 - 5)
            if c5 < best_c:
                best_c = c5
                best_h = h_cur.copy()

    # Phase 5: Final multi-restart polish (remaining time)
    p5 = deadline - 2
    if time.time() < p5:
        for trial in range(5):
            if time.time() > p5:
                break
            noise_scale = 0.003 * (trial + 1)
            h_trial = project_feasible(best_h + noise_scale * rng.randn(best_n), best_n)
            for alpha in [5000, 50000]:
                if time.time() > p5:
                    break
                h_trial, c5 = proj_grad_loop(h_trial, best_n, alpha, 150, lr_base, p5)
                if c5 < best_c:
                    best_c = c5
                    best_h = h_trial.copy()

    # Also try: go back to binary, greedy, then smooth
    if time.time() < deadline - 1:
        hb2, cb2 = binary_greedy(best_h, best_n, deadline - 3, max_sweeps=100)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
