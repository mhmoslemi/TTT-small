from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:

        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

    y = h_curr.copy()  # Nesterov momentum variable
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

    beta = 0.9  # momentum
    c_lower = 0.37  # lower bound for Polyak step size
    n_smooth = 10  # number of top lags to average over
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 100 == 0 and int(compute_c5(h_curr, n) * 1e8) == int(best_c * 1e8):
            if it > 1000:
                break
        
        c = compute_all_lags(y, n)
        m = c.max()
        
        # Identify active set (top n_smooth lags within some tolerance of max)
        tol = 0.001
        active_mask = c >= (m - tol)
        if active_mask.sum() == 0:
            active_mask = c >= (m - 0.01)
        if active_mask.sum() == 0:
            # Just use the top ones
            top_idx = np.argsort(c)[-n_smooth:]
            active_mask = np.zeros(2*n-1, dtype=bool)
            active_mask[top_idx] = True

        subgrad = np.zeros(n)

        if grad_norm_sq < 1e-20:

        alpha = (m - c_lower) / grad_norm_sq
        alpha = min(alpha, 0.5)  # cap the step

        y_new = project_feasible(h_curr + beta * (y - h_curr) - alpha * subgrad, n)
        
        # Check improvement

        if c_new < best_c:

            best_h = y_new.copy()
        
        h_curr = y_new
        y = y_new
        beta = min(beta * 0.9999, 0.999)  # slowly increase momentum

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_constraint_jac(x, n):
    return np.ones(n)

def smooth_max_obj(x, n, alpha, top_idx=None):

    if top_idx is not None:
        c = c[top_idx]

    shifted = alpha * (c - m)

    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

    """Nesterov accelerated projected subgradient method."""
    h_curr = h.copy()
    y = h_curr.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    beta = 0.9
    stall_count = 0
    patience = 2000
    
    for it in range(200000):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(y, n)
        m = c.max()
        
        # Polyak step size
        gap = m - c_lower
        if gap < 1e-10:
            break
        
        # Active set: top lags within tolerance
        tol = 0.002
        active_mask = c >= (m - tol)
        if active_mask.sum() == 0:
            active_mask = c >= (m - 0.01)
        n_active = max(active_mask.sum(), 1)
        
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in range(2 * n - 1):
            if active_mask[k]:
                lag = k - (n - 1)
                subgrad += compute_lag_gradient(y, n, lag)
        subgrad /= n_active
        
        grad_norm_sq = np.dot(subgrad, subgrad)
        if grad_norm_sq < 1e-20:
            break
        
        alpha = gap / grad_norm_sq
        alpha = min(alpha, 0.3)

        y = y_new
    
    return best_h, best_c

            cons.append(cp.sum(grad * h_var) <= t_var - c[k] + cp.sum(grad * h_curr))
