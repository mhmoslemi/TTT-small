from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lag_gradient(h, n, lag):
    """Compute gradient of c_lag with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        return grad
    elif lag > 0:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

    lags = np.correlate(h, 1 - h, mode='full')  # don't multiply by dx for now

    soft_max_val = (max_lag + log_sum / alpha)  # this is (1/alpha)*log(sum(exp(alpha*lags)))
    
    # Weights

        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += weights[k] * lag_gradient(h, n, lag)
    
    return soft_max_val, grad

            if tot < 1e-14:
