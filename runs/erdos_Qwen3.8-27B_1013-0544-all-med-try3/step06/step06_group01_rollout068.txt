import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Linear correction
        interior = (h_proj > 1e-12) & (h_proj < 1 - 1e-12)
        if interior.sum() > 0:
            h_proj[interior] += diff / interior.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h_p.sum()
    if abs(diff) > 1e-10:
        interior = (h_p > 1e-14) & (h_p < 1.0 - 1e-14)
        cnt = interior.sum()
        if cnt > 0:
            h_p[interior] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p
