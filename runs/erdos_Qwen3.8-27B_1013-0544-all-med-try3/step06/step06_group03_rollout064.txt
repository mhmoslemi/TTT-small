def subgradient_step(h, n, step_size, top_frac=0.1):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(c)
    n_top = max(1, int(n_top_frac * n_lags))
    top_indices = np.argsort(c)[-n_top:]
    
    # Compute subgradient
    grad = np.zeros(n)
    for m in top_indices:
        lag = m - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j < n and 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j < n and 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += g
    grad *= dx
    
    # Take step in negative gradient direction
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

- j-lag ≥ 0 means j ≥ lag, so j = lag, ..., n-1
- j+lag ≥ 0 is always true for j ≥ 0

So for lag ≥ 0:
- grad[0:n-lag] += (1 - h[lag:n])  [from first term: 1-h[j+lag] for j=0..n-1-lag, j+lag=lag..n-1]
- grad[lag:n] -= h[0:n-lag]  [from second term: -h[j-lag] for j=lag..n-1, j-lag=0..n-1-lag]

For lag < 0, let lag' = -lag > 0:
- j+lag = j-lag' ≥ 0 means j ≥ lag', so j = lag', ..., n-1
- j-lag = j+lag' < n means j < n-lag', so j = 0, ..., n-1-lag'
- j+lag < n is always true for j < n

So for lag < 0 (lag' = -lag):
- grad[lag':n] += (1 - h[0:n-lag'])  [from first term: 1-h[j+lag] = 1-h[j-lag'] for j=lag'..n-1, j-lag'=0..n-1-lag']
- grad[0:n-lag'] -= h[lag':n]  [from second term: -h[j-lag] = -h[j+lag'] for j=0..n-1-lag', j+lag'=lag'..n-1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_func(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max_val(c, alpha)

def sum_fun(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

                       bounds=bounds, constraints=cons,

        if res.x is not None:

def subgradient_search(h, n, deadline, n_iter_max=50000):
    """Subgradient-based local search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step_size = 0.01
    top_frac = 0.05
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        current_max = c.max()
        
        if current_max < best_c - 1e-14:
            best_c = current_max
        
        # Select top lags
        n_top = max(3, int(top_frac * len(c)))
        top_idx = np.argsort(c)[-n_top:]
        
        # Compute subgradient (vectorized)
        grad = np.zeros(n)
        for m in top_idx:
            lag = int(m) - (n - 1)
            if lag >= 0:
                # grad[0:n-lag] += (1 - h[lag:n])
                # grad[lag:n] -= h[0:n-lag]
                if n - lag > 0:
                    grad[0:n-lag] += (1.0 - h[lag:n])
                    grad[lag:n] -= h[0:n-lag]
            else:
                neg_lag = -lag
                if n - neg_lag > 0:
                    grad[neg_lag:n] += (1.0 - h[0:n-neg_lag])
                    grad[0:n-neg_lag] -= h[neg_lag:n]
        grad *= dx
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Nelder-Mead style: try the step, if it doesn't help, halve step
        h_new = project_feasible(h - step_size * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:

            best_c = c_new
            step_size = min(step_size * 1.1, 0.1)
        else:
            step_size *= 0.7
            if step_size < 1e-6:
                # Try smaller top_frac
                top_frac = max(0.01, top_frac * 0.5)
    
    return h, best_c
