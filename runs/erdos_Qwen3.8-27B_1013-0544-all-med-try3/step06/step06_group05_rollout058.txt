import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project to {0 ≤ h ≤ 1, sum(h) = n/2}
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP',

def project_feasible(h, n):

    k = np.arange(L)  # shape (L,)
    # W[i,k] = (1-h[i-k+n-1]) * [0 <= i-k+n-1 < n] - h[i+k-n+1] * [0 <= i+k-n+1 < n]
    # i-k+n-1: shape (n, L) when i is (n,1) and k is (1,L)
    i_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = k[None, :]  # (1, L)
    p1 = i_idx - k_idx + (n - 1)  # (n, L)
    p2 = i_idx + k_idx - (n - 1)  # (n, L)
    mask1 = (p1 >= 0) & (p1 < n)  # (n, L)
    mask2 = (p2 >= 0) & (p2 < n)  # (n, L)
    # Need h[p1] where mask1 is True, and h[p2] where mask2 is True
    # Use np.where to handle invalid indices
    p1_safe = np.where(mask1, p1, 0)
    p2_safe = np.where(mask2, p2, 0)
    W = mask1 * (1.0 - h[p1_safe]) - mask2 * h[p2_safe]
    return W

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))
