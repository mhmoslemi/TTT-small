        if step < 0.0001:
            break
        step *= 0.95
    return best_h, best_c

def coordinate_refine(h, n, deadline):
    """Efficient coordinate descent: try small adjustments to individual coordinates."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    delta = 0.01
    while delta > 0.0001 and time.time() < deadline:
        improved = False
        indices = list(range(n))
        rng.shuffle(indices)
        for i in indices:
            if time.time() > deadline:
                break
            for d in [delta, -delta]:
                h[i] += d
                if h[i] < 0 or h[i] > 1:
                    h[i] -= d
                    continue
                # Fix sum constraint by adjusting a random other element
                diff = -d
                j = i
                while j == i:
                    j = rng.randint(n)
                h[j] += diff
                if h[j] < 0 or h[j] > 1:
                    h[i] -= d
                    h[j] -= diff
                    continue
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= d
                    h[j] -= diff
        if not improved:
            delta *= 0.5
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline, seed):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # SLSQP with gradient, alpha sweep
    alphas = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000, 1000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        h_try = slsqp_with_grad(best_h, n, a, maxiter=3000, ftol=1e-18)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try.copy()
    
    # Gradient descent fine-tuning
    if time.time() < deadline - 8:
        h_try, c_try = gradient_descent(best_h, n, 500000, steps=500, lr=0.05,
                                        deadline=deadline - 5)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try.copy()
    
    # Coordinate refinement
    if time.time() < deadline - 5:
        h_try, c_try = coordinate_refine(best_h, n, deadline - 2)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 250
        h_init = np.full(n_init, 0.5)
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at the initial resolution
    h_opt, c_opt = optimize_at_resolution(h_init, n_init, deadline - 80, seed)
    best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try perturbed restarts at same resolution
    if time.time() < deadline - 60:
        remaining = deadline - 60 - time.time()
        n_restarts = min(4, max(1, int(remaining / 60)))
        scales = [0.02, 0.05, 0.01, 0.08]
        for r in range(n_restarts):
            if time.time() > deadline - 55:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_init)
            hp = project_feasible(hp, n_init)
            alloc = min(55, deadline - 55 - time.time())
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n_init, time.time() + alloc, seed + 100 + r)
            if c_opt < best_c - 1e-12:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 3: Try a higher resolution
    if time.time() < deadline - 60:
        n_high = n_init + 100
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_high)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_high)
        alloc = min(80, deadline - 40 - time.time())
        if alloc > 30:
            h_opt, c_opt = optimize_at_resolution(h_interp, n_high, time.time() + alloc, seed + 500)
            if c_opt < best_c - 1e-12:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final fine-tuning on the best solution
    if best_h is not None and time.time() < deadline - 5:
        n_best = len(best_h)
        # Final SLSQP with very high alpha
        h_try = slsqp_with_grad(best_h, n_best, 5000000, maxiter=5000, ftol=1e-18)
        c_try = compute_c5(h_try, n_best)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
        
        # Gradient descent polish
        if time.time() < deadline - 2:
            h_try, c_try = gradient_descent(best_h, n_best, 2000000, steps=300, lr=0.02,
                                            deadline=deadline - 1)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
