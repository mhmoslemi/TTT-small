    return np.correlate(h, 1.0 - h, mode='full') * dx

    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += -h[:n-lag]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] += -h[m:]

    all_g = compute_all_g(h, n)
    m = all_g.max()
    if alpha < 1e6:
        exp_vals = np.exp(alpha * (all_g - m))
        val = m + np.log(exp_vals.sum()) / alpha

    m = all_g.max()

        exp_vals = np.exp(alpha * (all_g - m))
        w = exp_vals / exp_vals.sum()

            lag = k - (n - 1)
            grad += w[k] * analytical_subgrad(h, n, lag)

        if diff > 0:

            s = room.sum()
            if s < 1e-13:

            s = h.sum()
            if s < 1e-13:
                break

    def fun(x):

        return smooth_grad(x, n, alpha)

                       constraints=cons,

                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def pair_refine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    h_work = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi, hj = h_work[i] + s * d, h_work[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h_work[i], h_work[j] = hi, hj
                            c = compute_c5(h_work, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                prog = True
                            else:
                                h_work[i], h_work[j] = hi - s * d, hj + s * d
    return h_work, best_c

def interpolate_to(h_old, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(xn, xo, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_dl = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h0)
    except Exception:
        n_base = 250
        h0 = np.full(n_base, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n_base)
    n = n_base

    # Phase 1: SLSQP with analytical gradient at base resolution (35% budget)
    p1_end = t0 + 0.35 * (budget_s - 3)
    h = best_h.copy()
    alpha_seq = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for alpha in alpha_seq:
        if time.time() > p1_end:
            break
        h = slsqp_optimize(h, n, alpha, 1500, p1_end)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Exact subgradient steps (10% budget)
    p2_end = t0 + 0.45 * (budget_s - 3)
    lr = 0.5 / n
    for trial_lr in [lr, lr * 0.5, lr * 0.2, lr * 0.1]:
        if time.time() > p2_end:
            break
        h_sg, c_sg = exact_subgrad_step(best_h.copy(), n, trial_lr, p2_end, max_iter=15000)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg.copy()

    # Phase 3: Projected momentum with high alpha (10% budget)
    p3_end = t0 + 0.55 * (budget_s - 3)
    for alpha_pm in [50000, 200000, 1000000]:
        if time.time() > p3_end:
            break
        lr_pm = 1.0 / (alpha_pm * 10.0 / n)
        h_pm, c_pm = projected_momentum(best_h.copy(), n, alpha_pm, lr_pm, 0.9, p3_end, max_iter=20000)
        if c_pm < best_c:
            best_c = c_pm
            best_h = h_pm.copy()

    # Phase 4: Higher resolution optimization (25% budget)
    p4_end = t0 + 0.80 * (budget_s - 3)
    for n_new in [n + 50, n + 100, n + 150, n + 250]:
        if time.time() > p4_end - 30:
            break
        if n_new > 600:
            break
        h_new = interpolate_to(best_h, n, n_new)
        h_new = slsqp_optimize(h_new, n_new, 100, 500, p4_end - 20)
        c_new = compute_c5(h_new, n_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            n = n_new
        if time.time() > p4_end - 60:
            break
        h_new = slsqp_optimize(h_new, n, 1000, 800, p4_end - 30)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if time.time() > p4_end - 30:
            break
        h_new = slsqp_optimize(h_new, n, 10000, 1000, p4_end - 10)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 5: Binary local search (8% budget)
    p5_end = t0 + 0.88 * (budget_s - 3)
    if time.time() < p5_end - 5:
        hb, cb = binary_local_search(best_h.copy(), n, p5_end, max_sweeps=300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Try from perturbed starting point
        hp = project_feasible(best_h + 0.05 * rng.randn(n), n)
        hb2, cb2 = binary_local_search(hp, n, p5_end, max_sweeps=100)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 6: Final pair refinement + SLSQP polish (7% budget)
    if time.time() < total_dl - 20:
        h_fin = slsqp_optimize(best_h.copy(), n, 500000, 1000, total_dl - 20)
        c_fin = compute_c5(h_fin, n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    if time.time() < total_dl - 3:
        best_h, best_c = pair_refine(best_h, n, total_dl,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
