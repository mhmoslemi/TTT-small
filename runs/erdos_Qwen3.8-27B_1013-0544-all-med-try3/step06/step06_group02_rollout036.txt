    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    # Sort by value, threshold
    sorted_idx = np.argsort(-h)  # descending
    target = n // 2
    b = np.zeros(n)

    m = corr.max()
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

```python
import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level, no closures)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_max(vals, alpha):
    m = vals.max()
    if alpha < 1:
        alpha = 1.0
    return float(m + np.log(np.sum(np.exp(-alpha * (m - vals)))) / alpha)

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth_max of correlation."""
    corr = compute_corr(h, n)
    m = corr.max()
    w = np.exp(-alpha * (m - corr))
    w = w / w.sum()
    # grad of corr[k] w.r.t. h[i]
    dx = 2.0 / n
    grad = np.zeros(n)
    # corr[m_idx] = dx * sum_i h[i] * (1-h[i+k]) where k = m_idx - (n-1)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_smooth(h, n, alpha, deadline, maxiter=300):
    """L-BFGS-B with smooth max objective and barrier for integral constraint."""
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x, n)
        m = corr.max()
        val = m + np.log(np.sum(np.exp(-alpha * (m - corr)))) / alpha
        val += 1000.0 * (x.sum() - target) ** 2
        return float(val)
    
    def grad(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2000.0 * (x.sum() - target)
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def binary_anneal(b, n, deadline, seed=42):
    """Simulated annealing on binary vectors."""
    b = b.copy()
    dx = 2.0 / n
    ones = list(np.nonzero(b)[0])
    zeros = list(np.nonzero(1 - b)[0])
    
    corr = np.correlate(b, 1.0 - b, mode='full')
    current_smooth = smooth_max(corr * dx, 10.0)
    current_exact = float(np.max(corr)) * dx
    
    best_b = b.copy()
    best_exact = current_exact
    best_smooth = current_smooth
    
    T = 1.0
    T_min = 0.01
    # Cooling: use geometric with rate based on available time
    # Target: reach T_min in about 80% of available time
    avail = max(deadline - time.time(), 1.0)
    # Estimate steps: ~50k per second
    est_steps = int(avail * 40000)
    beta = (T_min / T) ** (1.0 / max(est_steps, 1000))
    
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline:
        T = max(T * beta, T_min)
        
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        b[i] = 0.0
        b[j] = 1.0
        
        new_corr = np.correlate(b, 1.0 - b, mode='full')
        new_smooth = smooth_max(new_corr * dx, 10.0)
        
        delta = new_smooth - current_smooth
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_smooth = new_smooth
            new_exact = float(np.max(new_corr)) * dx
            if new_exact < best_exact - 1e-12:
                best_exact = new_exact
                best_b = b.copy()
                best_smooth = new_smooth
            # Update lists
            ones = list(np.nonzero(b)[0])
            zeros = list(np.nonzero(1.0 - b)[0])
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_exact
