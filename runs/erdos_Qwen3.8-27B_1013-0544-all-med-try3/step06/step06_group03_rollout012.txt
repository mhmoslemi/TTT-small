def pair_search_simulated_annealing(h, n, deadline, deltas, T_init=0.01, T_decay=0.99, radius=None):
    """Simulated annealing pair search - can escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    T = T_init
    
    while time.time() < deadline and T > 1e-6:
        d = deltas[np.random.randint(len(deltas))]
        improved = False
        for _ in range(n * 2):  # n*2 random moves per temperature
            if time.time() > deadline:
                break
            i = np.random.randint(n)
            if radius is not None:
                j = max(0, min(n-1, i + np.random.randint(-radius, radius+1)))
            else:
                j = np.random.randint(n)
            if i == j:
                continue
            
            # Random direction
            if np.random.rand() < 0.5:
                di, dj = d, -d
            else:
                di, dj = -d, d
            
            if cur_h[i] + di < -1e-12 or cur_h[i] + di > 1 + 1e-12:
                continue
            if cur_h[j] + dj < -1e-12 or cur_h[j] + dj > 1 + 1e-12:
                continue
            
            cur_h[i] += di
            cur_h[j] += dj
            new_c = compute_c5(cur_h, n)
            delta = new_c - cur_c
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                cur_c = new_c
                improved = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
            else:
                cur_h[i] -= di
                cur_h[j] -= dj
        
        T *= T_decay
        if not improved:
            # Restart from best
            cur_h = best_h.copy()
            cur_c = best_c
    
    return best_h, best_c

def binary_project(h, n):
    """Project to a binary-like solution."""
    hb = (h > 0.5).astype(float)
    # Adjust to meet sum constraint
    target = int(n / 2)
    curr_sum = int(hb.sum())
    if curr_sum > target:
        # Turn off some 1s (those with lowest h values)
        ones = np.where(hb == 1)[0]
        sorted_ones = ones[np.argsort(h[ones])]
        for idx in sorted_ones[:curr_sum - target]:
            hb[idx] = 0
    elif curr_sum < target:
        # Turn on some 0s (those with highest h values)
        zeros = np.where(hb == 0)[0]
        sorted_zeros = zeros[np.argsort(-h[zeros])]
        for idx in sorted_zeros[:target - curr_sum]:
            hb[idx] = 1
    return project_feasible(hb, n)

**Alternative: Formulate as a SOCP or QCQP**

**Another idea: Use the structure of the problem**

The correlation at lag 0 is: c_0 = Σ_i h[i]*(1-h[i]) = Σ_i h[i] - Σ_i h[i]² = n/2 - Σ_i h[i]².

To minimize c_0, we want to maximize Σ h[i]², which means making h as "binary" as possible (0 or 1).

- Close to binary (0 or 1)

- With the runs arranged to minimize the maximum correlation at ALL lags
