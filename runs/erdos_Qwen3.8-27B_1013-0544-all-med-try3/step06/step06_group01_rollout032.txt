    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

    idx = np.argsort(h)[::-1]

            min_val = new_maxes[min_idx]

                if min_val < best_swap_c:
                    best_swap_c = min_val
                    best_a = a
                    best_b = ones[min_idx]
                    best_new_corr = new_corrs[min_idx]
            else:

                    best_swap_c = min_val
                    best_a = a
                    best_b = ones[min_idx]
                    best_new_corr = new_corrs[min_idx]

            best_c = cur_c
            best_h = h.copy()
        
        T *= T_decay
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_search_fast(h, n, deadline, max_sweeps=10000, seed=42):
    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)
    h = to_binary(h, n)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    n_corr = 2 * n - 1
    stalling = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(zeros)
        best_swap_c = cur_c + 1e-10
        best_a = -1
        best_b = -1
        best_nc = None

        for a in zeros:
            pa = np.zeros(n_corr)
            pa[n - 1 - a : 2 * n - a - 1] += (1.0 - h[:n - a])
            pa[a : n + a] -= h[::-1]
            d_a = corr + pa
            num_ones = len(ones)
            P = np.zeros((num_ones, n_corr))
            for j in range(num_ones):
                b = ones[j]
                P[j, b:b + n] = h[::-1]
                P[j, n - 1 - b : 2 * n - b - 1] += (h[:n - b] - 1.0)
            new_corrs = d_a[None, :] + P
            new_maxes = new_corrs.max(axis=1) * dx
            min_idx = new_maxes.argmin()
            min_val = new_maxes[min_idx]
            if min_val < best_swap_c:
                best_swap_c = min_val

                best_b = ones[min_idx]

            cur_c = best_swap_c
            stalling = 0

                best_h = h.copy()
        else:
            stalling += 1
            if stalling > 5:

            if stalling >= 1:

def simulated_annealing(h, n, deadline, seed=42, T_init=0.002, T_final=0.0001):

    target = n // 2
    rng = np.random.RandomState(seed)
    h = to_binary(h, n)

    best_h = h.copy()

    n_corr = 2 * n - 1

    iter_step = log_T / total_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        if len(ones) == 0 or len(zeros) == 0:
            break

def slsqp_optimize(h, n, alpha, deadline, maxiter=500):
    dx = 2.0 / n
    n_corr = 2 * n - 1

    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        max_c = c.max()
        exp_c = np.exp(alpha * (c - max_c))
        F = (max_c + np.log(exp_c.sum())) / alpha
        w = exp_c / exp_c.sum()
        S1 = np.zeros(n)
        S2 = np.zeros(n)
        for k in range(-(n - 1), n):
            wk = w[n - 1 + k]
            if wk < 1e-15:
                continue
            i_s = max(0, -k)
            i_e = min(n, n - k)
            S1[i_s:i_e] += wk * x[i_s + k:i_e + k]
            i_s2 = max(0, k)
            i_e2 = min(n, n + k)
            S2[i_s2:i_e2] += wk * x[i_s2 - k:i_e2 - k]
        grad = dx * (1.0 - S1) - dx * S2
        return F, grad

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj_grad, h, jac=True, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
