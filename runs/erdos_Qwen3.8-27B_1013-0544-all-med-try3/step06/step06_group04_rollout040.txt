import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp
import warnings
warnings.filterwarnings('ignore')

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project to feasible set: 0<=h<=1, sum(h)=n/2"""
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                h = h + diff / n
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                h = h + diff / n
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    """Log-sum-exp approximation of max"""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    """SLSQP optimization with smooth max objective"""
    def obj(x):
        c = compute_all_lags(x, n)
        return smooth_max(c, alpha)
    
    def obj_grad(x):
        c = compute_all_lags(x, n)
        # Gradient of smooth max
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            weights = np.exp(shifted - mx)
            weights /= weights.sum()
        else:
            weights = np.exp(shifted)
            weights /= weights.sum()
        # Gradient of each c[k] w.r.t. x
        # c[k] = dx * correlate(x, 1-x, 'full')[k]
        # For lag = k - (n-1):
        # If lag > 0: c = dx * sum_{i=0}^{n-lag-1} x[i]*(1-x[i+lag])
        # grad_x[i] = dx*(1-x[i+lag]) for i < n-lag, and -dx*x[i-lag] for i >= lag
        # If lag < 0: similar with sign flip
        # If lag = 0: c = dx * sum x[i]*(1-x[i]), grad = dx*(1-2x[i])
        dx = 2.0 / n
        grad = np.zeros(n)
        for k_idx in range(len(c)):
            lag = k_idx - (n - 1)
            if abs(weights[k_idx]) < 1e-15:
                continue
            if lag == 0:
                grad += weights[k_idx] * dx * (1 - 2*x)
            elif lag > 0:
                m = lag
                grad[:n-m] += weights[k_idx] * dx * (1 - x[m:])
                grad[m:] += weights[k_idx] * dx * (-x[:n-m])
            else:
                m = -lag
                grad[:n-m] += weights[k_idx] * dx * (1 - x[m:])
                grad[m:] += weights[k_idx] * dx * (-x[:n-m])
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad_weights(c, alpha):
    """Returns weights w_k such that grad smooth_max = sum_k w_k * grad c_k"""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        w /= w.sum()
    else:
        w = np.exp(shifted)
        w /= w.sum()
    return w

def c_lag_gradient(h, n, lag):
    """Gradient of C_lag with respect to h. C_lag = dx * correlate(h, 1-h, 'full')[lag+n-1]"""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1 - 2*h)
    elif m >= n:
        pass  # zero
    else:
        if lag > 0:
            # C = dx * sum_{i=0}^{n-m-1} h[i]*(1-h[i+m])
            # dC/dh[i] = dx*(1-h[i+m]) for i < n-m
            # dC/dh[i+m] = -dx*h[i] for i < n-m, i.e., dC/dh[j] = -dx*h[j-m] for j >= m
            grad[:n-m] += dx * (1 - h[m:])
            grad[m:] += dx * (-h[:n-m])
        else:
            # lag < 0, m = -lag
            # C = dx * sum_{i=0}^{n-m-1} h[i+m]*(1-h[i])
            # dC/dh[i+m] = dx*(1-h[i]) for i < n-m
            # dC/dh[i] = -dx*h[i+m] for i < n-m
            grad[m:] += dx * (1 - h[:n-m])
            grad[:n-m] += dx * (-h[m:])
    return grad

def weighted_lag_gradient(h, n, lags, weights):
    """Sum of weighted gradients of C_lag for given lags and weights"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag, w in zip(lags, weights):
        if abs(w) < 1e-15:
            continue
        m = abs(lag)
        if m == 0:
            grad[:] += w * dx * (1 - 2*h)
        elif m < n:
            if lag > 0:
                grad[:n-m] += w * dx * (1 - h[m:])
                grad[m:] += w * dx * (-h[:n-m])
            else:
                grad[m:] += w * dx * (1 - h[:n-m])
                grad[:n-m] += w * dx * (-h[m:])
    return grad

def slsqp_with_grad(h, n, alpha, maxiter=3000, ftol=1e-15):
    """SLSQP with analytical gradient of smooth max"""
    def obj(x):
        c = compute_all_lags(x, n)
        return smooth_max_value(c, alpha)
    
    def obj_grad(x):
        c = compute_all_lags(x, n)
        w = smooth_max_grad_weights(c, alpha)
        # Only compute gradient for lags with significant weight
        lags = np.arange(-(n-1), n)
        mask = w > 1e-10
        return weighted_lag_gradient(x, n, lags[mask], w[mask])
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent_active(h, n, deadline, top_m=5, lr=0.001, n_iter=5000):
    """Gradient descent on smooth max of top_m lags"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        # Get top_m lags
        top_idx = np.argsort(c)[-top_m:]
        top_lags = top_idx - (n - 1)
        
        # Equal weights for top_m
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / top_m
        
        # Gradient
        grad = weighted_lag_gradient(h_curr, n, top_lags, w[top_idx])
        
        # Move in negative gradient direction (with sum constraint)
        # Project gradient to maintain sum: subtract mean
        grad_centered = grad - grad.mean()
        
        # Adaptive learning rate
        h_new = h_curr - lr * grad_centered
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_curr = h_new
        lr *= 0.999  # Slow decay
    
    return best_h, best_c

**Idea: Find a direction that decreases all top-m lags**

- d maintains the sum constraint (sum(d) = 0)

This is a linear feasibility problem! We can solve it with linprog.

If such a direction exists, we take a step in it. If not, we're at a local minimum (for the smooth max approximation).

This is essentially the "Frank-Wolfe" or "conditional gradient" approach applied to the max of functions.

```python
def find_improving_direction(h, n, top_idx, lags_all):
    """Find a direction that decreases all top lags, or None if infeasible"""
    from scipy.optimize import linprog
    
    # Gradient of each top lag
    grads = []
    for k_idx in top_idx:
        lag = int(lags_all[k_idx])
        grads.append(c_lag_gradient(h, n, lag))
    grads = np.array(grads)  # shape (m, n)
    
    # We want d such that:
    # 1. grads @ d < -1 (decreases all lags by at least 1/n)
    # 2. sum(d) = 0 (maintain integral)
    # 3. -h/d_max <= d <= (1-h)/d_max (maintain bounds, for some step size)
    
    # Simplify: find d with ||d||_1 <= 1, sum(d)=0, and grads @ d <= -1
    # This is an LP
    
    m, nn = grads.shape
    # Variables: d (nn terms)
    # Constraints:
    # -grads @ d <= 1  (i.e., grads @ d >= -1)
    # sum(d) = 0
    # |d_i| <= 1 (for simplicity)
    
    c_obj = np.zeros(nn)  # We just want feasibility
    A_ub = np.vstack([-grads, np.eye(nn), -np.eye(nn)])
    b_ub = np.concatenate([np.ones(m), np.ones(nn), np.ones(nn)])
    A_eq = np.array([np.ones(nn)])
    b_eq = np.array([0.0])
    bounds = [(-1, 1)] * nn
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='hihops')
    
    if res.status == 0:
        return res.x
    return None
```

def line_search_decrease(h, n, top_idx, lags_all, max_step=0.1):

        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha > 0:

    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time
import warnings
warnings.filterwarnings('ignore')

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return w / w_sum

def c_lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif m < n:
        if lag > 0:
            grad[:n - m] += dx * (1.0 - h[m:])
            grad[m:] += -dx * h[:n - m]
        else:
            grad[m:] += dx * (1.0 - h[:n - m])
            grad[:n - m] += -dx * h[m:]
    return grad

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14):
    def obj(x):
        c = compute_all_lags(x, n)
        return smooth_max_value(c, alpha)

    def obj_grad(x):
        c = compute_all_lags(x, n)
        w = smooth_max_weights(c, alpha)
        lags_arr = np.arange(-(n - 1), n)
        mask = w > 1e-10
        if not np.any(mask):
            return np.zeros(n)
        grad = np.zeros(n)
        for idx in np.where(mask)[0]:
            lag = int(lags_arr[idx])
            grad += w[idx] * c_lag_gradient(x, n, lag)
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_descent(h, n, deadline, top_m=5, max_iters=3000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lags_arr = np.arange(-(n - 1), n)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        top_c = c[top_idx]
        current_max = top_c.max()

        grads = np.array([c_lag_gradient(h_curr, n, int(lags_arr[k_idx])) for k_idx in top_idx])
        avg_grad = grads.mean(axis=0)
        avg_grad -= avg_grad.mean()

        if np.linalg.norm(avg_grad) < 1e-12:
            break

        direction = -avg_grad
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-12:
            break
        direction /= norm_d

        # Backtracking line search
        alpha = 0.05
        found = False
        for _ in range(30):
            h_trial = project_feasible(h_curr + alpha * direction, n)
            c_trial = compute_all_lags(h_trial, n)
            trial_max = c_trial[top_idx].max()
            if trial_max < current_max - 1e-10:
                h_curr = h_trial
                c_new = compute_c5(h_curr, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_curr.copy()
                found = True
                break
            alpha *= 0.5
        if not found:
            # Try smaller top_m
            if top_m > 2:
                top_m = max(2, top_m - 1)
            else:
                break

    return best_h, best_c

def local_search_steepest(h, n, deadline, top_m=3):
    h = h.copy()
    best_c = compute_c5(h, n)
    lags_arr = np.arange(-(n - 1), n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    n_sample = min(200, n)

    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_m:]

            for k_idx in top_idx:
                if time.time() > deadline:
                    break
                lag = int(lags_arr[k_idx])
                if lag == 0:
                    continue
                m = abs(lag)
                if n - m < 2:
                    continue

                positions = np.random.choice(n - m, size=min(n_sample, n - m), replace=False)
                for i in positions:
                    j = i + m
                    h_i, h_j = h[i], h[j]
                    if h_i > h_j + 1e-7:
                        delta = min(d, (h_i - h_j) * 0.4)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h_j > h_i + 1e-7:
                        delta = min(d, (h_j - h_i) * 0.4)
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta

    return h, best_c

def heavier_smoothing(h, n, deadline, n_outer=50):
    """Iteratively average with nearby points to smooth out sharp features"""
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_outer):
        if time.time() > deadline:
            break
        h_new = h.copy()
        for i in range(1, n - 1):
            h_new[i] = 0.3 * h[i - 1] + 0.4 * h[i] + 0.3 * h[i + 1]
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            break
    return h, best_c

def subgradient_method(h, n, deadline, top_m=5, n_iter=5000):
    """Subgradient method with Polyak step size"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lags_arr = np.arange(-(n - 1), n)
    best_known = compute_c5(h_curr, n)
    lr = 0.02

    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]

        grads = np.array([c_lag_gradient(h_curr, n, int(lags_arr[k_idx])) for k_idx in top_idx])
        subgrad = grads.sum(axis=0) / top_m
        subgrad -= subgrad.mean()

        h_new = project_feasible(h_curr - lr * subgrad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.95

        if lr < 1e-6:
            break

    return best_h, best_c

def cvx_outer_approx_step(h_curr, n, trust):
    """One outer approximation step using cvxpy LP"""
    import cvxpy as cp
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    lags_arr = np.arange(-(n - 1), n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k_idx in range(2 * n - 1):
        lag = int(lags_arr[k_idx])
        grad = c_lag_gradient(h_curr, n, lag)
        cons.append(c[k_idx] + grad @ (h_var - h_curr) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.CBC, time_limit=30)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return h_curr.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return h_curr.copy()
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

def cvx_outer_approx(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = cvx_outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust *= 0.85
        if trust < 0.003:
            break
    return best_h, best_c
