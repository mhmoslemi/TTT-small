import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def lag_gradient_arr(h, n, k_idx):
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n - m] * dx
    else:
        grad[m:] += (1.0 - h[:n - m]) * dx
        grad[:n - m] -= h[m:] * dx
    return grad


def sum_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_con_jac(x, n):
    return np.ones(n)

    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def _lse_weights(active, alpha):
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 300:
        ev = np.exp(shifted - mx)
    else:
        ev = np.exp(shifted)
    total = np.sum(ev)
    return ev / total


def full_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return _lse_val(c, alpha)


def full_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    w = _lse_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * lag_gradient_arr(x, n, k)
    return grad


def active_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return _lse_val(c[top_idx], alpha)


def active_grad(x, n, alpha, top_idx):

    grad = np.zeros(n)

    return grad


def slsqp_run(h, n, alpha, top_idx, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is None:
        try:
            res = minimize(full_obj, h, args=(n, alpha), jac=full_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,

        # Final gradient search
        if time.time() < deadline - 5:
            h_new, c_new = grad_direction_search(best_h, n_best, 5, deadline - 2, base_step=0.008)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

        # Final pairwise
        if time.time() < deadline - 2:
            h_new, c_new = pairwise_local_search(best_h, n_best, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
