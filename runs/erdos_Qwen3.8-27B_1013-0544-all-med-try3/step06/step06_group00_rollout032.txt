def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return np.clip(h, 0.0, 1.0)
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                return np.clip(h, 0.0, 1.0)
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                return np.clip(h, 0.0, 1.0)
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha, penalty=1000.0):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_s = np.exp(shifted)
    exp_sum = exp_s.sum()
    obj = c_max + np.log(exp_sum) / alpha + penalty * (h.sum() - n/2.0)**2
    
    w = exp_s / exp_sum

    for k in sig:
        m = k - (n - 1)
        wi = w[k]
        # First term: (1-h[i-m]) for valid i
        i_start = max(0, m)
        i_end = min(n, m + n)
        if i_start < i_end:

        # Second term: -h[i+m] for valid i
        j_start = max(0, -m)
        j_end = min(n, n - m)
        if j_start < j_end:
            grad[j_start:j_end] -= wi * h[j_start+m:j_end+m]
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n/2.0)
    return obj, grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                return h
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                return h
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha, penalty):
    dx = 2.0 / n

    obj = c_max + log_sum / alpha
