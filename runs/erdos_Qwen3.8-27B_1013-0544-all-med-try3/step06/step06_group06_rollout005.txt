1. Start with the given h
2. Use L-BFGS-B with the log-sum-exp objective (with penalty for constraint)
3. Gradually increase alpha (from smooth to sharp)
4. Track the best (non-smoothed) C5 value

```python
def optimize_lse(h, n, t_start, t_end, initial_alpha=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mu = 100.0  # penalty parameter
    
    for alpha in [initial_alpha, initial_alpha*10, initial_alpha*100, initial_alpha*1000]:
        if time.time() > t_end - 10:
            break
        
        def obj_and_grad(x):
            x = np.clip(x, 1e-10, 1-1e-10)  # keep in interior for L-BFGS-B
            c = compute_overlaps(x, n)
            ac = alpha * c
            s = logsumexp(ac)
            f = s / alpha + mu * (x.sum() - n/2.0)**2
            p = np.exp(ac - s)
            grad = compute_overlap_grad(x, n, p)
            grad += 2 * mu * (x.sum() - n/2.0)
            return f, grad
        
        x0 = best_h.copy()
        try:
            res = minimize(obj_and_grad, x0, jac=True, method='L-BFGS-B',
                         bounds=[(0,1)]*n,
                         options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
```
