import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot > 0:
                h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot > 0:
                h[mask] += diff / tot * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    
    sh = alpha * (c - m)
    mx_sh = sh.max()
    exp_sh = np.exp(sh - mx_sh)
    w_sum = exp_sh.sum()
    w = exp_sh / w_sum
    
    lse = m + (mx_sh + np.log(w_sum)) / alpha
    
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    thresh = 1e-10
    sig_idx = np.where(w > thresh)[0]
    
    for idx in sig_idx:
        s = int(shifts[idx])
        wi = w[idx]
        j = np.arange(n)
        js = j + s
        jm = j - s
        mask1 = (js >= 0) & (js < n)
        mask2 = (jm >= 0) & (jm < n)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if mask1.any():
            term1[mask1] = 1.0 - h[js[mask1]]
        if mask2.any():
            term2[mask2] = h[jm[mask2]]
        grad += wi * dx * (term1 - term2)
    
    grad /= alpha
    return lse, grad

def projected_gradient_optimize(h, n, deadline, alpha_start=100, alpha_end=100000, 
                                  n_iters=500, beta=100.0):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    momentum = 0.9
    v = np.zeros(n)
    lr = 0.01
    
    target = n / 2.0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        obj, grad = smooth_max_and_grad(h, n, alpha_start)
        # Add constraint penalty gradient
        penalty_grad = 2.0 * beta * (h.sum() - target) * np.ones(n)
        total_grad = grad + penalty_grad
        
        # Momentum update
        v = momentum * v - lr * total_grad
        h_new = h + v
        
        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
        # Decay learning rate

def guided_pair_search(h, n, deadline, deltas, max_sweeps=50):
    """Pair search guided by gradient information"""
    h = h.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    
    for d in deltas:
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            
            # Compute gradient to guide search
            _, grad = smooth_max_and_grad(h, n, 10000)
            
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Determine direction for h[i]
                # If grad[i] > 0, decreasing h[i] helps; if grad[i] < 0, increasing helps
                if grad[i] > 0:
                    # Try decreasing h[i], increasing some j with grad[j] < 0
                    for j in range(n):
                        if j == i:
                            continue
                        if grad[j] < 0 and h[i] - d >= 0 and h[j] + d <= 1:
                            h[i] -= d
                            h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] += d
                                h[j] -= d
                else:
                    # Try increasing h[i], decreasing some j with grad[j] > 0
                    for j in range(n):
                        if j == i:
                            continue
                        if grad[j] > 0 and h[i] + d <= 1 and h[j] - d >= 0:
                            h[i] += d
                            h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
            if not improved:
                break
    
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    """Full pair search - try all pairs"""
    h = h.copy()

def slsqp_optimize(h, n, alpha, maxiter=2000):
    """SLSQP optimization with smooth max objective"""
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        m = c.max()
        sh = alpha * (c - m)
        if sh.max() > 500:
            return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def obj_grad(x):
        return smooth_max_and_grad(x, n, alpha)[1]
    
    def constraint(x):
        return x.sum() - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

            if tot < 1e-15:

        if m2.any():
            t2[m2] = h[jm[m2]]
        grad += wi * (t1 - t2)
    
    return grad / alpha * dx

def slsqp(h, n, alpha, maxiter=2000):
    target = n / 2.0
    def obj(x):
        return smooth_obj(x, n, alpha)
    def obj_grad(x):
        return smooth_obj_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
