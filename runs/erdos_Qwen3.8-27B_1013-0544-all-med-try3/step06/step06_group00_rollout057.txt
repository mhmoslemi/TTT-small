from scipy.optimize import minimize
from scipy.signal import fftconvolve

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    g = np.asarray(h, dtype=float)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(g - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - hp.sum()
    if diff > 1e-14:
        room = 1.0 - hp
        s = room.sum()
        if s > 1e-15:
            hp += (diff / s) * room
    elif diff < -1e-14:
        s = hp.sum()
        if s > 1e-15:
            hp += (diff / s) * hp
    return np.clip(hp, 0.0, 1.0)

def grad_lse(h, alpha):
    """Gradient of logsumexp(alpha * corr) / alpha."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scores = alpha * corr
    smax = scores.max()
    exp_s = np.exp(scores - smax)
    w = exp_s / exp_s.sum()
    
    # FFT-based gradient
    A = fftconvolve(w, 1.0 - h, mode='full')
    B = np.correlate(w, h, mode='full')
    grad = dx * (A[n-1:2*n-1] - B[n-1:2*n-1])
    return grad

def lbgfss_optimize(h, n, alpha, deadline, maxiter=5000):
    """L-BFGS-B optimization of logsumexp(alpha*corr)/alpha with penalty for sum constraint."""
    dx = 2.0 / n
    target = n / 2.0
    pen = 50.0
    
    def fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        scores = alpha * corr
        smax = scores.max()
        lse = smax + np.log(np.exp(scores - smax).sum())
        return lse / alpha + pen * (x.sum() - target) ** 2
    
    def grad(x):
        g = grad_lse(x, alpha)
        g += 2.0 * pen * (x.sum() - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                             'gtol': 1e-8, 'maxfun': maxiter * 3})
        result = project_feasible(res.x, n)
        return result
    except Exception:
        return project_feasible(h, n)

def proj_grad_optimize(h, n, alpha, deadline, maxiter=100000, lr=None):
    """Projected gradient descent."""
    dx = 2.0 / n
    cur = h.copy()
    best = h.copy()
    best_c = compute_c5(h, n)
    
    if lr is None:
        lr = 2.0 / (alpha * dx * n)  # Lipschitz-based
    
    lr_schedule = lr
    for it in range(maxiter):
        if time.time() > deadline:
            break
        if it % 1000 == 0 and lr_schedule > lr * 0.1:
            lr_schedule *= 0.99  # gradual decrease
        
        g = grad_lse(cur, alpha)
        proposal = cur - lr_schedule * g
        cur = project_feasible(proposal, n)
        
        c = compute_c5(cur, n)
        if c < best_c:
            best_c = c
            best = cur.copy()
    
    return best, best_c

def binary_local_search(h_init, n, deadline, max_sweeps=200):
    """Binary swap local search."""
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = h_init.copy()
    # Make binary
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    h = h_b
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                # Compute delta for swapping i and j
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
