```python
def compute_gradient(h, w, n, dx):
    """Compute gradient of smoothed objective.
    h: array of length n
    w: softmax weights, array of length 2n-1 (for the full correlation)
    Returns: gradient array of length n
    """
    grad = np.zeros(n)
    # t_m = m - (n-1) for m = 0, ..., 2n-2
    # dC[m]/dh[j] = (1-h[j-t_m]) * 1{0 <= j-t_m < n} - h[j+t_m] * 1{0 <= j+t_m < n}
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in range(2*n - 1):
            t = m - (n - 1)
            if 0 <= j - t < n:
                s1 += w[m] * (1.0 - h[j - t])
            if 0 <= j + t < n:
                s2 += w[m] * h[j + t]
        grad[j] = dx * (s1 - s2)
    return grad
```
