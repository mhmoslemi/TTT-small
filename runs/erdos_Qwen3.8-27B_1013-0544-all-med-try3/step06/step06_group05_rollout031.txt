            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    mx = c.max()
    shifted = alpha * (c - mx)
    s = shifted.max()
    return float(mx + (s + np.log(np.sum(np.exp(shifted - s)))) / alpha)

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    mx = c.max()
    shifted = alpha * (c - mx)
    s = shifted.max()
    p = np.exp(shifted - s)

            grad[lo1:hi1+1] += w * (1.0 - x[u])
        # Term 2: -h[m+l] for m in [max(0,-l), min(n-1, n-1-l)]
        lo2 = max(0, -l)
        hi2 = min(n1, n1 - l)

            v = np.arange(lo2, hi2 + 1) + l
            grad[lo2:hi2+1] -= w * x[v]

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h = project_feasible(res.x, n)
        return h
    except Exception:
        return h0.copy()

def pair_descent(h, n, deadline, radius=30, deltas=None):
    """Coordinate descent: for each pair (i,j) within radius, find optimal transfer."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    
    n1 = n - 1
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(50):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    # Try transferring d from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try transferring d from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    
    return h, best_c

def optimize_at_n(h0, n, deadline, rng):
    """Run full optimization pipeline at resolution n."""
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [100, 500, 2000, 10000, 50000, 200000]

                best_c = c
                best_h = h_new
    
    # Phase 3: Pair descent

        r = min(30, n // 5)

    # Phase 4: Final SLSQP polish
