def min_1d_smooth_max(c, coeffs, alpha, s_lo, s_hi):
    """Minimize (1/alpha)*log(sum(exp(alpha*(c + s*coeffs)))) for s in [s_lo, s_hi]"""
    v_lo = c + s_lo * coeffs
    v_hi = c + s_hi * coeffs
    
    def deriv(s):
        v = c + s * coeffs
        mx = v.max()
        e = np.exp(alpha * (v - mx))
        return np.dot(coeffs, e) / e.sum()
    
    d_lo = deriv(s_lo)
    d_hi = deriv(s_hi)
    
    if d_lo >= 0:
        return s_lo
    if d_hi <= 0:
        return s_hi
    
    lo, hi = s_lo, s_hi
    for _ in range(50):
        mid = (lo + hi) / 2
        d_mid = deriv(mid)
        if d_mid > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for m in range(n_corr):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + np.log(np.sum(np.exp(sh - mx)) + mx) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (2000, 10.0), (10000, 50.0), (50000, 100.0),
               (200000, 500.0), (1000000, 2000.0)]
    for alpha, mu in configs:

def pair_search(h, n, deadline, deltas=None, radius=80):

                best_h = hf.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
