        best_move = None
        best_nc = cur_c
        for _ in range(n_cand):
            i = zeros_list[rng.randint(len(zeros_list))]
            j = ones_list[rng.randint(len(ones_list))]
            if tabu_until[i] > it or tabu_until[j] > it:
                continue
            delta = compute_delta_swap(h, n, i, j)
            nc = (corr + delta).max() * dx
            if nc < best_nc - 1e-15:
                best_nc = nc
                best_move = (i, j, delta, nc)
        if best_move is not None:
            i, j, delta, nc = best_move
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            ones_list.remove(j)
            zeros_list.remove(i)
            ones_list.append(i)
            zeros_list.append(j)
            tabu_until[i] = it + tabu_len
            tabu_until[j] = it + tabu_len
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    T = 0.02
    cool = 0.99997
    scale = 5000.0
    for it in range(5000000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        if h[i] == 1.0:
            i, j = j, i
        delta = compute_delta_swap(h, n, i, j)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    xo = np.linspace(0, 2.0, n_old)
    xn = np.linspace(0, 2.0, n_new)
    return project_feasible(np.interp(xn, xo, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation from parent
    p1 = t0 + 0.30 * total
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 5:
            break
        h = slsqp_step(h, n, alpha, maxiter=2500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Binary optimization (greedy + tabu)
    p2 = t0 + 0.50 * total
    hb, cb = greedy_binary(best_h, n, t0 + 0.38 * total, max_sweeps=250)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    ht, ct = tabu_search(best_h, n, t0 + 0.45 * total, seed=seed, tabu_len=30, n_cand=50)
    if ct < best_c:
        best_c, best_h = ct, ht.copy()
    ha, ca = anneal_binary(best_h, n, p2, seed=seed + 7)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 3: Re-smooth from best binary
    p3 = t0 + 0.65 * total
    h = best_h.copy()
    alphas2 = [10000, 100000, 1000000, 10000000, 100000000, 1000000000]
    for alpha in alphas2:
        if time.time() > p3 - 4:
            break
        h = slsqp_step(h, n, alpha, maxiter=2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Try n=600
    p4 = t0 + 0.84 * total
    n2 = 600
    h2 = interp_h(best_h, n, n2)
    c2_init = compute_c5(h2, n2)
    hb2, cb2 = greedy_binary(h2, n2, t0 + 0.72 * total, max_sweeps=150)
    if cb2 < c2_init:
        h2 = hb2
    ht2, ct2 = tabu_search(h2, n2, p4 - 30, seed=seed + 2, tabu_len=25, n_cand=40)
    if ct2 < (best_c if n == n2 else float('inf')):
        pass
    if ct2 < best_c:
        best_c, best_h, n = ct2, ht2.copy(), n2
        h2 = best_h.copy()
    elif cb2 < best_c:
        best_c, best_h, n = cb2, hb2.copy(), n2
        h2 = best_h.copy()
    else:
        h2 = interp_h(best_h, n, n2)
    
    alphas3 = [50000, 500000, 5000000, 50000000, 500000000]
    cur_n = n
    for alpha in alphas3:
        if time.time() > p4 - 5:
            break
        h2 = slsqp_step(h2, cur_n, alpha, maxiter=1500)
        c2 = compute_c5(h2, cur_n)
        if c2 < best_c:
            best_c, best_h, n = c2, h2.copy(), cur_n

    # Phase 5: Try n=800 if time permits
    p5 = t0 + 0.93 * total
    if time.time() < p5 - 60:
        n3 = 800
        h3 = interp_h(best_h, n, n3)
        hb3, cb3 = greedy_binary(h3, n3, p5 - 40, max_sweeps=100)
        if cb3 < compute_c5(h3, n3):
            h3 = hb3
        alphas4 = [100000, 1000000, 10000000]
        for alpha in alphas4:
            if time.time() > p5 - 5:
                break
            h3 = slsqp_step(h3, n3, alpha, maxiter=1000)
            c3 = compute_c5(h3, n3)
            if c3 < best_c:
                best_c, best_h, n = c3, h3.copy(), n3

    # Phase 6: Final polish with noise restarts
    p6 = dl - 3
    for trial in range(8):
        if time.time() > p6 - 1:
            break
        noise_level = 0.002 + 0.003 * trial
        h = best_h + noise_level * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [50000000, 500000000, 5000000000]:
            if time.time() > p6 - 0.5:
                break
            h = slsqp_step(h, n, alpha, maxiter=1500)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Final tabu on best solution
    if time.time() < dl - 5:
        hf, cf = tabu_search(best_h, n, dl - 4, seed=seed + 999, tabu_len=20, n_cand=60)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
