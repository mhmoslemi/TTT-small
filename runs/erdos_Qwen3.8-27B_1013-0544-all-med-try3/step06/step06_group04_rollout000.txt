import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of the overlap at a given lag."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def projected_gradient_descent(h, n, deadline, max_iter=50000, lr_init=0.01):

    prev_f = best_c

            break

        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))

    return best_h, best_c

def smooth_max_obj(x, n, alpha, active_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if active_idx is not None:
        c = c[active_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h, n, alpha, maxiter=5000, ftol=1e-16):
    target_sum = n / 2.0
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def obj_con(x):
        return float(np.sum(x) - target_sum)
    def obj_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj_con, 'jac': obj_con_jac}]
    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 2, n)
    roll = rng.randint(1, 8)
    
    if roll == 1:
        # Alternating blocks
        block = n // 4
        h = np.zeros(n)
        for i in range(0, n, block * 2):
            h[i:i+block] = 1.0
    elif roll == 2:
        # Three blocks
        h = np.zeros(n)
        h[:n//4] = 1.0
        h[n//2:3*n//4] = 1.0
        h[7*n//8:] = 1.0
    elif roll == 3:
        # Sawtooth
        h = (np.sin(np.pi * x) > 0.3).astype(float)
    elif roll == 4:
        # Concentrated in center
        center = n // 2
        width = n // 4
        h = np.zeros(n)
        h[center-width:center+width] = 1.0
    elif roll == 5:
        # Two peaks
        h = np.zeros(n)
        h[n//6:n//3] = 1.0
        h[4*n//6:5*n//6] = 1.0
    elif roll == 6:
        # Alternating with period 3
        h = (np.arange(n) % 3 == 0).astype(float)
    else:
        # Random
        h = (rng.rand(n) > 0.5).astype(float)
    
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    
    # === Phase 1: Quick refinement at parent resolution ===
    phase1_deadline = min(deadline - 200, t0 + 80)
    h_ref, c_ref = optimize_at_resolution(best_h, n_init, phase1_deadline, rng)
    if c_ref < best_c:
        best_c = c_ref
        best_h = h_ref
    
    # === Phase 2: Multi-resolution search ===
    resolutions = [200, 250, 350, 400, 450, 500]
    # Sort by priority: closer to n_init first, then higher
    resolutions.sort(key=lambda r: abs(r - n_init) + r * 0.1)
    
    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 60
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        
        if alloc < 40:
            break
        
        # Interpolate best solution to new resolution
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # === Phase 3: Diverse restarts at best resolution ===
    n_best = len(best_h)
    if time.time() < deadline - 60:
        n_restarts = max(2, int((deadline - time.time() - 40) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            
            # Diverse initialization
            if r == 0:
                hp = best_h + 0.01 * rng.randn(n_best)
            elif r == 1:
                hp = best_h + 0.02 * rng.randn(n_best)
            elif r == 2:
                hp = structured_init(n_best, rng)
            elif r == 3:
                # Perturbed parent
                h_init_n = np.interp(np.linspace(0, 2, n_best),
                                     np.linspace(0, 2, n_init), h_init)
                hp = h_init_n + 0.01 * rng.randn(n_best)
            else:
                hp = structured_init(n_best, rng) + 0.01 * rng.randn(n_best)
            
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 40) * 0.35)
            if alloc < 25:
                break
            
            h_opt, c_opt = optimize_at_resolution(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # === Phase 4: Try one more higher resolution if time allows ===
    if time.time() < deadline - 80:
        n_hi = min(600, n_best + 100)
        if n_hi != n_best:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_hi)
            h_start = np.interp(x_new, x_best, best_h)
            h_start = project_feasible(h_start, n_hi)
            alloc = min(90, deadline - time.time() - 50)
            if alloc > 40:
                h_opt, c_opt = optimize_at_resolution(h_start, n_hi, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    
    # === Phase 5: Final intensive polish ===
    if time.time() < deadline - 10:
        n = len(best_h)
        # High-alpha SLSQP
        h_new = slsqp_refine(best_h, n, 1000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # More projected gradient iterations
        if time.time() < deadline - 5:
            h_pg, c_pg = projected_gradient_descent(best_h, n, deadline - 3, max_iter=30000)
            if c_pg < best_c:
                best_c = c_pg
                best_h = h_pg
        
        # Final local search
        if time.time() < deadline - 2:
            best_h, best_c = local_search_focused(best_h, n, deadline - 1, top_lags=5)
    
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
