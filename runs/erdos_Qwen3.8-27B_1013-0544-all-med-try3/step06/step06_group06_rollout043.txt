- The term for m where m+l*=i (i.e., m=i-l*): was h[i-l*](1-h[i]) = 0 (since h[i]=1). Now it's h[i-l*](1-0) = h[i-l*]. Change: +h[i-l*]

- The term for m where m+l*=j (i.e., m=j-l*): was h[j-l*](1-h[j]) = h[j-l*]. Now it's h[j-l*](1-1) = 0. Change: -h[j-l*]

= h[i+l*] + h[i-l*] - h[j+l*] - h[j-l*]

                h[j] = 0

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def sa_search(h_init, n, rng, deadline, max_iters=200000):
    """Simulated annealing on binary solution."""
    h = h_init.copy()
    current_val = compute_c5(h, n)
    best_val = current_val
    best_h = h.copy()
    
    T_init = 0.05
    T_end = 1e-8
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Pick random swap
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        
        new_val = compute_c5(h, n)
        
        if new_val <= current_val:
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            T = T_init * (T_end / T_init) ** (float(it) / max_iters)
            if T > 0 and rng.rand() < np.exp(-(new_val - current_val) / T):
                current_val = new_val
            else:
                h[i] = 1.0
                h[j] = 0.0

        if h[i] == 0:
            # i moved from ones to zeros
            if i in ones_list:
                ones_list.remove(i)
            if i not in zeros_list:
                zeros_list.append(i)

            # j moved from zeros to ones
            if j in zeros_list:
                zeros_list.remove(j)
            if j not in ones_list:
                ones_list.append(j)
