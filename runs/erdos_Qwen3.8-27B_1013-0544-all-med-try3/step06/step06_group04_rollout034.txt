def project_feasible(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """
    Gauss-Newton step to equalize the top_m lag values.
    Returns new h and the equalized value.
    """
    h = h.copy()
    dx = 2.0 / n
    
    for it in range(max_iters):
        c = np.correlate(h, 1 - h, mode='full') * dx
        # Find top m lags
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        t_target = top_vals.mean()
        
        # Build the linear system

        #           sum(delta) = 0

        if top_vals[0] - top_vals[-1] < tol:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return dx * grad

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

def gauss_newton_equalize(h, n, top_m=5, max_iters=100, tol=1e-14):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        c = np.correlate(h, 1 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        t_min = top_vals.min()
        t_max = top_vals.max()
        
        if t_max - t_min < tol:
            # Try to reduce further
            t_target = t_max - 1e-6
        else:
            t_target = t_min
        
        m_top = len(top_idx)
        A = np.zeros((m_top + 1, n))
        b = np.zeros(m_top + 1)
        
        for i, idx in enumerate(top_idx):
            lag = idx - (n - 1)
            A[i] = lag_gradient(h, n, lag)
            b[i] = t_target - c[idx]
        
        A[m_top] = np.ones(n) / np.sqrt(n)
        b[m_top] = 0.0
        
        try:
            AAT = A @ A.T
            AAT += 1e-14 * np.eye(m_top + 1)
            x = np.linalg.solve(AAT, b)
            delta = A.T @ x
        except np.linalg.LinAlgError:
            break
        
        norm_delta = np.linalg.norm(delta)

            break

        step = 1.0
        improved = False

                best_c = c_new
                improved = True
                break
            step *= 0.5
            if step < 1e-10:
                break
        if not improved:
            break
    
    return h, best_c
