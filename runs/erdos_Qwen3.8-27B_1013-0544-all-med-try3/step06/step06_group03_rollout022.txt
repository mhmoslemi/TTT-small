def optimal_transfer(h, n, i, j):
    """Find optimal t to transfer from h[i] to h[j]."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    
    # Compute g_k(i) and g_k(j)
    # g_k(i) is the coefficient of h[i] in corr[n-1+k]
    # corr[n-1+k] = sum_i h[i] * (1-h[i+k])
    # d/dh[i] corr[n-1+k] = (1-h[i+k]) - h[i-k] (for valid indices)
    
    g_i = np.zeros(2*n-1)
    g_j = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        lag = k - (n-1)
        # h[i] appears in corr[k] as:
        # - as first factor: h[i]*(1-h[i+lag]) if 0 <= i+lag < n
        # - as second factor: h[i-lag]*(1-h[i]) if 0 <= i-lag < n, contributes -h[i-lag]*dh[i]
        coeff = 0.0
        if 0 <= i + lag < n:
            coeff += (1 - h[i + lag])
        if 0 <= i - lag < n:
            coeff -= h[i - lag]
        g_i[k] = coeff
        
        coeff = 0.0
        if 0 <= j + lag < n:
            coeff += (1 - h[j + lag])
        if 0 <= j - lag < n:
            coeff -= h[j - lag]
        g_j[k] = coeff
    
    # Transfer t: h[i] += t, h[j] -= t
    # corr[k] changes by t * (g_i[k] - g_j[k])
    slope = g_i - g_j
    base = corr  # current values
    
    # t range: h[i]+t <= 1 => t <= 1-h[i]; h[j]-t >= 0 => t >= -h[j]
    # Also h[i]+t >= 0 => t >= -h[i]; h[j]-t <= 1 => t <= 1-h[j]
    t_min = max(-h[i], -(1-h[j]))  # h[i]+t>=0 => t>=-h[i]; h[j]-t<=1 => t>=h[j]-1
    t_max = min(1-h[i], h[j])       # h[i]+t<=1 => t<=1-h[i]; h[j]-t>=0 => t<=h[j]

    # Minimize over t ∈ [t_min, t_max]
    # Use ternary search (f is convex as max of linear functions)
    
    a, b = t_min, t_max
    for _ in range(50):  # 50 iterations gives precision ~ (b-a)/3^50, more than enough

def compute_all_g(h, n):
    g = np.zeros((2*n-1, n))
    indices = np.arange(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        # h[i+lag] valid where 0 <= i+lag < n
        valid_plus = (indices + lag >= 0) & (indices + lag < n)
        g[k, valid_plus] += (1 - h[indices[valid_plus] + lag])
        # h[i-lag] valid where 0 <= i-lag < n
        valid_minus = (indices - lag >= 0) & (indices - lag < n)
        g[k, valid_minus] -= h[indices[valid_minus] - lag]
    return g

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:

            if tot < 1e-15:

        if mask.sum() > 0:

            h = np.clip(h, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            # Distribute among all
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha < 1:
        return float(m)
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    return float(x.sum() - n / 2.0)

    return np.ones(n)

    """Compute g[k,i] - g[k,j] for all k, vectorized."""
    lags = np.arange(-(n-1), n)
    
    idx_im = i - lags
    vi = (idx_im >= 0) & (idx_im < n)
    g_i = np.zeros(2*n-1)
    g_i[vi] += (1.0 - h[idx_im[vi]])
    
    idx_ip = i + lags
    vp = (idx_ip >= 0) & (idx_ip < n)
    g_i[vp] -= h[idx_ip[vp]]
    
    idx_jm = j - lags
    vj = (idx_jm >= 0) & (idx_jm < n)
    g_j = np.zeros(2*n-1)
    g_j[vj] += (1.0 - h[idx_jm[vj]])
    
    idx_jp = j + lags
    vpj = (idx_jp >= 0) & (idx_jp < n)
    g_j[vpj] -= h[idx_jp[vpj]]
    
    return g_i - g_j
