        infl[:n-m_abs] += (1.0 - h[m_abs:])
        infl[m_abs:] += -h[:n-m_abs]
        return infl

        infl = np.zeros(n)
        infl[m_abs:] += (1.0 - h[:n-m_abs])
        infl[:n-m_abs] += -h[m_abs:]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c


def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _sum_fun(x):
    return x.sum() - len(x) / 2.0


def _sum_jac(x):
    return np.ones(len(x))


def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + logsum / alpha)


def slsqp_run(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()


def compute_influence(h, n, m):
    m_abs = abs(m)
    if m_abs == 0:
        return 1.0 - 2.0 * h
    elif m > 0:
        infl = np.zeros(n)
        infl[:n - m_abs] += (1.0 - h[m_abs:])
        infl[m_abs:] += -h[:n - m_abs]
        return infl
    else:
        infl = np.zeros(n)
        infl[m_abs:] += (1.0 - h[:n - m_abs])
        infl[:n - m_abs] += -h[m_abs:]
        return infl


def lag_targeted_search(h, n, deadline, top_k=4, max_stagnant=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0
    rng = np.random.RandomState(777)

    while time.time() < deadline:
        c_all = compute_c5_vec(best_h, n)

        sorted_lags = np.argsort(c_all)[::-1]
        candidates = [l for l in sorted_lags[:top_k + 2] if l != n - 1][:top_k]
        if not candidates:
            candidates = [sorted_lags[0]]

        improved = False
        for lag_idx in candidates:
            if time.time() > deadline:
                break
            m = lag_idx - (n - 1)
            if m == 0:
                continue

            infl = compute_influence(best_h, n, m)

            can_inc = best_h < 0.9999
            can_dec = best_h > 0.0001

            inc_idx = np.where(can_inc)[0]
            dec_idx = np.where(can_dec)[0]
            if len(inc_idx) == 0 or len(dec_idx) == 0:
                continue

            inc_order = inc_idx[np.argsort(infl[inc_idx])][:20]
            dec_order = dec_idx[np.argsort(infl[dec_idx])[::-1]][:20]

            found = False
            for p in inc_order:
                if time.time() > deadline:
                    break
                for q in dec_order:
                    if p == q:
                        continue
                    gain = infl[p] - infl[q]
                    if gain >= -1e-12:
                        continue
                    max_d = min(1.0 - best_h[p], best_h[q])
                    if max_d < 1e-8:
                        continue
                    for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                        d = max_d * frac
                        old_p, old_q = best_h[p], best_h[q]
                        best_h[p] = old_p + d
                        best_h[q] = old_q - d
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            found = True
                            break
                        best_h[p], best_h[q] = old_p, old_q
                if found:
                    break
            if improved:
                break

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > max_stagnant:
                escaped = False
                for _ in range(30):
                    if time.time() > deadline:
                        break
                    mask_inc = best_h < 0.85
                    mask_dec = best_h > 0.15
                    if not mask_inc.any() or not mask_dec.any():
                        break
                    p = rng.choice(np.where(mask_inc)[0])
                    q = rng.choice(np.where(mask_dec)[0])
                    if p == q:
                        continue
                    d = rng.uniform(0.003, 0.015) * (1 if rng.rand() < 0.5 else -1)
                    best_h[p] += d
                    best_h[q] -= d
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c:
                        best_c = c_new
                        stagnant = 0
                        escaped = True
                        break
                if not escaped:
                    break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def fine_pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    radius = min(40, n // 5)

    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                            best_c = c
                            improved = True
                        else:
                            best_h[i] = hi - d
                            best_h[j] = hj + d
                if improved:
                    break
            if not improved:
                break

    return best_h, best_c


def optimize_resolution(h_start, n, deadline, alphas=None):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000]

    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for a in alphas:
        if time.time() > deadline - 15:

        ht = slsqp_run(best_h, n, a, maxiter=2500)

            best_h = ht.copy()

    if time.time() < deadline - 5:
