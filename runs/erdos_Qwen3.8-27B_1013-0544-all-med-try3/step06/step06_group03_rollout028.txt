```python
    c = np.correlate(h, 1-h, mode='full') * dx

    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
```

3. We want ΔF < 0, so we want grad[i] < grad[j] (increase h where gradient is low, decrease where it's high)

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha
```

**Let me also think about the "active set" approach more carefully.**

The idea:
1. Compute all correlations c[L]
2. Identify the top-k shifts (active set A)
3. Solve: minimize sum_{L in A} c[L] subject to 0<=h<=1, sum(h)=n/2
4. This is a (non-convex) QP
5. Update active set, repeat

For the QP, the objective is:
sum_{L in A} c[L] = sum_{L in A} dx * sum_i h[i]*(1-h[i+L])
= dx * sum_i h[i] * |{L in A: i+L valid}| - dx * sum_{L in A} sum_i h[i]*h[i+L]

The first term is linear in h, the second is quadratic. The Hessian is:
H[j,j'] = -dx * |{L in A: j+L=j' or j'+L=j, both valid}|

This is a symmetric matrix that's negative semidefinite (it's a negative multiple of a graph Laplacian-like matrix). So the QP is actually maximizing a concave function, i.e., minimizing a non-convex function. SLSQP should handle this.

Let me implement this:

```python
def active_set_qp(h, n, k_active, maxiter=500):
    """Minimize sum of top-k correlations using SLSQP."""
    dx = 2.0 / n
    c_all = np.correlate(h, 1-h, mode='full') * dx
    active_idx = np.argsort(c_all)[-k_active:]
    
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1-x, mode='full') * dx
        return c[active_idx].sum()
    
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth_max(x, n, alpha):
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    c = np.correlate(xc, 1.0 - xc, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    s = sh - sh.max()
    return float(m + (sh.max() + np.log(np.sum(np.exp(s)))) / alpha)

def obj_active_set(x, n, active_idx):
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    c = np.correlate(xc, 1.0 - xc, mode='full') * dx
    return float(np.sum(c[active_idx]))

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_grad_matrix(h, n):
    """Compute G[j, idx] = dc[L]/dh[j] for all j and L."""
    dx = 2.0 / n
    G = np.zeros((n, 2 * n - 1))
    for idx in range(2 * n - 1):
        L = idx - (n - 1)
        j1s = max(0, -L)
        j1e = min(n, n - L)
        if j1s < j1e:
            G[j1s:j1e, idx] += (1.0 - h[j1s + L: j1e + L])
        j2s = max(0, L)
        j2e = min(n, n + L)
        if j2s < j2e:
            G[j2s:j2e, idx] -= h[j2s - L: j2e - L]
    G *= dx
    return G

def compute_smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_corr(h, n)
    m = c.max()
    sh = alpha * (c - m)
    s = sh - sh.max()
    w = np.exp(s)
    w /= w.sum()
    G = compute_grad_matrix(h, n)
    return w @ G

def projected_gradient_descent(h, n, alpha, eta0, deadline, patience=200):

    best_h = h.copy()

    no_improve = 0

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:

            best_c = c_new

        else:
            no_improve += 1
            eta *= 0.75
            if eta < 1e-12 or no_improve > patience:

    return best_h, best_c

    # Phase D: SLSQP polish after PGD
    if time.time() < deadline - 80:
        h_new = slsqp_smooth(best_h, n_init, 100000, maxiter=1500)
        c_new = compute_c5(h_new, n_init)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase E: Fast pair search at current resolution
    if time.time() < deadline - 40:
        if n_init <= 300:
            h_new, c_new = fast_pair_search(
                best_h, n_init, deadline - 20,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            h_new, c_new = near_pair_search_fast(
                best_h, n_init, deadline - 20,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
                radius=80)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase F: Try upsampled resolution
    for n_try in [400, 500]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new_arr = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new_arr, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        
        # Optimize at new resolution
        h_new = slsqp_smooth(h_interp, n_try, 5000, maxiter=500)
        c_new = compute_c5(h_new, n_try)
        
        if c_new < best_c:
            h_new = slsqp_smooth(h_new, n_try, 50000, maxiter=800)
            c_new = compute_c5(h_new, n_try)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                
                # Pair search at new resolution
                if time.time() < deadline - 10:
                    h_new2, c_new2 = near_pair_search_fast(
                        best_h, n_try, deadline - 3,
                        [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                        radius=100)
                    if c_new2 < best_c:
                        best_c = c_new2
                        best_h = h_new2
    
    # Phase G: Multi-restart at best resolution
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        h_new, c_new = multi_restart_optimize(best_h, n_cur, deadline - 5, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase H: Final polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_new = slsqp_smooth(best_h, n_final, 200000, maxiter=1000)
        c_new = compute_c5(h_new, n_final)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        if time.time() < deadline - 2:
            if n_final <= 300:
                h_new, c_new = fast_pair_search(
                    best_h, n_final, deadline - 1,
                    [0.001, 0.0005, 0.0002, 0.0001])
            else:
                h_new, c_new = near_pair_search_fast(
                    best_h, n_final, deadline - 1,
                    [0.001, 0.0005, 0.0002], radius=50)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
```
