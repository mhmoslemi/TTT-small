import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    if mx > 500:
        return float(c_max + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha)
    else:
        return float(c_max + np.log(np.sum(np.exp(c_shift))) / alpha)

def smooth_max_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_topm_value(h, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_m < len(c):
        c = np.sort(c)[-top_m:]
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    if mx > 500:
        return float(c_max + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha)
    else:
        return float(c_max + np.log(np.sum(np.exp(c_shift))) / alpha)

def smooth_max_topm_gradient(h, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_m < len(c):
        top_idx = np.argpartition(c, -top_m)[-top_m:]
    else:
        top_idx = np.arange(len(c))
    c_top = c[top_idx]
    c_max = c_top.max()
    c_shift = alpha * (c_top - c_max)
    mx = c_shift.max()
    if mx > 500:
        w = np.exp(c_shift - mx)
    else:
        w = np.exp(c_shift)
    w /= w.sum()
    grad = np.zeros(n)
    for ii, m in enumerate(top_idx):
        if w[ii] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    val = smooth_max_value(h, n, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    g = smooth_max_gradient(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000]
    mus = [1.0, 5.0, 10.0, 50.0, 100.0, 500.0]
    for i in range(len(alphas)):
        if time.time() > deadline - 3:
            break
        alpha = alphas[i]
        mu = mus[min(i, len(mus)-1)]
        t_limit = min(time.time() + 25, deadline - 3)
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def proj_grad_optimize(h, n, deadline, alpha_init=100, top_m=20, lr_init=0.02):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    stall_count = 0
    max_stall = 80
    
    while time.time() < deadline - 2:
        g = smooth_max_topm_gradient(h, n, alpha, top_m)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        step = lr
        improved = False
        for _ in range(60):
            h_try = project_feasible(h - step * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if improved:
            stall_count = 0

def grad_coord_search(h, n, deadline, base_step=0.05):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000]
    for alpha in alphas:
        step = base_step
        while time.time() < deadline - 2:

            g = smooth_max_topm_gradient(h, n, alpha, top_m=10)

            high_idx = np.argsort(g)[-15:]
            low_idx = np.argsort(g)[:15]
            for i in high_idx:
                for j in low_idx:

                        continue
                    d = min(step, h[i], 1.0 - h[j]) * 0.5
                    if d < 1e-12:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)

                        improved = True
                        break
                    else:

                        h[j] -= d
                        h = project_feasible(h, n)
                if improved:

                step *= 0.5
                if step < 1e-9:
                    break
    return h, best_c

def interp_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_interp, n_new)

def pair_local_search(h, n, deadline, max_rounds=50):

    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
