    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    c = compute_all_lags(h, n)
    return float(np.max(c))

def project_box_sum(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_gradient(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        m = -k
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad * dx

    c = compute_all_lags(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

            h = h_new

                m_lags = min(m_lags * 2, 100)

def pair_search_efficient(h, n, deadline, deltas, max_passes=100):
    """Efficient pair search with early termination."""
    best_c = compute_c5(h, n)
    h = h.copy()

    target = n / 2.0

        base = smooth_obj(x, n, alpha)
        sum_pen = (np.sum(x) - target) ** 2 * 100.0
        bound_pen = 0.0
