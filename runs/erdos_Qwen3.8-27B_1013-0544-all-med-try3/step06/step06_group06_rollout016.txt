def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_fft(h, n):
    """Compute C5 using FFT for speed."""
    dx = 2.0 / n
    N = 2 * n  # padding
    H = np.fft.fft(h, N)
    V = np.fft.fft(1 - h, N)
    c = np.real(np.fft.ifft(H * np.conj(V)))[:2*n-1]
    return float(np.max(c) * dx), c
