    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    """Exact projection onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    if abs(h.sum() - target) < 1e-15 and h.min() >= -1e-15 and h.max() <= 1 + 1e-15:
        return np.clip(h, 0.0, 1.0)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad * dx

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max_val(c, alpha)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max_val(c[top_idx], alpha)

    return float(np.sum(x) - n / 2.0)

        if time.time() > deadline - 3:
            break
        
        c = compute_all_lags(h_curr, n)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        cons = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Trust region (L-infinity ball)
        # cp.norm(h_var - h_curr, np.inf) <= trust  # This creates an LP
        # Alternative: h_curr - trust <= h_var <= h_curr + trust
        cons.append(h_var >= h_curr - trust)
        cons.append(h_var <= h_curr + trust)
        
        # Linearized constraints
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            grad = lag_gradient(h_curr, n, lag)
            # c[k] + grad · (h_var - h_curr) <= t_var
            cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        if h_new is None:
            break
        
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        
        # Update trust region
        if step < 1e-6:
            break
        trust = max(0.002, trust * 0.9)
    
    return best_h, best_c

def local_search(h, n, deadline):
    """Coordinate-wise local search on the most active lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                # For lag 0, we can't reduce by moving mass
                # (it's h(1-h) which is minimized at 0 or 1)
                # Try to push values toward 0 or 1
                for i in range(n):
                    if time.time() > deadline:
                        break
                    if 0.05 < h[i] < 0.95:
                        # Push toward nearest boundary
                        if h[i] < 0.5:
                            delta = min(d, h[i])
                            h[i] -= delta
                            h = project_feasible(h, n)
                        else:
                            delta = min(d, 1.0 - h[i])
                            h[i] += delta
                            h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            # Revert (approximately)
                            h = project_feasible(h, n)  # This won't exactly revert
                            # We need a different approach
                            break
                continue
            m = abs(lag)
            if m >= n:
                break
            # Try moving mass from high to low along the active lag
            if lag > 0:
                pairs = [(i, i+m) for i in range(n-m)]
            else:
                pairs = [(i+m, i) for i in range(n-m)]
            
            for (i, j) in pairs:
                if time.time() > deadline:
                    break
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.5)
                    if delta < 1e-10:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.5)
                    if delta < 1e-10:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
    
    return h, best_c

def binary_pattern_thue_morse(n):
    """Generate Thue-Morse binary pattern of length n with n/2 ones."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    # Ensure exactly n//2 ones
    num_ones = tm.sum()
    if num_ones > n // 2:
        # Flip some 1s to 0s
        one_indices = np.where(tm == 1)[0]
        flip = one_indices[:num_ones - n // 2]
        tm[flip] = 0
    elif num_ones < n // 2:
        zero_indices = np.where(tm == 0)[0]
        flip = zero_indices[:n // 2 - num_ones]
        tm[flip] = 1
    return tm.astype(float)

def binary_local_search(h_bin, n, deadline, max_swaps=5000):
    """Local search on binary patterns: swap pairs of 0/1 to reduce max correlation."""
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    
    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        
        m = abs(lag)
        if m >= n:
            break
        
        # Find the worst pair to swap
        if lag > 0:
            # Want to reduce sum h[i]*(1-h[i+m])
            # If h[i]=1, h[i+m]=0: swap to h[i]=0, h[i+m]=1 reduces this lag
            # But might increase other lags
            best_improve = 0
            best_pair = None
            for i in range(0, n - m, max(1, (n-m)//100)):  # Sample for speed
                if h[i] == 1 and h[i+m] == 0:
                    # Try swapping
                    h[i] = 0
                    h[i+m] = 1
                    c_new = compute_c5(h, n)
                    improvement = best_c - c_new
                    if improvement > best_improve:
                        best_improve = improvement
                        best_pair = (i, i+m)
                    h[i] = 1
                    h[i+m] = 0
        else:
            best_improve = 0
            best_pair = None
            for i in range(0, n - m, max(1, (n-m)//100)):
                if h[i+m] == 1 and h[i] == 0:
                    h[i+m] = 0
                    h[i] = 1
                    c_new = compute_c5(h, n)
                    improvement = best_c - c_new
                    if improvement > best_improve:
                        best_improve = improvement
                        best_pair = (i+m, i)
                    h[i+m] = 1
                    h[i] = 0
        
        if best_pair is not None and best_improve > 1e-12:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            best_c = compute_c5(h, n)
        else:
            # Try random swaps
            for _ in range(10):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if h[i] == 1 and h[j] == 0:
                    h[i], h[j] = 0, 1
                    c_new = compute_c5(h, n)
                    if c_new < best_c:
                        best_c = c_new
                        break
                    else:
                        h[i], h[j] = 1, 0
    
    return h, best_c
