    target_ones = n // 2
    best_h = h01.copy()
    best_c5 = compute_c5(h01)
    
    for restart in range(n_restarts):
        if (time.time() - t0) > time_limit:
            break
        h = h01.copy()
        if restart > 0:
            # Perturb: random pair swaps
            for _ in range(50):
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                if len(zeros) == 0 or len(ones) == 0:
                    break
                i = zeros[np.random.randint(len(zeros))]
                j = ones[np.random.randint(len(ones))]
                h[i], h[j] = h[j], h[i]
            h = round_to_01(h, target_ones)
        
        current_c5 = compute_c5(h)
        T = 0.04
        T_min = 0.0005
        cooling = 0.9997
        stuck = 0
        max_stuck = 5000
        
        for it in range(500000):
            if (time.time() - t0) > time_limit:
                break
            if stuck > max_stuck:
                break
            
            r = np.random.random()
            if r < 0.7:
                # Single pair swap
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                if len(zeros) == 0 or len(ones) == 0:
                    break
                i = zeros[np.random.randint(len(zeros))]
                j = ones[np.random.randint(len(ones))]
                if i == j:
                    continue
                h[i] = 1.0
                h[j] = 0.0
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < 0 or np.random.random() < np.exp(-delta / T):
                    current_c5 = new_c5
                    stuck = 0
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i] = 0.0
                    h[j] = 1.0
                    stuck += 1
            elif r < 0.9:
                # Block swap: swap two blocks of consecutive positions
                blen = np.random.randint(1, 6)
                if blen >= n // 2:
                    continue
                i = np.random.randint(0, n - blen)
                j = np.random.randint(0, n - blen)
                if abs(i - j) < blen:
                    continue
                # Ensure the blocks have same number of ones (for valid swap)
                block_i = h[i:i+blen].astype(int)
                block_j = h[j:j+blen].astype(int)
                if block_i.sum() != block_j.sum():
                    continue
                h[i:i+blen] = block_j
                h[j:j+blen] = block_i
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < 0 or np.random.random() < np.exp(-delta / (T * 0.5)):
                    current_c5 = new_c5
                    stuck = 0
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i:i+blen] = block_i
                    h[j:j+blen] = block_j
                    stuck += 1
            else:
                # Reversal
                seg_len = np.random.randint(2, 15)
                i = np.random.randint(0, max(1, n - seg_len))
                seg = h[i:i+seg_len].copy()
                h[i:i+seg_len] = seg[::-1]
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < 0 or np.random.random() < np.exp(-delta / (T * 0.3)):
                    current_c5 = new_c5
                    stuck = 0
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i:i+seg_len] = seg
                    stuck += 1
            
            T = max(T_min, T * cooling)
    
    return best_h, best_c5


def slsqp_refine(h, n, dx, target_sum, t0, time_limit, betas=None):
    """SLSQP refinement with smooth max."""
    if betas is None:
        betas = [50, 200, 1000, 5000]
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        if (time.time() - t0) > time_limit:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(3):
            if (time.time() - t0) > time_limit:
                break
            x0 = best_h.copy()
            if trial > 0:
                scale = 0.01 / trial
                x0 += np.random.randn(n) * scale
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    
    return best_h, best_c5


def proj_grad_refine(h, n, dx, target_sum, t0, time_limit):
    """Projected gradient with momentum on the non-smooth max."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.002
    
    for it in range(200000):
        if (time.time() - t0) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        lag = k_worst - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
        
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, target_sum)
        
        if it % 1000 == 999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.99
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Round to 0-1
    h01 = round_to_01(h_init, n // 2)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Also evaluate the 0-1 version
    c5_01 = compute_c5(h01)
    if c5_01 < best_c5:
        best_c5 = c5_01
        best_h = h01.copy()
    
    # Phase 1: Simulated annealing on 0-1 (n=70)
    if remaining() > 200:
        h01_best, c5_01_best = sa_01(h01, n, t0, 180, n_restarts=5)
        if c5_01_best < best_c5:
            best_c5 = c5_01_best
            best_h = h01_best.copy()
    
    # Phase 2: Continuous refinement from 0-1 point at n=70
    if remaining() > 100:
        # Start from slightly perturbed 0-1
        h_start = best_h.copy()
        h_cont, c5_cont = slsqp_refine(h_start, n, dx, target_sum, t0, 80,
                                         betas=[20, 100, 500, 2000])
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
    
    # Phase 3: Projected gradient refinement
    if remaining() > 60:
        h_pg, c5_pg = proj_grad_refine(best_h, n, dx, target_sum, t0, 50)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Phase 4: Higher resolutions
    for n_try in [100, 140, 200, 100, 160]:
        if remaining() < 120:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        
        # 0-1 SA at this resolution
        if remaining() > 100:
            h01_t = round_to_01(h_t, n_try // 2)
            h01_t_best, c5_01_t = sa_01(h01_t, n_try, t0, 60, n_restarts=3)
            if c5_01_t < best_c5:
                best_c5 = c5_01_t
                best_h = h01_t_best.copy()
        
        # Continuous refinement
        if remaining() > 50:
            h_t2, c5_t2 = slsqp_refine(h_t, n_try, dx_t, ts_t, t0, 40,
                                         betas=[50, 500, 3000])
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2.copy()
                n = n_try
                dx = dx_t
                target_sum = ts_t
    
    # Phase 5: Try structured initializations at n=70
    if remaining() > 80:
        n_s = 70
        ts_s = n_s / 2.0
        dx_s = 2.0 / n_s
        patterns = []
        # Alternating
        alt = np.zeros(n_s)
        alt[::2] = 1.0
        if alt.sum() != ts_s:
            alt = round_to_01(alt, n_s // 2)
        patterns.append(alt)
        # Block pattern: 111000111000...
        blk = np.zeros(n_s)
        for i in range(n_s):
            blk[i] = 1.0 if (i % 6) < 3 else 0.0
        blk = round_to_01(blk, n_s // 2)
        patterns.append(blk)
        # Two-scale: 11001100 within groups
        two_scale = np.zeros(n_s)
        for i in range(n_s):
            g = i % 4
            two_scale[i] = 1.0 if g < 2 else 0.0
        two_scale = round_to_01(two_scale, n_s // 2)
        patterns.append(two_scale)
        
        for pat in patterns:
            if remaining() < 30:
                break
            c5_p = compute_c5(pat)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = pat.copy()
            # Quick SA from this pattern
            if remaining() > 20:
                h_sa, c5_sa = sa_01(pat, n_s, t0, 15, n_restarts=1)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
    
    # Phase 6: Final SLSQP polish
    if remaining() > 20:
        bounds = Bounds(np.zeros(len(best_h)), np.ones(len(best_h)))
        ts = len(best_h) / 2.0
        dx_f = 2.0 / len(best_h)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts, len(best_h))}]
        for beta in [5000, 20000, 100000]:
            if remaining() < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx_f, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Ensure exact feasibility
    final_n = len(best_h)
    final_h = project_feasible(best_h, final_n / 2.0)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, final_n
