def soft_max(c, alpha):
    """Compute soft max and its gradient weights."""
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    weights = exp_c / exp_c.sum()
    value = c.max() + np.log(exp_c.sum()) / alpha
    return value, weights

            if tot < 1e-15:

For a given h, computing all of G is O(n²). Then the gradient is G^T p which is O(n²).

Alternatively, I can compute the gradient directly without forming G:
∇L[j] = dx * sum_k p_k * [(1-h[j+k]) 1_{valid} - h[j-k] 1_{valid}]

This is O(n²) per component... no, O(n²) total since we sum over k for each j.

Wait: for each j, we sum over all k (2n-1 values). Each term is O(1). So total is O(n * 2n) = O(n²). For n=250, that's 125K operations. In a tight numpy loop, this is fast.

But in Python, a loop over j (250 iterations) with numpy operations inside should be fast enough.

Let me just write it as:

```python
def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)  # actual lag
            # First term: (1-h[j+k]) if 0 <= j+k < n
            if 0 <= j+k < n:
                grad[j] += p[k_idx] * (1 - h[j+k])
            # Second term: -h[j-k] if 0 <= j-k < n
            if 0 <= j-k < n:
                grad[j] -= p[k_idx] * h[j-k]
        grad[j] *= dx
    
    return grad
```

This is O(n²) Python-level operations (250 * 499 = 124750), each with a few arithmetic ops. In Python, this might take ~50-100ms per gradient evaluation. That's a bit slow for SLSQP.

def compute_grad(h, n, alpha):

        term2 = np.where(valid2, h[j - lags[valid2]], 0)

        # For each lag value l in lags:
        #   if 0 <= j+l < n: contributes (1-h[j+l])
        #   if 0 <= j-l < n: contributes -h[j-l]
        
        # j + lags gives an array of length 2n-1

        jl_neg = j - lags  # j - each lag
        
        t1 = np.zeros(2*n-1)
        mask1 = (jl >= 0) & (jl < n)
        t1[mask1] = 1 - h[jl[mask1]]
        
        t2 = np.zeros(2*n-1)
        mask2 = (jl_neg >= 0) & (jl_neg < n)
        t2[mask2] = h[jl_neg[mask2]]
        
        grad[j] = dx * (p @ (t1 - t2))

```python
def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    p = exp_c / exp_c.sum()
    
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each j in 0..n-1 and each lag l:
    # term1[j, l] = (1-h[j+l]) if 0<=j+l<n else 0
    # term2[j, l] = h[j-l] if 0<=j-l<n else 0
    
    # Create index arrays
    js = np.arange(n)[:, None]  # shape (n, 1)
    ls = lags[None, :]  # shape (1, 2n-1)
    
    jl = js + ls  # shape (n, 2n-1)
    jl_neg = js - ls  # shape (n, 2n-1)
    
    mask1 = (jl >= 0) & (jl < n)  # shape (n, 2n-1)
    mask2 = (jl_neg >= 0) & (jl_neg < n)  # shape (n, 2n-1)
    
    # term1: (1-h[j+l]) where valid
    term1 = np.zeros((n, 2*n-1))
    term1[mask1] = 1 - h[jl[mask1]]
    
    # term2: h[j-l] where valid
    term2 = np.zeros((n, 2*n-1))
    term2[mask2] = h[jl_neg[mask2]]
    
    # grad[j] = dx * sum_l p[l] * (term1[j,l] - term2[j,l])
    grad = dx * ((term1 - term2) @ p)
    
    return grad
```

This is fully vectorized! The main cost is creating the (n, 2n-1) arrays, which for n=250 is 250*499 = 125K elements. That's fast in numpy.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_push(h, n, t):
    pushed = 0.5 + (h - 0.5) * (1.0 + t)
    return project_feasible(pushed, n)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    ep = exp_c.sum()
    p = exp_c / ep
    lags = np.arange(-(n - 1), n)
    js = np.arange(n)[:, None]
    ls = lags[None, :]
    jl = js + ls
    jl_neg = js - ls
    mask1 = (jl >= 0) & (jl < n)
    mask2 = (jl_neg >= 0) & (jl_neg < n)
    term1 = np.zeros((n, 2 * n - 1))
    term1[mask1] = 1.0 - h[jl[mask1]]
    term2 = np.zeros((n, 2 * n - 1))
    term2[mask2] = h[jl_neg[mask2]]
    return dx * ((term1 - term2) @ p)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cs = c - c.max()
    return c.max() + np.log(np.sum(np.exp(alpha * cs))) / alpha

def slsqp_opt(h, n, alpha, maxiter=1500, use_grad=True):
    bounds = [(0.0, 1.0)] * n
    cons_list = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn / 2.0,
                  'jac': lambda x, nn=n: np.ones(nn)}]
    if use_grad:
        def obj_and_grad(x):
            v = soft_max_obj(x, n, alpha)
            g = compute_grad(x, n, alpha)
            return v, g
        try:
            res = minimize(obj_and_grad, h, method='SLSQP', bounds=bounds,
                           constraints=cons_list, options={'maxiter': maxiter, 'ftol': 1e-15})
            if res.success or res.fun < soft_max_obj(h, n, alpha):
                return project_feasible(res.x, n)
            return h
        except Exception:
            pass
    res = minimize(lambda x: soft_max_obj(x, n, alpha), h, method='SLSQP',
                   bounds=bounds, constraints=cons_list,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.success:
        return project_feasible(res.x, n)
    return h

def local_search_sa(h, n, deadline, rng, T0=0.005, cooling=0.97, min_T=0.0002):
    best_c = compute_c5(h, n)
    current_c = best_c
    T = T0
    dx = 2.0 / n
    c_arr = get_corr(h, n)
    while time.time() < deadline:
        if T < min_T:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dirs = [1, -1]
        d = rng.choice(dirs) * rng.choice([0.01, 0.02, 0.05, 0.0])
        if d == 0:
            d = 0.01
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi
        h[j] = hj
        new_c = compute_c5(h, n)
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
        else:
            h[i] -= d
            h[j] += d
        T *= cooling
    return h, best_c

def pair_search_fine(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_from_start(h_start, n, deadline, rng, budget_frac=1.0):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000, 80000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        alloc_per = max(5, (deadline - time.time()) / max(1, len(alphas) + 1))
        iters = min(1500, int(n * 10))
        ht = slsqp_opt(h, n, a, maxiter=iters)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    return h, best_c

def try_resolution(h_best, n_best, n_try, deadline, rng):
    if n_try == n_best:
        return h_best, compute_c5(h_best, n_best)
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, n_try)
    alloc = min(60, (deadline - time.time()) * 0.5)
    if alloc < 15:
        return h_best, compute_c5(h_best, n_best)
    h_opt, c_opt = optimize_from_start(h_interp, n_try, time.time() + alloc, rng)
    if c_opt < compute_c5(h_best, n_best):
        return h_opt, c_opt
    return h_best, compute_c5(h_best, n_best)

def fourier_init(n, rng):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    n_harmonics = rng.randint(3, 8)
    for _ in range(n_harmonics):
        f = rng.randint(1, 15)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * f * x + phase)
    h = 0.5 + h * 0.6
    return project_feasible(h, n)
