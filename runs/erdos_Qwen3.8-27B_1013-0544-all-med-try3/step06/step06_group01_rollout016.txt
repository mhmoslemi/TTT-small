def lbfgs_optimize(h, n, alpha, penalty, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        soft = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        pen = penalty * (x.sum() - target)**2
        return soft + pen
    
    bounds = [(0.0, 1.0)] * n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            # All at boundary, use the old method
            if diff > 0:
                room = 1.0 - h
                s = room.sum()
                if s > 1e-14:
                    h += diff / s * room
            else:
                s = h.sum()
                if s > 1e-14:
                    h *= target / s
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    soft = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    pen = penalty * (h.sum() - target) ** 2
    return soft + pen

    bounds = [(0.0, 1.0)] * n
    try:

import numpy as np
from scipy.optimize import minimize
import time

_deadline = 0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            if diff > 0:
                room = 1.0 - h
                s = room.sum()
                if s > 1e-14:
                    h += diff / s * room
            else:
                s = h.sum()
                if s > 1e-14:
                    h *= target / s
    return np.clip(h, 0.0, 1.0)

def obj_func(x, n, alpha, penalty):
    if time.time() > _deadline:
        raise ValueError("Deadline exceeded")
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    soft = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    pen = penalty * (x.sum() - target) ** 2
    return soft + pen

def lbfgs_optimize(h, n, alpha, penalty, maxiter, deadline):
    global _deadline
    _deadline = deadline
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds,
                       args=(n, alpha, penalty),
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_optimize(h, n, deadline, n_iters=50000, step0=0.05, decay=0.9999):
    dx = 2.0 / n
    target = n / 2.0
    cur_h = h.copy()
    cur_c = compute_c5(cur_h, n)
    best_h = cur_h.copy()
    best_c = cur_c
    step = step0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(cur_h, 1 - cur_h, mode='full') * dx
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        
        # Compute subgradient
        grad = np.zeros(n)
        if lag >= 0:
            # h[j] as h-position: (1-h[j-lag]) for j >= lag
            grad[lag:] += (1.0 - cur_h[:n-lag])
            # h[j] as 1-h-position: -h[j+lag] for j < n-lag
            grad[:n-lag] -= cur_h[lag:]
        else:
            ll = -lag
            # h[j] as h-position: (1-h[j+ll]) for j < n-ll
            grad[:n-ll] += (1.0 - cur_h[ll:])
            # h[j] as 1-h-position: -h[j-ll] for j >= ll
            grad[ll:] -= cur_h[:n-ll]
        
        cur_h = cur_h - step * grad
        cur_h = project_feasible(cur_h, n)
        
        c = compute_c5(cur_h, n)
        if c < best_c:
            best_c = c
            best_h = cur_h.copy()
        
        step = max(0.001, step * decay)
    
    return best_h, best_c

def compute_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_sa_search(h, n, deadline, seed, n_greedy_sweeps=100, sa_iterations=200000):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    
    # Round to binary
    target_count = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target_count]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Greedy phase
    for sweep in range(n_greedy_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = zeros.copy()
        o_list = ones.copy()
        for i in z_list:
            if h[i] != 0:
                continue
            for j in o_list:
                if h[j] != 1:
                    continue
                delta = compute_swap_delta(h, n, i, j, m_range)
                new_corr = corr + delta
                nc = float(new_corr.max() * dx)
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = new_corr
                    cur_c = nc
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
            if improved:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    
    # SA phase
    T0 = 0.05
    T_end = 0.001
    n_sa = sa_iterations
    for it in range(n_sa):
        if time.time() > deadline:
            break
        progress = it / n_sa
        T = T0 * (T_end / T0) ** progress
        
        # Random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        delta = compute_swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = float(new_corr.max() * dx)
        delta_c = nc - cur_c
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c
