def fast_correlate(a, b):
    """Compute full linear cross-correlation using FFT"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Next power of 2 for efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.fft(a, fsize)
    B = np.fft.fft(b, fsize)
    corr = np.fft.ifft(A * np.conj(B)).real
    # The first n+m-1 elements give the 'full' correlation
    # But we need to match np.correlate's convention
    # np.correlate(a, b, 'full')[k] = sum_j a[j+k] * b[j] (with appropriate ranges)
    # FFT gives circular correlation, which for zero-padded input gives linear correlation
    return corr[:size]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def sm_obj(h, n, alpha):
    c = compute_c5_all(h, n)
    return smooth_max_val(c, alpha)

def sm_grad(h, n, alpha):
    c = compute_c5_all(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_continuation(h, n, deadline, alphas=None):

        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h = project_sum(h, n / 2.0)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        if remaining < 3:
            break
        max_iter = min(2000, max(50, int(remaining * 20)))
        
        def obj(x):
            return sm_obj(x, n, alpha)
        def grad(x):
            return sm_grad(x, n, alpha)
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iter * 10,
                                   'maxls': 40, 'iprint': -1})
            h_new = project_sum(res.x, n / 2.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=2000):

        return sm_obj(x, n, alpha)

def fine_pair_search(h, n, deadline, steps=None, radius=None):
    if steps is None:
        steps = [0.001, 0.0005, 0.0001, 0.00005, 0.00001, 0.000005, 0.000001]
    
    h = project_sum(h, n / 2.0)
    best_c = compute_c5(h, n)
    
    for delta in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move from i to j
                    if h[i] >= delta and h[j] <= 1.0 - delta:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    # Move from j to i
                    elif h[j] >= delta and h[i] <= 1.0 - delta:
                        h[j] -= delta
                        h[i] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
    
    return h, best_c

def two_point_refine(h, n, deadline, alpha=500000):
    """Refine by trying to optimally pair up two points."""
    h = project_sum(h, n, n / 2.0)
    best_c = compute_c5(h, n)
    
    # Find the lags that achieve (near) maximum
    c = compute_c5_all(h, n)
    top_lags = np.argsort(c)[-10:]
    
    # For each top lag, try to redistribute to reduce it
    for lag in top_lags:
        if time.time() > deadline:
            return h, best_c
        k = lag - (n - 1)
        if k >= 0:
            pairs = [(i, i + k) for i in range(n - k)]
        else:
            kk = -k
            pairs = [(i + kk, i) for i in range(n - kk)]
        
        for (i, j) in pairs:
            if time.time() > deadline:
                return h, best_c
            # Try to reduce correlation at this lag by adjusting h[i] and h[j]
            # The contribution to c_k from this pair is h[i]*(1-h[j])
            # We want to reduce it while maintaining sum constraint
            # Try small adjustments
            for eps in [0.001, 0.0005, 0.0001, 0.00005]:
                # Increase h[j] (reduces h[i]*(1-h[j])) and decrease h[i]
                if h[j] + eps <= 1.0 and h[i] - eps >= 0.0:
                    h[i] -= eps
                    h[j] += eps
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                    else:
                        h[i] += eps
                        h[j] -= eps
                # Decrease h[j] and increase h[i] (might help other lags)
                if h[i] + eps <= 1.0 and h[j] - eps >= 0.0:
                    h[i] += eps
                    h[j] -= eps
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                    else:
                        h[i] -= eps
                        h[j] += eps
    
    return h, best_c

def binary_init(n, seed):

            best_h = h2
    
    # Phase 3: Gradient line search
    if time.time() < deadline - 150:
        t3 = min(time.time() + 50, deadline - 150)
        h3, c3 = grad_line_search(best_h, best_n, t3, alpha=200000, step_init=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Fine pair search
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 120)
        h4, c4 = fine_pair_search(best_h, best_n, t4,
            steps=[0.001, 0.0005, 0.0001, 0.00005, 0.00001, 0.000005],
            radius=40)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Try higher resolution
    for n_try in [best_n + 50, best_n + 100, best_n + 200]:
        if time.time() > deadline - 100:
            break
        remaining = deadline - time.time()
        if remaining < 80:
            break
        
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)
        
        alloc = min(remaining - 60, 100)
        if alloc < 40:
            break
        
        h_t, c_t = lbfgs_continuation(h_interp, n_try, time.time() + alloc * 0.6,
            alphas=[100, 1000, 10000, 100000, 500000])
        
        if time.time() < deadline - 60:
            h_t, c_t = slsqp_optimize(h_t, n_try, time.time() + min(alloc * 0.2, 30),
                alpha=500000, maxiter=1000)
        
        if time.time() < deadline - 50:
            h_t, c_t = fine_pair_search(h_t, n_try, min(time.time() + 20, deadline - 40),
                steps=[0.001, 0.0001, 0.00001], radius=30)
        
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try
    
    # Phase 6: Random restarts (small perturbations)
    for restart in range(3):
        if time.time() > deadline - 80:
            break
        noise_scale = [0.02, 0.01, 0.005][restart]
        h_restart = best_h + rng.randn(best_n) * noise_scale
        h_restart = project_sum(h_restart, best_n / 2.0)
        
        alloc = min(40, deadline - time.time() - 40)
        if alloc < 20:
            break
        
        h_r, c_r = lbfgs_continuation(h_restart, best_n, time.time() + alloc * 0.7,
            alphas=[1000, 10000, 100000])
        
        if time.time() < deadline - 30:
            h_r, c_r = fine_pair_search(h_r, best_n, min(time.time() + 10, deadline - 25),
                steps=[0.001, 0.0001], radius=20)
        
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    
    # Phase 7: Final SLSQP + very fine pair search
    if time.time() < deadline - 15:
        h_f, c_f = slsqp_optimize(best_h, best_n, deadline - 10, alpha=2000000, maxiter=1000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    if time.time() < deadline - 3:
        h_f, c_f = fine_pair_search(best_h, best_n, deadline - 1,
            steps=[0.00005, 0.00001, 0.000005, 0.000001],
            radius=20)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    # Final verification
    best_h = project_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
