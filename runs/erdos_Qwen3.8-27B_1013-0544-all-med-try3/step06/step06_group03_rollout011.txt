def gradient_informed_pair_search(h, n, deadline, max_rounds=100):
    best_c = compute_c5(h, n)
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = np.correlate(h, 1-h, mode='full') * (2.0/n)
        current_max = c_all.max()
        active_idx = np.argmax(c_all)
        lag = active_idx - (n-1)
        
        # Compute gradient of active lag
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1 - h[i+lag])
                grad[i+lag] -= h[i]
        else:
            llag = -lag
            for i in range(n - llag):
                grad[i] += (1 - h[i+llag])
                grad[i+llag] -= h[i]
        
        # Sort by gradient (high to low)
        order = np.argsort(-grad)
        
        # Try moving mass from high-grad to low-grad
        step = 0.01 / (1 + 0.05 * round_num)
        improved = False
        
        for a_pos in range(min(50, n)):
            if time.time() > deadline:
                break
            i = order[a_pos]
            for b_pos in range(n - min(50, n), n):
                j = order[b_pos]
                if i == j:
                    continue
                # Try moving from i to j
                if h[i] - step >= 0 and h[j] + step <= 1:
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] += step
                        h[j] -= step
        
        if not improved:
            # Try smaller step
            step = step * 0.5
            if step < 1e-6:
                break
    
    return h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    max_shifted = shifted.max()
    if max_shifted > 500:
        return float(m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(shifted)))) / alpha)

def _sum_fn(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < compute_c5(h, n):
            return h_new
    except:
        pass
    return h.copy()

def active_lag_search(h, n, deadline, n_top=30):
    """Pair search focused on the active lag, using gradient information."""
    best_c = compute_c5(h, n)
    c_all = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            
            c_all = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            current_max = c_all.max()
            active_idx = np.argmax(c_all)
            lag = active_idx - (n - 1)
            llag = abs(lag)
            
            # Compute gradient of active lag w.r.t. h
            grad = np.zeros(n)
            for i in range(n - llag):
                grad[i] += (1.0 - h[i + llag])
                grad[i + llag] -= h[i]
            
            # Also compute "risk" - how much other lags would increase
            # Approximate: for each coordinate, the max gradient magnitude over other lags
            # (This is expensive, so skip for now)
            
            # Sort by gradient: highest first (most beneficial to decrease)
            order = np.argsort(-grad)
            
            # Try moving from high-grad to low-grad coordinates
            tried = set()
            for ai in range(min(n_top, n)):
                if time.time() > deadline:
                    return h, best_c
                i = order[ai]
                for bi in range(max(0, n - n_top), n):
                    j = order[bi]
                    if i == j or (min(i,j), max(i,j)) in tried:
                        continue
                    tried.add((min(i,j), max(i,j)))
                    
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += step
                            h[j] -= step
                    elif h[j] - step >= 0 and h[i] + step <= 1:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[j] += step
                            h[i] -= step
    
    return h, best_c

def near_pair_search(h, n, deadline, radius=50):
    """Near pair search (from previous, slightly modified)."""
    best_c = compute_c5(h, n)
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c
