def compute_profiles(h, n, ks):
    ts = np.arange(n)
    TK = ts[:, None] + ks[None, :]
    TMK = ts[:, None] - ks[None, :]
    valid_TK = (TK >= 0) & (TK < n)
    valid_TMK = (TMK >= 0) & (TMK < n)
    TK_safe = np.clip(TK, 0, n-1)
    TMK_safe = np.clip(TMK, 0, n-1)
    P = np.zeros((n, 2*n-1))
    P[valid_TK] = (1.0 - h[TK_safe[valid_TK]])
    P[valid_TMK] += h[TMK_safe[valid_TMK]]
    return P
