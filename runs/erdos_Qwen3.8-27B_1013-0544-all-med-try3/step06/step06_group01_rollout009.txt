    dx = 2.0 / n

    best_c = compute_c5(h, n)

        shifts = top_idx - (n - 1)

        weights = np.exp(100 * (corr[top_idx] - corr.max()))
        weights /= weights.sum()

        for w, shift in zip(weights, shifts):
            for i in range(n):
                val = 0.0
                j1 = i + shift
                j2 = i - shift
                if 0 <= j1 < n:
                    val += (1.0 - curr[j1])
                if 0 <= j2 < n:
                    val -= curr[j2]
                sg[i] += w * val

        step = step_base / np.sqrt(it + 1)
        curr = project_feasible(curr - step * sg, n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
