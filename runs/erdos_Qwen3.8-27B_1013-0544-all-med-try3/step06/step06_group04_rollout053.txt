import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0 or m >= n:
        if m == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        return grad
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n-m] * dx
    else:
        grad[:n-m] -= (1.0 - h[m:]) * dx
        grad[m:] += h[:n-m] * dx
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.minimum(shifted, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    return np.sum(x) - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, deadline, alpha=50000, max_iter=50000):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 2.0 / (alpha * dx)  # ~1/L where L is Lipschitz constant
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        m = c.max()
        shifted = alpha * (c - m)
        shifted = np.minimum(shifted, 500)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-12:
                continue
            lag = k - (n - 1)
            grad += w * lag_gradient(h, n, lag)

        step = lr
        for _bt in range(20):
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new <= m + 1e-12 or step < 1e-10:

def binary_snap(h, n, threshold=0.01):
    """Snap values close to 0 or 1 to exactly 0 or 1, then re-optimize."""
    h = h.copy()
    h[h < threshold] = 0
    h[h > 1 - threshold] = 1
    # Adjust the remaining values to satisfy the sum constraint
    free = (h > 0) & (h < 1)
    if free.sum() == 0:
        return project_feasible(h, n)
    target = n/2.0 - h.sum()
    # Distribute the adjustment among free variables
    if target > 0:
        room = (1.0 - h)[free]
        tot = room.sum()
        if tot < 1e-14:
            return project_feasible(h, n)
        h[free] += (target / tot) * room
    else:
        tot = h[free].sum()
        if tot < 1e-14:
            return project_feasible(h, n)
        # Reduce free variables proportionally
        h[free] += (target / tot) * h[free]
    return project_feasible(h, n)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0 or m >= n:
        if m == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        return grad
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n - m] * dx
    else:
        grad[:n - m] -= (1.0 - h[m:]) * dx
        grad[m:] += h[:n - m] * dx
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = np.minimum(alpha * (c - m), 500.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def projected_gradient(h, n, deadline, alpha=30000, max_iter=20000):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    initial_c = best_c
    lr = 1.0 / (alpha * dx * 0.5)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        m = c.max()
        shifted = np.minimum(alpha * (c - m), 500.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        grad = np.zeros(n)
        significant = weights > 1e-10
        for k in np.where(significant)[0]:
            lag = k - (n - 1)
            grad += weights[k] * lag_gradient(h, n, lag)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 3.0 / (alpha * dx))

            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-12:

    return best_h, best_c

def worst_lag_descent(h, n, deadline, max_iter=50000):
