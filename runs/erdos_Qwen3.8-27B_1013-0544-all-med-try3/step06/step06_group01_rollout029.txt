    Assumes h[pos] != new_val before the call.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def flip_update(h, corr, pos, n):
    """
    Flip h[pos] (0->1 or 1->0). Update h and corr in-place.
    Must be called with h[pos] being 0 or 1 (binary).
    """
    a = pos
    if h[a] == 0:
        # 0 -> 1
        # Term A: for k in [-a, n-1-a], k != 0: corr[k+n-1] -= h[a+k]
        for k in range(-a, n - a):
            if k == 0:
                continue
            corr[k + n - 1] -= h[a + k]
        # Term B: for k in [a-n+1, a], k != 0: corr[k+n-1] += 1 - h[a-k]
        for k in range(a - n + 1, a + 1):
            if k == 0:
                continue
            corr[k + n - 1] += 1.0 - h[a - k]
    else:
        # 1 -> 0
        for k in range(-a, n - a):
            if k == 0:
                continue
            corr[k + n - 1] += h[a + k]
        for k in range(a - n + 1, a + 1):
            if k == 0:
                continue
            corr[k + n - 1] -= 1.0 - h[a - k]
    h[a] = 1.0 - h[a]

def sa_binary(h_init, n, deadline, seed=42, T0=0.05, double_prob=0.2, max_time=None):
    """Simulated annealing on binary solutions with incremental correlation updates."""
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    dx = 2.0 / n
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    t_start = time.time()
    total_time = max(deadline - t_start, 1.0)
    T_end = 1e-8
    cool_rate = np.log(T0 / T_end) / total_time
    
    n_no_improve = 0
    patience = 50000
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = T0 * np.exp(-cool_rate * elapsed)
        
        # Choose move type
        if double_prob > 0 and len(ones) >= 2 and len(zeros) >= 2 and rng.random() < double_prob:
            # Double swap
            ia = rng.randint(len(zeros))
            ib = rng.randint(len(zeros) - 1)
            if ib >= ia:
                ib += 1
            a, b = int(zeros[ia]), int(zeros[ib])
            ii = rng.randint(len(ones))
            ij = rng.randint(len(ones) - 1)
            if ij >= ii:
                ij += 1
            i1, i2 = int(ones[ii]), int(ones[ij])
            
            old_h = h.copy()
            old_corr = corr.copy()
            
            # Apply 4 flips
            flip_update(h, corr, a, n)
            flip_update(h, corr, b, n)
            flip_update(h, corr, i1, n)
            flip_update(h, corr, i2, n)
            
            new_c = float(corr.max() * dx)
            delta = new_c - cur_c
            
            if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
                cur_c = new_c
                # Update sets
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    n_no_improve = 0
            else:
                # Revert
                h[:] = old_h
                corr[:] = old_corr
        else:
            # Single swap
            a = int(rng.choice(zeros))
            i1 = int(rng.choice(ones))
            
            old_h_a = h[a]
            old_h_i = h[i1]
            old_corr = corr.copy()
            
            flip_update(h, corr, a, n)
            flip_update(h, corr, i1, n)
            
            new_c = float(corr.max() * dx)
            delta = new_c - cur_c
            
            if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
                cur_c = new_c
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    n_no_improve = 0
            else:
                h[a] = old_h_a
                h[i1] = old_h_i
                corr[:] = old_corr
        
        n_no_improve += 1
        # Optional: adaptive cooling
        # if n_no_improve > patience:
        #     break
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max() if sh.size > 0 else 0
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
