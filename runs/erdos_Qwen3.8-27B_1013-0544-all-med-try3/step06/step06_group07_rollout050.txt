import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_sum(h, target_sum):
    return h.sum() - target_sum

def con_sum_jac(h, target_sum):
    return np.ones_like(h)

def corr_subgrad(h, lag_index, n):
    """Compute subgradient of corr[lag_index] w.r.t. h"""
    # lag_index is index in the full correlation array (length 2n-1)
    # Center is at n-1
    k = lag_index - (n - 1)  # actual lag
    grad = np.zeros(n)
    if k >= 0:
        # corr = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        # d/dh[i]: (1-h[i-k]) if k<=i<=n-1, minus h[i+k] if 0<=i<=n-1-k
        for i in range(n):
            val = 0.0
            if k <= i <= n - 1:
                val += (1.0 - h[i - k])
            if 0 <= i <= n - 1 - k:
                val -= h[i + k]
            grad[i] = val
    else:
        # negative lag, let l = -k
        l = -k
        # corr = sum_{j=l}^{n-1} h[j] * (1-h[j-l])
        # d/dh[i]: (1-h[i-l]) if l<=i<=n-1, minus h[i+l] if 0<=i<=n-1-l
        for i in range(n):
            val = 0.0
            if l <= i <= n - 1:
                val += (1.0 - h[i - l])
            if 0 <= i <= n - 1 - l:
                val -= h[i + l]
            grad[i] = val
    return grad

def corr_subgrad_vec(h, lag_index, n):
    """Vectorized version of corr_subgrad"""
    k = lag_index - (n - 1)
    l = abs(k)
    grad = np.zeros(n)
    # Contribution from h[i] as the "first" factor: (1-h[i-l]) for i in [l, n-1]
    idx1 = np.arange(l, n)
    grad[idx1] += (1.0 - h[idx1 - l])
    # Contribution from h[i] as the "second" factor: -h[i+l] for i in [0, n-1-l]
    idx2 = np.arange(0, n - l)
    grad[idx2] -= h[idx2 + l]
    return grad

def run_slsqp(h0, n, beta, maxiter, target_sum, t0, budget_s):
    """Run SLSQP with smooth max objective"""
    if time.time() - t0 > budget_s - 5:
        return None, float('inf')
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, float('inf')

def run_focused_slsqp(h0, n, top_k, maxiter, target_sum, t0, budget_s):
    """Run SLSQP minimizing sum of top_k worst lags"""
    if time.time() - t0 > budget_s - 5:
        return None, float('inf')
    dx = 2.0 / n
    corr = get_corr(h0)
    top_indices = np.argsort(corr)[-top_k:]
    lag_weights = np.zeros(2 * n - 1)
    lag_weights[top_indices] = 1.0 / top_k
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac, 'args': (target_sum,)}]

        h_new = project_feasible(res.x, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, float('inf')

    """Weighted sum of specific lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.dot(lag_weights, corr) * dx
