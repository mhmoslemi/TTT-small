def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:

            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _logsumexp(arr):
    m = arr.max()
    if m > 500:
        return m + np.log(np.sum(np.exp(arr - m)))
    return m + np.log(np.sum(np.exp(arr - m)))

    def obj(x):
        c = compute_all_lags(x, n)
        if top_idx is not None:
            return _logsumexp(alpha * c[top_idx]) / alpha
        return _logsumexp(alpha * c) / alpha
    return obj

                      method='SLSQP', bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time
import warnings
warnings.filterwarnings('ignore')


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)


def _sum_con_jac(x, n):
    return np.ones(n)


def _obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def _obj_active(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    act = c[top_idx]
    m = act.max()
    shifted = alpha * (act - m)
    sm = shifted.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_step(h, n, alpha, top_idx=None, maxiter=5000, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        try:
            res = minimize(_obj_active, h, args=(n, top_idx, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()
    else:
        try:
            res = minimize(_obj_full, h, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            return project_feasible(res.x, n)
        except Exception:
            return h.copy()

    # Stage 1: Coarse full optimization
    for alpha in [200, 1000, 5000]:

            best_c = compute_c5(h, n)

def perturb_and_optimize(h, n, deadline, rng, scale=0.02):
    """Perturb and re-optimize to escape local optima."""
    h_pert = h + rng.normal(0, scale, n)
    h_pert = project_feasible(h_pert, n)
    c0 = compute_c5(h_pert, n)

        h_pert = project_feasible(h_pert, n)
