import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Numerical stability
    max_shifted = shifted.max()
    if max_shifted > 500:
        log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return float(m + log_sum / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

Let me also think about the ternary search more carefully. The function C5(δ) = max_t(c_t + δ*a_t) is convex but piecewise linear. Ternary search works for convex functions, but it's slow to converge near the minimum (since the function is flat or has kinks). 

A better approach for piecewise-linear convex functions:
- The minimum is at a kink (where two lines tie) or at a boundary
- We can find it by: start at δ=0, find the max line, then move in the direction of decreasing max until we hit a kink or boundary
- At each kink, re-evaluate

But this requires finding kinks, which is O(n²) in the worst case. For practical purposes, ternary search with 30-50 iterations should be sufficient.

Actually, here's an even better idea: since C5(δ) is the max of linear functions, the subgradient at δ is a_t for the argmax t. We can use a subgradient method:
- Start at δ = 0
- Compute subgradient g = a[argmax_t(c_t + δ*a_t)]
- If g < 0, decrease δ; if g > 0, increase δ
- Step size: η * |g| (or some adaptive scheme)
- Stop when |g| is small or we hit a boundary

This is like projected subgradient descent in 1D, which should converge fast for piecewise-linear convex functions.

But actually, for 1D convex optimization, the simplest and most reliable method is just bisection on the subgradient:
- If subgradient > 0, the minimum is to the left → b_tern = mid
- If subgradient < 0, the minimum is to the right → a_tern = mid
- If subgradient = 0 (or contains 0), we're at the minimum

This converges in O(log(range/precision)) iterations, which is about 30-40 for our precision needs.

        vals = c + mid * a
        max_val = vals.max()

        # For bisection, we just need the sign
        # If a[argmax] > 0, function is increasing → min is to the left
        # If a[argmax] < 0, function is decreasing → min is to the right
        # If a[argmax] == 0, we're at the min
        # But there might be multiple maximizers. Use the average subgradient.
        mask = vals >= max_val - 1e-15
        subgrad = a[mask].mean()
        
        if subgrad > 0:
            hi = mid
        elif subgrad < 0:
            lo = mid
        else:
            break

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return float(m + log_sum / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

                       bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return float(m + log_sum / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_a_vector(h, n, i, j):
    t = np.arange(2 * n - 1)
    idx_ip = i + t - (n - 1)
    idx_im = i - t + (n - 1)
    idx_jp = j + t - (n - 1)
    idx_jm = j - t + (n - 1)
    a = np.zeros(2 * n - 1)
    m1 = (idx_ip >= 0) & (idx_ip < n)
    a[m1] -= h[idx_ip[m1]]
    m2 = (idx_im >= 0) & (idx_im < n)
    a[m2] -= h[idx_im[m2]]
    m3 = (idx_jp >= 0) & (idx_jp < n)
    a[m3] += h[idx_jp[m3]]
    m4 = (idx_jm >= 0) & (idx_jm < n)
    a[m4] += h[idx_jm[m4]]
    return a

def find_optimal_delta(c, a, lo, hi, dx):
    if lo >= hi:
        return 0.0, float(c.max() * dx)
    a_lo = lo
    a_hi = hi
    for _ in range(45):
        mid = (a_lo + a_hi) * 0.5
        vals = c + mid * a
        max_val = vals.max()
        mask = vals >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        subgrad = a[mask].mean()
        if subgrad > 1e-15:
            a_hi = mid
        elif subgrad < -1e-15:
            a_lo = mid
        else:
            a_lo = a_hi = mid
            break
    candidates = [a_lo, a_hi, (a_lo + a_hi) * 0.5, 0.0]
    best_delta = 0.0
    best_val = float(c.max() * dx)
    for d in candidates:
        if a_lo - 1e-15 <= d <= a_hi + 1e-15:
            d = max(a_lo, min(a_hi, d))
            v = float((c + d * a).max() * dx)
            if v < best_val:
                best_val = v
                best_delta = d
    return best_delta, best_val

def local_search_random(h, n, c, best_c, deadline, rng, max_iter):
    h = h.copy()
    c = c.copy()
    dx = 2.0 / n
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        a = compute_a_vector(h, n, i, j)
        lo1 = max(-h[i], h[j] - 1.0)
        hi1 = min(1.0 - h[i], h[j])
        if lo1 < hi1:
            d1, v1 = find_optimal_delta(c, a, lo1, hi1, dx)
            if v1 < best_c - 1e-15:
                h[i] += d1
                h[j] -= d1
                c = np.correlate(h, 1.0 - h, mode='full')
                best_c = float(c.max() * dx)
                continue
        lo2 = max(-h[j], h[i] - 1.0)
        hi2 = min(1.0 - h[j], h[i])
        if lo2 < hi2:
            d2, v2 = find_optimal_delta(c, -a, lo2, hi2, dx)
            if v2 < best_c - 1e-15:
                h[i] -= d2
                h[j] += d2
                c = np.correlate(h, 1.0 - h, mode='full')
                best_c = float(c.max() * dx)
    return h, c, best_c

def systematic_sweep(h, n, c, best_c, deadline, dx):
    h = h.copy()
    c = c.copy()
    for i in range(n):
        if time.time() > deadline:
            break
        for j in range(i + 1, n):
            if time.time() > deadline:
                break
            a = compute_a_vector(h, n, i, j)
            lo1 = max(-h[i], h[j] - 1.0)
            hi1 = min(1.0 - h[i], h[j])
            if lo1 < hi1:
                d1, v1 = find_optimal_delta(c, a, lo1, hi1, dx)
                if v1 < best_c - 1e-15:
                    h[i] += d1
                    h[j] -= d1
                    c = np.correlate(h, 1.0 - h, mode='full')
                    best_c = float(c.max() * dx)
                    continue
            lo2 = max(-h[j], h[i] - 1.0)
            hi2 = min(1.0 - h[j], h[i])
            if lo2 < hi2:
                d2, v2 = find_optimal_delta(c, -a, lo2, hi2, dx)
                if v2 < best_c - 1e-15:
                    h[i] -= d2
                    h[j] += d2
                    c = np.correlate(h, 1.0 - h, mode='full')
                    best_c = float(c.max() * dx)
    return h, c, best_c

def multi_resolution_optimize(h_start, n_start, deadline, rng):
    h = project_feasible(h_start.copy(), n_start)
    best_h = h.copy()
    best_c = compute_c5(best_h, n_start)
    n = n_start

    # SLSQP phase
    alphas = [1000, 5000, 20000]
    for a in alphas:
        if time.time() > deadline - 800:
            break
        ht = slsqp_step(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Try higher resolution
    if time.time() < deadline - 400:
        n2 = 500
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2)
        c_interp = compute_c5(h_interp, n2)
        if c_interp < best_c + 0.005:
            ht = slsqp_step(h_interp, n2, 5000, maxiter=1500)
            c2 = compute_c5(ht, n2)
            if c2 < best_c:
                best_c = c2
                best_h = ht.copy()
                n = n2

    # Back to original n if we went to 500
    if n != n_start and time.time() < deadline - 300:
        x_new = np.linspace(0, 2, n)
        x_old = np.linspace(0, 2, n_start)
        h_back = np.interp(x_old, x_new, best_h)
        h_back = project_feasible(h_back, n_start)
        c_back = compute_c5(h_back, n_start)
        if c_back < best_c:
            best_c = c_back
            best_h = h_back.copy()
            n = n_start

    # Random local search
    c_cur = np.correlate(best_h, 1.0 - best_h, mode='full')
    n_now = len(best_h)
    dx = 2.0 / n_now
    best_c2 = float(c_cur.max() * dx)
    
    if time.time() < deadline - 200:
        n_iter = int((deadline - 200 - time.time()) * 1200)
        best_h, c_cur, best_c2 = local_search_random(
            best_h, n_now, c_cur, best_c2, deadline - 200, rng, n_iter)

    # Systematic sweeps

        best_h, c_cur, best_c2 = systematic_sweep(
            best_h, n_now, c_cur, best_c2, time.time() + 30, 2.0 / n_now)
        if time.time() > deadline - 10:
            break

    # Final random polish
    if time.time() < deadline - 2:
        n_iter = int((deadline - 2 - time.time()) * 1200)
        best_h, c_cur, best_c2 = local_search_random(
            best_h, n_now, c_cur, best_c2, deadline - 1, rng, n_iter)

    best_h = project_feasible(best_h, n_now)
    best_c2 = compute_c5(best_h, n_now)
    return best_h, best_c2, n_now
