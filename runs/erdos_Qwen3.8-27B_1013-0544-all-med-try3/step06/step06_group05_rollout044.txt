```python
        grad[d:] = 1 - h[:n-d]
        # j+d in [0, n-1] means j in [0, n-1-d]
        grad[:n-d] -= h[d:]
    else:
        dd = -d
        # j-d = j+dd in [0, n-1] means j in [0, n-1-dd]
        grad[:n-dd] = 1 - h[dd:]
        # j+d = j-dd in [0, n-1] means j in [dd, n-1]
        grad[dd:] -= h[:n-dd]
    return grad
```

Let me verify: for d > 0, j ranges from d to n-1 for the first term (j-d from 0 to n-1-d).
h[j-d] for j=d..n-1 is h[0..n-1-d] = h[:n-d]. ✓
For the second term, j+d < n means j < n-d, so j ranges from 0 to n-1-d.
h[j+d] for j=0..n-1-d is h[d..n-1] = h[d:]. ✓

For d < 0, let dd = -d > 0:
j-d = j+dd < n means j < n-dd, so j ranges from 0 to n-1-dd.
h[j+dd] for j=0..n-1-dd is h[dd..n-1] = h[dd:]. ✓
j+d = j-dd ≥ 0 means j ≥ dd, so j ranges from dd to n-1.
h[j-dd] for j=dd..n-1 is h[0..n-1-dd] = h[:n-dd]. ✓

Now the full gradient of the smooth max:
grad_smooth = (1/alpha) * sum_d exp(alpha * c_d) / sum_d' exp(alpha * c_d') * ∂c_d/∂h
= sum_d w_d * ∂c_d/∂h

where w_d = exp(alpha * c_d) / sum exp(alpha * c_d')

This is O(n^2) total (n lags, each taking O(n) to compute its gradient contribution). For n=500, that's 250,000 operations per gradient evaluation, which is very fast.

Actually wait, for each d, computing the gradient vector is O(n), and we have 2n-1 values of d, so total is O(n^2). For n=500, that's 500*1000 = 500,000, still very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_c_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlation vector wrt h."""
    c = compute_c_vector(h, n)
    m = c.max()
    log_w = alpha * (c - m)
    log_w -= log_w.max()  # for numerical stability
    w = np.exp(log_w)
    w /= w.sum()
    
    grad = np.zeros(n)
    # c has length 2n-1, index m corresponds to lag d = m - (n-1)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        d = idx - (n - 1)
        if d >= 0:
            # First term: (1 - h[j-d]) for j in [d, n-1]
            grad[d:] += wi * (1.0 - h[:n - d])
            # Second term: -h[j+d] for j in [0, n-1-d]
            grad[:n - d] -= wi * h[d:]
        else:
            dd = -d
            # First term: (1 - h[j+dd]) for j in [0, n-1-dd]
            grad[:n - dd] += wi * (1.0 - h[dd:])
            # Second term: -h[j-dd] for j in [dd, n-1]
            grad[dd:] -= wi * h[:n - dd]
    
    return grad

def projected_gradient_step(h, grad, n, lr):
    """Take a projected gradient descent step."""
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def backtracking_line_search(h, grad, c_cur, n, alpha, lr_init, max_iter=30):
    """Backtracking line search for smooth max."""
    lr = lr_init
    dx = 2.0 / n
    c_cur_val = smooth_max_value(compute_c_vector(h, n), alpha)
    for _ in range(max_iter):
        h_new = projected_gradient_step(h, grad, n, lr)
        c_new = smooth_max_value(compute_c_vector(h_new, n), alpha)
        if c_new < c_cur_val - 0.01 * lr * np.sum(np.abs(grad) ** 2):
            return h_new, c_new, lr
        lr *= 0.5
    return h, c_cur_val, lr

def pgd_optimize(h, n, alpha, max_iter, deadline):
    """Projected gradient descent optimization."""
    h = h.copy()
    lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_c_vector(h, n)
        grad = smooth_max_gradient(h, n, alpha)
        h_new, _, lr = backtracking_line_search(h, grad, c, n, alpha, lr)
        h = h_new
    return h
```

```python
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h

            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_obj(h, n, alpha):
    c = compute_c_vector(h, n)
    return smooth_max_value(c, alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c_vector(h, n)
    m = c.max()
    log_w = alpha * (c - m)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        d = idx - (n - 1)
        if d >= 0:
            if d < n:
                grad[d:] += wi * (1.0 - h[:n - d])
                grad[:n - d] -= wi * h[d:]
        else:
            dd = -d
            if dd < n:
                grad[:n - dd] += wi * (1.0 - h[dd:])
                grad[dd:] -= wi * h[:n - dd]
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    c = compute_c_vector(h, n)
    cur_val = smooth_max_value(c, alpha)
    grad = smooth_max_grad(h, n, alpha)
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    new_val = compute_c5(h_new, n)

def greedy_local_search(h, n, deadline, alpha=5000):
    """Local search guided by gradient: try moves that align with gradient direction."""
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

            improved = False
            # Compute gradient to guide search
            grad = smooth_max_grad(h, n, alpha)
            # Sort indices by gradient (high grad = should decrease, low grad = should increase)
            # Moving mass from high-grad to low-grad index decreases smooth max
            order = np.argsort(grad)
            # Try pairs: increase h at low-grad indices, decrease at high-grad indices
            n_try = min(200, n)
            low_idx = order[:n_try]
            high_idx = order[-n_try:]

        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
```
