    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618

import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            can = 1.0 - h
            s = can.sum()
            if s < 1e-13:
                break
            h = h + diff / s * can
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    """Compute gradient of the log-sum-exp smoothed objective."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, 'full') * dx
    t = g.max()

    T2 = np.correlate(w[::-1], h, 'full')

    best_h = h.copy()
    best_c = compute_c5(h, n)

            break
        h = h.copy()
        lr = lr_init

            else:

            # Apply the move
            h_bin[start:start+block_len] = 0.0
            h_bin[new_start:new_start+block_len] = block
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_bin.copy()
                improved = True
            else:
                # Undo
                h_bin[new_start:new_start+block_len] = 0.0
                h_bin[start:start+block_len] = block
        
        if not improved:
            break
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp smoothing."""
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        # Gradient of the smoothed objective
        # d/dx_i [sum_j x[j](1-x[j+k])] = (1-x[i+k]) - x[i-k] (with boundary)
        # Full gradient = sum_k alpha*w[k] * [(1-x[i+k-(n-1)]) - x[i-k+(n-1)]] * (2/n)
        dx = 2.0 / n
        T1 = np.correlate(w, 1 - x, 'full')
        T2 = np.correlate(w[::-1], x, 'full')
        grad = alpha * dx * (T1[n-1::-1] - T2[n-1::-1])
        return grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
