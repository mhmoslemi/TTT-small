        weights /= weights.sum()
        
        grad = np.zeros(n)
        one_minus_x = 1.0 - x
        neg_x = -x
        
        for k_idx in range(2 * n - 1):
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = k_idx - (n - 1)
            if k == 0:
                grad += w * dx * (1 - 2 * x)
            elif k > 0:
                grad[:n-k] += w * dx * one_minus_x[k:]
                grad[k:] += w * dx * neg_x[:n-k]
            else:
                k2 = -k
                grad[k2:] += w * dx * one_minus_x[:n-k2]
                grad[:n-k2] += w * dx * neg_x[k2:]
        
        grad += 2 * mu * (x.sum() - n/2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h, method='L-BFGS-B', jac=gradient,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    return res.x

def greed_binary(h_init, n, deadline, max_sweeps=500):

    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:

                im = i + m_range

                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                # Apply the best swap found
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)

                jm = j + m_range

                delta[msk] -= 1.0 - h[jm[msk]]

                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)

        if not improved:

        if cur_c < best_c:

def anneal_binary(h_init, n, deadline, seed=42, T_init=0.02, cool=0.99993):

        delta[n - 1] -= 2.0

def project_feasible(h, n):
    target = n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(diff) < 1e-14:

            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room

def smooth_obj(x, n, alpha, mu):

    sm = shifted.max()

def smooth_grad(x, n, alpha, mu):

    m = c.max()

    grad = np.zeros(n)

    n1 = n - 1
