    v = np.zeros(n)  # momentum

            worst_lag = int(np.argmax(c))
            lag = worst_lag - (n - 1)
            improved = False
            
            # Targeted moves on worst lag
            if abs(lag) > 0 and abs(lag) < n:
                if lag > 0:
                    idx_range = range(0, n - lag)
                else:
                    idx_range = range(-lag, n)
                for i in idx_range:
                    j = i + lag
                    for di in [d, -d]:
                        hi = h[i] + di
                        hj = h[j] - di
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                improved = True
                            else:
                                h[i], h[j] = hi - di, hj + di
                    
            # Also try second-worst lag
            sorted_lags = np.argsort(c)[::-1]
            for sl in sorted_lags[1:3]:
                if time.time() > deadline:
                    return h, best_c
                lag2 = int(sl) - (n - 1)
                if abs(lag2) > 0 and abs(lag2) < n:
                    if lag2 > 0:
                        idx_range2 = range(0, n - lag2)
                    else:
                        idx_range2 = range(-lag2, n)
                    for idx in idx_range2[:min(len(idx_range2), 100)]:
                        j2 = idx + lag2
                        for di in [d, -d]:
                            hi = h[idx] + di
                            hj = h[j2] - di
                            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                                h[idx], h[j2] = hi, hj
                                c_new = compute_c5(h, n)
                                if c_new < best_c - 1e-14:
                                    best_c = c_new
                                    improved = True
                                else:
                                    h[idx], h[j2] = hi - di, hj + di
            
            # Random moves as fallback
            for _ in range(50):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for di in [d, -d]:
                    hi = h[i] + di
                    hj = h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i], h[j] = hi - di, hj + di
            
            if not improved:
                stagnant += 1
                if stagnant > 3:
                    break
            else:
                stagnant = 0
    
    return h, best_c

def upsample_step(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Nearest-neighbor upsample (preserves step structure)
    h_new = np.interp(x_new, x_old, h, left=0, right=0)
    # Round to preserve step nature
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n_new)

def full_optimize(h_start, n, deadline, rng=None):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    # Multi-alpha L-BFGS-B
    alphas = [50000, 10000, 2000, 500]
    M = 500.0
    
    for a in alphas:
        if time.time() > deadline - 10:
            break
        alloc = min(80, (deadline - time.time()) * 0.3)
        h_opt = lbfgsb_optimize(best_h, n, a, M, time.time() + alloc, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Subgradient on exact max
    if time.time() < deadline - 30:
        alloc_sg = min(120, (deadline - time.time()) * 0.4)
        h_sg, c_sg = subgradient_descent(best_h, n, time.time() + alloc_sg,
                                         max_iter=50000, initial_step=0.05)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg.copy()
    
    # L-BFGS-B polish
    if time.time() < deadline - 20:
        h_opt = lbfgsb_optimize(best_h, n, 100000, 1000.0, time.time() + min(60, (deadline - time.time()) * 0.3), maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Targeted pair search
    if time.time() < deadline - 5:
        best_h, best_c = targeted_pair_search(best_h, n, deadline, max_passes=3000)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        # Phase 1: Optimize at original resolution
        alloc1 = min(350, (deadline - t0) * 0.35)
        h1, c1 = full_optimize(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h1, c1
        
        # Phase 2: Try at n=500
        if time.time() < deadline - 250:
            n500 = 500
            h_500 = upsample_step(h_init, n_init, n500)
            # Also try upsampled from optimized
            h_500b = upsample_step(best_h, n_init, n500)
            
            alloc2 = min(300, (deadline - time.time()) * 0.35)
            h2, c2 = full_optimize(h_500, n500, time.time() + alloc2, rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2
            
            if time.time() < deadline - 150:
                alloc2b = min(150, (deadline - time.time()) * 0.4)
                h2b, c2b = full_optimize(h_500b, n500, time.time() + alloc2b, rng)
                if c2b < best_c:
                    best_c = c2b
                    best_h = h2b
        
        # Phase 3: Try at n=400
        if time.time() < deadline - 200:
            n400 = 400
            h_400 = upsample_step(best_h, len(best_h), n400)
            alloc3 = min(200, (deadline - time.time()) * 0.4)
            h3, c3 = full_optimize(h_400, n400, time.time() + alloc3, rng)
            if c3 < best_c:
                best_c = c3
                best_h = h3
        
        # Phase 4: Restart with perturbation
        if time.time() < deadline - 100:
            n = len(best_h)
            for scale in [0.02, 0.05, 0.1]:
                if time.time() > deadline - 40:
                    break
                hp = best_h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)
                alloc_r = min(80, (deadline - time.time()) * 0.3)
                if alloc_r < 30:
                    break
                hr, cr = full_optimize(hp, n, time.time() + alloc_r, rng)
                if cr < best_c:
                    best_c = cr
                    best_h = hr
    
    # Final polish
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = targeted_pair_search(best_h, n, deadline,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005], max_passes=10000)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n = n_init if h_init is not None else 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
