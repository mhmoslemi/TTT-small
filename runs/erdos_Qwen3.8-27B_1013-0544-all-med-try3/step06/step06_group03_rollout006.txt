```python
def project_feasible(h, n):

            jm = j - lag
            g = 0.0

                g -= h[jm]
            grad[j] += g
```

```python
def compute_lag_gradient(h, lag, n):
    """Gradient of correlation at a specific lag w.r.t. h."""
    grad = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # dc/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag] - h[j-lag] for j in [lag, n-1]
        grad[:n-lag] += (1 - h[lag:])       # first term
        grad[lag:] -= h[:n-lag]             # second term
    else:
        m = -lag
        # c = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # dc/dh[j] = (1-h[j-m]) for j in [m, n-1] - h[j+m] for j in [0, n-1-m]
        grad[m:] += (1 - h[:n-m])           # first term
        grad[:n-m] -= h[m:]                 # second term
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh)))) / alpha

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_lag_grad(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad

def subgradient_momentum(h, n, deadline, eta0=0.03, momentum=0.7, top_k=5):
```
