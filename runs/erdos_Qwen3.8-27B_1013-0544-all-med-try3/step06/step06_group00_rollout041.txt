import numpy as np
import time
from scipy.optimize import minimize


def compute_counts(h):
    return np.correlate(h, 1.0 - h, mode='full')


def objective_c(h, n):
    return float(np.max(compute_counts(h)) * (2.0 / n))


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            rs = float(room.sum())
            if rs <= 1e-15:
                break
            h = h + diff * room / rs
        else:
            if s <= 1e-15:
                break
            h = h + diff * h / s
        h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        frac = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(frac))
        if cnt == 0:
            break
        h[frac] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def logsum_value(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if alpha <= 0.0:
        return m
    z = alpha * (c - m)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / alpha


def weights_from_counts(c, alpha):
    c = np.asarray(c, dtype=float)
    if alpha <= 0.0:
        w = np.zeros_like(c)
        w[c == np.max(c)] = 1.0
        return w / max(1.0, float(w.sum()))
    m = float(np.max(c))
    z = alpha * (c - m)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        e = np.zeros_like(c)
        e[np.argmax(c)] = 1.0
        s = 1.0
    return e / s


def grad_from_weights(w, h, n):
    w = np.asarray(w, dtype=float)
    h = np.asarray(h, dtype=float)
    a = 1.0 - h
    f = np.convolve(w, a, mode='full')
    g = np.convolve(w[::-1], h, mode='full')
    return f[n - 1:2 * n - 1] - g[n - 1:2 * n - 1]


def slq_obj(x, alpha, n):
    c = compute_counts(x)
    return logsum_value(c, alpha)


def slq_grad(x, alpha, n):
    c = compute_counts(x)
    w = weights_from_counts(c, alpha)
    return grad_from_weights(w, x, n)


def slq_cons(x, n):
    return float(x.sum()) - 0.5 * n


def slq_cons_jac(x, n):
    return np.ones(n, dtype=float)


def slsqp_opt(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    try:
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = {
            'type': 'eq',
            'fun': slq_cons,
            'jac': slq_cons_jac,
            'args': (n,)
        }
        res = minimize(
            slq_obj,
            h,
            method='SLSQP',
            bounds=bounds,
            jac=slq_grad,
            args=(alpha, n),
            constraints=[cons],
            options={'maxiter': int(maxiter), 'ftol': 1e-11}
        )
        x = getattr(res, 'x', None)
        if x is None:
            return None
        x = np.asarray(x, dtype=float)
        if not np.all(np.isfinite(x)):
            return None
        return project_feasible(x, n)
    except Exception:
        return None


def lbfgs_fun(x, alpha, n, rho):
    c = compute_counts(x)
    f = logsum_value(c, alpha)
    s = float(x.sum()) - 0.5 * n
    return f + 0.5 * rho * s * s


def lbfgs_grad(x, alpha, n, rho):
    c = compute_counts(x)
    w = weights_from_counts(c, alpha)
    g = grad_from_weights(w, x, n)
    s = float(x.sum()) - 0.5 * n
    return g + (rho * s) * np.ones(n, dtype=float)


def lbfgs_opt(h, n, alpha, rho, maxiter):
    h = project_feasible(h, n)
    try:
        bounds = [(0.0, 1.0) for _ in range(n)]
        res = minimize(
            lbfgs_fun,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=lbfgs_grad,
            args=(alpha, n, rho),
            options={
                'maxiter': int(maxiter),
                'ftol': 1e-11,
                'gtol': 1e-8,
                'maxfun': max(1000, int(20 * maxiter))
            }
        )
        x = getattr(res, 'x', None)
        if x is None:
            return None
        x = np.asarray(x, dtype=float)
        if not np.all(np.isfinite(x)):
            return None
        return project_feasible(x, n)
    except Exception:
        return None


def linear_delta_pair(i, j, h, n, m_range):
    dl = np.zeros(2 * n - 1, dtype=float)

    p = i
    amt = 1.0
    pos = p - m_range
    msk = (pos >= 0) & (pos < n)
    dl[msk] += amt * (1.0 - h[pos[msk]])
    pos = p + m_range
    msk = (pos >= 0) & (pos < n)
    dl[msk] += amt * (-h[pos[msk]])

    p = j
    amt = -1.0
    pos = p - m_range
    msk = (pos >= 0) & (pos < n)
    dl[msk] += amt * (1.0 - h[pos[msk]])
    pos = p + m_range
    msk = (pos >= 0) & (pos < n)
    dl[msk] += amt * (-h[pos[msk]])

    return dl


def pair_descent(h, n, deadline, active_k=16, pair_k=8, fracs=None):
    h = np.asarray(h, dtype=float).copy()
    if fracs is None:
        fracs = (0.25, 0.5, 0.75, 1.0)
    m_range = np.arange(-(n - 1), n, dtype=int)
    c = compute_counts(h)
    mval = float(np.max(c)) * (2.0 / n)
    best_h = h.copy()
    best_val = mval

    for _sweep in range(100000):
        if time.time() > deadline:
            break
        c = compute_counts(h)
        mval = float(np.max(c)) * (2.0 / n)
        if mval < best_val:
            best_val = mval
            best_h = h.copy()

        order = np.argsort(c)
        top = order[-active_k:]
        improved = False

        for li in top:
            if time.time() > deadline:
                break
            li = int(li)
            lag = li - (n - 1)
            if lag >= 0:
                start = 0
                end = n - lag
            else:
                start = -lag
                end = n
            if end <= start:
                continue
            ii = np.arange(start, end, dtype=int)
            jj = ii + lag
            score = h[jj] * (1.0 - h[ii])
            if float(np.max(score)) <= 1e-7:
                continue
            loc = np.argsort(score)
            for t in range(min(pair_k, len(loc))):
                idx = int(loc[len(loc) - 1 - t])
                i = int(ii[idx])
                j = int(jj[idx])
                dmax = min(float(h[j]), float(1.0 - h[i]))
                if dmax <= 1e-9:
                    continue
                dl = linear_delta_pair(i, j, h, n, m_range)
                cen = n - 1
                idx1 = cen + (j - i)
                idx2 = cen + (i - j)
                bd = 0.0
                bcand = None
                bm = mval
                for frac in fracs:
                    d = float(frac) * dmax
                    if d <= 1e-10:
                        continue
                    cand = c + d * dl
                    cand[cen] -= 2.0 * d * d
                    cand[idx1] += d * d
                    cand[idx2] += d * d
                    mn = float(np.max(cand)) * (2.0 / n)
                    if mn < bm - 1e-12:
                        bm = mn
                        bd = d
                        bcand = cand
                if bd > 0.0:
                    h[i] += bd
                    h[j] -= bd
                    c = bcand
                    mval = bm
                    improved = True
                    if mval < best_val:
                        best_val = mval
                        best_h = h.copy()

        if not improved:
            break

        if abs(float(h.sum()) - 0.5 * n) > 1e-10:
            h = project_feasible(h, n)
            c = compute_counts(h)
            mval = float(np.max(c)) * (2.0 / n)
            if mval < best_val:
                best_val = mval
                best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_val = objective_c(best_h, n)
    return best_h, best_val

def resample_feasible(h, n, nn):
    if nn == int(n):
        return project_feasible(h, int(n))
    xo = np.linspace(0.0, 2.0, int(n))
    xn = np.linspace(0.0, 2.0, int(nn))
    hh = np.interp(xn, xo, h)
    return project_feasible(hh, int(nn))


def binary_from(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    b = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1]
    b[idx[:n // 2]] = 1.0
    return b
