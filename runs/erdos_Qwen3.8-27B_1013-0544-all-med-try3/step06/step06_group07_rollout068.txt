def compute_grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-12:
            continue
        l = k - (n - 1)
        if l >= 0:
            if n - l > 0:
                grad[:n-l] += w[k] * (1.0 - h[l:])
            if l > 0:
                grad[l:] += -w[k] * h[:n-l]
            elif l == 0:
                grad[:] += -w[k] * h[:]
        else:
            lp = -l
            if n - lp > 0:
                grad[lp:] += w[k] * (1.0 - h[:n-lp])
            if lp > 0:
                grad[:n-lp] += -w[k] * h[lp:]
    
    return grad * dx

The projection of a point x onto {h: 0 ≤ h ≤ 1, sum(h) = S} can be done in O(n log n) by:

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def compute_grad_smooth(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        l = k - (n - 1)
        if l >= 0:
            if n - l > 0:
                grad[:n - l] += wk * (1.0 - h[l:n])
            if l == 0:
                grad[:] -= wk * h[:]
            else:
                grad[l:] -= wk * h[:n - l]
        else:
            p = -l
            grad[p:] += wk * (1.0 - h[:n - p])
            grad[:n - p] -= wk * h[p:]

    return grad * dx


def project_feasible(x, S):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - S) < 1e-14:
        return x
    lo = -2.0
    hi = 3.0
    mid = 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - mid, 0.0, 1.0)
    # Fine-tune
    for _ in range(50):
        s = h.sum()
        diff = S - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx


def make_con_fun(target_sum):
    return sum([0])  # placeholder, not used


def con_eq(h, target_sum, n):
    return h.sum() - target_sum


def con_eq_jac(h, target_sum, n):
    return np.ones(n)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def slsqp_optimize(h0, n, dx, target_sum, beta, maxiter=5000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum, n)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return h0


def momentum_gradient_descent(h0, n, dx, target_sum, beta, max_iters, time_limit_func, base_lr=0.003, momentum_init=0.9):
    h = h0.copy()
    v = np.zeros(n)
    momentum = momentum_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    prev_grad = None
    prev_h = h.copy()

    for it in range(max_iters):
        if time_limit_func() < 30:
            break
        # BB step size computation
        lr = base_lr / (1.0 + it * 0.0003)
        if prev_grad is not None:
            s = h - prev_h
            y = compute_grad_smooth(h, dx, beta) - prev_grad
            sy = s.dot(y)
            yy = y.dot(y)
            if yy > 1e-16:
                lr_bb = sy / yy
                if 1e-6 < lr_bb < 10 * lr:
                    lr = 0.5 * lr + 0.5 * lr_bb

        # Nesterov lookahead
        y = h + momentum * v
        y = project_feasible(y, target_sum)

        grad = compute_grad_smooth(y, dx, beta)
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, target_sum)

        if it % 300 == 299:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    momentum = max(0.5, momentum * 0.85)
                    v *= 0.5
                    no_improve = 0

        prev_grad = grad
        prev_h = h.copy()

    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return best_h, best_c5


def pairwise_swap(h, n, dx, target_sum, n_pairs, time_limit_func, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    max_corr = corr.max()

    for _ in range(n_pairs):
        if time_limit_func() < 15:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if i == j:
            continue

        # Line search along direction e_i - e_j
        # h[i] + delta, h[j] - delta
        lo = max(-best_h[i], best_h[j] - 1.0)
        hi = min(1.0 - best_h[i], best_h[j])
        if lo >= hi:
            continue

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = lo, hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)

        def obj_delta(delta):
            h[i] = best_h[i] + delta
            h[j] = best_h[j] - delta
            h_c = np.clip(h, 0.0, 1.0)
            corr_t = np.correlate(h_c, 1.0 - h_c, mode='full')
            return corr_t.max() * dx

        h = best_h.copy()
        fc = obj_delta(c)
        fd = obj_delta(d)

        for _ in range(30):
            if b - a < 1e-8:
                break
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = obj_delta(c)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = obj_delta(d)

        opt_delta = (a + b) / 2.0
        h_new = best_h.copy()
        h_new[i] = best_h[i] + opt_delta
        h_new[j] = best_h[j] - opt_delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    return best_h, best_c5
