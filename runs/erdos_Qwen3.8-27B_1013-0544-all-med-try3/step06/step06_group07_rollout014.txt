import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        if diff > 0:
            space = 1.0 - h[mask]
            total = space.sum()
            if total > 1e-15:
                h[mask] += diff * space / total
            else:
                h[mask] += diff / cnt
        else:
            space = h[mask]
            total = space.sum()
            if total > 1e-15:
                h[mask] += diff * space / total
            else:
                h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def grad_f_k(h, l, n, dx):
    """Exact gradient of f_k (correlation at lag l) w.r.t. h."""
    grad = np.zeros(n)
    # term1: (1-h[i-l]) * 1_{0<=i-l<=n-1}
    i_range = np.arange(n)
    idx1 = i_range - l
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    # term2: -h[i+l] * 1_{0<=i+l<=n-1}
    idx2 = i_range + l
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad * dx


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def obj_exact_k(h, l, n, dx):
    """Exact f_k value for lag l where l = k - (n-1)."""
    if l >= 0:
        m = n - l
        return dx * np.sum(h[:m] * (1.0 - h[l:]))
    else:
        la = -l
        m = n - la
        return dx * np.sum(h[la:] * (1.0 - h[:m]))


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def pair_cd_step(h, i, j, n, dx):
    """Optimize h[i] and h[j] with h[i]+h[j] fixed. Returns improved h."""
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi_val = min(1.0, c)
    if lo >= hi_val:
        return h
    h_save = h.copy()
    h_save[i] = 0.5 * (lo + hi_val)
    h_save[j] = c - h_save[i]
    best_t = 0.5 * (lo + hi_val)
    best_val = compute_c5(h_save)
    # Ternary search
    a = lo
    b = hi_val
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h_save[i] = m1
        h_save[j] = c - m1
        v1 = compute_c5(h_save)
        h_save[i] = m2
        h_save[j] = c - m2
        v2 = compute_c5(h_save)
        if v1 < v2:
            b = m2
            if v1 < best_val:
                best_val = v1
                best_t = m1
        else:
            a = m1
            if v2 < best_val:
                best_val = v2
                best_t = m2
    h[i] = best_t
    h[j] = c - best_t
    return h


def pair_cd_full(h, n, dx, max_sweeps, time_limit_fn):
    """Full pair coordinate descent."""
    h = h.copy()
    order = np.random.permutation(n)
    for sweep in range(max_sweeps):
        if time_limit_fn() < 5:
            break
        for idx in range(n):
            i = order[idx]
            j = (i + 1) % n
            h = pair_cd_step(h, i, j, n, dx)
    return h


def active_constraint_opt(h, n, dx, target, max_iters, time_limit_fn):
    """Iteratively minimize the worst constraint."""
    h = h.copy()
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target, n)}]
    for it in range(max_iters):
        if time_limit_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        l = k_star - (n - 1)

        def obj_act(x):
            if l >= 0:
                m = n - l
                return dx * np.sum(x[:m] * (1.0 - x[l:]))
            else:
                la = -l
                m = n - la
                return dx * np.sum(x[la:] * (1.0 - x[:m]))

        def grad_act(x):
            return grad_f_k(x, l, n, dx)

        try:
            res = minimize(obj_act, h, jac=grad_act, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new < c5_old:
                h = h_new
        except Exception:
            pass
    return h


def subgradient_avg(h, n, dx, target, max_iters, time_limit_fn):
    """Subgradient method with iterate averaging."""
    h = h.copy()
    h_avg = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c_lower = 0.0
    for it in range(max_iters):
        if time_limit_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        l = k_worst - (n - 1)
        grad = grad_f_k(h, l, n, dx)
        f_val = compute_c5(h)
        # Polyak step size
        gap = f_val - c_lower
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-20:
            step = 0.0
        else:
            step = max(gap / gnorm2, 0.01)
            step = min(step, 0.5)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        h_avg = ((it * h_avg) + h_new) / (it + 1)
        h = h_new
        if it % 500 == 499:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = project_feasible(h_avg, target).copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    c5_avg = compute_c5(project_feasible(h_avg, target))
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = project_feasible(h_avg, target).copy()
    return best_h
