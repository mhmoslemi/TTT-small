def local_pair_search(h, n, deadline, deltas=(0.005, 0.002, 0.001, 0.0005, 0.0001), max_passes=100):
    """Local search with pair moves."""
    h = h.copy()
    best_c = compute_c5_full(h, n)[0]
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i+1, min(i+50, n)):
                    # Try moving mass from j to i
                    if h[i] < 1.0 and h[j] > 0.0:
                        delta = min(d, 1.0 - h[i], h[j])
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5_full(h, n)[0]
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    # Try moving mass from i to j
                    if h[j] < 1.0 and h[i] > 0.0:
                        delta = min(d, 1.0 - h[j], h[i])
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5_full(h, n)[0]
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            if not improved:
                break
    
    return h, best_c

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
