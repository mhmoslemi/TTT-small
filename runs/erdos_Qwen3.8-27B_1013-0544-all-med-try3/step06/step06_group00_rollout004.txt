def compute_gradient(h, n, weights=None):
    """Compute gradient of weighted sum of correlations.
    If weights is None, compute gradient of max (using uniform weights on active set).
    weights: array of length 2n-1, the weights for each correlation lag.

        weights = np.ones(2*n-1) / (2*n-1)
    m_range = np.arange(-(n-1), n)

        if s >= 0:

                grad[:n-s] += w * (1 - h[s:])
                grad[s:] -= w * h[:n-s]
        else:

            if n - sq > 0:
                grad[sq:] += w * (1 - h[:n-sq])
