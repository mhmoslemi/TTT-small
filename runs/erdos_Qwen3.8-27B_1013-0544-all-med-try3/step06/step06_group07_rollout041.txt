The theoretical minimum of R(k) for k≥1 is at most -n/(n-1) ≈ -1 on average, but individual values can be much lower.

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def subgrad_h(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def multi_grad_h(h, k_list, n):
    grad = np.zeros(n)
    for k_worst in k_list:
        lag = k_worst - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad / len(k_list)


def slsqp_refine(h, n, ts, dx, betas, t0, time_limit, max_iter=3000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts, n)}]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        for trial in range(3):
            if time.time() - t0 > time_limit:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * (0.008 / trial)
                x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': ftol})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    return best_h, best_c5


def subgrad_refine(h, n, ts, t0, time_limit, lr_base=0.003, use_multi=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h = h.copy()
    for it in range(300000):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        if use_multi:
            top_k = np.argsort(corr)[-5:]
            grad = multi_grad_h(h, top_k, n)
        else:
            k_w = np.argmax(corr)
            grad = subgrad_h(h, k_w, n)
        lr = lr_base / (1.0 + it * 0.00015)
        h = h - lr * grad
        h = project_feasible(h, ts)
        if it % 2000 == 1999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5


def binary_local_search(h, t0, time_limit):
    n = len(h)
    ts = n // 2
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:ts]] = 1.0
    best_c5 = compute_c5(h_b)
    best_h = h_b.copy()
    h_cur = h_b.copy()
    no_improve = 0
    for it in range(80000):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
            no_improve = 0
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
            no_improve += 1
            if no_improve > 8000:
                pert = 0.05 * np.random.randn(n)
                h_pert = h + pert
                order2 = np.argsort(-h_pert)
                h_cur = np.zeros(n)
                h_cur[order2[:ts]] = 1.0
                no_improve = 0
    return best_h, best_c5
