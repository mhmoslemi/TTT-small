    dx = 2.0 / n
    best_c = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c = np.correlate(h, 1-h, mode='full')
            M_raw = c.max()  # without dx factor
            M = M_raw * dx

            # G has shape (n, 2n-1)
            G = np.zeros((n, 2*n-1))
            for m in range(2*n-1):
                lag = lags[m]
                # h[j + lag]: valid for j in [max(0,-lag), min(n, n-lag))
                lo = max(0, -lag)
                hi = min(n, n - lag)
                if hi > lo:
                    G[lo:hi, m] = h[lag + np.arange(hi-lo)]  # h[lag:lag+hi-lo]
                # h[j - lag]: valid for j in [max(0,lag), min(n, n+lag))
                lo2 = max(0, lag)
                hi2 = min(n, n + lag)
                if hi2 > lo2:
                    G[lo2:hi2, m] += h[lag + np.arange(hi2-lo2)]  # h[lo2-lag : hi2-lag] = h[0:hi2-lag]
            
            # For each i, compute upper and find valid j
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                
                # upper[m] = (M_raw - c[m]) / d + G[i, m]
                # (using raw correlation values, no dx factor)

        threshold = M * 0.999
        active = np.where(c_vals >= threshold)[0]

        grad = np.zeros(n)
        for m in active:
            lag = m - (n-1)
            # ∂c[m]/∂h[j] = (1-h[j-lag]) - h[j+lag] (padded)

            else:
                g[:n+lag] += (1 - h[-lag:])
            # -h[j+lag] part
            if lag >= 0:
                g[:n-lag] -= h[lag:]
            else:
                g[-lag:] -= h[:n+lag]
            grad += g
        
        grad /= len(active)
        
        # Adaptive learning rate
        lr = base_lr / (1 + it * 0.001)
        it += 1
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

**Targeted Local Search Idea:**
- Identify the worst lag k*
- The correlation at k* is: c_{k*} = ∑_i h[i] * (1-h[i+k*]) (approximately, for the relevant range)
- To reduce c_{k*}, we want to increase ∑ h[i]*h[i+k*]
- This means: where h[i] is high and h[i+k*] is low, try to increase h[i+k*] and decrease h[i]
- But we need to check that this doesn't increase other correlations too much

**Algorithm:**
1. Find worst lag k*
2. For indices i where h[i] > 0.5 and h[i+k*] < 0.5, try transferring mass from h[i] to h[i+k*]
3. Check if the overall max improves
4. If yes, accept; if no, try a smaller transfer

This is essentially a guided pair search focused on the worst lag.

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:

            if tot < 1e-14:

    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=8000, ftol=1e-18):

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

def subgradient_step(h, n, active_lags, lr):
    """One subgradient step targeting the active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active_lags:
        lag = m - (n - 1)
        # ∂(c[m] * dx)/∂h[j] = dx * [(1-h[j-lag]) - h[j+lag]]
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n-lag])
        else:
            g[:n+lag] += (1.0 - h[-lag:])
        if lag >= 0:
            g[:n-lag] -= h[lag:]
        else:
            g[-lag:] -= h[:n+lag]
        grad += g
    grad *= dx / len(active_lags)
    return grad

def subgradient_optimize(h, n, deadline, base_lr=0.05, frac=0.999):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    it = 0
    
    while time.time() < deadline:
        c_vals = np.correlate(h, 1-h, mode='full') * (2.0/n)
        M = c_vals.max()
        
        # Active lags: within frac of max
        threshold = M * frac
        active = np.where(c_vals >= threshold)[0]
        
        # Also include the argmax
        am = c_vals.argmax()
        if am not in active:
            active = np.append(active, am)
        
        grad = subgradient_step(h, n, active, 1.0)
        lr = base_lr / (1.0 + 0.01 * it)
        it += 1
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

def targeted_search(h, n, deadline, max_rounds=10000):

    h = h.copy()

    best_c = compute_c5(h, n)

        # Find pairs (i, i+|lag|) where h[i] is high and h[i+lag] is low
        # Transfer from high to low at the worst lag
        candidates = []
        if lag > 0:
            for i in range(n - lag):

                    candidates.append((i, i+lag, h[i] - 0.5, 0.01))
        elif lag < 0:

                    candidates.append((i, i+lag, h[i] - 0.5, 0.01))
        
        if not candidates:
            break
        
        # Try the most extreme candidates first
        candidates.sort(key=lambda x: abs(x[2]), reverse=True)
        
        improved = False
        for (i, j, bias, d) in candidates[:50]:
            # Try h[i] -= d, h[j] += d (transfer from i to j)
            hi = h[i] - d
            hj = h[j] + d
            if 0 <= hi <= 1 and 0 <= hj <= 1:

                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i], h[j] = hi + d, hj - d
        
        if not improved:
            # Try smaller steps

                        break
                    else:
                        h[i], h[j] = hi + d, hj - d
            if not improved:

    return best_h, best_c

That's too slow! SLSQP with numerical gradient is O(n) evaluations per iteration.

Let me provide an analytical gradient to SLSQP. The smooth objective is:
f(h) = m + (1/α) * log(∑ exp(α*(c_k - m)))

where m = max_k c_k and c_k = (2/n) * ∑_i h[i]*(1-h[i+k-(n-1)])

The gradient is:
∂f/∂h[j] = (1/α) * ∑_k [α * ∂c_k/∂h[j] * exp(α*(c_k - m))] / ∑_k exp(α*(c_k - m))
           = ∑_k [∂c_k/∂h[j] * exp(α*(c_k - m))] / ∑_k exp(α*(c_k - m))

Let w_k = exp(α*(c_k - m)) / ∑_l exp(α*(c_l - m)) (softmax weights)

Then ∂f/∂h[j] = ∑_k w_k * ∂c_k/∂h[j] = ∑_k w_k * (2/n) * [(1-h[j-lag_k]) - h[j+lag_k]]

This can be computed in O(n * (2n-1)) = O(n²) time, which for n=250 is 125K operations ≈ 0.1ms.

So with analytical gradient, each SLSQP iteration is much faster (~0.5ms), and 8000 iterations take ~4s. That's very manageable!

Let me implement the analytical gradient:

```python
def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx  # length 2n-1
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    
    # Numerically stable softmax
    exp_shifted = np.exp(shifted - mx)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp  # softmax weights
    
    # Objective value
    log_sum_exp = mx + np.log(sum_exp)
    obj = m + log_sum_exp / alpha
    
    # Gradient
    # ∂f/∂h[j] = ∑_k weights[k] * dx * [(1-h[j-lag_k]) - h[j+lag_k]]
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        lag = lags[k]
        w = weights[k] * dx
        # (1-h[j-lag]) part
        if lag >= 0:
            grad[lag:] += w * (1.0 - x[:n-lag])
        else:
            grad[:n+lag] += w * (1.0 - x[-lag:])
        # -h[j+lag] part
        if lag >= 0:
            grad[:n-lag] -= w * x[lag:]
        else:
            grad[-lag:] -= w * x[:n+lag]
    
    return obj, grad
```
