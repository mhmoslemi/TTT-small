def project_feasible(h, n):
    target = n / 2.0

            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _solve_1d(a, b):
    lb = np.where(b >= 0, a, a + b)
    lo = float(np.max(lb))
    hi = float(min(np.max(a), np.max(a + b)))
    if hi - lo < 1e-15:
        if float(np.max(a + b)) <= lo + 1e-13:
            return lo, 1.0
        elif float(np.max(a)) <= lo + 1e-13:
            return lo, 0.0
        return lo, 0.5
    eps = 1e-14
    for _ in range(50):
        mid = (lo + hi) * 0.5
        lower = 0.0
        upper = 1.0
        pos = b > eps
        if np.any(pos):
            upper = min(upper, float(np.min((mid - a[pos]) / b[pos])))
        neg = b < -eps
        if np.any(neg):
            lower = max(lower, float(np.max((mid - a[neg]) / b[neg])))
        zero = np.abs(b) <= eps
        if np.any(zero) and float(np.max(a[zero])) > mid:
            lo = mid
            continue
        if lower <= upper:
            hi = mid
        else:
            lo = mid
    lower = 0.0
    upper = 1.0
    pos = b > eps
    if np.any(pos):
        upper = min(upper, float(np.min((hi - a[pos]) / b[pos])))
    neg = b < -eps
    if np.any(neg):
        lower = max(lower, float(np.max((hi - a[neg]) / b[neg])))
    if lower > upper:
        t_opt = 0.5
    else:
        t_opt = (lower + upper) * 0.5
    t_opt = float(np.clip(t_opt, 0.0, 1.0))
    return hi, t_opt

def coord_descent(h, n, deadline, rng, max_sweeps=50000):
    h = h.copy()
    dx = 2.0 / n
    total_lags = 2 * n - 1
    k_vals = np.arange(-(n - 1), n)
    raw_c = np.correlate(h, 1 - h, mode='full')
    best_c = float(raw_c.max() * dx)
    best_h = project_feasible(h, n)
    best_c_val = compute_c5(best_h, n)
    sweep = 0
    stall = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        order = rng.permutation(n)
        for p in order:
            valid1 = (k_vals >= -p) & (k_vals < n - p)
            valid2 = (k_vals > p - n) & (k_vals <= p)
            b = np.zeros(total_lags)
            if np.any(valid1):
                b[valid1] = 1.0 - h[p + k_vals[valid1]]
            if np.any(valid2):
                b[valid2] -= h[p - k_vals[valid2]]
            a = raw_c - h[p] * b
            val, t_opt = _solve_1d(a, b)
            delta = t_opt - h[p]
            if abs(delta) > 1e-14:
                h[p] = t_opt
                raw_c += delta * b
                nm = float(raw_c.max() * dx)
                if nm < best_c - 1e-14:
                    best_c = nm
                    best_h = h.copy()
                    improved = True
        h = project_feasible(h, n)
        raw_c = np.correlate(h, 1 - h, mode='full')
        pm = float(raw_c.max() * dx)
        if pm < best_c - 1e-14:
            best_c = pm
            best_h = h.copy()
        if not improved:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_polish(h, n, deadline, max_iters=200):
    from scipy.optimize import minimize
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(1000, 50.0), (5000, 200.0), (50000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        def obj(x):
            c = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full') * (2.0 / n)
            sh = alpha * (c - c.max())
            mx = sh.max()
            if mx > 500:
                val = c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = c.max() + np.log(np.sum(np.exp(sh))) / alpha
            val += mu * (np.clip(x,0,1).sum() - n / 2.0) ** 2
            return float(val)
        def jac(x):
            x = np.clip(x, 0, 1)
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
