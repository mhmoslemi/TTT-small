import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_raw_overlap(h, n):
    return np.correlate(h, 1 - h, mode='full')

def swap_delta_vec(h, i, j, n):
    k = np.arange(2*n - 1) - (n - 1)
    jk = j + k
    hjk = np.where((jk >= 0) & (jk < n), h[np.clip(jk, 0, n-1)], 0.0)
    jmk = j - k
    hjmk = np.where((jmk >= 0) & (jmk < n), h[np.clip(jmk, 0, n-1)], 0.0)
    ik = i + k
    hik = np.where((ik >= 0) & (ik < n), h[np.clip(ik, 0, n-1)], 0.0)
    imk = i - k
    himk = np.where((imk >= 0) & (imk < n), h[np.clip(imk, 0, n-1)], 0.0)
    return hjk + hjmk - hik - himk

def simulated_annealing(h_bin, n, deadline, max_iters=100000):
    rng = np.random.RandomState(42)
    dx = 2.0 / n
    O = get_raw_overlap(h_bin, n)
    current_c5 = float(np.max(O) * dx)
    best_c5 = current_c5
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    T_start = 2.0  # in units of dx (so T=2 means we accept changes up to 2*dx)
    T_end = 0.01
    decay = (T_end / T_start) ** (1.0 / max_iters)
    T = T_start
    
    accepted = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= decay
        
        # Pick random swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        delta = swap_delta_vec(h_bin, i, j, n)
        new_O = O + delta
        new_c5 = float(np.max(new_O) * dx)
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / (T * dx)):
            # Accept
            h_bin[i], h_bin[j] = 1.0, 0.0
            O = new_O
            current_c5 = new_c5
            
            # Update ones and zeros
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            
            accepted += 1
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_bin.copy()
    
    return best_h, best_c5

def swap_delta(h, i, j, n, k_arr):
    """Compute ΔO(k) for all k when moving 1 from j to i (both binary)."""
    jk = j + k_arr
    hjk = np.where((jk >= 0) & (jk < n), h[np.clip(jk, 0, n - 1)], 0.0)
    jmk = j - k_arr
    hjmk = np.where((jmk >= 0) & (jmk < n), h[np.clip(jmk, 0, n - 1)], 0.0)

            if current_max < best_max:
                best_max = current_max
                best_h = h_bin.copy()

    return best_h, best_max * (2.0 / n)

def greedy_binary_search(h_bin, n, deadline, k_arr, max_sweeps=100):
    O = get_raw_overlap(h_bin)
    current_max = float(np.max(O))
    best_c5 = current_max * (2.0 / n)
    best_h = h_bin.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]

        best_improvement = 0
        best_i = -1
        best_j = -1
        best_new_O = None

        for j in ones:
            for i in zeros:
                delta = swap_delta(h_bin, i, j, n, k_arr)
                new_max = float(np.max(O + delta))
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
                    best_new_O = O + delta

        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            O = best_new_O
            current_max -= best_improvement
            best_c5 = current_max * (2.0 / n)
            best_h = h_bin.copy()
        else:
            break

    return best_h, best_c5

def slsqp_single(h_start, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h_start

def slsqp_continuation(h_start, n, deadline, alpha_list, maxiter_each=400):
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_new = slsqp_single(best_h, n, alpha, maxiter_each, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def multi_start_slsqp(h_starts, n, deadline, alpha_list, maxiter_each=300):
    best_h = None
    best_c = float('inf')
    for hs in h_starts:
        if time.time() > deadline:
            break
        h_new, c_new = slsqp_continuation(hs, n, deadline, alpha_list, maxiter_each)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = budget_s - 10
    dl = t0 + total_budget

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Binary SA (25% budget)
    p1_end = t0 + 0.25 * total_budget
    h_bin = binary_project(h0, n0)
    k_arr = precompute_k_array(n0)

    h_sa, c_sa = simulated_annealing_binary(h_bin.copy(), n0, p1_end, seed=seed, max_iters=10000000)
    if c_sa < best_c:
        best_c = c_sa
        best_h = h_sa.copy()
        best_n = n0

    # Greedy from SA result
    h_g, c_g = greedy_binary_search(h_sa.copy(), n0, p1_end, k_arr, max_sweeps=50)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g.copy()
        best_n = n0

    # Also try SA from random start
    rng = np.random.RandomState(seed + 1)
    h_rand = np.zeros(n0)
    rand_idx = rng.choice(n0, size=n0 // 2, replace=False)
    h_rand[rand_idx] = 1.0
    h_sa2, c_sa2 = simulated_annealing_binary(h_rand, n0, p1_end, seed=seed + 2, max_iters=2000000)
    if c_sa2 < best_c:
        best_c = c_sa2
        best_h = h_sa2.copy()
        best_n = n0

    # Phase 2: Continuous SLSQP (40% budget)
    p2_end = t0 + 0.65 * total_budget

    # Prepare multiple starting points
    starts = []
    starts.append(h_sa.copy())  # From SA binary
    starts.append(h0.copy())    # From parent
    starts.append(project_feasible(h_sa + 0.1 * rng.randn(n0), n0))  # Perturbed
    starts.append(project_feasible(h_sa + 0.05 * rng.randn(n0), n0))  # Less perturbed

    h_best_cont, c_best_cont = multi_start_slsqp(starts, n0, p2_end,
                                                  alpha_list=[100, 500, 2000, 10000, 50000, 200000],
                                                  maxiter_each=300)
    if c_best_cont < best_c:
        best_c = c_best_cont
        best_h = h_best_cont.copy()
        best_n = n0

    # Phase 3: Multi-resolution (20% budget)
    p3_end = t0 + 0.85 * total_budget

    for nh in [200, 300, 350]:
        if time.time() > p3_end - 20:
            break
        if nh == best_n:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)

        # Binary + SA at new resolution
        hh_bin = binary_project(hh, nh)
        kh = precompute_k_array(nh)
        hh_sa, c_hh_sa = simulated_annealing_binary(hh_bin, nh, p3_end - 15,
                                                     seed=seed + nh, max_iters=3000000)
        if c_hh_sa < best_c:
            best_c = c_hh_sa
            best_h = hh_sa.copy()
            best_n = nh

        # SLSQP from best binary at new res
        hh_c, c_hh_c = slsqp_continuation(hh_sa, nh, p3_end - 5,
                                           alpha_list=[500, 2000, 10000, 50000, 200000],
                                           maxiter_each=250)
        if c_hh_c < best_c:
            best_c = c_hh_c
            best_h = hh_c.copy()
            best_n = nh

    # Phase 4: Final polish (15% budget)
    if time.time() < dl - 5:
        # High alpha SLSQP on best solution
        h_final, c_final = slsqp_continuation(best_h.copy(), best_n, dl,
                                               alpha_list=[100000, 500000, 2000000, 5000000],
                                               maxiter_each=500)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
            best_n = best_n

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
