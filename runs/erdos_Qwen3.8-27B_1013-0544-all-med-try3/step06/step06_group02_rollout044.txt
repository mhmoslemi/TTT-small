import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    c_shift = alpha * c
    mx = c_shift.max()
    if mx > 500:
        return c.max() + (mx - 500 + np.log(np.sum(np.exp(c_shift - mx - 500)))) / alpha
    val = (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha
    return val

def smooth_max_weighted_grad(h, n, alpha):
    c = compute_correlation(h, n)
    c_shift = alpha * c
    mx = c_shift.max()
    if mx > 500:
        exp_c = np.exp(c_shift - mx - 500)
    else:
        exp_c = np.exp(c_shift - mx)
    w = exp_c / exp_c.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad, c

def _obj_penalty(x, n, alpha, mu):
    c = compute_correlation(x, n)
    c_shift = alpha * c
    mx = c_shift.max()
    if mx > 500:
        val = c.max() + (mx - 500 + np.log(np.sum(np.exp(c_shift - mx - 500)))) / alpha
    else:
        val = (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_penalty(x, n, alpha, mu):
    g, _ = smooth_max_weighted_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _obj_slsqp(x, n, alpha):
    c = compute_correlation(x, n)
    c_shift = alpha * c
    mx = c_shift.max()
    if mx > 500:
        return float(c.max() + (mx - 500 + np.log(np.sum(np.exp(c_shift - mx - 500)))) / alpha)
    return float((mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha)

def _con_fun(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def lbfgs_multi_alpha(h, n, deadline, max_iters_per_alpha=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 50000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = max(0.1, alpha * 1e-6)
        t_alloc = min(45, deadline - time.time() - 2)
        if t_alloc < 5:
            break
        try:
            res = minimize(_obj_penalty, h, jac=_grad_penalty,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters_per_alpha, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters_per_alpha * 8,
                                   'maxls': 40, 'maxlsmlu': 0.05,
                                   'disp': False})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_coordinate_descent(h, n, deadline, n_top=50, step_init=0.02):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = step_init
    alpha = 2000
    stuck_count = 0
    while time.time() < deadline:
        g, c = smooth_max_weighted_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        top_idx = np.argsort(np.abs(g))[-n_top:]
        improved = False
        for idx in top_idx:
            if time.time() > deadline:
                break
            for direction in [-1, 1]:
                trial = h.copy()
                trial[idx] += direction * step * g[idx]
                if trial[idx] < 0 or trial[idx] > 1:
                    continue
                trial = project_feasible(trial, n)
                c_trial = compute_c5(trial, n)
                if c_trial < best_c - 1e-15:
                    h = trial
                    best_c = c_trial
                    best_h = h.copy()
                    improved = True
                    break
        if improved:
            stuck_count = 0
            step = min(step * 1.1, 0.05)
        else:
            stuck_count += 1
            step *= 0.7
            alpha *= 1.3
            if step < 1e-6:
                step = 0.01
                alpha = 5000
            if stuck_count > 20:
                break
    return best_h, best_c

def pair_refinement(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        passes = 0
        while time.time() < deadline:
            passes += 1
            if passes > 50:
                break
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003, alpha=5000):
    best_c = compute_c5(h, n)
    while time.time() < deadline:
        g, _ = smooth_max_weighted_grad(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        top_idx = np.argsort(np.abs(g))[-40:]
        improved = False
        for i in top_idx:
            if time.time() > deadline:
                return h, best_c
            for j_idx in top_idx:
                j = j_idx
                if i == j:
                    continue
                move = -step_size * (g[i] - g[j])
                if abs(move) < 1e-12:
                    continue
                if 0 <= h[i] + move <= 1.0 and 0 <= h[j] - move <= 1.0:
                    h[i] += move; h[j] -= move
                    h = project_feasible(h, n)
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= move; h[j] += move
                        h = project_feasible(h, n)
        if not improved:
            step_size *= 0.8
            if step_size < 1e-5:
                break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
