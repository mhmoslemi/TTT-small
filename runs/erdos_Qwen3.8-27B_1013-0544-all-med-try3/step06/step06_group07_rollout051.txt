import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')

        c = compute_coefficients(h, n, lag)

        i_best = np.argmax(c)
        j_best = np.argmin(c)
        
        if i_best == j_best:
            break
        
        c_diff = c[i_best] - c[j_best]
        if abs(c_diff) < 1e-12:
            break
        
        # Determine max step
        if c_diff > 0:
            # Increase h[i_best], decrease h[j_best]
            max_t = min(1.0 - h[i_best], h[j_best])
        else:

            max_t = min(h[i_best], 1.0 - h[j_best])
            i_best, j_best = j_best, i_best

At iteration t, solve: min g_k*(h) - ∇p_k*(h^t) · h s.t. h ∈ C
where k* is the active lag and C is the feasible set {0 ≤ h ≤ 1, 1^T h = n/2}.

s.t. 0 ≤ h ≤ 1, 1^T h = n/2

This is a quadratic programming problem (QP) with linear constraints! We can solve it efficiently.

The objective is:
∑ h_i + (1/4)∑(h_i - h_{i+k*})² - ∇p_k*(h^t) · h

Let me expand:
= ∑ h_i + (1/4)∑(h_i² - 2h_i h_{i+k*} + h_{i+k*}²) - ∇p_k*(h^t) · h
= (1/4)∑ h_i² + (1/4)∑ h_{i+k*}² - (1/2)∑ h_i h_{i+k*} + ∑ h_i (1 - [∇p_k*(h^t)]_i)

So the objective is:
(1/2)∑ h_i² - (1/2)∑ h_i h_{i+k*} + ∑ h_i (1 - [∇p_k*(h^t)]_i)

Let d_i = 1 - [∇p_k*(h^t)]_i (a constant vector computed from h^t).

The objective is: (1/2)h^T h - (1/2)∑ h_i h_{i+k*} + d^T h

This is a definite Hessian (I minus a symmetric matrix with eigenvalues in [-1, 1]... actually the shift matrix has eigenvalues on the unit circle, so (A+A^T)/2 has eigenvalues in [-1,1], meaning I - (A+A^T)/2 has eigenvalues in [0, 2], so the Hessian is positive semidefinite).

def dc_qp_step(h_t, n, lag):
    # Compute the DC QP step
    # First, compute grad p_k*(h_t)
    # p_k(h) = (1/4) sum_{valid i} (h_i + h_{i+k})^2
    # grad p_k(h_t)[j] = (1/2) * [(h_t[j] + h_t[j+k]) (if valid) + (h_t[j-k] + h_t[j]) (if valid)]

    if lag >= 0:
        for i in range(n - lag):
            grad_p[i] += 0.5 * (h_t[i] + h_t[i + lag])
            grad_p[i + lag] += 0.5 * (h_t[i] + h_t[i + lag])
    else:
        k = -lag
        for i in range(n - k):
            grad_p[i + k] += 0.5 * (h_t[i] + h_t[i + k])
            grad_p[i] += 0.5 * (h_t[i] + h_t[i + k])
    
    d = 1.0 - grad_p  # The linear coefficient in the QP objective
    
    # Build the QP
    h_var = cp.Variable(n)
    # Objective: (1/2) h^T h - (1/2) sum h_i h_{i+k} + d^T h
    # In cvxpy: cp.quad_form(h_var, Q) + d @ h_var

    else:

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    
    return h_var.value
