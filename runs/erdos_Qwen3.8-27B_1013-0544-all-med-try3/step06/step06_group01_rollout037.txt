    C = np.correlate(h, 1-h, 'full').astype(float)
    current_max = C.max()
    best_h = h.copy()
    best_max = current_max
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / n_iterations)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        T *= cooling
        
        # Random swap
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        
        # Compute delta
        delta = compute_delta(h, i, j, n)
        
        # New max
        new_C_max_candidates = C + delta
        new_max = new_C_max_candidates.max()
        
        # Accept?
        if new_max <= current_max or np.random.rand() < np.exp((current_max - new_max) / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            C = C + delta
            current_max = new_max
            # Update sets
            zeros = np.setdiff1d(zeros, [i])
            ones = np.setdiff1d(ones, [j])
            ones = np.append(ones, i)
            zeros = np.append(zeros, j)
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    
    return best_h, best_max * dx

S_k ≈ (n-k)/2 (if the 1s are uniformly distributed).
A_linear[k] should be close to (n-k)/2 as well.

This means we want the autocorrelation to be as "flat" as possible, with A_linear[k] ≈ (n-k)/2 for all k.

3. **Zadoff-Chu sequences**: Have constant amplitude zero autocorrelation (CAZAC) but are complex-valued

def full_sweep(h, n, C):
    """Do one full sweep: try all pairs (a in ones, b in zeros) and apply the best improvement."""
    dx = 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n1 = len(ones)
    n0 = len(zeros)
    
    if n1 == 0 or n0 == 0:
        return h, C, C.max(), False
    
    # Precompute delta for flipping each 1-position (1→0)
    m_range = np.arange(-(n-1), n)
    
    # For position a (1→0): delta[k] = -(1-h[a+k]) + h[a-k]
    # Index in full correlate: n-1+k
    # a+k is valid when 0 <= a+k < n, i.e., -a <= k < n-a
    # a-k is valid when 0 <= a-k < n, i.e., a-n+1 < k <= a
    
    d1 = np.zeros((n1, 2*n-1))
    d0 = np.zeros((n0, 2*n-1))
    
    for idx, a in enumerate(ones):
        im = a + m_range  # a+k for each k
        msk = (im >= 0) & (im < n)
        d1[idx, msk] -= (1.0 - h[im[msk]])
        imn = a - m_range  # a-k for each k
        msk = (imn >= 0) & (imn < n)
        d1[idx, msk] += h[imn[msk]]
    
    for idx, b in enumerate(zeros):
        bm = b + m_range
        msk = (bm >= 0) & (bm < n)
        d0[idx, msk] += (1.0 - h[bm[msk]])
        bmn = b - m_range
        msk = (bmn >= 0) & (bmn < n)
        d0[idx, msk] -= h[bmn[msk]]
    
    # Find best pair
    cur_max = C.max()
    best_val = cur_max
    best_ia = -1
    best_ib = -1
    
    for ia in range(n1):
        Ca = C + d1[ia]  # shape (2n-1,)
        # For each b: new_max = max(Ca + d0[ib])
        # = max over k of (Ca[k] + d0[ib, k])
        new_maxs = np.max(Ca[None, :] + d0, axis=1)  # shape (n0,)
        min_idx = np.argmin(new_maxs)
        if new_maxs[min_idx] < best_val - 1e-15:
            best_val = new_maxs[min_idx]
            best_ia = ia
            best_ib = min_idx
    
    if best_ia >= 0:
        a = ones[best_ia]
        b = zeros[best_ib]
        h[a] = 0.0
        h[b] = 1.0
        C = C + d1[best_ia] + d0[best_ib]
        return h, C, best_val, True
    
    return h, C, cur_max, False

def compute_delta_pair(h, a, b, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = a + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] -= (1.0 - h[im[msk]])
    imn = a - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] += h[imn[msk]]
    bm = b + m_range
    msk = (bm >= 0) & (bm < n)
    delta[msk] += (1.0 - h[bm[msk]])
    bmn = b - m_range
    msk = (bmn >= 0) & (bmn < n)
    delta[msk] -= h[bmn[msk]]
    return delta

def best_improvement_sweep(h, n, C, m_range, max_pairs_check=5000):

        h_bin_final = make_binary(best_h, best_n)
        c_fb = compute_c5(h_bin_final, best_n)
        if c_fb < best_c:
            update_best(h_bin_final, c_fb, best_n)
        else:
            h_bin_final = best_h.copy()
            c_fb = best_c

        # Quick Local search
        sub_dl = polish_deadline - 5
        h_pol, c_pol, _ = iterated_local_search(h_bin_final, best_n, sub_dl, rng, n_restarts=5)
        if c_pol < best_c:
            update_best(h_pol, c_pol, best_n)

    # Final: one more SLSQP polish
    if time.time() < polish_deadline - 2:
        h_final = project_feasible(best_h + 0.01 * rng.randn(best_n), best_n)
        h_final = fast_slsqp(h_final, best_n, 500000, 1000, polish_deadline - 1)
        c_final = compute_c5(h_final, best_n)
        if c_final < best_c:
            update_best(h_final, c_final, best_n)

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
