```python
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').copy()
    cur_c = corr.max() * dx
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    m_range = np.arange(-(n-1), n)
    
    temp = initial_temp
    best_h = h.copy()
    best_c = cur_c
    
    while time.time() < deadline:
        # Randomly pick a zero and a one
        i = random.choice(zeros)
        j = random.choice(ones)
        
        # Compute delta
        delta = compute_delta(h, i, j, n)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        
        # Accept if better or with probability exp((cur_c - nc) / temp)
        if nc < cur_c or random.random() < np.exp((cur_c - nc) * n / temp):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = nc
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        temp *= cooling_rate
    
    return best_h, best_c
```
