        ts_best = best_n / 2.0
        for beta in [30000, 100000, 200000]:
            if remaining() < 10:
                break
            h_try, c5_try = slsqp_run(best_h, best_n, ts_best, beta, maxiter=5000, ftol=1e-16)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Phase 7: One more subgradient + polish cycle if time remains
    if remaining() > 30:
        ts_best = best_n / 2.0
        sub_deadline = time.time() + min(remaining() - 15, 50)
        h_sg, c5_sg = subgrad_run(best_h, best_n, ts_best, max_iter=40000,
                                  base_lr=0.001, decay=0.001, deadline=sub_deadline)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg
        if remaining() > 15:
            h_try, c5_try = slsqp_run(best_h, best_n, ts_best, 100000, maxiter=5000, ftol=1e-16)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Final projection and evaluation
    ts_final = best_n / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
