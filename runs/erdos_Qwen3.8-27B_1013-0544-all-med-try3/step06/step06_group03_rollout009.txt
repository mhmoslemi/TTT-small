```python
def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    full_corr = np.correlate(h, 1-h, mode='full') * dx
    # softmax weights
    shifted = alpha * full_corr - alpha * full_corr.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        for delta in range(-(n-1), n):
            s = w[delta + n - 1]
            if 0 <= j + delta <= n-1:
                grad[j] += s * (1 - h[j+delta])
            if 0 <= j - delta <= n-1:
                grad[j] -= s * h[j-delta]
    grad *= dx
    return grad
```

    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    full_corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = full_corr.max()
    shifted = alpha * (full_corr - max_c)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    log_sum = np.log(total)
    
    sum_diff = h.sum() - n / 2.0
    obj = max_c + log_sum / alpha + penalty * sum_diff * sum_diff
    
    w = exp_shifted / total
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = dx * (w[n - 1 - j: 2 * n - 1 - j] @ (1 - h) - w[j: j + n][::-1] @ h)
    grad += 2.0 * penalty * sum_diff
    return obj, grad

def lgbfsopt(h, n, time_limit, alphas=None, penalty=1e6):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > time_limit:
            break
        try:
            res = minimize(
                smooth_obj_grad, h, args=(n, alpha, penalty),
                method='L-BFGS-B', jac=True,
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 20000}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c

def subgradient_polish(h, n, time_limit, steps=20000):
    """Direct subgradient method on the true max."""
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.005
    
    for step in range(steps):
        if time.time() > time_limit:
            break
        full_corr = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(full_corr))
        delta = k_star - (n - 1)
        
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j + delta <= n - 1:
                g[j] += (1 - h[j + delta])
            if 0 <= j - delta <= n - 1:
                g[j] -= h[j - delta]
        g *= dx
        
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            lr *= 0.95
    
    return h, best_c

def gridded_local_search(h, n, time_limit, deltas=None, radius=150):
    """Local search trying nearby pairs with various step sizes."""
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < time_limit:
            improved = False
            for i in range(n):
                if time.time() > time_limit:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
        if time.time() > time_limit:
            break
    
    return h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_random_init(n, rng, mean=0.5):
    h = np.clip(rng.normal(mean, 0.25, n), 0, 1)
    return project_feasible(h, n)

def make_checkerboard(n, rng, phase=0):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i + phase) % 2 == 0) else 0.0
    noise = rng.normal(0, 0.05, n)
    h = np.clip(h + noise, 0, 1)
    return project_feasible(h, n)

def make_vdc_sequence(n):
    """Van der Corput based low-discrepancy pattern."""
    h = np.zeros(n)
    for i in range(n):
        # Binary van der Corput
        v = 0.0
        d = 1.0
        m = i + 1
        while m > 0:
            d /= 2.0
            v += (m % 2) * d
            m //= 2
        h[i] = 1.0 if v > 0.5 else 0.0
    return project_feasible(h, n)

def optimize_at_resolution(h_start, n, deadline, rng, use_subgrad=True):
    """Full optimization pipeline at a given resolution."""
    h = project_feasible(h_start.copy(), n)

        if c2 < best_c:
            best_c = c2
            h = h2
