def subgradient_step(h, n, lr, n_top=5):
    """One subgradient step for the max overlap problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx  # (2n-1,)
    
    # Get top n_top lags
    top_idx = np.argsort(g)[::-1][:n_top]
    
    # Compute subgradient (average of gradients at top lags)
    m_range = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k_idx in top_idx:
        lag = m_range[k_idx]
        # d/dh[j] of sum_i h[i]*v[i+lag] = v[j+lag] - h[j-lag] (with boundary=0)
        # = (1-h[j+lag])*[valid] - h[j-lag]*[valid]
        jp = np.arange(n) + lag
        jm = np.arange(n) - lag
        vp = np.zeros(n)
        hp = np.zeros(n)
        valid_p = (jp >= 0) & (jp < n)
        valid_m = (jm >= 0) & (jm < n)
        vp[valid_p] = 1 - h[jp[valid_p]]
        hp[valid_m] = h[jm[valid_m]]
        grad += vp - hp
        # Also need: -h[j-lag] from the v change
        # Wait, I already included that. Let me recheck.
        # d/dh[j] of sum_i h[i]*v[i+lag]:
        # = v[j+lag] + sum_{i: i+lag=j} h[i]*(-1)  [from v[j] changing]
        # = v[j+lag] - h[j-lag]
        # = (1-h[j+lag])*[valid] - h[j-lag]*[valid]
        # That's what I have. Good.
    
    grad /= len(top_idx)
    
    # Update
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, g.max()
