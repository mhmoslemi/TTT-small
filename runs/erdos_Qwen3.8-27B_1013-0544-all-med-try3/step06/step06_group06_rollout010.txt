import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    """Soft-max objective with penalty for sum constraint."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    if alpha < 1e10:
        s = g - m
        logsumexp = m + np.log(np.exp(alpha * s).sum()) / alpha
    else:
        logsumexp = m
    penalty = mu * (h.sum() - n/2.0)**2
    return logsumexp + penalty

def smooth_max_grad(h, n, alpha, mu):
    """Gradient of soft-max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    g = c * dx
    
    m = g.max()
    if alpha < 1e10:
        s = alpha * (g - m)
        # Shift for numerical stability
        s_max = s.max()
        exp_s = np.exp(s - s_max)
        weights = exp_s / exp_s.sum()
    else:
        weights = np.zeros(len(g))
        weights[np.argmax(g)] = 1.0
    
    grad = np.zeros(n)
    nn = n
    for kk in range(2 * nn - 1):
        w = weights[kk]
        if w < 1e-15:
            continue
        # h[i] as h[n+kk]: n = i-kk, valid when 0<=i-kk<nn and 0<=i<nn
        i_lo = max(0, kk)
        i_hi = min(nn, kk + nn)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * dx * (1.0 - h[i_lo - kk:i_hi - kk])
        
        # h[i] in (1-h[n]): n = i, valid when 0<=i<nn and 0<=i+kk<nn
        i_lo2 = max(0, -kk)
        i_hi2 = min(nn, nn - kk)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * dx * h[i_lo2 + kk:i_hi2 + kk]
    
    # Add penalty gradient
    grad += 2 * mu * (h.sum() - nn / 2.0)
    
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_and_grad(h, n, alpha):
    """Returns (objective, gradient) of soft-max approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    g = c * dx
    
    m = g.max()
    if alpha > 1e15:
        # Hard max
        weights = np.zeros(len(g))
        weights[np.argmax(g)] = 1.0
        obj = m
    else:
        s = alpha * (g - m)
        s_max = s.max()
        exp_s = np.exp(s - s_max)
        weights = exp_s / exp_s.sum()
        obj = m + np.log(exp_s.sum()) / alpha
    
    # Gradient
    N = n
    grad = np.zeros(N)
    for kk in range(2 * N - 1):
        w = weights[kk]

        k = kk - (N - 1)
        # Term 1: v[i-k] where v = 1-h, valid range: max(0,k) <= i < min(N, k+N)
