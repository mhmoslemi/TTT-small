```
def simulated_annealing_binary(h, n, T_start, T_end, n_steps, deadline):
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_c = cur_max * dx
    T = T_start
    cool = (T_end / T_start) ** (1.0 / n_steps)
    
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    
    for step in range(n_steps):
        if time.time() > deadline: break
        # Pick random i from zeros, j from ones
        i = random choice from zeros
        j = random choice from ones
        
        # Compute delta
        delta = compute_delta(i, j, h, n)
        new_corr = corr + delta
        new_max = new_corr.max()
        
        # Accept/reject
        delta_max = new_max - cur_max
        if delta_max < 0 or random() < exp(-delta_max / T):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_max = new_max
            zeros.remove(i); zeros.append(j)
            ones.remove(j); ones.append(i)
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        
        T *= cool
    return best_h, best_c
```

```python
    # h[i]*(1-h[j]) is now 1*1=1 (after: h[i]=1, h[j]=0, so (1-h[j])=1)
```

```python
    """Simulated annealing on binary h with exactly n/2 ones"""
    dx = 2.0 / n
    ones = set(np.where(h == 1)[0].tolist())
    zeros = list(np.where(h == 0)[0].tolist())
    ones_list = list(ones)
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max_val = corr.max()
    cur_c = cur_max_val * dx
    best_c = cur_c
    best_h = h.copy()
    
    T = t_start
    if n_steps > 1:
        cool_rate = (t_end / t_start) ** (1.0 / n_steps)
    else:
        cool_rate = 1.0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Pick random i from zeros, j from ones
        if len(zeros) == 0 or len(ones_list) == 0:
            break
        i = int(rng.randint(len(zeros)))
        j_idx = int(rng.randint(len(ones_list)))
        i_pos = zeros[i]
        j_pos = ones_list[j_idx]
        
        # Make the swap
        h[i_pos] = 1.0
        h[j_pos] = 0.0
        
        # Recompute correlation
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max_val = new_corr.max()
        new_c = new_max_val * dx
        
        # Accept/reject
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T * n)):
            # Accept
            corr = new_corr
            cur_max_val = new_max_val
            cur_c = new_c
            # Update sets
            ones.add(i_pos)
            ones.discard(j_pos)
            zeros[i] = j_pos
            ones_list[j_idx] = i_pos
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Reject, undo
            h[i_pos] = 0.0
            h[j_pos] = 1.0
        
        T *= cool_rate
    
    return best_h, best_c
```

```python
    pos_new = np.linspace(0, n_new, n_new, endpoint=False)

    # Actually, for binary, just use np.interp with rounding
    new_h = np.interp(pos_new, pos_old, h)

    zeros = list(np.where(h == 0)[0])

    T_min = 0.1
    cool = 0.9995
    no_improve = 0
    
    step = 0
    while time.time() < deadline:
        i_idx = int(rng.randint(len(zeros)))
        j_idx = int(rng.randint(len(ones)))
        i_pos = zeros[i_idx]
        j_pos = ones[j_idx]

        h[j_pos] = 0.0
        new_max_val = np.correlate(h, 1 - h, mode='full').max()

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_max_val = new_max_val
            ones[j_idx] = i_pos
            zeros[i_idx] = j_pos
            if new_max_val * dx < best_c:
```

```python
        h7 = slsqp_smooth(h7, n_cur, 100000, 1000, dl - 5)
        c7 = compute_c5(h7, n_cur)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    best_h = project_feasible(best_h, n_cur)
    best_c = compute_c5(best_h, n_cur)
    return best_h, best_c, n_cur
```
