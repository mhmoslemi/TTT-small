import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    """Smooth approximation of max using log-sum-exp."""
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_smooth_max(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        return smooth_max(g, alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def slsqp_barrier(h, n, T, alpha, maxiter, deadline):

    dx = 2.0 / n

        barrier = np.sum(np.maximum(0, g - T)**2)
        sm = smooth_max(g, alpha)

def binary_sa(h_bin, n, deadline, rng, max_iters=None):

    h = h_bin.copy()
    dx = 2.0 / n
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    g = np.correlate(h, 1-h, 'full') * dx
    current_cost = g.max()
    best_h = h.copy()
    best_cost = current_cost
    
    T_start = 0.05  # Initial temperature (in units of C5)
    T_end = 0.0001
    n_total = 2*n - 1
    
    if max_iters is None:
        max_iters = 500000
    
    it = 0
    while it < max_iters and time.time() < deadline:
        # Periodic reheat
        if it % 50000 == 49999:
            T_start = max(T_start * 0.5, 0.001)
        
        frac = it / max_iters
        T = T_start * (T_end / T_start) ** frac
        
        # Pick random swap
        i = int(rng.randint(len(ones_list)))
        j = int(rng.randint(len(zeros_list)))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oz] = 1.0

        new_cost = g_new.max()

            g = g_new
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_cost < best_cost - 1e-14:
                best_cost = new_cost
                best_h = h.copy()
        else:

            h[oz] = 0.0
        
        it += 1
    
    return best_h, best_cost

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_smooth_max(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(g, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

        oz = zeros_list[j_idx]

            g = g_new
            ones_list[i_idx] = oz
            zeros_list[j_idx] = oi
            if new_cost < best_cost - 1e-15:

                best_h = h.copy()

**Approach: Direct formulation as a quadratic program**

For a given target C (we want to check if C5 ≤ C is feasible):
- Variables: h[i] for i=0,...,n-1
- Constraints: 
  - ∑ h[i] = n/2
  - 0 ≤ h[i] ≤ 1
  - ∑_i h[i]*(1-h[i+k]) ≤ C/dx for all k

The last constraint is quadratic: ∑_i h[i] - ∑_i h[i]*h[i+k] ≤ C/dx

This is a QCQP (quadratically constrained quadratic program). We could use cvxpy if the problem is convex, but it's not (the constraint ∑ h[i]*h[i+k] ≥ ∑ h[i] - C/dx is not convex in general).

However, for binary h, the constraints become linear:
- h[i] ∈ {0, 1}
- ∑ h[i] = n/2
- ∑_i h[i]*(1-h[i+k]) ≤ C/dx for all k

The third constraint for binary h: ∑_i h[i]*(1-h[i+k]) = #{i: h[i]=1, h[i+k]=0} ≤ C/dx

This is a 0-1 integer program, which is NP-hard in general. But for n=250, we might be able to solve it with a MILP solver if we have one available (like SCIP or CBC).

Let me think about this: the MILP would have:
- 250 binary variables
- 1 linear constraint (sum = 125)
- 499 linear constraints (one for each lag k)
- Box constraints (implicit in binary variables)

This is a relatively small MILP! Modern solvers like CBC or SCIP should be able to handle this. The question is whether cvxpy can interface with these solvers for binary variables.

            ones[ii] = oz
            zeros[jj] = oi
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        elif delta < 1e-15:
            cur = new_c
            g = g_new
            ones[ii] = oz
            zeros[jj] = oi
        else:
            if rng.random() < np.exp(-delta / T):
                cur = new_c
                g = g_new
                ones[ii] = oz
                zeros[jj] = oi
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oz] = 0.0
        it += 1
    return best_h, best_c

def greedy_binary_full(h, n, deadline, max_passes=50):
    h = h.copy().astype(float)
    cur = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        for i in ones:
            if time.time() > deadline:
                return best_h, best_c
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h, n)
                if c < cur - 1e-14:
                    cur = c
                    improved = True
                    zeros.remove(j)
                    ones.remove(i)
                    ones.append(j)
                    zeros.append(i)
                else:
                    h[i], h[j] = 1.0, 0.0
        if improved:
            best_h = h.copy()
            best_c = cur
        else:
            break
    return best_h, best_c

def targeted_greedy(h, n, deadline):
    h = h.copy().astype(float)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, 'full') * dx
    cur = float(g.max())
    best_h = h.copy()
    best_c = cur
    for _ in range(200):
        if time.time() > deadline:
            break
        k_max = int(np.argmax(g))
        offset = k_max - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        witnesses = []
        for i in range(i_lo, i_hi + 1):
            j = i + offset
            if 0 <= j < n and h[i] > 0.5 and h[j] < 0.5:
                witnesses.append((i, j))
        if not witnesses:
            break
        improved = False
        for (i, j) in witnesses:
            h[i], h[j] = 0.0, 1.0
            g_new = np.correlate(h, 1 - h, 'full') * dx
            new_c = float(g_new.max())
            if new_c < cur - 1e-14:
                cur = new_c
                g = g_new
                improved = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_c

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = project_feasible(np.full(n0, 0.5), n0)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0

    # Phase 1: Continuous SLSQP on initial (12% budget)
    p1_end = t0 + 0.12 * (budget_s - 3)
    h = best_h.copy()
    for p in [8, 16, 32]:
        if time.time() > p1_end:
            break
        h = slsqp_pnorm(h, n0, p, 2000, p1_end)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    for alpha in [50000, 500000]:
        if time.time() > p1_end:
            break
        h = slsqp_lse(h, n0, alpha, 2000, p1_end)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary SA (55% budget)
    p2_end = t0 + 0.67 * (budget_s - 3)
    h_bin = make_binary(best_h, n0)
    c_bin = compute_c5(h_bin, n0)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    # Intensive SA with restarts
    sa_total = p2_end - time.time()
    n_restarts = 4
    sa_each = sa_total / n_restarts
    for r in range(n_restarts):
        t_dead = time.time() + sa_each
        if t_dead > p2_end:
            t_dead = p2_end
        if t_dead - time.time() < 20:
            break
        if r == 0:
            h_sa = h_bin.copy()
        else:
            h_sa = make_binary(best_h, n0)
            n_pert = 5 * (r + 1)
            for _ in range(n_pert):
                ol = list(np.where(h_sa == 1)[0])
                zl = list(np.where(h_sa == 0)[0])
                i = ol[int(rng.randint(len(ol)))]
                j = zl[int(rng.randint(len(zl)))]
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        h_sa, c_sa = sa_binary(h_sa, n0, t_dead, rng)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()

    # Phase 3: Greedy + targeted (8% budget)
    p3_end = t0 + 0.75 * (budget_s - 3)
    if time.time() < p3_end - 5:
        h_g, c_g = greedy_binary_full(best_h.copy(), n0, min(time.time() + 60, p3_end - 30), max_passes=30)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g.copy()
        if time.time() < p3_end - 5:
            h_t, c_t = targeted_greedy(best_h.copy(), n0, p3_end)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()

    # Phase 4: Continuous refinement from best binary (10% budget)
    p4_end = t0 + 0.85 * (budget_s - 3)
    if time.time() < p4_end - 10:
        h_cont = best_h.copy().astype(float)
        if np.all((h_cont == 0) | (h_cont == 1)):
            h_cont = h_cont + 0.01 * rng.randn(n0)
            h_cont = project_feasible(h_cont, n0)
        for p in [16, 32, 64]:
            if time.time() > p4_end:
                break
            h_cont = slsqp_pnorm(h_cont, n0, p, 3000, p4_end)
            c = compute_c5(h_cont, n0)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()
        for alpha in [100000, 1000000]:
            if time.time() > p4_end:
                break
            h_cont = slsqp_lse(h_cont, n0, alpha, 2000, p4_end)
            c = compute_c5(h_cont, n0)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()

    # Phase 5: Higher resolution (10% budget)
    p5_end = dl - 5
    for n_new in [400]:
        if time.time() > p5_end - 40:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_new)
        t_dead = min(time.time() + 70, p5_end - 5)
        for p in [16, 32]:
            if time.time() > t_dead:
                break
            h_hi = slsqp_pnorm(h_hi, n_new, p, 2000, t_dead)
        c_hi = compute_c5(h_hi, n_new)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi.copy()
            n_best = n_new

    # Final polish
    if time.time() < dl - 3:
        n_fin = len(best_h)
        h_fin = slsqp_pnorm(best_h.copy(), n_fin, 64, 2000, dl - 2)
        c_fin = compute_c5(h_fin, n_fin)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
