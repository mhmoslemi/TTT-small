            if tot < 1e-15:

        grad[m:] -= dx * h[:n - m]
    else:
        grad[m:] += dx * (1.0 - h[:n - m])
        grad[:n - m] -= dx * h[m:]

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def projected_gradient_bb(h, n, alpha, deadline, max_iters=10000):
    """Projected gradient with Barzilai-Borwein step size."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    x_prev = None
    g_prev = None
    eta_bb = 0.01  # initial step size
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        g = _slsqp_obj_grad(h_curr, n, alpha)
        # Project gradient onto the null space of the sum constraint
        # The constraint is sum(h) = n/2, so the normal is ones
        # Projected gradient: g - (g . ones / n) * ones
        g_proj = g - np.mean(g) * np.ones(n)
        
        if x_prev is not None and g_prev is not None:
            s = h_curr - x_prev
            y = g - g_prev
            sy = s @ y
            yy = y @ y
            if yy > 1e-20:
                if sy > 1e-10:
                    eta_bb = sy / yy
                else:
                    eta_bb = 1.0 / max(np.linalg.norm(y), 1e-10)
                eta_bb = np.clip(eta_bb, 1e-6, 0.1)
        
        x_prev = h_curr.copy()
        g_prev = g.copy()
        
        # Line search (backtracking)
        eta = eta_bb
        for ls in range(20):
            h_trial = h_curr - eta * g_proj
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c or c_trial < compute_c5(h_curr, n) - 1e-16:
                break
            eta *= 0.5
        else:
            break
        
        h_new = project_feasible(h_curr - eta * g_proj, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
        
        if eta < 1e-12:
            break
    
    return best_h, best_c

        grad[:n - m] -= dx * h[m:]
    return grad

def smooth_max_vals(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _obj_func(x, n, alpha):
    c = compute_all_lags(x, n)
    return smooth_max_vals(c, alpha)

def _obj_grad_func(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        weights = np.exp(shifted - mx)
        weights /= weights.sum()
    else:
        ew = np.exp(shifted)
        weights = ew / ew.sum()
    grad = np.zeros(n)
    for idx in range(len(weights)):
        if weights[idx] < 1e-18:
            continue
        lag = idx - (n - 1)
        grad += weights[idx] * compute_lag_gradient(x, n, lag)
    return grad

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_func, h, args=(n, alpha), jac=_obj_grad_func,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def proj_grad_bb(h, n, alpha, deadline, max_iters=5000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    x_prev = None
    g_prev = None
    eta = 0.01
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        g = _obj_grad_func(h_curr, n, alpha)
        g_proj = g - np.mean(g)  # project onto null space of sum constraint
        
        if x_prev is not None and g_prev is not None:
            s = h_curr - x_prev
            y = g - g_prev
            sy = float(s @ y)
            yy = float(y @ y)
            if yy > 1e-20:
                if sy > 1e-10:
                    eta_new = sy / yy
                else:
                    eta_new = 1.0 / max(np.linalg.norm(y), 1e-10)
                eta = float(np.clip(eta_new, 1e-8, 0.05))
        
        x_prev = h_curr.copy()
        g_prev = g.copy()
        
        # Backtracking line search
        step = h_curr - eta * g_proj
        h_trial = project_feasible(step, n)
        c_curr = compute_c5(h_curr, n)
        for _ in range(15):
            c_trial = compute_c5(h_trial, n)
            if c_trial < c_curr - 1e-14:
                break
            eta *= 0.5
            step = h_curr - eta * g_proj
            h_trial = project_feasible(step, n)
        
        h_new = project_feasible(h_curr - eta * g_proj, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        if not np.allclose(h_new, h_curr, atol=1e-12):
            h_curr = h_new
        else:
            break
    
    return best_h, best_c

def local_search_fine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-12:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-12:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [5000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Projected gradient with BB
    if time.time() < deadline - 20:
        pg_end = min(time.time() + 60, deadline - 10)
        h_pg, c_pg = proj_grad_bb(best_h, n, 500000, pg_end)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    
    # Phase 3: SLSQP final polish
    if time.time() < deadline - 10:
        h_new = slsqp_optimize(best_h, n, 5000000, maxiter=5000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Local search
    if time.time() < deadline - 3:
        ls_end = min(deadline - 1, time.time() + 30)
        best_h, best_c = local_search_fine(best_h, n, ls_end)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
