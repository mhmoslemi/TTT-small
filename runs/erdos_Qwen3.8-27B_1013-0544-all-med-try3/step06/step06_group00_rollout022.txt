def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_polytope(y, target_sum):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Bisection on the threshold
    s = y.sum()
    if abs(s - target_sum) < 1e-15:
        return np.clip(y, 0, 1)
    lo = y.min() - 1
    hi = y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def fista_smooth(h_init, n, alpha, n_iter, deadline, step_override=None):
    """FISTA on the log-sum-exp smoothing of the max correlation."""
    dx = 2.0 / n
    target = n / 2.0
    x = h_init.copy()
    z = h_init.copy()
    t = 1.0
    L = step_override if step_override else 8.0 * alpha / n  # Lipschitz estimate

        if time.time() > deadline:

        max_c = corr.max()

    dx = 2.0 / n
    target = n / 2.0
    x = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)

        corr = np.correlate(x, 1.0 - x, mode='full').astype(np.float64) * dx

        exp_c = np.exp(alpha * (corr - max_c))

        B = np.convolve(p[::-1], x, mode='full')[2*n-3 : 3*n-3]
        grad = dx * (A - B)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (k + 1))
        v_hat = v / (1 - beta2 ** (k + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x = project_onto_polytope(x, target)
    
    return x

def greed_binary(h_init, n, deadline, max_sweeps=200):
    """Greedy binary local search: swap a 0 with a 1 if it reduces the max."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)

    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]

            break
        improved = False

            best_delta = np.inf

                if nc < best_delta:

                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_delta
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def _compute_swap_delta(h, corr, i, j, n, m_range):
    """Compute the change in correlation when swapping h[i] from 0 to 1 and h[j] from 1 to 0."""
    delta = np.zeros(2 * n - 1)
    # Turning on position i (was 0, now 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    # Turning off position j (was 1, now 0)
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    # Correction for the self-interaction at lag n-1
    delta[n - 1] -= 2.0
    # Correction for the cross terms
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta
