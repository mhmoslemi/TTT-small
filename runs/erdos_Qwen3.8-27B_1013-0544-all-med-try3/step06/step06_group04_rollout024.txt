def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp(x, alpha):
    m = x.max()
    shifted = alpha * (x - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    if mx < -500:
        return float(m + np.log(len(x)) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        lags = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(lags))
        L_star = k_star - (n - 1)
        grad = get_lag_gradient(h_curr, n, L_star)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad_unit = grad / grad_norm
        for _ in range(30):
            step_try = step
            h_try = h_curr - step_try * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_h = h_try.copy()
                h_curr = h_try
                stall = 0
                step = min(step * 1.1, 0.5)
                break
            step_try *= 0.5
            if step_try < 1e-10:
                break
        else:
            stall += 1
            step *= 0.7
            if stall > 200:
                h_curr = project_feasible(h_curr + 0.01 * np.random.randn(n), n)
                stall = 0
    return best_h, best_c

def pairwise_swap(h, n, L, max_rounds=200):
    h_curr = h.copy()
    m = abs(L)
    if m == 0 or m >= n:
        return h_curr
    best_c = compute_c5(h_curr, n)
    for _ in range(max_rounds):
        improved = False
        if L > 0:
            for i in range(n - m):
                if h_curr[i] > h_curr[i + m] + 1e-10:
                    delta = min(h_curr[i], 1.0 - h_curr[i + m])
                    if delta > 1e-10:
                        h_curr[i] -= delta
                        h_curr[i + m] += delta
                        improved = True
        else:
            for i in range(m, n):
                if h_curr[i] > h_curr[i - m] + 1e-10:
                    delta = min(h_curr[i], 1.0 - h_curr[i - m])
                    if delta > 1e-10:
                        h_curr[i] -= delta
                        h_curr[i - m] += delta
                        improved = True
        if not improved:
            break
        h_curr = project_feasible(h_curr, n)
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c - 1e-14:
            best_c = c_curr
        else:
            break
    return h_curr

def partial_swap(h, n, L, frac, max_rounds=100):
    h_curr = h.copy()
    m = abs(L)
    if m == 0 or m >= n:
        return h_curr
    best_c = compute_c5(h_curr, n)
    for _ in range(max_rounds):
        improved = False
        if L > 0:
            for i in range(n - m):
                if h_curr[i] > h_curr[i + m] + 1e-10:
                    delta = min(h_curr[i], 1.0 - h_curr[i + m]) * frac
                    if delta > 1e-10:
                        h_curr[i] -= delta
                        h_curr[i + m] += delta
                        improved = True
        else:
            for i in range(m, n):
                if h_curr[i] > h_curr[i - m] + 1e-10:
                    delta = min(h_curr[i], 1.0 - h_curr[i - m]) * frac
                    if delta > 1e-10:
                        h_curr[i] -= delta
                        h_curr[i - m] += delta
                        improved = True
        if not improved:
            break
        h_curr = project_feasible(h_curr, n)
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c - 1e-14:
            best_c = c_curr
        else:
            break
    return h_curr

def slsqp_solve(h, n, top_idx, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj_and_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        active = c[top_idx]
        m_val = active.max()
        shifted = alpha * (active - m_val)
        mx = shifted.max()
        if mx > 500:
            val = float(m_val + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
            probs = np.exp(shifted - mx)
            probs /= probs.sum()
        else:
            val = float(m_val + np.log(np.sum(np.exp(shifted))) / alpha)
            probs = np.exp(shifted)
            probs /= probs.sum()
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            L_i = top_idx[i] - (n - 1)
            grad += probs[i] * get_lag_gradient(x, n, L_i)
        grad *= dx
        return val, grad

    def sum_fun(x):
        return np.sum(x) - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        val, g = obj_and_grad(h)
        res = minimize(obj_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() > deadline:
            return h.copy()
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search_refine(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    for frac in [1.0, 0.5, 0.3, 0.1, 0.05, 0.01]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            lags = compute_all_lags(h_curr, n)
            k_star = int(np.argmax(lags))
            L_star = k_star - (n - 1)
            if L_star == 0:
                break
            h_new = partial_swap(h_curr, n, L_star, frac, max_rounds=50)
            if h_new is not h_curr:
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h_curr = h_new
                    improved = True
    return h_curr, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def optimize_manager(h_start, n, deadline, use_slsqp=True):
    h_curr = project_feasible(h_start.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)

    sg_deadline = min(deadline - 30, time.time() + (deadline - time.time()) * 0.4)
    h_sg, c_sg = subgradient_optimize(h_curr, n, sg_deadline)
    if c_sg < best_c:
        best_c = c_sg
        best_h = h_sg

    if time.time() < deadline - 20:
        lags = compute_all_lags(best_h, n)
        L_star = int(np.argmax(lags)) - (n - 1)
        h_sw = pairwise_swap(best_h, n, L_star, max_rounds=100)
        c_sw = compute_c5(h_sw, n)
        if c_sw < best_c:
            best_c = c_sw
            best_h = h_sw

    if use_slsqp and n <= 2000 and time.time() < deadline - 30:
        for alpha in [50000, 500000]:
            if time.time() > deadline - 20:
                break
            lags = compute_all_lags(best_h, n)
            top_m = 5
            top_idx = np.argsort(lags)[-top_m:]
            h_sl = slsqp_solve(best_h, n, top_idx, alpha, maxiter=3000, deadline=deadline - 15)
            c_sl = compute_c5(h_sl, n)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl

    if time.time() < deadline - 5:
        h_ls, c_ls = local_search_refine(best_h, n, deadline - 1)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    best_h = None
    best_c = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    if h_init is not None:
        n_init_use = n_init
        alloc = min(120, (deadline - t0) * 0.12)
        if alloc > 30:
            h_opt, c_opt = optimize_manager(h_init, n_init_use, t0 + alloc, use_slsqp=True)
            best_h, best_c = h_opt, c_opt

    n_targets = [500, 1000, 2000, 3000]
    for n_try in n_targets:
        if time.time() > deadline - 60:
            break
        remaining = deadline - time.time() - 30
        if remaining < 40:
            break
        alloc_frac = 0.22
        alloc = min(remaining * alloc_frac, 200)
        if alloc < 30:
            break

        if best_h is not None:
            h_start = upsample(best_h, len(best_h), n_try)
        elif h_init is not None:
            h_start = upsample(h_init, n_init, n_try)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)

        use_sl = n_try <= 2000
        h_opt, c_opt = optimize_manager(h_start, n_try, time.time() + alloc, use_slsqp=use_sl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 40:
        n_best = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 20) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 15) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_manager(hp, n_best, time.time() + alloc, use_slsqp=(n_best <= 1500))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        h_new = slsqp_solve(best_h, n_best, np.argsort(compute_all_lags(best_h, n_best))[-5:],
                            1000000, 5000, deadline - 5)
        c_new = compute_c5(h_new, n_best)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

        best_h, best_c = local_search_refine(best_h, n_best, deadline - 2)
        best_h = project_feasible(best_h, n_best)
        best_c = compute_c5(best_h, n_best)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
