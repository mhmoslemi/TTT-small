If h is binary (0 or 1), then h(x)(1-h(x+k)) = 1 only when h(x)=1 and h(x+k)=0. So c_k = (number of positions where h[i]=1 and h[i+k]=0) * dx.

For a binary h with n/2 ones and n/2 zeros (to satisfy the sum constraint), we want to minimize the maximum over k of the number of "1 followed by 0 at distance k" patterns.

This is related to the problem of arranging 1s and 0s to minimize the maximum "disagreement" at all shifts.

For a periodic pattern with period p, the correlations at shifts that are multiples of p will be the same. So we want a period-p pattern that minimizes the max disagreement.

For period 2 (alternating 1,0,1,0,...): at shift 1, all positions disagree (c_1 = n/2 * dx = 1). At shift 2, all positions agree (c_2 = 0). So max = 1, which is bad.

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(m + lse)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-10:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            grad[l:n] += dx * w[k_idx] * (1 - x[0:n-l])
            grad[0:n-l] -= dx * w[k_idx] * x[l:n]
        else:
            ll = -l
            grad[0:n-ll] += dx * w[k_idx] * (1 - x[ll:n])
            grad[ll:n] -= dx * w[k_idx] * x[0:n-ll]
    return grad

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    
    def constraint_fun(x):
        return x.sum() - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(obj, h, args=(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
