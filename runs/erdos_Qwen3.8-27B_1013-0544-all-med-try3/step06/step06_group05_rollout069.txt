2. Coordinate descent: for each p, try δ ∈ {±0.001, ±0.005, ±0.01} and pick the one that decreases the objective (evaluated exactly)

So the change in the MAX is bounded by max_k |∂c[k]/∂h[p]| * |δ|, but I need to check if it actually changes the max.

import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_active(h, k_star, n):
    """Compute gradient of c[k_star] with respect to h."""
    # k_star is the index in the full correlation (0 to 2n-2)
    # lag = k_star - (n-1)
    ell = k_star - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        # c = sum_{m=0}^{n-1-ell} h[m+ell] * (1-h[m])
        # ∂c/∂h[p] = (1-h[p-ell]) * [ell <= p <= n-1] - h[p+ell] * [0 <= p <= n-1-ell]
        p_range1 = range(ell, n)  # where h[p] = h[m+ell]
        g[p_range1] += (1 - h[p_range1 - ell])
        p_range2 = range(0, n - ell)  # where h[p] = h[m]
        g[p_range2] -= h[p_range2 + ell]
    else:
        s = -ell
        # c = sum_{m=s}^{n-1} h[m-s] * (1-h[m])
        # ∂c/∂h[p] = (1-h[p+s]) * [0 <= p <= n-1-s] - h[p-s] * [s <= p <= n-1]
        p_range1 = range(0, n - s)  # where h[p] = h[m-s]
        g[p_range1] += (1 - h[p_range1 + s])
        p_range2 = range(s, n)  # where h[p] = h[m]
        g[p_range2] -= h[p_range2 - s]
    return g

def subgradient_descent(h, n, deadline, n_iter=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    eta0 = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        k_star = np.argmax(c)
        g = grad_active(h, k_star, n)
        
        # Polyak step size
        # We want to decrease the max, so use eta = (t_k - t*) / ||g||^2
        # Since we don't know t*, use a heuristic
        eta = eta0 / (it + 1) ** 0.5
        
        h_new = h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        # Backtracking if it didn't improve
        if c_new > best_c:
            # Try smaller step
            h_new = h - eta * 0.1 * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Don't change h, just reduce eta
            eta0 *= 0.95
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, deltas=None):
    """Coordinate descent: change one element, adjust another for sum constraint."""
    if deltas is None:

    n = len(h)

            break
        for _pass in range(200):
            if time.time() > deadline:
                return h, best_c
            improved = False
            # Compute current correlation
            c = compute_corr(h)
            c_max = c.max()
            
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_corr(h)
                        c_new = c.max() * dx
                        if c_new < best_c - 1e-15:

                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d

    return h, best_c

Let me use the gradient to determine good (i, j) pairs:

**Gradient-guided coordinate descent:**
1. Compute gradient g of the active constraint
2. The steepest descent direction is: increase h where g is most negative, decrease h where g is most positive
3. So: i* = argmin(g) (increase h[i*]), j* = argmax(g) (decrease h[j*])
4. Take a step in that direction

This is O(n) per step (to compute g and find min/max), and I can do many steps.

def gradient_descent_pair(h, n, deadline, n_iter=10000):

    best_c = compute_c5(h, n)
    best_h = h.copy()

        if time.time() > deadline:

        k_star = np.argmax(c)
        g = grad_active(h, k_star, n)

        i = np.argmin(g)  # increase h[i]
        j = np.argmax(g)  # decrease h[j]
        
        if i == j:
            break
        
        # Max step: limited by bounds
        d_max = min(1.0 - h[i], h[j])
        
        # Try a fraction of the max step
        for frac in [1.0, 0.5, 0.1, 0.01]:
            d = d_max * frac
            h[i] += d
            h[j] -= d
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                break
            else:
                h[i] -= d
                h[j] += d
        
    return best_h, best_c
