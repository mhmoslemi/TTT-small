import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def make_binary_pattern(n, block_size, n_blocks, offset=0):
    """Create a pattern of 1-blocks and 0-blocks."""
    h = np.zeros(n)
    period = 2 * block_size
    for i in range(n):
        pos = (i + offset) % period
        h[i] = 1.0 if pos < block_size else 0.0
    return h

def obj_fn(h, dx, beta):
    corr = get_corr(h)
    return smooth_max(corr, beta) * dx

def sum_fun(h, target_sum):
    return h.sum() - target_sum

def sum_jac(h, target_sum):
    return np.ones(len(h))

def pairwise_search(h, target_sum, n, max_iter=5000, time_limit=30):
    """Pairwise coordinate descent."""
    t_start = time.time()
    best = compute_c5(h)
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        # Pick random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h[i] <= 1e-8 or h[i] >= 1 - 1e-8 or h[j] <= 1e-8 or h[j] >= 1 - 1e-8:
            continue
        
        # Try moving mass from j to i
        # h[i] += d, h[j] -= d
        # Optimal d: try a few values
        max_d_pos = min(1.0 - h[i], h[j])
        max_d_neg = min(h[i], 1.0 - h[j])
        
        candidates_d = np.concatenate([
            np.linspace(0.01, max_d_pos, 5),
            -np.linspace(0.01, max_d_neg, 5)
        ])
        
        best_d = 0.0
        best_c5 = best
        
        for d in candidates_d:
            if abs(d) < 1e-6:
                continue
            test = h.copy()
            test[i] = h[i] + d
            test[j] = h[j] - d
            if test[i] < -1e-12 or test[i] > 1 + 1e-12 or test[j] < -1e-12 or test[j] > 1 + 1e-12:
                continue
            test = np.clip(test, 0, 1)
            c5_test = compute_c5(test)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_d = d
        
        if best_d != 0:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            best = best_c5
    
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Resolutions to try (sorted by priority)
    # Higher resolution = more degrees of freedom = potentially lower C5
    # But also slower optimization
    if n_init >= 100:
        resolutions = [n_init, n_init + 50, n_init + 100, n_init + 150]
    else:
        resolutions = [n_init, n_init * 2, n_init * 3, n_init * 4]
    resolutions = sorted(set(r for r in resolutions if 40 <= r <= 400))
    
    for n in resolutions:
        if remaining() < 80:
            break
        
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        
        # Multiple initializations
        inits = [resample_h(best_h, n)]
        
        # Add a binary block pattern
        if n % 4 == 0:
            bs = n // 8  # block size
            inits.append(make_binary_pattern(n, bs, 4))
        
        # Add a random initialization
        h_rand = np.random.rand(n)
        h_rand = (h_rand - h_rand.min()) / (h_rand.max() - h_rand.min() + 1e-10)
        inits.append(h_rand)
        
        for h_start in inits:
            if remaining() < 30:
                break
            
            h = project_feasible(h_start, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # SLSQP optimization with increasing beta
            for beta in [10, 50, 200, 1000, 5000, 20000]:
                if remaining() < 20:
                    break
                try:
                    res = minimize(
                        obj_fn, h, args=(dx, beta),
                        method='SLSQP', bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac,
                                     'args': (target_sum,)}],
                        options={'maxiter': 3000, 'ftol': 1e-14}
                    )
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    if c5_new < compute_c5(h):
                        h = h_new.copy()
                except Exception:
                    pass
    
    # Final pairwise search
    if remaining() > 20:
        n = len(best_h)
        h = best_h.copy()
        target_sum = n / 2.0
        h, c5 = pairwise_search(h, target_sum, n, 
                                max_iter=100000, 
                                time_limit=max(10, remaining() - 10))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, len(best_h)

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    m = corr.max()

    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    obj = dx * (m + np.log(exp_shifted.sum()) / beta)
    
    # Gradient: ∂f/∂h[j] = dx * Σ_k w_k * ((1-h[j-ℓ_k]) - h[j+ℓ_k])
    # where ℓ_k = k - (n-1)

            # (1-h[j-lag]): j-lag in [0, n-1] => j in [lag, n-1+lag], intersect [0,n-1] => j in [lag, n-1]
            # h[j+lag]: j+lag in [0, n-1] => j in [-lag, n-1-lag], intersect [0,n-1] => j in [0, n-1-lag]
            # For the first term: j in [lag, n-1]
            # For the second term: j in [0, n-1-lag]
            idx1 = np.arange(lag, n)  # for (1-h[j-lag])

            idx2 = np.arange(0, n - lag)  # for h[j+lag]

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_and_grad(h, dx, beta):
    """Compute objective and analytical gradient of smooth_max(corr)*dx."""
    n = len(h)
    corr = get_corr(h)
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    wsum = exp_shifted.sum()
    weights = exp_shifted / wsum
    obj = dx * (m + np.log(wsum) / beta)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        else:
            la = -lag
            if la == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[:n - la] += wk * (1.0 - h[la:])
                grad[la:] -= wk * h[:n - la]
    grad *= dx
    return obj, grad


def sum_con(x, target_sum):
    return x.sum() - target_sum


def sum_jac(x, target_sum):
    return np.ones(len(x))


def pairwise_search(h, target_sum, n, time_limit):
    """Aggressive pairwise coordinate descent."""
    t_start = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    delta_set = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]
    
    while time.time() - t_start < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        hi = h[i]
        hj = h[j]
        if hi <= 1e-10 or hi >= 1.0 - 1e-10 or hj <= 1e-10 or hj >= 1.0 - 1e-10:
            continue
        if i == j:
            continue
        
        best_d = 0.0
        best_val = best_c5
        for d in delta_set:
            for sign in [1, -1]:
                dd = sign * d
                new_hi = hi + dd
                new_hj = hj - dd
                if new_hi < -1e-12 or new_hi > 1.0 + 1e-12:
                    continue
                if new_hj < -1e-12 or new_hj > 1.0 + 1e-12:
                    continue
                test = h.copy()
                test[i] = new_hi
                test[j] = new_hj
                c5_test = compute_c5(test)
                if c5_test < best_val - 1e-14:
                    best_val = c5_test
                    best_d = dd
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0.0, 1.0)
            best_c5 = best_val
    
    return h, best_c5

        if 40 <= n <= 400 and n not in resolutions_to_try:
            resolutions_to_try.append(n)
    resolutions_to_try = sorted(resolutions_to_try)

    beta_schedule = [20, 80, 300, 1000, 4000, 15000, 50000]

    for n in resolutions_to_try:

        target_sum = n / 2.0

        else:

                    return obj_and_grad(x, dx, beta)[0]

                    return obj_and_grad(x, dx, beta)[1]

    # Phase 2: More aggressive optimization with very high beta on best resolution

                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)

                except Exception:

    # Phase 3: Try slightly different resolutions for final push

                continue

                    return obj_and_grad(x, dx_t, beta)[1]
