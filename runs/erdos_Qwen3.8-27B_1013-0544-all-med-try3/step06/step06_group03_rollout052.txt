import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    p = np.exp(sh)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p = p / p_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        j = np.arange(n)
        idx1 = j - k + (n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        if m1.any():
            grad[m1] += pk * (1.0 - h[idx1[m1]])
        idx2 = j + k - (n - 1)
        m2 = (idx2 >= 0) & (idx2 < n)
        if m2.any():
            grad[m2] -= pk * h[idx2[m2]]
    return grad

def gradient_optimize(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [50, 150, 500, 1500, 5000, 15000, 50000, 150000]
    n_phases = len(alphas)
    for a_idx in range(n_phases):
        if time.time() > deadline - 3:
            break
        alpha = alphas[a_idx]
        remaining_phases = n_phases - a_idx
        time_left = deadline - time.time() - 3
        if time_left <= 0:
            break
        phase_time = time_left / remaining_phases
        phase_deadline = time.time() + phase_time
        lr = 1.0 / (alpha * np.sqrt(n) * 0.5)
        momentum = 0.85
        v = np.zeros(n)
        h_cur = h.copy()
        c_cur = compute_c5(h_cur, n)
        stall = 0
        while time.time() < phase_deadline:
            h_look = project_feasible(h_cur - momentum * v, n)
            grad = compute_gradient(h_look, n, alpha)
            v = momentum * v - lr * grad
            h_new = project_feasible(h_cur + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur - 1e-13:
                c_cur = c_new
                stall = 0
                if rng.random() < 0.05:
                    lr = min(lr * 1.03, 10.0 / alpha)
            else:
                stall += 1
                lr *= 0.92
                v *= 0.4
            h_cur = h_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
            if stall > 3000:
                break
        h = h_cur
    return best_h, best_c

def block_move_search(h, n, deadline):
    best_c = compute_c5(h, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n - 2):
            if time.time() > deadline:
                return h, best_c
            a, b, c = h[i], h[i+1], h[i+2]
            h[i], h[i+1], h[i+2] = b, c, a
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                h[i], h[i+1], h[i+2] = a, b, c
        for i in range(n - 2):
            if time.time() > deadline:
                return h, best_c
            a, b, c = h[i], h[i+1], h[i+2]
            h[i], h[i+1], h[i+2] = c, a, b
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                h[i], h[i+1], h[i+2] = a, b, c
    return h, best_c

def triple_move_search(h, n, deadline):
    best_c = compute_c5(h, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n - 2):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 2, min(i + 6, n)):
                a, b, c = h[i], h[j-1], h[j]
                h[i], h[j-1], h[j] = b, c, a
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i], h[j-1], h[j] = a, b, c
    return h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

    t0 = time.time()
