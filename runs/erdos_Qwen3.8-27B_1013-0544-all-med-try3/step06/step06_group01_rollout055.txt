def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    shifted = alpha * (c - c.max())
    sm = shifted.max()
    return c.max() + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def soft_max_grad(x, n, alpha, lam=0.0):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    shifted = alpha * (c - c.max())
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights = weights / weights.sum()
    weights = weights * dx  # include dx factor
    
    g = 1 - x
    grad = np.zeros(n)
    m_range = np.arange(2 * n - 1)
    
    for i in range(n):

        t2 = np.dot(weights[mask2], x[idx2[mask2]])
