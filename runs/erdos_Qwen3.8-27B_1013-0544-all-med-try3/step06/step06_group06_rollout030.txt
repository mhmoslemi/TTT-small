```python
def fast_correlate(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Pad to next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    # Correlation: flip b
    c = np.fft.irfft(fa * np.conj(fb), fsize)[:size]
    return c
```

```python
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_gradient_at_max(h, n, g):
    """Compute gradient of max(g) w.r.t. h using analytical formula.
    g[k] = sum_i h[i] * (1 - h[i + k - n + 1]) * dx (for valid i)
    """
    dx = 2.0 / n
    kn = len(g)  # 2n-1
    km = int(np.argmax(g))
    # shift s = km - (n-1)
    s = km - (n - 1)
    grad = np.zeros(n)
    # d g[km] / d h[j] = dx * [(1 - h[j+s]) - h[j-s]]
    # where h[idx] = 0 if idx out of [0, n-1]
    for j in range(n):

        # First term: 1 - h[j+s] (if j+s in range)
        jp = j + s
        if 0 <= jp < n:

        # Second term: -h[j-s] (if j-s in range)
        jm = j - s
        if 0 <= jm < n:
            val -= h[jm]
        grad[j] = dx * val
    return grad

    g = compute_all_g(h, n)
    m = g.max()
    # Soft weights
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    w = exp_sh / exp_sh.sum()  # weights for each lag

    dx = 2.0 / n

            jm = j - s

                val -= h[jm]
            grad[j] += w[k] * dx * val
```

```python
    best_t = 0.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=100, deadline=None, verbose=False):
    """Coordinate descent with exact 1D subproblems."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline and time.time() > deadline:
                break
            
            old_val = h[j]
            h[j] = 0.0
            g_base = np.correlate(h, 1 - h, mode='full')
            
            # Compute D: coefficient of t in g(m) when h[j] changes from 0 to t
            D = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                s = m - (n - 1)
                d = 0.0
                if 0 <= j + s < n:
                    d += 1.0 - h[j + s]
                if 0 <= j - s < n:
                    d -= h[j - s]
                D[m] = d
            
            # Note: for m = n-1 (s=0):
            # g(n-1) = g_base[n-1] + t*(1-t) = g_base[n-1] + t - t²
            # D[n-1] = (1-h[j]) - h[j] = 1 (since h[j]=0)
            # So g_candidate[n-1] = g_base[n-1] + t*D[n-1] - t² = g_base[n-1] + t - t²✓
            
            # Find optimal t: minimize max(g_base + t*D) with correction at n-1
            # Use coarse grid + refinement
            best_t = old_val  # default: keep current value
            best_val = np.inf
            
            # Coarse grid
            for t in np.linspace(0, 1, 51):
                g_cand = g_base + t * D
                g_cand[n-1] = g_base[n-1] + t - t*t
                val = g_cand.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine
            lo = max(0, best_t - 0.02)
            hi = min(1, best_t + 0.02)
            for t in np.linspace(lo, hi, 51):
                g_cand = g_base + t * D
                g_cand[n-1] = g_base[n-1] + t - t*t
                val = g_cand.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            
            h[j] = best_t
            c = compute_c5(h, n)
            if c < best_c - 1e-13:
                best_c = c
                improved = True
        
        if not improved:
            break
    
    return h, best_c

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    """L-BFGS-B on the smooth (log-sum-exp) approximation."""
    dx = 2.0 / n
    
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        w = exp_sh / exp_sh.sum()
        val = m + (np.log(exp_sh.sum()) - sm) / alpha
        
        # Gradient (vectorized)
        grad = np.zeros(n)
        for k in range(len(g)):
            if w[k] < 1e-14:
                continue
            s = k - (n - 1)
            wk = w[k] * dx
            if s == 0:
                grad += wk * (1.0 - 2.0 * x)
            elif s > 0:
                if n - s > 0:
                    grad[:n-s] += wk * (1.0 - x[s:])
                if s < n:
                    grad[s:] += wk * (-x[:n-s])
            else:
                ss = -s
                if n - ss > 0:
                    grad[ss:] += wk * (1.0 - x[:n-ss])
                if ss < n:
                    grad[:n-ss] += wk * (-x[ss:])
        
        return val, grad
    
    # Ensure feasibility
    h = project_feasible(h, n)
    
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', jac=True,
                      bounds=[(0, 1)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = h
    
    return h_out

def pair_refine(h, n, deadline, deltas=None):
    """Fast pair refinement: try small transfers between pairs."""
    if deltas is None:
        deltas = [0.01, 0.005, 0.001, 0.0005]
    
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            g = compute_g(h, n)
            m = g.max()
            # Focus on lags near the maximum
            k_max = np.argmax(g)
            s_max = k_max - (n-1)
            # Try to reduce the max by adjusting coordinates involved in the worst lag
            for j in range(n):
                if time.time() > deadline:
                    break
                # Try transferring d from j to j+s_max (or j-s_max)
                for target in [j + s_max, j - s_max]:
                    if 0 <= target < n and target != j:
                        if h[j] >= d and h[target] <= 1.0 - d:
                            h[j] -= d
                            h[target] += d
                            c = compute_c5(h, n)
                            if c < best_c:
                                best_c = c
                                prog = True
                            else:
                                h[j] += d
                                h[target] -= d
    return h, best_c
```

```python
            #     g_base[n-1] = g[n-1] - h[j] + h[j]²
            
            hj = h[j]
            g_base = g - hj * D  # this is correct for m≠n-1
            # For m=n-1: g_base[n-1] should be g[n-1] - hj + hj²
            # But g - hj*D gives g[n-1] - hj*1 = g[n-1] - hj
            # We need g_base[n-1] = g[n-1] - hj + hj²
            # So: g_base[n-1] += hj²
            g_base = g.copy()
            g_base -= hj * D
            g_base[n-1] += hj * hj  # correction for the quadratic term
            
            # Now find optimal t: minimize max(g_base + t*D) with g_cand[n-1] = g_base[n-1] + t - t²
            # Note: g_base + t*D gives the correct value for m≠n-1
            # For m=n-1: g_base[n-1] + t*D[n-1] - t² = g_base[n-1] + t - t²
            
            best_t = hj
            best_val = np.inf
            
            # Grid search
            for t in np.linspace(0, 1, 51):
                g_cand = g_base + t * D
                g_cand[n-1] = g_base[n-1] + t - t*t
                val = g_cand.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine
            if 0 < best_t < 1:
                lo = max(0, best_t - 0.02)
                hi = min(1, best_t + 0.02)
                for t in np.linspace(lo, hi, 51):
                    g_cand = g_base + t * D
                    g_cand[n-1] = g_base[n-1] + t - t*t
                    val = g_cand.max()
                    if val < best_val:
                        best_val = val
                        best_t = t
            
            # Update
            if best_t != hj:
                delta = best_t - hj
                h[j] = best_t
                g = g_base + best_t * D
                g[n-1] = g_base[n-1] + best_t - best_t*best_t
                c = g.max() * (2.0/n)
                if c < best_c - 1e-13:
                    best_c = c
                    improved = True
        
        if not improved:
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)


def coord_descent_fast(h, n, max_sweeps=100, deadline=None, grid_pts=81, refine_pts=61):
    best_c = compute_c5(h, n)
    g = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    kn = 2 * n - 1

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break

            # Compute D: d g[m] / d h[j] for current h (with h[j] treated as variable)
            D = np.zeros(kn)
            for m in range(kn):
                s = m - (n - 1)
                d = 0.0
                jp = j + s
                jm = j - s
                if 0 <= jp < n:
                    d += 1.0 - h[jp]
                if 0 <= jm < n:
                    d -= h[jm]
                D[m] = d

            # g_base: values of g when h[j] = 0
            hj = h[j]
            g_base = g.copy()
            g_base -= hj * D
            g_base[n - 1] += hj * hj  # quadratic correction

            # Find optimal t via grid search (vectorized)
            ts = np.linspace(0.0, 1.0, grid_pts)
            g_cand = g_base[np.newaxis, :] + ts[:, np.newaxis] * D[np.newaxis, :]
            g_cand[:, n - 1] = g_base[n - 1] + ts - ts * ts
            vals = g_cand.max(axis=1)
            best_idx = np.argmin(vals)
            best_t = ts[best_idx]
            best_val = vals[best_idx]

            # Refine around best_t
            if 0.0 < best_t < 1.0:
                lo = max(0.0, best_t - 1.0 / grid_pts)
                hi = min(1.0, best_t + 1.0 / grid_pts)
                ts2 = np.linspace(lo, hi, refine_pts)
                g_cand2 = g_base[np.newaxis, :] + ts2[:, np.newaxis] * D[np.newaxis, :]
                g_cand2[:, n - 1] = g_base[n - 1] + ts2 - ts2 * ts2
                vals2 = g_cand2.max(axis=1)
                best_idx2 = np.argmin(vals2)
                if vals2[best_idx2] < best_val:
                    best_t = ts2[best_idx2]
                    best_val = vals2[best_idx2]

            # Update
            if abs(best_t - hj) > 1e-12:
                h[j] = best_t
                g = g_base + best_t * D
                g[n - 1] = g_base[n - 1] + best_t - best_t * best_t
                c = g.max() * dx
                if c < best_c - 1e-13:
                    best_c = c
                    improved = True
            # else: no change, g remains valid

        if not improved:
            break

    return h, best_c

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    def wrapper(x):
        return smooth_obj_grad(x, n, alpha)

    h = project_feasible(h, n)
    try:
        res = minimize(wrapper, h, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 20})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = h
    return h_out


def pair_refine(h, n, deadline):
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001]
    best_c = compute_c5(h, n)
    g = np.correlate(h, 1 - h, mode='full')

    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            k_max = np.argmax(g)
            s_max = k_max - (n - 1)
            # Try transfers along the worst lag direction
            tried = False
            for j in range(n):
                if time.time() > deadline:
                    break
                # Try moving mass from j toward j+s_max (reduces overlap at worst lag)
                for target in [j + s_max, j - s_max]:
                    if 0 <= target < n and target != j:
                        if h[j] >= d and h[target] <= 1.0 - d:
                            h[j] -= d
                            h[target] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                prog = True
                                g = np.correlate(h, 1 - h, mode='full')
                            else:
                                h[j] += d
                                h[target] -= d
    return h, best_c


def multi_start_coord_descent(h, n, deadline, n_restarts=20, scale=0.05):
    rng = np.random.RandomState(12345)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for i in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        h_pert, c = coord_descent_fast(h_pert, n, max_sweeps=30, deadline=deadline)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h_pert.copy()
            scale = max(0.01, scale * 0.7)
    return best_h, best_c
```
