def anneal_local_search(h, n, deadline, T0=0.001, T1=0.00001):
    """Simulated annealing style local search"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    total_steps = 20000
    current_T = T0
    
    for step in range(total_steps):
        if time.time() > deadline:
            break
        # Anneal temperature
        current_T = T0 * (T1 / T0) ** (step / max(total_steps, 1))
        
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        m = abs(k_star - (n - 1))
        if m == 0:
            break
        
        # Pick a random pair for the worst lag
        i = np.random.randint(0, n - m)
        j = i + m
        
        if abs(h[i] - h[j]) < 1e-5:
            continue
        
        delta = 0.0001 * np.sign(h[i] - h[j])
        if delta > h[i] or h[j] + delta > 1:
            continue
        
        h[i] -= delta
        h[j] += delta
        c_new = compute_c5(h, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        elif c_new > best_c:
            # Accept with probability exp(-(c_new - best_c)/T)
            if np.random.random() < np.exp(-(c_new - best_c) / current_T):
                pass  # accept
            else:
                h[i] += delta
                h[j] -= delta
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 300:
        shifted = shifted - mx
    s = np.sum(np.exp(shifted))
    if s < 1e-300:
        return float(m)
    return float(m + np.log(s) / alpha)

def _active_obj(x, n, top_idx, alpha):
