    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_anneal(h, n, deadline):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    alphas = [200, 800, 3000, 10000, 30000, 100000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new = single_slsqp(h_curr, n, alpha, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new
            h_curr = h_new
    return best_h, best_c

def lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=30):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 5e-9:
            trust = max(0.002, trust * 0.5)
        else:
            trust = max(0.002, trust * 0.85)
        h_curr = h_new
    return best_h, best_c

def multi_lag_local_search(h, n, deadline, top_k=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_idx = np.argsort(c)[-top_k:]
            for k_star in top_idx:
                if time.time() > deadline:
                    break
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                if m == 0:
                    break
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-10:
                        delta = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.8, (1.0 - h[j]) * 0.8)
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-10:
                        delta = min(d, (h[j] - h[i]) * 0.3, h[j] * 0.8, (1.0 - h[i]) * 0.8)
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return best_h, best_c

def swap_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-3:]
    for k_star in top_idx:
        if time.time() > deadline:
            break
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            continue
        for i in range(n - m):
            if time.time() > deadline:
                break
            j = i + m
            if abs(h[i] - h[j]) < 1e-6:
                continue
            trial = h.copy()
            trial[i], trial[j] = h[j], h[i]
            trial = project_feasible(trial, n)
            c_new = compute_c5(trial, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = trial
                h = trial
    return best_h, best_c

def pair_optimize(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-5:]
    for k_star in top_idx:
        if time.time() > deadline:
            break
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            continue
        for i in range(n - m):
            if time.time() > deadline:
                break
            j = i + m
            if h[i] <= 0.01 or h[i] >= 0.99 or h[j] <= 0.01 or h[j] >= 0.99:
                continue
            for frac in [0.3, 0.5, 0.7]:
                trial = h.copy()
                trial[i] = h[i] * (1 - frac)
                trial[j] = h[j] * (1 + frac) * (1 - h[i]) / max(h[j], 1e-10)
                trial[i] = min(max(trial[i], 0), 1)
                trial[j] = min(max(trial[j], 0), 1)
                trial = project_feasible(trial, n)
                c_new = compute_c5(trial, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = trial
                    h = trial
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    if time.time() < deadline - 20:
        alloc = min(60, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx(best_h, n, time.time() + alloc)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    
    if time.time() < deadline - 15:
        h_an, c_an = slsqp_anneal(best_h, n, time.time() + min(40, (deadline - time.time()) * 0.25))
        if c_an < best_c:
            best_c = c_an
            best_h = h_an
    
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-5:]
        for alpha in [50000, 200000]:
            if time.time() > deadline - 10:
                break
            h_new = active_slsqp(best_h, n, top_idx, alpha, maxiter=3000, ftol=1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-5:]
    
    if time.time() < deadline - 8:
        h_new = single_slsqp(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 3:
        ls_end = min(deadline - 1, time.time() + 20)
        best_h, best_c = multi_lag_local_search(best_h, n, ls_end, top_k=5)
    
    if time.time() < deadline - 2:
        best_h, best_c = swap_search(best_h, n, deadline - 0.5)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    
    h_parent = np.array(initial_h_values, dtype=float).copy()
    n_parent = len(h_parent)
    
    best_h = h_parent.copy()
    best_c = compute_c5(best_h, n_parent)
    
    # Phase 1: Optimize at parent resolution
    alloc1 = min(250, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_at_n(h_parent, n_parent, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Phase 2: Try other resolutions
    n_candidates = [250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 40
        if remaining < 40:
            break
        alloc = min(80, remaining * 0.22)
        if alloc < 30:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt2, c_opt2 = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2
    
    # Phase 3: Multi-start at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 60:
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            if n_restarts > 1:
                scale = 0.01 * (1.0 + 0.5 * r)
            else:
                scale = 0.008
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(40, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt3, c_opt3 = optimize_at_n(hp, n_best, time.time() + alloc, rng)
            if c_opt3 < best_c:
                best_c = c_opt3
                best_h = h_opt3
    
    # Phase 4: Final intensive polish
    if time.time() < deadline - 10:
        n = len(best_h)
        # Very fine SLSQP
        h_new = single_slsqp(best_h, n, 1000000, maxiter=6000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # Active set with very large alpha
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-7:]
        h_new = active_slsqp(best_h, n, top_idx, 2000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # Fine local search
        if time.time() < deadline - 2:
            best_h, best_c = multi_lag_local_search(best_h, n, deadline - 0.5, top_k=7)
        
        # Swap search
        best_h, best_c = swap_search(best_h, n, deadline - 0.2)
        
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
