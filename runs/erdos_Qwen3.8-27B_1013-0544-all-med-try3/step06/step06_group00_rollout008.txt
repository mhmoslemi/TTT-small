    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]

    try:
        res = minimize(obj_grad, x0, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return x0
        return project_feasible(res.x, n)
    except Exception:
        return x0

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary_fast(h_init, n, deadline, seed=42, n_iters=2000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].astype(int)
    zeros_arr = np.where(h == 0)[0].astype(int)
    T = 0.01
    cool = 0.99997
    scale = 2000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = False
        if nc <= cur_c:
            accept = True
        elif T > 1e-12:
            diff = scale * (nc - cur_c) / T
            if diff < 50 and rng.random() < np.exp(-diff):
                accept = True
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def multi_alpha_optimize(h, n, alphas, deadline, maxiter_per_alpha=800):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        h_new = slsqp_opt(best_h, n, alpha, maxiter_per_alpha, 1e-16, deadline - 1)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def perturbed_restart(base_h, n, alpha, deadline, rng, n_trials=3, noise_scale=0.02):
    best_h = base_h.copy()
    best_c = compute_c5(base_h, n)
    for _ in range(n_trials):
        if time.time() > deadline - 5:
            break
        h = base_h + noise_scale * rng.randn(n)
        h = project_feasible(h, n)
        h_new = slsqp_opt(h, n, alpha, 500, 1e-16, deadline - 1)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0 = np.zeros(n0)
        h0[idx] = 1.0

    n = n0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick binary local search
    p1 = t0 + 0.06 * total
    hb, cb = greed_binary(best_h, n, p1, max_sweeps=150)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Main SLSQP optimization with increasing alpha
    p2 = t0 + 0.45 * total
    alphas_main = [30, 100, 300, 1000, 3000, 10000, 30000, 100000, 500000]
    best_h, best_c = multi_alpha_optimize(best_h, n, alphas_main, p2, maxiter_per_alpha=600)

    # Phase 3: Annealing for diverse binary solution
    p3 = t0 + 0.58 * total
    ha, ca = anneal_binary_fast(best_h, n, p3, seed=seed + 1, n_iters=2000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
        # Re-smooth from new binary
        p3b = t0 + 0.68 * total
        alphas_re = [500, 5000, 50000, 500000]
        best_h, best_c = multi_alpha_optimize(best_h, n, alphas_re, p3b, maxiter_per_alpha=400)

    # Phase 4: Perturbation restarts at high alpha
    p4 = t0 + 0.82 * total
    best_h, best_c = perturbed_restart(best_h, n, 200000, p4, rng, n_trials=4, noise_scale=0.03)
    best_h, best_c = perturbed_restart(best_h, n, 500000, p4, rng, n_trials=3, noise_scale=0.01)

    # Phase 5: Try different initializations
    p5 = t0 + 0.92 * total
    # Random binary start
    h_rand = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h_rand[idx] = 1.0
    h_rand = project_feasible(h_rand, n)
    # Quick optimize from random
    alphas_rand = [100, 1000, 10000, 100000]
    hr, cr = multi_alpha_optimize(h_rand, n, alphas_rand, p5, maxiter_per_alpha=300)
    if cr < best_c:
        best_c = cr
        best_h = hr.copy()

    # Structured initialization (periodic-ish)
    h_struct = np.zeros(n)
    period = n // 4
    for i in range(n):
        if (i % (2 * period)) < period:
            h_struct[i] = 1.0
    h_struct = project_feasible(h_struct, n)
    hs, cs = multi_alpha_optimize(h_struct, n, alphas_rand, p5, maxiter_per_alpha=200)
    if cs < best_c:
        best_c = cs
        best_h = hs.copy()

    # Phase 6: Final polish with very high alpha
    p6 = dl - 2
    if time.time() < p6:
        best_h, best_c = multi_alpha_optimize(best_h, n, [1000000, 5000000, 10000000], p6, maxiter_per_alpha=2000)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
