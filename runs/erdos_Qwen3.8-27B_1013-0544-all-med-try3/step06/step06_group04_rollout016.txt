    S = h_pad[idx_plus] + h_pad[idx_minus]
    return S, idx_plus, idx_minus, h_pad

def binary_local_search(h_bin, n, deadline, max_passes=2000, use_sa=False, sa_T0=0.02, sa_T_end=0.001):
    """Efficient binary local search with swap moves."""
    h = h_bin.copy()
    c = np.correlate(h, 1 - h, mode='full')
    c5 = float(np.max(c))
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    S, idx_plus, idx_minus, h_pad = build_S_matrix(h, n)
    
    if use_sa:
        total_time = max(1, deadline - time.time())
    else:
        total_time = 1
    
    pass_count = 0
    no_improve_count = 0
    max_no_improve = 50
    
    while pass_count < max_passes:
        if time.time() > deadline:
            break
        pass_count += 1
        no_improve_count += 1
        
        # Find active lags (within some tolerance of max)
        tol = 0.005
        active_idx = np.where(c >= c5 - tol)[0]
        M = len(active_idx)
        if M == 0:
            active_idx = np.array([int(np.argmax(c))])
            M = 1
        
        # Filter: for swap (i,j), need S[j, active] > S[i, active] for all active
        Si_act = S[ones, :][:, active_idx] if len(ones) > 0 else np.empty((0, M))
        Sj_act = S[zeros, :][:, active_idx] if len(zeros) > 0 else np.empty((0, M))
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # diff[i_local, j_local, m] = Sj_act[j_local, m] - Si_act[i_local, m]
        # We want all > 0 (necessary condition for improvement)
        diff_act = Sj_act[None, :, :] - Si_act[:, None, :]
        filter_mask = np.all(diff_act > 0, axis=2)
        
        candidate_pairs = np.argwhere(filter_mask)
        
        best_new_c5 = c5
        best_pair = None
        best_delta_c = None
        
        for i_local, j_local in candidate_pairs:
            i = ones[i_local]
            j = zeros[j_local]
            delta_c = S[i] - S[j]
            new_c5 = float(np.max(c + delta_c))
            if new_c5 < best_new_c5 - 1e-12:
                best_new_c5 = new_c5
                best_pair = (i, j)
                best_delta_c = delta_c
        
        # Simulated annealing: also consider slightly worse moves
        if use_sa and best_pair is None and len(candidate_pairs) > 0:
            frac = min(1.0, pass_count / max(1, max_passes // 4))
            T = sa_T0 * (sa_T_end / sa_T0) ** frac
            # Try random candidates with small worsening
            for _ in range(5):
                if time.time() > deadline:
                    break
                idx = np.random.randint(len(candidate_pairs))
                i_local, j_local = candidate_pairs[idx]
                i = ones[i_local]
                j = zeros[j_local]
                delta_c = S[i] - S[j]
                new_c5 = float(np.max(c + delta_c))
                if new_c5 < c5 + T:
                    if np.random.random() < np.exp(-(new_c5 - c5) / max(T, 1e-10)):
                        best_new_c5 = new_c5
                        best_pair = (i, j)
                        best_delta_c = delta_c
                        break
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c = c + best_delta_c
            old_c5 = c5
            c5 = best_new_c5
            
            # Update ones/zeros
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            
            # Rebuild S
            h_pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
            S = h_pad[idx_plus] + h_pad[idx_minus]
            
            no_improve_count = 0
        else:
            if no_improve_count > max_no_improve:
                break
    
    return h, c5

def smooth_max_obj(h, n, alpha, active_idx=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if active_idx is not None and len(active_idx) > 0:
        c_use = c[active_idx]
    else:
        c_use = c
    m = c_use.max()
    shifted = alpha * (c_use - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    val = float(m + np.log(np.sum(np.exp(shifted - mx + mx)))) / alpha
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_refine(h, n, deadline, alpha=50000, maxiter=3000):
    """SLSQP continuous refinement."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
