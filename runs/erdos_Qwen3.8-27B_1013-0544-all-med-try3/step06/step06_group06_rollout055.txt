def fast_binary_optimize(h, n, deadline, max_sweeps=10000):
    """Local search for binary h using bottleneck-guided swaps."""
    c = np.correlate(h, 1 - h, mode='full') + n  # shift to 0-based indexing
    # Actually, np.correlate(h, 1-h, 'full') has length 2n-1
    # Index 0 corresponds to lag -(n-1), index n-1 corresponds to lag 0, etc.
    # Let me just use the raw correlation
    c = np.correlate(h, 1 - h, mode='full')
    best_c = c.max()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        k_star = int(np.argmax(c))
        # k_star is in [0, 2n-2], corresponding to lag k_star - (n-1)
        lag = k_star - (n - 1)
        
        # Try all swaps that might reduce c[k_star]
        # Delta in c[k_star] for swapping i (1→0) and j (0→1):
        # We need to compute the change in c at position k_star
        
        found = False
        for i in ones:
            for j in zeros:
                # Compute delta in c[k_star]
                # c[k_star] = sum_n h[n+lag]*(1-h[n]) over valid n
                # Term where i is n+lag: n = i - lag, valid if 0 <= i-lag < n and 0 <= i < n
                delta = 0
                # i goes 1→0
                # As n+lag = i: contribution was (1-h[i-lag]), now 0. Delta: -(1-h[i-lag])
                ni = i - lag
                if 0 <= ni < n and ni >= max(0, -lag) and ni <= min(n-1, n-1-lag):
                    delta -= (1 - h[ni])
                # As n = i: contribution was h[i+lag]*(1-1)=0, now h[i+lag]*(1-0)=h[i+lag]. Delta: +h[i+lag]
                ni2 = i + lag
                if 0 <= ni2 < n:
                    # Need n=i to be valid: max(0,-lag) <= i <= min(n-1, n-1-lag)
                    if i >= max(0, -lag) and i <= min(n-1, n-1-lag):
                        delta += h[ni2]
                # j goes 0→1
                # As n+lag = j: contribution was 0, now (1-h[j-lag]). Delta: +(1-h[j-lag])
                nj = j - lag
                if 0 <= nj < n and nj >= max(0, -lag) and nj <= min(n-1, n-1-lag):
                    delta += (1 - h[nj])
                # As n = j: contribution was h[j+lag]*1, now 0. Delta: -h[j+lag]
                nj2 = j + lag
                if 0 <= nj2 < n:
                    if j >= max(0, -lag) and j <= min(n-1, n-1-lag):
                        delta -= h[nj2]
                
                if delta < 0:  # c[k_star] would decrease
                    # Full evaluation
                    h[i], h[j] = 0, 1
                    c_new = np.correlate(h, 1 - h, mode='full')
                    c_max_new = c_new.max()
                    if c_max_new < best_c - 1e-15:
                        c = c_new
                        best_c = c_max_new
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                        improved = True
                        found = True
                        break
                    else:
                        h[i], h[j] = 1, 0
            if found:
                break
        
        if not improved:
            # Try all swaps (not just bottleneck-targeting)
            # Or break
            break
    
    return h, best_c * (2.0/n)

def anneal_binary(h, n, deadline, T_start=1.0, T_end=0.01, n_iter=None):
    """Simulated annealing for binary h."""
    if n_iter is None:
        n_iter = 10_000_000  # will be limited by deadline
    
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    best_c = c.max()
    best_h = h.copy()
    cur_c = best_c
    
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooling
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = 0, 1
        c_new = np.correlate(h, 1 - h, mode='full')
        new_max = c_new.max()
        
        delta = new_max - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_max
            c = c_new
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    
    return best_h, best_c * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def anneal_binary(h, n, deadline, T_start=0.5, T_end=0.01, rng=None):
    """Simulated annealing for binary h."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cur_c = c.max()
    best_c = cur_c
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    n_iter = max(1, int((deadline - time.time()) * 200000))  # ~200k iters/sec
    cooling = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    T = T_start
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooling
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0, 1
        c_new = np.correlate(h, 1 - h, mode='full')
        new_max = c_new.max()
        
        delta = new_max - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_max
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i], h[j] = 1, 0
    
    return best_h, best_c * dx

def slsqp_optimize(h, n, deadline, alphas=(100, 1000, 10000, 100000)):
    """SLSQP optimization with soft-max approximation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        res = minimize(obj, best_h, method='SLSQP',
                      bounds=[(0, 1)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2}],
                      options={'maxiter': 2000, 'ftol': 1e-14})
        
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def anneal_binary(h, n, deadline, T_start=0.5, T_end=0.005, rng=None, restart_patience=5000):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    cur_c = c.max()
    best_c = cur_c
    best_h = h.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_iter_est = max(1000, int((deadline - time.time()) * 80000))
    cooling = (T_end / T_start) ** (1.0 / n_iter_est)
    T = T_start
    stall = 0
    for it in range(n_iter_est * 3):
        if time.time() > deadline:
            break
        T *= cooling
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i], h[j] = 0.0, 1.0
        c_new = np.correlate(h, 1 - h, mode='full')
        new_max = c_new.max()
        delta = new_max - cur_c
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c = new_max
            if new_max < best_c - 1e-15:
                best_c = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            ii = ones_list.index(i)
            jj = zeros_list.index(j)
            ones_list[ii] = j
            zeros_list[jj] = i
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > restart_patience and T > T_end * 5:
            T = max(T_end, T * 3.0)
            stall = 0
            for _ in range(10):
                ii = rng.randint(len(ones_list))
                jj = rng.randint(len(zeros_list))
                h[ones_list[ii]], h[zeros_list[jj]] = 0.0, 1.0
                c_new = np.correlate(h, 1 - h, mode='full')
                new_max = c_new.max()
                if new_max < best_c:
                    cur_c = new_max
                else:
                    h[ones_list[ii]], h[zeros_list[jj]] = 1.0, 0.0
                ones_list[ii], zeros_list[jj] = zeros_list[jj], ones_list[ii]
    return best_h, best_c * dx

def greedy_bottleneck(h, n, deadline, max_rounds=2000):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    best_c = c.max()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        improved = False
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in ones_list:
            d_i = 0.0
            ni = i - lag
            if lag >= 0:
                if 0 <= ni <= n - 1 - lag and 0 <= i <= n - 1:
                    d_i -= (1.0 - h[ni])
                if 0 <= i <= n - 1 - lag and 0 <= i + lag < n:
                    d_i += h[i + lag]
            else:
                m = -lag
                if 0 <= ni <= n - 1 and m <= i <= n - 1:
                    d_i -= (1.0 - h[ni])
                if m <= i <= n - 1 and 0 <= i + lag < n:
                    d_i += h[i + lag]
            for j in zeros_list:
                d_j = 0.0
                nj = j - lag
                if lag >= 0:
                    if 0 <= nj <= n - 1 - lag and 0 <= j <= n - 1:
                        d_j += (1.0 - h[nj])
                    if 0 <= j <= n - 1 - lag and 0 <= j + lag < n:
                        d_j -= h[j + lag]
                else:
                    m = -lag
                    if 0 <= nj <= n - 1 and m <= j <= n - 1:
                        d_j += (1.0 - h[nj])
                    if m <= j <= n - 1 and 0 <= j + lag < n:
                        d_j -= h[j + lag]
                gain = -(d_i + d_j)
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        if best_i >= 0 and best_j >= 0:
            h[best_i], h[best_j] = 0.0, 1.0

                zeros_list[jj] = best_i
                improved = True

                h[best_i], h[best_j] = 1.0, 0.0

def slsqp_polish(h, n, deadline, alpha_list):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alpha_list:
        if time.time() > deadline - 20:
            break
        def obj_fn(x):
            g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        def con_fun(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(obj_fn, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con_fun}],
                          options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        m = g.max()
        k_star = int(np.argmax(g))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(lag, n):
                grad[i] += (1.0 - h[i - lag])
            for i in range(0, n - lag):
                grad[i] -= h[i + lag]
        else:
            m2 = -lag
            for i in range(0, n - m2):
                grad[i] += (1.0 - h[i + m2])
            for i in range(m2, n):
                grad[i] -= h[i - m2]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c
