import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2} via bisection on uniform shift."""
    target = n / 2.0
    # Edge case
    h_clipped = np.clip(h, 0.0, 1.0)
    if h_clipped.sum() == target:
        return h_clipped
    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any tiny numerical errors
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining diff to non-boundary entries
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_step(h, n, alpha, step):
    """One projected subgradient step."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')  # no dx factor
    shifted = alpha * (c_raw - c_raw.max())
    max_sh = shifted.max()
    weights = np.exp(shifted - max_sh)
    weights /= weights.sum()
    
    one_minus_h = 1.0 - h
    g = np.zeros(n)
    n1 = n - 1
    for s_idx in range(2 * n - 1):
        w = weights[s_idx]
        if w < 1e-13:
            continue
        s = s_idx - n1
        if s >= 0:
            g[:n - s] += w * one_minus_h[s:]
            if s > 0:
                g[s:] -= w * h[:n - s]
        else:
            t = -s
            g[t:] += w * one_minus_h[:n - t]
            g[:n - t] -= w * h[t:]
    
    h_new = h - step * dx * g
    return project_feasible(h_new, n)

def proj_subgrad(h, n, iters, alpha):
    """Projected subgradient with Polyak-style step."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

        h_new = subgrad_step(h, n, alpha, step)
        c = compute_c5(h_new, n)

            best_h = h_new.copy()

    best_c = compute_c5(h, n)
