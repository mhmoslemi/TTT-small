import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:

        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value

        return h_curr.copy()
    return project_feasible(h_new, n)

def gradient_local_search(h, n, deadline):
    """Gradient-based local search using the max subgradient."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for ss in step_sizes:
        if time.time() > deadline - 2:
            break
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            c = compute_all_lags(h, n)
            c_max = c.max()
            # Find active lags (within 1% of max)
            active_mask = c > c_max * 0.995
            active_idx = np.where(active_mask)[0]
            
            # Compute composite gradient
            grad = np.zeros(n)
            for k_idx in active_idx:
                lag = k_idx - (n - 1)
                grad += _compute_lag_gradient(h, n, lag)
            grad *= dx / len(active_idx)
            
            # Project gradient: subtract mean to maintain sum constraint
            grad -= np.mean(grad)
            
            # Clip to feasible direction
            for i in range(n):
                if grad[i] < 0 and h[i] <= 1e-8:
                    grad[i] = 0
                if grad[i] > 0 and h[i] >= 1 - 1e-8:
                    grad[i] = 0
            
            # Take negative gradient step
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            direction = -grad / gnorm
            
            # Line search
            best_alpha = 0
            best_c_try = best_c
            for alpha in [ss, ss * 0.5, ss * 0.25, ss * 0.1]:
                h_try = h + alpha * direction
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c_try - 1e-15:
                    best_c_try = c_try
                    best_alpha = alpha
            
            if best_alpha > 0:
                h = project_feasible(h + best_alpha * direction, n)
                best_c = compute_c5(h, n)
                improved = True
    
    return h, best_c

        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(150, remaining * 0.35)
        
        # Choose initial guess
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase C: Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        n_restarts = 0
        while time.time() < deadline - 50:
            if n_restarts >= 5:
                break
            scale = 0.008 * (1.0 + 0.6 * n_restarts)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 40) * 0.4)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            n_restarts += 1
    
    # Phase D: Final aggressive polish
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        
        # Ultra-high alpha SLSQP
        h_new = slsqp_optimize(best_h, n, 2000000, maxiter=10000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # Gradient local search
        if time.time() < deadline - 5:
            h_gls, c_gls = gradient_local_search(best_h, n, deadline - 3)
            if c_gls < best_c:
                best_c = c_gls
                best_h = h_gls
        
        # Pairwise local search
        if time.time() < deadline - 2:
            h_pls, c_pls = pairwise_local_search(best_h, n, deadline - 1)
            if c_pls < best_c:
                best_c = c_pls
                best_h = h_pls
        
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
