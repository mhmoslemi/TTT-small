```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_unscaled(h, n):
    """Correlation without dx scaling"""
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h *= target / s
        else:
            room = 1.0 - h
            room_sum = room.sum()
            if room_sum < 1e-14:
                break
            h += (target - s) / room_sum * room
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, alpha):
    c = compute_c5_unscaled(h, n) * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    c = compute_c5_unscaled(h, n) * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, max_iters_per_phase=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (500, 10.0), (2000, 20.0), (10000, 50.0),
        (50000, 200.0), (200000, 1000.0), (1000000, 5000.0),
        (5000000, 10000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def obj(x):
            val = smooth_max_obj(x, n, alpha)
            val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad(x):
            g = smooth_max_grad(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters_per_phase, 'gtol': 1e-12,
                                   'maxfun': max_iters_per_phase * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_unscaled(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h *= target / s
        else:
            room = 1.0 - h
            room_sum = room.sum()
            if room_sum < 1e-14:
                break
            h += (target - s) / room_sum * room
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, n, alpha):
    c = compute_c5_unscaled(h, n) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad_fn(h, n, alpha):
    c = compute_c5_unscaled(h, n) * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad_fn(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (500, 5.0, 200),
        (2000, 10.0, 200),
        (10000, 20.0, 300),
        (50000, 50.0, 300),
        (200000, 100.0, 500),
        (1000000, 500.0, 500),
        (5000000, 2000.0, 300),
    ]
    for alpha, mu, maxit in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'gtol': 1e-11,
                                   'maxfun': maxit * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
```
