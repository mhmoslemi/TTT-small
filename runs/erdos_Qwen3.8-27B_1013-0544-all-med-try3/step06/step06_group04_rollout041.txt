def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-14:
                break
            h = h + diff * room / tot
        else:
            tot = float(h.sum())
            if tot < 1e-14:
                break
            h = h + diff * h / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size > 0:
            # distribute over many small? just uniform
            step = diff / idx.size
            h[idx] = h[idx] + step
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - float(h.sum())
            if abs(diff2) > 1e-14 and idx.size > 0:
                # place residual into one with room
                if diff2 > 0:
                    k = int(np.argmax(1.0 - h))
                    h[k] = min(1.0, h[k] + diff2)

                    k = int(np.argmax(h))

def subset_smooth_grad(x,n,alpha,idx):
    idx = np.asarray(idx, dtype=int)
    c = compute_all_lags(x,n)
    sub = c[idx]
    m = float(np.max(sub))
    v = alpha*(sub - m)
    mx = float(np.max(v))
    z = np.exp(v-mx)
    w = z / float(np.sum(z))
    g = np.zeros(n, dtype=float)
    for ii, wi in zip(idx, w):
        if wi <= 1e-12: continue
        r = int(ii) - (n - 1)

            g[:n-mabs] += a * (1.0 - x[mabs:])

import time
import math
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-15:
                break
            h = h + diff * room / tot
        else:
            tot = float(h.sum())
            if tot < 1e-15:
                break
            h = h + diff * h / tot
        h = np.clip(h, 0.0, 1.0)

    for _ in range(20):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        h[idx] = h[idx] + diff / float(idx.size)
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                if diff <= 0.0:
                    break
                add = min(diff, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
                    diff -= add
        else:
            idx = np.argsort(h)
            for i in idx:
                if diff >= 0.0:
                    break
                add = max(diff, -h[i])
                if add < 0.0:
                    h[i] += add
                    diff -= add

    return np.clip(h, 0.0, 1.0)


def compute_all_lags(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    c = compute_all_lags(h, n)
    return float(np.max(c))


def sum_con_fun(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_con_jac(x, n):
    return np.ones(n, dtype=float)


def logsumexp_scaled(c, alpha):
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    v = float(alpha) * (c - m)
    mx = float(np.max(v))
    if not np.isfinite(mx):
        return m
    z = np.exp(v - mx)
    s = float(np.sum(z))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + (mx + math.log(s)) / float(alpha))


def subset_smooth_obj(x, n, alpha, idx):
    c = compute_all_lags(x, n)
    idx = np.asarray(idx, dtype=int)
    if idx.size == 0:
        return 0.0
    return logsumexp_scaled(c[idx], alpha)


def subset_smooth_grad(x, n, alpha, idx):
    idx = np.asarray(idx, dtype=int)
    c = compute_all_lags(x, n)
    sub = c[idx]
    m = float(np.max(sub))
    v = float(alpha) * (sub - m)
    mx = float(np.max(v))
    z = np.exp(v - mx)
    s = float(np.sum(z))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(idx.size, dtype=float)
        if idx.size > 0:
            w[0] = 1.0
    else:
        w = z / s

    g = np.zeros(n, dtype=float)
    for ii, wi in zip(idx, w):
        if wi <= 1e-12:
            continue
        r = int(ii) - (n - 1)
        mabs = abs(r)
        a = float(wi)
        if r == 0:
            g += a * (1.0 - 2.0 * x)
        elif r > 0:
            g[:n - mabs] -= a * x[mabs:]
            g[mabs:] += a * (1.0 - x[:n - mabs])
        else:
            g[:n - mabs] += a * (1.0 - x[mabs:])
            g[mabs:] -= a * x[:n - mabs]
    return g


def choose_active(c, m, max_count=35, tol=0.02):
    order = np.argsort(-c)
    selected = []
    maxc = float(np.max(c))
    top_count = max(1, int(m))
    for ii in order[:top_count]:
        ii = int(ii)
        if ii not in selected:
            selected.append(ii)
    for ii in order:
        if len(selected) >= max_count:
            break
        ii = int(ii)
        if ii in selected:
            continue
        if c[ii] >= maxc - float(tol):
            selected.append(ii)
        elif len(selected) >= top_count and c[ii] < maxc - 0.10:
            break
    if len(selected) == 0:
        selected = [int(order[0])]
    return np.asarray(selected, dtype=int)


def lag_gradient_unit(h, n, idx):
    r = int(idx) - (n - 1)
    mabs = abs(r)
    g = np.zeros(n, dtype=float)
    if r == 0:
        g[:] = 1.0 - 2.0 * h
    elif r > 0:
        g[:n - mabs] -= h[mabs:]
        g[mabs:] += 1.0 - h[:n - mabs]
    else:
        g[:n - mabs] += 1.0 - h[mabs:]
        g[mabs:] -= h[:n - mabs]
    return g


def unique_index_array(idx, length):
    seen = np.zeros(length, dtype=bool)
    out = []
    for ii in idx:
        ii = int(ii)
        if 0 <= ii < length and not seen[ii]:
            seen[ii] = True
            out.append(ii)
    return np.asarray(out, dtype=int)


def slsqp_subset(h, n, idx, alpha, maxiter):
    h = project_feasible(h, n)
    idx = unique_index_array(idx, 2 * n - 1)
    if idx.size == 0:
        return h.copy()

    constraints = [
        {
            "type": "eq",
            "fun": sum_con_fun,
            "jac": sum_con_jac,
            "args": (n,)
        }
    ]
    bounds = [(0.0, 1.0)] * int(n)
    try:
        res = minimize(
            subset_smooth_obj,
            h,
            args=(n, float(alpha), idx),
            jac=subset_smooth_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": 0}
        )
        x = res.x
    except Exception:
        return h.copy()
    return project_feasible(x, n)


def eval_budget(n, factor=1.0):
    base = 220000000.0 * float(factor)
    val = base / max(1.0, float(n) * float(n))
    if val < 300:
        val = 300
    if val > 8000:
        val = 8000
    return int(val)


def local_pair_polish(h, n, deadline, active_lags=12, top_pairs=5,
                      max_evals=6000, min_diff=1e-5):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    evals = 0
    min_diff_cur = max(1e-7, float(min_diff))
    stuck = 0
    center = n - 1

    while time.time() < deadline and evals < max_evals and stuck < 3:
        c = compute_all_lags(h, n)
        cur = float(np.max(c))
        if cur < best_c - 1e-13:
            best_c = cur

        order = np.argsort(-c)
        lags = []
        for ii in order:
            ii = int(ii)
            if len(lags) >= active_lags:
                break
            r = ii - center
            if r == 0:
                continue
            if c[ii] >= best_c - 1e-6:
                lags.append(ii)

        if not lags:
            for ii in order[:active_lags * 2]:
                ii = int(ii)
                if ii != center:
                    lags.append(ii)
                if len(lags) >= active_lags:
                    break

        found = False
        for ii in lags:
            ii = int(ii)
            r = ii - center
            mabs = abs(r)
            hi = h[:n - mabs]
            hj = h[mabs:]

            if r > 0:
                diff = hj - hi
            else:
                diff = hi - hj

            cand = np.where(diff > min_diff_cur)[0]
            if cand.size == 0:
                continue

            if cand.size > top_pairs:
                cand = cand[np.argsort(-diff[cand])[:top_pairs]]

            for i in cand:
                i = int(i)
                if r > 0:
                    dmax = min(float(diff[i]), float(hj[i]), 1.0 - float(hi[i]))
                else:
                    dmax = min(float(diff[i]), float(hi[i]), 1.0 - float(hj[i]))
                if dmax <= 1e-9:
                    continue

                for frac in (0.9, 0.55, 0.30, 0.15):
                    if evals >= max_evals or time.time() >= deadline:
                        break
                    d = frac * dmax
                    if d < 1e-9:
                        continue

                    if r > 0:
                        h[i] += d
                        h[i + mabs] -= d
                    else:
                        h[i] -= d
                        h[i + mabs] += d

                    evals += 1
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        found = True
                        break
                    else:
                        if r > 0:
                            h[i] -= d
                            h[i + mabs] += d
                        else:
                            h[i] += d
                            h[i + mabs] -= d

                if found:
                    break
            if found:
                break

        if found:
            stuck = 0
        else:
            stuck += 1
            min_diff_cur *= 0.6
            if min_diff_cur < 1e-8:
                break

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def gd_polish(h, n, deadline, max_iters=800, top=12, alphas=(500, 2000), base=0.03):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    step = float(base)
    center = n - 1

    for alpha in alphas:
        for _ in range(int(max_iters)):
            if time.time() >= deadline:
                break
            c = compute_all_lags(h, n)
            cur = float(np.max(c))
            if cur < best_c - 1e-14:
                best_c = cur

            idx = np.argsort(-c)[:top]
            g = subset_smooth_grad(h, n, float(alpha), idx)
            g -= float(np.mean(g))
            gn = float(np.max(np.abs(g)))
            if gn < 1e-10:
                break

            direction = -g / gn
            found = False
            for t in (step, 2.0 * step, 0.5 * step, 4.0 * step, 8.0 * step):
                if t <= 1e-8:
                    continue
                cand = project_feasible(h + t * direction, n)
                cc = compute_c5(cand, n)
                if cc < best_c - 1e-14:
                    h = cand
                    best_c = cc
                    found = True
                    step = min(0.25, step * 1.05)
                    break

            if not found:
                step *= 0.7
                if step < 1e-5:
                    break

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def sorted_targets(targets, center):
    pairs = []
    for t in targets:
        pairs.append((abs(int(t) - int(center)), int(t)))
    pairs.sort()
    out = []
    for p in pairs:
        out.append(p[1])
    return out


def resample_to(h, n_new):
    n_old = int(len(h))
    if n_old == int(n_new):
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, int(n_new))
    return project_feasible(np.interp(x_new, x_old, h), int(n_new))


def combo_line_search(h, h1, n):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    fs = (0.1, 0.2, 0.25, 0.35, 0.5, 0.6, 0.75, 1.0, 0.4, 0.7, 0.8)
    for s in fs:
        if time.time() > 1e100:
            break
        cand = (1.0 - s) * h + s * h1
        cand = project_feasible(cand, n)
        cc = compute_c5(cand, n)
        if cc < best_c - 1e-14:
            best_c = cc
            best_h = cand
    return best_h, best_c


def trust_lp_step(h, n, trust, active_m=25):
    X = cp.Variable(n)
    t = cp.Variable()
    target = 0.5 * float(n)

    cons = [
        cp.sum(X) == target,
        X >= 0.0,
        X <= 1.0,
        X <= h + trust,
        X >= h - trust
    ]

    c = compute_all_lags(h, n)
    idx = choose_active(c, active_m, max(12, active_m + 8), 0.02)
    for ii in idx:
        ii = int(ii)
        g = lag_gradient_unit(h, n, ii)
        cons.append(float(c[ii]) + cp.sum(g * (X - h)) <= t)

    prob = cp.Problem(cp.Minimize(t), cons)

    solvers = []
    for name in ("GLOP", "ECOS", "SCS", "CBC"):
        s = getattr(cp, name, None)
        if s is not None:
            solvers.append(s)

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ("optimal", "optimal_inaccurate"):
                if X.value is not None:
                    return project_feasible(X.value, n)
        except Exception:
            pass
    return None


def trust_lp_polish(h, n, deadline, trust=0.20, active_m=25):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    tr = float(trust)

    for _ in range(6):
        if time.time() > deadline - 5:
            break
        if tr < 0.01:
            break
        h1 = trust_lp_step(h, n, tr, active_m)
        if h1 is None:
            break
        step = float(np.max(np.abs(h1 - h)))
        if step < 1e-7:
            break
        h_new, c_new = combo_line_search(h, h1, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = project_feasible(h_new, n)
            tr *= 0.75
        else:
            tr *= 0.5

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def slsqp_refine(h, n, end_time):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if n > 600:
        return best_h, best_c

    current = best_h.copy()
    specs = [
        (8, 12, 0.0, 5000, 420),
        (18, 24, 0.02, 1200, 380),
        (28, 34, 0.04, 300, 320),
        (40, 40, 0.06, 100, 280)
    ]
    if n > 420:
        specs = specs[:3]

    for m, max_count, tol, alpha, base_iter in specs:
        if time.time() > end_time - 35:
            break
        c = compute_all_lags(current, n)
        idx = choose_active(c, m, max_count, tol)
        if idx.size == 0:
            break
        rem = end_time - time.time()
        maxiter = int(min(base_iter, max(80, int(rem * 4))))
        if maxiter < 50:
            break

        h_new = slsqp_subset(current, n, idx, alpha, maxiter)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = project_feasible(h_new, n)
            current = best_h.copy()
        elif c_new < best_c + 0.0015:
            current = project_feasible(h_new, n)
        else:
            current = best_h.copy()

    return best_h, best_c


def polish_candidate(h, n, end_time):
    n = int(n)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)

    e1 = eval_budget(n, 1.0)
    e2 = eval_budget(n, 2.0)

    if time.time() < end_time - 20:
        local_end = min(end_time - 15.0, time.time() + 55.0)
        if local_end > time.time() + 5.0:
            h1, c1 = local_pair_polish(
                h, n, local_end,
                active_lags=12, top_pairs=5,
                max_evals=e1, min_diff=1e-5
            )
            if c1 < best_c - 1e-14:
                best_c = c1
                h = h1

    if time.time() < end_time - 70:
        h2, c2 = slsqp_refine(h, n, end_time)
        if c2 < best_c - 1e-14:
            best_c = c2
            h = h2

    if n <= 460 and time.time() < end_time - 55:
        h3, c3 = trust_lp_polish(
            h, n,
            min(end_time - 20.0, time.time() + 60.0),
            trust=0.20, active_m=20
        )
        if c3 < best_c - 1e-14:
            best_c = c3
            h = h3

    if time.time() < end_time - 8:
        h4, c4 = local_pair_polish(
            h, n, end_time,
            active_lags=16, top_pairs=6,
            max_evals=e2, min_diff=2e-6
        )
        if c4 < best_c - 1e-14:
            best_c = c4
            h = h4

    if time.time() < end_time - 15:
        h5, c5 = gd_polish(
            h, n, end_time - 2.0,
            max_iters=250, top=12,
            alphas=(800, 2500), base=0.02
        )
        if c5 < best_c - 1e-14:
            best_c = c5
            h = h5

    h = project_feasible(h, n)
    return h, compute_c5(h, n)


def default_start(n, rng):
    x = np.linspace(0.0, 2.0, int(n))
    h = 0.5 + 0.5 * np.sin(math.pi * x)
    h = h + 0.10 * np.sin(2.0 * math.pi * x + 0.7)
    h = h + 0.05 * np.sin(5.0 * math.pi * x - 1.3)
    h = h + 0.04 * rng.normal(0.0, 1.0, int(n))
    return project_feasible(h, int(n))


def run_restarts(best_h, best_c, rng, deadline):
    n = int(len(best_h))
    h_best = best_h.copy()
    c_best = float(best_c)
    it = 0

    while time.time() < deadline - 40:
        it += 1
        mode = it % 4
        scale = 0.02 + 0.008 * min(it, 20)

        if mode == 0:
            h_start = h_best + rng.normal(0.0, scale, n)
        elif mode == 1:
            f = int(rng.randint(1, 5))
            amp = float(rng.uniform(0.02, 0.10))
            phase = float(rng.uniform(0.0, 2.0 * math.pi))
            x = np.linspace(0.0, 2.0, n)
            h_start = h_best + amp * np.sin(2.0 * math.pi * f * x + phase)
            h_start = h_start + rng.normal(0.0, 0.01, n)
        elif mode == 2:
            z = h_best + rng.normal(0.0, 0.03, n)
            q = np.clip((z - 0.5) / 0.07, -1.0, 1.0) * 0.5 + 0.5
            h_start = q + 0.02 * rng.normal(0.0, 1.0, n)
        else:
            shiftc = float(rng.uniform(-0.05, 0.05))
            gain = float(rng.uniform(0.7, 1.4))
            h_start = (h_best + shiftc) * gain + shiftc

        h_start = project_feasible(h_start, n)
        rem = deadline - time.time()
        if rem < 50:
            break
        alloc = min(45.0, 0.18 * rem)
        if alloc < 12:
            break

        h_q, c_q = local_pair_polish(
            h_start, n, time.time() + alloc,
            active_lags=10, top_pairs=4,
            max_evals=eval_budget(n, 1.0), min_diff=1e-5
        )

        improved = False
        if c_q < c_best - 1e-7:
            if time.time() < deadline - 80:
                deep_end = time.time() + min(120.0, 0.35 * (deadline - time.time()))
                h_r, c_r = polish_candidate(h_q, n, deep_end)
                if c_r < c_best - 1e-14:
                    h_best = h_r.copy()
                    c_best = c_r
                    improved = True
            if not improved:
                h_best = h_q.copy()
                c_best = c_q
                improved = True
        elif c_q < c_best + 0.0015 and time.time() < deadline - 100 and (it % 6 == 0):
            deep_end = time.time() + min(90.0, 0.25 * (deadline - time.time()))
            h_r, c_r = polish_candidate(h_q, n, deep_end)
            if c_r < c_best - 1e-14:
                h_best = h_r.copy()
                c_best = c_r
                improved = True

        if improved:
            it = 0

    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(30.0, budget_s - 12.0)
    rng = np.random.RandomState(int(seed))

    h_init = None
    n_init = 300
    try:
        if "initial_h_values" in globals():
            h_raw = np.asarray(initial_h_values, dtype=float)
        else:
            raise Exception("no initial")
        if h_raw.ndim != 1:
            raise Exception("bad shape")
        n_init = int(h_raw.size)
        if n_init < 20 or not np.all(np.isfinite(h_raw)):
            raise Exception("bad values")
        h_init = project_feasible(h_raw, n_init)
    except Exception:
        h_init = None
        n_init = 300

    if h_init is None:
        n_init = 300
        h_init = default_start(n_init, rng)

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = int(n_init)
    tried_ns = set()
    tried_ns.add(best_n)

    initial_alloc = min(240.0, max(60.0, 0.35 * (deadline - t0)))
    h_opt, c_opt = polish_candidate(h_init, n_init, t0 + initial_alloc)
    if c_opt < best_c - 1e-14:
        best_c = c_opt
        best_h = h_opt
        best_n = int(n_init)

    target_list = [224, 256, 320, 384, 448, 512, 576, 640]
    candidate_ns = [n_init]
    for nt in sorted_targets(target_list, n_init):
        if len(candidate_ns) >= 6:
            break
        nt = int(nt)
        if nt < 128 or nt > 640:
            continue
        if nt % 2 != 0:
            nt += 1
        if nt not in candidate_ns:
            candidate_ns.append(nt)

    for n_try in candidate_ns:
        n_try = int(n_try)
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        if time.time() > deadline - 90:
            break
        remaining = deadline - time.time() - 90
        if remaining < 45:
            break

        alloc = min(200.0, 0.35 * remaining)
        alloc = max(45.0, alloc * (384.0 / float(n_try)))
        alloc = min(220.0, alloc)
        if alloc < 30:
            break

        if best_h is not None:
            h_try = resample_to(best_h, n_try)
        else:
            h_try = default_start(n_try, rng)
        h_try = project_feasible(h_try, n_try)

        h_opt, c_opt = polish_candidate(h_try, n_try, time.time() + alloc)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
            best_n = n_try

    if best_h is not None and time.time() < deadline - 80:
        best_h, best_c = run_restarts(best_h, best_c, rng, deadline - 30.0)
        best_n = int(len(best_h))

    if best_h is not None and time.time() < deadline - 15:
        n = int(len(best_h))
        final_end = deadline - 2.0
        h1, c1 = local_pair_polish(
            best_h, n, final_end,
            active_lags=18, top_pairs=6,
            max_evals=eval_budget(n, 2.0), min_diff=1e-6
        )
        if c1 < best_c - 1e-14:
            best_c = c1
            best_h = h1
            best_n = n

        if n <= 560 and time.time() < final_end - 35:
            h2, c2 = slsqp_refine(best_h, n, final_end - 2.0)
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h2
                best_n = int(len(h2))

        if time.time() < final_end - 5:
            h3, c3 = local_pair_polish(
                best_h, int(len(best_h)), final_end,
                active_lags=18, top_pairs=6,
                max_evals=eval_budget(int(len(best_h)), 1.5), min_diff=1e-7
            )
            if c3 < best_c - 1e-14:
                best_c = c3
                best_h = h3
                best_n = int(len(h3))

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_n = 300

    best_h = project_feasible(best_h, int(len(best_h)))
    best_c = compute_c5(best_h, int(len(best_h)))
    return best_h, float(best_c), int(len(best_h))
