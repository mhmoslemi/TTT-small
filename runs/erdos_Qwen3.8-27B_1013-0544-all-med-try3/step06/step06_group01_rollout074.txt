```python
def greedy_binary_optimize(h, n, max_iters=1000):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max() * dx
    
    for iteration in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute rem[i] for all i in ones
        # rem[i][k] = -(1-h[i+k]) * [0<=i+k<n] + h[i-k] * [0<=i-k<n]
        # k ranges from -(n-1) to n-1, index = k + (n-1)
        
        # For efficiency, vectorize
        # i is shape (n_ones,), k is shape (2n-1,)
        # i+k is shape (n_ones, 2n-1)
        
        k_vals = np.arange(-(n-1), n)  # shape (2n-1,)
        
        # rem: for each one position i, the effect of setting h[i]=0
        # rem[i_idx, k_idx] = -(1-h[i+k]) * [0<=i+k<n] + h[i-k] * [0<=i-k<n]
        
        i_plus_k = ones[:, None] + k_vals[None, :]  # (n_ones, 2n-1)
        i_minus_k = ones[:, None] - k_vals[None, :]  # (n_ones, 2n-1)
        
        rem = np.zeros((len(ones), len(k_vals)))
        mask_plus = (i_plus_k >= 0) & (i_plus_k < n)
        mask_minus = (i_minus_k >= 0) & (i_minus_k < n)
        
        # For valid i+k: -(1-h[i+k])
        rem[mask_plus] += -(1 - h[i_plus_k[mask_plus]])
        # For valid i-k: +h[i-k]
        rem[mask_minus] += h[i_minus_k[mask_minus]]
        
        # add: for each zero position j, the effect of setting h[j]=1
        # add[j_idx, k_idx] = (1-h[j+k]) * [0<=j+k<n] - h[j-k] * [0<=j-k<n]
        
        j_plus_k = zeros[:, None] + k_vals[None, :]
        j_minus_k = zeros[:, None] - k_vals[None, :]
        
        add = np.zeros((len(zeros), len(k_vals)))
        mask_jplus = (j_plus_k >= 0) & (j_plus_k < n)
        mask_jminus = (j_minus_k >= 0) & (j_minus_k < n)
        
        add[mask_jplus] += (1 - h[j_plus_k[mask_jplus]])
        add[mask_jminus] += -h[j_minus_k[mask_jminus]]
        
        # Find best swap
        best_score = np.inf
        best_i = None
        best_j = None
        
        # For each j in zeros:
        for j_idx in range(len(zeros)):
            base = corr + add[j_idx]
            # For each i in ones: max(base + rem[i])
            scores = np.max(base[None, :] + rem, axis=1)
            i_best = np.argmin(scores)
            if scores[i_best] * dx < best_score:
                best_score = scores[i_best] * dx
                best_i = ones[i_best]
                best_j = zeros[j_idx]
        
        if best_score >= cur_max - 1e-15:
            break  # No improvement
        
        # Apply swap
        h[best_i] = 0
        h[best_j] = 1
        corr = np.correlate(h, 1-h, mode='full').astype(float)
        cur_max = corr.max() * dx
    
    return h, cur_max
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
    return float(np.max(c) * dx)

def make_binary(h, n):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    h = np.round(h)
    # Count ones
    n_ones = int(h.sum())
    if n_ones > target:
        # Need to remove some ones
        ones = np.where(h == 1)[0]
        # Remove the "least important" ones (those with lowest contribution)
        # For now, just remove from the end
        to_remove = ones[n_ones - target:]
        h[to_remove] = 0
    elif n_ones < target:
        # Need to add some ones
        zeros = np.where(h == 0)[0]
        to_add = zeros[:target - n_ones]
        h[to_add] = 1
    return h

def greedy_binary_search(h, n, deadline, max_iters=500):
    """Vectorized greedy binary local search."""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
    cur_max = corr.max()
    k_vals = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    
    for iteration in range(max_iters):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Compute rem matrix: effect of setting h[ones[i]]=0
        i_plus_k = ones[:, None] + k_vals[None, :]  # (n_ones, n_corr)
        i_minus_k = ones[:, None] - k_vals[None, :]
        
        rem = np.zeros((n_ones, n_corr))
        mask_p = (i_plus_k >= 0) & (i_plus_k < n)
        mask_m = (i_minus_k >= 0) & (i_minus_k < n)
        rem[mask_p] -= (1.0 - h[i_plus_k[mask_p]])
        rem[mask_m] += h[i_minus_k[mask_m]]
        
        # Compute add matrix: effect of setting h[zeros[j]]=1
        j_plus_k = zeros[:, None] + k_vals[None, :]
        j_minus_k = zeros[:, None] - k_vals[None, :]
        
        add = np.zeros((n_zeros, n_corr))
        mask_jp = (j_plus_k >= 0) & (j_plus_k < n)
        mask_jm = (j_minus_k >= 0) & (j_minus_k < n)
        add[mask_jp] += (1.0 - h[j_plus_k[mask_jp]])
        add[mask_jm] -= h[j_minus_k[mask_jm]]
        
        # Find best swap: minimize max_k(corr + rem[i] + add[j])
        # Vectorized computation
        # combined[i, j, k] = corr[k] + rem[i, k] + add[j, k]
        # We want argmin_{i,j} max_k combined[i,j,k]
        
        # To save memory, compute in chunks
        # For n=250: n_ones * n_zeros * n_corr = 125 * 125 * 499 ≈ 7.8M (62MB) - OK
        # For n=500: 250 * 250 * 999 ≈ 62.4M (500MB) - too much!
        
        if n_ones * n_zeros * n_corr < 50_000_000:
            # Full vectorized computation
            combined = corr[None, None, :] + rem[:, None, :] + add[None, :, :]
            scores = np.max(combined, axis=2)
            del combined
        else:
            # Chunked computation over zeros
            scores = np.zeros((n_ones, n_zeros))
            chunk_size = max(1, 50_000_000 // (n_ones * n_corr))
            for j_start in range(0, n_zeros, chunk_size):
                j_end = min(j_start + chunk_size, n_zeros)
                combined = corr[None, None, :] + rem[:, None, :] + add[None, j_start:j_end, :]
                scores[:, j_start:j_end] = np.max(combined, axis=2)
                del combined
        
        flat_idx = np.argmin(scores)
        i_idx, j_idx = np.unravel_index(fat_idx, scores.shape)
        
        if scores[i_idx, j_idx] >= cur_max - 1e-10:
            break
        
        # Apply swap and verify
        h[ones[i_idx]] = 0
        h[zeros[j_idx]] = 1
        corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
        new_max = corr.max()
        
        if new_max >= cur_max:
            h[ones[i_idx]] = 1
            h[zeros[j_idx]] = 0
            corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
            break
        
        cur_max = new_max
    
    return h, cur_max * dx

def simulated_annealing(h, n, deadline, T_start=0.05, T_end=0.001, steps=50000):
    """Simulated annealing on binary h."""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
    cur_val = corr.max() * dx
    best_h = h.copy()
    best_val = cur_val
    
    rng = np.random.RandomState(123)
    n_steps = steps
    k_vals = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (step / max(n_steps, 1))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        pos_i = ones[i]
        pos_j = zeros[j]
        
        # Compute change
        h[pos_i] = 0
        h[pos_j] = 1
        new_corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
        new_val = new_corr.max() * dx
        
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            corr = new_corr
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # Undo
            h[pos_i] = 1
            h[pos_j] = 0
    
    return best_h, best_val
```

```python
    """Compute the change in correlation vector after h[i]: 1->0, h[j]: 0->1."""
    k_vals = np.arange(-(n-1), n)  # lags
    n_corr = 2*n - 1
    delta = np.zeros(n_corr)
    
    # h[i] goes from 1 to 0:
    # For each lag k: the term h[i] appears when x=i (contribution 1*(1-h[i+k]) -> 0)
    #                 and when x+k=i, i.e., x=i-k (contribution h[i-k]*(1-1)=0 -> h[i-k]*(1-0)=h[i-k])
    
    # Term at x=i: -(1-h[i+k]) for each k where 0<=i+k<n
    ik = i + k_vals  # positions i+k for each lag
    mask = (ik >= 0) & (ik < n)

    ik2 = i - k_vals
```
