```python
    w = np.exp(s - sm)
    w /= w.sum()
    return w

def grad_obj(h, n, alpha):
    """Gradient of smooth max of overlaps."""
    c = all_overlaps(h, n)
    w = lse_weights(c, alpha)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            g[:nv] += wi * dx * (1.0 - h[k:])
            g[k:] += -wi * dx * h[:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wi * dx * (1.0 - h[:nv])
            g[:nv] += -wi * dx * h[kk:]
    return g

def obj_penalty(h, n, alpha, mu):
    c = all_overlaps(h, n)
    val = lse_max(c, alpha)
    val += mu * (h.sum() - 0.5 * n) ** 2
    return val

def grad_penalty(h, n, alpha, mu):
    g = grad_obj(h, n, alpha)
    g += 2.0 * mu * (h.sum() - 0.5 * n)
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} exactly."""
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            if not mask.any():
                break
            weights = np.where(mask, 1.0 - h, 0.0)
            total_w = weights.sum()
            if total_w < 1e-15:
                break
            frac = min(1.0, diff / total_w)
            h += frac * weights
        else:
            mask = h > 1e-16
            if not mask.any():
                break
            weights = np.where(mask, h, 0.0)
            total_w = weights.sum()
            if total_w < 1e-15:
                break
            frac = min(1.0, -diff / total_w)
            h -= frac * weights
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return float(m + (sm + np.log(np.exp(s - sm).sum())) / alpha)


def lse_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    return w


def grad_obj(h, n, alpha):
    """Gradient of smooth max of all overlaps w.r.t. h."""
    c = all_overlaps(h, n)
    w = lse_weights(c, alpha)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wi * dx * (1.0 - h[k:])
                g[k:] += -wi * dx * h[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wi * dx * (1.0 - h[:nv])
                g[:nv] += -wi * dx * h[kk:]
    return g


def obj_pen(h, n, alpha, mu):
    c = all_overlaps(h, n)
    val = lse_max(c, alpha)
    val += mu * (h.sum() - 0.5 * n) ** 2
    return val


def grad_pen(h, n, alpha, mu):
    g = grad_obj(h, n, alpha)
    g += 2.0 * mu * (h.sum() - 0.5 * n)
    return g


def optimize_lbfgsb(h, n, alpha_seq, mu_seq, deadline):
    bounds = [(0.0, 1.0)] * n
    best_h = project(h, n)
    best_c = compute_c5(best_h, n)
    
    for i in range(len(alpha_seq)):
        alpha = alpha_seq[i]
        mu = mu_seq[i]
        remaining = deadline - time.time()
        if remaining < 3:
            break
        max_iter = max(50, int(remaining * 3))
        try:
            res = minimize(obj_pen, best_h, jac=grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter * 30,
                                   'maxls': 50})
            h_new = project(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c


def proj_gradient_descent(h, n, deadline, alpha=10000, lr_init=0.1):
    h = project(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    for _ in range(200000):
        if time.time() > deadline:
            break
        g = grad_obj(h, n, alpha)
        d = g - g.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        
        step = lr
        improved = False
        for _ in range(30):
            h_new = project(h + step * d, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c


def steepest_pair_descent(h, n, deadline, alpha=20000):
    h = project(h, n)
    best_c = compute_c5(h, n)
    
    for _ in range(100000):
        if time.time() > deadline:
            break
        g = grad_obj(h, n, alpha)
        
        # Find best pair: maximize g[i] - g[j] with h[i]>0 and h[j]<1
        g_i = np.where(h > 1e-10, g, -np.inf)
        g_j = np.where(h < 1.0 - 1e-10, g, np.inf)
        
        i = int(np.argmax(g_i))
        j = int(np.argmin(g_j))
        
        if i == j or h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
            # Try removing the best i from j's candidates
            g_i2 = g_i.copy()
            g_i2[j] = -np.inf
            g_j2 = g_j.copy()
            g_j2[i] = np.inf
            if np.all(np.isinf(g_i2)) or np.all(np.isinf(g_j2)):
                break
            i = int(np.argmax(g_i2))
            j = int(np.argmin(g_j2))
            if i == j:
                break
        
        if g[i] - g[j] < 1e-12:
            break
        
        # Line search
        max_delta = min(h[i], 1.0 - h[j])
        best_delta = 0.0
        best_c_try = best_c
        
        for _ in range(40):
            delta = max_delta * 0.5
            if delta < 1e-12:
                break
            h[i] -= delta
            h[j] += delta
            c = compute_c5(h, n)
            if c < best_c_try:
                best_c_try = c
                best_delta = delta
            else:
                h[i] += delta
                h[j] -= delta
                # Try smaller
                max_delta = delta
                continue
        
        if best_delta > 0:
            h[i] -= best_delta
            h[j] += best_delta
            best_c = best_c_try
        else:
            break
    
    return h, best_c


def pair_search_local(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    h = project(h, n)
    best_c = compute_c5(h, n)
    target = 0.5 * n
    
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try moving mass from i to nearby j
                lo = max(0, i - 50)
                hi = min(n, i + 51)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move d from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
                    # Move d from j to i
                    if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[j] += d
                            h[i] -= d
    return h, best_c


def interpolate_to_n(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return h_new


def generate_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project(h, n)


def generate_block_init(n, seed):
    """Generate a block-structured initialization."""
    rng = np.random.RandomState(seed)
    # Create blocks of high and low values
    num_blocks = rng.randint(4, 10)
    h = np.zeros(n)
    block_size = n // num_blocks
    positions = rng.choice(n, num_blocks, replace=False)
    for pos in positions:
        start = (pos // block_size) * block_size
        end = min(start + block_size, n)
        h[start:end] = rng.uniform(0.6, 1.0)
    return project(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(250, 0.5)
        n0 = 250
    
    best_h = project(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: L-BFGS-B at parent resolution
    t_phase1_end = min(t0 + 180, deadline - 400)
    if t_phase1_end > t0 + 30:
        alpha_seq = [200, 1000, 5000, 20000, 100000]
        mu_seq = [50, 200, 1000, 5000, 20000]
        best_h, best_c = optimize_lbfgsb(best_h, best_n, alpha_seq, mu_seq, t_phase1_end)
    
    # Phase 2: Projected gradient descent
    if time.time() < deadline - 300:
        t_pg = min(time.time() + 60, deadline - 300)
        h_pg, c_pg = proj_gradient_descent(best_h, best_n, t_pg, alpha=50000, lr_init=0.05)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    
    # Phase 3: Steepest pair descent
    if time.time() < deadline - 250:
        t_sp = min(time.time() + 80, deadline - 250)
        h_sp, c_sp = steepest_pair_descent(best_h, best_n, t_sp, alpha=50000)
        if c_sp < best_c:
            best_c = c_sp
            best_h = h_sp
    
    # Phase 4: Try higher resolutions
    if time.time() < deadline - 150:
        resolutions = sorted(set([int(best_n * 1.5), int(best_n * 2), 400, 500, 600]))
        for n_try in resolutions:
            if time.time() > deadline - 80:
                break
            if n_try > 800 or n_try < 100:
                continue
            if n_try == best_n:
                continue
            # Interpolate
            h_interp = interpolate_to_n(best_h, best_n, n_try)
            h_interp = project(h_interp, n_try)
            c_interp = compute_c5(h_interp, n_try)
            
            alloc = min(120, deadline - time.time() - 50)
            if alloc < 30:
                break
            
            # Optimize at new resolution
            a_seq = [1000, 10000, 100000]
            m_seq = [500, 5000, 50000]
            h_new, c_new = optimize_lbfgsb(h_interp, n_try, a_seq, m_seq, time.time() + alloc * 0.6)
            
            if time.time() < deadline - 50:
                h_new, c_new = steepest_pair_descent(h_new, n_try, min(time.time() + 30, deadline - 40),
                                                     alpha=100000)
            if time.time() < deadline - 40:
                h_new, c_new = pair_search_local(h_new, n_try, deadline - 40,
                                                 deltas=[0.005, 0.002, 0.001, 0.0005])
            
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                best_n = n_try
    
    # Phase 5: Final optimization at best resolution
    if time.time() < deadline - 20:
        remaining = deadline - time.time() - 10
        a_final = [100000, 500000]
        m_final = [10000, 50000]
        best_h, best_c = optimize_lbfgsb(best_h, best_n, a_final, m_final, time.time() + remaining * 0.4)
    
    # Phase 6: Final local search
    if time.time() < deadline - 5:
        h_ps, c_ps = steepest_pair_descent(best_h, best_n, deadline - 3, alpha=200000)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps
    
    # Phase 7: Tiny pair search
    if time.time() < deadline - 2:
        h_ps, c_ps = pair_search_local(best_h, best_n, deadline - 1,
                                       deltas=[0.001, 0.0005, 0.0002, 0.0001])
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps
    
    best_h = project(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
