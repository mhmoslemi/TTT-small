def compute_c5(h, n):
    if n == 0:

    dx = 2.0 / n

def project_feasible(h, n):

    deadline = t0 + total_budget - 5

        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.zeros(n)
        h[:n // 2] = 1.0
        
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    rng = np.random.RandomState(seed)
    
    def optimize_smooth(x, alpha, maxiter=2000, gtol=1e-8):
        x = project_feasible(x, n)
        dx = 2.0 / n
        
        def obj(val):
            g = np.correlate(val, 1 - val, mode='full') * dx
            return smooth_max(g, alpha)
            
        def cons_fun(val):
            return val.sum() - n / 2.0
            
        def cons_jac(val):
            return np.ones(n)
            
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, x, method='trust-constr',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'gtol': gtol})
            return res.x
        except Exception as e:
            return x

    def fast_binary_search(h_init, max_sweeps=1500, n_restarts=3, seed_gen=42):
        target = n // 2
        rng_local = np.random.RandomState(seed_gen)
        
        # Binarize
        h0 = np.zeros(n)
        idx = np.argsort(h_init)[::-1]
        h0[idx[:target]] = 1.0
        
        corr = np.correlate(h0, 1 - h0, mode='full').copy()
        dx = 2.0 / n
        cur_c = corr.max() * dx
        
        best_bh = h0.copy()
        best_c_local = cur_c
        current_bh = h0.copy()
        
        m_range = np.arange(-(n - 1), n)
        
        for restart in range(n_restarts):
            if time.time() > deadline:
                break
            if restart > 0:
                for _ in range(rng_local.randint(3, 9)):
                    oi = np.where(current_bh == 1)[0]
                    zi = np.where(current_bh == 0)[0]
                    if len(oi) > 0 and len(zi) > 0:
                        a = rng_local.choice(oi)
                        b = rng_local.choice(zi)
                        current_bh[a], current_bh[b] = current_bh[b], current_bh[a]
                        
            h = current_bh.copy()
            ones = set(np.where(h == 1)[0].tolist())
            zeros = set(np.where(h == 0)[0].tolist())
            corr = np.correlate(h, 1 - h, mode='full').copy()
            cur_c = corr.max() * dx
            
            for sweep in range(max_sweeps):
                if time.time() > deadline:
                    break
                improved = False
                z_list = sorted(zeros)
                o_list = sorted(ones)
                
                for i in z_list:
                    if time.time() > deadline:
                        break
                    if improved:
                        break
                    for j in o_list:
                        im = i + m_range
                        imn = i - m_range
                        jm = j + m_range
                        jmn = j - m_range
                        
                        delta = np.zeros(2 * n - 1)
                        msk = (im >= 0) & (im < n)
                        delta[msk] += 1.0 - h[im[msk]]
                        msk = (imn >= 0) & (imn < n)
                        delta[msk] -= h[imn[msk]]
                        msk = (jm >= 0) & (jm < n)
                        delta[msk] -= 1.0 - h[jm[msk]]
                        msk = (jmn >= 0) & (jmn < n)
                        delta[msk] += h[jmn[msk]]
                        
                        # Bilinear cross terms
                        delta[n - 1] -= 2.0
                        k1 = n - 1 + (j - i)
                        if 0 <= k1 < 2 * n - 1:
                            delta[k1] += 1.0
                        k2 = n - 1 + (i - j)
                        if 0 <= k2 < 2 * n - 1:
                            delta[k2] += 1.0
                            
                        new_corr = corr + delta
                        nc = new_corr.max() * dx
                        if nc < cur_c - 1e-14:
                            h[i], h[j] = 1.0, 0.0
                            corr = new_corr
                            cur_c = nc
                            ones.discard(j)
                            ones.add(i)
                            zeros.discard(i)
                            zeros.add(j)
                            improved = True
                            break
                            # restart inner sweeps on improvement
                            if not improved:
                                break
                    if improved:
                        break
                        
                if not improved:
                    break
            if cur_c < best_c_local:
                best_c_local = cur_c
                best_bh = h.copy()
                current_bh = h.copy()
                
        return best_bh, best_c_local

    # 1. Initial continuous annealing
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        h = optimize_smooth(h, alpha, maxiter=3000, gtol=1e-10)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # 2. Binary exact peak tinker
    if time.time() < deadline - 100:
        hb, cb = fast_binary_search(best_h, max_sweeps=2000, n_restarts=5, seed_gen=seed)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # 3. Re-optimize from best (including fractional) and backtrack jump
    if time.time() < deadline:
        h = best_h.copy()
        for alpha in [20000, 100000, 500000, 2000000]:
            h = optimize_smooth(h, alpha, maxiter=3000, gtol=1e-10)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

            if time.time() > deadline - 10:
                break

    # 4. Random warping restarts to find global dips
    while time.time() < deadline - 15:
        # perturb the active boundary of best_h
        idx = np.argsort(np.abs(best_h - 0.5))[:60]
        noise = rng.randint(0, 2, size=len(idx)).astype(float) - 0.5
        h_pert = best_h.copy()
        h_pert[idx] += noise
        
        h_pert = project_feasible(h_pert, n)
        h_pert = optimize_smooth(h_pert, alpha=50000, maxiter=1000, gtol=1e-9)
        h_pert = project_feasible(h_pert, n)
        c = compute_c5(h_pert, n)
        
        if c < best_c:
            best_c = c
            best_h = h_pert.copy()

    # Final guarantee
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
