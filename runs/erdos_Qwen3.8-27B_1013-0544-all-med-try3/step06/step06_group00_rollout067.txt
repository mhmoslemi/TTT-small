                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                if i == j:
                    continue
                delta = compute_swap_delta(h, i, j, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                delta = compute_swap_delta(h, i, best_j, n, m_range)
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=10000000, T0=0.01, cool=0.99992, scale=5000.0):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T0
    no_improve = 0
    patience = 200000
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 10000 == 0:
            no_improve += 1
            if no_improve > patience // 10000:
                T = T0 * 0.3
                no_improve = 0
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        if i == j:
            continue
        delta = compute_swap_delta(h, i, j, n, m_range)
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c
        if not accept and T > 1e-14:
            exp_arg = scale * (cur_c - nc) / T
            if exp_arg > 700:
                accept = True
            else:
                accept = rng.random() < np.exp(exp_arg)
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
        T *= cool
    return best_h, best_c

def slsqp_phase(h_start, n, deadline, alphas, maxiter=3000):
    best_h = h_start.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h = slsqp_optimize(best_h, n, alpha, maxiter=maxiter, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        hb = to_binary(h, n)
        cb = compute_c5(hb, n)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary greedy (25s)
    p1 = t0 + 25
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: SLSQP main phase with progressive alpha (350s)
    p2 = t0 + 375
    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 300000, 1000000, 3000000, 10000000]
    best_h, best_c = slsqp_phase(best_h, n, p2, alphas, maxiter=4000)

    # Phase 3: Binary annealing from best (150s)
    p3 = t0 + 525
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=10000000, T0=0.008, cool=0.99993, scale=8000.0)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 4: SLSQP re-refinement (120s)
    p4 = t0 + 645
    alphas4 = [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    best_h, best_c = slsqp_phase(best_h, n, p4, alphas4, maxiter=5000)

    # Phase 5: Another anneal round with different params (100s)
    p5 = t0 + 745
    ha2, ca2 = anneal_binary(best_h, n, p5, seed=seed + 1, n_iters=5000000, T0=0.005, cool=0.99996, scale=10000.0)
    if ca2 < best_c:
        best_c, best_h = ca2, ha2.copy()

    # Phase 6: SLSQP polish (80s)
    p6 = t0 + 825
    alphas6 = [100000, 1000000, 10000000, 100000000]
    best_h, best_c = slsqp_phase(best_h, n, p6, alphas6, maxiter=6000)

    # Phase 7: Try different resolutions (remaining time - 20s buffer)
    p7 = dl - 20
    resolutions = [480, 500, 520, 600]
    for nh in resolutions:
        if time.time() > p7 - 40:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        ch = compute_c5(hh, nh)
        # Quick binary greedy
        hh_b, ch_b = greedy_binary(hh, nh, min(time.time() + 15, p7 - 40), max_sweeps=40)
        if ch_b < ch:
            hh, ch = hh_b, ch_b
        # SLSQP at new resolution
        if time.time() < p7 - 25:
            hh2, ch2 = slsqp_phase(hh, nh, p7 - 25, [1000, 10000, 100000, 1000000, 10000000], maxiter=2000)
            if ch2 < ch:
                hh, ch = hh2, ch2
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
