    if d >= 0:
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        ad = -d
        grad[ad:] += (1.0 - h[:n-ad])
        grad[:n-ad] -= h[ad:]
    
    grad *= dx
    return grad, c[k_max] * dx

def anderson_subgradient(h, n, max_iter=200, m=5, deadline=None):
    """Anderson acceleration of subgradient method."""
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    # Store for Anderson
    X = [h_curr.copy()]
    G = []
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        grad, val = subgradient_max_c(h_curr, n)
        
        if len(X) < m:
            # Standard subgradient step
            alpha = 2.0 / (it + 2)  # diminishing step size
            h_new = project_feasible(h_curr - alpha * grad, n)
            X.append(h_new)
            G.append(grad)
            if compute_c5(h_new, n) < best_c:
                best_c = compute_c5(h_new, n)
                h_curr = h_new
        else:
            # Anderson step
            # Extrapolation
            h_extrap = h_curr.copy()
            G_arr = np.array(G[-(m-1):])
            # Solve for weights that minimize ||h_extrap - h_{k+1}||
            # Simplified: use the differences
            H = np.array(X[-(m-1):] - h_curr)
            # Find weights w minimizing ||G_arr @ w + grad||
            A = G_arr.T @ G_arr + 1e-8 * np.eye(m-1)
            b = -G_arr.T @ grad
            try:
                w = np.linalg.solve(A, b)
            except:
                w = np.zeros(m-1)
            h_anderson = h_curr - grad
            for i in range(m-1):
                h_anderson += w[i] * (X[-(m-1)+i] - h_curr)
            h_anderson = project_feasible(h_anderson, n)
            
            c_new = compute_c5(h_anderson, n)
            if c_new < best_c:
                best_c = c_new
                h_curr = h_anderson
            
            X.append(h_anderson)
            G.append(grad)
            if len(X) > m:
                X.pop(0)
                G.pop(0)
    
    return h_curr, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def eq_fun(x):
        return x.sum() - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, deadline, min_delta=0.001):
    """Local search focused on reducing the worst shift."""
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        dx = 2.0 / n
        c = np.correlate(h, 1-h, mode='full') * dx
        d_star = int(np.argmax(c)) - n + 1
        c_star = c.max()
        
        # Compute gradient of c_{d_star}
        grad = np.zeros(n)
        if d_star >= 0:
            grad[:n-d_star] += (1.0 - h[d_star:])
            grad[d_star:] -= h[:n-d_star]
        else:
            ad = -d_star
            grad[ad:] += (1.0 - h[:n-ad])
            grad[:n-ad] -= h[ad:]
        grad *= dx
        
        # Find beneficial moves: increase h where grad < 0, decrease where grad > 0
        # Try small transfers from high-grad to low-grad positions
        beneficial = h[grad < -min_delta]  # positions where decreasing h helps
        harmful = h[grad > min_delta]  # positions where increasing h helps
        
        if len(beneficial) == 0 or len(harmful) == 0:
            break
        
        best_move_c = best_c
        best_move = None
        
        # Sample a few candidate moves
        n_bench = min(50, len(beneficial))
        n_harm = min(50, len(harmful))
        
        bench_idx = np.where(grad < -min_delta)[0][:n_bench]
        harm_idx = np.where(grad > min_delta)[0][:n_harm]
        
        improved = False
        for i in bench_idx:
            if time.time() > deadline:
                break
            for j in harm_idx:
                if i == j:
                    continue
                # Move mass from i to j
                for delta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                    if h[i] - delta >= 0 and h[j] + delta <= 1:
                        h[i] -= delta
                        h[j] += delta

                            improved = True
                            best_move = (i, j, delta)
                            break

                            h[i] += delta
                            h[j] -= delta
                if best_move is not None:
                    break
            if best_move is not None:
                break
        
        if not improved:
            break

def generic_pair_search(h, n, deadline, deltas):

def near_pair_search(h, n, deadline, deltas, radius=50):
    """Near-neighbor pair search (for larger n)."""
    best_c = compute_c5(h, n)

        while time.time() < deadline:

            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break

The smooth max: f(h) = m + (1/alpha) * log(sum exp(alpha * (c_k - m)))
where m = max_k c_k, c_k = (correlate(h, 1-h, 'full'))[k] * dx

= sum_k [w_k * dc_k/dh_j]
where w_k = exp(alpha*(c_k - m)) / sum_l exp(alpha*(c_l - m)) (softmax weights)

And dc_k/dh_j = dx * [(1-h_{j+d_k}) - h_{j-d_k}] (with boundary handling)

∂f/∂h_j = dx * sum_k w_k * [(1-h_{j+d_k}) - h_{j-d_k}]

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += ((target - s) / tot) * room
        else:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    max_s = s.max()
    exp_s = np.exp(s - max_s)
    w_sum = exp_s.sum()
    weights = exp_s / w_sum
    val = m + (max_s + np.log(w_sum)) / alpha

    if d >= 0:
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        ad = -d
        grad[ad:] += (1.0 - h[:n-ad])
        grad[:n-ad] -= h[ad:]
    grad *= dx
    return grad, c[k_max] * dx

def anderson_method(h, n, max_iter=300, memory=5, deadline=None):
    """Anderson acceleration of subgradient method."""
    best_c = compute_c5(h, n)
    x_curr = h.copy()
    
    # History
    x_hist = [x_curr.copy()]
    s_hist = []  # steps
    y_hist = []  # gradient differences
    
    gamma = 1.0  # step size
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        g_curr, val = subgradient_step(x_curr, n)
        
        if len(x_hist) == 1:
            # First step
            x_next = project_feasible(x_curr - gamma * g_curr, n)
        else:
            # Anderson step
            m = min(memory, len(x_hist) - 1)
            # Build matrices
            S = np.column_stack([x_hist[-(i+1)] - x_curr for i in range(m)])  # n x m
            Y = np.column_stack([g_hist[-(i+1)] for i in range(m)])  # n x m
            # Actually let me use the proper Anderson formulation
            # x_{k+1} = x_k - gamma * g_k + sum_i alpha_i * (x_k - x_{k-1-i})
            # where alpha minimizes ||g_k + sum_i alpha_i * (g_k - g_{k-1-i})||
            
            # Differences
            d_x = np.array([x_curr - x_hist[-(i+2)] for i in range(m-1)]) if m > 1 else np.array([])
            d_g = np.array([g_curr - s_hist[-(i+1)] for i in range(m-1)]) if m > 1 and len(s_hist) >= m else np.array([])
            
            if len(d_x) > 0:
                # Solve (d_g^T d_g + mu*I) alpha = -d_g^T g_curr
                mu = 1e-6
                A = d_g.T @ d_g + mu * np.eye(m-1)
                b = -d_g.T @ g_curr
                try:
                    alpha = np.linalg.solve(A, b)
                except:
                    alpha = np.zeros(m-1)
                x_next = project_feasible(x_curr - gamma * g_curr + d_x @ alpha, n)
            else:
                x_next = project_feasible(x_curr - gamma * g_curr, n)
        
        c_new = compute_c5(x_next, n)
        
        x_hist.append(x_next)
        s_hist.append(g_curr)
        
        if len(x_hist) > memory + 1:
            x_hist.pop(0)
            if len(s_hist) > memory:
                s_hist.pop(0)
        
        if c_new < best_c:
            best_c = c_new
            x_curr = x_next
            # Reduce step size on improvement? Or keep it?
        else:
            gamma *= 0.9
    
    return x_curr, best_c

def targeted_search(h, n, deadline):
    """Targeted local search focused on worst shift."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_full = np.correlate(h, 1-h, mode='full') * dx
            d_star = int(np.argmax(c_full)) - n + 1
            
            # Compute "benefit" of moving mass at each position
            # benefit[j] = how much c_{d_star} decreases if we increase h[j]
            grad = np.zeros(n)
            if d_star >= 0:
                grad[:n-d_star] += (1.0 - h[d_star:])
                grad[d_star:] -= h[:n-d_star]
            else:
                ad = -d_star
                grad[ad:] += (1.0 - h[:n-ad])
                grad[:n-ad] -= h[ad:]
            
            # We want to increase h where grad is most negative (reduces c_{d_star})
            # and decrease h where grad is most positive
            # But we also need to check the overall max doesn't increase
            
            # Try top candidates
            order = np.argsort(grad)  # most negative first
            n_try = min(30, n // 3)
            
            for idx in order[:n_try]:
                if time.time() > deadline:
                    return h, best_c
                if h[idx] + d > 1.0:
                    continue
                # Find where to take mass from
                # Preferentially from positions where grad is positive
                source_order = np.argsort(-grad)
                for sidx in source_order[:n_try]:
                    if sidx == idx:
                        continue
                    if h[sidx] - d < 0:
                        continue
                    h[idx] += d
                    h[sidx] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[idx] -= d
                        h[sidx] += d
                if improved:
                    break
            
            if not improved:
                break
    
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_full(h_start, n, deadline):
    """Full optimization pipeline at resolution n."""
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Stage 1: SLSQP with increasing alpha
    alphas = [100, 300, 1000, 3000, 10000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_with_grad(h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    # Stage 2: Anderson subgradient
    if time.time() < deadline - 10:
        h, c = anderson_method(h, n, max_iter=200, deadline=deadline - 8)
        if c < best_c:
            best_c = c
    
    # Stage 3: Targeted search
    if time.time() < deadline - 5:
        h, best_c = targeted_search(h, n, deadline - 3)
    
    # Stage 4: Final SLSQP
    if time.time() < deadline - 2:
        ht = slsqp_with_grad(h, n, 50000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
