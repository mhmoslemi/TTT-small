                # Move from j_max to i
                if j_max != i and h[j_max] >= step and h[i] + step <= 1.0:
                    h[j_max] -= step
                    h[i] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[j_max] += step
                        h[i] -= step
            if not improved:
                break
    return h, best_c

def fine_pair_search(h, n, deadline, steps=None, window=15):
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for step in steps:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= step and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    if h[j] >= step and h[i] + step <= 1.0:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
    return h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def target_shift_optimize(h, n, deadline, alpha_max=5000000, maxiter=200):
    """Focus optimization on the shift that achieves the current maximum."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [alpha_max, alpha_max // 2, alpha_max // 5]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_func, h, jac=obj_grad, args=(n, alpha, 0.01),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def randomized_restart(h, n, deadline, rng, alpha=100000):
    """Perturb and re-optimize to try escaping local minimum."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    noise_scales = [0.05, 0.02, 0.01, 0.005]
    for ns in noise_scales:
        if time.time() > deadline - 10:
            break
        h_pert = h + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, n)
        h_new, c_new = lbfgs_optimize(h_pert, n, deadline - 5, max_iters=80,
                                      alpha_seq=[5000, 20000, 100000, 500000],
                                      mu_seq=[10.0, 10.0, 1.0, 0.1])
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize at parent resolution (confirm and improve baseline)
    if time.time() < deadline - 500:
        t1 = min(t0 + 100, deadline - 500)
        h1, c1 = lbfgs_optimize(best_h, n_init, t1, max_iters=100,
                                alpha_seq=[5000, 20000, 100000, 500000, 2000000],
                                mu_seq=[10.0, 10.0, 1.0, 0.1, 0.01])
        if c1 < best_c:
            best_c = c1
            best_h = h1
        # SLSQP polish
        if time.time() < deadline - 480:
            h1s, c1s = slsqp_optimize(best_h, n_init, min(time.time() + 50, deadline - 480),
                                      alpha=2000000, maxiter=800)
            if c1s < best_c:
                best_c = c1s
                best_h = h1s

    # Phase 2: Try n=500
    n2 = 500
    if time.time() < deadline - 350:
        h2 = upsample(best_h, len(best_h), n2)
        t2 = min(time.time() + 180, deadline - 200)
        h2, c2 = lbfgs_optimize(h2, n2, t2, max_iters=120,
                                alpha_seq=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
                                mu_seq=[1.0, 5.0, 50.0, 100.0, 50.0, 5.0, 0.5, 0.05])
        if time.time() < deadline - 180:
            h2, c2 = gradient_guided_search(h2, n2, min(time.time() + 60, deadline - 180),
                                            alpha=200000, steps=[0.002, 0.001, 0.0005, 0.0002],
                                            window=20)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try n=700
    n3 = 700
    if time.time() < deadline - 200:
        h3 = upsample(best_h, len(best_h), n3)
        t3 = min(time.time() + 120, deadline - 100)
        h3, c3 = lbfgs_optimize(h3, n3, t3, max_iters=80,
                                alpha_seq=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
                                mu_seq=[1.0, 5.0, 50.0, 100.0, 50.0, 5.0, 0.5, 0.05])
        if time.time() < deadline - 80:
            h3, c3 = gradient_guided_search(h3, n3, min(time.time() + 40, deadline - 80),
                                            alpha=100000, steps=[0.001, 0.0005, 0.0002],
                                            window=15)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Try complement (1-h) as restart at best resolution
    nf = len(best_h)
    if time.time() < deadline - 100:
        h_comp = 1.0 - best_h
        h_comp = project_feasible(h_comp, nf)
        t4 = min(time.time() + 60, deadline - 80)
        h4, c4 = lbfgs_optimize(h_comp, nf, t4, max_iters=80,
                                alpha_seq=[5000, 50000, 500000, 2000000],
                                mu_seq=[5.0, 1.0, 0.1, 0.01])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Randomized restarts
    if time.time() < deadline - 80:
        t5 = min(time.time() + 60, deadline - 60)
        h5, c5 = randomized_restart(best_h, nf, t5, rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Targeted optimization on the worst shift
    if time.time() < deadline - 50:
        t6 = min(time.time() + 40, deadline - 30)
        h6, c6 = target_shift_optimize(best_h, nf, t6, alpha_max=5000000, maxiter=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final fine polishing
    if time.time() < deadline - 5:
        nf = len(best_h)
        # Gradient-guided with very small steps
        t7 = min(time.time() + 30, deadline - 3)
        h7, c7 = gradient_guided_search(best_h, nf, t7, alpha=500000,
                                        steps=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                        window=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # Final pair search with smallest steps
        if time.time() < deadline - 2:
            h8, c8 = fine_pair_search(best_h, nf, deadline - 1,
                                      steps=[0.0001, 0.00005, 0.00002],
                                      window=8)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
