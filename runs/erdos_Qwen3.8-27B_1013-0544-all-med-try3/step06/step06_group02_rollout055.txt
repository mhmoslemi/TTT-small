```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Gradient of the log-sum-exp approximation to max(c)"""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    return mx + np.log(np.sum(np.exp(sh - mx))) / alpha

def smooth_max_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def lbfgs_optimize(h, n, deadline, max_alphas=None):
    """Run L-BFGS-B with increasing alpha"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    if max_alphas is None:
        max_alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    
    for alpha in max_alphas:
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time() - 2
        if remaining < 5:
            break
        
        # Scale maxiter based on available time
        max_iters = min(500, int(remaining * 50))
        
        def obj(x):
            return smooth_max_obj(x, n, alpha)
        
        def grad(x):
            return smooth_max_grad(x, n, alpha)
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 20,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c
```

Let me think about the block move search. The idea is:
- Select a random block of indices [i, i+w) where w is the block width
- Shift all values in the block by some amount d (maintaining the sum constraint by shifting a corresponding block in the opposite direction)
- Or: move a "bump" - add d to a block and subtract from another block

```python
def block_move_search(h, n, deadline):
    best_c = compute_c5(h, n)
    widths = [2, 3, 5, 10, 20]
    ds = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    while time.time() < deadline:
        improved = False
        for w in widths:
            for d in ds:
                for _ in range(n // w):  # Try a few random positions
                    i = rng.randint(0, n - w)
                    j = rng.randint(0, n - w)
                    if abs(i - j) < w:
                        continue
                    # Try moving d from block i to block j
                    if np.all(h[i:i+w] - d >= 0) and np.all(h[j:j+w] + d <= 1):
                        h[i:i+w] -= d
                        h[j:j+w] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i:i+w] += d
                            h[j:j+w] -= d
                    # Try the reverse
                    if np.all(h[j:j+w] - d >= 0) and np.all(h[i:i+w] + d <= 1):
                        h[j:j+w] -= d
                        h[i:i+w] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j:j+w] += d
                            h[i:i+w] -= d
        if not improved:
            # Try smaller d
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    if mx > 500:
        v = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        v = np.log(np.sum(np.exp(sh)))
    return v / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_with_penalty(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = compute_c5_vec(x, n)
        sh = alpha * (c - c.max())
        mx = sh.max()
        if mx > 500:
            val = mx + np.log(np.sum(np.exp(sh - mx)))
        else:
            val = np.log(np.sum(np.exp(sh)))
        return val / alpha + mu * (x.sum() - n / 2.0) ** 2

    def grad(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-15, 'maxfun': maxiter * 30, 'maxls': 100})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lbfgs_schedule(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alphas_mus = [
        (100, 1.0), (500, 5.0), (2000, 10.0), (10000, 50.0),
        (50000, 200.0), (200000, 1000.0), (1000000, 5000.0), (5000000, 20000.0)
    ]
    for alpha, mu in alphas_mus:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 3
        if remaining < 5:
            break
        maxit = min(400, int(remaining * 20))
        if maxit < 10:
            break
        h_new = lbfgs_with_penalty(h, n, alpha, mu, maxit, time.time() + remaining)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def periodic_obj(z, p, n, alpha):
    h = np.tile(z, n // p + 1)[:n]
    return smooth_max_val(h, n, alpha) + 10.0 * (z.sum() - p / 2.0) ** 2

def periodic_grad(z, p, n, alpha):
    h = np.tile(z, n // p + 1)[:n]
    g_full = smooth_max_grad(h, n, alpha)
    g_z = np.zeros(p)
    for j in range(n):
        g_z[j % p] += g_full[j]
    g_z += 20.0 * (z.sum() - p / 2.0)
    return g_z

def optimize_periodic(z_init, p, n, deadline):
    z = z_init.copy()
    h_init = np.tile(z, n // p + 1)[:n]
    best_c = compute_c5(h_init, n)
    best_z = z.copy()
    bounds = [(0.0, 1.0)] * p
    alphas = [100, 1000, 10000, 100000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time() - 2
        if remaining < 3:
            break
        maxit = min(200, int(remaining * 10))
        if maxit < 5:
            break
        try:
            res = minimize(periodic_obj, best_z, jac=periodic_grad,
                         args=(p, n, alpha), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-14})
            z_new = res.x
            h_new = np.tile(z_new, n // p + 1)[:n]
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_z = z_new.copy()
        except Exception:
            pass
    h_best = np.tile(best_z, n // p + 1)[:n]
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    return h_best, best_c

def block_move_search(h, n, deadline, rng, max_passes=500):
    best_c = compute_c5(h, n)
    widths = [2, 3, 5, 8, 15, 30]
    ds = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    pass_count = 0
    while time.time() < deadline and pass_count < max_passes:
        improved = False
        n_trials = 300
        for _ in range(n_trials):
            if time.time() > deadline:
                return h, best_c
            w = widths[rng.randint(len(widths))]
            d = ds[rng.randint(len(ds))]
            i = rng.randint(0, n - w)
            j = rng.randint(0, n - w)
            if abs(i - j) < w:
                continue
            for (src, dst) in [(i, j), (j, i)]:
                block_src = h[src:src + w]
                block_dst = h[dst:dst + w]
                if np.all(block_src - d >= 0) and np.all(block_dst + d <= 1):
                    h[src:src + w] -= d
                    h[dst:dst + w] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[src:src + w] += d
                        h[dst:dst + w] -= d
        pass_count += 1
        if not improved and pass_count > 20:
            break
    return h, best_c

def pair_search_fine(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    ds = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05]
    for d in ds:
        while time.time() < deadline:
            improved = False
            for _ in range(2000):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    old_i, old_j = h[i], h[j]
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    old_i, old_j = h[i], h[j]
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
            if not improved:
                break
    return h, best_c

def random_perturb_reopt(h, n, rng, deadline, alpha=50000, mu=1000.0):
    h_pert = h + rng.randn(n) * 0.05
    h_pert = project_feasible(h_pert, n)
    remaining = deadline - time.time()
    if remaining < 10:
        return h, compute_c5(h, n)
    maxit = min(200, int(remaining * 15))
    h_new = lbfgs_with_penalty(h_pert, n, alpha, mu, maxit, time.time() + remaining)
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    c_old = compute_c5(h, n)
    if c_new < c_old:
        return h_new, c_new
    return h, c_old

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
```
