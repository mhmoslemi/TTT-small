def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    exponents = alpha * (c - c_max)
    exponents = np.clip(exponents, -700, 700)
    exps = np.exp(exponents)
    w = exps / exps.sum()
    
    F = c_max + np.log(exps.sum()) / alpha
    
    # Gradient
    # c_full[m] = sum_i x[i] * (1-x[i+m-(n-1)]) for valid i
    # dc_full[m]/dx[j] = (1-x[j+m-(n-1)]) if 0<=j+m-(n-1)<n
    #                    + (-x[j-(m-(n-1))]) if 0<=j-(m-(n-1))<n
    # Wait, c_full[m] = sum_i x[i] * (1-x[i+m-(n-1)])
    # d/dx[j]: 
    #   if j appears as i: term is (1-x[j+m-(n-1)]) if 0<=j+m-(n-1)<n
    #   if j appears as i+m-(n-1), i.e., i = j-m+(n-1): term is -x[j-m+(n-1)] if 0<=j-m+(n-1)<n
    #   
    # Let k = m-(n-1) (the lag)
    # dc_k/dx[j] = (1-x[j+k]) * 1[0<=j+k<n] - x[j-k] * 1[0<=j-k<n]
    
    # So grad_x[j] = dx * sum_m w[m] * dc_k/dx[j] where k = m-(n-1)
    # = dx * sum_k w[k+(n-1)] * [(1-x[j+k])*1[0<=j+k<n] - x[j-k]*1[0<=j-k<n]]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        k = m - (n-1)
        wm = w[m] * dx
        if wm < 1e-18:
            continue
        # Part 1: (1-x[j+k]) for j where 0<=j+k<n, i.e., -k <= j < n-k
        j1_start = max(0, -k)
        j1_end = min(n, n-k)  # exclusive
        if j1_end > j1_start:
            grad[j1_start:j1_end] += wm * (1.0 - x[j1_start+k:j1_end+k])
        # Part 2: -x[j-k] for j where 0<=j-k<n, i.e., k<=j<n+k
        j2_start = max(0, k)
        j2_end = min(n, n+k)  # exclusive
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= wm * x[j2_start-k:j2_end-k]
    
    return F, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c_vector(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    exponents = alpha * (c - c_max)
    exponents = np.clip(exponents, -700, 700)
    exps = np.exp(exponents)
    exps_sum = exps.sum()
    w = exps / exps_sum
    F = c_max + np.log(exps_sum) / alpha
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        wm = w[m] * dx
        if wm < 1e-18:
            continue
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            grad[j1s:j1e] += wm * (1.0 - x[j1s + k:j1e + k])
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            grad[j2s:j2e] -= wm * x[j2s - k:j2e - k]
    
    return F, grad

def penalty_obj_and_grad(x, n, alpha, penalty):
    F, g = smooth_obj_and_grad(x, n, alpha)
    diff = x.sum() - n / 2.0
    F += penalty * diff * diff
    g += 2.0 * penalty * diff * np.ones(n)
    return F, g

def projected_gradient_step(x, n, alpha, lr, deadline, max_iter=5000):
    h = x.copy()
    best_F = smooth_obj_and_grad(h, n, alpha)[0]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        F, g = smooth_obj_and_grad(h, n, alpha)
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        F_new = smooth_obj_and_grad(h_new, n, alpha)[0]
        if F_new < best_F:
            best_F = F_new
            h = h_new
            # Decrease learning rate slowly
            lr *= 0.999
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, compute_c5(h, n)

def gradient_guided_local_search(h, n, deadline, max_iter=200000):
    best_c = compute_c5(h, n)
    h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in steps:
        while time.time() < deadline:
            improved = False
            for _ in range(max_iter):
                if time.time() > deadline:
                    break
                F, g = smooth_obj_and_grad(h, n, 50000)
                # Find best increase and decrease
                # We want to increase h where g is most negative, decrease where g is most positive
                # But we must maintain sum constraint: increase one, decrease another
                # Effective gradient for a transfer from i to j: -g[i] + g[j] (we want this negative)
                # i.e., g[i] > g[j] means transferring from i to j helps
                
                # Find top candidates
                pos_idx = np.argsort(g)[::-1][:20]  # most positive grad -> should decrease
                neg_idx = np.argsort(g)[:20]       # most negative grad -> should increase
                
                best_improve = 1e-14
                best_i, best_j = -1, -1
                for i in pos_idx:
                    if h[i] < step:
                        continue
                    for j in neg_idx:
                        if i == j:
                            continue
                        if h[j] > 1.0 - step:
                            continue
                        # Try transfer
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - best_improve:
                            best_improve = best_c - c
                            best_i, best_j = i, j
                        h[i] += step
                        h[j] -= step
                
                if best_i >= 0:
                    h[best_i] -= step
                    h[best_j] += step
                    best_c = compute_c5(h, n)
                    improved = True
                else:
                    break
            if not improved:
                break
    return h, best_c

def simulated_annealing(h, n, deadline, t_start=0.01, t_end=0.0001, max_iter=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    rng = np.random.RandomState(123)
    
    n_iter = 0
    while time.time() < deadline and n_iter < max_iter:
        n_iter += 1
        t = t_start * (t_end / t_start) ** (n_iter / max_iter)
        
        # Random pair transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        step = rng.uniform(0.001, 0.01)
        
        if direction == 1:
            if h[i] < step or h[j] > 1.0 - step:
                continue
            h[i] -= step
            h[j] += step
        else:
            if h[j] < step or h[i] > 1.0 - step:
                continue
            h[j] -= step
            h[i] += step
        
        c = compute_c5(h, n)
        delta = c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            if direction == 1:
                h[i] += step
                h[j] -= step
            else:
                h[j] += step
                h[i] -= step
    
    return best_h, best_c

def upsolve(h, n, n_new, deadline):
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = project_feasible(h_interp, n_new)
    return h_interp

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Phase A: L-BFGS-B with moderate alpha
    alphas = [500, 2000, 10000, 50000, 200000, 500000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj_and_grad, h, args=(n, a),
                          method='L-BFGS-B', jac=True,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 50})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                h = ht
        except Exception:
            pass
    
    # Phase B: Projected gradient
    if time.time() < deadline - 30:
        ht, c = projected_gradient_step(h, n, 100000, 0.001, time.time() + min(60, deadline - time.time() - 10))
        if c < best_c:
            best_c = c
            h = ht
    
    # Phase C: Gradient-guided local search
    if time.time() < deadline - 5:
        h, best_c = gradient_guided_local_search(h, n, deadline - 2)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
