        threshold = R_min + 0.01  # 1% above minimum

        # Q_A @ h = (1/|A|) sum_{k in A} (Q_k @ h)

        # (Q_k @ h)[i] = h[i-lag_k] (shifted h)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_autocorr(h, n):
    dx = 2.0 / n
    return np.correlate(h, h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def obj_lbfgs(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -80, 80)
    obj = m + float(np.log(np.sum(np.exp(s)))) / alpha
    constr = x.sum() - n / 2.0
    return obj + penalty * constr * constr

def grad_lbfgs(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -80, 80)
    exp_s = np.exp(s)
    w = exp_s / exp_s.sum()
    
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        wk = w[k] * dx
        m_s = max(0, lag)
        m_e = min(n, n + lag)
        if m_e > m_s:
            idx = np.arange(m_s, m_e)
            g[idx] += wk * (1.0 - x[idx - lag])
        m_s2 = max(0, -lag)
        m_e2 = min(n, n - lag)
        if m_e2 > m_s2:
            idx2 = np.arange(m_s2, m_e2)
            g[idx2] -= wk * x[idx2 + lag]
    
    constr = x.sum() - n / 2.0
    g += 2.0 * penalty * constr
    return g

def optimize_lbfgs(h0, n, alphas, penalty=1e4, maxiter_per=3000):
    h = h0.copy()
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_lbfgs, h, jac=grad_lbfgs, method='L-BFGS-B',
                       args=(n, alpha, penalty),
                       bounds=bounds,
                       options={'maxiter': maxiter_per, 'ftol': 1e-14,
                                'gtol': 1e-12, 'maxls': 50})
        h = project_feasible(res.x, n)
    return h

def active_set_ascent(h, n, max_rounds=200, deadline=None):
    """Gradient ascent on the min autocorrelation (active set method)."""
    best_c = compute_c5(h, n)
    
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        
        R = compute_autocorr(h, n)
        R_min = R.min()
        current_c = 1.0 - R_min
        if current_c < best_c:
            best_c = current_c
        
        # Active set: lags near the minimum
        margin = max(0.005, 0.02 * R_min)
        active = np.where(R <= R_min + margin)[0]
        
        # Gradient of sum of R(k) for k in active
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            m_s = max(0, lag)
            m_e = min(n, n + lag)
            if m_e > m_s:
                idx = np.arange(m_s, m_e)
                grad[idx] += h[idx - lag]
            m_s2 = max(0, -lag)
            m_e2 = min(n, n - lag)
            if m_e2 > m_s2:
                idx2 = np.arange(m_s2, m_e2)
                grad[idx2] += h[idx2 + lag]
        grad /= len(active)
        
        # Center the gradient (to maintain sum constraint)
        grad_c = grad - grad.mean()
        
        # Line search
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_t = project_feasible(h + step * grad_c, n)
            c_t = compute_c5(h_t, n)
            if c_t < best_c - 1e-15:
                best_c = c_t
                h = h_t
                improved = True
                break
        if not improved:
            # Try smaller neighborhood
            for step in [0.0005, 0.0002, 0.0001]:
                h_t = project_feasible(h + step * grad_c, n)
                c_t = compute_c5(h_t, n)
                if c_t < best_c - 1e-15:
                    best_c = c_t
                    h = h_t
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c

def targeted_pair_search(h, n, deadline, deltas, radius=80):
