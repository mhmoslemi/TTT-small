def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):

        if abs(diff) < 1e-12:

            if s < 1e-14:

            if s < 1e-14:

        total = h.sum()
        if total < 1e-14:

        scale = diff / total
        h = h + scale * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_gradient(h, n, alpha):
    """
    Compute gradient of (1/alpha) * log(sum_k exp(alpha * c_k * dx))
    Returns (gradient, function_value)
    """
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * corr * dx
    # Softmax
    s_max = scaled.max()
    exp_s = np.exp(scaled - s_max)
    w = exp_s / exp_s.sum()  # softmax weights, shape (2n-1,)
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)  # shift
        wm = w[m] * dx
        # First term: (1-h[i-k]) for valid i
        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] += wm * (1.0 - h[lo - k:hi - k])
        # Second term: -h[i+k] for valid i
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 + k:hi2 + k]
    
    f_val = (s_max / alpha) + np.log(exp_s.sum()) / alpha
    return grad, f_val

def fista_smooth(h_init, n, alpha, max_iter, step_size=None, deadline=None):
    """FISTA with smoothed max objective"""
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    h_prev = h.copy()
    t = 1.0
    
    if step_size is None:
        # Lipschitz constant estimate
        # |grad f| <= sum_k w_k * ||grad c_k|| * dx <= max_k ||grad c_k|| * dx * sum_k w_k
        # ||grad c_k||_2 <= 2*sqrt(n) (rough estimate)
        # So L <= 2*sqrt(n)*dx = 4*sqrt(n)/n = 4/sqrt(n)
        step_size = 1.0 / (4.0 / np.sqrt(n))
    
    best_h = h.copy()
    best_f = compute_c5(h, n)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        grad, f_smooth = smooth_max_gradient(h, n, alpha)
        
        # Also compute actual C5
        actual_f = compute_c5(h, n)
        if actual_f < best_f:
            best_f = actual_f
            best_h = h.copy()
        
        # Extrapolation
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        y = h + ((t_old - 1.0) / t) * (h - h_prev)
        
        # Gradient step + projection
        h_prev = h.copy()
        h = project_feasible(y - step_size * grad, n)
    
    return best_h, best_f

    """Fast gradient computation using FFT convolution"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * corr * dx
    s_max = scaled.max()
    exp_s = np.exp(scaled - s_max)
    w = exp_s / exp_s.sum()  # shape (2n-1,)
    
    # w[m] corresponds to shift k = m - (n-1)
    # We need: grad[i] = sum_m w[m] * [(1-h[i-(m-(n-1))]) * 1{valid} - h[i+(m-(n-1))] * 1{valid}]
    
    # Let's use the loop approach for clarity (it's fast enough for n=250)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        wm_dx = wm * dx
        # Term 1: (1-h[i-k]) for i such that 0 <= i-k < n
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_start < i_end:
            grad[i_start:i_end] += wm_dx * (1.0 - h[i_start - k:i_end - k])
        # Term 2: -h[i+k] for i such that 0 <= i+k < n
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= wm_dx * h[i_start2 + k:i_end2 + k]
    
    f_val = (s_max + np.log(exp_s.sum())) / alpha
    return grad, f_val

1. The current max value: c_max
2. For which k's the max is achieved
3. Whether any of those k's (or potentially other k's) get worse

A sufficient condition for a swap to improve: max(c+Δc) < c_max, which is equivalent to c[k] + Δc[k] < c_max for ALL k.

A necessary condition: for the k* that achieves the max, c[k*] + Δc[k*] < c_max, i.e., Δc[k*] < 0.

So a quick necessary check is: Δc[k*] < 0 for all k* that achieve the max. If this fails, the swap doesn't improve.

If it passes, we still need to check all other k's, but we can skip k's where c[k] + max_possible_Δc < c_max.

max_possible_Δc for any k is at most 4 (since at most 4 terms change, each by at most 1). Actually for binary h, each term changes by at most 1, and there are at most 4 affected terms, so |Δc[k]| ≤ 4.

So if c[k] + 4 < c_max for some k, we don't need to check that k.

This means we only need to check k's where c[k] ≥ c_max - 4. This is a small set!
