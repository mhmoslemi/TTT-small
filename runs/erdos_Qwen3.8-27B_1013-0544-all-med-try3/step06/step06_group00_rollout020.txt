import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lsq_obj_and_grad(h, n, alpha, pen_coeff, dx):
    """Log-sum-exp objective and gradient with L2 penalty for sum constraint."""
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    p = exp_shifted / exp_shifted.sum()
    logsum = (shifted_max + np.log(exp_shifted.sum())) / alpha
    lse = m + logsum
    
    # Gradient
    ft = np.convolve(1.0 - h, p, mode='full')
    p_rev = p[::-1]
    st = np.convolve(h, p_rev, mode='full')
    
    # ft[j] for j in [0, n-1]
    # st[j] for j in [2n-2, 2n-2+n-1] = [2n-2, 3n-3]
    grad = dx * (ft[:n] - st[2*n-2:3*n-2])
    grad += 2.0 * pen_coeff * (h.sum() - n/2.0)
    
    return lse + pen_coeff * (h.sum() - n/2.0)**2, grad

def lbfgsb_optimize(h_init, n, alpha, pen_coeff, maxiter, deadline, ftol=1e-15):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        val, _ = lsq_obj_and_grad(x, n, alpha, pen_coeff, dx)
        return val
    
    def grad(x):
        _, g = lsq_obj_and_grad(x, n, alpha, pen_coeff, dx)
        return g
    
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10,
                              'maxls': 50})

    except Exception:
        return h_init

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lsq_val_grad(h, n, alpha, pen, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    p = exp_s / exp_s.sum()
    logsum = (sm + np.log(exp_s.sum())) / alpha
    sum_diff = h.sum() - n / 2.0
    val = m + logsum + pen * sum_diff * sum_diff
    ft = np.convolve(1.0 - h, p, mode='full')
    st = np.convolve(h, p[::-1], mode='full')
    grad = dx * (ft[:n] - st[2 * n - 2:3 * n - 2])
    grad += 2.0 * pen * sum_diff
    return val, grad
