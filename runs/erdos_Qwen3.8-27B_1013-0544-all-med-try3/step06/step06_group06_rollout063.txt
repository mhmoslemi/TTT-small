```python
import numpy as np
import time

def compute_c5_binary(h, n):
    """Compute C5 for a binary (or general) function h of length n."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_fast(h, n):
    """Fast computation using integer arithmetic for binary h."""
    dx = 2.0 / n
    c = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
    return float(np.max(c)) * dx

def binarize(h, n):
    """Convert h to binary, keeping exactly n/2 ones."""
    target = int(n // 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[idx[:target]] = 1
    return h_bin

def simulated_annealing(h_bin, n, deadline, seed=42):
    """Simulated annealing on binary function."""
    rng = np.random.RandomState(seed)
    n_ones = int(n // 2)
    
    # Initial state
    h = h_bin.copy().astype(np.int8)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c = compute_c5_binary(h.astype(float), n)
    current_c = best_c
    
    # Temperature schedule
    T_start = 0.5 * n / n  # scale relative to typical C5 change
    # A single swap changes at most O(n) in the correlation, so C5 changes by at most O(1)
    # Actually, a swap changes overlap_k by at most 2 for each affected k
    # So C5 changes by at most 2 * 2/n = 4/n per swap
    T_start = 4.0 / n * 2  # start a bit high
    T_end = 1e-6
    n_iterations = 2000000  # will be limited by time
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        
        # Pick a random 1 and a random 0
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[a] = 0
        h[b] = 1
        ones = ones[ones != a]
        ones = np.append(ones, b)
        zeros = zeros[zeros != b]
        zeros = np.append(zeros, a)
        
        # Evaluate
        new_c = compute_c5_binary(h.astype(float), n)
        
        # Metropolis criterion
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T_start * (T_end/T_start)**(it/n_iterations), T_end)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            h[a] = 1
            h[b] = 0
            ones = ones[ones != b]
            ones = np.append(ones, a)
            zeros = zeros[zeros != a]
            zeros = np.append(zeros, b)
    
    return best_h, best_c
```

h[i](1-h[j]) - h[i] + h[j] = -h[i]h[j] + h[j] = h[j](1-h[i]) ≥ 0. ✓

So the McCormick constraints for z_{i,j} = h[i] * (1-h[j]) are:
- z ≤ h[i]
- z ≤ 1 - h[j]
- z ≥ 0
- z ≥ h[i] - h[j]

These are all linear! So the LP is:
min C
s.t.
- C ≥ dx * ∑_{valid n} z_{n+k, n} for each shift k
- z_{i,j} ≤ h[i] for all i ≠ j
- z_{i,j} ≤ 1 - h[j] for all i ≠ j
- z_{i,j} ≥ h[i] - h[j] for all i ≠ j
- z_{i,j} ≥ 0 for all i ≠ j
- h[i] ∈ [0,1]
- ∑h = n/2

This IS a linear program! And it's the convex relaxation of the original (non-convex) bilinear problem. The LP optimum is a lower bound on the true minimum.

But here's the key: for this specific problem, the McCormick relaxation might be tight (i.e., the LP optimum is achievable with h[i] ∈ {0,1}). If so, solving the LP gives the exact optimum.

import cvxpy as cp

def solve_erdos_lp(n, solver=cp.GLPK):
    h = cp.Variable(n)
    C = cp.Variable()
    dx = 2.0 / n
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    
    # z_{i,j} for i != j: z = h[i] * (1-h[j])
    # We'll use a 2D variable
    z = cp.Variable((n, n))
    constraints.append(z >= 0)
    constraints.append(z <= h)  # z[i,j] <= h[i] (broadcasting)
    # z[i,j] <= 1 - h[j]
    constraints.append(z <= 1 - h.T)  # need to handle this carefully
    
    # z[i,j] >= h[i] - h[j]
    constraints.append(z >= h[:, None] - h[None, :])
    
    # Diagonal: z[i,i] = h[i](1-h[i]) = 0 for binary, but for continuous it's h[i](1-h[i])
    # We can set z[i,i] = 0 (it's the maximum possible for the McCormick envelope since h[i](1-h[i]) ≤ h[i] and ≤ 1-h[i] and ≥ h[i]-h[i] = 0)
    # Actually for the LP, z[i,i] is free within the constraints, but it doesn't appear in any overlap constraint (k=0 gives overlap=0)
    # So it doesn't matter what z[i,i] is. Let's set it to 0 for simplicity.
    for i in range(n):
        constraints.append(z[i, i] == 0)
    
    # Overlap constraints: C >= dx * sum_{valid n} z[n+k, n]
    for k in range(1, n):  # k from 1 to n-1 (positive shifts)
        # c[k] = sum_{n=0}^{n-1-k} h[n+k](1-h[n]) = sum_{n=0}^{n-1-k} z[n+k, n]
        total = cp.sum(z[k:n, 0:n-k])  # z[n+k, n] for n=0 to n-1-k
        constraints.append(C >= dx * total)
    
    for k in range(1, n):  # k from -(n-1) to -1 (negative shifts)
        # c[-k] = sum_{n=k}^{n-1} h[n-k](1-h[n]) = sum_{n=k}^{n-1} z[n-k, n]
        total = cp.sum(z[0:n-k, k:n])  # z[n-k, n] for n=k to n-1
        constraints.append(C >= dx * total)
    
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    return h.value, C.value

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5_float(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n, dtype=np.int32)
    hb[idx[:target]] = 1
    return hb

def compute_c5_int(h_int, n):
    c = np.correlate(h_int, 1 - h_int, mode='full')
    return float(np.max(c)) * (2.0 / n)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.int32)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1
    return h

def sa_optimize(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5_int(h, n)
    current_c = best_c
    T = 0.05
    T_min = 1e-9
    no_improve = 0
    reheat_at = 10000
    
    while time.time() < deadline:
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b or h[a] == h[b]:
            continue
        h[a], h[b] = h[b], h[a]
        new_c = compute_c5_int(h, n)
        delta = new_c - current_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[a], h[b] = h[b], h[a]
            no_improve += 1
        if no_improve >= reheat_at:
            T = 0.04
            no_improve = 0
        else:
            T = max(T * 0.9999, T_min)
    return best_h, best_c

def solve_erdos_lp(n, solver=None):
    """Solve the LP relaxation of the Erdős overlap problem."""
    h = cp.Variable(n)
    C = cp.Variable()
    z = cp.Variable((n, n))
    dx = 2.0 / n
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
        z >= 0,
    ]
    
    # McCormick: z[i,j] <= h[i], z[i,j] <= 1-h[j], z[i,j] >= h[i] - h[j]
    # Using broadcasting
    h_col = h[:, None]  # shape (n, 1)
    h_row = h[None, :]  # shape (1, n)
    
    constraints.append(z <= h_col)       # z[i,j] <= h[i]
    constraints.append(z <= 1 - h_row)   # z[i,j] <= 1 - h[j]
    constraints.append(z >= h_col - h_row)  # z[i,j] >= h[i] - h[j]
    
    # Diagonal = 0
    for i in range(n):
        constraints.append(z[i, i] == 0)
    
    # Overlap constraints
    for k in range(1, n):
        # overlap_k = sum_{i=0}^{n-1-k} z[k+i, i]
        s = cp.Constant(0)
        for
