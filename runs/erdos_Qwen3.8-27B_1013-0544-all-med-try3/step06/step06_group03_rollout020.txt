def active_lag_search(h, n, deadline, K=50, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(np.max(c_full))
    
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_full = np.correlate(h, 1 - h, mode='full') * dx
            m_full = np.max(c_full)
            
            # Identify active lags
            thresh = m_full - dx * 0.1  # within 0.1*dx of max
            active_idx = np.where(c_full >= thresh)[0]
            if len(active_idx) > K:
                # Keep top K
                top_order = np.argsort(c_full[active_idx])[::-K:]
                active_idx = active_idx[top_order]
            
            # Precompute diff coefficients
            # diff[i, k] = (1-h[i-k]) if valid - h[i+k] if valid
            n_active = len(active_idx)
            lags = active_idx - (n - 1)  # actual lag values
            
            # For efficiency, compute diff for all i and active lags
            diff = np.zeros((n, n_active))
            for ai, lag in enumerate(lags):
                # diff[i, ai] = (1-h[i-lag]) if 0<=i-lag<n - h[i+lag] if 0<=i+lag<n
                # For i range where i-lag is valid: i in [max(0,lag), min(n, n+lag))
                i_lo = max(0, lag)
                i_hi = min(n, n + lag)
                if i_hi > i_lo:
                    diff[i_lo:i_hi, ai] += (1.0 - h[i_lo - lag:i_hi - lag])
                
                i_lo2 = max(0, -lag)
                i_hi2 = min(n, n - lag)
                if i_hi2 > i_lo2:
                    diff[i_lo2:i_hi2, ai] -= h[i_lo2 + lag:i_hi2 + lag]
            
            # Now for each pair (i,j), check if the move helps
            # We'll do a greedy approach: for each i, find best j
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0:
                    continue
                # Find candidates j where h[j] >= d
                candidates = np.where(h >= d)[0]
                if len(candidates) == 0:
                    continue
                # For each candidate, compute new max
                # new_c[ai] = c_full[active_idx[ai]] + d * (diff[i,ai] - diff[j,ai])
                # We want to minimize the max
                base = c_full[active_idx] + d * diff[i, :]
                # For each j, new_max = max(base - d*diff[j,:])
                # Vectorize: for all j at once
                # base[j_idx] - d * diff[j_idx, :] for each j in candidates
                # This is: (base[None,:] - d * diff[candidates,:]).max(axis=1)
                new_vals = base[None, :] - d * diff[candidates, :]
                new_maxes = new_vals.max(axis=1)
                
                # Also need to check that non-active lags don't exceed
                # For safety, check the overall max (approximate)
                best_j_idx = np.argmin(new_maxes)
                best_new_max = new_maxes[best_j_idx]
                
                if best_new_max < m_full - 1e-13:
                    j = candidates[best_j_idx]
                    h[i] += d
                    h[j] -= d
                    # Verify
                    c_new = np.max(np.correlate(h, 1-h, mode='full')) * dx
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
    
    return h, best_c

    active = np.where(c_full >= m - 1e-3)[0]

    lags = active - (n - 1)

            if 0 <= i + lag < n:
                g -= h[i + lag]
            grad[i] += g / len(active)
    
    # Projected step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        s = np.sum(np.exp(sh - mx))
        return float(m + (mx + np.log(s)) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_lag_search(h, n, deadline, K=50, step_sizes=None):
    """Focused local search on active lags."""
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_full = compute_c5_full(h, n)
            m_full = float(np.max(c_full))
            
            # Identify active lags (top K or all within epsilon of max)
            thresh = m_full - dx * 0.05
            active_idx = np.where(c_full >= thresh)[0]
            if len(active_idx) > K:
                top_order = np.argsort(c_full[active_idx])[::-K:]
                active_idx = active_idx[top_order]
            elif len(active_idx) == 0:
                active_idx = np.array([np.argmax(c_full)])
            
            n_active = len(active_idx)
            lags = active_idx - (n - 1)
            
            # Precompute diff[i, ai] for all i and active lags
            diff = np.zeros((n, n_active))
            for ai, lag in enumerate(lags):
                # (1-h[i-lag]) valid when 0 <= i-lag < n
                lo1 = max(0, lag)
                hi1 = min(n, n + lag)
                if hi1 > lo1:
                    diff[lo1:hi1, ai] += (1.0 - h[lo1 - lag:hi1 - lag])
                # -h[i+lag] valid when 0 <= i+lag < n
                lo2 = max(0, -lag)
                hi2 = min(n, n - lag)
                if hi2 > lo2:
                    diff[lo2:hi2, ai] -= h[lo2 + lag:hi2 + lag]
            
            # For each i, find best j to pair with
            # base[ai] = c_full[active_idx[ai]] + d * diff[i, ai]
            # new_max for pair (i,j) = max_ait (c_full[active_idx[ait]] + d*(diff[i,ait] - diff[j,ait]))
            
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0 + 1e-12:
                    continue
                
                base = c_full[active_idx] + d * diff[i, :]
                
                # Candidates: j where h[j] >= d
                candidates = np.where(h >= d - 1e-12)[0]
                # Exclude i itself (or allow it? moving d from i to i is a no-op)
                candidates = candidates[candidates != i]
                if len(candidates) == 0:
                    continue
                
                # Compute new max for all candidates at once
                # Shape: (len(candidates), n_active)
                new_vals = base[None, :] - d * diff[candidates, :]
                new_maxes = new_vals.max(axis=1)
                
                best_j_idx = int(np.argmin(new_maxes))
                best_new_max = new_maxes[best_j_idx]
                
                if best_new_max < m_full - 1e-12:
                    j = int(candidates[best_j_idx])
                    h[i] += d
                    h[j] -= d
                    # Verify with full computation
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def try_resolution(h_start, n, deadline, rng):
    """Optimize at a given resolution."""
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time() - 30
        if remaining < 10:
            break
        ht = slsqp(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    
    # Active-lag local search
    if time.time() < deadline - 15:
        h, best_c = active_lag_search(h, n, deadline - 10, K=50,
                                      step_sizes=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    # Final SLSQP polish
    if time.time() < deadline - 5:
        ht = slsqp(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    
    # Final active-lag search with small steps
    if time.time() < deadline - 2:
        h, best_c = active_lag_search(h, n, deadline - 1, K=30,
                                      step_sizes=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

        if n_try == n_init:

        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(180, (deadline - time.time()) * 0.25)

        if c_opt < best_c:

            best_h = h_opt
    
    # Phase 3: Random restarts from best solution
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(2, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = try_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final intensive optimization on best
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        h_opt, c_opt = try_resolution(best_h, n, deadline - 5, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Ultra-fine active-lag search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = active_lag_search(best_h, n, deadline - 2, K=30,
                                           step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

But c_k(h) = ∑_i h[i](1-h[i-τ]) is not convex in h (it's bilinear/quadratic). So CVXPY won't directly work.

What if we relax it? If we replace c_k(h) with an upper bound that is convex...

h[i](1-h[i-τ]) = h[i] - h[i]*h[i-τ]

By AM-GM: h[i]*h[i-τ] ≤ ((h[i] + h[i-τ])/2)²

So c_k(h) ≥ ∑h[i] - ∑((h[i]+h[i-τ])/2)²

But we need an upper bound, not a lower bound. We want c_k ≤ t.

h[i](1-h[i-τ]) ≤ ? 

This is a product of two terms in [0,1]. The max of h(1-h') for h,h' ∈ [0,1] is 1 (when h=1, h'=0). That's not helpful.

What about: h[i](1-h[i-τ]) ≤ h[i] (since 1-h[i-τ] ≤ 1)
So c_k ≤ ∑_i h[i] over valid range. That's a very loose bound.

Or: h[i](1-h[i-τ]) ≤ (h[i]² + (1-h[i-τ])²)/2 by AM-GM on the product
= (h[i]² + 1 - 2h[i-τ] + h[i-τ]²)/2

This is convex in (h[i], h[i-τ])! So:
c_k ≤ ∑_i (h[i]² + 1 - 2h[i-τ] + h[i-τ]²)/2

We could minimize the max of these upper bounds. But the bound is quite loose, so optimizing it might not give a tight answer.

Actually, one more idea: **What about using the fact that the optimal h is likely to be "nearly binary"?**

If h is binary (0 or 1), then h[i](1-h[i-τ]) = 1 if h[i]=1 and h[i-τ]=0, else 0.
So c_k(τ) = number of i where h[i]=1 and h[i-τ]=0 (in the valid range).

This is a combinatorial problem! We want to choose a subset S of {0,...,n-1} with |S| = n/2 that minimizes:
max_τ |S ∩ (S-τ)^c| = max_τ (|S ∩ [valid range for τ]| - |S ∩ (S+τ)|)

Hmm, this is related to the problem of finding a binary sequence with low autocorrelation.

Actually, for binary h:
c(τ) = ∑_i h[i] * (1-h[i-τ]) over valid i
      = (number of 1s in valid range for τ) - (number of positions i where h[i]=1 and h[i-τ]=1)

The first term depends on τ (boundary effects). The second term is the autocorrelation.

For a "random" binary sequence with n/2 ones, the autocorrelation at lag τ is approximately n/4 (expected value), with standard deviation ~√(n)/4. So c(τ) ≈ n/2 - n/4 = n/4 for the interior, and C₅ ≈ n/4 * (2/n) = 1/2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        s = np.sum(np.exp(sh - mx))
        return float(m + (mx + np.log(s)) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_diff_matrix(h, n, lags):
    """Compute diff[i, ai] = (1-h[i-lag]) if valid - h[i+lag] if valid for each active lag."""
    n_a = len(lags)
    diff = np.zeros((n, n_a))
    for ai in range(n_a):
        lag = lags[ai]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            diff[lo1:hi1, ai] += (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            diff[lo2:hi2, ai] -= h[lo2 + lag:hi2 + lag]
    return diff

def active_lag_search(h, n, deadline, K=50, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_full = compute_c5_full(h, n)
            m_full = float(np.max(c_full))
            thresh = m_full - dx * 0.1
            active_idx = np.where(c_full >= thresh)[0]
            if len(active_idx) > K:
                top_order = np.argsort(c_full[active_idx])[::-K:]
                active_idx = active_idx[top_order]
            elif len(active_idx) == 0:
                active_idx = np.array([int(np.argmax(c_full))])
            lags = (active_idx - (n - 1)).astype(int)
            diff = compute_diff_matrix(h, n, lags)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0 + 1e-12:
                    continue
                base = c_full[active_idx] + d * diff[i, :]
                candidates = np.where(h >= d - 1e-12)[0]
                mask = candidates != i
                if not np.any(mask):
                    continue
                candidates = candidates[mask]
                if len(candidates) == 0:
                    continue
                new_vals = base[None, :] - d * diff[candidates, :]
                new_maxes = new_vals.max(axis=1)
                best_j_idx = int(np.argmin(new_maxes))
                best_new_max = new_maxes[best_j_idx]
                if best_new_max < m_full - 1e-12:
                    j = int(candidates[best_j_idx])
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def line_search_pair(h, n, i, j, c_full, diff_i, diff_j, active_idx, dx):
    """Find optimal step d for pair (i,j) to minimize max over active lags."""
    # c(τ) + d * (diff_i(τ) - diff_j(τ))
    g = diff_i - diff_j
    c_act = c_full[active_idx]
    # Valid range for d
    d_max = min(1.0 - h[i], h[j])
    d_min = max(-h[i], -(1.0 - h[j]))
    if d_max <= d_min:
        return 0.0, compute_c5(h, n)
    # Try to find d that minimizes max(c_act + d*g)
    # This is a piecewise linear convex function in d
    # Try the candidate d that equalizes the two largest c_act values
    # Sort by c_act descending
    order = np.argsort(c_act)[::-1]
    best_d = 0.0
    best_val = float('inf')
    # Check d=0
    best_val = float(np.max(c_act))
    # Try equalizing top pairs
    for a in range(min(3, len(order))):
        for b in range(a + 1, min(5, len(order))):
            ia = order[a]
            ib = order[b]
            if abs(g[ia] - g[ib]) < 1e-15:
                continue
            d_cand = (c_act[ib] - c_act[ia]) / (g[ia] - g[ib])
            d_cand = max(d_min, min(d_max, d_cand))
            val = float(np.max(c_act + d_cand * g))
            if val < best_val:
                best_val = val
                best_d = d_cand
    # Also try small steps
    for d_try in [0.001, 0.005, 0.01, 0.02, 0.05]:
        for sign in [1, -1]:
            d_cand = sign * d_try
            if d_cand < d_min or d_cand > d_max:
                continue
            val = float(np.max(c_act + d_cand * g))
            if val < best_val:
                best_val = val
                best_d = d_cand
    return best_d, best_val

def greedy_line_search(h, n, deadline, K=30, max_rounds=200):
    """Greedy line search: for each i, find best j and optimal step."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        c_full = compute_c5_full(h, n)
        m_full = float(np.max(c_full))
        thresh = m_full - dx * 0.05
        active_idx = np.where(c_full >= thresh)[0]
        if len(active_idx) > K:
            top_order = np.argsort(c_full[active_idx])[::-K:]
            active_idx = active_idx[top_order]
        elif len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c_full))])
        lags = (active_idx - (n - 1)).astype(int)
        diff = compute_diff_matrix(h, n, lags)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            candidates = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            mask = candidates != i
            if not np.any(mask):
                continue
            candidates = candidates[mask]
            if len(candidates) == 0:
                continue
            # For each candidate j, compute the optimal step and resulting max
            # Vectorize: g[j_idx] = diff[i] - diff[j_idx]
            g_matrix = diff[i, :][None, :] - diff[candidates, :]
            c_act = c_full[active_idx]
            # For each candidate, the max at d: max(c_act + d * g)
            # We want to find d that minimizes this
            # Approximate: try d = -(c_act_argmax) / g[argmax] for the argmax lag
            # Or just use the linear search approach
            # For efficiency, try a few d values per candidate
            current_max = m_full
            best_cand_j = -1
            best_cand_d = 0.0
            best_cand_val = current_max
            # Try the candidate that most reduces the current max lag
            max_lag_local = int(np.argmax(c_act))
            g_at_max = g_matrix[:, max_lag_local]
            # We want d * g_at_max < 0 to reduce the max
            # Best j is argmin of g_at_max (most negative, so positive d helps)
            # or argmax (most positive, so negative d helps)
            if np.min(g_at_max) < -1e-10:
                # Use positive d with the j that has most negative g_at_max
                best_j_pos = candidates[int(np.argmin(g_at_max))]
                d_try = min(0.05, 1.0 - h[i], h[int(best_j_pos)])
                g_test = diff[i, :] - diff[best_j_pos, :]
                # Binary search for optimal d
                lo, hi = 0.0, d_try
                for _ in range(10):
                    mid = (lo + hi) / 2
                    val = float(np.max(c_act + mid * g_test))
                    if val > float(np.max(c_act + hi * g_test)):
                        lo = mid
                    else:
                        hi = mid
                d_opt = (lo + hi) / 2
                val_opt = float(np.max(c_act + d_opt * g_test))
                if val_opt < best_cand_val:
                    best_cand_val = val_opt
                    best_cand_j = int(best_j_pos)
                    best_cand_d = d_opt
            if np.max(g_at_max) > 1e-10:
                # Use negative d with the j that has most positive g_at_max
                best_j_neg = candidates[int(np.argmax(g_at_max))]
                d_try = min(0.05, h[i], 1.0 - h[int(best_j_neg)])
                g_test = diff[i, :] - diff[best_j_neg, :]
                lo, hi = -d_try, 0.0
                for _ in range(10):
                    mid = (lo + hi) / 2
                    val = float(np.max(c_act + mid * g_test))
                    if val > float(np.max(c_act + lo * g_test)):
                        hi = mid
                    else:
                        lo = mid
                d_opt = (lo + hi) / 2
                val_opt = float(np.max(c_act + d_opt * g_test))
                if val_opt < best_cand_val:
                    best_cand_val = val_opt
                    best_cand_j = int(best_j_neg)
                    best_cand_d = d_opt
            if best_cand_j >= 0 and best_cand_val < m_full - 1e-12:
                h[i] += best_cand_d
                h[best_cand_j] -= best_cand_d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                    h = project_feasible(h, n)
                else:
                    h[i] -= best_cand_d
                    h[best_cand_j] += best_cand_d
                    h = project_feasible(h, n)
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    # Active-lag local search
    if time.time() < deadline - 15:
        h, best_c = active_lag_search(h, n, deadline - 10, K=50,
                                      step_sizes=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    # Greedy line search
    if time.time() < deadline - 10:
        h, c2 = greedy_line_search(h, n, deadline - 5, K=30, max_rounds=100)
        if c2 < best_c:
            best_c = c2
            h = c2 and h
    # Final SLSQP polish
    if time.time() < deadline - 5:
        ht = slsqp(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    # Ultra-fine active-lag search
    if time.time() < deadline - 2:
        h, best_c = active_lag_search(h, n, deadline - 1, K=30,
                                      step_sizes=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
