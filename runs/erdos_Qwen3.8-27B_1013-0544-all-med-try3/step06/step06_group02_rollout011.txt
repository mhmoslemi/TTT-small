import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    """Compute smooth approximation of max of c5 values"""
    c = compute_c5_all(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    """Gradient of smooth max approximation"""
    c = compute_c5_all(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

            # Quick optimize from perturbed
            h5b, c5b = lbfgs_phase(h5, best_n, time.time() + 15, max_iters=100)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Refine the best solution
    if time.time() < deadline - 100:
        t9 = min(time.time() + 30, deadline - 100)
        h6, c6 = lbfgs_phase(best_h, best_n, t9, max_iters=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    if time.time() < deadline - 60:
        h6, c6 = slsqp_phase(best_h, best_n, time.time() + 20, alpha=5000000, maxiter=800)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final gradient descent
    if time.time() < deadline - 50:
        t10 = min(time.time() + 20, deadline - 50)
        h7, c7 = projected_gradient(best_h, best_n, t10, max_iters=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final pair search (most important for last drop)
    if time.time() < deadline - 10:
        t11 = deadline - 5
        h8, c8 = gradient_pair_search(best_h, best_n, t11, max_rounds=30000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
