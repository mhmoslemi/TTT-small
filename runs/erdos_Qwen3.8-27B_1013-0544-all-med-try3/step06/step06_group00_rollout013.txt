import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_sum_box(v, target_sum, n_iter=80):
    lo = -np.abs(v).max() - 1.0
    hi = np.abs(v).max() + 1.0
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_shift(h, n, s):
    g = np.zeros(n)
    if 0 <= s < n:
        g[:n-s] += (1.0 - h[s:n])
        g[s:] -= h[:n-s]
    elif -n < s <= 0:
        s2 = -s
        if s2 > 0:
            g[s2:] += (1.0 - h[:n-s2])
            g[:n-s2] -= h[s2:]
    return g

def projected_gradient_phase(h, n, deadline, alpha_start=50, alpha_end=500000, n_phases=6):
    """Run projected gradient with increasing alpha (log-sum-exp sharpness)."""
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(h, n)
    best_h = h.copy()
    best_c = corr.max() * dx
    shifts = np.arange(-(n-1), n)
    
    for phase in range(n_phases):
        frac = phase / max(1, n_phases - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        # Step size: for log-sum-exp with sharpness alpha, the function is alpha-Lipschitz gradient
        # So step size ~ 1/alpha * some constant
        lr = 2.0 / (alpha * n)  # conservative
        momentum = 0.85
        v = np.zeros(n)
        
        phase_time = (deadline - time.time()) * 0.8 / (n_phases - phase)
        phase_deadline = time.time() + min(phase_time, 120)
        
        best_phase_c = best_c
        no_improve = 0
        
        while time.time() < phase_deadline:
            corr = compute_corr(h, n)
            shifted = alpha * corr
            max_val = shifted.max()
            exp_shifted = np.exp(shifted - max_val)
            weights = exp_shifted / exp_shifted.sum()
            
            # Compute weighted gradient
            g = np.zeros(n)
            for idx in range(len(shifts)):
                w = weights[idx]
                if w > 1e-8:
                    g += w * grad_for_shift(h, n, shifts[idx])
            
            # Momentum update
            v = momentum * v + g
            h_new = project_sum_box(h - lr * v, target)
            
            c_new = compute_corr(h_new, n).max() * dx
            h = h_new
            if c_new < best_c - 1e-10:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    lr *= 0.5
                    no_improve = 0
                    if lr < 1e-10:
                        break
            
    return best_h, best_c
