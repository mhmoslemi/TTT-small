import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += (d/t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d/t) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_step(h, n, alpha, maxiter=3000):
    bounds = [(0,1)] * n
    def obj(x):
        return smooth_objective(x, n, alpha)
    def cons_fun(x):
        return np.sum(x) - n/2.0
    def cons_jac(x):
        return np.ones(n)
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type':'eq','fun':cons_fun,'jac':cons_jac}],
                   options={'maxiter':maxiter, 'ftol':1e-18})
    return project_feasible(res.x, n)

def subgradient_step(h, n, k_active, step_size):
    """Take one subgradient step for lag k_active."""
    m = k_active - (n - 1)
    # grad[j] = (1-h[j-m])*I(0<=j-m<n) - h[j+m]*I(0<=j+m<n)
    grad = np.zeros(n)
    # Term 1: (1-h[j-m]) for j where 0<=j-m<n, i.e., m<=j<m+n, intersected with [0,n-1]
    j_lo = max(0, m)
    j_hi = min(n, m + n)
    if j_hi > j_lo:
        grad[j_lo:j_hi] += (1 - h[j_lo - m:j_hi - m])
    # Term 2: -h[j+m] for j where 0<=j+m<n, i.e., -m<=j<n-m, intersected with [0,n-1]
    j_lo2 = max(0, -m)
    j_hi2 = min(n, n - m)
    if j_hi2 > j_lo2:
        grad[j_lo2:j_hi2] -= h[j_lo2 + m:j_hi2 + m]
    return grad

    v = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c_vals = corr * dx
        
        # Pick active lag (optimum or random among top few)
        k_star = np.argmax(c_vals)
        
        # Compute subgradient
        grad = subgradient_step(h, n, k_star, 1.0)
        
        # Project to sum constraint: subtract mean
        grad -= grad.mean()
        
        # Nesterov momentum
        v = beta * v + (1 - beta) * grad
        step = gamma / (1 + 0.01 * it)
        
        h_new = h + step * v  # gradient ascent on negative, so plus
        # Wait, we want to MINIMIZE the max, so we go in the direction that DECREASES c_{k*}
        # c_{k*} increases in direction grad, so we go in direction -grad
        h_new = h - step * v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c
