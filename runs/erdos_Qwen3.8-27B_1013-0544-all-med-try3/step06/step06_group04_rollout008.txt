def compute_all_lags(h, n):

def project_feasible(h, n):

            if tot < 1e-15:

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        if tried == 0:
            break
    
    return h, best_c

def outer_approx_step(h_curr, n, trust, top_k=10):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    # Only use top-k lags for the LP (faster)
    top_indices = np.argsort(c)[-top_k:]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    
    for k_idx in top_indices:
        lag = k_idx - (n - 1)
        m = abs(lag)
        # Linearize C(lag) at h_curr: C(lag) ≈ C_h_curr(lag) + grad @ (h - h_curr)
        # C(lag) = sum of h[i]*h[j] type terms, so gradient is:
        if m == 0:
            grad = (1.0 - 2.0 * h_curr) * dx
        elif m >= n:
            continue
        elif lag > 0:
            grad = np.zeros(n)
            grad[:n-m] -= h_curr[m:] * dx
            grad[m:] += (1.0 - h_curr[:n-m]) * dx
        else:
            grad = np.zeros(n)
            grad[:n-m] += (1.0 - h_curr[m:]) * dx
            grad[m:] -= h_curr[:n-m] * dx
        
        cons.append(c[k_idx] + grad @ (h_var - h_curr) <= t_var)
    
    # Also add the zeroth lag constraint
    if 0 not in [k - (n-1) for k in top_indices]:
        c0 = c[n-1]
        grad0 = (1.0 - 2.0 * h_curr) * dx
        cons.append(c0 + grad0 @ (h_var - h_curr) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx_loop(h, n, deadline, max_iters=50):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        if step < 1e-7:
            if trust > 0.001:
                trust *= 0.7
            else:
                break
        else:
            trust = max(0.002, trust * 0.8)
    
    return best_h, best_c

def optimize_at_resolution(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase A: SLSQP coarse
    if time.time() < deadline - 30:
        for alpha in [5000, 50000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase B: Outer approximation
    if time.time() < deadline - 25:
        oo_deadline = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_new, c_new = outer_approx_loop(best_h, n, oo_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase C: Subgradient descent
    if time.time() < deadline - 20:
        sg_deadline = time.time() + min(40, (deadline - time.time()) * 0.25)
        h_new, c_new = subgradient_descent(best_h, n, sg_deadline, top_m=5)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Phase D: Active set SLSQP
    if time.time() < deadline - 15:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 10:
                break
            h_new = active_set_slsqp(best_h, n, top_m, 10000, maxiter=1500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase E: Greedy local search
    if time.time() < deadline - 5:
        ls_deadline = min(deadline - 1, time.time() + min(40, (deadline - time.time()) * 0.4))
        h_new, c_new = greedy_local_search(best_h, n, ls_deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def interpolate_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial resolution
    if h_init is not None:
        alloc = min(120, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try different resolutions
    n_candidates = []
    if h_init is not None:
        n_candidates = [n_init]
    for n_try in [200, 300, 400, 500, 600, 800]:
        if h_init is None or n_try != n_init:
            n_candidates.append(n_try)
    
    # Remove duplicates, sort
    n_candidates = sorted(set(n_candidates))
    
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 40
        if remaining < 60:
            break
        
        if best_h is not None and n_try == len(best_h):
            continue
        
        alloc = min(100, remaining * 0.2)
        if alloc < 40:
            break
        
        if best_h is not None and len(best_h) != n_try:
            h_start = interpolate_to_n(best_h, len(best_h), n_try)
        elif h_init is not None:
            h_start = interpolate_to_n(h_init, n_init, n_try)
        else:
            h_start = np.full(n_try, 0.5)
        
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_resolution(h_start, n_try, time.time() + alloc)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Multiple restarts with perturbation
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 50))
        n_restarts = min(n_restarts, 5)
        
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final intensive local search on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Final SLSQP with high alpha
        if time.time() < deadline - 15:
            h_new = slsqp_optimize(best_h, n, 200000, maxiter=3000, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Final local search
        if time.time() < deadline - 5:
            h_new, c_new = greedy_local_search(best_h, n, deadline - 2)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        
        # Active set polish
        if time.time() < deadline - 5:
            for top_m in [3, 5]:
                if time.time() > deadline - 3:
                    break
                h_new = active_set_slsqp(best_h, n, top_m, 50000, maxiter=1000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    
    # Final projection and computation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
