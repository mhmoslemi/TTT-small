def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

def simulated_annealing(h, n, deadline, t0=0.5, t_end=0.001, cool_rate=0.9995):
    """SA with single-point perturbations and block swaps."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    T = t0
    rng = np.random.RandomState(42)
    
    steps = 0
    while time.time() < deadline and T > t_end:
        # Generate a random move
        move_type = rng.random()
        if move_type < 0.6:
            # Single point change
            j = rng.randint(n)
            delta = rng.uniform(-0.05, 0.05)
            new_val = np.clip(h[j] + delta, 0, 1)
            old_val = h[j]
            h[j] = new_val
            c_new = compute_c5(h, n)
        elif move_type < 0.85:
            # Pair swap
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c_new = compute_c5(h, n)
            old_i, old_j = h[i], h[j]  # This is wrong for undo
        else:
            # Block swap
            bs = rng.choice([2, 3, 4, 5])
            i = rng.randint(0, max(1, n - bs))
            j = rng.randint(0, max(1, n - bs))
            if abs(i - j) < bs:
                continue
            block1 = h[i:i+bs].copy()
            block2 = h[j:j+bs].copy()
            h[i:i+bs] = block2
            h[j:j+bs] = block1
            c_new = compute_c5(h, n)
        
        diff = c_new - current_c
        if diff < 0 or rng.random() < np.exp(-diff / T):
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Undo
            if move_type < 0.6:
                h[j] = old_val
            elif move_type < 0.85:
                h[i], h[j] = h[j], h[i]
            else:
                h[i:i+bs] = block1
                h[j:j+bs] = block2
            c_new = current_c
        
        T *= cool_rate
        steps += 1
    
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

Hmm, that's a problem. But wait - SLSQP might converge in far fewer iterations. And the previous code worked fine at n=300. Let me look at what the previous code did: it used `maxiter=5000` and `ftol=1e-16`. The ftol=1e-16 means it stops when the change in objective is less than 1e-16, which is extremely tight. In practice, it probably converges (or hits maxiter) in a few hundred iterations.

Actually, for the SLSQP to work well, we should provide the gradient of the objective. The gradient of the smooth max:
f(h) = m + (1/α) * log(∑ exp(α*(c_k - m)))

where m = max_k c_k.

∂f/∂h[j] = (1/α) * ∑_k exp(α*(c_k - m)) / ∑_l exp(α*(c_l - m)) * ∂c_k/∂h[j]

= ∑_k softmax_k * ∂c_k/∂h[j]

where softmax_k = exp(α*(c_k - m)) / ∑_l exp(α*(c_l - m))

And ∂c_k/∂h[j] can be computed analytically as I described earlier.

If I provide the analytical gradient, SLSQP only needs 1 function eval + 1 gradient eval per iteration, which is much faster.

def _smooth_max_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    log_sum = mx + np.log(np.sum(exp_shifted))
    val = m + log_sum / alpha

    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            # dc/dh[j] = dx*(1-h[j+lag]) for j=0..n-lag-1  MINUS  dx*h[j-lag] for j=lag..n-1
            grad[:n-lag] += w * dx * (1.0 - x[lag:])
            grad[lag:] -= w * dx * x[:n-lag]
        else:
            m = -lag
            # dc/dh[j] = -dx*h[j+m] for j=0..n-m-1  PLUS  dx*(1-h[j-m]) for j=m..n-1
            grad[:n-m] -= w * dx * x[m:]
            grad[m:] += w * dx * (1.0 - x[:n-m])
    
    return val, grad
