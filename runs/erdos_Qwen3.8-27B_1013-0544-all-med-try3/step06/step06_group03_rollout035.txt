            tot = room.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    """Log-sum-exp smoothed objective for SLSQP/L-BFGS."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def subgradient(h, n, temp=0.01):
    """Compute weighted subgradient of max C_k using softmax weighting."""
    dx = 2.0 / n
    corr = compute_corr(h, n)
    m = corr.max()
    # Softmax weights
    sh = (corr - m) / temp
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    
    n_active = len(corr)
    lags = np.arange(n_active) - (n - 1)
    
    grad = np.zeros(n)
    for idx in np.where(w > 1e-12)[0]:
        lag = lags[idx]
        wk = w[idx] * dx
        if lag > 0:
            # dC/dh[j] = dx*(1-h[j+lag]) for j in [0, n-1-lag]
            #           = -dx*h[j-lag] for j in [lag, n-1]
            #           (both apply when both are valid)
            # For j in [0, n-1-lag]: both terms present
            #   grad[j] += wk * (1 - h[j+lag] - h[j-lag]) if j-lag >= 0
            #   grad[j] += wk * (1 - h[j+lag]) if j-lag < 0
            # For j in [n-lag, n-1]: only second term
            #   grad[j] += -wk * h[j-lag]
            
            # Vectorized:
            # First part: j in [0, n-1-lag], term = wk*(1-h[j+lag])
            grad[:n-lag] += wk
            grad[:n-lag] -= wk * h[lag:]
            # Second part: j in [lag, n-1], term = -wk*h[j-lag]
            grad[lag:] -= wk * h[:n-lag]
        elif lag < 0:
            la = -lag
            # By symmetry, same as lag > 0 case (correlation is symmetric for this problem? No, it's not)
            # Actually for our problem: C_lag = dx * sum_i h[i]*(1-h[i+lag]) for valid i
            # If lag < 0, let la = -lag > 0
            # C_{-la} = dx * sum_i h[i]*(1-h[i-la]) for valid i (0<=i<n, la<=i<n)
            # = dx * sum_i h[i+la]*(1-h[i]) for valid i (0<=i<n-la)
            # dC_{-la}/dh[j]:
            # j appears as h[i+la] when i = j-la (valid if 0<=j-la<n-la, i.e., la<=j<n)
            #   contrib: -dx*h[j-la] * ... wait no
            # j appears as h[i] when i=j (valid if la<=j<n)
            #   contrib: dx*(1-h[j-la])
            # j appears as h[i+la] when i=j-la (valid if 0<=j-la<n-la)
            #   contrib: -dx*h[j] ... no wait
            
            # Let me be more careful.
            # C_{-la} = dx * sum_{i=la}^{n-1} h[i]*(1-h[i-la])
            # dC_{-la}/dh[j]:
            # If j is in "h[i]" position: i=j, valid if la<=j<n
            #   contrib: dx*(1-h[j-la])
            # If j is in "h[i-la]" position: i-la=j, i=j+la, valid if la<=j+la<n, i.e., 0<=j<n-la
            #   contrib: -dx*h[j+la]
            
            # So:
            # For j in [0, n-1-la]: -dx*h[j+la]  (second position)
            # For j in [la, n-1]: dx*(1-h[j-la])  (first position)
            # For j in [la, n-1-la]: both!
            
            grad[:n-la] -= wk * h[la:]
            grad[la:] += wk
            grad[la:] -= wk * h[:n-la]
        else:
            # lag = 0: C_0 = dx*sum h[i]*(1-h[i])
            # dC_0/dh[j] = dx*(1-2*h[j])
            grad += wk * (1 - 2*h)
    
    return grad

def projected_subgrad(h, n, lr, momentum, max_iter, deadline):
    """Projected subgradient descent with momentum."""
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    temp = 0.05
    
    for it in range(max_iter):

        h_new = project_feasible(h_new, n)

        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
        h_curr = h_new
        lr = max(0.001, lr * 0.999)
    
    return best_h, best_c

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _con_sum(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old + 1e-15:
            return h_new, c_new

        pass
    return h.copy(), compute_c5(h, n)

def subgradient(h, n, temp):
    dx = 2.0 / n
    corr = compute_corr(h, n)
    m = corr.max()
    sh = (corr - m) / temp
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w.sum()
    n_lags = len(corr)
    lags = np.arange(n_lags) - (n - 1)
    grad = np.zeros(n)
    for idx in np.where(w > 1e-12)[0]:
        lag = int(lags[idx])
        wk = w[idx] * dx
        if lag > 0:
            grad[:n-lag] += wk
            grad[:n-lag] -= wk * h[lag:]
            grad[lag:] -= wk * h[:n-lag]
        elif lag < 0:
            la = -lag
            grad[:n-la] -= wk * h[la:]
            grad[la:] += wk
            grad[la:] -= wk * h[:n-la]
        else:
            grad += wk * (1.0 - 2.0 * h)
    return grad
