    target = n // 2
    idx = np.argsort(h)[::-1]

    h_b[idx[:target]] = 1.0

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP optimization with log-sum-exp smoothing"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def swap_delta(h, i, j, n, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h)[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h)[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def simulated_annealing(h, n, deadline, seed=42, T_init=None, cool_rate=0.998, patience=80000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = compute_corr(h, n)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    n_ones = len(ones)
    n_zeros = len(zeros)

    if T_init is None:
        T_init = cur_c * 0.12
    T = T_init
    no_improve = 0
    steps = 0

    while time.time() < deadline:
        if n_ones == 0 or n_zeros == 0:
            break

        idx_i = rng.randint(n_zeros)
        idx_j = rng.randint(n_ones)
        i = zeros[idx_i]
        j = ones[idx_j]

        delta = swap_delta(h, i, j, n, m_range)
        new_corr = corr + delta
        new_c = new_corr.max() * dx

        dc = new_c - cur_c
        if dc < -1e-15 or (T > 1e-14 and rng.random() < np.exp(-dc / T)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            zeros[idx_i] = j

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

    if best_n <= 600:
        h_sls = project_feasible(best_h + 0.03 * rng.randn(best_n), best_n)
        for alpha in [500, 5000, 50000, 500000]:
            if time.time() > p4_end - 15:
                break
            h_sls = slsqp_optimize(h_sls, best_n, alpha, 1500, p4_end - 15)
            c_sls = compute_c5(h_sls, best_n)
            if c_sls < best_c:
                best_c = c_sls
                best_h = h_sls.copy()
    else:
        # For large n, do more SA instead
        h_sa2, c_sa2 = simulated_annealing(best_h, best_n, p4_end, seed=seed + 500,
                                           T_init=best_c * 0.05, cool_rate=0.9995)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2.copy()

    # Phase 5: Final SA + greedy (13% of time)
    p5_end = dl
    h_fin, c_fin = simulated_annealing(best_h, best_n, p5_end - 30, seed=seed + 999,
                                       T_init=best_c * 0.04, cool_rate=0.9995,
                                       patience=40000)
    if c_fin < best_c:
        best_c = c_fin
        best_h = h_fin.copy()

    if time.time() < dl - 10:
        h_g2, c_g2 = greedy_local_search(best_h, best_n, dl - 5, max_rounds=200)
        if c_g2 < best_c:
            best_c = c_g2
            best_h = h_g2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    # Ensure exact sum
    if best_n % 2 == 0:
        target = best_n / 2.0
        if abs(best_h.sum() - target) > 1e-10:
            best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
