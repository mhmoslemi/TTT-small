```python
def smooth_max_gradient(h, n, alpha):
    """Compute gradient of (1/alpha) * log(sum(exp(alpha * c_k)))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Soft max
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / exp_c.sum()  # probabilities
    # Gradient
    # grad[j] = sum_k p[k] * d(c_k)/dh[j] * dx
    # d(c_k)/dh[j] = (1 - h[j+k]) - h[j-k] where k = corr_index - (n-1)
    m_vals = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for t in range(2*n-1):
        k = m_vals[t]
        if p[t] < 1e-15:
            continue
        # d(c_t)/dh[j] for all j:
        # = (1 - h[j+k]) - h[j-k] (with boundary)
        # For j+k in [0,n): (1-h[j+k])
        # For j-k in [0,n): -h[j-k]
        dcdh = np.zeros(n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        dcdh[lo1:hi1] += (1 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        dcdh[lo2:hi2] -= h[lo2-k:hi2-k]
        grad += p[t] * dcdh
    grad *= dx
    return grad, corr.max()
```

This is O(n²) per iteration (the loop over t with O(n) work each, but only active p[t] matter). For large alpha, only a few t have significant p[t], so it's actually O(n * |active_set|).

def compute_c5(h, n):

def project_feasible(h, n, max_iter=500):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def build_effect(h):
    n = len(h)
    effect = np.ones((n, 2 * n - 1))
    for t in range(2 * n - 1):
        k = t - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            effect[lo1:hi1, t] -= h[lo1 + k:hi1 + k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            effect[lo2:hi2, t] -= h[lo2 - k:hi2 - k]

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    h = best_h.copy()
    for alpha in [200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if time.time() > p3 - 5:
            break
        h = slsqp_smooth(h, best_n, alpha, 2000, 1e-16)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 4: Annealing with effect matrix (fast)
    p4 = t0 + 0.62 * total
    ha, ca = anneal_binary_effect(best_h, best_n, p4, seed=seed, n_iters=8000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Second anneal from different seed
    if time.time() < p4 + 0.08 * total:
        p4b = p4 + 0.08 * total
        ha2, ca2 = anneal_binary_effect(best_h, best_n, p4b, seed=seed + 1, n_iters=5000000)
        if ca2 < best_c:
            best_c, best_h = ca2, ha2.copy()
    
    # Phase 5: Re-smooth from annealed solution
    p5 = t0 + 0.75 * total
    h = best_h.copy()
    for alpha in [1000, 10000, 100000, 1000000, 10000000]:
        if time.time() > p5 - 5:
            break
        h = slsqp_smooth(h, best_n, alpha, 1500, 1e-16)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Projected gradient polish
    if time.time() < p5 + 0.05 * total:
        p5b = p5 + 0.05 * total
        hpg2, cpg2 = projected_gradient(best_h, best_n, p5b,
                                        alphas=[5000, 20000, 100000, 500000])
        if cpg2 < best_c:
            best_c, best_h = cpg2, hpg2.copy()
    
    # Phase 6: Try other resolutions
    p6 = t0 + 0.88 * total
    for nh in [200, 600, 800]:
        if time.time() > p6 - 10:
            break
        hh = interpolate_to_resolution(best_h, best_n, nh)
        # Binary search at new resolution
        hb_new, cb_new = vectorized_binary_search(hh, nh, p6 - 8, max_sweeps=100)
        if cb_new < compute_c5(hh, nh):
            hh = hb_new
        # Smoothing
        for alpha in [5000, 50000, 500000]:
            if time.time() > p6 - 5:
                break
            hh = slsqp_smooth(hh, nh, alpha, 1000, 1e-16)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh
    
    # Phase 7: Final perturbation restarts
    p7 = dl - 3
    if time.time() < p7:
        rng = np.random.RandomState(seed + 100)
        for trial in range(5):
            if time.time() > p7 - 1:
                break
            noise_scale = 0.003
            h = best_h + noise_scale * rng.randn(best_n)
            h = project_feasible(h, best_n)
            for alpha in [50000, 500000, 5000000]:
                if time.time() > p7 - 0.5:
                    break
                h = slsqp_smooth(h, best_n, alpha, 2000, 1e-18)
            c = compute_c5(h, best_n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
