import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha, penalty):
    """Compute smooth objective and its gradient."""
    dx = 2.0 / n
    one_minus_x = 1.0 - x
    c = np.correlate(x, one_minus_x, mode='full') * dx
    M = np.max(c)
    shifted = alpha * (c - M)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    Z = np.sum(exp_shifted)
    log_sum_exp = max_shifted + np.log(Z)
    
    sum_dev = np.sum(x) - n / 2.0
    obj = M + log_sum_exp / alpha + penalty * sum_dev ** 2
    
    if penalty == 0 and abs(sum_dev) > 1e-6:
        # Just return the smooth obj gradient without penalty
        pass
    
    # Weights
    w = exp_shifted / Z  # shape (2n-1,)
    
    # A[j] = sum_l w_l * x[j+l] (valid)
    # B[j] = sum_l w_l * x[j-l] (valid)
    A = np.correlate(x, w, mode='full')[n-1:2*n-1]
    B = np.correlate(x, w[::-1], mode='full')[n-1:2*n-1]
    
    grad = dx * (1.0 - A - B) + 2.0 * penalty * sum_dev
    return obj, grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter=2000):
    """Run L-BFGS-B optimization."""
    h = h.copy()
    def obj_and_grad(x):
        return smooth_obj_and_grad(x, n, alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_search_fast(h, n, deadline, deltas, radius=100, max_passes=500):
    """Local search: move mass between pairs of coordinates."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def coordinate_descent(h, n, deadline, max_iter=100000):
    """Coordinate descent: optimize one coordinate at a time."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        j = it % n
        # Try moving h[j] to 0 or 1 (or some intermediate value)
        # The effect on c_l: only lags where j is involved change
        # c_l changes by: dx * (h[j]_new - h[j]_old) * (1 - h[j+l]) - dx * h[j+l] * (h[j]_new - h[j]_old) ... 
        # Actually it's easier to just try a few candidate values.
        h_old = h[j]
        # Candidate: 0, 1, or the value that minimizes the max
        candidates = [0.0, 1.0]
        # Also try small adjustments
        for adj in [-0.01, -0.005, 0.005, 0.01]:
            val = h_old + adj
            if 0 <= val <= 1:
                candidates.append(val)
        
        best_val = h_old
        best_c_candidate = best_c
        for val in candidates:
            h[j] = val
            c = compute_c5(h, n)
            if c < best_c_candidate:
                best_c_candidate = c
                best_val = val
        h[j] = best_val
        if best_c_candidate < best_c:
            best_c = best_c_candidate
    return h, best_c
