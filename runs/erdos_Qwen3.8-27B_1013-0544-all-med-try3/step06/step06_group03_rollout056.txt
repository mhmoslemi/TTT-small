    if gnorm < 1e-15:
        return h
    h_new = h - step_size * grad / gnorm
    return project_feasible(h_new, n)

    """Subgradient method with decreasing step sizes."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    c_lb = 0.25  # known lower bound
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Adaptive step size
        step = 2 * (best_c - c_lb) / max(1, it)
        step = min(step, 0.5)  # cap to avoid overshooting
        
        h_new = subgradient_step(h, n, step)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            h = h_new  # still move (subgradient methods don't require descent)
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_passes=20):
    """Coordinate descent with golden section line search."""
    best_c = compute_c5(h, n)
    h = h.copy()
    phi = (1 + np.sqrt(5)) / 2
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Golden section search on h[i] ∈ [0, 1]
            a, b = 0.0, 1.0
            h_old = h[i]
            s = 1.0 / phi
            t = 1.0 - s
            x1 = a + s * (b - a)
            x2 = a + t * (b - a)
            
            h[i] = x1
            f1 = compute_c5(h, n)
            h[i] = x2
            f2 = compute_c5(h, n)
            
            for _ in range(30):  # 30 iterations of golden section
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = a + s * (b - a)
                    h[i] = x1
                    f1 = compute_c5(h, n)
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = a + t * (b - a)
                    h[i] = x2
                    f2 = compute_c5(h, n)
            
            h[i] = (a + b) / 2
            f_new = compute_c5(h, n)
            
            if f_new < best_c - 1e-12:
                best_c = f_new
                improved = True
            else:
                h[i] = h_old  # revert if no improvement
    
        if not improved:
            break
    
    return h, best_c

Actually, for the SA, I can speed things up by noting that when I swap positions i and j (going from h[i]=1, h[j]=0 to h[i]=0, h[j]=1), the change in the correlation can be computed locally:

For the correlation at lag c (using the convention corr[k] = sum h[i+k]*(1-h[i]) for the appropriate range):
- The swap affects terms where either position i or j is involved
- For position i: it appears in lag c terms where i is the "first" position (i.e., i is valid and i is in range) or the "second" position
- Similarly for j

The key insight is that for a binary sequence, h[i]*(1-h[j]) at a given lag is either 0 or 1. When we swap i and j:
- Terms involving i change: previously h[i]=1, now h[i]=0
- Terms involving j change: previously h[j]=0, now h[j]=1

For the correlation at lag c > 0 (index n-1+c in the full output):
corr[c] = sum_{i=0}^{n-1-c} h[i+c] * (1-h[i])

If i is in range [0, n-1-c] (as the "first" index), then the term h[i+c]*(1-h[i]) changes:
- Before: h[i+c] * (1-1) = 0
- After: h[i+c] * (1-0) = h[i+c]
- Change: +h[i+c]

If i is in range [c, n-1] (as the "second" index, i.e., i = j+c for some j), then the term h[i]*(1-h[i-c]) changes:
- Before: 1 * (1-h[i-c]) = 1-h[i-c]
- After: 0 * (1-h[i-c]) = 0
- Change: -(1-h[i-c]) = h[i-c]-1

Similarly for j (which goes from 0 to 1):
If j is in range [0, n-1-c]:
- Before: h[j+c] * (1-0) = h[j+c]
- After: h[j+c] * (1-1) = 0
- Change: -h[j+c]

If j is in range [c, n-1]:
- Before: 0 * (1-h[j-c]) = 0
- After: 1 * (1-h[j-c]) = 1-h[j-c]
- Change: +(1-h[j-c])

So the total change in corr at lag c is:
Δ_c = [contribution from i as "first"] + [contribution from i as "second"] + [contribution from j as "first"] + [contribution from j as "second"]

This can be computed in O(1) per lag, and there are O(n) lags, so O(n) total per swap. That's faster than O(n²) for the full correlation!

    """Create binary sequence from pattern, truncated/padded to n with exactly n_ones ones."""
    h = np.zeros(n)
    if len(pattern) >= n:
        h = pattern[:n].copy()
    else:
        # Repeat and truncate
        reps = n // len(pattern) + 1
        h = np.tile(pattern, reps)[:n]
    # Adjust to have exactly n_ones ones
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]

        # Remove some ones

        h[ones_idx[perm]] = 0

        need = n_ones - len(ones_idx)
        perm = np.random.permutation(len(zeros_idx))[:need]
        h[zeros_idx[perm]] = 1
    return h

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def sa_binary(h_start, n, deadline, T_start=0.3, T_end=0.003, reheat_every=200000):
    h = (h_start >= 0.5).astype(np.float64)
    target = n // 2
    ones_count = int(h.sum())
    if ones_count > target:
        ones_idx = np.where(h == 1)[0]
        np.random.shuffle(ones_idx)
        h[ones_idx[:ones_count - target]] = 0
    elif ones_count < target:
        zeros_idx = np.where(h == 0)[0]
        np.random.shuffle(zeros_idx)
        h[zeros_idx[:target - ones_count]] = 1
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    no_improve = 0
    total_target = 2000000
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Cooling schedule with reheat
        progress = min(it, total_target) / total_target
        T = T_start * (T_end / T_start) ** progress
        
        # Reheat if stuck
        if no_improve > reheat_every:
            T = max(T, T_start * 0.5)
            no_improve = 0
        
        i_idx = np.random.randint(len(ones_list))
        j_idx = np.random.randint(len(zeros_list))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - curr_c
        
        if delta <= 0 or np.random.random() < np.exp(-delta / T):
            curr_c = new_c
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return best_h, best_c

def coord_descent(h, n, deadline, max_passes=100, tol=1e-10):
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            # Golden section search
            ratio = 0.6180339887498949
            a, b = 0.0, 1.0
            c = b - ratio * (b - a)
            d = a + ratio * (b - a)
            h[i] = c
            fc = compute_c5(h, n)
            h[i] = d
            fd = compute_c5(h, n)
            
            for _ in range(40):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - ratio * (b - a)
                    h[i] = c
                    fc = compute_c5(h, n)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + ratio * (b - a)
                    h[i] = d
                    fd = compute_c5(h, n)
            
            h[i] = (a + b) / 2
            f_final = compute_c5(h, n)
            
            if f_final < best_c - 1e-13:
                best_c = f_final
                pass_improved = True
            else:
                h[i] = old_val
        
        if not pass_improved:
            break
    
    return h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        # Log-sum-exp for smooth max
        sh = (corr - m) * alpha
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = (corr - m) * alpha
        mx = sh.max()
        weights = np.exp(sh - mx)
        weights /= weights.sum()
        # Gradient of log-sum-exp part
        # d/dh[j] of corr[k] for each k
        # This is complex; use numerical gradient or just use SLSQP's internal
        return None  # Let SLSQP use numerical gradients
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    
    return h, best_c
