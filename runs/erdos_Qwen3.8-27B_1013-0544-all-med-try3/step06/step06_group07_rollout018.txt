   ```python
   import numpy as np
   from scipy.optimize import minimize, linprog, Bounds
   import time

   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)

   def project_feasible(h, target_sum):
       h = h.copy()
       h = np.clip(h, 0.0, 1.0)
       for _ in range(200):
           s = h.sum()
           if abs(s - target_sum) < 1e-14:
               break
           diff = target_sum - s
           if diff > 0:
               mask = h < 1.0 - 1e-15
           else:
               mask = h > 1e-15
           cnt = mask.sum()
           if cnt == 0:
               break
           h[mask] += diff / cnt
           h = np.clip(h, 0.0, 1.0)
       # Final adjustment
       s = h.sum()
       diff = target_sum - s
       if abs(diff) > 1e-14:
           mask = (h > 1e-10) & (h < 1.0 - 1e-10)
           if mask.sum() > 0:
               h[mask] += diff / mask.sum()
               h = np.clip(h, 0.0, 1.0)
       return h

   def resample_h(h_old, n_new):
       x_old = np.linspace(0, 2, len(h_old), endpoint=False)
       x_new = np.linspace(0, 2, n_new, endpoint=False)
       # Linear interpolation
       result = np.interp(x_new, x_old, h_old)
       return result

   def compute_alpha(h_curr):
       n = len(h_curr)
       num_lags = 2 * n - 1
       k = np.arange(-(n-1), n, dtype=np.float64)  # shape (2n-1,)
       i = np.arange(n, dtype=np.float64)  # shape (n,)
       
       i_minus_k = i[:, None] - k[None, :]  # shape (n, 2n-1)
       i_plus_k = i[:, None] + k[None, :]  # shape (n, 2n-1)
       
       mask1 = (i_minus_k >= 0) & (i_minus_k < n)
       mask2 = (i_plus_k >= 0) & (i_plus_k < n)
       
       # Clip indices for safe indexing
       imk = np.clip(i_minus_k.astype(int), 0, n-1)
       ipk = np.clip(i_plus_k.astype(int), 0, n-1)
       
       term1 = np.where(mask1, 1.0 - h_curr[imk], 0.0)
       term2 = np.where(mask2, -h_curr[ipk], 0.0)
       
       alpha = (term1 + term2).T  # shape (2n-1, n)
       return alpha

   def slp_iteration(h_curr, target_sum, dx):
       """One SLP iteration: solve the LP relaxation."""
       n = len(h_curr)
       corr_curr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
       alpha = compute_alpha(h_curr)
       
       b_ub = alpha @ h_curr - corr_curr
       
       A_ub = np.hstack([alpha, -np.ones((2*n-1, 1)) / dx])
       A_eq = np.zeros((1, n+1))
       A_eq[0, :n] = 1.0
       b_eq = np.array([target_sum])
       
       c_obj = np.zeros(n+1)
       c_obj[n] = 1.0
       
       bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
       
       try:
           result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds_lp, method='highs',
                           options={'disp': False, 'presolve': True})
           if result.success:
               h_new = result.x[:n]
               return h_new, result.x[n]
       except Exception:
           pass
       return h_curr.copy(), None

   def smooth_max(x, beta):
       m = np.max(x)
       shifted = np.clip(beta * (x - m), -700.0, 700.0)
       return m + np.log(np.sum(np.exp(shifted))) / beta

   def obj_smooth(h, dx, beta):
       corr = np.correlate(h, 1.0 - h, mode='full')
       return smooth_max(corr, beta) * dx

                           best_c5 = c5_try
                       stalled = 0
   ```
