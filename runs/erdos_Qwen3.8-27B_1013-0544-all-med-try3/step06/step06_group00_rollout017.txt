    c = np.correlate(h, 1-h, mode='full')
    # Compute softmax weights
    z = alpha * c * dx
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    W = exp_z / exp_z.sum()  # length 2n-1
    
    # Compute gradient directly
    # grad[j] = dx * sum_s W_s * [(1-h[j+s])*1_{valid} - h[j-s]*1_{valid}]

        if Ws < 1e-15:

        # First term: (1-h[j+s]) for valid j+s
        if s >= 0:
            grad[:n-s] += Ws * (1-h[s:])
        else:
            grad[-s:] += Ws * (1-h[:n+s])

        if s >= 0:
            grad[s:] -= Ws * h[:n-s]
        else:
            grad[:n+s] -= Ws * h[-s:]
    grad *= dx
    return grad, W, c

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            h += diff / s * room

            s = h.sum()
            if s < 1e-14:
