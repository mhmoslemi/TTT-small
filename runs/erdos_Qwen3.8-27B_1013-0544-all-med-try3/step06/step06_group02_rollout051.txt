import time

def compute_c5_values(h, n):
    """Returns array of all correlation values (length 2n-1)"""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_values(h, n)))

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_for_pair(h, n, i, j):
    """Compute d_m = ∂c_m/∂h_i - ∂c_m/∂h_j for all m"""
    # c_m = ∑_a h[a] * (1-h[a+p]) where p = m-(n-1)
    # ∂c_m/∂h[x] = (1-h[x+p])*[x+p valid] - h[x-p]*[x-p valid]
    # Let's compute this for x=i and x=j
    
    d = np.zeros(2*n-1)
    p = np.arange(2*n-1) - (n-1)  # p[m] = m-(n-1), ranges from -(n-1) to n-1

    p_valid_minus = (i - p >= 0) & (i - p < n)
    grad_i = np.zeros(2*n-1)
    grad_i[p_valid_plus] = 1.0 - h[i + p[p_valid_plus]]
    grad_i[p_valid_minus] -= h[i - p[p_valid_minus]]

    grad_j[p_valid_minus_j] -= h[j - p[p_valid_minus_j]]
    
    return grad_i - grad_j

    return np.correlate(h, 1.0 - h, mode='full') * dx

                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_matrix(h, n):
    """Compute G[i, m] = ∂c_m/∂h_i for all i, m"""
    p = np.arange(-(n-1), n)  # shape (2n-1,)
    I = np.arange(n)[:, None]  # shape (n, 1)
    P = p[None, :]  # shape (1, 2n-1)
    ip = I + P  # i + p
    im = I - P  # i - p
    valid_plus = (ip >= 0) & (ip < n)
    valid_minus = (im >= 0) & (im < n)
    # Clip for indexing
    ip_clip = np.clip(ip, 0, n-1)
    im_clip = np.clip(im, 0, n-1)
    G = np.where(valid_plus, 1.0 - h[ip_clip], 0.0) - np.where(valid_minus, h[im_clip], 0.0)
    return G

def smooth_max_obj(x, n, alpha):
    c = compute_c5_arr(x, n)
    m = c.max()
    s = alpha * (c - m)
    return m + np.log(np.sum(np.exp(s - s.max())) + 1e-30) / alpha

def smooth_max_grad(x, n, alpha):
    c = compute_c5_arr(x, n)
    m = c.max()
    s = alpha * (c - m)
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w /= w.sum()
    G = compute_gradient_matrix(x, n)
    # grad_j = sum_m w[m] * G[j, m]
    grad = G.T @ w
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [100, 1000, 10000, 100000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha)
        def grad(x):
            return smooth_max_grad(x, n, alpha)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except:
            pass
    return h, best_c

def lp_iteration(h, n, M_active=30):
    """One iteration of the linearized LP method."""
    c = compute_c5_arr(h, n)
    C = c.max()
    
    # Get active shifts (top M)
    idx = np.argsort(c)[::-1][:M_active]
    
    # Compute gradient matrix
    G = compute_gradient_matrix(h, n)
    
    # Build LP: minimize t
    # s.t. c_m + G[:, m] @ (h_new - h) <= t  for m in active
    #      0 <= h_new <= 1
    #      sum(h_new) = n/2
    #
    # Variables: h_new (n vars) + t (1 var) = n+1 vars
    #
    # For each active m:
    #   G[:, m] @ h_new - t <= c[m] - G[:, m] @ h
    #
    # c[m] - G[:, m] @ h = c[m] - c[m] = 0? Let me check.
    # Actually c[m] = sum_i h[i] * (1-h[i+p_m]) and G[:, m] = ∂c_m/∂h
    # G[:, m] @ h is NOT equal to c[m] in general (c is bilinear, not linear)
    
    A_ub = np.zeros((M_active, n + 1))
    b_ub = np.zeros(M_active)
    for i_m, m in enumerate(idx):
        A_ub[i_m, :n] = G[:, m]
        A_ub[i_m, n] = -1.0  # coefficient of t
        b_ub[i_m] = c[m] - G[:, m] @ h  # = 0? Let's verify
    
    # Actually: c[m] ≈ c[m] + G[:,m] @ (h_new - h) = G[:,m] @ h_new - G[:,m] @ h + c[m]
    # So: G[:,m] @ h_new - t <= c[m] - G[:,m] @ h
    # And c[m] - G[:,m] @ h is NOT zero (since c is bilinear)
    # Let's just compute it numerically
    
    # Hmm, let me reconsider. The first-order Taylor approximation:
    # c_m(h_new) ≈ c_m(h) + ∇c_m(h) · (h_new - h)
    # = c_m(h) + G[:, m] @ (h_new - h)
    # We want: c_m(h) + G[:, m] @ (h_new - h) <= t
    # => G[:, m] @ h_new - t <= t - c_m(h) + ... wait
    # => G[:, m] @ h_new - t <= c_m(h) - G[:, m] @ h
    # RHS = c_m(h) - G[:, m] @ h
    
    # Let me just compute this:
    # G[:, m] @ h = sum_i G[i, m] * h[i]
    # For m = n-1 (p=0): G[i, n-1] = (1-h[i]) - h[i] = 1-2h[i]
    # G[:, n-1] @ h = sum_i (1-2h[i])*h[i] = sum_i h[i] - 2*sum_i h[i]^2
    # c[n-1] = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]^2
    # So G[:, n-1] @ h = sum h - 2*sum h^2, c[n-1] = sum h - sum h^2
    # c[n-1] - G[:, n-1] @ h = sum h^2 > 0. OK, not zero.
    
    # This is fine. The LP is:
    # min t
    # s.t. G[:, m] @ h_new - t <= c[m] - G[:, m] @ h  (for active m)
    #      0 <= h_new <= 1
    #      sum h_new = n/2
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(C - 0.1, C + 0.1)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_new = np.clip(res.x[:n], 0, 1)
            h_new = project_feasible(h_new, n)
            return h_new
    except:
        pass
    return h

def lp_optimize(h, n, deadline, M_active=30, max_iters=500):
    """Iterate the LP method with line search."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        h_lp = lp_iteration(h, n, M_active)
        
        # Line search: try h = (1-t)*h + t*h_lp for t in [0,1]
        best_t = 0
        best_c_try = best_c
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible((1-t)*h + t*h_lp, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_try:
                best_c_try = c_try
                best_t = t
        
        if best_t > 0:
            h = project_feasible((1-best_t)*h + best_t*h_lp, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            # No improvement, try smaller M_active
            break
    
    return best_h, best_c

def pair_descent(h, n, deadline, n_pairs_per_sweep=500, ternary_iters=40):
    """Exact 1D pair coordinate descent."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = compute_c5_arr(h, n)
    
    while time.time() < deadline:
        # Recompute c and G
        c = compute_c5_arr(h, n)
        current_max = c.max()
        G = compute_gradient_matrix(h, n)
        
        # Generate candidate pairs
        # Strategy: use gradient to identify good pairs
        # Active set
        M = min(30, 2*n-1)
        active_idx = np.argsort(c)[::-1][:M]
        
        # Subgradient (uniform over active set)
        s = G[:, active_idx].mean(axis=1)  # shape (n,)
        s_centered = s - s.mean()
        
        # Donors: high s (decrease h), Receivers: low s (increase h)
        donors = np.argsort(s)[::-1][:30]  # top 30 by s
        receivers = np.argsort(s)[:30]  # bottom 30 by s
        
        # Generate pairs
        pairs = []
        for d in donors:
            for r in receivers:
                if d != r:
                    pairs.append((d, r))
        
        # Add some random pairs
        rng = np.random.randint(0, n, size=(100, 2))
        for p in rng:
            if p[0] != p[1]:
                pairs.append((p[0], p[1]))
        
        # Add local pairs
        for i in range(0, n, 5):  # every 5th
            for offset in [1, 2, 3, -1, -2, -3]:
                j = i + offset
                if 0 <= j < n:
                    pairs.append((i, j))
        
        # Shuffle
        np.random.shuffle(pairs)
        
        # Limit pairs
        if len(pairs) > n_pairs_per_sweep:
            pairs = pairs[:n_pairs_per_sweep]
        
        improved = False
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            
            # Optimal: increase h_j, decrease h_i (based on subgradient)
            # Or vice versa. Try both directions.
            
            d = G[i] - G[j]  # ∂c/∂h_i - ∂c/∂h_j
            
            # Direction 1: h_i decreases by δ, h_j increases by δ
            # δ ∈ [0, min(h_i, 1-h_j)]
            # New c: c + (-δ) * G[i] ... wait
            
            # If h_i -= δ and h_j += δ:
            # Δc_m = -δ * ∂c_m/∂h_i + δ * ∂c_m/∂h_j = δ * (G[j,m] - G[i,m]) = -δ * d[m]
            # New c_m = c[m] - δ * d[m]
            # Minimize max_m (c[m] - δ*d[m]) for δ ∈ [0, min(h_i, 1-h_j)]
            
            # Direction 2: h_i += δ and h_j -= δ
            # Δc_m = δ * G[i,m] - δ * G[j,m] = δ * d[m]
            # Minimize max_m (c[m] + δ*d[m]) for δ ∈ [0, min(1-h_i, h_j)]
            
            # Solve both and pick the better one
            
            # Direction 1: minimize max(c - δ*d), δ ∈ [0, δ_max1]
            δ_max1 = min(h[i], 1 - h[j])
            if δ_max1 > 1e-12:
                # Ternary search for δ*
                a, b = 0.0, δ_max1
                for _ in range(ternary_iters):
                    m1 = a + (b-a)/3
                    m2 = b - (b-a)/3
                    f1 = np.max(c - m1 * d)
                    f2 = np.max(c - m2 * d)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                δ1 = (a+b)/2
                c1 = compute_c5_arr(h, n)  # just to be safe, recompute
                # Actually let me apply and check
                h[i] -= δ1; h[j] += δ1
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    c = compute_c5_arr(h, n)
                    G = compute_gradient_matrix(h, n)
                    improved = True
                else:
                    h[i] += δ1; h[j] -= δ1  # undo
            
            # Direction 2: minimize max(c + δ*d), δ ∈ [0, δ_max2]
            δ_max2 = min(1 - h[i], h[j])
            if δ_max2 > 1e-12:
                a, b = 0.0, δ_max2
                for _ in range(ternary_iters):
                    m1 = a + (b-a)/3
                    m2 = b - (b-a)/3
                    f1 = np.max(c + m1 * d)
                    f2 = np.max(c + m2 * d)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                δ2 = (a+b)/2
                h[i] += δ2; h[j] -= δ2
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    c = compute_c5_arr(h, n)
                    G = compute_gradient_matrix(h, n)
