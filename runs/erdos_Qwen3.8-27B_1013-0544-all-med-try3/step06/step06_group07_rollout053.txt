def solve_1d(base, coeff, m_nminus1, S, target, lam):
    """Solve min_{v in [0,1]} max_m c[m](v) + lam*(S+v-target)^2"""
    def F(v):

        vals[m_nminus1] = base[m_nminus1] + v - v*v
        return np.max(vals) + lam * (S + v - target)**2
    
    # Golden section search
    res = minimize_scalar(F, bounds=(0, 1), method='bounded', options={'xatol': 1e-12})
    v_best = res.x

            #   For m = n-1: c[m] = c_base[m] + h[i] - h[i]^2  (since h[i]*(1-h[i]) = h[i] - h[i]^2)
            #     But c_one[n-1] - c_base[n-1] = 1 - 0 = 1 (since h[i]*(1-h[i]) at h[i]=1 is 0, at h[i]=0 is 0)
            #     Hmm, that gives 0. Let me reconsider.
            
            # When h[i] = 0: contribution to c[n-1] is 0*(1-0) = 0
            # When h[i] = 1: contribution to c[n-1] is 1*(1-1) = 0
            # So c_base[n-1] and c_one[n-1] have the SAME contribution from position i (which is 0).
            # The actual contribution for h[i] = v is v*(1-v) = v - v^2.
            # So c[n-1] = c_base[n-1] + v - v^2 (where c_base has h[i]=0, contributing 0)
            
            # For m != n-1:
            # When h[i] = 0: contribution is 0
            # When h[i] = 1: contribution is (1 - h[i+k]) + (-h[i-k]) (from the two cases)
            #   = c_one[m] - c_base[m]
            # For h[i] = v: contribution is v * ((1 - h[i+k]) - h[i-k]) (linear!)
            # So c[m] = c_base[m] + v * (c_one[m] - c_base[m])
            
            # Great! So the algorithm is:
            # 1. Save h[i], compute c0 (with h[i]=0) and c1 (with h[i]=1)
            # 2. For m != n-1: c[m] = c0[m] + v*(c1[m]-c0[m])
            # 3. For m = n-1: c[m] = c0[m] + v - v^2
            # 4. Minimize max(c) + penalty over v ∈ [0,1]
            
            # But this requires TWO correlate computations per coordinate, which is expensive.
            # Alternatively, I can compute base and coeff analytically.
            
            # Let me compute analytically:
            k_vals = np.arange(2*n-1) - (n-1)  # lag values: -(n-1), ..., n-1
            # For coordinate i, the contribution to c at lag k is:
            #   (1 - h[i+k]) if 0 <= i+k < n  (h[i] appears as the first factor)
            #   - h[i-k] if 0 <= i-k < n  (h[i] appears in (1-h[.]) as the second factor)
            #   BUT for k=0: the contribution is (1-h[i]) - 0 = 1-h[i] from first case,
            #     and -h[i] from second case. Total: 1 - h[i] - h[i] = 1 - 2*h[i]?
            #     No wait. For k=0: t=i gives h[i]*(1-h[i]). The "base" (h[i]=0) gives 0.
            #     The "one" (h[i]=1) gives 1*0 = 0.
            #     So c0[n-1] already has the correct base (it's the sum of all OTHER terms).
            #     And the contribution of h[i] to c[n-1] is h[i]*(1-h[i]) = h[i] - h[i]^2.
            
            # For k ≠ 0:
            #   If 0 <= i+k < n: h[i] * (1-h[i+k]) → linear in h[i] with coeff (1-h[i+k])
            #   If 0 <= i-k < n: (1-h[i]) * h[i-k] → linear in h[i] with coeff (-h[i-k])
            #   Total coeff: (1-h[i+k])*[valid] + (-h[i-k])*[valid]
            #   Base: c[m] - h[i]*coeff = rest of the sum
            
            # Let me just compute c[m] for the current h[i], then subtract h[i]*coeff to get base.
            
            # Actually, the simplest and most robust approach:
            # Compute c = np.correlate(h, 1-h, 'full') for the current h.
            # For coordinate i:
            #   For m != n-1: coeff[m] = (1-h[(i+k)%n if valid]) ... 
            #   This is getting complicated with boundary conditions.
            
            # Let me just use the two-evaluation approach but vectorize across coordinates.
            # Actually, for n=70 and ~10000 coordinates to update, that's 20000 correlate calls.
            # Each correlate is O(n^2) = O(4900). Total: 100 million operations. Should be OK.
            
            # But wait, I don't need to do this for every coordinate every cycle.
            # I can compute base and coeff once per cycle (for all coordinates) and reuse.
            # But base and coeff change every time I update a coordinate...
            
            # OK let me just go with the two-evaluation approach per coordinate update.
            # For n=70, correlate is fast (O(n^2) = 4900 ops). Two per coordinate = 9800 ops.
            # 70 coordinates per cycle = 686,000 ops per cycle.
            # 10,000 cycles = 6.86 billion ops. At ~1 Gops in numpy, that's ~7 seconds.
            # Actually numpy correlate for n=70 is probably ~1μs (it's highly optimized).
            # So 2 * 70 * 10000 = 1.4 million correlate calls, each ~1μs = 1.4 seconds. Fine!
            
            hi_save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hi_save
            
            # For m != n-1: c[m](v) = c0[m] + v * (c1[m] - c0[m])
            # For m = n-1: c[m](v) = c0[m] + v - v^2
            diff = c1 - c0  # this is the "slope" for linear parts
            
            # The 1D objective:
            # F(v) = max(c0 + v*diff, with c0[n-1] + v - v^2 replacing c0[n-1] + v*diff[n-1])
            #       + lam * (S + v - target)^2
            # Note: diff[n-1] should be 0 (as we discussed), so c0[n-1] + v*diff[n-1] = c0[n-1]
            #       and the actual value is c0[n-1] + v - v^2
            
            # Let's verify: diff[n-1] = c1[n-1] - c0[n-1]
            # c1[n-1] = sum_j h_j(1-h_j) with h[i]=1 → includes 1*0=0 from position i
            # c0[n-1] = sum_j h_j(1-h_j) with h[i]=0 → includes 0*1=0 from position i
            # So diff[n-1] = 0. ✓
            
            # Now minimize F(v) = max( [c0[m] + v*diff[m] for m≠n-1], c0[n-1] + v - v^2 ) + lam*(S+v-target)^2
            # over v ∈ [0,1]
            
            # Use minimize_scalar
            S_rest = h.sum() - hi_save  # S = sum of all other h[j]
            
            def obj_1d(v):
                vals = c0 + v * diff
                vals[m_np1] = c0[m_np1] + v - v * v
                return np.max(vals) + lam * (S_rest + v - target) ** 2
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
            h[i] = res.x
    
    return h

import numpy as np
from scipy.optimize import minimize_scalar, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta, target, lam):
    corr = np.correlate(h, 1.0 - h, mode='full')
    pen = lam * (h.sum() - target) ** 2
    return smooth_max(corr, beta) * dx + pen

def coord_descent(h, n, lam, t0, max_time):
    target = n / 2.0
    m_np1 = n - 1
    h = h.copy()
    
    while time.time() - t0 < max_time:
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() - t0 > max_time:
                return h
            hi = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hi
            diff = c1 - c0
            S_rest = h.sum() - hi
            
            # 1D minimization
            def obj_1d(v):
                vals = c0 + v * diff
                vals[m_np1] = c0[m_np1] + v - v * v
                return np.max(vals) + lam * (S_rest + v - target) ** 2
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            h[i] = res.x
    return h
