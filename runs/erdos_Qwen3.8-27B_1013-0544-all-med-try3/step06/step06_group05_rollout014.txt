import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, penalty):
    """Objective and gradient for L-BFGS-B."""
    dx = 2.0 / n
    target_sum = n / 2.0
    
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    w = exp_c / Z  # softmax weights, sum to 1
    
    obj = (c_max + np.log(Z) / alpha) + penalty * (h.sum() - target_sum)**2
    
    # Gradient of soft max part
    # For each shift m, s_m = m - (n-1)
    # d(c_m)/d(h[j]) = dx * [(1 - h[j-s_m]) * 1[0<=j-s_m<n] - h[j+s_m] * 1[0<=j+s_m<n]]
    
    grad_sm = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)  # shift, ranges from -(n-1) to n-1
        wm = w[m]
        if wm < 1e-15:
            continue
        
        # Term 1: (1 - h[j-s]) for j where 0 <= j-s < n
        # j ranges from max(0, s) to min(n-1, s+n-1) = n-1 (since s < n)
        # Actually: 0 <= j-s < n means s <= j < s+n, intersected with [0, n-1]
        # So j from max(0, s) to min(n-1, s+n-1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)  # exclusive
        if lo1 < hi1:
            # indices in h for the "1-h" part: j-s, ranging from lo1-s to hi1-1-s
            idx_h = np.arange(lo1, hi1) - s
            grad_sm[lo1:hi1] += wm * dx * (1.0 - h[idx_h])
        
        # Term 2: -h[j+s] for j where 0 <= j+s < n
        # j ranges from max(0, -s) to min(n-1, n-1-s)
        lo2 = max(0, -s)
        hi2 = min(n, n - s)  # exclusive
        if lo2 < hi2:
            idx_h = np.arange(lo2, hi2) + s
            grad_sm[lo2:hi2] -= wm * dx * h[idx_h]
    
    # Add penalty gradient
    grad = grad_sm + 2.0 * penalty * (h.sum() - target_sum)
    
    return obj, grad

def lbfgsb_optimize(h, n, alpha, penalty, maxiter=2000, tol=1e-12):
    """Optimize using L-BFGS-B."""
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        obj, _ = smooth_max_grad(x, n, alpha, penalty)
        return obj
    
    def jac(x):
        _, g = smooth_max_grad(x, n, alpha, penalty)
        return g
    
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    if verbose:
        print(f"  n={n}, initial C5={best_c:.6f}")
    
    # Phase 1: L-BFGS-B with increasing alpha
    alphas = [200, 500, 1000, 3000, 10000, 30000, 100000]
    for a in alphas:
        if time.time() > deadline - 10:

        # Adaptive penalty
        pen = 100.0
        h_new = lbfgsb_optimize(best_h, n, a, pen, maxiter=1000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            if verbose:
                print(f"    alpha={a}: C5={c_new:.6f}")
    
    # Phase 2: Near pair search
    if time.time() < deadline - 15:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 15,

        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 15,
                                              [0.01, 0.005, 0.002], radius=40)
    
    # Phase 3: Final L-BFGS-B polish
    if time.time() < deadline - 10:

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 4: Final fine pair search
    if time.time() < deadline - 3:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], max_passes=300)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    target_sum = n / 2.0
    L = n
    
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    w = exp_c / Z
    
    obj = (c_max + np.log(Z) / alpha) + penalty * (h.sum() - target_sum)**2
    
    # Gradient
    grad = np.zeros(L)
    for m in range(2 * L - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        lag = m - (L - 1)
        
        # Term 1: (1 - h[j-lag]) for j where 0 <= j-lag < L
        # j-lag >= 0 => j >= lag; j-lag < L => j < lag + L
        # Combined with 0 <= j < L: max(0,lag) <= j < min(L, lag+L)
        lo1 = max(0, lag)
        hi1 = min(L, lag + L)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) - lag
            grad[lo1:hi1] += wm * dx * (1.0 - h[idx])
        
        # Term 2: -h[j+lag] for j where 0 <= j+lag < L
        # j+lag >= 0 => j >= -lag; j+lag < L => j < L-lag
        # Combined: max(0,-lag) <= j < min(L, L-lag)
        lo2 = max(0, -lag)
        hi2 = min(L, L - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) + lag
            grad[lo2:hi2] -= wm * dx * h[idx]
    
    grad += 2.0 * penalty * (h.sum() - target_sum)
    
    return obj, grad

def lbfgsb_step(h, n, alpha, penalty, maxiter=1500):
    def fun(x):
        o, _ = smooth_max_and_grad(x, n, alpha, penalty)
        return o
    def jac(x):
        _, g = smooth_max_and_grad(x, n, alpha, penalty)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # L-BFGS-B with increasing alpha
    alphas = [100, 300, 1000, 3000, 10000, 30000, 50000, 100000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        pn = 50.0
        h_new = lbfgsb_step(best_h, n, a, pn, maxiter=1000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new
    
    # Near pair search (or full for small n)
    remaining = deadline - time.time()
    if remaining > 20:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, time.time() + remaining * 0.5,
                                         [0.01, 0.005, 0.002, 0.001], max_passes=80)
        else:
            best_h, best_c = near_pair_search(best_h, n, time.time() + remaining * 0.5,
                                              [0.01, 0.005], radius=30)
    
    # Final L-BFGS-B
    if time.time() < deadline - 10:
        h_new = lbfgsb_step(best_h, n, 100000, 100.0, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new
    
    # Final fine pair search
    if time.time() < deadline - 3:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], max_passes=200)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002], radius=40)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_passes=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    target_sum = n / 2.0
    L = n

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    Z = exp_c.sum()
    w = exp_c / Z

    obj = (c_max + np.log(Z) / alpha) + penalty * (h.sum() - target_sum) ** 2

    grad = np.zeros(L)
    for m in range(2 * L - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        lag = m - (L - 1)

        lo1 = max(0, lag)
        hi1 = min(L, lag + L)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) - lag
            grad[lo1:hi1] += wm * dx * (1.0 - h[idx])

        lo2 = max(0, -lag)
        hi2 = min(L, L - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) + lag
            grad[lo2:hi2] -= wm * dx * h[idx]

    grad += 2.0 * penalty * (h.sum() - target_sum)

    return obj, grad

def lbfgsb_step(h, n, alpha, penalty, maxiter=1500):
    def fun(x):
        o, _ = smooth_max_and_grad(x, n, alpha, penalty)
        return o
    def jac(x):
        _, g = smooth_max_and_grad(x, n, alpha, penalty)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=100):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(80):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [100, 300, 1000, 3000, 10000, 30000, 50000, 100000, 200000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        pn = 50.0
        h_new = lbfgsb_step(best_h, n, a, pn, maxiter=1200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new

    remaining = deadline - time.time()
    if remaining > 25:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, time.time() + remaining * 0.5,
                                         [0.01, 0.005, 0.002, 0.001], max_passes=80)
        else:
            best_h, best_c = near_pair_search(best_h, n, time.time() + remaining * 0.5,
                                              [0.01, 0.005], radius=35)

    if time.time() < deadline - 12:
        h_new = lbfgsb_step(best_h, n, 200000, 100.0, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 3:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], max_passes=200)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002], radius=45)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def coarse_search(n_coarse, deadline, rng):
    best_h = None
    best_c = float('inf')

    inits = []
    inits.append(np.full(n_coarse, 0.5))
    block = np.zeros(n_coarse)
    block[:n_coarse // 2] = 1.0
    inits.append(block)
    for a in [0.3, 0.4, 0.5, 0.6]:
        blk = np.zeros(n_coarse)
        w = int(a * n_coarse)
        blk[:w] = 1.0 / a
        inits.append(blk)
    inits.append(rng.uniform(0, 1, n_coarse))

    for h0 in inits:
        if time.time() > deadline:
            break
        h0 = project_feasible(h0, n_coarse)
        alphas = [200, 1000, 5000, 20000]
        h_cur = h0.copy()
        c_cur = compute_c5(h_cur, n_coarse)
        for a in alphas:
            if time.time() > deadline - 5:
                break
            h_new = lbfgsb_step(h_cur, n_coarse, a, 50.0, maxiter=800)
            c_new = compute_c5(h_new, n_coarse)
            if c_new < c_cur - 1e-10:
                c_cur = c_new
                h_cur = h_new
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()

    return best_h, best_c
