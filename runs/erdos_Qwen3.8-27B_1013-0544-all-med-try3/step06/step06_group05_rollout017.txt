    # Step 1: SLSQP with increasing alpha
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        remaining = deadline - time.time() - 15
        maxiter = min(3000, max(500, int(remaining * 30)))
        ht = slsqp_optimize(best_h, n, a, maxiter=maxiter)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    # Step 2: Subgradient descent
    if time.time() < deadline - 20:
        dur = min(30, (deadline - time.time() - 20) * 0.5)
        ht, ct = subgradient_descent(best_h, n, time.time() + dur)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    
    # Step 3: Local pair search
    if time.time() < deadline - 5:
        best_h, best_c = pair_search_local(best_h, n, deadline - 2,
                                           radius=min(40, n // 5))
    
    # Step 4: One more SLSQP pass
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    
    # Step 5: Final pair search
    if time.time() < deadline - 1:
        best_h, best_c = pair_search_local(best_h, n, deadline,
                                           deltas=[0.001, 0.0005, 0.0002, 0.0001],
                                           radius=min(50, n // 4))
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
