    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def solve_1d_convex(base, coeff):
    """Solve min_{x in [0,1]} max_j (base[j] + coeff[j]*x).
    This is a 1D convex optimization problem."""
    # Compute values at x=0 and x=1
    f0 = np.max(base)
    f1 = np.max(base + coeff)
    
    # Check if the unconstrained minimum is at a boundary
    # Subgradient at x: set of {coeff[j] : base[j] + coeff[j]*x = max}
    # At x=0: subgradient is {coeff[j] : base[j] = max(base)}
    sg0 = np.mean(coeff[base == f0])
    # At x=1: subgradient is {coeff[j] : base[j] + coeff[j] = max(base+coeff)}
    sg1 = np.mean(coeff[base + coeff == f1])
    
    if sg0 >= 0:
        return 0.0  # Minimum at x=0
    if sg1 <= 0:
        return 1.0  # Minimum at x=1
    
    # Binary search for where 0 enters the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        vals = base + coeff * mid
        m = np.max(vals)
        active = np.where(np.abs(vals - m) < 1e-10)[0]
        sg = np.mean(coeff[active])
        if sg > 0:
            hi = mid
        else:
            lo = mid
    
    return (lo + hi) / 2

def coord_descent(h, n, n_sweeps=500, time_limit=None, t0=None):
    """Coordinate descent for the Erdős problem.
    Ignores sum constraint during updates, projects after each sweep."""
    target_sum = n / 2.0
    dx_val = 2.0 / n
    
    for sweep in range(n_sweeps):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        
        for i in range(n):
            # Compute coefficient of h[i] in each corr entry
            coeff = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                c = 0.0
                if 0 <= i + s < n:
                    c += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    c += (-h[i - s])
                coeff[n - 1 + s] = c
            
            base = corr_full - coeff * h[i]
            x_opt = solve_1d_convex(base, coeff)
            
            if x_opt != h[i]:
                delta = x_opt - h[i]
                h[i] = x_opt
                corr_full += delta * coeff
        
        # Project onto sum constraint
        h = project_feasible(h, target_sum)
    
    return h
