def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_worst_lags(h, n, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argsort(c)[::-1][:top_k]
    return idx, c[idx], c

def gradient_worst_lag(h, n, lag_idx):
    """Compute gradient of c[lag_idx] with respect to h"""
    # lag_idx is index into the full correlation (length 2n-1)
    shift = lag_idx - (n - 1)
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0

        if 0 <= j - shift < n:
            val += (1.0 - h[j - shift])
        # Contribution from h[j] as second factor (1-h[i-shift]) where i-shift=j
        if 0 <= j + shift < n:
            val += (-h[j + shift])
        grad[j] = val
    return grad * (2.0 / n)  # multiply by dx

        return float(np.sum(x) - n / 2.0)

        return np.ones(n)

    for d in deltas:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c_all = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            idx_k = np.argsort(c_all)[::-1][:top_k]

            for lag_idx in idx_k:
                if time.time() > deadline:
                    break
                lag_idx = int(lag_idx)
                shift = lag_idx - (n - 1)
                if shift == 0:
                    continue
                if shift > 0:
                    pairs = [(i, i + shift) for i in range(n - shift)]
                else:
                    pairs = [(i - shift, i) for i in range(-shift, n)]

                for (i, j) in pairs:
                    if time.time() > deadline:
                        break
                    hi, hj = h[i] - d, h[j] + d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i], h[j] = hi + d, hj - d
                    hi, hj = h[i] + d, h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i], h[j] = hi - d, h[j] + d

    return h, best_c

def fine_pair_search(h, n, deadline, max_passes=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    radius = 50 if n > 200 else n

    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi, hj = h[i] + d, h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, h[j] + d
            if not improved:
                break

    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    for alpha in [5000, 50000, 500000, 5000000, 20000000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp_refine(h, n, alpha, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht.copy()

    if time.time() < deadline - 25:
        gd_dl = min(time.time() + 50, deadline - 15)
        h, best_c = gradient_descent_refine(h, n, gd_dl, top_k=3)

    if time.time() < deadline - 15:
        h, best_c = lag_focused_search(h, n, deadline - 10, top_k=3)

    if time.time() < deadline - 8:
        h, best_c = fine_pair_search(h, n, deadline - 5, max_passes=150)

    if time.time() < deadline - 4:
        ht = slsqp_refine(h, n, 50000000, maxiter=4000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    return project_feasible(h, n), best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = min(220, (deadline - t0) * 0.25)
        h1, c1 = optimize_at_n(h_init, n_init, t0 + alloc1)
        best_h, best_c = h1, c1

        resolutions_to_try = []
        if n_init != 400:
            resolutions_to_try.append(400)
        if n_init != 350:
            resolutions_to_try.append(350)
        if n_init != 500:
            resolutions_to_try.append(500)

        for n_try in resolutions_to_try:
            if time.time() > deadline - 80:
                break
            remaining = deadline - time.time()
            alloc = min(180, remaining * 0.25)
            if alloc < 50:
                break
            x_ref = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_ref, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_new, c_new = optimize_at_n(h_interp, n_try, time.time() + alloc)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            x = np.linspace(0, 2, n)
            freq = (r + 1) * 2 + int(rng.randint(0, 3))
            amp = 0.015 + 0.01 * (r % 4)
            phase = rng.uniform(0, 2 * np.pi)
            pert = amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            hp = best_h + pert
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 35) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if time.time() < deadline - 3:
            best_h, best_c = lag_focused_search(best_h, n, deadline - 2, top_k=5)
        if time.time() < deadline - 1:
            best_h, best_c = fine_pair_search(best_h, n, deadline, max_passes=100)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
