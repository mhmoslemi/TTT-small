    target = n / 2.0

        if abs(diff) < 1e-14:

            # d/dh[j] = dx*(1-h[j+lag]) for j in [0, n-1-lag]
            #           - dx*h[j-lag] for j in [lag, n-1]

            grad[j_range1] += dx * (1 - h[j_range1 + lag])
            j_range2 = np.arange(lag, n)
            grad[j_range2] += -dx * h[j_range2 - lag]

            j_range2 = np.arange(0, n - m)
            grad[j_range2] += -dx * h[j_range2 + m]

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        return project_feasible(res.x, n)

        return h.copy()

def grad_of_top_lags(h, n, top_k=5):
    """Gradient of the average of the top-k correlations."""
    dx = 2.0 / n
    full_c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(full_c)
    
    if top_k >= n_lags:
        active_idx = range(n_lags)
    else:
        active_idx = np.argsort(full_c)[::-1][:top_k]
    
    grad = np.zeros(n)
    
    for k in active_idx:
        lag = k - (n - 1)
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            grad[j1] += dx * (1.0 - h[j1 + lag])
            if lag > 0:
                j2 = np.arange(lag, n)
                grad[j2] += -dx * h[j2 - lag]
        else:
            m = -lag
            j1 = np.arange(m, n)
            grad[j1] += dx * (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            grad[j2] += -dx * h[j2 + m]
    
    return grad / len(list(active_idx))

    step = initial_step

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        return float(m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_of_lags(h, n, top_k=5):
    dx = 2.0 / n
    full_c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(full_c)
    if top_k >= n_lags:
        idx = range(n_lags)
    else:
        idx = np.argsort(full_c)[::-1][:top_k]
    idx_list = list(idx)
    grad = np.zeros(n)
    for k in idx_list:
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += dx * (1.0 - h)
        elif lag > 0:
            j1 = np.arange(0, n - lag)
            grad[j1] += dx * (1.0 - h[j1 + lag])
            j2 = np.arange(lag, n)
            grad[j2] += -dx * h[j2 - lag]
        else:
            m = -lag
            j1 = np.arange(m, n)
            grad[j1] += dx * (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            grad[j2] += -dx * h[j2 + m]
    return grad / len(idx_list)

def gradient_descent(h, n, deadline, initial_step=0.05, min_step=1e-7, top_k=3,
                     restart_every=200, restart_scale=0.02):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = initial_step
    stall = 0
    it = 0
    rng = np.random.RandomState(12345)

    while time.time() < deadline:
        grad = grad_of_lags(h, n, top_k)
        gc = grad - grad.mean()
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_t = h - step * scale * gc
            h_t = project_feasible(h_t, n)
            c_t = compute_c5(h_t, n)
            if c_t < best_c - 1e-16:
                h = h_t
                best_c = c_t
                best_h = h.copy()
                improved = True
                break
        if improved:
            stall = 0
            step = min(step * 1.005, 0.3)
        else:
            stall += 1
            step *= 0.95
            if stall == restart_every:
                h = best_h + rng.normal(0, restart_scale, n)
                h = project_feasible(h, n)
                c_h = compute_c5(h, n)
                if c_h < best_c:
                    best_c = c_h
                    best_h = h.copy()
                stall = 0
                step = initial_step * 0.5
            if step < min_step and stall > 500:

        it += 1
    return best_h, best_c

def make_random_smooth(n, rng, kernel_width=0.15):
    raw = rng.uniform(0.1, 0.9, n)
    x = np.linspace(0, 2, n, endpoint=False)
    xk = np.linspace(0, 2, max(n, 5), endpoint=False)
    raw_k = np.interp(xk, x, raw)
    smoothed = np.zeros_like(raw_k)
    for i in range(len(raw_k)):
        mask = np.abs(xk - xk[i]) <= kernel_width
        if mask.sum() > 0:
            w = np.exp(-0.5 * ((xk[mask] - xk[i]) / (kernel_width / 3.0)) ** 2)
            smoothed[i] = np.dot(raw_k[mask], w) / w.sum()
        else:
            smoothed[i] = raw_k[i]
    return project_feasible(smoothed, n)

        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()

    if time.time() < deadline - 20:
        gd_deadline = min(time.time() + min(120, (deadline - time.time()) * 0.5), deadline - 15)
        best_h, best_c = gradient_descent(best_h, n, gd_deadline,
                                           initial_step=0.08, top_k=3,
                                           restart_every=300, restart_scale=0.015)

    if time.time() < deadline - 8:
        pd = min(time.time() + 50, deadline - 3)
        best_h, best_c = fine_pair_search(best_h, n, pd,
                                           step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                           radius=40)

    if time.time() < deadline - 3:
        h2 = slsqp(best_h, n, 2000000, maxiter=3000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        if time.time() < deadline - 1:
            best_h, best_c = gradient_descent(best_h, n, deadline - 0.5,
                                               initial_step=0.005, min_step=1e-8, top_k=2)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize initial construction (largest allocation)
    alloc1 = min(400, (deadline - t0) * 0.35)
    d1 = t0 + alloc1
    h1, c1 = optimize_single(h_init, n_init, d1, rng)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Try different resolutions
    tried_n = {n_init}
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)

    for n_try in [80, 120, 160, 200, 300, 400, 500]:
        if time.time() > deadline - 60:
            break
        if n_try in tried_n:
            continue
        if n_try > 600:
            continue
        remaining = deadline - time.time() - 30
        if remaining < 50:
            break
        x_try = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_try, x_best, best_h)
        h_try = project_feasible(h_try, n_try)
        alloc = min(80, remaining * 0.3)
        if alloc < 25:
            break
        h_try, c_try = optimize_single(h_try, n_try, time.time() + alloc, rng)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            x_best = np.linspace(0, 2, n_try, endpoint=False)
        tried_n.add(n_try)

    # Phase 3: Alternative initializations
    inits = []
    n_cur = len(best_h)
    for freq in [1, 2, 3, 4, 6]:
        inits.append(make_sin_init(n_cur, freq, 0.0))
        inits.append(make_sin_init(n_cur, freq, np.pi / 3.0))
    inits.append(make_random_smooth(n_cur, rng, 0.12))
    inits.append(make_random_smooth(n_cur, rng, 0.25))

    for h_try in inits:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time() - 20
        if remaining < 40:
            break
        alloc = min(60, remaining * 0.35)
        if alloc < 20:
            break
        h_try, c_try = optimize_single(h_try, n_cur, time.time() + alloc, rng)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

    # Phase 4: Final intensive polish on the best solution
    if time.time() < deadline - 15:
        n_f = len(best_h)
        # High-alpha SLSQP
        h_f = slsqp(best_h, n_f, 5000000, maxiter=5000)
        c_f = compute_c5(h_f, n_f)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

        # Long gradient descent
        if time.time() < deadline - 8:
            gd_dl = min(time.time() + 40, deadline - 5)
            best_h, best_c = gradient_descent(best_h, n_f, gd_dl,
                                               initial_step=0.02, min_step=1e-8,
                                               top_k=2, restart_every=500,
                                               restart_scale=0.005)

        # Very fine pair search
        if time.time() < deadline - 3:
            best_h, best_c = fine_pair_search(best_h, n_f, deadline - 1,
                                               step_sizes=[0.0005, 0.0002, 0.0001, 5e-5],
                                               radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
