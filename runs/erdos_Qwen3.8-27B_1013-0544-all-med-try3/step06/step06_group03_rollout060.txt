import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection on Lagrange multiplier."""
    n = len(h)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    # Return sum of top M values (proportional to their concentration)
    idx = np.argpartition(c, -M)[-M:]
    return float(c[idx].sum())

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

                       bounds=bounds, constraints=cons,

    h = h.copy().astype(float)
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def project_box_sum(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - target_sum / n - 2
    hi = h.max() + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj


def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return float(c[top_indices].sum())

    return float(x.sum() - n / 2.0)


def _sum_j(x, n):
    return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]

                       bounds=bounds, constraints=cons,

    h = h.copy().astype(float)
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    y = h.copy()
    t = 1

    while time.time() < deadline:
        c = compute_all_c(h, n)
        idx = np.argpartition(c, -topM)[-topM:]
        grad = np.zeros(n)
        for k in idx:
            grad += gradient_c(h, n, k)
        grad /= len(idx)

        t_new = (1.0 + (1.0 + 4.0 * t * t) ** 0.5) / 2.0
        momentum = (t - 1.0) / t_new
        y_new = h + momentum * (h - y)
        lr = lr_init / (1.0 + 0.05 * t)
        h_new = project_box_sum(y_new - lr * grad, target)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        y = h.copy()
        h = h_new
        t = t_new

    return best_h, best_c


def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    if time.time() < deadline - 20:
        h, best_c = active_set_optimize(h, n, time.time() + min(80, (deadline - time.time()) * 0.3), M=5, max_outer=20)

    alpha_end = deadline - 30
    if time.time() < alpha_end:
        h, best_c = multi_alpha_slsqp(h, n, alpha_end)

    if time.time() < deadline - 15:
        h, best_c = gradient_descent_topM(h, n, time.time() + min(30, (deadline - time.time()) * 0.2), topM=3, lr_init=0.002)

    if time.time() < deadline - 10:
        h, best_c = active_set_optimize(h, n, time.time() + min(40, (deadline - time.time()) * 0.3), M=3, max_outer=15)

    if time.time() < deadline - 5:
        n_local = len(h)
        if n_local <= 300:
            h, best_c = near_pair_search(h, n_local, deadline - 2,
                                         [0.01, 0.005, 0.002, 0.001], radius=60)
        else:
            h, best_c = near_pair_search(h, n_local, deadline - 2,
                                         [0.005, 0.002, 0.001], radius=40)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Main optimization at initial resolution
    alloc1 = min(250, (deadline - t0) * 0.3)
    h1, c1 = full_optimize(h_init, n_init, t0 + alloc1, rng)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Active set + gradient from current best
    if time.time() < deadline - 150:
        alloc2 = min(80, (deadline - time.time()) * 0.15)
        h2, c2 = active_set_optimize(best_h, len(best_h), time.time() + alloc2, M=3, max_outer=20)
        if c2 < best_c:
            best_h, best_c = h2, c2
        if time.time() < deadline - 100:
            alloc2b = min(40, (deadline - time.time()) * 0.1)
            h2b, c2b = gradient_descent_topM(best_h, len(best_h), time.time() + alloc2b, topM=3, lr_init=0.003)
            if c2b < best_c:
                best_h, best_c = h2b, c2b

    # Phase 3: Subgradient with momentum (escape local optima)
    if time.time() < deadline - 100:
        alloc3 = min(50, (deadline - time.time()) * 0.1)
        h3, c3 = subgradient_momentum(best_h, len(best_h), time.time() + alloc3, topM=3, lr_init=0.003)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-restart with small perturbations
    if best_h is not None and time.time() < deadline - 80:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.04 * (r + 1) / n_restarts
            hp = best_h + rng.normal(0, scale, n_cur)
            hp = project_feasible(hp, n_cur)
            alloc = min(50, (deadline - time.time() - 20) * 0.3)
            if alloc < 20:
                break
            h_r, c_r = full_optimize(hp, n_cur, time.time() + alloc, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Phase 5: Intensive near-pair search
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        h5, c5 = near_pair_search(best_h, n_cur, deadline - 15,
                                   [0.02, 0.01, 0.005, 0.002, 0.001], radius=70)
        if c5 < best_c:
            best_h, best_c = h5, c5

    # Phase 6: Fine active set + gradient
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h6, c6 = active_set_optimize(best_h, n_cur, deadline - 10, M=3, max_outer=10)
        if c6 < best_c:
            best_h, best_c = h6, c6
        if time.time() < deadline - 8:
            h6b, c6b = gradient_descent_topM(best_h, n_cur, deadline - 5, topM=2, lr_init=0.001)
            if c6b < best_c:
                best_h, best_c = h6b, c6b

    # Phase 7: Ultra-fine pair search
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        if n_cur <= 300:
            best_h, best_c = pair_search(best_h, n_cur, deadline - 2,
                                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair_search(best_h, n_cur, deadline - 2,
                                               [0.0005, 0.0002, 0.0001], radius=50)

    # Phase 8: Final SLSQP polish with very high alpha
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        ht = slsqp(best_h, n_cur, 2000000, maxiter=3000)
        c = compute_c5(ht, n_cur)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            n_cur = len(best_h)
            if n_cur <= 300:
                best_h, best_c = pair_search(best_h, n_cur, deadline,
                                              [0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = near_pair_search(best_h, n_cur, deadline,
                                                   [0.0001, 0.00005], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
