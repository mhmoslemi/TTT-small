import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_arr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_steepest_descent(h, n, deadline, max_steps=50000):

        if time.time() > deadline:
            break

        for j in ones:

                J.append(j)

            if no_improve_count > max_no_improve:
                break
            continue

        I = []
        for i in zeros:
            id_ = i + d_star
            if not (0 <= id_ < n) or h[id_] == 1:

            no_improve_count += 1
            if no_improve_count > max_no_improve:

def slsqp_softmax(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        logsumexp = sm + np.log(np.exp(sh - sm).sum())
        return m + logsumexp / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_refine_continuum(h, n, deadline):
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi_val = h[i] + s * d
                        hj_val = h[j] - s * d
                        if 0.0 <= hi_val <= 1.0 and 0.0 <= hj_val <= 1.0:
                            h[i], h[j] = hi_val, hj_val
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                prog = True
                            else:
                                h[i], h[j] = hi_val - s * d, hj_val + s * d
    return h, best_c
