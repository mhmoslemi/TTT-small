        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c + 1e-15 or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_list[ai] = j
            ones_list[bi] = i
            no_improve += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
        else:
            no_improve += 1
        T *= cool
        if no_improve > max_no_improve and T < 0.003:
            T = 0.03
            no_improve = 0
    return best_h, best_c

def multi_restart_slsqp(h, n, deadline, base_alphas, n_restarts=3, noise_scale=0.003):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    best_h_local = h.copy()
    best_c_local = compute_c5(h, n)
    for restart in range(n_restarts):
        if time.time() > deadline - 5:
            break
        h_start = best_h_local + noise_scale * rng.randn(n)
        h_start = project_feasible(h_start, n)
        hh = h_start
        for alpha in base_alphas:
            if time.time() > deadline - 3:
                break
            hh = slsqp_smooth(hh, n, alpha, 2000, 1e-16)
            c = compute_c5(hh, n)
            if c < best_c_local:
                best_c_local = c
                best_h_local = hh.copy()
    return best_h_local, best_c_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP progressive alpha (main workhorse)
    p1_end = t0 + 0.25 * total
    h = best_h.copy()
    alphas1 = [10, 50, 200, 1000, 5000, 20000, 100000, 500000,
               2000000, 10000000, 50000000, 200000000]
    for alpha in alphas1:
        if time.time() > p1_end - 8:
            break
        h = slsqp_smooth(h, n, alpha, 5000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: L-BFGS-B on non-smooth max
    p2_end = t0 + 0.32 * total
    if time.time() < p2_end:
        h = lbfgsb_polish(best_h, n, p2_end, lam=5000.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Binary annealing (long)
    p3_end = t0 + 0.55 * total
    hb, cb = anneal_binary_fast(best_h, n, p3_end, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: Re-smooth from best
    p4_end = t0 + 0.72 * total
    h = project_feasible(best_h, n)
    alphas2 = [50, 200, 1000, 5000, 20000, 100000, 500000,
               2000000, 10000000, 50000000, 200000000, 500000000]
    for alpha in alphas2:
        if time.time() > p4_end - 5:
            break
        h = slsqp_smooth(h, n, alpha, 4000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Multi-restart SLSQP from best
    p5_end = t0 + 0.82 * total
    if time.time() < p5_end:
        alphas3 = [10000, 100000, 1000000, 10000000, 100000000, 500000000]
        h_r, c_r = multi_restart_slsqp(best_h, n, p5_end, alphas3, n_restarts=4, noise_scale=0.002)
        if c_r < best_c:
            best_c, best_h = c_r, h_r.copy()

    # Phase 6: Try higher resolutions
    p6_end = t0 + 0.93 * total
    for nh in [500, 600]:
        if time.time() > p6_end - 15:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        alphas_h = [10000, 100000, 1000000, 10000000, 100000000]
        for alpha in alphas_h:
            if time.time() > p6_end - 10:
                break
            hh = slsqp_smooth(hh, nh, alpha, 2000, 1e-16)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Final polish
    if time.time() < dl - 2:
        h = lbfgsb_polish(best_h, n, dl - 2, lam=10000.0)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() < dl - 1:
            h = slsqp_smooth(best_h, n, 500000000, 3000, 1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
