```python
def compute_gradient(x, n, alpha, dx):
    corr = np.correlate(x, 1-x, mode='full') * dx
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    p = exp_s / exp_s.sum()
    
    # p has length 2n-1, indexed 0 to 2n-2
    # p[k] corresponds to offset o = k - (n-1)
    
    g = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            o = k - (n-1)
            term = 1.0
            if 0 <= i+o < n:
                term -= x[i+o]
            if 0 <= i-o < n:
                term -= x[i-o]
            s += p[k] * term
        g[i] = dx * s
    return g
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1-x, mode='full') * dx
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    val = max_c + (max_s + np.log(exp_s.sum())) / alpha
    if mu > 0:
        s = x.sum() - n/2.0
        val += 0.5 * mu * s * s
    return val

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1-x, mode='full') * dx
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    p = exp_s / exp_s.sum()
    
    # Fast gradient using FFT-based correlation
    corr_px = np.correlate(p, x, mode='full')
    conv_xp = np.convolve(x, p, mode='full')
    
    A = corr_px[2*(n-1) - np.arange(n)]
    B = conv_xp[np.arange(n) + (n-1)]
    
    g = dx * (1.0 - A - B)
    
    if mu > 0:
        s = x.sum() - n/2.0
        g += mu * s
    return g

def lbfgs_smooth(h, n, alpha, mu, maxiter=5000, deadline=None):
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj(x, n, alpha, mu)
    
    def grad(x):
        return smooth_grad(x, n, alpha, mu)
    
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15})
        return project_feasible(res.x, n)
```

Now let me think about the binary local search. The previous implementation was O(n³) per sweep which is too slow. Let me think of a faster approach.

For binary h, the correlation at lag k is:
c[k] = number of positions i where h[i]=1 and h[i+k-(n-1)]=0 (with appropriate bounds)

The maximum of c is what we want to minimize.

A swap (turn i from 0 to 1, j from 1 to 0) changes c by:
delta[k] = (1-h[i+o_k])*[valid, ≠j] + [i+o_k=j] - h[i-o_k]*[valid, ≠j] - (1-h[j+o_k])*[valid, ≠i] + h[j-o_k]*[valid, ≠i]

where o_k = k-(n-1).

The key question: can we compute the maximum of (c + delta) faster than O(n)?

In general, no, because delta is an arbitrary vector. But we can be smarter:

1. Only consider swaps that are "promising" - i.e., where the current c at the relevant lags is high.
