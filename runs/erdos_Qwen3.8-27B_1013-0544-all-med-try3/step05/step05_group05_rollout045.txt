    # h[i] = 1, h[j] = 0
    # Compute change in corr[k] for all k = 1..n-1
    # corr[k] = sum_{valid l} h[l] * (1-h[l+k])
    # After swap: h[i]=0, h[j]=1

    # As first element of pair (l, l+k): term h[i]*(1-h[i+k]) changes from (1-h[i+k]) to 0
    #   => delta_i_first[k] = -(1-h[i+k]) for k where i+k < n, i.e., k < n-i

    #   => delta_i_second[k] = h[i-k] for k where i-k >= 0, i.e., k <= i

    # As first element: term changes from 0 to (1-h[j+k])
    #   => delta_j_first[k] = (1-h[j+k]) for k < n-j

    #   => delta_j_second[k] = -h[j-k] for k <= j

    # i as second element: k in 1..i

    # j as first element: k in 1..n-j-1

    return delta

    """Compute delta in full correlation vector for swap h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2*n - 1)
    
    # Term 1: i as h[n+s] (first factor), h[i] was 1 now 0
    # m range: [max(0, i-1), min(2n-2, i+n-1)]
    # delta[m] = -(1 - h[i - (m - n + 1)]) = -(1 - h[i - m + n - 1])

        delta[lo1:hi1+1] -= (1.0 - h[idx1])
    
    # Term 2: j as h[n+s] (first factor), h[j] was 0 now 1
    # m range: [max(0, j), min(2n-2, j+n-1)]

    lo2 = max(0, j)

    # delta[m] = +h[i + (m-n+1)] = +h[i + m - n + 1]
    lo3 = max(0, n - 1 - i)

    return delta
