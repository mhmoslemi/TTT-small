```python
import numpy as np
from scipy.optimize import minimize
import time

# ... helper functions ...

def slsqp_smooth_max(h, n, alpha, maxiter=3000):
    """SLSQP optimization with smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        # Soft max: m + (1/alpha) * log(sum(exp(alpha*(c-m))))
        shifted = alpha * (c - m)
        mx = shifted.max()
        softmax_part = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + softmax_part
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        # Gradient of soft max
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)
        weights /= weights.sum()  # softmax weights
        # dc_k/dx for each k, then weighted sum
        # This is expensive for n=250, might skip gradient and let SLSQP use numerical
        return None  # Let SLSQP compute numerical gradient
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)
```

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Gradient of c_k w.r.t. h (including dx factor)."""
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        hi1 = n - 1 - s
        g[0:hi1+1] += dx * (1.0 - h[s:hi1+1+s])  # wait, h[s:hi1+1+s] has length hi1+1, but I need h[j+s] for j in [0, hi1]
        # h[j+s] for j in [0, hi1] = h[s: s+hi1+1]
        g[0:hi1+1] += dx * (1.0 - h[s:s+hi1+1])
        g[s:n] -= dx * h[0:n-s]
    else:
        p = -s
        # c_{-p} = dx * sum_{i=0}^{n-1-p} h[i+p]*(1-h[i])
        # dc/dh[j] = dx*(1-h[j-p]) for j in [p, n-1] minus dx*h[j+p] for j in [0, n-1-p]
        g[p:n] += dx * (1.0 - h[0:n-p])
        g[0:n-p] -= dx * h[p:n]
    return g

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max value and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()

    mx = shifted.max()
    weights = np.exp(shifted - mx)

            continue
        s = k - (n - 1)

            hi1 = n - 1 - s
            grad[0:hi1+1] += w * (1.0 - h[s:s+hi1+1])
            grad[s:n] -= w * h[0:n-s]
        else:
            p = -s
            grad[p:n] += w * (1.0 - h[0:n-p])
            grad[0:n-p] -= w * h[p:n]
    return val, grad

def slsqp_round(h, n, alpha, maxiter=2000):
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        weights = np.exp(shifted - mx)

            else:
                p = -s
                grad[p:n] += w * (1.0 - x[0:n-p])
                grad[0:n-p] -= w * x[p:n]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:

                       jac=obj_grad, options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        return h.copy()

def sa_polish(h, n, t_start, T_init, T_final, n_steps, deadline):
    """Simulated annealing with directed pair moves."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    T = T_init
    T_ratio = (T_final / T_init) ** (1.0 / max(1, n_steps))
    rng = np.random.RandomState(42)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T *= T_ratio
        c = get_corr(h, n)
        curr_c = float(c.max())
        if curr_c < best_c:
            best_c = curr_c
            best_h = h.copy()
        
        k_star = int(np.argmax(c))
        g = lag_gradient(h, n, k_star)  # already includes dx
        
        # Find best pair: transfer from max gradient to min gradient
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        
        # Determine step size
        max_step = min(h[i_src] * 0.5, (1.0 - h[i_dst]) * 0.5)
        if max_step < 1e-10:
            continue
        d = np.clip(max_step, 1e-6, 0.1)
        
        # Try the move
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
        else:
            h[i_src] += d
            h[i_dst] -= d
    
    return best_h, best_c

def multi_lag_sa(h, n, deadline, T_init=0.005, n_lags=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T_init
    rng = np.random.RandomState(7)
    step_size = 0.05
    no_improve = 0
    
    while time.time() < deadline:
        T *= 0.999
        c = get_corr(h, n)
        curr_c = float(c.max())
        if curr_c < best_c - 1e-15:
            best_c = curr_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top n_lags
        top_k = np.argpartition(c, -n_lags)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += lag_gradient(h, n, int(k))
        
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        
        max_step = min(h[i_src] * 0.5, (1.0 - h[i_dst]) * 0.5)
        if max_step < 1e-10:
            break
        d = min(step_size, max_step)
        
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        
        delta = new_c - curr_c
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            step_size = min(step_size * 1.1, 0.1)  # encourage larger steps
        else:
            h[i_src] += d
            h[i_dst] -= d
            step_size = max(step_size * 0.95, 0.001)
        
        # Periodic random kick to escape plateaus
        if no_improve > 500:
            idx = rng.choice(n, size=n//10, replace=False)
            h[idx] += rng.normal(0, 0.02, len(idx))
            h = project_feasible(h, n)
            no_improve = 0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        hi1 = n - 1 - s
        g[0:hi1+1] += dx * (1.0 - h[s:s+hi1+1])
        g[s:n] -= dx * h[0:n-s]
    else:
        p = -s
        g[p:n] += dx * (1.0 - h[0:n-p])
        g[0:n-p] -= dx * h[p:n]
    return g

def _obj(x):
    n, alpha = _G['n'], _G['alpha']
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def _obj_grad(x):
    n, alpha = _G['n'], _G['alpha']
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-14:
            continue
        s = k - (n - 1)
        w = weights[k] * dx
        if s >= 0:
            hi1 = n - 1 - s
            grad[0:hi1+1] += w * (1.0 - x[s:s+hi1+1])
            grad[s:n] -= w * x[0:n-s]
        else:
            p = -s
            grad[p:n] += w * (1.0 - x[0:n-p])
            grad[0:n-p] -= w * x[p:n]
    return grad

def _cons_fun(x):
    return float(x.sum() - _G['n'] / 2.0)

def _cons_jac(x):
    return np.ones(_G['n'])

def slsqp_smooth(h, n, alpha, maxiter=3000):
    _G['n'] = n
    _G['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       jac=_obj_grad, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def directed_polish(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stall = 0
    c = get_corr(h, n)
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        g = lag_gradient(h, n, k_star)
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = min(steps[step_idx], h[i_src] * 0.9, (1.0 - h[i_dst]) * 0.9)
        if d < 1e-10:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 200:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        c = get_corr(h, n)
        new_max = float(c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 200:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
    return best_h, best_c

def sa_multi_lag(h, n, deadline, T_init=0.008, n_lags=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T_init
    rng = np.random.RandomState(99)
    step_size = 0.04
    no_improve = 0
    c = get_corr(h, n)
    curr_c = float(c.max())
    while time.time() < deadline:
        T *= 0.998
        if curr_c < best_c - 1e-15:
            best_c = curr_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_k = np.argpartition(c, -n_lags)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += lag_gradient(h, n, int(k))
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        max_step = min(h[i_src] * 0.5, (1.0 - h[i_dst]) * 0.5)
        if max_step < 1e-10:
            break
        d = min(step_size, max_step)
        h[i_src] -= d
        h[i_dst] += d
        c = get_corr(h, n)
        new_max = float(c.max())
        delta = new_max - curr_c
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            curr_c = new_max
            step_size = min(step_size * 1.05, 0.1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            step_size = max(step_size * 0.97, 0.001)
        if no_improve > 800:
            n_pert = max(5, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            h[idx] += rng.normal(0, 0.03, n_pert)
            h = project_feasible(h, n)
            c = get_corr(h, n)
            curr_c = float(c.max())
            no_improve = 0
            step_size = 0.04
    return best_h, best_c

def optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        ht = slsqp_smooth(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    if time.time() < deadline - 20:
        sa_deadline = min(deadline - 20, time.time() + 60)
        best_h, best_c = sa_multi_lag(best_h, n, sa_deadline, T_init=0.008, n_lags=5)
    if time.time() < deadline - 5:
        best_h, best_c = directed_polish(best_h, n, min(deadline - 5, time.time() + 40))
    if time.time() < deadline - 3:
        ht = slsqp_smooth(best_h, n, 5000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 2:
        best_h, best_c = directed_polish(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    try:

        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    if h_init is not None:
        alloc = (deadline - t0) * 0.55
        h_opt, c_opt = optimize(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
        if time.time() < deadline - 100:
            for n_try in [100, 150]:
                if time.time() > deadline - 80:
                    break
                x_init = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new, x_init, best_h)
                h_interp = project_feasible(h_interp, n_try)
                alloc = min(100, (deadline - time.time() - 40) * 0.4)
                if alloc < 30:
                    break
                h_opt, c_opt = optimize(h_interp, n_try, time.time() + alloc)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    else:
        n = n_init
        h_start = np.full(n, 0.5)
        best_h, best_c = optimize(h_start, n, deadline - 10)
    if best_h is not None:
        n = len(best_h)
        if time.time() < deadline - 60:
            remaining = deadline - time.time() - 30
            n_restarts = max(3, int(remaining / 50))
            for r in range(n_restarts):
                if time.time() > deadline - 30:
                    break
                scale = 0.003 + 0.08 * (r / max(1, n_restarts - 1))
                hp = best_h + rng.normal(0, scale, n)
                hp = project_feasible(hp, n)

        if time.time() < deadline - 5:

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
