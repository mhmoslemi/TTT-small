def project_box_sum(y, target_sum):
    """Project y onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(y)
    def f(tau):
        h = np.clip(y - tau, 0, 1)
        return h.sum() - target_sum
    # f is decreasing in tau
    # f(-inf) = n - target_sum > 0 (since target_sum = n/2)
    # f(+inf) = -target_sum < 0
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix any numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 0)[0]
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 0)[0]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0, 1)
    return h

    c = np.correlate(h, 1-h, mode='full') * dx

**Approach: Coordinate descent with projection**

For each coordinate p:
1. Fix all other h values
2. The objective as a function of h[p] is: max_m (A_m * h[p] + B_m) * dx
   where A_m and B_m are precomputed
3. Find h[p]* that minimizes this (1D convex optimization)
4. After updating all coordinates, project to satisfy sum = n/2

Step 3: The function f(t) = max_m (A_m * t + B_m) for t ∈ [0,1] is convex PL.
Its minimum can be found by:
- Evaluating at t=0, t=1
- Evaluating at all breakpoints (intersections of pairs of lines)
- Taking the minimum

For n=250, there are O(n^2) ≈ 62500 breakpoints. Evaluating f at each is O(n) = 250. Total: 62500*250 = 15.6M per coordinate. With 250 coordinates: 3.9B. Too slow!

Better: Use the fact that the minimum of max of lines is at a point where at least 2 lines are "active" and their slopes straddle 0. We can find this in O(n log n) using a sorting-based approach:

1. Sort lines by slope A_m
2. The upper envelope changes at specific points
3. The minimum is where the envelope's slope changes sign

Actually, there's a simpler O(n) approach:
- The subgradient of f(t) at any point is the convex hull of {A_m : m ∈ argmax}
- At the minimum, 0 ∈ subgradient, meaning min(A_m for active m) ≤ 0 ≤ max(A_m for active m)
- We can binary search on t to find where this happens

But binary search needs O(n) per evaluation (to find the active set), and we need O(log(precision)) evaluations. With precision 1e-10, that's ~37 evaluations, each O(n). Total: O(n log(1/ε)) per coordinate.

Total for all coordinates: O(n^2 log(1/ε)) = 250^2 * 37 ≈ 2.3M. That's fast!

def project_box_sum(y, target_sum):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

        else:

            h[mask] += diff / mask.sum()

def pair_move_exact(h, c, i, j):
    """Try a pair move h[i]+=d, h[j]-=d, find optimal d exactly."""
    n = len(h)
    # Compute v[m] = dc[m]/dh[i] - dc[m]/dh[j]
    # dc[m]/dh[p] = (1-h[p+s])*I(0<=p+s<n) + (-h[p-s])*I(0<=p-s<n) where s=m-(n-1)
    
    # For efficiency, compute v directly
    v = np.zeros(2*n-1)
    for m in range(2*n-1):
        s = m - (n-1)
        # Contribution from h[i]:
        vi = 0.0
        if 0 <= i+s < n:
            vi += (1.0 - h[i+s])
        if 0 <= i-s < n:
            vi += (-h[i-s])
        # Contribution from h[j]:
        vj = 0.0
        if 0 <= j+s < n:
            vj += (1.0 - h[j+s])
        if 0 <= j-s < n:
            vj += (-h[j-s])
        v[m] = vi - vj
    
    # Find d that minimizes max(c + d*v) subject to d_lo <= d <= d_hi
    d_lo = max(-h[i], -h[j])  # h[i]+d >= 0 => d >= -h[i], h[j]-d >= 0 => d <= h[j]
    d_hi = min(1.0 - h[i], h[j])  # h[i]+d <= 1 => d <= 1-h[i], h[j]-d <= 1 => d >= h[j]-1
    
    # Wait: d_lo = max(-h[i], h[j]-1.0) and d_hi = min(1.0-h[i], h[j])
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    
    if d_lo >= d_hi:
        return 0.0
    
    # Minimize max(c + d*v) over d in [d_lo, d_hi]
    # Use bisection on the subgradient
    def f(d):
        return np.max(c + d * v)
    
    # Check endpoints
    f_lo = f(d_lo)
    f_hi = f(d_hi)
    
    # Binary search for where subgradient crosses 0
    # At point d, the subgradient is {v[m] : m in argmax(c+d*v)}
    # We want min(v[m] for m in argmax) <= 0 <= max(v[m] for m in argmax)
    
    best_d = d_lo if f_lo < f_hi else d_hi
    best_f = min(f_lo, f_hi)
    
    # Try bisection
    lo, hi = d_lo, d_hi
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = c + mid * v
        max_val = vals.max()
        active = np.where(vals >= max_val - 1e-12)[0]
        v_active = v[active]
        if v_active.min() <= 0 and v_active.max() >= 0:
            # 0 is in the subgradient, this is optimal (or very close)
            break
        elif v_active.min() > 0:
            # All active slopes are positive, decrease d
            hi = mid
        else:
            # All active slopes are negative, increase d
            lo = mid
    
    d_opt = (lo + hi) / 2.0
    f_opt = f(d_opt)
    
    if f_opt < best_f:
        best_d = d_opt
        best_f = f_opt
    
    return best_d

        p3, p4 = max(0, s), min(n, n + s)
        if p3 < p4:
            grad[p3:p4] += wm * (-h[p3-s:p4-s])
    return grad

def project_grad_descent(h, n, deadline, max_iter=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        progress = min(1.0, it / (max_iter * 0.8))
        alpha = 50.0 * (5000.0 / 50.0) ** progress
        grad = smooth_gradient(h, n, alpha)
        # Backtracking
        improved = False
        for bt in range(10):
            trial_lr = lr * (0.5 ** bt)
            h_new = project_box_sum(h - trial_lr * grad, n / 2.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
        if improved:
            stuck = 0
            lr = min(0.1, lr * 1.1)
        else:
            stuck += 1
            lr = max(1e-5, lr * 0.5)
            if stuck > 30:
                break
    return best_h, best_c

def pair_sweep(h, n, c, d_max, n_pairs, rng):
    """Try n_pairs random pair moves with exact 1D optimization."""
    improved = False
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Compute v[m] = dc[m]/dh[i] - dc[m]/dh[j]
        # For efficiency, compute incrementally
        v = np.zeros(2*n-1)
        for m in range(2*n-1):
            s = m - (n-1)
            vi = 0.0
            if 0 <= i+s < n: vi += 1.0 - h[i+s]
            if 0 <= i-s < n: vi -= h[i-s]
            vj = 0.0
            if 0 <= j+s < n: vj += 1.0 - h[j+s]
            if 0 <= j-s < n: vj -= h[j-s]
            v[m] = vi - vj
        
        d_lo = max(-h[i], h[j]-1.0)
        d_hi = min(1.0-h[i], h[j])
        d_lo = max(d_lo, -d_max)
        d_hi = min(d_hi, d_max)
        if d_lo >= d_hi:
            continue
        
        # Bisection for optimal d
        lo, hi = d_lo, d_hi
        for _ in range(30):
            mid = (lo + hi) / 2.0
            vals = c + mid * v
            mx = vals.max()
            active = vals >= mx - 1e-12
            if active.sum() == 0:
                break
            if v[active].min() <= 0 and v[active].max() >= 0:
                break
            elif v[active].min() > 0:
                hi = mid
            else:
                lo = mid
        
        d_opt = (lo + hi) / 2.0
        # Also check endpoints
        f_opt = (c + d_opt * v).max()
        f_lo = (c + d_lo * v).max()
        f_hi = (c + d_hi * v).max()
        if f_lo < f_opt:
            d_opt = d_lo
            f_opt = f_lo
        if f_hi < f_opt:
            d_opt = d_hi
            f_opt = f_hi
        
        if f_opt < c.max() - 1e-14:
            h[i] += d_opt
            h[j] -= d_opt
            c = get_c(h, n)
            improved = True
    
    return h, c, improved

def optimize_combined(h, n, deadline, rng):

    c = get_c(h, n)
    best_c = float(c.max())

def project_box_sum(y, target):
