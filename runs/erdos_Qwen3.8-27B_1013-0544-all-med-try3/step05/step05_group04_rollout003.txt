    idx = np.argsort(c)[-n_active:]

        s = k - n + 1
        g = np.zeros(n)

            val = 0.0

    A_ub = np.zeros((n_active, n + 1))

def project_feasible(h, n):

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_corr(h, n, s):
    """Gradient of correlation with shift s, scaled by dx."""
    dx = 2.0 / n
    g = np.zeros(n)
    js_start = max(0, -s)
    js_end = min(n, n - s)
    if js_start < js_end:
        g[js_start:js_end] += (1.0 - h[js_start + s:js_end + s])
    jm_start = max(0, s)
    jm_end = min(n, n + s)
    if jm_start < jm_end:
        g[jm_start:jm_end] -= h[jm_start - s:jm_end - s]
    return g * dx

def grad_descent_phase(h, n, deadline, max_iters=50000):
    """Steepest descent on the max correlation using projected gradient steps."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1 - h, 'full') * dx
        k_star = c.argmax()
        s = k_star - n + 1
        
        g = grad_corr(h, n, s)
        # Project gradient to sum-0 subspace
        g = g - g.mean()
        
        # Line search
        improved = False
        for factor in [step, step*0.5, step*0.25, step*0.1]:
            h_new = project_feasible(h - factor * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
                improved = True
                break
        
        if not improved:
            # Try reducing step size
            step *= 0.5
            if step < 1e-8:
                break
        else:
            step = min(step * 1.1, 0.1)
    
    return best_h, best_c
