def slsqp_smooth(h, n, alpha, maxiter=2000):

        m = c.max()

        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(lo + s, 0)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
    return g

def _sum_fun(x, n):

        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    def obj_grad(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)

            g += w[k] * lag_gradient(hx, n, k)
        return g
    bounds = [(0.0, 1.0)] * n
