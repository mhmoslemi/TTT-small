def slsqp_tophk(h, n, k, maxiter, deadline):
    """SLSQP with top-k approximation of max."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        # Top-k approximation
        idx = np.argpartition(c, -k)[-k:]
        return np.mean(c[idx]) * dx
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        idx = np.argpartition(c, -k)[-k:]
        grad = np.zeros(n)
        for ki in idx:

            for i in range(n):
                im = i - kk
                ip = i + kk
                g = 0.0
                if 0 <= im < n: g += (1 - x[im])
                if 0 <= ip < n: g -= x[ip]
                grad[i] += g
        grad /= k
        return grad * dx
    bounds = [(0.0, 1.0)] * n

    try:

                flip[i0, m + n - 1] -= h[ip]
        return flip

```python
def get_flip_up(h, n):
    """Vectorized computation of flip vectors."""
    # flip[i0][m_idx] where m_idx = m + (n-1), m in [-(n-1), n-1]
    # = (1-h[i0-m]) [if 0<=i0-m<n] - h[i0+m] [if 0<=i0+m<n]
    
    # First term: for each i0, the values (1-h[i0-m]) for m in valid range
    # i0-m = p, so m = i0-p, p in [0, n-1]
    # Array index: (i0-p) + (n-1) = i0 + n - 1 - p
    # So flip[i0, i0+n-1-p] += (1-h[p]) for p in [0, n-1]
    # This is: flip[i0, (i0+n-1)::-1] += (1-h)  [reversed]
    
    # Actually, let's use a different approach with index arithmetic
    i0 = np.arange(n)[:, None]  # (n, 1)
    m = np.arange(-(n-1), n)[None, :]  # (1, 2n-1)
    
    # First term: (1-h[i0-m]) where 0 <= i0-m < n
    im = i0 - m  # (n, 2n-1)
    mask1 = (im >= 0) & (im < n)
    term1 = np.where(mask1, 1 - h[im.clip(0, n-1)], 0)
    
    # Second term: -h[i0+m] where 0 <= i0+m < n
    ip = i0 + m  # (n, 2n-1)
    mask2 = (ip >= 0) & (ip < n)
    term2 = np.where(mask2, -h[ip.clip(0, n-1)], 0)
    
    return term1 + term2
```

This creates (n, 2n-1) = (250, 499) arrays. Memory: 250*499*8 = ~1MB per array. Very manageable.

Time: a few numpy operations on 250x499 arrays, should be ~1-2 ms.

            elif delta_c < 3*T and rng.random() < np.exp(-delta_c / max(T, 1e-15)):

Wait, there's an issue. After accepting a swap, I break out of the inner loop and recompute everything. This means each "round" (inner loop over zeros) does at most one swap. That's very slow.

Better approach: after a swap, update the flip vectors incrementally and continue.

When we swap h[i] from 0 to 1 and h[j] from 1 to 0:
- flip[i] changes: it was computed for h[i]=0, now h[i]=1. The flip vector for position i is the change when h[i] flips. But now h[i] is 1, so flip[i] would be the change when it goes 1→0 = -flip[i] (old). But we don't need flip[i] anymore since i is now a one.
- For all other positions p: flip[p] might change because h[i] and h[j] changed.

This incremental update is complex. Let me just recompute flip after each swap. For n=250, it takes ~2ms. If we do 100 swaps per "phase", that's 200ms. Fine.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def get_flip_up(h, n):
    i0 = np.arange(n)[:, None]
    m = np.arange(-(n - 1), n)[None, :]
    im = i0 - m
    mask1 = (im >= 0) & (im < n)
    term1 = np.where(mask1, 1.0 - h[im.clip(0, n - 1)], 0.0)
    ip = i0 + m
    mask2 = (ip >= 0) & (ip < n)
    term2 = np.where(mask2, -h[ip.clip(0, n - 1)], 0.0)
    return term1 + term2

def sa_binary(h_init, n, deadline, seed=42, T_init=0.015):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T = T_init
    T_min = 0.0003
    cooling = 0.9995

    while time.time() < deadline:
        corr = compute_corr(h, n)
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        flip = get_flip_up(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        flip_i = flip[zeros]
        flip_j = flip[ones]
        n_swaps = 0

        z_order = rng.permutation(len(zeros))
        for zi in z_order:
            if time.time() > deadline:
                return best_h, best_c
            i = zeros[zi]
            fi = flip_i[zi]
            deltas = fi[None, :] - flip_j
            scores = (corr[None, :] + deltas).max(axis=1) * dx
            bi = np.argmin(scores)
            bj = ones[bi]
            sc = scores[bi]
            dc = sc - cur_c
            accept = dc < -1e-14 or (dc < 4 * T and rng.random() < np.exp(-dc / max(T, 1e-15)))
            if accept:
                h[i] = 1.0
                h[bj] = 0.0
                cur_c = sc
                n_swaps += 1
                corr = compute_corr(h, n)
                flip = get_flip_up(h, n)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                flip_i = flip[zeros]
                flip_j = flip[ones]
                if n_swaps >= 30:
                    break
        if n_swaps == 0:
            T = max(T * cooling, T_min)
            if T <= T_min * 1.01:
                c1 = np.where(h == 1)[0]
                c0 = np.where(h == 0)[0]
                for _ in range(rng.randint(2, 7)):
                    a = rng.choice(c1)
                    b = rng.choice(c0)
                    h[a], h[b] = 0.0, 1.0
                    c1 = np.where(h == 1)[0]
                    c0 = np.where(h == 0)[0]
                T = T_init
        else:
            T = max(T * cooling, T_min)

    return best_h, best_c

def slsqp_tophk(h, n, k, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        idx = np.argpartition(c, -k)[-k:]
        return float(np.mean(c[idx]) * dx)
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        idx = np.argpartition(c, -k)[-k:]
        grad = np.zeros(n)
        for ki in idx:
            kk = ki - (n - 1)
            im = np.arange(n) - kk
            m1 = (im >= 0) & (im < n)
            grad += np.where(m1, 1.0 - x[im[m1].clip(0, n-1)], 0.0) * m1
            ip = np.arange(n) + kk
            m2 = (ip >= 0) & (ip < n)
            grad -= np.where(m2, x[ip[m2].clip(0, n-1)], 0.0) * m2
        return grad / k * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_method(h_init, n, deadline, lr_init=0.05):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    while time.time() < deadline:
        corr = compute_corr(h, n)
        M = corr.max()
        active = np.where(corr >= M - max(0.05 * M, 0.5))[0]
        grad = np.zeros(n)
        for ki in active:
            kk = ki - (n - 1)
            im = np.arange(n) - kk
            m1 = (im >= 0) & (im < n)
            grad += np.where(m1, 1.0 - h[im[m1].clip(0, n - 1)], 0.0)
            ip = np.arange(n) + kk
            m2 = (ip >= 0) & (ip < n)
            grad -= np.where(m2, h[ip[m2].clip(0, n - 1)], 0.0)
        grad /= len(active)
        h = project_feasible(h - lr * grad, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lr = max(lr * 0.998, 0.001)
    return best_h, best_c
