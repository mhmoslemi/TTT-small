import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_overlaps(h, n):
    """Compute all overlap values c[k] * dx for k = -(n-1), ..., n-1"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c  # shape (2n-1,), index 0 corresponds to shift -(n-1)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def solve_qp_for_shift(h, n, k, method='SLSQP'):
    """
    Solve: min sum_i h[i](1-h[i+k]) s.t. sum h[i] = n/2, 0 <= h[i] <= 1
    This is a QP since the objective is quadratic in h.
    """
    dx = 2.0 / n
    target = n / 2.0
    
    # Objective: sum_{i} h[i](1-h[i+k]) = sum_i h[i] - sum_i h[i]h[i+k]
    # (with appropriate ranges)
    
    # For k > 0: valid i is 0 to n-1-k
    #   obj = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]h[i+k]
    #   = sum_{i=0}^{n-1-k} h[i] - h[0:n-k] @ h[k:n]
    
    # For k < 0 (k' = -k): 
    #   obj = sum_{i=k'}^{n-1} h[i] - sum_{i=k'}^{n-1} h[i]h[i-k']
    #   = sum_{i=k'}^{n-1} h[i] - h[k':n] @ h[0:n-k']
    
    # For k = 0:
    #   obj = sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]^2
    
    # This is a QP. Let's use scipy minimize with the explicit quadratic.
    
    if k > 0:
        idx1 = np.arange(n - k)
        idx2 = np.arange(k, n)
        A = np.zeros((n, n))  # Hessian of -obj (we want to minimize obj, so Hessian of obj)
        # obj = sum_{i in idx1} h[i] - h[idx1] @ h[idx2]
        # d²obj/dh[i]dh[j]: for i in idx1, j = i+k in idx2: -1
        # So Hessian = -matrix with -1 at (i, i+k) and (i+k, i) for i in idx1
        # Actually: obj = sum_i h[i] - sum_i h[i]h[i+k]
        # ∂obj/∂h[i] = 1[i in idx1] - h[i+k][i+k in range] - h[i-k][i-k in range]
        # ∂²obj/∂h[i]∂h[j] = -1 if j = i+k (i in idx1) or i = j+k (j in idx2)
        
        # For SLSQP, we need obj and grad:
        def obj(x):
            return np.sum(x[idx1]) - np.sum(x[idx1] * x[idx2])
        def grad(x):
            g = np.zeros(n)
            g[idx1] += 1.0
            g[idx1] -= x[idx2]
            g[idx2] -= x[idx1]
            return g
    elif k < 0:
        kk = -k
        idx1 = np.arange(kk, n)
        idx2 = np.arange(0, n - kk)
        def obj(x):
            return np.sum(x[idx1]) - np.sum(x[idx1] * x[idx2])
        def grad(x):

            g[idx1] += 1.0
            g[idx1] -= x[idx2]
            g[idx2] -= x[idx1]
            return g
    else:
        def obj(x):
            return np.sum(x) - np.sum(x * x)
        def grad(x):
            return 1.0 - 2.0 * x
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                      'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', jac=grad,

            m_star = int(np.argmax(all_c_arr))
            k_act = m_star - (n-1)
            g = grad_active(h, n, k_act)
            v = mom*v - lr*g
            h_new = project(h + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr*1.1, 0.1)
            else:
                v *= 0.5
                lr *= 0.9
                if lr < 1e-8:
                    h = project(best_h + 0.02*rng.randn(n), n)
                    v = np.zeros(n)
                    lr = 0.01
    
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

Instead of random selection of i and j, I can:
1. Compute for each position i (with h[i]=1), the "cost" of moving it: how much it contributes to the maximum overlap
2. Compute for each position j (with h[j]=0), the "benefit" of adding mass there
3. Pick the pair (i,j) that gives the best improvement

The "cost" of position i is: for the active shift k*, h[i] contributes (1-h[i+k*]) to c[k*]. So positions with h[i+k*]=0 are "expensive" (they contribute fully to the overlap).

Let me think about it differently. For a swap of i (1→0) and j (0→1):
Change in c[k] = h[i+k] - h[j+k] (for the case where both i+k and j+k are in [0,n))

For the active k*, we want h[i+k*] < h[j+k*].

So the "best" swap for the active constraint is one where h[i+k*] is as small as possible and h[j+k*] is as large as possible.

I can precompute:
- For each i in ones: cost_i = h[i+k*] (lower is better)
- For each j in zeros: benefit_j = h[j+k*] (higher is better)

Then the best swap is the one that minimizes cost_i - benefit_j, subject to the constraint that the swap actually reduces the maximum (not just c[k*]).

Checking the full constraint is expensive, but as a heuristic, if h[i+k*] is very small and h[j+k*] is very large, the swap is likely to reduce the max.
