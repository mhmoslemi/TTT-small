For the piecewise linear function f(δ) = max_m (a_m + b_m * δ) where a_m = c[m], b_m = coef[m]:
The minimum is at a point where two lines cross, or at a boundary.
We can find it by: for each pair (m1, m2), the crossing point is δ = (a_m2 - a_m1) / (b_m1 - b_m2). The minimum is at the crossing point where both lines are achieving the max.

But this is O(n²) to enumerate all pairs. For n=250, that's 62500 pairs, which is fine.

```python
def find_optimal_delta(c, coef, lo, hi):
    # f(δ) = max(c + δ*coef), convex
    # Find δ that minimizes f
    
    # Evaluate at boundaries
    f_lo = np.max(c + lo * coef)
    f_hi = np.max(c + hi * coef)
    
    # Bisection on subgradient
    # Subgradient at δ: if c[m*]+δ*coef[m*] is unique max, subgrad = coef[m*]
    # Otherwise, subgrad ∈ [min active coef, max active coef]
    
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c + mid * coef
        mval = vals.max()
        active = np.where(vals >= mval - 1e-14)[0]
        if len(active) == 1:
            sg = coef[active[0]]
        else:
            sg_lo = coef[active].min()
            sg_hi = coef[active].max()
            if sg_lo <= 0 <= sg_hi:
                return mid  # 0 is in subgradient, optimal
            elif sg_hi < 0:
                # f is decreasing at mid, need to go right
                lo = mid
            else:  # sg_lo > 0
                # f is increasing at mid, need to go left
                hi = mid
        # For the unique active case:
        if len(active) == 1:
            if coef[active[0]] > 0:
                hi = mid  # f increasing, go left
            elif coef[active[0]] < 0:
                lo = mid  # f decreasing, go right
            else:
                return mid  # already optimal
    
    return (lo + hi) / 2
```

This is O(n) per iteration, 50 iterations, so O(50n) = O(n) per coordinate. Total O(n²) per pass.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=500, deadline=None, verbose=False):
    """Fast coordinate descent for min max_k c[k]."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    target = n / 2.0
    best_c5 = np.max(c) * (2.0/n)
    
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Compute sensitivity vector
            coef = np.zeros(2*n-1)
            # term1: (1-h[j-i+n-1]) for valid i
            # j-i+n-1 in [0, n-1] => i in [j+1-n, j]
            i_lo = max(0, j+1-n)
            i_hi = min(2*n-2, j)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi+1)
                h_idx = j - idx + n - 1
                coef[idx] += (1.0 - h[h_idx])
            # term2: -h[j+i-n+1] for valid i
            # j+i-n+1 in [0, n-1] => i in [n-1-j, n-1+j]
            i_lo2 = max(0, n-1-j)
            i_hi2 = min(2*n-2, n-1+j)
            if i_lo2 <= i_hi2:
                idx2 = np.arange(i_lo2, i_hi2+1)
                h_idx2 = j + idx2 - n + 1
                coef[idx2] -= h[h_idx2]
            
            # Find optimal delta
            lo = -h[j]
            hi = 1.0 - h[j]
            if lo >= hi - 1e-15:
                continue
            
            # Bisection on subgradient
            for _ in range(30):
                if hi - lo < 1e-12:
                    break
                mid = 0.5 * (lo + hi)
                vals = c + mid * coef
                mval = vals.max()
                active = vals >= mval - 1e-14
                n_active = active.sum()
                if n_active == 1:
                    j_act = np.argmax(active)
                    sg = coef[j_act]
                    if sg > 0:
                        hi = mid
                    else:
                        lo = mid
                else:
                    sg_lo = coef[active].min()
                    sg_hi = coef[active].max()
                    if sg_lo <= 0 <= sg_hi:
                        lo = mid  # optimal, tighten bounds
                        hi = mid
                    elif sg_hi < 0:
                        lo = mid  # function decreasing, go right
                    else:
                        hi = mid  # function increasing, go left
            
            delta = 0.5 * (lo + hi)
            if abs(delta) > 1e-10:
                h[j] += delta
                h[j] = np.clip(h[j], 0.0, 1.0)
                c = c + delta * coef
                improved = True
        
        # Project to sum constraint
        h = project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_coef(j, h, n):
    coef = np.zeros(2 * n - 1)
    # term1: (1-h[j-i+n-1]) for i where 0 <= j-i+n-1 <= n-1
    i_lo = max(0, j + 1 - n)
    i_hi = min(2*n - 2, j)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        h_idx = j - idx + n - 1
        coef[idx] += (1.0 - h[h_idx])
    # term2: -h[j+i-n+1] for i where 0 <= j+i-n+1 <= n-1
    i_lo2 = max(0, n - 1 - j)
    i_hi2 = min(2*n - 2, n - 1 + j)
    if i_lo2 <= i_hi2:
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        h_idx2 = j + idx2 - n + 1
        coef[idx2] -= h[h_idx2]
    return coef

def find_optimal_delta(c, coef, lo, hi):
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        mid = 0.5 * (lo + hi)
        vals = c + mid * coef
        mval = vals.max()
        active = vals >= mval - 1e-14
        cnt = active.sum()
        if cnt == 1:
            sg = coef[np.argmax(active)]
            if sg > 1e-15:
                hi = mid
            elif sg < -1e-15:
                lo = mid
            else:
                break
        else:
            sg_lo = coef[active].min()
            sg_hi = coef[active].max()
            if sg_lo <= 0 and sg_hi >= 0:
                break  # 0 in subgradient
            elif sg_hi < 0:
                lo = mid
            else:
                hi = mid
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_passes=5000, deadline=None):
    h = h.astype(float).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    target = n / 2.0
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-14:
                continue
            coef = compute_coef(j, h, n)
            delta = find_optimal_delta(c, coef, lo, hi)
            if abs(delta) > 1e-10:
                h[j] += delta
                h[j] = np.clip(h[j], 0.0, 1.0)
                c += delta * coef
                improved = True
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_refine(h, n, deadline, max_sweeps=500):
    """Fast binary swap local search."""
    target = int(n // 2)
    # Binarize
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full').copy()
    cur_c = corr.max() * dx
    best_h = h_bin.copy()
    best_c = cur_c
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2*n-1)
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h_bin[im[msk]])
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_bin[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h_bin[jm[msk]])
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_bin[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h_bin[i], h_bin[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_bin.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Phase 1: Coordinate descent (fast, many passes)
    p1_end = t0 + 0.25 * (budget_s - 5)
    h1, c1 = coord_descent(h0, n, max_passes=10000, deadline=p1_end)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    
    # Phase 2: SLSQP polish
    p2_end = t0 + 0.4 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [1000, 10000, 100000, 1000000]:
        if time.time() > p2_end:
            break
        h = slsqp_smooth(h, n, alpha, 2000, p2_end)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Coordinate descent again (from SLSQP result)
    p3_end = t0 + 0.55 * (budget_s - 5)
    h3, c3 = coord_descent(best_h, n, max_passes=10000, deadline=p3_end)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()
    
    # Phase 4: Binary refinement
    p4_end = t0 + 0.75 * (budget_s - 5)
    hb, cb = binary_refine(best_h, n, p4_end, max_sweeps=2000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 5: Multi-restart with perturbations
    p5_end = t0 + 0.9 * (budget_s - 5)
    for restart in range(20):
        if time.time() > p5_end:
            break
        h_r = best_h + 0.05 * rng.randn(n)
        h_r = project_feasible(h_r, n)
        h_r, c_r = coord_descent(h_r, n, max_passes=2000, deadline=p5_end)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    
    # Phase 6: Final polish
    if time.time() < dl - 10:
        h_f, c_f = coord_descent(best_h, n, max_passes=5000, deadline=dl - 5)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
        if time.time() < dl - 2:
            hb2, cb2 = binary_refine(best_h, n, dl, max_sweeps=500)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_coef(j, h, n):
    N = n
    coef = np.zeros(2 * N - 1)
    i_lo = max(0, j + 1 - N)
    i_hi = min(2 * N - 2, j)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        h_idx = j - idx + N - 1
        coef[idx] += (1.0 - h[h_idx])
    i_lo2 = max(0, N - 1 - j)
    i_hi2 = min(2 * N - 2, N - 1 + j)
    if i_lo2 <= i_hi2:
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        h_idx2 = j + idx2 - N + 1
        coef[idx2] -= h[h_idx2]
    return coef

def find_optimal_delta(c, coef, lo, hi):
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        mid = 0.5 * (lo + hi)
        vals = c + mid * coef
        mval = vals.max()
        active = vals >= mval - 1e-14
        cnt = active.sum()
        if cnt == 1:
            sg = coef[np.argmax(active)]
            if sg > 1e-15:

            elif sg_hi < 0.0:
                lo = mid
            else:

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

                       constraints=cons,
