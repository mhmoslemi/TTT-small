```python
    """Update correlation array when h[i]:0->1 and h[j]:1->0"""
    # h[i] goes from 0 to 1: adds (1-h[i+lag]) for valid lags
    # h[j] goes from 1 to 0: removes (1-h[j+lag]) for valid lags
    
    # For position i (0->1):
    # c[m] += (1 - h[lag + i]) where m = n-1+lag, valid when 0<=lag+i<n and 0<=i<n
    # lag ranges from -i to n-1-i

        idx = i + lag
        if 0 <= idx < n:
            c[m] += (1 - h[idx])

    # c[m] -= (1 - h[lag + j]) where valid
    for lag in range(-j, n-j):
        m = n - 1 + lag
        idx = j + lag
        if 0 <= idx < n:
            c[m] -= (1 - h[idx])
    
    return c
```

```python
def update_correlation(c, h, i, j, n):
    # For position i (0->1): lag from -i to n-1-i, m = n-1+lag
    # idx = i + lag = i - i + ... = i + lag
    # m ranges from n-1-i to n-1-i+n-1 = 2n-2-i
    # idx = i + (m - n + 1) = m - n + 1 + i
    m_start = n - 1 - i
    m_end = 2 * n - 2 - i + 1  # exclusive
    idx_start = 0
    idx_end = n - i
    c[m_start:m_end] += (1 - h[idx_start:idx_end])
    
    # For position j (1->0):
    m_start = n - 1 - j
    m_end = 2 * n - 2 - j + 1
    idx_start = 0
    idx_end = n - j
    c[m_start:m_end] -= (1 - h[idx_start:idx_end])
    
    return c
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=200):
    """Greedy local search on binary vectors."""
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cmax = float(c.max())
    best_h = h.copy()
    best_cmax = cmax
    
    target = n // 2
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i_idx in range(len(zeros)):
            if time.time() > deadline:
                break
            zi = zeros[i_idx]
            best_j = -1
            best_local_cmax = cmax
            for j_idx in range(len(ones)):
                oj = ones[j_idx]
                h[zi] = 1.0
                h[oj] = 0.0

                h[zi] = 0.0
```
