def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

            break
        max_change = 0.0

            k_lo2 = max(0, n - 1 - i)

                beta[idx] -= h[i + idx - n + 1]

            grid = np.linspace(0.0, 1.0, 64)
