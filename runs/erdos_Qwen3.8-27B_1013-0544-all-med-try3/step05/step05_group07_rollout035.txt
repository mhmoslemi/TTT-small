def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), np.max(c), int(np.argmax(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break

    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def raw_max_c(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    half = n / 2.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e4 * (x.sum() - half) ** 2
        return float(val)

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e4 * (x.sum() - half)
        return g

    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h
