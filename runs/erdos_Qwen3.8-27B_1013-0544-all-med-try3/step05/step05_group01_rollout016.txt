    """Compute delta in O[d] for swapping h[i] and h[j] (assuming h[i]=1, h[j]=0, j=i+d)."""
    # i < j, d = j - i
    idx1 = i - d
    idx2 = j + d
    val1 = h[idx1] if 0 <= idx1 < len(h) else 0.0
    val2 = h[idx2] if 0 <= idx2 < len(h) else 0.0
    return val1 - val2

def binary_greedy(h, n, deadline):
    """Greedy local search: repeatedly find best swap at the max lag."""
    dx = 2.0 / n
    h = h.copy()
    O = compute_O(h, n)
    best_c = float(O.max())
    best_h = h.copy()
    stall_count = 0
    
    while time.time() < deadline:
        d_star = int(np.argmax(O))
        max_val = O[d_star]
        
        # Find best swap at lag d_star
        best_delta = 0.0
        best_i = -1
        for i in range(n - d_star):
            j = i + d_star
            if h[i] == 1 and h[j] == 0:
                delta = swap_delta(h, i, j, d_star)
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
        
        if best_i >= 0:
            j = best_i + d_star
            h[best_i] = 0
            h[j] = 1
            O[d_star] += best_delta
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                break
    
    O = compute_O(h, n)  # Recompute to be safe
    c = float(O.max())
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def binary_sa(h, n, deadline, rng):
    """Simulated annealing on binary sequences."""
    dx = 2.0 / n
    h = h.copy()
    O = compute_O(h, n)
    best_c = float(O.max())
    best_h = h.copy()
    
    T0 = 0.5
    n_iters = 200000
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (0.01) ** (it / n_iters)
        
        # Random swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        
        # Ensure h[i]=1, h[j]=0
        if h[i] == 0:
            i, j = j, i
        
        d = j - i
        if d <= 0:
            d = -d
        
        # Compute delta
        if h[i] == 1 and h[j] == 0:
            # i is the 1, j is the 0
            i1, j1 = min(i,j), max(i,j)
            dd = j1 - i1
            idx1 = i1 - dd
            idx2 = j1 + dd
            val1 = h[idx1] if 0 <= idx1 < n else 0.0
            val2 = h[idx2] if 0 <= idx2 < n else 0.0
            delta = val1 - val2
        else:
            continue  # shouldn't happen
        
        old_max = float(O.max())
        O[d] += delta
        new_max = float(O.max())
        
        accept = False
        if new_max <= old_max:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-(new_max - old_max) / T):
                accept = True
        
        if accept:
            h[i], h[j] = h[j], h[i]
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
        else:
            O[d] -= delta
    
    return best_h, best_c

def smooth_obj(x, n, alpha):

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h.copy()

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room

            if tot < 1e-14:
