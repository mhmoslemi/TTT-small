```python
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
```

d[i] = -(1 - h[i-k*]) + h[i+k*] = h[i+k*] + h[i-k*] - 1

Wait, that's the negative gradient. So we want to move h in the direction:
h[i] += step * ((1 - h[i-k*]) - h[i+k*])

This increases h[i] when h[i-k*] is low (meaning h[i-k*] contributes little to the mismatch) and h[i+k*] is high (meaning the term h[i+k*](1-h[i]) is large and we want to reduce it by increasing h[i]).

This makes sense! If h[i+k*] is high and h[i] is low, the term h[i+k*](1-h[i]) is large. Increasing h[i] reduces this term.

So the subgradient step is:
h[i] += step * (1 - h[i-k*] - h[i+k*]) for the active lag k*

But we need to handle multiple active lags and the boundary terms carefully.

Let me implement a "multi-active" subgradient method:
1. Find all k where c[k] is within ε of the maximum
2. Compute the negative gradient of the average of these c[k]'s
3. Take a step in that direction
4. Project to [0,1] and sum constraint
5. Adapt the step size

This could be very effective for the final polishing stage.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def subgrad_polish(h, n, deadline, tol=1e-6):
    """Subgradient method targeting active peaks."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    patience = 50
    no_improve = 0
    
    while time.time() < deadline:
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = c.max()
        if cur_max < best_c:
            best_c = cur_max
            no_improve = 0
            step = min(step * 1.2, 0.05)
        else:
            no_improve += 1
            if no_improve > patience:
                step *= 0.7
                no_improve = 0
                if step < 1e-8:
                    break
        
        # Find active lags
        active = np.where(c > cur_max - tol * cur_max)[0]
        
        # Compute negative gradient (average over active lags)
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)  # actual lag
            # dc[k]/dh[i] for each i
            # c[k] = sum_m h[m+k]*(1-h[m])
            # d/dh[i]: (1-h[i-k]) if i-k valid (from h[i] in first factor)
            #           -h[i+k] if i+k valid (from h[i] in second factor)
            for i in range(n):
                g = 0.0
                ik = i - k
                ikp = i + k
                if 0 <= ik < n and 0 <= i < n:
                    # h[i] appears as h[m+k] when m=i-k: term h[i]*(1-h[i-k])
                    g += (1 - h[ik])
                if 0 <= ikp < n and 0 <= i < n:
                    # h[i] appears as h[m] when m=i: term h[i+k]*(1-h[i])
                    g += (-h[ikp])
                grad[i] += g
        
        grad /= len(active)
        
        # Normalize and scale
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = -grad / gnorm * step
        
        h_new = h + grad
        h_new = project_feasible(h_new, n)
        
        new_c = compute_c5(h_new, n)
        if new_c <= best_c + 1e-12:
            h = h_new
            if new_c < best_c - 1e-15:
                best_c = new_c
        # else: don't update h (keep best)
    
    return h, best_c
```
