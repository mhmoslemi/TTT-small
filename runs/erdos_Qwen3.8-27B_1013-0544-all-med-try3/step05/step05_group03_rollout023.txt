def project_feasible(h, n):

    for _ in range(2000):

    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        used = np.zeros(n, dtype=bool)
        for idx in order:
            if used[idx]:
                continue
            j_idx = None
            start = int(idx) + 1
            if start >= n:
                start = 0
            for k in range(n):
                j_c = (start + k) % n
                if j_c != int(idx) and not used[j_c]:
                    j_idx = j_c
                    break
            if j_idx is None:
                break
            used[idx] = True
            used[j_idx] = True
            i, j = int(idx), int(j_idx)
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            a, b = lo, hi
            c1 = a + (1.0 - gr) * (b - a)
            c2 = a + gr * (b - a)
            h[i] = c1; h[j] = s - c1
            f1 = compute_c5(h, n)
            h[i] = c2; h[j] = s - c2
            f2 = compute_c5(h, n)
            for _ in range(inner_iters):
                if b - a < 1e-10:
                    break
                if f1 <= f2:
                    b = c2
                else:
                    a = c1
                c1 = a + (1.0 - gr) * (b - a)
                c2 = a + gr * (b - a)
                h[i] = c1; h[j] = s - c1
                f1 = compute_c5(h, n)
                h[i] = c2; h[j] = s - c2
                f2 = compute_c5(h, n)
            x_opt = (a + b) * 0.5
            h[i] = x_opt; h[j] = s - x_opt
            new_c = compute_c5(h, n)
            if new_c < best_c - 1e-15:
                best_c = new_c
                improved = True
        if not improved and sweep > 3:
            break
    return h, best_c
