import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_swap_search(h_bin, n, deadline, max_iter=5000000):
    """Simulated annealing on binary representation."""
    dx = 2.0 / n
    g = compute_g(h_bin.astype(float), n)
    current_c = g.max()
    best_c = current_c
    best_h = h_bin.copy()
    
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    
    rng = np.random.RandomState(0)
    T = 0.001  # initial temperature
    T_min = 1e-7
    cool = 0.9999
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            T = max(T_min, T * cool)
        
        # Pick random swap
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        o = ones[oi]
        z = zeros[zi]
        
        # Compute effect of swapping o and z
        # h[o] goes 1→0, h[z] goes 0→1
        # Effect on g: for each lag m, g[m] changes by:
        #   -contribution of h[o]=1 at position o
        #   +contribution of h[z]=1 at position z
        # The contribution of h[j]=1 at position j to g[m] is:
        #   (1-h[j+d_m]) if j+d_m valid, minus h[j-d_m] if j-d_m valid
        # where d_m = m - (n-1)

        h_test = h_bin.copy()
        h_test[o] = 0.0
        h_test[z] = 1.0
        g_new = compute_g(h_test, n)
        c_new = g_new.max()
        
        if c_new < current_c - 1e-14:
            # Always accept
            h_bin[:] = h_test
            g = g_new
            current_c = c_new
            ones[oi] = z
            zeros[zi] = o
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()

            delta = c_new - current_c
            if delta < 0 or rng.random() < np.exp(-delta / T):

                g = g_new
                current_c = c_new
                ones[oi] = z
                zeros[zi] = o
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_bin.copy()

Let me try yet another approach: **Use the subgradient method with a smart step size.**

∇f(h) at a non-differentiable point is any convex combination of ∇g_m for m in argmax.

where s_t is a subgradient and α_t is the step size.

With α_t = C / sqrt(t) (or using Polyak step size if we have a lower bound), this converges.

def subgradient_method(h, n, deadline, max_iter=100000):
    dx = 2.0 / n
    dms = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        if current_max < best_c:
            best_c = current_max
            best_h = h.copy()
        
        # Subgradient: gradient of the active Lag(s)
        # Find all lags achieving the max (within tolerance)
        active = np.where(g >= current_max - 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(g)])
        
        # Average gradient over active lags
        s = np.zeros(n)
        for m in active:
            d = dms[m]
            for p in range(n):
                if 0 <= p + d < n:
                    s[p] += (1 - h[p + d])
                if 0 <= p - d < n:
                    s[p] -= h[p - d]
        s *= dx / len(active)
        
        # Step size
        alpha = 0.01 / np.sqrt(t)
        
        h_new = h - alpha * s
        h_new = project_feasible(h_new, n)
        h = h_new
    
    return best_h, best_c

Let me implement FISTA:

f_smooth(h) = (1/α) * log(Σ_k exp(α * g_k(h)))

∇f_smooth(h) = (1/Σ_k exp(α*g_k)) * Σ_k α * exp(α*g_k) * ∇g_k(h)
             = Σ_k w_k * ∇g_k(h)  where w_k = exp(α*g_k) / Σ_j exp(α*g_j)

This is a weighted average of the individual gradients, with weights given by softmax.

Computing ∇f_smooth:
- Compute g = correlate(h, 1-h) * dx → O(n)
- Compute weights w_k → O(n)
- Compute ∇g_k for each k and combine → O(n²)

Total per iteration: O(n²). With FISTA's O(1/t²) convergence, we need about O(sqrt(n)) iterations to get good accuracy, which is very feasible.

Wait, the convergence of FISTA is O(f(x_0) - f*) / t² for the function value, but this is for the smoothed version. As α → ∞, the smoothed version approaches the true max, but the condition number gets worse.

    for _ in range(200):

        if abs(diff) < 1e-12:

            s = room.sum()
            if s < 1e-14:

            s = h.sum()
            if s < 1e-14:

    return np.clip(h, 0.0, 1.0)

    g = compute_g(h, n)
    # Log-sum-exp for numerical stability
    g_max = g.max()

    log_sum = np.log(np.exp(shifted).sum())

        pd2 = p - d
        valid2 = (pd2 >= 0) & (pd2 < n)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            s = mask.sum()
            if s < 1e-14:
                break
            h[mask] += diff / s
        else:
            mask = h > 0.0
            s = mask.sum()
            if s < 1e-14:
                break
            h[mask] += diff / s
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    dx = 2.0 / n
    dms = np.arange(2 * n - 1) - (n - 1)
    A = np.zeros((n, 2 * n - 1))
    p = np.arange(n)
    for m in range(2 * n - 1):
        d = dms[m]
        pd = p + d
        v1 = (pd >= 0) & (pd < n)
        A[v1, m] += (1.0 - h[pd[v1]])
        pd2 = p - d
        v2 = (pd2 >= 0) & (pd2 < n)
        A[v2, m] -= h[pd2[v2]]
    A *= dx
    return A

def incremental_A_update(A, n, i, j, delta):
    dx = 2.0 / n
    dms = np.arange(2 * n - 1) - (n - 1)
    for m in range(2 * n - 1):
        d = dms[m]
        ip = i - d
        if 0 <= ip < n:
            A[ip, m] -= dx * delta
        ip2 = i + d
        if 0 <= ip2 < n:
            A[ip2, m] -= dx * delta
        jp = j - d
        if 0 <= jp < n:
            A[jp, m] += dx * delta
        jp2 = j + d
        if 0 <= jp2 < n:
            A[jp2, m] += dx * delta

def smooth_grad(h, n, alpha):
    g = compute_g(h, n)
    g_max = g.max()
    shifted = alpha * (g - g_max)
    log_sum = np.log(np.exp(shifted).sum())
    w = np.exp(shifted - log_sum)
    dms = np.arange(2 * n - 1) - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    p = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        d = dms[m]
        pd = p + d
        v1 = (pd >= 0) & (pd < n)
        grad[v1] += wm * (1.0 - h[pd[v1]])
        pd2 = p - d
        v2 = (pd2 >= 0) & (pd2 < n)
        grad[v2] -= wm * h[pd2[v2]]
    grad *= dx
    return grad

def fista_smoothed(h, n, alpha, deadline, max_iter=3000):
    x = h.copy()
    y = h.copy()
    t_val = 1.0
    L = max(2.0 * alpha / np.sqrt(n), 0.5)
    step = 1.0 / L
    best_c = compute_c5(x, n)
    best_x = x.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad_y = smooth_grad(y, n, alpha)
        x_new = project_feasible(y - step * grad_y, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        x = x_new
        t_val = t_new
        if it % 300 == 0:
            c = compute_c5(x, n)
            if c < best_c:
                best_c = c
                best_x = x.copy()
    c = compute_c5(x, n)
    if c < best_c:
        best_c = c
        best_x = x.copy()
    return best_x, best_c

def pair_descent(h, n, deadline, max_sweeps=500, K=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    A = compute_A(h, n)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        for i in range(n):
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                A_diff = A[i] - A[j]
                # Quick reject: if top lag would increase, skip
                if A_diff[int(top_idx[0])] > 1e-12:
                    continue
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(a + 1, K):
                        ia = int(top_idx[a])
                        ib = int(top_idx[b])
                        ds = A_diff[ib] - A_diff[ia]
                        if abs(ds) > 1e-15:
                            dc = (g[ia] - g[ib]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * A_diff))
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * A_diff
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    incremental_A_update(A, n, i, j, best_delta)
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c

        else:

    return best_h, best_c
