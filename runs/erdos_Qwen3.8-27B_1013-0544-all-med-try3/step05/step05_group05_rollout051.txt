        return m + (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum())) / alpha

        vals = g[active_lags]

def precompute_contribs(a, n, active_k):
    """For each position p and each active lag k, compute a[p+k] + a[p-k]."""
    # active_k: array of lag values (can be negative)

        for p in range(n):

            if 0 <= p+k < n: c += a[p+k]
            if 0 <= p-k < n: c += a[p-k]
            contribs[p, l] = c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    a = np.zeros(n)
    a[idx[:target]] = 1.0
    return a

def top_lags_idx(g, m):
    order = np.argsort(g)[::-1]
    return order[:m]

def precompute_contribs(a, n, active_k):
    m = len(active_k)
    contribs = np.zeros((n, m))
    for l in range(m):
        k = active_k[l]
        for p in range(n):
            c = 0.0
            pk = p + k
            pm = p - k
            if 0 <= pk < n:
                c += a[pk]
            if 0 <= pm < n:
                c += a[pm]
            contribs[p, l] = c
    return contribs

def binary_local_search_v2(a, n, deadline, max_sweeps=5000, m_start=3, seed=42):
    g = compute_all_g(a, n)
    cur_max = g.max()
    dx = 2.0 / n
    offset = n - 1
    rng = np.random.RandomState(seed)
    m_active = m_start
    stuck = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        active_idx = top_lags_idx(g, m_active)
        active_k = active_idx - offset
        contribs = precompute_contribs(a, n, active_k)
        ones = np.where(a == 1)[0]
        zeros = np.where(a == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in ones:
            if time.time() > deadline:
                break
            val_i = g[active_idx] + contribs[i] * dx
            diff = val_i[None, :] - contribs[zeros] * dx
            new_maxes = np.max(diff, axis=1)
            candidates = np.where(new_maxes < cur_max - 1e-14)[0]
            if len(candidates) > 0:
                j = zeros[candidates[0]]
                a[i] = 0.0
                a[j] = 1.0
                g = compute_all_g(a, n)
                cur_max = g.max()
                improved = True
                break
        if not improved:
            stuck += 1
            if stuck <= 5:
                m_active = min(m_active + 2, 25)
            elif stuck <= 10:
                idx1 = np.where(a == 1)[0]
                idx0 = np.where(a == 0)[0]
                p = int(rng.randint(len(idx1)))
                q = int(rng.randint(len(idx0)))
                a[idx1[p]], a[idx0[q]] = a[idx0[q]], a[idx1[p]]
                g = compute_all_g(a, n)
                cur_max = g.max()
                stuck = 0
            else:
                break
        else:
            stuck = 0
    return a, cur_max

def slsqp_grad_active(x, n, active_idx, alpha):
    dx = 2.0 / n
    offset = n - 1
    g_raw = np.correlate(x, 1.0 - x, mode='full')
    g = g_raw * dx
    vals = g[active_idx]
    m = vals.max()
    shifted = vals - m
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    grad = np.zeros(n)
    for idx, k_idx in enumerate(active_idx):
        k = k_idx - offset
        wi = w[idx] * dx
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += wi * (1.0 - x[k:n])
            if k > 0:
                grad[k:] -= wi * x[:nk]
        else:
            k2 = -k
            nk = n - k2
            if nk > 0:
                grad[k2:] += wi * (1.0 - x[:nk])
            if k2 > 0:
                grad[:nk] -= wi * x[k2:]
    return vals, grad

def slsqp_active_run(h0, n, active_idx, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        vals = g[active_idx]
        m = vals.max()
        sh = vals - m
        return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

    def obj_grad(x):
        _, grad = slsqp_grad_active(x, n, active_idx, alpha)
        return grad

    bounds = [(0.0, 1.0)] * n
    ones_n = np.ones(n)

    def cons_fun(x):
        return x.sum() - n / 2.0

    def cons_jac(x):
        return ones_n

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgrad_step(h, n, deadline, max_iter=50000, m_active=5):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    offset = n - 1
    lr = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        active_idx = top_lags_idx(g, m_active)
        active_k = active_idx - offset
        grad = np.zeros(n)
        for k_idx in active_idx:
            k = k_idx - offset
            if k >= 0:
                nk = n - k
                if nk > 0:
                    grad[:nk] += dx * (1.0 - best_h[k:n])
                if k > 0:
                    grad[k:] -= dx * best_h[:nk]
            else:
                k2 = -k
                nk = n - k2
                if nk > 0:
                    grad[k2:] += dx * (1.0 - best_h[:nk])
                if k2 > 0:
                    grad[:nk] -= dx * best_h[k2:]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                best_h = ht
                found = True
                break
        if not found:
            lr *= 0.85
            if lr < 1e-14:
                break
        else:
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
    return best_h, best_c

def thue_morse_init(n, seed=0):
    a = np.zeros(n)
    for i in range(n):
        a[i] = float(bin(i).count('1') % 2)
    target = n // 2
    ones = int(a.sum())
    if ones > target:
        idx1 = np.where(a == 1)[0]
        idx0 = np.where(a == 0)[0]
        rng = np.random.RandomState(seed)
        rng.shuffle(idx1)
        for k in range(ones - target):
            a[idx1[k]] = 0.0
    elif ones < target:
        idx0 = np.where(a == 0)[0]
        rng = np.random.RandomState(seed)
        rng.shuffle(idx0)
        for k in range(target - ones):
            a[idx0[k]] = 1.0
    return a
