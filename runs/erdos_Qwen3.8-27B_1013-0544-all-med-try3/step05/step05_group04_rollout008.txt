import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    return float(m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha)

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    sm = np.exp(shifted - smax)
    p = sm / sm.sum()
    s_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        idx_b = j + s_vals
        mask_b = (idx_b >= 0) & (idx_b < n)
        term_b = np.dot(p[mask_b], x[idx_b[mask_b]])
        idx_a = j - s_vals
        mask_a = (idx_a >= 0) & (idx_a < n)
        term_a = np.dot(p[mask_a], x[idx_a[mask_a]])
        grad[j] = dx * (1.0 - term_a - term_b)
    return grad

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h.copy(), jac=lse_grad, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 1e10:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h.copy()

def fine_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_opt(h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht

    remaining = deadline - time.time()
    if remaining > 5 and n <= 300:
        h, best_c = fine_pair_search(h, n, deadline, [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    elif remaining > 5:
        h, best_c = near_pair_search(h, n, deadline, [0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def structured_starts(n, rng):
    starts = []
    x = np.linspace(0, 2, n)
    h1 = np.where(x < 1.0, 1.0, 0.0)
    starts.append(project_feasible(h1, n))
    h2 = np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1)
    starts.append(project_feasible(h2, n))
    h3 = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x), 0, 1)
    starts.append(project_feasible(h3, n))
    h4 = np.clip(0.5 + rng.normal(0, 0.3, n), 0, 1)
    starts.append(project_feasible(h4, n))
    return starts
