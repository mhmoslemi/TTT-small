    shifted = alpha * (c - m)
    # Clip for numerical stability

import numpy as np
import time

def _compute_overlap(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, n):
    return float(np.max(_compute_overlap(h, n)))

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-14: break
            h += d/t * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += d/t * h
    return np.clip(h, 0.0, 1.0)

def _grad(h, n, alpha):
    """Return gradient of smooth max overlap w.r.t. h."""
    dx = 2.0 / n
    O = _compute_overlap(h, n)
    m = O.max()
    s = np.clip(alpha*(O - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(n)
    nl1 = n - 1
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-14: continue
        lag = k - nl1
        if lag >= 0:
            g[:n-lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:n-lag]
        else:
            ll = -lag
            g[ll:] += w * (1.0 - h[:n-ll])
            g[:n-ll] -= w * h[ll:]
    return dx * g
