def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full')
    g = g * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_grad(x, n, alpha):
    dx = 2.0 / n
    F = compute_F(x, n)
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w = w / w.sum()
    return (F @ w) / alpha

def slsqp_cons(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': lambda x, n=n: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), jac=lambda x: slsqp_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP warm start
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, p1, maxiter=100)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Binary steepest descent
    p2 = max(p1 + 1, t0 + 0.45 * (budget_s - 5))
    h_bin = np.zeros(n)
    h_bin[np.argsort(best_h)[::-1][:n // 2]] = 1.0
    h_bin, c_bin = binary_steepest(h_bin, n, p2)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    # Phase 3: Pair descent from best
    p3 = max(p2 + 1, t0 + 0.78 * (budget_s - 5))
    h_pd = best_h.copy()
    h_pd, c_pd = pair_descent(h_pd, n, p3, K_top=16)
    if c_pd < best_c:
        best_c = c_pd
        best_h = h_pd.copy()
    # Phase 4: Random kicks + pair descent
    p4 = max(p3 + 1, t0 + 0.92 * (budget_s - 5))
    for kick in range(10):
        if time.time() > p4:
            break
        h_kick = best_h.copy()
        idx = rng.choice(n, size=n // 10, replace=False)
        for ii in idx:
            h_kick[ii] = float(np.clip(h_kick[ii] + rng.randn() * 0.3, 0.0, 1.0))
        h_kick = project_feasible(h_kick, n)
        # Round to binary for kick
        h_kick_bin = np.zeros(n)
        h_kick_bin[np.argsort(h_kick)[::-1][:n // 2]] = 1.0
        h_kick_bin, c_kb = binary_steepest(h_kick_bin, n, min(time.time() + 5, p4))
        if c_kb < best_c:
            best_c = c_kb
            best_h = h_kick_bin.copy()
        # Try pair descent from kicked binary
        if time.time() < p4 - 2:
            h_kick_pd, c_kp = pair_descent(h_kick_bin.copy(), n, p4, K_top=12)
            if c_kp < best_c:
                best_c = c_kp
                best_h = h_kick_pd.copy()
    # Phase 5: Final SLSQP polish
    if time.time() < dl - 20:
        h_fin = slsqp_optimize(best_h.copy(), n, 1000000, dl - 20, maxiter=150)
        c_fin = compute_c5(h_fin, n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
