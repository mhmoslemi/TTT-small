import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, shift):
    dx = 2.0 / n
    j = np.arange(n)
    g = np.zeros(n)
    v1 = (j + shift >= 0) & (j + shift < n)
    g[v1] += (1.0 - h[np.clip(j[v1] + shift, 0, n - 1)])
    v2 = (j - shift >= 0) & (j - shift < n)
    g[v2] -= h[np.clip(j[v2] - shift, 0, n - 1)]
    return g * dx

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx)) + 1e-300) / alpha)

    if time.time() > deadline:
        return h
    return project_feasible(res.x, n)

            h[j] -= d
