def project_onto_feasible(h_prime, n):
    """Project h_prime onto {h : 0<=h[i]<=1, sum(h)=n/2}."""
    target = n / 2.0
    
    def sum_clipped(t):
        return np.clip(h_prime + t, 0, 1).sum()
    
    # Find t such that sum_clipped(t) = target
    # sum_clipped is non-decreasing in t
    # If sum_clipped(0) = target, return clip(h_prime, 0, 1)
    # Otherwise, bisect
    
    lo, hi = -max(abs(h_prime)), max(abs(1 - h_prime)) + 1
    # Expand range if needed
    while sum_clipped(lo) > target:
        lo *= 2
    while sum_clipped(hi) < target:
        hi *= 2
    
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clipped(mid) > target:
            hi = mid
        else:
            lo = mid
    
    t = (lo + hi) / 2
    return np.clip(h_prime + t, 0, 1)

                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, step, radius=30, max_passes=3000):
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            return h, best_c
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if h[i] + step > 1.0:
                continue
            j_lo = max(0, i - radius)
            j_hi = min(n, i + radius + 1)
            for j in range(j_lo, j_hi):
                if j == i or h[j] < step:
                    continue
                h[i] += step
                h[j] -= step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i] -= step
                    h[j] += step
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)
    n = n_init

    # Phase 1: PGD with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    for a in alphas:
        if time.time() > deadline - 120:
            break
        alloc = min(80, (deadline - time.time() - 60) * 0.25)
        if alloc < 8:
            break
        h_opt, c_opt = pgd_optimize(best_h, n, a, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Gradient-guided pair search
    if time.time() < deadline - 60:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        remaining = (deadline - time.time() - 30) * 0.3
        alloc = min(remaining, 150)
        if alloc > 10:
            best_h, best_c = grad_pair_search(best_h, n, time.time() + alloc, steps)

    # Phase 3: Near pair search for local refinement
    if time.time() < deadline - 20:
        steps = [0.001, 0.0005, 0.0002, 0.0001]
        for s in steps:
            if time.time() > deadline - 10:
                break
            best_h, best_c = near_pair_search(best_h, n, time.time() + 30, s, radius=40)

    # Phase 4: Try different n values
    for n_try in [200, 300, 150]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 40
        if remaining < 40:
            break
        for a in [10000, 100000, 1000000]:
            if time.time() > deadline - 30:
                break
            h_opt, c_opt = pgd_optimize(h_interp, n_try, a, time.time() + min(50, remaining * 0.2))
            if c_opt < c_opt and False:
                pass
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                n = n_try
                h_interp = h_opt

    # Phase 5: Final PGD + pair search on best solution
    n = len(best_h)
    if time.time() < deadline - 15:
        alloc = min(deadline - time.time() - 8, 80)
        best_h, best_c = pgd_optimize(best_h, n, 10000000, time.time() + alloc)

    if time.time() < deadline - 8:
        steps = [0.0005, 0.0002, 0.0001, 0.00005]
        best_h, best_c = grad_pair_search(best_h, n, deadline - 3, steps, top_k=40)

    # Phase 6: Random restarts with remaining time
    if time.time() < deadline - 20:
        remaining = deadline - time.time() - 10
        n_restarts = max(1, int(remaining / 40))
        scales = [0.01, 0.025, 0.05, 0.08]
        for r in range(n_restarts):
            if time.time() > deadline - 10:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 8) * 0.6)
            if alloc < 10:
                break
            h_opt, c_opt = pgd_optimize(hp, n, 5000000, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = len(best_h)

    # Final polish
    if time.time() < deadline - 4:
        n = len(best_h)
        best_h, best_c = near_pair_search(best_h, n, deadline - 1, 0.0001, radius=50)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, float(best_c), len(best_h)
