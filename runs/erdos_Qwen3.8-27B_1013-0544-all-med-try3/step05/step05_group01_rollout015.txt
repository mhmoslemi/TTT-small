from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(0.0, h * (1.0 + diff / tot))
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, top_k=5):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    return float(c[idx].mean())

def topk_grad(x, n, top_k=5):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        j = k - (n - 1)
        # dc/dh[a] = (1-h[a+j]) * [0<=a+j<n] - h[a-j] * [0<=a-j<n]
        # For a in [0, n-1]:
        # term1: (1-h[a+j]) if 0 <= a+j < n
        # term2: -h[a-j] if 0 <= a-j < n
        a = np.arange(n)
        # term1
        mask1 = (a + j >= 0) & (a + j < n)
        grad[mask1] += (1 - x[a[mask1] + j])
        # term2
        mask2 = (a - j >= 0) & (a - j < n)
        grad[mask2] -= x[a[mask2] - j]
    grad *= dx / top_k
    return grad

def lbfgs_optimize(h, n, top_k, deadline, maxiter=5000):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return topk_obj(x, n, top_k)
    
    def grad(x):
        return topk_grad(x, n, top_k)
    
    # L-BFGS-B doesn't handle equality constraints well
    # Let's use a penalty for the sum constraint
    penalty = 100.0
    
    def obj_pen(x):
        base = topk_obj(x, n, top_k)
        sum_err = x.sum() - n / 2.0
        return base + penalty * sum_err ** 2

        base_grad = topk_grad(x, n, top_k)
        sum_err = x.sum() - n / 2.0
        return base_grad + 2 * penalty * sum_err * np.ones(n)

def bottleneck_search(h, n, deadline, max_iter=5000):

    best_c = compute_c5(h, n)
    a = np.arange(n)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        k_star = np.argmax(c)
        j_off = k_star - (n - 1)
        # Compute gradient of c_{k_star}
        # dc/dh[i] = (1-h[i+j_off])*[valid] - h[i-j_off]*[valid]
        grad = np.zeros(n)
        m1 = (a + j_off >= 0) & (a + j_off < n)
        grad[m1] = (1.0 - h[a[m1] + j_off])
        m2 = (a - j_off >= 0) & (a - j_off < n)
        grad[m2] -= h[a[m2] - j_off]
        # Find pair (i,j) to transfer from i to j that reduces c_{k_star}
        # We want grad[i] - grad[j] to be as negative as possible
        # (transferring from i to j changes c by d*(grad[i] - grad[j]))
        # So we want grad[i] < grad[j], i.e., transfer from low-grad to high-grad
        # Wait: if we do h[i] += d, h[j] -= d, then Δc = d*(grad[i] - grad[j])
        # To decrease c, we need grad[i] - grad[j] < 0, i.e., grad[i] < grad[j]
        # So we transfer from i (low grad) to j (high grad)? No wait...
        # Actually: Δc = d * grad[i] + (-d) * grad[j] = d * (grad[i] - grad[j])
        # To decrease c: need d * (grad[i] - grad[j]) < 0
        # If d > 0: need grad[i] < grad[j]
        # So: increase h[i] (where grad is low) and decrease h[j] (where grad is high)
        
        # Find the pair with the most negative (grad[i] - grad[j]) subject to 0<=h[i]+d<=1, 0<=h[j]-d<=1
        # For d > 0: need h[i] < 1 and h[j] > 0
        # Best pair: min grad[i] (among h[i]<1) and max grad[j] (among h[j]>0)
        
        # But we also need to check that this doesn't increase other c_k above max
        # For now, let's just try the greedy move and check
        
        # Find i with smallest grad (where h[i] < 1 - epsilon)
        # Find j with largest grad (where h[j] > epsilon)
        eps = 1e-10

        valid_j = np.where(h > eps)[0]
        if len(valid_i) == 0 or len(valid_j) == 0:
            break
        
        i_best = valid_i[np.argmin(grad[valid_i])]
        j_best = valid_j[np.argmax(grad[valid_j])]
        
        if i_best == j_best:
            # Try second best
            break
        
        # Determine step size
        max_d = min(1.0 - h[i_best], h[j_best])
        for d in [0.01, 0.005, 0.001, 0.0001, min(0.01, max_d*0.5)]:

            else:

                h[j_best] += d

            # No improvement with any step size, try a different approach
            # Pick random pairs
            break

    h = np.clip(h.copy(), 0.0, 1.0)

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

        else:
            tot = h.sum()

                break
            h = np.maximum(0.0, h * (1.0 + diff / tot))

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    cap = shifted - shifted.max()
    return m + np.log(np.sum(np.exp(cap))) / alpha

    m = c.max()
    shifted = alpha * (c - m)
    cap = shifted - shifted.max()

    grad = np.zeros(n)
