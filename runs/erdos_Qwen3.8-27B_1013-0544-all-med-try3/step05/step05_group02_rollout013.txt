def project_feasible_bisect(v, n):
    """Project v onto {h : 0 <= h <= 1, sum(h) = n/2} using bisection on the simplex constraint."""
    S = n / 2.0
    # We want to find h_i = clip(v_i - lambda, 0, 1) such that sum(h) = S
    # As lambda decreases, sum increases (since more v_i - lambda become positive and less get clipped at 1)
    # Find lambda by bisection
    
    def sum_at_lambda(lam):
        h = np.clip(v - lam, 0, 1)
        return h.sum()
    
    # Lambda range: if lam -> -inf, h -> 1 for all, sum -> n
    # if lam -> +inf, h -> 0 for all, sum -> 0
    # We want sum = S = n/2, so lambda is somewhere in the range
    
    lo = v.min() - S  # conservative lower bound
    hi = v.max() + S  # conservative upper bound
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > S:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fix any tiny numerical error
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        # Add/subtract from elements not at boundary
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

        top_lags = np.argsort(c)[-top_n:]
        g = np.zeros(n)
        for k in top_lags:
            g += subgrad_one(h, n, int(k))
        if g.max() - g.min() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        t_max = min(h[i_src], 1 - h[i_dst])
        if t_max < 1e-10:
            # Try second best
            g2 = g.copy()
            g2[i_src] = -np.inf
            g2[i_dst] = np.inf
            i_src2 = int(np.argmax(g2))
            i_dst2 = int(np.argmin(g2))
            if i_src2 == i_dst2:
                stall += 1
                if stall > 200:
                    break
                continue
            t_max = min(h[i_src2], 1 - h[i_dst2])
            i_src, i_dst = i_src2, i_dst2
            if t_max < 1e-10:
                stall += 1
                if stall > 200:
                    break
                continue
        accepted = False
        for frac in fracs:
            t = frac * t_max
            if t < 1e-10:
                continue
            h[i_src] -= t
            h[i_dst] += t
            c_new = compute_c5(h, n)
            if c_new < curr - 1e-14:
                accepted = True
                break
            h[i_src] += t
            h[i_dst] -= t
        if not accepted:
            h[i_src] += t
            h[i_dst] -= t
            stamp = np.where(g > 0)[0]
            if len(stamp) < 2:
                stall += 1
                if stall > 300:
                    break
                pi = int(rng_state.choice(stamp))
            else:
                pi = int(rng_state.choice(stamp))
            dn = np.where(g < 0)[0]
            if len(dn) < 1:
                stall += 1
                if stall > 300:
                    break
                di = int(rng_state.choice(dn))
            else:
                di = int(rng_state.choice(dn))
            if pi == di:
                stall += 1
                if stall > 300:
                    break
                continue
            t2 = 0.03 * min(h[pi], 1 - h[di])
            if t2 < 1e-10:
                stall += 1
                if stall > 300:
                    break
                continue
            h[pi] -= t2
            h[di] += t2
            c_new = compute_c5(h, n)
            if c_new >= curr:
                h[pi] += t2
                h[di] -= t2
            stall += 1
            if stall > 300:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global rng_state
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng_state = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = project_feasible(np.full(250, 0.5), 250)
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Phase 1: Smooth descent on initial
    h = project_feasible(h_init, n_init)
    alloc1 = min(200, (deadline - t0) * 0.25)
    h, c = smooth_descent(h, n_init, t0 + alloc1)
    best_h, best_c = h.copy(), c
    print(f"Phase1 n={n_init}: {best_c:.6f}")

    # Phase 2: Transfer polish
    if time.time() < deadline - 30:
        h, c = transfer_polish(h, n_init, min(deadline - 30, time.time() + 180))
        if c < best_c:
            best_h, best_c = h.copy(), c
        print(f"Phase2: {best_c:.6f}")

    # Phase 3: Try other n
    for n_try in [200, 300, 150, 350]:
        if n_try == n_init or time.time() > deadline - 100:
            continue
        x_o = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_try)
        hi = np.interp(x_n, x_o, best_h)
        hi = project_feasible(hi, n_try)
        alloc3 = min(100, (deadline - time.time()) * 0.2)
        if alloc3 < 40:
            break
        ht, ct = smooth_descent(hi, n_try, time.time() + alloc3)
        if time.time() < deadline - 20:
            ht, ct = transfer_polish(ht, n_try, min(deadline - 20, time.time() + 60))
        if ct < best_c:
            best_h, best_c = ht.copy(), ct
            print(f"Phase3 n={n_try}: {best_c:.6f}")

    # Phase 4: Restart perturbations
    if time.time() < deadline - 40:
        nb = len(best_h)
        n_restarts = max(3, int((deadline - time.time() - 20) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.005 + 0.03 * (r / max(1, n_restarts - 1))
            hp = best_h + rng_state.normal(0, scale, nb)
            hp = project_feasible(hp, nb)
            alloc = min(50, (deadline - time.time() - 15) * 0.35)
            if alloc < 20:
                break
            hr, cr = smooth_descent(hp, nb, time.time() + alloc)
            if time.time() < deadline - 10:
                hr, cr = transfer_polish(hr, nb, deadline - 8)
            if cr < best_c:
                best_h, best_c = hr.copy(), cr
                print(f"Restart {r}: {best_c:.6f}")

    # Final polish
    if time.time() < deadline - 3:
        h = project_feasible(best_h, len(best_h))
        h, c = transfer_polish(h, len(best_h), deadline - 1)
        if c < best_c:
            best_h, best_c = h.copy(), c

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    print(f"FINAL: C5={best_c:.6f}, n={len(best_h)}")
    return best_h, best_c, len(best_h)
