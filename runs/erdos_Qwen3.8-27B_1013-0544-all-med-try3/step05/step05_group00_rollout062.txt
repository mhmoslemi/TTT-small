The key insight for speed: precompute the "profile" vectors for each position, then for each pair (i,j) the delta is just a sum of two precomputed vectors plus a point correction.

Let me define:
- For each position p, define A_p[s] = (1-h[p+s])*[0<=p+s<n] for s in range -(n-1)..(n-1)
  This is a vector of length 2n-1.
- For each position p, define B_p[s] = -h[p-s]*[0<=p-s<n]
- For each position p, define C_p[s] = -(1-h[p+s])*[0<=p+s<n] = -A_p[s]
- For each position p, define D_p[s] = h[p-s]*[0<=p-s<n] = -B_p[s]

Then for swap (i: 0→1, j: 1→0):
delta[s] = A_i[s] + B_i[s] + C_j[s] + D_j[s] + 2*[s==j-i]
         = (A_i + B_i)[s] + (C_j + D_j)[s] + 2*[s==j-i]
         = P_i[s] + Q_j[s] + 2*[s==j-i]

where P_i = A_i + B_i and Q_j = C_j + D_j = -A_j - B_j = -P_j!

Wait: C_j = -A_j and D_j = -B_j? Let me check.
A_j[s] = (1-h[j+s])*[0<=j+s<n]
C_j[s] = -(1-h[j+s])*[0<=j+s<n] = -A_j[s] ✓

B_j[s] = -h[j-s]*[0<=j-s<n]
D_j[s] = h[j-s]*[0<=j-s<n] = -B_j[s] ✓

So Q_j = -A_j - B_j = -P_j.

Therefore: delta[s] = P_i[s] - P_j[s] + 2*[s==j-i]

where P_p[s] = (1-h[p+s])*[0<=p+s<n] - h[p-s]*[0<=p-s<n]

This is beautiful! The delta for a swap (i,j) is just P_i - P_j + 2*e_{j-i} where e_k is a unit vector.

So the algorithm becomes:
1. Precompute P_p for all p (each is a vector of length 2n-1): O(n²) total
2. For each pair (i,j): delta = P_i - P_j + 2*e_{j-i}: O(1) to set up, O(n) to add to corr

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        result = project_feasible(res.x, n)
        return result
    except Exception:
        return h

def precompute_P(h, n):
    """Precompute P_p vectors for all positions p.
    P_p[s] = (1-h[p+s])*[0<=p+s<n] - h[p-s]*[0<=p-s<n]
    where s ranges from -(n-1) to (n-1), stored at index n-1+s.
    """
    P = np.zeros((n, 2*n-1))
    for p in range(n):
        # (1-h[p+s]) for s where 0<=p+s<n, i.e., s in [-p, n-1-p]
        lo = max(-p, -(n-1))
        hi = min(n-1-p, n-1)
        if lo <= hi:
            s_vals = np.arange(lo, hi+1)
            idx = n-1 + s_vals
            P[p, idx] += 1.0 - h[p + s_vals]
        # -h[p-s] for s where 0<=p-s<n, i.e., s in [p-(n-1), p]
        lo2 = max(p-(n-1), -(n-1))
        hi2 = min(p, n-1)
        if lo2 <= hi2:
            s_vals2 = np.arange(lo2, hi2+1)
            idx2 = n-1 + s_vals2
            P[p, idx2] -= h[p - s_vals2]
    return P

    dx = 2.0 / n
    target = n // 2
    # Round to binary
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    P = precompute_P(h, n)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        rng.shuffle(z_list)
        o_list = sorted(ones)
        
        for i in z_list:
            if time.time() > deadline:
                break
            if i not in zeros:
                continue
            # For this i, try all j in ones
            bi = corr + P[i]  # base for this i
            best_j = -1
            best_new_c = cur_c - 1e-15  # need strict improvement
            
            for j in o_list:
                if j not in ones:
                    continue
                # delta = P[i] - P[j] + 2*e_{j-i}
                # new_corr = corr + P[i] - P[j] + 2*e_{j-i}
                # = bi - P[j] + 2*e_{j-i}
                nc = bi - P[j]
                s_extra = j - i
                idx_extra = n - 1 + s_extra
                nc[idx_extra] += 2.0
                new_c = nc.max() * dx
                if new_c < best_new_c:
                    best_new_c = new_c
                    best_j = j
            
            if best_j >= 0:
                # Do the swap
                j = best_j
                h[i] = 1.0
                h[j] = 0.0
                corr = bi - P[j]
                corr[n-1 + (j-i)] += 2.0
                cur_c = best_new_c
                zeros.discard(i)
                zeros.add(j)
                ones.discard(j)
                ones.add(i)
                # Update P vectors (only P[i] and P[j] change)
                # P_i changes because h[i] changed from 0 to 1
                # P_p depends on h values at positions p±s
                # When h[i] changes: P_p changes if |p-i| < n (always) 
                # Specifically: P_p[s] = (1-h[p+s])*[0<=p+s<n] - h[p-s]*[0<=p-s<n]
                # h[i] appears in P_p when p+s=i (s=i-p) or p-s=i (s=p-i)

                o_list = sorted(ones)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                continue  # restart the i loop since o_list changed

            if anneal_temp > 0 and len(zeros) > 0 and len(ones) > 0:
                i = rng.choice(list(zeros))
                j = rng.choice(list(ones))
                nc = corr + P[i] - P[j]
                nc[n-1 + (j-i)] += 2.0
                new_c = nc.max() * dx
                if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / anneal_temp):
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = nc
                    cur_c = new_c
                    zeros.discard(i)
                    zeros.add(j)
                    ones.discard(j)
                    ones.add(i)
                    P = precompute_P(h, n)
                    anneal_temp *= 0.99  # cool down
    
    # Final: update best
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
