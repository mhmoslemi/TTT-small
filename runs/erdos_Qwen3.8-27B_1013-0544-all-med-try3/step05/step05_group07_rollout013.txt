def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
