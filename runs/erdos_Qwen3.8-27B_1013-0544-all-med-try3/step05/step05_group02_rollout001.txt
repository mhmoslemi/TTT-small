```python
        gb = int(np.argmax(fitnesses))
        gb_c = -fitnesses[gb]
        if gb_c < best_c - 1e-15:
            best_c = gb_c
            best_h = pop[gb].copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reseed if stuck
        if no_improve > 15:
            no_improve = 0
            for i in range(pop_size // 3):
                scale = 0.03 + 0.04 * rng.random()
                pop[i] = project_feasible(best_h + rng.normal(0, scale, n), n)
                fitnesses[i] = -compute_c5(pop[i], n)

    # Phase 3: Thorough SLSQP refinement
    if time.time() < deadline - 30:
        for a in [20000, 100000, 500000, 2000000]:
            if time.time() > deadline - 25:
                break
            ht = slsqp_refine(best_h, n, a, maxiter=800)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht

    # Phase 4: Directed polish
    if time.time() < deadline - 3:
        best_h, best_c = directed_polish(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
