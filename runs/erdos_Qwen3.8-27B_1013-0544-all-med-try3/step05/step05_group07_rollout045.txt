**The "peeling" method**:
1. Find the current max lag k*
2. Optimize h to minimize g_{k*}(h) (a single quadratic) using projected gradient
3. Re-find the max lag
4. Repeat

This is essentially a **block coordinate descent** on the (h, k) pair, where k is discrete. At each step, we minimize a quadratic (which has a known structure), then update k.

For step 2, minimizing g_k(h) = dx * ∑_{i} h[i](1-h[i+k]) subject to constraints:
- This is a quadratic with box constraints and one equality constraint
- We can use projected gradient: ∇g_k(h) is easy to compute
- Or we can use the fact that it's a quadratic and solve via KKT conditions

The projected gradient for a single lag is O(n) per step, very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    lse = smax + np.log(np.exp(shifted - smax).sum())
    return m + lse / alpha + 1e4 * (x.sum() - n / 2.0) ** 2

def smooth_grad_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    probs = np.exp(shifted - smax)
    probs /= probs.sum()
    grad = np.zeros(n)
    k_values = np.arange(-(n - 1), n)
    for idx in range(len(k_values)):
        w_k = probs[idx]
        if w_k < 1e-10:
            continue
        k = k_values[idx]
        if k >= 0:
            grad[:n - k] += w_k * dx * (1.0 - x[k:])
            grad[k:] -= w_k * dx * x[:n - k]
        else:
            j = -k
            grad[:n - j] += w_k * dx * (1.0 - x[j:])
            grad[j:] -= w_k * dx * x[:n - j]
    grad += 2e4 * (x.sum() - n / 2.0)
    return grad

def run_smooth_lbfgs(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj_full(x, n, alpha)
    def grad(x):
        return smooth_grad_full(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def peel_optimize(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 1.0
    stall = 0
    for outer in range(500000):
        if time.time() > deadline:
            break
        c_arr = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(c_arr)) - (n - 1)
        lr = max(lr, 0.01)
        for inner in range(100):
            if time.time() > deadline:
                break
            grad = np.zeros(n)
            if k_star >= 0:
                grad[:n - k_star] = dx * (1.0 - h[k_star:])
                grad[k_star:] -= dx * h[:n - k_star]
            else:
                j = -k_star
                grad[:n - j] = dx * (1.0 - h[j:])
                grad[j:] -= dx * h[:n - j]
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < compute_c5(h, n) - 1e-14:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                lr = min(lr * 1.01, 5.0)
                stall = 0
            else:
                lr *= 0.8
                if lr < 0.001:
                    break
    return best_h, best_c

def binary_local_search(h, n, deadline, rng):
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    temp = 0.005
    for it in range(200000):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i] = 1.0
        h_bin[j] = 0.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_bin.copy()
            ones.remove(j)
            ones.append(i)
            zeros = [x for x in range(n) if h_bin[x] == 0]
            temp = max(temp * 0.999, 1e-7)
        elif rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-9)):
            ones.remove(j)
            ones.append(i)
            zeros = [x for x in range(n) if h_bin[x] == 0]
            best_h = h_bin.copy()
            best_c = c_new
        else:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        temp *= 0.9997
        if temp < 1e-8:
            temp = 1e-8
    return best_h, best_c

def continuous_pair_refine(h, n, deadline, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] = hi - s * d
                                h[j] = hj + s * d
    return best_h, best_c

def make_structured_init(n, rng):
    candidates = []
    h1 = np.zeros(n)
    block = max(1, n // 10)
    for b in range(0, n, 2 * block):
        h1[b:b + block] = 1.0
    candidates.append(project_feasible(h1, n))
    h2 = np.zeros(n)
    for i in range(n):
        if (i // max(1, n // 14)) % 2 == 0:
            h2[i] = 1.0
    candidates.append(project_feasible(h2, n))
    h3 = np.sin(2 * np.pi * np.arange(n) * 3.0 / n)
    h3 = (h3 > 0).astype(float)
    candidates.append(project_feasible(h3, n))
    h4 = np.zeros(n)
    h4[rng.permutation(n)[:n // 2]] = 1.0
    candidates.append(project_feasible(h4, n))
    return candidates
