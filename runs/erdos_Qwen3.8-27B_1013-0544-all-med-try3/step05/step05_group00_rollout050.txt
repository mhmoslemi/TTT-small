    """Compute delta in correlation when h[i]: 0->1, h[j]: 1->0.
    corr = np.correlate(h, 1-h, mode='full')
    Returns delta vector of length 2n-1.
    """
    ones_minus_h = 1.0 - h
    delta = np.zeros(2 * n - 1)
    
    # Effect of h[i] going from 0 to 1 (first arg changes at index i)
    # corr[k] = sum_m h[m] * (1-h)[m - k + n - 1]
    # Adding h[i]=1: corr[k] += (1-h)[i - k + n - 1]
    # Valid when 0 <= i - k + n - 1 < n, i.e., n-1-i <= k <= n-1+i

    for k in range(max(0, n-1-i), min(2*n-1, n+i)):
        idx = i - k + n - 1
        delta[k] += ones_minus_h[idx]
    
    # Effect of (1-h)[i] going from 1 to 0 (second arg changes at index i)
    # corr[k] = sum_m h[m] * (1-h)[m - k + n - 1]

def get_swap_delta(h, n, i, j):
    """Compute delta in correlation for swap h[i]:0->1, h[j]:1->0.
    Assumes h[i]=0 and h[j]=1."""
    g = 1.0 - h  # g = 1-h
    d = np.zeros(2*n - 1)
