            g = compute_g(h, n)
            infl = compute_influence(h, n)
            temp = 0.03
    return best_h, best_c

def pair_descent(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    g = compute_g(h, n)
    best_c = float(g.max())
    best_h = h.copy()
    dm = np.arange(2 * n - 1) - (n - 1)
    K = min(12, 2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argsort(g)[::-1][:K]
        sweep_imp = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = np.zeros(2 * n - 1)
                v1 = (i + dm >= 0) & (i + dm < n)
                s[v1] -= 1.0 - h[(i + dm)[v1]]
                v1b = (i - dm >= 0) & (i - dm < n)
                s[v1b] -= h[(i - dm)[v1b]]
                v2 = (j + dm >= 0) & (j + dm < n)
                s[v2] += 1.0 - h[(j + dm)[v2]]
                v2b = (j - dm >= 0) & (j - dm < n)
                s[v2b] += h[(j - dm)[v2b]]
                if np.allclose(s, 0):
                    continue
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = float(g.max())
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    sweep_imp += 1
        h = project_feasible(h, n)
        g = compute_g(h, n)
        c = float(g.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8:
            break
    return best_h, best_c

def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rp = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm start (10%)
    p1 = t0 + 0.10 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 20000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, p1, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binarize + combinatorial descent (40%)
    p2 = t0 + 0.50 * (budget_s - 5)
    h_bin = to_binary(best_h, n)
    h_bin, c_bin = binary_descent(h_bin, n, p2, max_sweeps=5000)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    # Phase 3: Pair descent on continuous (25%)
    p3 = t0 + 0.75 * (budget_s - 5)
    h_cont = best_h.copy()
    h_cont, c_cont = pair_descent(h_cont, n, p3, max_sweeps=100)
    if c_cont < best_c:
        best_c = c_cont
        best_h = h_cont.copy()

    # Phase 4: Re-binarize + annealed combinatorial (10%)
    p4 = t0 + 0.88 * (budget_s - 5)
    h_bin2 = to_binary(best_h, n)
    h_bin2, c_bin2 = binary_descent_anneal(h_bin2, n, p4, max_sweeps=3000)
    if c_bin2 < best_c:
        best_c = c_bin2
        best_h = h_bin2.copy()

    # Phase 5: Final polish (remaining)
    if time.time() < dl - 30:
        h_fin = best_h.copy()
        h_fin, c_fin = pair_descent(h_fin, n, dl - 5, max_sweeps=100)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
