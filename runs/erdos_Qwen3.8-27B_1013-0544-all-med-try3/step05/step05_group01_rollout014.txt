To reduce c[k*], we want to:

So the gradient tells us: if (1-h[m+k*]) > h[m-k*], increasing h[m] increases c[k*], and we should decrease it.

For the max objective, the subgradient is the gradient of the active constraint (the shift achieving the max).

So a gradient descent step would be:
h_new = h - α · ∇c[k*] (projected onto feasible set)

One key improvement I can make: **use the gradient information to guide the pair search**. Instead of trying random pairs, try the pairs that the gradient suggests would be most effective.

From the gradient grad[i], we know:
- If grad[i] > 0, increasing h[i] increases the objective → we should decrease h[i]
- If grad[i] < 0, increasing h[i] decreases the objective → we should increase h[i]

So the best pair is (i, j) where grad[i] < 0 (we want to increase h[i]) and grad[j] > 0 (we want to decrease h[j]). The improvement is approximately -lr * (grad[i] - grad[j]) * d for a transfer of d from j to i.

This gives a very targeted pair search!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_active(h, n, top_k):
    c_raw = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(c_raw)[-top_k:]
    top_c = c_raw[top_idx]
    total = top_c.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_c / total
    grad = np.zeros(n)
    for t in range(len(top_idx)):
        idx = int(top_idx[t])
        w = weights[t]
        k = idx - (n - 1)
        valid = n - abs(k)
        if valid <= 0:
            continue
        if k >= 0:
            grad[:valid] += w * (1.0 - h[k:])
            grad[k:] += -w * h[:valid]
        else:
            ke = -k
            grad[ke:] += w * (1.0 - h[:valid])
            grad[:valid] += -w * h[ke:]
    return grad

def gradient_polish(h, n, deadline, top_k=5):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.012
    no_imp = 0
    while time.time() < deadline:
        grad = grad_active(h, n, top_k)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_imp = 0
            lr = min(lr * 1.01, 0.06)
        else:
            no_imp += 1
            if no_imp < 50:
                lr *= 0.99
            else:
                lr *= 0.95
                if lr < 1e-6:
                    h = best_h.copy()
                    lr = 0.012
                    no_imp = 0
        h = h_new
    return best_h, best_c

def gradient_guided_pairs(h, n, deadline, max_passes=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        grad = grad_active(h, n, 5)
        order = np.argsort(grad)
        inc_cand = order[:n // 2]
        dec_cand = order[n // 2:]
        improved = False
        for ii in inc_cand[:40]:
            if time.time() > deadline:
                return h, best_c
            for jj in dec_cand[:40]:
                i, j = int(ii), int(jj)
                if i == j:
                    continue
                for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i], h[j] = hi - d, hj + d
        if not improved:
            break
    return h, best_c

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=10000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        for _pass in range(300):

                return h, best_c
            improved = False

                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c
```
