    for _ in range(200):

        if abs(diff) < 1e-14:
            break

        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def barrier_obj_grad(h, T, dx):
    """Objective and gradient for barrier method."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(corr - T, 0.0)
    obj = np.sum(excess**2)
    # Gradient
    n = len(h)
    g = np.zeros(n)
    mask = corr > T
    if np.any(mask):
        # For each lag k where corr_k > T, add (corr_k - T) * d(corr_k)/dh
        for k in np.where(mask)[0]:
            lag = k - (n - 1)
            w = 2.0 * (corr[k] - T) * dx
            if lag >= 0:
                # corr[k] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
                g[:n-lag] += w * (1.0 - h[lag:])
                g[lag:] -= w * h[:n-lag]
            else:
                la = -lag
                g[la:] += w * (1.0 - h[:n-la])
                g[:n-la] -= w * h[la:]
    return obj, g

        return g

        return np.sum(x) - target

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize, Bounds
from itertools import combinations
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _barrier_obj(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(corr - T, 0.0)
    return float(np.sum(excess * excess))

def _barrier_grad(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    n = len(h)
    g = np.zeros(n)
    mask = corr > T
    if not np.any(mask):
        return g
    idx = np.where(mask)[0]
    for k in idx:
        lag = int(k) - (n - 1)
        w = 2.0 * (corr[k] - T) * dx
        if lag >= 0:
            g[:n - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            g[la:] += w * (1.0 - h[:n - la])
            g[:n - la] -= w * h[la:]
    return g

def _sum_func(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones_like(x)

def do_barrier_slsqp(h0, n, target, T, maxiter=2000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_func, 'jac': _sum_jac, 'args': (target,)}]

    def bo(x):
        return _barrier_obj(x, T, dx)

    def bg(x):
        return _barrier_grad(x, T, dx)

    try:
        r = minimize(bo, h0, jac=bg, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except Exception:
        return h0

def _smax_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    return m + np.log(np.mean(np.exp(s))) / beta

def _smax_grad(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    exps = np.exp(s)
    wts = exps / np.sum(exps)
    n = len(h)
    g = np.zeros(n)
    active = wts > 1e-10
    for k in np.where(active)[0]:
        w = wts[k] * dx
        lag = int(k) - (n - 1)
        if lag >= 0:
            g[:n - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:n - lag]
        else:
            la = -lag
            g[la:] += w * (1.0 - h[:n - la])
            g[:n - la] -= w * h[la:]
    return g

def do_smax_slsqp(h0, n, target, beta, maxiter=3000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_func, 'jac': _sum_jac, 'args': (target,)}]

    def so(x):
        return _smax_obj(x, beta, dx)

    def sg(x):
        return _smax_grad(x, beta, dx)

    try:
        r = minimize(so, h0, jac=sg, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_targeted(h, n, target, n_iter, deadline):
    cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n - lag] += (1.0 - cur[lag:])
            g[lag:] -= cur[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - cur[:n - la])
            g[:n - la] -= cur[la:]
        gm = np.linalg.norm(g)
        if gm < 1e-14:
            break
        step = 0.008 / (1.0 + 0.005 * it)
        new_h = project_h(cur - step * g / gm, target)
        c = compute_c5(new_h)
        if c < best_c5 - 1e-16:
            best_h, best_c5 = new_h.copy(), c
        cur = new_h
    return best_h, best_c5

def block_search(n, target, deadline, max_m=16):
    best_h, best_c5 = None, float('inf')
    for m in range(4, max_m + 1):
        if time.time() > deadline - 60:
            break
        if n % m != 0:
            continue
        bs = n // m
        n_ones = int(round(target / bs))
        if n_ones * bs != target or n_ones < 1 or n_ones >= m:
            continue
        for combo in combinations(range(m), n_ones):
            if time.time() > deadline - 60:
                break
            h = np.zeros(n)
            for b in combo:
                h[b * bs:(b + 1) * bs] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def coord_descent(cur, target, n, sweeps, deadline):
    best_c5 = compute_c5(cur)
    deltas = (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() > deadline - 3:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 2:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-16:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
