def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, target, beta, maxiter=10000):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})

    if remaining > 20:
        alloc = min(remaining - 5, 80)
        best_h, best_c5 = exact_cd_fast(best_h, target, t0, t0 + alloc, max_sweeps=15000, shake_every=300, seed=seed+99)
    
    # Final verification
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_cur
