    corr = np.correlate(h, 1.0 - h, mode='full')

        lag = k - (n - 1)
        grad += w[j] * subgrad_for_lag(h, lag, n)
    return grad

def fast_top_k_gradient(h, n, top_k=10, temp=30.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[top_idx]
    shifted = temp * (vals - vals.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    lags = top_idx - (n - 1)
    pos_lags = lags[lags >= 0]
    neg_lags = lags[lags < 0]
    
    grad = np.zeros(n)

    # Combined: for m in [0, n-1]:
    #   (1-h[m+k]) * 1[m < n-k] - h[m-k] * 1[m >= k]
    
    # Let's compute using the direct formula for small top_k
    for i, k in enumerate(lags):

        if k >= 0:
            m = n - k
            if m > 0:
                grad[:m] += wi * (1.0 - h[k:k+m])  # (1-h[m+k]) for m in [0, n-1-k]

                grad[:m] -= wi * h[la:]              # -h[m+k] = -h[m-la] for m in [0, n-1-la]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

    return np.ones_like(h)

        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]

                            options={'maxiter': maxit, 'ftol': 1e-14})
                hn = project_feasible(r.x, ts)
                cn = compute_c5(hn)
                if cn < c5:
                    h = hn
                    c5 = cn
            except Exception:
                pass
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(350, 0.5)
    pn = len(parent)

    # Phase 1: SLSQP + gradient at parent resolution (38%)
    d1 = t0 + budget_s * 0.38
    h1 = parent.copy()
    h1, c1 = slsqp_run(h1, pn, d1, seed)
    h1, c1 = grad_polish(h1, pn, d1, seed + 1)
    if c1 < best_c:
        best_h, best_c, best_n = h1.copy(), c1, pn

    # Also try reversed parent if time allows
    if time.time() < t0 + budget_s * 0.35 and pn > 1:
        h_rev = project_feasible(parent[::-1].copy(), pn / 2.0)
        h_rev, c_rev = slsqp_run(h_rev, pn, t0 + budget_s * 0.38, seed + 50)
        h_rev, c_rev = grad_polish(h_rev, pn, t0 + budget_s * 0.38, seed + 51)
        if c_rev < best_c:
            best_h, best_c, best_n = h_rev.copy(), c_rev, pn

    # Phase 2: Other resolutions (30%)
    d2 = t0 + budget_s * 0.68
    try_ns = [280, 350, 420, 500, 580]
    try_ns = [n for n in try_ns if n != pn and 60 <= n <= 800]
    remaining = d2 - time.time()
    per_res = remaining / max(len(try_ns), 1)
    for n in try_ns:
        if time.time() > d2:
            break
        src = best_h if best_h is not None else parent
        x_old = np.linspace(0, 2, len(src), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0 = project_feasible(np.interp(x_new, x_old, src), n / 2.0)
        sub_dl = min(d2, time.time() + per_res * 0.95)
        h0, cn = slsqp_run(h0, n, sub_dl, seed + n)
        h0, cn = grad_polish(h0, n, sub_dl, seed + n + 1)
        if cn < best_c:
            best_h, best_c, best_n = h0.copy(), cn, n

    # Phase 3: Final high-precision SLSQP + gradient (32%)
    d3 = t0 + budget_s * 0.98
    if best_h is not None:
        h3 = best_h.copy()
        h3, c3 = slsqp_run(h3, best_n, d3, seed + 500)
        h3, c3 = grad_polish(h3, best_n, d3, seed + 501)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Final safety
    if best_h is None:
        best_h = parent.copy()
        best_n = pn
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
