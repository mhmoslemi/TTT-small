def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_max_lags(h, n, tol=0.01):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    return c, m, np.where(c > m - tol)[0]

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

Actually, computing the full gradient is O(n^2) which is fine for n=200. Let me implement it properly.

For position a, dC_k/dh[a] (where k is the shift s = i - (N-1)):

From our earlier analysis:
For s >= 0 (i = N-1+s):
- If s <= a: (1-h[a-s])
- If s <= N-1-a: -h[a+s]

For s < 0 (m = -s, i = N-1-m):
- If m <= N-1-a: (1-h[a+m])
- If m <= a: -h[a-m]

So the gradient of C_k with respect to h[a] is:
g[a][k] = (1-h[a-s]) * [s<=a, s>=0] + (-h[a+s]) * [s<=N-1-a, s>=0] + (1-h[a+m]) * [m<=N-1-a, m>0] + (-h[a-m]) * [m<=a, m>0]

where s = k-(N-1) for k >= N-1, and m = (N-1)-k for k < N-1.

The gradient of the soft max is:
[grad(sm)][a] = sum_k w[k] * g[a][k]

where w[k] = softmax(alpha * C)[k].

This is O(n^2) to compute, which is fine for n=200.

= sum_k w_k * dC_k/dh[a]

where w_k = exp(alpha*C_k) / sum_j exp(alpha*C_j).

So I need to compute sum_k w_k * dC_k/dh[a] for each a.

```python
def compute_sm_gradient(h, n, alpha):
    """Compute gradient of the soft max of the correlation array."""
    dx = 2.0 / n
    C = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    C_shifted = C - C.max()
    w = np.exp(alpha * C_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for a in range(n):
        g = 0.0
        # s >= 0 part
        s_max1 = min(a, n-1)
        for s in range(0, s_max1 + 1):
            k = n - 1 + s
            g += w[k] * (1 - h[a - s])
        s_max2 = min(n-1-a, n-1)
        for s in range(0, s_max2 + 1):
            k = n - 1 + s
            g += w[k] * (-h[a + s])
        # s < 0 part
        m_max1 = min(n-1-a, n-1)
        for m in range(1, m_max1 + 1):
            k = n - 1 - m
            g += w[k] * (1 - h[a + m])
        m_max2 = min(a, n-1)
        for m in range(1, m_max2 + 1):
            k = n - 1 - m
            g += w[k] * (-h[a - m])
        grad[a] = g
    return grad, C

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sm_gradient(h, n, alpha):
    dx = 2.0 / n
    C = np.correlate(h, 1 - h, mode='full') * dx
    Csh = C - C.max()
    w = np.exp(alpha * Csh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(n)
    for a in range(n):
        if a >= 0:
            s = np.arange(a + 1)
            grad[a] += np.dot(w[n - 1 + s], 1 - h[a - s])
        if n - 1 - a >= 0:
            s = np.arange(n - a)
            grad[a] += np.dot(w[n - 1 + s], -h[a + s])
        if n - 1 - a >= 1:
            m = np.arange(1, n - a)
            grad[a] += np.dot(w[n - 1 - m], 1 - h[a + m])
        if a >= 1:
            m = np.arange(1, a + 1)
            grad[a] += np.dot(w[n - 1 - m], -h[a - m])
    return grad

def projected_grad(h, n, deadline, alpha=500, lr_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    v = np.zeros(n)
    lr = lr_init
    for _ in range(50000):
        if time.time() > deadline:
            break
        g = sm_gradient(best_h, n, alpha)
        v = 0.9 * v - lr * g
        new_h = project_feasible(best_h + v, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = new_h
        else:
            lr *= 0.95
            v *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c

def local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    C = np.correlate(best_h, 1 - best_h, mode='full')
    dx = 2.0 / n
    for d in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
        for _pass in range(50):
            if time.time() > deadline:
                return best_h, best_c
            improved = False
            step = max(1, n // 60)
            for i in range(0, n, step):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n, step):
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c_new = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            C = np.correlate(best_h, 1 - best_h, mode='full')
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c_new = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
                        if c_new < best_c - 1e-12:

                            C = np.correlate(best_h, 1 - best_h, mode='full')
                            improved = True
                        else:
                            best_h[i] += d
