    """Compute autocorrelation A[d] for d = -(n-1), ..., (n-1)"""
    n = len(h)
    A = np.correlate(h, h, 'full')  # A[k] where k=0..2n-2, corresponds to d=k-(n-1)
    return A

def min_autocorr(h):
    """Return min of autocorrelation (excluding d=0 which is always n/2)"""
    A = np.correlate(h, h, 'full')
    # A[0] corresponds to d=-(n-1), A[n-1] to d=0, A[2n-2] to d=n-1

    h = h.copy()
    y = h.copy()  # Nesterov momentum
    t = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov extrapolation
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (1 - t) / t_new

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            if s < 1e-14:
                break

            if s < 1e-14:

def smooth_max_grad(h, n, alpha):

    dx = 2.0 / n
