        grad = topk_gradient(cur, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        trial = project_h(cur - step * grad, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            best_c5 = c_try
            best_h = trial.copy()
            cur = trial
            stall = 0
            step = min(step * 1.03, 0.15)
        else:
            stall += 1
            step *= 0.7
            if stall > 200:
                break
    return best_h, best_c5

def pair_local_search(h, target, n, t0, budget, max_sweeps=80, step_size=0.04):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        improved = False
        lags = compute_lags(cur)
        top_idx = np.argsort(lags)[::-1][:24]
        for idx in top_idx:
            if budget - (time.time() - t0) < 2:
                break
            lag = int(idx) - (n - 1)
            if abs(lag) >= n:
                continue
            la = abs(lag)
            pairs = n - la
            stride = max(1, pairs // 8)
            for i in range(0, pairs, stride):
                if lag >= 0:
                    i1, i2 = i, i + la
                else:
                    i1, i2 = i + la, i
                if cur[i1] <= cur[i2] + 1e-6:
                    continue
                trial = cur.copy()
                trial[i1] -= step_size
                trial[i2] += step_size
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def smooth_max_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta) * (2.0 / n)

def do_slsqp(h0, n, target, beta, maxiter=6000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.full(n_init, 0.5)
        h_parent[:n_init // 2] = 0.7
        h_parent[n_init // 2:] = 0.3
        h_parent = project_h(h_parent, n_init / 2.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    n = best_n
    target = n / 2.0

    # Phase 1: Projected gradient with multiple top_k values
    for top_k in [16, 8, 32, 64, 4]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        h_try = best_h.copy()
        h_try, c_try = proj_grad_descent(h_try, target, n, top_k, 4000, t0, budget_s, step_init=0.02)
        if c_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c_try

    # Phase 2: Pair local search
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = pair_local_search(best_h, target, n, t0, budget_s, max_sweeps=60, step_size=0.04)

    # Phase 3: SLSQP medium beta
    for beta in [500, 2000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        best_h = do_slsqp(best_h, n, target, beta, maxiter=6000)
        best_c5 = compute_c5(best_h)

    # Phase 4: More pair search with smaller step
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = pair_local_search(best_h, target, n, t0, budget_s, max_sweeps=80, step_size=0.02)

    # Phase 5: SLSQP high beta
    for beta in [8000, 20000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        best_h = do_slsqp(best_h, n, target, beta, maxiter=8000)
        best_c5 = compute_c5(best_h)

    # Phase 6: Try different resolutions
    for n_hr in [320, 448, 512, 384]:
        if budget_s - (time.time() - t0) < 50:
            break
        if n_hr == best_n:
            continue
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample_fn(best_h, n_hr), target_hr)
        cur_hr, c_try = proj_grad_descent(cur_hr, target_hr, n_hr, 16, 2000, t0, budget_s, step_init=0.03)
        if c_try < best_c5:
            best_h = cur_hr.copy()
            best_c5 = c_try
            best_n = n_hr
            n = n_hr
            target = target_hr
            cur_hr = do_slsqp(cur_hr, n, target, 3000, maxiter=4000)
            c_try = compute_c5(cur_hr)
            if c_try < best_c5:
                best_h = cur_hr.copy()
                best_c5 = c_try
        else:
            cur_hr = do_slsqp(cur_hr, n_hr, target_hr, 2000, maxiter=3000)
            c_try = compute_c5(cur_hr)
            if c_try < best_c5:
                best_h = cur_hr.copy()
                best_c5 = c_try
                best_n = n_hr
                n = n_hr
                target = target_hr

    # Phase 7: Random restarts
    remaining = budget_s - (time.time() - t0)
    for att in range(12):
        if remaining < 25:
            break
        remaining = budget_s - (time.time() - t0)
        scale = 0.03 + 0.03 * (att % 4)
        hp = best_h + rng.standard_normal(n) * scale
        hp = project_h(hp, target)
        hp, c_try = proj_grad_descent(hp, target, n, 16, 2500, t0, budget_s, step_init=0.02)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try
        remaining = budget_s - (time.time() - t0)
        if c_try < best_c5 + 0.0005 and remaining > 12:
            best_h = do_slsqp(best_h, n, target, 5000, maxiter=5000)
            best_c5 = compute_c5(best_h)

    # Phase 8: Final pair search + SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = pair_local_search(best_h, target, n, t0, budget_s, max_sweeps=150, step_size=0.015)

    remaining = budget_s - (time.time() - t0)
    if remaining > 12:
        best_h = do_slsqp(best_h, n, target, 50000, maxiter=12000)
        best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h, best_c5 = pair_local_search(best_h, target, n, t0, budget_s, max_sweeps=100, step_size=0.01)

    return best_h, best_c5, best_n
