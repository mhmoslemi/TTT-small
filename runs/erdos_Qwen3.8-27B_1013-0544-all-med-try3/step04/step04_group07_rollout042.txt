def smooth_max_penalty(h, beta, dx):
    """Smooth max of correlation as objective for SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Use log-sum-exp for smooth max
    shifted = beta * (corr - m)

    val = m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta
    return val * dx

def sum_constraint(x, t):
    return np.sum(x) - t

def sum_constraint_jac(x, t):
    return np.ones(x.shape)

def slsqp_optimize(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (target,)}]
    dx = 2.0 / n
    
    def obj(x):
        return smooth_max_penalty(x, beta, dx)
    
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or r.fun < smooth_max_penalty(h0, beta, dx):
            return project_h(r.x, target)
    except:
        pass
    return h0

    n = len(h)

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")
