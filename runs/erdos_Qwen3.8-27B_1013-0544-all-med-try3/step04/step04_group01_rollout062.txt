```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_D(h, n):
    """Build the sensitivity matrix D where D[i,k] = dC[k]/dh[i]."""
    g = 1.0 - h
    D = np.zeros((n, 2 * n - 1))
    ks = np.arange(2 * n - 1)
    for i in range(n):
        m1 = i - ks
        valid1 = (m1 >= 0) & (m1 < n)
        D[i, valid1] += g[m1[valid1]]
        m2 = i + ks
        valid2 = (m2 >= 0) & (m2 < n)
        D[i, valid2] -= h[m2[valid2]]
    return D

def fast_pair_search(h, n, deadline, deltas, max_passes=500):
    """Vectorized pair search using precomputed sensitivity matrix."""
    best_c = float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))
    C = np.correlate(h, 1 - h, mode='full')
    current_max = C.max()
    D = build_D(h, n)
    
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            # Direction 1: h[i] += d, h[j] -= d
            for j in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[j] < d - 1e-12:
                    continue
                deltas_C = d * (D - D[j:j+1, :])
                new_maxes = (C[None, :] + deltas_C).max(axis=1)
                feasible = (h <= 1.0 - d + 1e-12)
                feasible[j] = False
                new_maxes[~feasible] = np.inf
                best_i = int(new_maxes.argmin())
                if new_maxes[best_i] < current_max - 1e-14:
                    h[best_i] += d
                    h[j] -= d
                    C += d * (D[best_i] - D[j])
                    current_max = C.max()
                    D = build_D(h, n)
                    improved = True
            # Direction 2: h[i] -= d, h[j] += d
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] < d - 1e-12:
                    continue
                deltas_C = -d * (D - D[i:i+1, :])
                new_maxes = (C[None, :] + deltas_C).max(axis=1)
                feasible = (h <= 1.0 - d + 1e-12)
                feasible[i] = False
                new_maxes[~feasible] = np.inf
                best_j = int(new_maxes.argmin())
                if new_maxes[best_j] < current_max - 1e-14:
                    h[i] -= d
                    h[best_j] += d
                    C -= d * (D[i] - D[best_j])
                    current_max = C.max()
                    D = build_D(h, n)
                    improved = True
            if not improved:
                break
    best_c = float(C.max() * (2.0 / n))
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Phase 2: Fast pair search
    if time.time() < deadline - 5:
        best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 5,
                                          [0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    # Phase 3: Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 4: Final pair search with small steps
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 1,
                                          [0.002, 0.001, 0.0005, 0.0002])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        # Optimize at the initial resolution
        alloc1 = (deadline - t0) * 0.50
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Try other resolutions
        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150, 300, 400]:
            if n_try == n_init or time.time() > deadline - 40:
                continue
            remaining = deadline - time.time()
            alloc = min(80, remaining * 0.2)
            if alloc < 20:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(3, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.02 + 0.1 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 15) * 0.25)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if time.time() < deadline - 2:
            best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001])
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 200
        x = np.linspace(0, 2, n)
        best_h = 0.5 + 0.3 * np.sin(np.pi * x)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
```

```python
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2500, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_D(h, n):
    g = 1.0 - h
    D = np.zeros((n, 2 * n - 1))
    ks = np.arange(2 * n - 1)
    off = n - 1
    for i in range(n):
        m1 = i - ks + off
        v1 = (m1 >= 0) & (m1 < n)
        D[i, v1] += g[m1[v1]]
        m2 = i + ks - off
        v2 = (m2 >= 0) & (m2 < n)
        D[i, v2] -= h[m2[v2]]
    return D

def fast_pair_search(h, n, deadline, deltas, max_passes=300):
    C = np.correlate(h, 1 - h, mode='full')
    current_max = C.max()
    D = build_D(h, n)
    tol = 1e-14
    ws = np.empty((n, 2 * n - 1))
    nm = np.empty(n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for fixed in range(n):
                if time.time() > deadline:
                    break
                if h[fixed] < d - 1e-12:
                    continue
                np.subtract(D, D[fixed:fixed + 1, :], out=ws)
                ws *= d
                np.add(ws, C[None, :], out=ws)
                    ws.max(axis=1, out=nm)
                feasible = (h <= 1.0 - d + 1e-12)
                feasible[fixed] = False
                nm[~feasible] = np.inf
                best_free = int(nm.argmin())
                if nm[best_free] < current_max - tol:
                    h[best_free] += d
                    h[fixed] -= d
                    C += d * (D[best_free] - D[fixed])
                    current_max = C.max()
                    D = build_D(h, n)
                    improved = True
            if not improved:
                break
    best_c = float(C.max() * (2.0 / n))
    return h, best_c

def optimize_at_resolution(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() > deadline - 5:
        return best_h, best_c

    # SLSQP phase - increasing alpha
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 3000000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Fast pair search - main improvement
    if time.time() < deadline - 5:
        best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 5,
                                          [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    # Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Final fine pair search
    if time.time() < deadline - 2:
        best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 0.5,
                                          [0.002, 0.001, 0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = min(350, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_resolution(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        # Try other resolutions by interpolation
        x_init = np.linspace(0, 2, n_init)
        temps = [100, 150, 300, 400]
        for n_try in temps:
            if n_try == n_init or time.time() > deadline - 50:
                continue
            remaining = deadline - time.time()
            alloc = min(100, remaining * 0.2)
            if alloc < 25:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_at_resolution(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Random restarts at best resolution
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.015 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_resolution(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish at best resolution
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = fast_pair_search(best_h.copy(), n, deadline - 0.5,
                                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 200
        x = np.linspace(0, 2, n)
        best_h = 0.5 + 0.3 * np.sin(np.pi * x)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
```
