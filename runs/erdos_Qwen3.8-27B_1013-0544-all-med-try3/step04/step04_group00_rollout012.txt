    full = np.correlate(h, 1 - h, mode='full')

        if wk < 1e-12:

        L = k - (n - 1)
        if L >= 0:

            # positions L to n-1: contribute -h[j-L]
            if n - 1 - L >= 0:

            if n - L > L:  # positions L to n-1 exist
                grad[L:n] += wk * dx * (-h[0:n-L])
        else:

            # positions M to n-1: contribute (1-h[j-M])
            # positions 0 to n-1-M: contribute -h[j+M]
            grad[M:n] += wk * dx * (1 - h[0:n-M])
            grad[0:n-M] += wk * dx * (-h[M:n])
    
    return grad

                            else:
