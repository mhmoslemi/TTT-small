            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 18) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001, 0.00005, 0.00002], max_passes=400)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.0005, 0.0002, 0.0001], radius=80)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
