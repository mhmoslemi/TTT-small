            h_pert2 = project_feasible(best_h + 0.15 * rng.randn(n), n)
            h_bin2, c_bin2 = binary_local_search(h_pert2, n, deadline - 2, max_iters=50000)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = h_bin2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
