```python
def solve_lp_for_h(g, n, dx):
    """Solve LP: min t s.t. dx*sum(h[i]*g[i+k]) <= t for all k, 0<=h<=1, sum(h)=n/2"""
    from scipy.optimize import linprog
    total_vars = n + 1  # h_0,...,h_{n-1}, t
    # Objective: minimize t
    c = np.zeros(total_vars)
    c[n] = 1.0  # t is the last variable
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: dx * sum_i h[i] * g[i+k] - t <= 0
    kags = range(-(n-1), n)
    A_rows = []
    b_vals = []
    for k in kags:
        # Valid i range
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        if i_lo > i_hi:
            continue
        row = np.zeros(total_vars)
        # row[n] = -1 (for the -t term)
        row[n] = -1.0
        for i in range(i_lo, i_hi+1):
            row[i] = dx * g[i+k]
        A_rows.append(row)
        b_vals.append(0.0)
    
    A_ub = np.array(A_rows) if A_rows else np.zeros((1, total_vars))
    b_ub = np.array(b_vals) if b_vals else np.zeros(1)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, total_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 <= h_i <= 1, t >= 0 (no upper bound on t)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res.x[:n], res.x[n]
```

import numpy as np
from scipy.optimize import minimize, linprog, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_x_given_y(y, n, dx):
    """Solve: min t s.t. dx*sum_j x[j]*y[j+lag] <= t for all lags, 0<=x<=1, sum(x)=n/2"""
    total_vars = n + 1
    c_obj = np.zeros(total_vars)
    c_obj[n] = 1.0
    rows = []
    for lag in range(-(n - 1), n):
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo > i_hi:
            continue
        row = np.zeros(total_vars)
        row[n] = -1.0
        idx = np.arange(i_lo, i_hi + 1)
        row[idx] = dx * y[idx + lag]
        rows.append(row)
    if not rows:
        return np.full(n, 0.5), 0.0
    A_ub = np.array(rows)
    b_ub = np.zeros(len(rows))
    A_eq = np.zeros((1, total_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def block_lp_descent(h, n, deadline, max_iter=300):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = 1.0 - h_cur
        h_new, t_val = solve_lp_x_given_y(g, n, dx)
        if h_new is not None:
            h_new = project_feasible(h_new, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
            h_cur = h_new
        else:
            break
        g = 1.0 - h_cur
        g_new, _ = solve_lp_x_given_y(h_cur, n, dx)
        if g_new is not None:
            h_cur = np.clip(1.0 - g_new, 0.0, 1.0)
            h_cur = project_feasible(h_cur, n / 2.0)
            c5_new = compute_c5(h_cur)
            if c5_new < best_c5 - 1e-12:
                best_h = h_cur.copy()
                best_c5 = c5_new
        else:
            break
        if it > 10 and abs(best_c5 - compute_c5(h_cur)) < 1e-12:
            break
    return best_h, best_c5

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def slsqp_refine(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    betas = [50, 200, 1000, 5000, 20000, 100000]
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.002, 0.008]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def subgrad_for_lag_vec(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:])
            grad[lag:] -= h[:m]
    else:
        la = -lag
        m = n - la
        if m > 0:
            grad[la:] += (1.0 - h[:m])
            grad[:m] -= h[la:]
    return grad

def subgradient_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = 8
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(15.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * subgrad_for_lag_vec(h, lag, n)
        velocity = 0.80 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.003, 0.006)
        else:
            stagnant += 1
            velocity *= 0.3
            lr *= 0.90
        if stagnant > 500 and lr < 1e-6:
            break
    return h, c5

def coordinate_descent(h, n, deadline, max_sweeps=30):
    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            saved = h[j]
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[j] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[j] = saved
            a = corr1 - corr0
            b = corr0 - saved * a
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = a * mid + b
                mx = np.max(vals)
                active = vals >= mx - 1e-15
                if not np.any(active):
                    active[np.argmax(vals)] = True
                sg = a[active]
                if np.max(sg) <= 1e-15:
                    hi = mid
                elif np.min(sg) >= -1e-15:
                    lo = mid
                else:
                    break
            h[j] = (lo + hi) / 2.0
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)
    parent = project_feasible(parent, pn / 2.0)

    # Phase 1: Block LP descent on parent resolution (40% time)
    d1 = t0 + budget_s * 0.40
    h1, c1 = block_lp_descent(parent.copy(), pn, d1, max_iter=400)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: Coordinate descent refinement (15%)
    d2 = t0 + budget_s * 0.55
    if best_h is not None:
        h2, c2 = coordinate_descent(best_h.copy(), best_n, d2, max_sweeps=40)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP refinement (20%)
    d3 = t0 + budget_s * 0.75
    if best_h is not None:
        h3, c3 = slsqp_refine(best_h.copy(), best_n, d3, seed)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try alternative resolutions (15%)
    d4 = t0 + budget_s * 0.90
    for n_alt in [pn - 50, pn + 50, pn - 100, pn + 100]:
        if n_alt < 100 or n_alt > 800:
            continue
        if time.time() > d4:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_alt / 2.0)
        hh, cc = block_lp_descent(h0, n_alt, d4, max_iter=100)
        if cc < best_c5:
            best_h, best_c5, best_n = hh, cc, n_alt

    # Phase 5: Final SLSQP + subgradient polish (10%)
    d5 = t0 + budget_s * 0.97
    if best_h is not None:
        h5, c5 = slsqp_refine(best_h.copy(), best_n, d5, seed + 7)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5
        best_h, best_c5 = subgradient_polish(best_h, best_n, d5, seed + 13)

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
