```python
        local_grid = np.linspace(max(0, best_s - 2*grid[1]), min(1, best_s + 2*grid[1]), 5)
        for s in local_grid:
            val = np.max(beta + s * alpha)
            if val < best_val:
                best_val = val
                best_s = s
    return best_s
```

This is 20 + 5 = 25 evaluations of max, each O(n). Total: 25n per coordinate. For n=500, that's 12500 per coordinate, 6.25M per sweep. In numpy, this should be fast (maybe 0.01-0.05s per sweep).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gauss(h, i):
    n = len(h)
    gauss = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - i)
    hi1 = min(2 * n - 2, 2 * n - 2 - i)
    if lo1 <= hi1:
        m = np.arange(lo1, hi1 + 1)
        idx = i + m - (n - 1)
        gauss[m] += (1.0 - h[idx])
    lo2 = max(0, i)
    hi2 = min(2 * n - 2, i + n - 1)
    if lo2 <= hi2:
        m = np.arange(lo2, hi2 + 1)
        idx = i - (m - (n - 1))
        gauss[m] -= h[idx]
    return gauss

def solve_1d_maxmin(beta, alpha):
    grid = np.linspace(0.0, 1.0, 25)
    vals = np.max(beta[None, :] + grid[:, None] * alpha[None, :], axis=1)
    idx = np.argmin(vals)
    best_s = grid[idx]
    best_val = vals[idx]
    sp = grid[1] * 3
    lo = max(0.0, best_s - sp)
    hi = min(1.0, best_s + sp)
    if lo < hi:
        fine_grid = np.linspace(lo, hi, 11)
        fine_vals = np.max(beta[None, :] + fine_grid[:, None] * alpha[None, :], axis=1)
        fidx = np.argmin(fine_vals)
        if fine_vals[fidx] < best_val:
            best_s = fine_grid[fidx]
            best_val = fine_vals[fidx]
    return best_s

def coord_descent(h, n_sweeps, time_fn, target_sum, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1.0 - h, mode="full")
    for sweep in range(n_sweeps):
        if time_fn() < 5:
            break
        order = rng.permutation(n)
        for i in order:
            if time_fn() < 2:
                return h, compute_c5(h)
            hi_old = h[i]
            gauss = compute_gauss(h, i)
            beta = corr - hi_old * gauss
            s_star = solve_1d_maxmin(beta, gauss)
            delta = s_star - hi_old
            if abs(delta) > 1e-14:
                h[i] = s_star
                corr += delta * gauss
        h = project_h(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
    return h, compute_c5(h)

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

_box_beta = [100.0]
_box_target = [0.0]

def _slsqp_obj(x):
    val, _ = smooth_max_obj_grad(x, _box_beta[0])
    return val

def _slsqp_grad(x):
    _, g = smooth_max_obj_grad(x, _box_beta[0])
    return g

def _slsqp_con(x):
    return x.sum() - _box_target[0]

def _slsqp_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter_each, time_fn):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    _box_target[0] = target
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    constraints = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    for beta in betas:
        if time_fn() < 10:
            break
        _box_beta[0] = beta
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP',
                          bounds=bounds, jac=_slsqp_grad,
                          constraints=constraints,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Coordinate descent on main resolution
    h_cd, c5_cd = coord_descent(best_h, 3000, time_left, n_best / 2.0, seed=seed)
    if c5_cd < best_c5:
        best_h, best_c5, n_best = h_cd, c5_cd, len(h_cd)

    # Phase 2: SLSQP with high beta
    if time_left() > 60:
        h_sl, c5_sl = do_slsqp(best_h, [200, 1000, 5000, 20000, 50000, 100000],
                                maxiter_each=5000, time_fn=time_left)
        if c5_sl < best_c5:
            best_h, best_c5, n_best = h_sl, c5_sl, len(h_sl)

    # Phase 3: More coordinate descent
    if time_left() > 30:
        h_cd2, c5_cd2 = coord_descent(best_h, 3000, time_left, n_best / 2.0, seed=seed + 1)
        if c5_cd2 < best_c5:
            best_h, best_c5, n_best = h_cd2, c5_cd2, len(h_cd2)

    # Phase 4: Multi-resolution exploration
    for n_res in [150, 200, 300, 400]:
        if time_left() < 80:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        h_r, c5_r = coord_descent(h_res, 1500, time_left, n_res / 2.0, seed=seed + n_res)
        if c5_r < best_c5:
            best_h, best_c5, n_best = h_r, c5_r, n_res
        if time_left() > 40:
            h_sl2, c5_sl2 = do_slsqp(h_r, [500, 5000, 30000],
                                      maxiter_each=3000, time_fn=time_left)
            if c5_sl2 < best_c5:
                best_h, best_c5, n_best = h_sl2, c5_sl2, len(h_sl2)

    # Phase 5: Random restarts at best resolution
    if time_left() > 60:
        for attempt in range(5):
            if time_left() < 30:
                break
            pert = np.random.randn(n_best) * 0.02
            h_try = project_h(best_h + pert, n_best / 2.0)
            h_try, c5_try = coord_descent(h_try, 1000, time_left, n_best / 2.0,
                                          seed=seed + 100 + attempt)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try

    # Phase 6: Final coordinate descent + SLSQP polish
    if time_left() > 20:
        h_f, c5_f = coord_descent(best_h, 2000, time_left, n_best / 2.0, seed=seed + 999)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    if time_left() > 15:
        h_f2, c5_f2 = do_slsqp(best_h, [50000, 100000, 200000],
                                maxiter_each=8000, time_fn=time_left)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2

    # Final coordinate descent polish
    if time_left() > 5:
        h_f3, c5_f3 = coord_descent(best_h, 1000, time_left, n_best / 2.0, seed=seed + 777)
        if c5_f3 < best_c5:
            best_h, best_c5 = h_f3, c5_f3

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
