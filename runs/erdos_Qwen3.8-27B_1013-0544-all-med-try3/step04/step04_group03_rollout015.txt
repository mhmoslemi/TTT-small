import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def weighted_lag_obj(h_in, q, lags_idx, n):
    """Compute sum_k q[k] * corr[k] * dx for given lag indices"""
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    val = np.sum(q[lags_idx] * corr[lags_idx]) * (2.0 / n)
    return val

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * (2.0 / n)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def do_slsqp_weighted(h0, n, target, q, maxiter=5000):
    """SLSQP on weighted sum of lags (smooth objective)"""
    lags_idx = np.where(q > 1e-8)[0]
    q_sub = q[lags_idx] / np.sum(q[lags_idx])
    
    def obj(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        return np.sum(q_sub * corr[lags_idx]) * (2.0 / n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=8, momentum=0.85):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    best_cur = cur.copy()
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            elif lag < 0:
                la = -lag
                if la < n:
                    grad[la:] += (1.0 - cur[:n - la])
                    grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = momentum * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            best_cur = cur.copy()
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.8
            mom *= 0.3
        if stall > 300:
            break
    return best_cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)

                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

    # Precompute initial correlation

    best_h = h.copy()
    best_max = cur_max

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):

            break
        
        # Pick random swap
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        
        # Compute delta for all correlation indices
        delta = np.zeros(2*n - 1)
        for k in range(2*n - 1):

            # a as first index: valid if 0 <= a < n and 0 <= a+lag < n
            da = 0.0
            if 0 <= a + lag < n:

            # a as second index: valid if 0 <= a-lag < n
            if 0 <= a - lag < n:

            # b as second index

                db -= h[b - lag]  # (1-h[b]) went from 1 to 0

        new_max = float(np.max(new_corr)) * dx

            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_max = new_max
            best_max = new_max

                zeros = np.where(h == 0)[0]

                pass  # reject
    
    return best_h, best_max

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * (2.0 / n)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def do_slsqp_weighted(h0, n, target, w, maxiter=3000):
    """Minimize weighted sum of correlations using SLSQP with analytical gradient"""
    full_corr_size = 2 * n - 1
    lags = np.arange(full_corr_size) - (n - 1)
    
    def obj(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        return np.dot(w, corr) * (2.0 / n)
    
    def obj_grad(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        val = np.dot(w, corr) * (2.0 / n)
        grad = np.zeros(n)
        for k in range(full_corr_size):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = lags[k]
            i_start = max(0, n - 1 - k)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = i_idx + lag
            grad[i_idx] += wk * (1.0 - h_in[j_idx])
            grad[j_idx] += -wk * h_in[i_idx]
        grad *= (2.0 / n)
        return val, grad
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(obj_grad, h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def dual_ascent(h, n, target, t0, budget, n_outer=10, beta_q=100):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    for outer in range(n_outer):
        if budget - (time.time() - t0) < 20:
            break
        lags = compute_lags(cur)
        w = np.exp(beta_q * (lags - np.max(lags)))
        w /= w.sum()
        cur = do_slsqp_weighted(cur, n, target, w, maxiter=2000)
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = cur.copy()
    return best_h, best_c5

def binary_anneal_fast(h_bin, n, t0, budget, n_iter=500000):
    """Simulated annealing on binary h with vectorized delta computation"""
    h = h_bin.copy()
    full_size = 2 * n - 1
    lags = np.arange(full_size) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    rng = np.random.default_rng(999)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        
        # Vectorized delta computation
        va_f = (a + lags >= 0) & (a + lags < n)
        va_s = (a - lags >= 0) & (a - lags < n)
        vb_f = (b + lags >= 0) & (b + lags < n)
        vb_s = (b - lags >= 0) & (b - lags < n)
        
        delta = np.zeros(full_size)
        delta[va_f] -= (1.0 - h[a + lags[va_f]])
        delta[va_s] += h[a - lags[va_s]]
        delta[vb_f] += (1.0 - h[b + lags[vb_f]])
        delta[vb_s] -= h[b - lags[vb_s]]
        
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        
        if new_max < best_max:
            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_max = new_max
            best_max = new_max
            best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.5 * (0.001) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp * full_size, 1e-30)):
                h[a], h[b] = 0.0, 1.0
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = new_max
    
    return best_h, best_max * dx

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=8):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    best_cur = cur.copy()
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            elif lag < 0:
                la = -lag
                if la < n:
                    grad[la:] += (1.0 - cur[:n - la])
                    grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.85 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            best_cur = cur.copy()
            stall = 0
            step = min(step * 1.03, 0.06)
        else:
            stall += 1
            step *= 0.8
            mom *= 0.3
        if stall > 200:
            break
    return best_cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.015, -0.015, 0.008, -0.008]:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5
