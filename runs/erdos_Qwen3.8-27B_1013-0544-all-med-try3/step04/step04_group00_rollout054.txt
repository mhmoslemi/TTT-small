def project_onto_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h) = n/2} using bisection on lambda."""
    target = n / 2.0
    # h_j = clip(h_j - lambda, 0, 1), find lambda so sum = target
    # If h_j - lambda >= 1: h_j = 1 (contribution 1)
    # If h_j - lambda <= 0: h_j = 0 (contribution 0)  
    # If 0 < h_j - lambda < 1: h_j = h_j - lambda (contribution h_j - lambda)

def grad_at_lag(h, n, tau):
    """Compute gradient of g(tau) with respect to h, where tau is the shift.
    tau > 0: g = sum_{i=0}^{n-1-tau} h[i](1-h[i+tau])
    tau < 0: let s=-tau, g = sum_{i=s}^{n-1} h[i](1-h[i-s])
    tau = 0: g = sum h[i](1-h[i])
    """
    grad = np.zeros(n)
    if tau == 0:
        grad[:] = 1.0 - 2.0 * h
    elif tau > 0:

        #   j in [0, n-1-tau]: contributes (1-h[j+tau])
        #   j in [tau, n-1]: contributes -h[j-tau]
        grad[:n-tau] += (1.0 - h[tau:n])
        grad[tau:] -= h[:n-tau]
    else:

        #   j in [0, n-1-s]: contributes -h[j+s]
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    return grad

def grad_lag(h, n, tau):
    """Analytical gradient of g(tau) w.r.t. h."""
    g = np.zeros(n)
    if tau == 0:
        g[:] = 1.0 - 2.0 * h
    elif tau > 0:
        g[:n - tau] += (1.0 - h[tau:n])
        g[tau:] -= h[:n - tau]
    else:
        s = -tau
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def subgrad_opt(h, n, deadline, top_k=3, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        # Top-k active lags
        if top_k >= len(g):
            ks = list(range(len(g)))
        else:
            ks = list(np.argsort(g)[-top_k:])
        grad = np.zeros(n)
        for k in ks:
            tau = k - (n - 1)
            grad += grad_lag(h, n, tau)
        grad /= len(ks)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        # Backtracking line search
        accepted = False
        for f in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            step = lr * f / ng
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new <= m + 1e-15:
                h = h_new
                accepted = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        if not accepted:
            lr *= 0.5
            if lr < 1e-12:
                break
        # Periodically boost lr to escape local optima
        elif it % 500 == 499:
            lr = min(lr * 1.5, 10.0)
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()  # numerical stability (should already be ≤ 0)
    p = np.exp(sh)
    p /= p.sum()

        if p[k] < 1e-15:
            continue
        tau = k - (n-1)
        grad += p[k] * grad_lag(h, n, tau)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            np.clip(result, 0.0, 1.0, out=result)
    return result

def grad_lag(h, n, tau):
    g = np.zeros(n)
    if tau == 0:
        g[:] = 1.0 - 2.0 * h
    elif tau > 0:
        g[:n - tau] += 1.0 - h[tau:]
        g[tau:] -= h[:n - tau]
    else:
        s = -tau
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def subgrad_opt(h, n, deadline, top_k=3, max_iter=100000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        if top_k >= len(g):
            ks = list(range(len(g)))
        else:
            ks = list(np.argsort(g)[-top_k:])
        grad = np.zeros(n)
        for k in ks:
            grad += grad_lag(h, n, k - (n - 1))
        grad /= len(ks)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        accepted = False
        for f in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new <= m + 1e-14:
                h = h_new
                accepted = True
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                break
        if not accepted:
            lr *= 0.5
            stall += 1
            if lr < 1e-12:
                break
        if stall > 300:
            h = project_feasible(h + 0.02 * rng.randn(n), n)
            lr = max(lr, 0.5)
            stall = 0
    return best_h, best_c

def slsqp_obj_grad(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    p = np.exp(sh)
    p /= p.sum()
    val = m + np.log(p.sum() * np.exp(sh.max())) / alpha
    val = m + (sh.max() + np.log(p.sum())) / alpha
    grad = np.zeros(n)
    for k in range(len(g)):
        if p[k] > 1e-14:
            tau = k - (n - 1)
            grad += p[k] * grad_lag(h, n, tau)
    return val, grad

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        return slsqp_obj_grad(x, n, alpha)[0]
    def jac(x):
        return slsqp_obj_grad(x, n, alpha)[1]
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       jac=jac,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': con}],
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            return project_feasible(res.x, n)
        except Exception:
            return h

def multiscale_slsqp(h, n, deadline, alphas, maxiter_each=5000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for a in alphas:
        if time.time() > deadline:
            break
        h2 = slsqp_step(best_h, n, a, maxiter_each, deadline)
        c = compute_c5(h2, n)
        if c < best_c:
            best_c = c
            best_h = h2.copy()
    return best_h, best_c
