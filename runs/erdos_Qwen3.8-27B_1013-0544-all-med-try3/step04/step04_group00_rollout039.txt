This is O(1) per shift, and there are 2n-1 shifts, so O(n) total per swap.

For n=250, that's about 500 operations per swap, which is very fast. We can do millions of swaps per second.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    h = h_init.copy()

    target = n // 2

    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    # Initial overlaps
    ov = np.correlate(h, 1 - h, mode='full').astype(float)
    # ov is indexed 0 to 2n-2, where ov[m] corresponds to shift s = m - (n-1)
    
    best_ov = ov.copy()
    best_c = ov.max() * dx
    best_h = h.copy()
    
    n_shifts = 2 * n - 1  # length of ov
    
    # Shift ranges: ov[m] for m in [0, 2n-2], shift s = m - (n-1)
    # s ranges from -(n-1) to (n-1)
    
    T = 0.5  # initial temperature in units of dx
    cool_rate = 0.9999
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(10000000):
        if time.time() > deadline:
            break
        
        # Pick random swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        # Compute delta for each shift
        # h[i] 1->0, h[j] 0->1
        # Δ(s) = -(1-h[i+s])*[0≤i+s<n] + h[i-s]*[0≤i-s<n] 
        #        + (1-h[j+s])*[0≤j+s<n] - h[j-s]*[0←j-s<n]
        
        # We need to update ov for all shifts s where any of these terms is non-zero
        # i+s valid: s in [-i, n-1-i]
        # i-s valid: s in [i-(n-1), i]  ... wait, 0≤i-s<n means 0≤i-s and i-s<n, so i-n<s≤i, i.e., s in [i-n+1, i]
        # j+s valid: s in [-j, n-1-j]
        # j-s valid: s in [j-n+1, j]
        
        # For each affected shift, compute delta
        # Let's vectorize this
        
        # Create delta array for affected shifts
        delta = np.zeros(n_shifts)
        
        # From i (1->0):
        # Term 1: -(1-h[i+s]) for s where 0<=i+s<n, i.e., s in [-i, n-1-i]
        #   In ov index: m = s + (n-1), so m in [n-1-i, n-1+n-1-i] = [n-1-i, 2n-2-i]
        s_min1 = max(-i, -(n-1))
        s_max1 = min(n-1-i, n-1)
        if s_min1 <= s_max1:
            m1 = np.arange(s_min1 + (n-1), s_max1 + (n-1) + 1)
            # h[i+s] for s in [s_min1, s_max1]
            idx_i_s = i + np.arange(s_min1, s_max1 + 1)
            delta[m1] -= (1 - h[idx_i_s])
        
        # Term 2: +h[i-s] for s where 0<=i-s<n, i.e., s in [i-n+1, i]
        s_min2 = max(i - n + 1, -(n-1))
        s_max2 = min(i, n-1)
        if s_min2 <= s_max2:
            m2 = np.arange(s_min2 + (n-1), s_max2 + (n-1) + 1)
            idx_i_ms = i - np.arange(s_min2, s_max2 + 1)
            delta[m2] += h[idx_i_ms]
        
        # From j (0->1):
        # Term 3: +(1-h[j+s]) for s where 0<=j+s<n, i.e., s in [-j, n-1-j]
        s_min3 = max(-j, -(n-1))
        s_max3 = min(n-1-j, n-1)
        if s_min3 <= s_max3:
            m3 = np.arange(s_min3 + (n-1), s_max3 + (n-1) + 1)
            idx_j_s = j + np.arange(s_min3, s_max3 + 1)
            delta[m3] += (1 - h[idx_j_s])
        
        # Term 4: -h[j-s] for s where 0<=j-s<n, i.e., s in [j-n+1, j]
        s_min4 = max(j - n + 1, -(n-1))
        s_max4 = min(j, n-1)
        if s_min4 <= s_max4:
            m4 = np.arange(s_min4 + (n-1), s_max4 + (n-1) + 1)
            idx_j_ms = j - np.arange(s_min4, s_max4 + 1)
            delta[m4] -= h[idx_j_ms]
        
        # new ov = ov + delta
        new_ov_max = np.max(ov + delta)
        
        # Metropolis accept
        if new_ov_max <= ov.max() or rng.random() < np.exp((ov.max() - new_ov_max) / max(T, 1e-10)):
            # Accept
            ov += delta
            h[i], h[j] = 0.0, 1.0
            ones[ii], zeros[jj] = j, i  # swap in lists
            # Actually need to be more careful with the lists
            
            if new_ov_max * dx < best_c:
                best_c = new_ov_max * dx
                best_h = h.copy()
                best_ov = ov.copy()
        
        # Cool
        T *= cool_rate
    
    return best_h, best_c

def project_feasible(h, n):

            if s < 1e-12:
