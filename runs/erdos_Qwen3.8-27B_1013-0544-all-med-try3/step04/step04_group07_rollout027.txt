def slsqp_opt(h0, n, target, beta, maxiter=3000, ftol=1e-14):
    def obj(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(corr)
        exps = np.exp(np.clip(beta * (corr - m), -500, 500))
        return (m + np.log(np.sum(exps)) / len(exps)) / beta * (2.0 / n)
    
    def sum_eq(x, t):
        return np.sum(x) - t
    def sum_eq_jac(x, t):
        return np.ones(x.shape)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except:
        return h0

def random_binary(n, target, seed):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, int(target), replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def sa_binary(h0, target, t0, budget):
    """Simulated annealing on binary patterns."""
    n = len(h0)
    rng = np.random.RandomState(123)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.005
    cooldown = 0.9997
    
    while time.time() - t0 < budget - 5:
        # Random swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        
        c_new = compute_c5(h_new)
        delta = c_new - compute_c5(h)
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c_new < best_c5:
                best_h = h.copy()
                best_c5 = c_new
        
        T *= cooldown
    
    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)
