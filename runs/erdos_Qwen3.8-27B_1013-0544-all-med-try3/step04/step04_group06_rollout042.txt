def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:

        else:

def grad_c(h, n_active=10):
    """Compute subgradient of max of top n_active correlations w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)

        else:

            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g / n_active

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_subgrad(h, k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:k]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g / k

def polish(h, target, t0, budget, max_iter=5000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_imp = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        k = min(3 + it // 200, 25)
        g = topk_subgrad(best, k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        for step in [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004, 0.0001]:
            ht = project_sum(best - step * g / gn, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best, best_c5 = ht, c5t
                improved = True
                break
        no_imp = 0 if improved else no_imp + 1
        if no_imp > 200:
            if no_imp > 2000:
                break
            if it % 80 == 79:
                sc = 0.02 + 0.04 * min(no_imp / 2000, 1.0)
                ht = project_sum(best + np.random.randn(n) * sc, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best, best_c5 = ht, c5t
                    no_imp = 0
    return best, best_c5

def slsqp_refine(h, n, beta, maxiter=2000):
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        bc = np.clip(beta * (corr - m), -400, 400)
        e = np.exp(bc)
        return (m + np.log(e.sum() / len(corr)) / beta) * (2.0 / n)
    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        bc = np.clip(beta * (corr - m), -400, 400)
        e = np.exp(bc)
        w = e / e.sum()
        g = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-12:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - x[:-lag])
                g[:n - lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n - m2]
        return g * (2.0 / n)
    def eq_fun(x):
        return x.sum() - target
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_sum(res.x, target)

def ga_optimize(h_seed, n, target, pop_size, n_gen, t0, budget):
    pop = np.array([h_seed.copy()])
    for i in range(1, pop_size):
        pop = np.append(pop, project_sum(h_seed + np.random.randn(n) * 0.12, target), axis=0)
    best_h = h_seed.copy()
    best_c5 = compute_c5(best_h)
    for gen in range(n_gen):
        if budget - (time.time() - t0) < 10:
            break
        scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
        bi = np.argmin(scores)
        if scores[bi] < best_c5:
            best_c5 = scores[bi]
            best_h = pop[bi].copy()
        order = np.argsort(scores)
        n_parents = max(4, pop_size // 3)
        parents = pop[order[:n_parents]]
        new_pop = parents.copy()
        for i in range(pop_size - n_parents):
            p1 = parents[np.random.randint(n_parents)]
            p2 = parents[np.random.randint(n_parents)]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mscale = 0.03 + 0.06 * (gen / max(n_gen, 1))
            child = project_sum(child + np.random.randn(n) * mscale, target)
            new_pop = np.append(new_pop, child.reshape(1, -1), axis=0)
        pop = new_pop
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: GA at moderate resolution for global exploration
    n_ga = 160
    xd = np.linspace(0, 2, len(best_h))
    xg = np.linspace(0, 2, n_ga)
    h_ga = project_sum(np.interp(xg, xd, best_h), n_ga / 2.0)
    h_ga, c5_ga = ga_optimize(h_ga, n_ga, n_ga / 2.0, 60, 1500, t0, budget_s)
    if c5_ga < best_c5:
        best_c5 = c5_ga
        best_h = project_sum(np.interp(xd, xg, h_ga), len(best_h) / 2.0)

    # Also try GA at n=200
    n_ga2 = 200
    xg2 = np.linspace(0, 2, n_ga2)
    h_ga2 = project_sum(np.interp(xg2, xd, best_h), n_ga2 / 2.0)
    h_ga2, c5_ga2 = ga_optimize(h_ga2, n_ga2, n_ga2 / 2.0, 50, 1000, t0, budget_s)
    if c5_ga2 < best_c5:
        best_c5 = c5_ga2
        best_h = project_sum(np.interp(xd, xg2, h_ga2), len(best_h) / 2.0)

    # Phase 2: SLSQP refinement at current resolution
    n = len(best_h)
    for beta in [50, 200, 800, 3000, 10000]:
        if tl() < 80:
            break
        h_new = slsqp_refine(best_h, n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        best_h, best_c5 = polish(best_h, n / 2.0, t0, budget_s, 2000)

    # Phase 3: Try higher resolutions
    for n2 in [256, 320, 400, 480]:
        if tl() < 180:
            break
        if n2 <= n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hi = project_sum(np.interp(xn, xo, best_h), n2 / 2.0)
        for beta in [100, 500, 2500]:
            if tl() < 70:
                break
            h_new = slsqp_refine(hi, n2, beta, maxiter=1200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5, n = h_new, c5_new, n2
            hi = h_new
        best_h, best_c5 = polish(best_h, n / 2.0, t0, budget_s, 3000)

    # Phase 4: Basin hopping at best resolution
    n_best = len(best_h)
    tgt = n_best / 2.0
    for kick in range(15):
        if tl() < 60:
            break
        scale = 0.03 + 0.05 * (kick / 15.0)
        h_k = project_sum(best_h + np.random.randn(n_best) * scale, tgt)
        h_k, c5_k = polish(h_k, tgt, t0, budget_s, 3000)
        if c5_k < best_c5:
            best_h, best_c5 = h_k, c5_k

    # Phase 5: Final SLSQP + micro-polish
    if tl() > 30:
        h_new = slsqp_refine(best_h, n_best, 15000, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    best_h, best_c5 = polish(best_h, n_best / 2.0, t0, budget_s, 8000)

    # Micro-polish with very small steps
    for it in range(5000):
        if tl() < 2:
            break
        g = topk_subgrad(best_h, 4)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0003, 0.0001]:
            ht = project_sum(best_h - step * g / gn, n_best / 2.0)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-17:
                best_h, best_c5 = ht, c5t
                improved = True
                break
        if not improved and it > 1000:
            break

    return best_h, best_c5, len(best_h)
