import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_corr(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_grad(h, n, k_star):
    """Subgradient of c_k_star w.r.t. h, where k_star is the lag index in the full correlate output."""
    # c_k = sum_i h[i] * (1 - h[i+k]) for valid i
    # where k = k_star - (n-1) (convert from full output index to actual lag)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    # For each valid i:
    #   h[i] contributes (1 - h[i+lag]) to grad[i]
    #   h[i+lag] contributes -h[i] to grad[i+lag]
    if lag >= 0:
        # valid i: 0 <= i < n and 0 <= i+lag < n => 0 <= i < n-lag
        valid_i = np.arange(0, n - lag)
        if len(valid_i) > 0:
            grad[valid_i] += (1.0 - h[valid_i + lag])
            grad[valid_i + lag] += -h[valid_i]
    else:
        # lag < 0, let l = -lag > 0
        # valid i: 0 <= i < n and -l <= i < n => l <= i < n
        l = -lag
        valid_i = np.arange(l, n)
        if len(valid_i) > 0:
            grad[valid_i] += (1.0 - h[valid_i - l])
            grad[valid_i - l] += -h[valid_i]
    return grad

def power_mean_obj(h, n, p):
    """Compute (sum_k |c_k|^p)^(1/p) as a smooth approximation of max c_k."""
    c = compute_all_corr(h, n)
    return float(np.sum(np.abs(c) ** p) ** (1.0 / p))

def subgradient_descent(h, n, max_iter, step_decay=0.999, epsilon=1e-6, deadline=None):
    """Subgradient method with heavy ball momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    step_size = 0.01
    mu = 0.9  # momentum coefficient
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_all_corr(h, n)
        c_max = c.max()
        # Find near-maximal lags
       thresh = c_max - epsilon
        active = np.where(c >= thresh)[0]
        # Average subgradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += c5_grad(h, n, k)
        grad /= len(active)
        
        # Momentum step
        v = mu * v - step_size * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step_size *= step_decay
    
    return best_h, best_c

def slsqp_power_mean(h, n, p, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = compute_all_corr(x, n)
        return float(np.sum(c ** p) ** (1.0 / p))
    def jac(x):
        c = compute_all_corr(x, n)
        # d/dx of (sum c_k^p)^(1/p) = (1/p) * (sum c_k^p)^(1/p - 1) * p * sum_k c_k^(p-1) * dc_k/dx
        # = (sum c_k^p)^(1/p - 1) * sum_k c_k^(p-1) * dc_k/dx
        norm = np.sum(c ** p) ** (1.0 / p - 1.0)
        g = np.zeros(n)
        for k in range(len(c)):
            if c[k] != 0:
                g += (c[k] ** (p - 1)) * c5_grad(x, n, k)
        return norm * g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_descent(h, n, deadline, max_rounds=100):
    """Optimize pairs of coordinates while maintaining sum constraint."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs (for n <= 200, this is at most 19900 pairs)
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                # Optimal delta: try a few values
                for d in [0.001, 0.005, 0.01]:
                    for sign in [1, -1]:
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
        if not improved:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def pm_objective(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.sum(c ** p) ** (1.0 / p))

def pm_gradient(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    s = float(np.sum(c ** p))
    if s < 1e-30:
        return np.zeros(n)
    base = s ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ck = c[k]
        if ck < 1e-15:
            continue
        w = ck ** (p - 1)
        lag = k - (n - 1)
        al = abs(lag)
        if al == 0:
            g += w * (1.0 - 2.0 * x)
        elif lag >= 0:
            g[:n - al] += w * (1.0 - x[al:])
            g[al:] += -w * x[:n - al]
        else:
            g[al:] += w * (1.0 - x[:n - al])
            g[:n - al] += -w * x[al:]
    return base * g

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def slsqp_pm(h, n, p, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(pm_objective, h, args=(n, p), jac=lambda x: pm_gradient(x, n, p),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_step(h, n):
    """One subgradient step with line search."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    # Active set: lags within epsilon of max
    eps = max(1e-4, 0.01 * c_max)
    active = np.where(c >= c_max - eps)[0]
    
    # Compute subgradient (average of active)
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        al = abs(lag)
        if al == 0:
            g += (1.0 - 2.0 * h)
        elif lag >= 0:
            g[:n - al] += (1.0 - h[al:])
            g[al:] += -h[:n - al]
        else:
            g[al:] += (1.0 - h[:n - al])
            g[:n - al] += -h[al:]
    g /= len(active)
    
    return g, c_max

def subgrad_descent(h, n, max_iter=10000, deadline=None):
    """Subgradient method with Polyak step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    mu = 0.85
    step = 0.005
    best_step = step
    patience = 0
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g, c_max = subgradient_step(h, n)
        
        # Momentum
        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
            best_step = step
        else:
            patience += 1
            if patience > 50:
                step *= 0.9
                patience = 0
        
        h = h_new
        step = min(0.02, best_step * (1.0 + 0.001 * np.random.randn()))
    
    return best_h, best_c

def targeted_refine(h, n, deadline):
    """Targeted pair search focusing on the active lag."""
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for step in steps:
        while time.time() < deadline:
            improved = False
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            k_max = int(np.argmax(c))
            lag = k_max - (n - 1)
            al = abs(lag)
            
            # Compute scores for the active lag
            if al == 0:
                scores = h * (1.0 - h)
            elif lag >= 0:
                scores = np.zeros(n)
                scores[:n-al] = h[:n-al] * (1.0 - h[al:])
            else:
                scores = np.zeros(n)
                scores[al:] = h[al:] * (1.0 - h[:n-al])
            
            # Try top-scoring positions
            top_idx = np.argsort(scores)[::-1][:30]
            for i in top_idx:
                if time.time() > deadline:
                    return h, best_c
                # Find j = i + lag (or i - lag) as the "partner"
                if lag >= 0:
                    j = i + lag if i + lag < n else i - lag
                else:
                    j = i + lag if i + lag >= 0 else i - lag
                if j < 0 or j >= n or j == i:
                    j = (i + al) % n  # fallback
                    if j == i:
                        continue
                
                # Try moving mass from i to j or j to i
                for direction in [1, -1]:
                    if direction == 1 and h[i] < h[j]:
                        continue
                    if direction == -1 and h[i] > h[j]:
                        continue
                    d = min(step, abs(h[i] - h[j]) * 0.5)
                    if d < 1e-8:
                        continue
                    h[i] += direction * d
                    h[j] -= direction * d
                    h = np.clip(h, 0.0, 1.0)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= direction * d
                        h[j] += direction * d
                        h = np.clip(h, 0.0, 1.0)
            if not improved:
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def pm_objective(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.sum(c ** p) ** (1.0 / p))

def pm_gradient(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    s = float(np.sum(c ** p))
    if s < 1e-30:
        return np.zeros(n)
    base = s ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ck = c[k]
        if ck < 1e-15:
            continue
        w = ck ** (p - 1)
        lag = k - (n - 1)
        al = abs(lag)
        if al == 0:
            g += w * (1.0 - 2.0 * x)
        elif lag >= 0:
            g[:n - al] += w * (1.0 - x[al:])
            g[al:] += -w * x[:n - al]
        else:
            g[al:] += w * (1.0 - x[:n - al])
            g[:n - al] += -w * x[al:]
    return base * g

def _sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_pm(h, n, p, maxiter=2000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(pm_objective, h, args=(n, p),
                       jac=lambda x: pm_gradient(x, n, p),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_subgrad(h, n, eps_frac=0.02):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    thresh = c_max - max(1e-4, eps_frac * c_max)
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        al = abs(lag)
        if al == 0:
            g += (1.0 - 2.0 * h)
        elif lag >= 0:
            g[:n - al] += (1.0 - h[al:])
            g[al:] += -h[:n - al]
        else:
            g[al:] += (1.0 - h[:n - al])
            g[:n - al] += -h[al:]
    g /= len(active)
    return g, c_max

def subgrad_optimize(h, n, max_iter=50000, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    mu = 0.9
    step = 0.003
    best_step = step
    no_improve = 0

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g, _ = active_subgrad(h, n, eps_frac=0.02)
        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
            best_step = step
        else:
            no_improve += 1
            if no_improve > 100:
                step *= 0.95
                no_improve = 0
        h = h_new
        step = max(1e-5, min(0.02, best_step * (1.0 + 0.01 * np.random.randn())))

    return best_h, best_c

            else:
                scores = np.zeros(n)

            top_idx = np.argsort(scores)[::-1][:40]

                        h = np.clip(h, 0.0, 1.0)

            else:
