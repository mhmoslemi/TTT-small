```python
    corr = np.correlate(h, 1.0 - h, mode="full")

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:

            h = np.clip(h, 0.0, 1.0)

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def lag_gradient(h, lag, n):
    """Exact gradient of overlap at a specific lag."""
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            # overlap = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[i] for i < n-lag: (1-h[i+lag])
            # d/dh[i] for i >= lag: -h[i-lag]
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        la = -lag
        if la < n:
            # overlap = sum_{i=la}^{n-1} h[i]*(1-h[i-la])
            # d/dh[i] for i >= la: (1-h[i-la])
            # d/dh[i] for i < n-la: -h[i+la]
            grad[la:] += (1.0 - h[:n - la])

    m = np.max(corr)

    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(corr)):

        if w < 1e-10:

        grad += w * lag_gradient(h, lag, n)

    return np.sum(x_in) - target

def nesterov_smooth_max(h0, target, n, beta, n_iter, step_init, t0, budget):
    """Nesterov accelerated gradient for smooth max objective with projection."""
    cur = h0.copy()
    y = cur.copy()
    best = cur.copy()
    best_c5 = compute_c5(cur)
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        t = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2  # Nesterov momentum
        # Actually standard Nesterov: t_k = (1+sqrt(1+4t_{k-1}^2))/2
        # Let me use the simpler form: alpha_k = (1 - t_{k-1})/t_k where t_k = (1+sqrt(1+4k^2))/2
        grad = smooth_max_grad(y, beta, n)
        # Gradient step
        new_y = project_h(y - step * grad, target)
        # Nesterov interpolation
        alpha = (1 - 2.0 / (it + 2))  # simplified momentum
        cur_new = alpha * new_y + (1 - alpha) * cur
        cur_new = project_h(cur_new, target)
        
        c_try = compute_c5(cur_new)
        if c_try < best_c5:
            best = cur_new.copy()
            best_c5 = c_try
        cur = cur_new
        y = cur_new
    return best, best_c5
```

```python
def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_lags_array(h):

    dx = 2.0 / n

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-15:
            break

        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()

    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

    return np.sum(x_in) - target
```

```python
def do_slsqp(h0, n, target, beta, maxiter, ftol=1e-16):

                     bounds=bounds, constraints=cons,
```

The problem is:
minimize max_k ∑_i h[i](1-h[i+k]) * dx
subject to: 0 ≤ h[i] ≤ 1, ∑h[i]*dx = 1

The objective is non-convex (bilinear terms h[i]*h[i+k]). But we can relax it:

Note that h[i](1-h[i+k]) ≤ h[i] (since 1-h[i+k] ≤ 1). This gives an upper bound but not a tight one.

Alternatively: h[i](1-h[i+k]) is the product of two variables in [0,1]. We can use the Relaxable Bilinear (RB) relaxation, but that would give a lower bound on the minimum, not an upper bound.

For an upper bound, we need a feasible solution. So CVXPY isn't directly helpful for the non-convex problem.

```python
            best_c5 = c_try

        # Decay learning rate
        lr = max(lr * 0.999, 0.001)
    
    return best, best_c5
```

For lag k ≥ 0:
overlap(k) = ∑_{i=0}^{n-1-k} h[i] * (1 - h[i+k])

∂overlap/∂h[j]:
- If j < n-k: h[j] appears in the term h[j]*(1-h[j+k]), derivative is (1-h[j+k])
- If j >= k: h[j] appears in the term h[j-k]*(1-h[j]), derivative is -h[j-k]
- These are not mutually exclusive (for k ≤ j < n-k, both apply)

So: ∂overlap/∂h[j] = (1-h[j+k]) * [j < n-k] - h[j-k] * [j ≥ k]

In vector form:
- First term: for j = 0 to n-k-1, add (1-h[j+k]) → grad[:n-k] += (1-h[k:])
- Second term: for j = k to n-1, subtract h[j-k] → grad[k:] -= h[:n-k]

For lag k < 0 (let la = -k > 0):
overlap(-la) = ∑_{i=la}^{n-1} h[i] * (1 - h[i-la])

∂overlap/∂h[j]:
- If j >= la: h[j] appears in term h[j]*(1-h[j-la]), derivative is (1-h[j-la])
- If j < n-la: h[j] appears in term h[j+la]*(1-h[j]), derivative is -h[j+la]

Vector form:
- First term: for j = la to n-1, add (1-h[j-la]) → grad[la:] += (1-h[:n-la])
- Second term: for j = 0 to n-la-1, subtract h[j+la] → grad[:n-la] -= h[la:]

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def lag_grad(h, lag, n):
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    elif -n < lag < 0:
        la = -lag
        grad[la:] += (1.0 - h[:n - la])
        grad[:n - la] -= h[la:]
    return grad

def binary_local_search(h_bin, n, t0, budget, max_iters=15000):
    cur = h_bin.copy()
    cur_lags = compute_lags_array(cur)
    cur_max = float(cur_lags.max())
    amax = int(cur_lags.argmax())
    best_h = cur.copy()
    best_max = cur_max
    rng = np.random.default_rng(789)
    stall = 0
    for it in range(max_iters):
        if budget - (time.time() - t0) < 3:
            break
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_h = cur.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        # Targeted: try moving ones that hurt the worst lag
        wl = amax - (n - 1)
        try_a = list(ones[rng.choice(len(ones), size=min(10, len(ones)), replace=False)])
        for a in try_a:
            if budget - (time.time() - t0) < 1:
                break
            try_b = list(zeros[rng.choice(len(zeros), size=min(10, len(zeros)), replace=False)])
            for b in try_b:
                if a == b:
                    continue
                cur[a] = 0.0
                cur[b] = 1.0
                nl = compute_lags_array(cur)
                nm = float(nl.max())
                if nm < cur_max - 1e-15:
                    cur_max = nm
                    amax = int(nl.argmax())
                    improved = True
                    break
                else:
                    cur[a] = 1.0
                    cur[b] = 0.0
            if improved:
                break
        if not improved:
            # Random escape
            a = int(ones[rng.integers(len(ones))])
            b = int(zeros[rng.integers(len(zeros))])
            cur[a] = 0.0
            cur[b] = 1.0
            nm = float(compute_lags_array(cur).max())
            if nm < cur_max + 0.03:
                cur_max = nm
                amax = int(compute_lags_array(cur).argmax())
            else:
                cur[a] = 1.0
                cur[b] = 0.0
    return best_h, best_max

def momentum_desc(h0, n, target, beta, n_iter, lr_init, t0, budget):
    cur = h0.copy()
    vel = np.zeros(n)
    best = cur.copy()
    best_c5 = compute_c5(cur)
    lr = lr_init
    mom = 0.85
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        m = corr.max()
        xs = np.clip(beta * (corr - m), -50, 50)
        w = np.exp(xs)
        w /= w.sum()
        sig = np.where(w > 0.005)[0]
        grad = np.zeros(n)
        for idx in sig:
            lag = int(idx) - (n - 1)
            grad += w[idx] * lag_grad(cur, lag, n)
        grad *= (2.0 / n)
        vel = mom * vel - lr * grad
        trial = project_h(cur + vel, target)
        c_try = compute_c5(trial)
        if c_try < best_c5:
            best = trial.copy()
            best_c5 = c_try
        cur = trial
        lr = max(lr * 0.998, 0.0005)
    return best, best_c5

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    stall = 0
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.015, -0.015, 0.007, -0.007, 0.003, -0.003, 0.001, -0.001]:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            if stall > 4:
                break
    return cur, best_c5
```
