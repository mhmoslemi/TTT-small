def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    """Compute gradient of smooth max of dx*correlate(h, 1-h)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    g = c * dx
    gs = g - g.max()
    eg = np.exp(alpha * gs)
    p = eg / eg.sum()
    
    # T1[i] = sum_{k=i}^{i+n-1} p[k]
    p_ext = np.concatenate([p, np.zeros(n)])
    cs = np.cumsum(p_ext)
    T1 = cs[n:n+n] - cs[:n]
    
    # T2[i] = sum_j p[i-j+n-1] * h[j] = convolve(p, h)[i+n-1]
    T2 = np.convolve(p, h, mode='full')[n-1:2*n-1]
    
    # T3[i] = sum_j p[j-i+n-1] * h[j] = correlate(p, h)[n-1-i]
    T3 = np.correlate(p, h, mode='full')[n-1::-1]
    
    return dx * (T1 - T2 - T3)

def subgrad_single(h, n, k):
    """Compute subgradient of G_k = dx*correlate(h,1-h)[k]."""
    dx = 2.0 / n
    s = k - (n - 1)
    grad = np.zeros(n)
    # (1-h[i-s]) for valid i-s
    if s >= 0:
        # i-s in [0, n-1] means i in [s, s+n-1], intersect [0, n-1] = [s, n-1]
        grad[s:] = (1 - h[:n-s])
    else:
        # i-s in [0, n-1] means i in [-s, n-1-s], intersect [0, n-1] = [0, n-1-s]
        grad[:n+s] += (1 - h[-s:])
    # -h[i+s] for valid i+s
    if s >= 0:
        # i+s in [0, n-1] means i in [0, n-1-s]
        grad[:n-s] -= h[s:]
    else:
        # i+s in [0, n-1] means i in [-s, n-1]
        grad[-s:] -= h[:n+s]
    return dx * grad

import numpy as np
import time

# Helper functions (top-level, no closures)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    """Return (smooth_obj_value, gradient) for log-sum-exp of dx*correlate(h,1-h)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    g = c * dx
    gs = g - g.max()
    eg = np.exp(alpha * gs)
    p = eg / eg.sum()
    S = g.max() + np.log(eg.sum()) / alpha
    
    # Gradient
    p_ext = np.concatenate([p, np.zeros(n)])
    cs = np.cumsum(p_ext)
    T1 = cs[n:2*n] - cs[:n]
    T2 = np.convolve(p, h, mode='full')[n-1:2*n-1]
    T3 = np.correlate(p, h, mode='full')[n-1::-1]
    grad = dx * (T1 - T2 - T3)
    return S, grad

def proj_grad(h, n, alpha, max_iter, deadline):
    """Projected gradient descent on smooth max with given alpha."""
    h_c = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_c, n)
    base_lr = 1.0 / (2.0 * alpha + 1.0)
    stall = 0

        if time.time() > deadline:

        _, grad = smooth_obj_grad(h_c, n, alpha)
        improved = False

            h_t = project_feasible(h_c - f * base_lr * grad, n)
