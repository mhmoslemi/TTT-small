    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_subgrad_at_lags(h, lag_indices, n):
    """Compute subgradient contribution from specific lag indices in correlation array"""
    grad = np.zeros(n)
    for li in lag_indices:
        lag = int(li) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def slsqp_optimize(h, target, beta, time_fn, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    
    def con_fn(h_in):
        return h_in.sum() - target
    
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fn, 'jac': lambda x: np.ones(n)}],
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        return project_h(res.x, target)
    except:
        return h

def subgrad_optimize(h, target, time_fn, max_iters=20000, m_act=20, momentum=0.85):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        corr = compute_corr(best)
        top = np.argsort(corr)[::-1][:m_act]
        g = compute_subgrad_at_lags(best, top, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in steps:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve % 500 == 0:
                scale = 0.01 * max(1.0, no_improve / 5000.0)
                p = np.random.randn(n) * scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.3
                no_improve = 0
            if no_improve > 5000:
                break
    return best, best_c5
