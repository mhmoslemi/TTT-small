```python
    corr = compute_corr(h)
    mc = np.max(corr)

            g[la:] += (1.0 - h[:n - la])

    m = np.max(corr)
    ex = np.exp(np.clip(beta * (corr - m), -500, 500))
    w = ex / ex.sum()
    val = (m + np.log(np.sum(ex) / len(corr)) / beta) * dx
    # Jacobian
    n = len(h_in)
    g = np.zeros(n)
    # Only compute for active (high weight) lags for efficiency

            g[lag:] -= wk * h_in[:n - lag]
        else:
            la = -lag
            g[la:] += wk * (1.0 - h_in[:n - la])
            g[:n - la] -= wk * h_in[la:]
    return val, g
```

- i in [start, end) (h[i] changed): affects c[k] for all k (via the h[i] factor)
- i+k in [start, end), i.e., i in [start-k, end-k) (h[i+k] changed): affects c[k]

So for each lag k, the change in c[k] is:
Δc[k] = sum_{i in overlap1(k)} (h_new[i] - h_old[i]) * (1-h[i+k]) + sum_{i in overlap2(k)} h[i] * (-(h_new[i+k] - h_old[i+k]))

where overlap1 and overlap2 are the relevant index ranges.

Let me try a different approach: instead of full SLSQP per block, use a simple gradient step per block (computed analytically).

For a block at [start, end), the gradient of C5 (smooth max) with respect to h[start:end] can be computed as:
- Compute full correlation c
- Get weights w from smooth max
- For each active lag k, compute the derivative of c[k] w.r.t. h[start:end]
- Sum weighted derivatives

The derivative of c[k] w.r.t. h[j] (if j is in the block) is:
- If k ≥ 0:
  - d c[k] / d h[j] = (1-h[j+k]) if j+k < n (i.e., j < n-k)
  - d c[k] / d h[j] += -h[j-k] if j-k ≥ 0 (i.e., j ≥ k)
- If k < 0 (let la = -k):
  - d c[k] / d h[j] = (1-h[j-la]) if j-la ≥ 0 (i.e., j ≥ la)
  - d c[k] / d h[j] += -h[j+la] if j+la < n (i.e., j < n-la)

So for a given block variable h[j], its gradient contribution from lag k is:
- (1-h[j+k]) * 1[j < n-k] - h[j-k] * 1[j ≥ |k|] (for k ≥ 0)
- (1-h[j-|k|]) * 1[j ≥ |k|] - h[j+|k|] * 1[j < n-|k|] (for k < 0)

This is O(1) per (j, k) pair. For a block of size B and |active| lags, computing the block gradient is O(B * |active|).

- Represent h in terms of its DFT (circular, with zero-padding for linear)

- Try to find a DFT that minimizes the peak of the correlation

Specifically, for zero-padded sequences:
H = FFT(h_padded, 2n), G = FFT((1-h)_padded, 2n)
c[k] = IFFT(H * conj(G))[k] (with appropriate indexing)

We want to minimize max |c[k]| (well, max c[k] since it should be positive).

In the frequency domain, c = IFFT(H * conj(G)). We want the time-domain signal c to be as flat as possible. This is like a "minimum peak-to-average power ratio" problem, which is related to the OD glyph problem in OFDM.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgrad(h, n, corr, active_frac):
    mc = np.max(corr)
    thr = mc - active_frac * (mc + 1e-10)
    active = np.where(corr >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    g = np.zeros(n)
    for i in active:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / len(active)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500))) / len(corr)) / beta

def sml_fn(h_in, beta, n):
    dx = 2.0 / n
    return smooth_max_val(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sml_fn_jac(h_in, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    ex = np.exp(np.clip(beta * (corr - m), -500, 500))
    w = ex / ex.sum()
    val = (m + np.log(np.sum(ex) / len(corr)) / beta) * dx
    n = len(h_in)
    g = np.zeros(n)
    top_k = np.argsort(w)[::-1][:max(10, int(0.03 * len(corr)))]
    for ki in top_k:
        wk = w[ki] * dx
        lag = int(ki) - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h_in[lag:])
            g[lag:] -= wk * h_in[:n - lag]
        else:
            la = -lag
            g[la:] += wk * (1.0 - h_in[:n - la])
            g[:n - la] -= wk * h_in[la:]
    return val, g

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_fn_jac, h0, method='SLSQP', jac=True,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_line_search(h, n, target, t0, deadline, n_iter=30000):
    h_curr = h.copy()
    h_mom = h.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = 0.04
    stall = 0
    active_frac = 0.015

    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        corr = compute_corr(h_mom)
        g = get_subgrad(h_mom, n, corr, active_frac)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        best_alpha = 0.0
        best_trial = best_c5
        alphas = [step, step * 0.5, step * 0.25, step * 1.5, step * 0.75]
        for a in alphas:
            ht = project_h(h_mom - a * g, target)
            ct = compute_c5(ht)
            if ct < best_trial:
                best_trial = ct
                best_alpha = a

        if best_alpha > 0:
            h_new = project_h(h_mom - best_alpha * g, target)
            mu = (1.0 - np.sqrt(1.0 + 8.0 / (it + 1))) / 2.0
            h_mom = h_new + mu * (h_new - h_curr)
            h_mom = np.clip(h_mom, 0, 1)
            h_curr = h_new
            step = min(0.15, step * 1.01)
            stall = 0
            if best_trial < best_c5:
                best_h, best_c5 = h_new.copy(), best_trial
        else:
            stall += 1
            step *= 0.75
            active_frac = min(0.08, active_frac * 1.3)
            if stall > 300:
                break
    return best_h, best_c5

def gaussian_smooth(h, sigma):
    n = len(h)
    k = int(2 * sigma) + 1
    if k < 3:
        return h.copy()
    xs = np.arange(-k, k + 1)
    kern = np.exp(-xs**2 / (2 * sigma**2))
    kern /= kern.sum()
    padded = np.concatenate([h[-k:], h, h[:k]])
    smoothed = np.convolve(padded, kern, mode='valid')[:n]
    return smoothed

def coord_descent(cur, target, n, sweeps, t0, deadline):
    best_c5 = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005)
    for sw in range(sweeps):
        if time.time() > deadline - 2:
            break
        improved = False
        for j in range(n):
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    resolutions = sorted(set([len(h_parent)] + [256, 320, 384, 448]))
    resolutions = [r for r in resolutions if 128 <= r <= 600]

    for n_pts in resolutions:
        rem = deadline - time.time()
        if rem < 100:
            break
        target = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

        # Smooth to reduce oscillations
        cur_s = gaussian_smooth(cur, 2.0)
        cur_s = project_h(cur_s, target)
        c5_s = compute_c5(cur_s)
        if c5_s < cur_c5:
            cur, cur_c5 = cur_s, c5_s
            if c5_s < best_c5:
                best_h, best_c5 = cur.copy(), c5_s

        # Subgradient with line search (main phase)
        rem = deadline - time.time()
        if rem > 150:
            cur, cur_c5 = subgrad_line_search(cur, n_pts, target, t0, deadline, n_iter=40000)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # SLSQP with multiple smoothness levels
        rem = deadline - time.time()
        if rem > 50:
            for beta in [3000, 20000]:
                rem = deadline - time.time()
                if rem < 20:
                    break
                cur = do_slsqp(cur, n_pts, target, beta, 4000, 1e-14)
                cur_c5 = compute_c5(cur)
                if cur_c5 < best_c5:
                    best_h, best_c5 = cur.copy(), cur_c5

        # Random perturbation + re-optimize (escape local optimum)
        rem = deadline - time.time()
        if rem > 80:
            noise = np.random.normal(0, 0.05, n_pts)
            perturbed = project_h(cur + noise, target)
            perturbed_c5 = compute_c5(perturbed)
            # Only keep perturbation if not too bad
            if perturbed_c5 < best_c5 + 0.02:
                cur2 = perturbed
            else:
                cur2 = cur
            cur2, c5_2 = subgrad_line_search(cur2, n_pts, target, t0, deadline, n_iter=20000)
            if c5_2 < best_c5:
                best_h, best_c5 = cur2.copy(), c5_2
                cur = cur2

        # Coordinate descent polish
        rem = deadline - time.time()
        if rem > 25:
            cur, cur_c5 = coord_descent(cur, target, n_pts, 8, t0, deadline)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

    # Final intensive polish at best resolution
    rem = deadline - time.time()
    if rem > 40:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = project_h(best_h, tf)
        # SLSQP with very high beta
        best_h = do_slsqp(best_h, nf, tf, 50000, 6000, 1e-15)
        best_c5 = compute_c5(best_h)

        rem = deadline - time.time()
        if rem > 25:
            best_h, best_c5 = subgrad_line_search(best_h, nf, tf, t0, deadline, n_iter=25000)

        rem = deadline - time.time()
        if rem > 15:
            best_h, best_c5 = coord_descent(best_h, tf, nf, 15, t0, deadline)

        # Final micro-snap: try pushing values toward 0 or 1 slightly
        rem = deadline - time.time()
        if rem > 8:
            for push in [0.02, -0.02, 0.01, -0.01]:
                trial = np.clip(best_h + push * (best_h - 0.5) * 2, 0, 1)
                trial = project_h(trial, tf)
                ct = compute_c5(trial)
                if ct < best_c5:
                    best_h, best_c5 = trial, ct

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
