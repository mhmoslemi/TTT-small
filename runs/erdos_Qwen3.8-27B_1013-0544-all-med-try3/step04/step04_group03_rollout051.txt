import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_subgrad(h, target, n, n_iter, t0, budget, top_k=16):
    """Nesterov accelerated subgradient with Polyak step."""
    cur = h.copy()
    y = h.copy()  # Nesterov's y point
    best_h = h.copy()
    best_c5 = compute_c5(cur)
    momentum = 0.0
    step_init = 0.05
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        # Compute gradient at y
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - y[lag:])
                    grad[lag:] -= y[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - y[:n - la])
                grad[:n - la] -= y[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Polyak step size
        f_y = compute_c5(y)
        gamma = max(1e-6, (f_y - best_c5) / (gn * gn + 1e-14))
        gamma = min(gamma, 0.5)
        
        # Nesterov update

        y_new = (1.0 - beta_t) * cur + beta_t * y + beta_t * (-gamma * grad / gn * gn)
        # Actually, standard Nesterov: y_{t+1} = x_{t+1} + (t/(t+2))(x_{t+1} - x_t)

        y = x_new + momentum_t * (x_new - cur)
        y = np.clip(y, 0.0, 1.0)
        
        cur = x_new
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = cur.copy()

def frank_wolfe(h, target, n, n_iter, t0, budget, top_k=8):
    """Frank-Wolfe (conditional gradient) method."""
    cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(cur)
    n_half = n // 2
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        # Compute gradient
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            elif lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        
        # Linear oracle: minimize grad^T v s.t. 0 <= v <= 1, sum(v) = n/2
        idx_sorted = np.argsort(grad)
        v = np.zeros(n)
        v[idx_sorted[:n_half]] = 1.0
        
        # Step size: backtracking line search or 2/(t+2)
        gamma = 2.0 / (it + 2.0)
        trial = (1.0 - gamma) * cur + gamma * v
        trial = np.clip(trial, 0.0, 1.0)
        # Re-project to fix sum
        trial = project_h(trial, n / 2.0)
        
        c_try = compute_c5(trial)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = trial.copy()
        cur = trial
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def get_subgrad(y, n, top_k):
    corr = np.correlate(y, 1.0 - y, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if 0 <= lag < n:
            grad[:n - lag] += (1.0 - y[lag:])
            grad[lag:] -= y[:n - lag]
        elif -n < lag < 0:
            la = -lag
            grad[la:] += (1.0 - y[:n - la])
            grad[:n - la] -= y[la:]
    grad /= top_k
    return grad

def nesterov_subgrad(h, target, n, n_iter, t0, budget, top_k=12):
    cur = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(cur)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        grad = get_subgrad(y, n, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        f_y = compute_c5(y)
        gamma = max(1e-7, (f_y - best_c5 + 1e-6) / (gn + 1e-14))
        gamma = min(gamma, 0.4)
        step = grad * min(gamma, 0.3)
        x_new = project_h(y - step, target)
        mu = (1.0 + it) / (3.0 + it)
        y = x_new + mu * (x_new - cur)
        y = np.clip(y, 0.0, 1.0)
        cur = x_new
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = cur.copy()
    return best_h, best_c5

def frank_wolfe(h, target, n, n_iter, t0, budget, top_k=8):
    cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(cur)
    n_half = n // 2
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        grad = get_subgrad(cur, n, top_k)
        idx_sorted = np.argsort(grad)
        v = np.zeros(n)
        v[idx_sorted[:n_half]] = 1.0
        gamma = 2.0 / (it + 2.0)
        trial = (1.0 - gamma) * cur + gamma * v
        trial = project_h(np.clip(trial, 0.0, 1.0), n / 2.0)
        c_try = compute_c5(trial)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = trial.copy()
        cur = trial
    return best_h, best_c5

def block_flip_diversify(h, n, target, rng, n_flips=8):
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_flips):
        blk_len = rng.integers(8, n // 4 + 1)
        start = rng.integers(0, n - blk_len)
        trial = h.copy()
        trial[start:start + blk_len] = 1.0 - trial[start:start + blk_len]
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5:
            best = trial.copy()
            best_c5 = c
    return best, best_c5
