def compute_A(h, n, i):
    """Compute A vector: A[k_idx] = coefficient of h[i] in c_k"""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    for k in range(-(n-1), n):
        k_idx = k + (n-1)
        a = 0.0
        if 0 <= i + k < n:
            a += dx * (1 - h[i+k])
        if 0 <= i - k < n:
            a -= dx * h[i-k]
        A[k_idx] = a
    return A
