    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

        return project_h(r.x, target)
    except:

    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        # Compute subgradient at active lags
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        thr = mc - 0.01 * mc
        idx = np.where(corr >= thr)[0]
        if len(idx) > 50:
            idx = idx[np.argsort(corr[idx])[::-1][:50]]
        g = np.zeros(n)

                g[:n-lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n-lag]
            else:
                la = -lag
                g[la:] += (1.0 - y[:n-la])

            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 0.5:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(cur[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(77)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 1:
            break
        improved = False
        for _ in range(n):
            if time.time() - t0 > budget - 0.5:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004):
                trial = cur.copy()
                trial[j] = np.clip(cur[j] + d, 0, 1)
                trial[k] = np.clip(cur[k] - d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def kick_and_refine(h0, n, target, bounds, t0, budget, kick_type, rng):
    h = h0.copy()
    if kick_type == 0:
        nflip = rng.randint(3, 12)
        idx = rng.choice(n, nflip, replace=False)
        h[idx] = 1.0 - h[idx]
    elif kick_type == 1:
        sigma = rng.choice([0.05, 0.1, 0.15, 0.25])
        mask = rng.random(n) < 0.25
        h[mask] += rng.randn(mask.sum()) * sigma
    else:
        blen = rng.randint(4, 18)
        i = rng.randint(0, max(1, n - blen * 2))
        j = i + blen + rng.randint(0, max(1, n - 2 * blen - i))
        if j + blen <= n:
            h[i:i + blen], h[j:j + blen] = h[j:j + blen].copy(), h[i:i + blen].copy()
    h = project_h(h, target)
    rem = budget - (time.time() - t0)
    if rem > 5:
        h = do_slsqp(h, n, target, 400, bounds, 1500, 1e-12)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    n_pts = best_n
    target = n_pts / 2.0
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    cur = project_h(best_h.copy(), target)

    # Phase 1: Progressive SLSQP with increasing smoothness
    for beta in [100, 300, 800, 2000, 5000]:
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        cur = do_slsqp(cur, n_pts, target, beta, bounds, 6000, 1e-16)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Nesterov
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        cur, c = nesterov_subgrad(cur, n_pts, target, 6000, 0.015, t0, budget_s)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Coordinate + pair
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        cur, c = coord_descent(cur, target, n_pts, 15, t0, budget_s)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        cur, c = pair_descent(cur, target, n_pts, 10, t0, budget_s)
        if c < best_c5:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

    # Phase 2: Multi-start kicks
    rem = budget_s - (time.time() - t0)
    kick_count = 0
    no_improve = 0
    while rem > 80:
        kt = kick_count % 3
        h_k = kick_and_refine(best_h, n_pts, target, bounds, t0, budget_s, kt, rng)
        c_k = compute_c5(h_k)
        if c_k < best_c5 - 1e-13:
            best_h, best_c5, best_n = h_k.copy(), c_k, n_pts
            no_improve = 0
        else:
            no_improve += 1
        kick_count += 1
        if no_improve > 15:
            # Try a random restart
            h_rand = rng.randint(0, 2, size=n_pts).astype(float)
            h_rand = project_h(h_rand, target)
            h_rand = do_slsqp(h_rand, n_pts, target, 300, bounds, 2000, 1e-12)
            c_r = compute_c5(h_rand)
            if c_r < best_c5:
                best_h, best_c5, best_n = h_rand.copy(), c_r, n_pts
                no_improve = 0
            else:
                no_improve = 0  # reset counter
        rem = budget_s - (time.time() - t0)

    # Phase 3: Try alternative resolutions
    rem = budget_s - (time.time() - t0)
    alt_res = [320, 256, 448, 384]
    for n_alt in alt_res:
        if rem < 35:
            break
        if n_alt == best_n:
            continue
        h_alt = project_h(resample(best_h, n_alt), n_alt / 2.0)
        bnd_alt = Bounds(np.zeros(n_alt), np.ones(n_alt))
        for beta in [400, 1500]:
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            h_alt = do_slsqp(h_alt, n_alt, n_alt / 2.0, beta, bnd_alt, 4000, 1e-15)
            c_alt = compute_c5(h_alt)
            if c_alt < best_c5:
                best_h, best_c5, best_n = h_alt.copy(), c_alt, n_alt

    # Phase 4: Final intensive polish
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [2000, 5000, 10000]:
            rem = budget_s - (time.time() - t0)
            if rem < 4:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 6000, 1e-16)
            best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 8:
        nf = len(best_h)
        best_h, best_c5 = coord_descent(best_h, nf / 2.0, nf, 25, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 5:
        nf = len(best_h)
        best_h, best_c5 = pair_descent(best_h, nf / 2.0, nf, 15, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 4:
        nf = len(best_h)
        best_h, best_c5 = nesterov_subgrad(best_h, nf, nf / 2.0, 3000, 0.005, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
