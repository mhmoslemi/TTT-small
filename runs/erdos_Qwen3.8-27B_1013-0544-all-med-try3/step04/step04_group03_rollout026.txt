    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):

    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def regret_binary_search(h_bin, n, t0, budget, n_iter=300000, window=None):
    """Binary local search using regret moves (try many, pick best)."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    no_improve = 0
    
    if window is None:
        window = n  # full window
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try regret: sample many swaps, pick the best
        n_try = min(200, len(ones) * len(zeros))
        best_delta = 0  # we want to minimize, so negative delta is good
        best_a, best_b = -1, -1
        
        # Focus on nearby swaps for efficiency
        for _ in range(n_try):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            c_try = compute_c5(h)
            delta = c_try - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_a, best_b = a, b
            h[a] = 1.0
            h[b] = 0.0
        
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            cur_c5 += best_delta
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Accept a random move (simulated annealing style)
            frac = it / max(n_iter, 1)
            temp = 0.1 * (0.01) ** (frac ** 0.5)
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            c_try = compute_c5(h)
            if c_try < cur_c5 or rng.random() < np.exp(-(c_try - cur_c5) / max(temp, 1e-10)):
                cur_c5 = c_try
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
            else:
                h[a] = 1.0
                h[b] = 0.0
                no_improve += 1
        
        if no_improve > 50000:
            break
    
    return best_h, best_c5

def lag_weighted_obj(h_in, n, top_k=10):
    """Objective that focuses on the top lags."""
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    # Use a power-sum approximation of the max
    c_abs = np.abs(c)
    # Sort and take top k
    c_sorted = np.sort(c_abs)[::-1]
    top = c_sorted[:top_k]
    # Weighted sum with emphasis on the maximum
    return np.sum(top ** 4) / max(np.sum(c_abs ** 4), 1e-30) * np.max(c_abs) * (2.0 / n)

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=8):
    """Subgradient descent with focus on top lags."""
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            stall += 1
            step *= 0.8
            if stall > 200:
                break
    
    return cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.001, -0.001]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                if trial[j] < 0 or trial[j] > 1:
                    continue
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

                ba, bb = a, b
        if ba >= 0:
            h[ba] = 0.0
            h[bb] = 1.0
            cur_c5 = best_c
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            frac = it / max(n_iter, 1)
            temp = 0.08 * (0.005) ** (frac ** 0.7)
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            ct = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0/n)
            if ct < cur_c5 or rng.random() < np.exp(-(ct - cur_c5) / max(temp, 1e-12)):
                cur_c5 = ct
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = ct
                    best_h = h.copy()
                    no_improve = 0
            else:
                h[a] = 1.0
                h[b] = 0.0
                no_improve += 1
        if no_improve > 40000:
            break
    return best_h, best_c5

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=10):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            elif lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            stall += 1
            step *= 0.85
            if stall > 300:
                break
    return cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                if trial[j] < -0.01 or trial[j] > 1.01:
                    continue
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.zeros(n_init)
        h_parent[:n_init // 2] = 1.0
        h_parent = project_h(h_parent, n_init / 2.0)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    n_cur = best_n
    target = n_cur / 2.0
    print(f"Start: n={n_cur} C5={best_c5:.6f}", flush=True)

    # Phase 1: Binary greedy + SA
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_bin = binary_snap(best_h, n_cur)
        h_bin, bc = greedy_lag_balance(h_bin, n_cur, t0, budget_s, max_rounds=80000)
        print(f"Greedy: {bc:.6f}", flush=True)
        h_bin2, bc2 = regret_sa(h_bin, n_cur, t0, budget_s, n_iter=80000, n_try=60)
        if bc2 < bc:
            bc = bc2
            h_bin = h_bin2
        print(f"SA: {bc:.6f}", flush=True)
        h_r = h_bin + 0.01 * np.random.randn(n_cur)
        h_r = project_h(h_r, target)
        for beta in [500, 2000, 8000]:
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            h_r = do_slsqp(h_r, n_cur, target, beta, maxiter=5000)
            ct = compute_c5(h_r)
            if ct < best_c5:
                best_h, best_c5 = h_r.copy(), ct
                print(f"  refine b={beta}: {ct:.6f}", flush=True)

    # Phase 2: Multi-start
    rem = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 777)
    for att in range(5):
        if rem < 35:
            break
        hp = best_h + rng.standard_normal(n_cur) * 0.12
        hb = binary_snap(hp, n_cur)
        hb, bc = regret_sa(hb, n_cur, t0, budget_s, n_iter=60000, n_try=50)
        h_r = hb + 0.008 * rng.standard_normal(n_cur)
        h_r = project_h(h_r, target)
        for beta in [1000, 5000]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            h_r = do_slsqp(h_r, n_cur, target, beta, maxiter=4000)
        ct = compute_c5(h_r)
        if ct < best_c5:
            best_h, best_c5 = h_r.copy(), ct
            print(f"  ms{att}: {ct:.6f}", flush=True)

    # Phase 3: Subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 12000, 0.015, t0, budget_s, top_k=12)
        print(f"Subgrad: {best_c5:.6f}", flush=True)

    # Phase 4: High-beta SLSQP
    rem = budget_s - (time.time() - t0)
    for beta in [5000, 20000, 60000]:
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=10000)
        best_c5 = compute_c5(best_h)
        print(f"  SLSQP b={beta}: {best_c5:.6f}", flush=True)

    # Phase 5: Try higher res
    rem = budget_s - (time.time() - t0)
    for n_hr in [448, 512]:
        if rem < 55:
            break
        th = n_hr / 2.0
        old_x = np.linspace(0, 2, n_cur, endpoint=False)
        new_x = np.linspace(0, 2, n_hr, endpoint=False)
        ch = np.interp(new_x, old_x, best_h)
        ch = project_h(ch, th)
        for beta in [500, 3000, 10000]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            ch = do_slsqp(ch, n_hr, th, beta, maxiter=4000)
        ct = compute_c5(ch)
        if ct < best_c5:
            best_h, best_c5, best_n, n_cur, target = ch.copy(), ct, n_hr, n_hr, th
            print(f"  res{n_hr}: {ct:.6f}", flush=True)

    # Phase 6: Coord descent + final SLSQP
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 80, t0, budget_s)
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h = do_slsqp(best_h, n_cur, target, 50000, maxiter=12000)
        best_c5 = compute_c5(best_h)
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 50, t0, budget_s)
    print(f"FINAL: n={best_n} C5={best_c5:.6f}", flush=True)
    return best_h, best_c5, best_n
