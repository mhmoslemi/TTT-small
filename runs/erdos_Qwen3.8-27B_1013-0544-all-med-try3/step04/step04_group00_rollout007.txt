    """Compute gradient of g(shift) w.r.t. h, where shift = k - (n-1)."""
    grad = np.zeros(n)

        grad[s:] += (1 - h[:n-s])
        # j-shift = j+s in [0,n) means j in [0, n-s)
        grad[:n-s] -= h[s:n]
