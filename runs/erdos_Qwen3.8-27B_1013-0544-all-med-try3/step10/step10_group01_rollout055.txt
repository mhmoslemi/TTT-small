import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def active_set_obj(h, n, active, alpha, mu):

    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_set_grad(h, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()

def get_active(c_vec, max_val, tol_frac, top_k):
    threshold = max_val - max(1e-10, tol_frac * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    return active

def lbfgs_smooth(h, n, deadline, alpha, mu, maxit=None):
    bounds = [(0.0, 1.0)] * n

                      bounds=bounds,

                               'gtol': 1e-18, 'maxfun': maxit * 12,
                               'maxls': 50, 'maxcor': 60})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def lbfgs_active(h, n, active, deadline, alpha, mu, maxit=None):

def slqp_polish(h, n, deadline, alpha, mu=0.0):
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0

    def sq_jac(x, nn):
        return np.ones(nn)

    def slq_obj(x, nn, al):
        c = compute_c5_vec(x, nn)
        m = c.max()
        sh = al * (c - m)
        smx = sh.max()
        if smx > 400:
            val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / al
        else:
            val = m + np.log(np.sum(np.exp(sh))) / al
        return float(val)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    rem = max(10, int(deadline - time.time()))
    maxit = min(2000, max(100, rem * 8))
    try:
        res = minimize(slq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'maxls': 100})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def grad_descent_pair(h, n, deadline, steps=None, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]

            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()

            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n

            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():

def simulate_anneal_binary(h, n, deadline, iterations=50000):

        return best_h, best_c
    T0 = 0.002
    T1 = 0.00001
    cur_c = best_c
    count = 0
    while time.time() < deadline and count < iterations:

        i = np.random.randint(len(ones))
        j = np.random.randint(len(zeros))

        h[oi] = 0.0
        h[oj] = 1.0
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        delta = c_new - cur_c

                best_h = h.copy()

            h[oi] = 1.0
            h[oj] = 0.0
            h = project_feasible(h, n)
        count += 1
