    n_freq = rng.randint(1, 4)
    for _ in range(n_freq):
        f = rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.45)
        d += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    h = 0.5 + d
    return project(h, n)

def directed_pairs(h, n, deadline, max_moves=50000):
    best_h = h.copy()
    best_c = c5_val(best_h, n)
    dx = 2.0 / n
    rng_p = np.random.RandomState(12345)
    delta = 0.005
    moves = 0
    stalls = 0

    while delta >= 1e-7 and time.time() < deadline - 10 and moves < max_moves:
        c_v = c5_vec(best_h, n)
        max_v = c_v.max()
        center = n - 1
        # Top active constraints
        top = np.argsort(-c_v)[:16]
        tol = max(1e-8, 0.002 * max_v)
        active = top[c_v[top] >= max_v - tol]
        
        # Compute gradients for active constraints
        g_active = np.array([grad_cs(best_h, n, int(idx) - center, dx) for idx in active])
        
        # Weighted gradient (emphasis on the max)
        w_act = c5_vec(best_h, n)[active]
        w_act = w_act / w_act.sum()
        g_w = (w_act[:, None] * g_active).sum(axis=0)
        
        # Sort by weighted gradient
        order = np.argsort(g_w)
        low_end = order[:n // 3]
        high_end = order[-(n // 3):]
        
        improved_this_level = False
        attempts = 0
        max_attempts = 300
        
        while attempts < max_attempts and time.time() < deadline - 5:
            attempts += 1
            i = high_end[rng_p.randint(len(high_end))]
            j = low_end[rng_p.randint(len(low_end))]
            if i == j:
                continue
            
            # Try moving delta from i to j
            if best_h[i] >= delta and best_h[j] <= 1.0 - delta:
                best_h[i] -= delta
                best_h[j] += delta
                c_new = c5_val(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved_this_level = True
                    moves += 1
                else:
                    best_h[i] += delta
                    best_h[j] -= delta
            
            # Try moving delta from j to i
            if best_h[j] >= delta and best_h[i] <= 1.0 - delta:
                best_h[j] -= delta
                best_h[i] += delta
                c_new = c5_val(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved_this_level = True
                    moves += 1
                else:
                    best_h[j] += delta
                    best_h[i] -= delta
        
        if improved_this_level:
            stalls = 0
        else:
            stalls += 1
            delta *= 0.65
            if stalls > 30:
                break
    
    return best_h, best_c

def fine_broadcast(h, n, deadline):
    """Very fine pair search focusing on neighborhoods."""
    best_h = h.copy()
    best_c = c5_val(best_h, n)
    dx = 2.0 / n
    c_v = c5_vec(best_h, n)
    center = n - 1
    top16 = np.argsort(-c_v)[:16]
    
    # Precompute gradients for top constraints
    g_list = [grad_cs(best_h, n, int(idx) - center, dx) for idx in top16]
    g_arr = np.array(g_list)
    
    delta = 0.001
    found = True
    while delta >= 1e-8 and found and time.time() < deadline - 5:
        found = False
        c_v = c5_vec(best_h, n)
        max_v = c_v.max()
        tol = max(1e-9, 0.001 * max_v)
        act = np.where(c_v >= max_v - tol)[0]
        
        for i in range(n):
            if time.time() > deadline - 3:
                return best_h, best_c
            if best_h[i] < delta * 2 or best_h[i] > 1.0 - delta * 2:
                continue
            # Check gradient directions
            g_plus = g_arr[:, i].mean()  # average gradient at i
                
            # Find best j to couple with
            for j in range(max(0, i - 50), min(n, i + 51)):
                if i == j:
                    continue
                if best_h[j] < delta or best_h[j] > 1.0 - delta:
                    continue
                # Move from i to j if g[i] > g[j] (reduces max)
                for direction in [1, -1]:
                    if direction == 1 and best_h[i] >= delta and best_h[j] <= 1.0 - delta:
                        best_h[i] -= delta
                        best_h[j] += delta
                        c_new = c5_val(best_h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            found = True
                            # Update gradients
                            g_list = [grad_cs(best_h, n, int(idx) - center, dx) for idx in top16]
                            g_arr = np.array(g_list)
                        else:
                            best_h[i] += delta
                            best_h[j] -= delta
                    elif direction == -1 and best_h[j] >= delta and best_h[i] <= 1.0 - delta:
                        best_h[j] -= delta
                        best_h[i] += delta
                        c_new = c5_val(best_h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            found = True
                            g_list = [grad_cs(best_h, n, int(idx) - center, dx) for idx in top16]
                            g_arr = np.array(g_list)
                        else:
                            best_h[j] += delta
                            best_h[i] -= delta
        if not found:
            delta *= 0.5
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)
    
    h0 = project(h0, n0)
    best_h = h0.copy()
    best_c = c5_val(h0, n0)
    best_n = n0
    
    # Generate diverse starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    candidates = [
        make_beatty(n0, golden * 0.5),
        make_beatty(n0, 0.4312),
        make_beatty(n0, np.sqrt(2.0) * 0.5),
        make_random_bin(n0, rng),
        make_random_bin(n0, rng),
        make_fourier(n0, rng),
        make_fourier(n0, rng),
    ]
    for hc in candidates:
        cc = c5_val(hc, n0)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
    
    # Phase 1: L-BFGS warmup with increasing alpha
    p1_end = min(t0 + 80, deadline - 850)
    bounds = [(0.0, 1.0)] * best_n
    alpha_schedule = [500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alpha_schedule:
        if time.time() > p1_end - 5:
            break
        mu = 0.01
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(best_n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 6000,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project(res.x, best_n)
            c_new = c5_val(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: Active set optimization with various top_k
    p2_end = min(time.time() + 250, deadline - 450)
    configs = [
        (4, 20000, 0.001),
        (6, 50000, 0.001),
        (8, 100000, 0.0005),
        (10, 200000, 0.0005),
        (12, 500000, 0.0003),
        (16, 1000000, 0.0003),
        (20, 2000000, 0.0002),
        (24, 5000000, 0.0002),
    ]
    for top_k, alpha, mu in configs:
        if time.time() > p2_end - 10:
            break
        for _ in range(2):
            if time.time() > p2_end - 8:
                break
            c_v = c5_vec(best_h, best_n)
            thr = c_v.max() - max(1e-8, 0.001 * c_v.max())
            active = np.where(c_v >= thr)[0]
            if len(active) > top_k:
                active = np.argsort(-c_v[active])[:top_k]
            if len(active) == 0:
                continue
            try:
                maxit = max(200, min(1500, int((p2_end - time.time() - 5) * 2)))
                res = minimize(obj_act, best_h, jac=grad_act,
                              args=(best_n, active, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project(res.x, best_n)
                c_new = c5_val(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 3: Directed pair search at current resolution
    p3_end = min(time.time() + 250, deadline - 250)
    if p3_end > time.time() + 20:
        h3, c3 = directed_pairs(best_h, best_n, p3_end, max_moves=80000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try higher resolution n=1000
    n_up = 1000
    p4_end = min(time.time() + 250, deadline - 150)
    if p4_end > time.time() + 60:
        h_up = upsample(best_h, best_n, n_up)
        c_up = c5_val(h_up, n_up)
        bounds_up = [(0.0, 1.0)] * n_up
        for alpha in [5000, 20000, 100000, 500000, 2000000]:
            if time.time() > p4_end - 10:
                break
            try:
                mu = 0.005
                res = minimize(obj_full, h_up, jac=grad_full,
                              args=(n_up, alpha, mu), method='L-BFGS-B',
                              bounds=bounds_up,
                              options={'maxiter': 600, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': 5000,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project(res.x, n_up)
                c_new = c5_val(h_new, n_up)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new
                    bounds_up = bounds
            except Exception:
                pass
        # Active set at higher res
        for top_k, alpha, mu in [(8, 200000, 0.001), (12, 1000000, 0.0005), (16, 5000000, 0.0003)]:
            if time.time() > p4_end - 10:
                break
            if best_n == n_up:
                c_v = c5_vec(best_h, best_n)
                thr = c_v.max() - max(1e-8, 0.001 * c_v.max())
                active = np.where(c_v >= thr)[0]
                if len(active) > top_k:
                    active = np.argsort(-c_v[active])[:top_k]
                if len(active) > 0:
                    try:
                        res = minimize(obj_act, best_h, jac=grad_act,
                                      args=(best_n, active, alpha, mu),
                                      method='L-BFGS-B', bounds=bounds_up,
                                      options={'maxiter': 800, 'ftol': 1e-20,
                                               'gtol': 1e-14, 'maxfun': 5000,
                                               'maxls': 40, 'maxcor': 40})
                        h_new = project(res.x, best_n)
                        c_new = c5_val(h_new, best_n)
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h_new
                    except Exception:
                        pass
    
    # Phase 5: Fine pair search
    p5_end = min(time.time() + 120, deadline - 40)
    if p5_end > time.time() + 15:
        h5, c5 = directed_pairs(best_h, best_n, p5_end, max_moves=40000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Ultra-fine local search
    p6_end = deadline - 3
    if p6_end > time.time() + 10:
        h6, c6 = fine_broadcast(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Final
    best_h = project(best_h, best_n)
    best_c = c5_val(best_h, best_n)
    return best_h, best_c, best_n
