import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, fun, jac, args, maxiter=None):
    if maxiter is None:
        maxiter = max(200, int((deadline - time.time()) * 2))
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, args=args, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20,

        c_new = compute_c5(h_new, n)

def subgradient_averaging(h, n, deadline, max_iter=50000, tol_active=0.001):

    best_c = compute_c5(h, n)
    h_avg = h.copy()
    dx = 2.0 / n
    nc = 2 * n - 1
    step_decay = 1.0
    initial_step = 0.05
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        cur_c = c_vec.max()
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - tol_active)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        polyak_gap = cur_c - best_c
        if polyak_gap < 1e-12:
            step = initial_step / np.sqrt(t) * step_decay
        else:
            step = min(initial_step, polyak_gap / (gnorm * gnorm + 1e-12))
        step = max(step, 1e-10)
        h_step = best_h - step * grad
        h_step = project_feasible(h_step, n)
        h_avg = ((t - 1) / t) * h_avg + (1.0 / t) * h_step
        c_avg = compute_c5(h_avg, n)
        if c_avg < best_c - 1e-16:
            best_c = c_avg
            best_h = h_avg.copy()
        c_step = compute_c5(h_step, n)
        if c_step < best_c - 1e-16:
            best_c = c_step
            best_h = h_step.copy()
        best_h = h_step
    return best_h, best_c

        if not ones_mask.any() or not zeros_mask.any():
            break
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        ones_grad = grad[ones_idx]
        zeros_grad = grad[zeros_idx]
        pos_ones = ones_idx[ones_grad > 0]
        neg_zeros = zeros_idx[zeros_grad < 0]
        if len(pos_ones) == 0 or len(neg_zeros) == 0:
            break
        pos_ones = pos_ones[np.argsort(-ones_grad[pos_ones])][:top_candidates]
        neg_zeros = neg_zeros[np.argsort(zeros_grad[neg_zeros])][:top_candidates]
        best_gain = -1e10
        best_i = -1
        best_j = -1
        for i in pos_ones:
            gi = grad[i]
            for j in neg_zeros:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        amt = min(0.01, best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-8:
            amt = min(best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-8:
                break
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
        rounds += 1
    return best_h, best_c

def fine_local_search(h, n, deadline, deltas=None, radius=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, 2):
                    if abs(i - j) < 3:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty_start(n, alpha_param, rng):
    h = np.zeros(n)
    n_ones = n // 2
    offsets = rng.uniform(0, 1, size=3)
    best_h = None
    best_c = np.inf
    for off in offsets:
        hb = np.zeros(n)
        positions = set()
        for k in range(1, 4 * n):
            pos = int((k * alpha_param + off) * n) % n
            positions.add(pos)
            if len(positions) >= n_ones:
                break
        for pos in positions:
            hb[pos] = 1.0
        remaining = n / 2.0 - hb.sum()
        if abs(remaining) > 1e-10:
            zeros = np.where(hb == 0)[0]
            if len(zeros) > 0:
                hb[zeros[0]] += remaining
        hb = project_feasible(hb, n)
        c = compute_c5(hb, n)
        if c < best_c:
            best_c = c
            best_h = hb
    return best_h if best_h is not None else h

def perturb_and_optimize(h, n, deadline, rng, num_perturbs=5, alpha=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    perturbation_scale = [0.02, 0.01, 0.005, 0.002, 0.001]
    for p in range(num_perturbs):
        if time.time() > deadline - 20:
            break
        scale = perturbation_scale[min(p, len(perturbation_scale) - 1)]
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        sub_deadline = min(time.time() + 40, deadline - 10)
        h_opt, c_opt = run_lbfgs(h_pert, n, sub_deadline,
                                 full_smooth_obj, full_smooth_grad,
                                 (n, alpha, 0.001), maxiter=500)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k_list=None, alphas=None, rounds_per=4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if top_k_list is None:
        top_k_list = [3, 5, 8, 12, 20]
    if alphas is None:
        alphas = [100000, 500000, 2000000]
    bounds = [(0.0, 1.0)] * n
    for top_k in top_k_list:
        if time.time() > deadline - 20:
            break
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            for rnd in range(rounds_per):
                if time.time() > deadline - 10:
                    break
                c_vec = compute_c5_vec(best_h, n)
                max_val = c_vec.max()
                threshold = max_val - max(1e-9, 0.0005 * max_val)
                active = np.where(c_vec >= threshold)[0]
                if len(active) > top_k:
                    order = np.argsort(-c_vec[active])
                    active = active[order[:top_k]]
                if len(active) == 0:
                    break
                maxit = max(200, int((deadline - 10 - time.time()) * 1.5))
                try:
                    obj_val = smooth_max_val(c_vec[active], alpha)
                    best_h_new = best_h
                    for inner in range(3):
                        if time.time() > deadline - 10:
                            break
                        c_cur = compute_c5_vec(best_h_new, n)
                        vals = c_cur[active]
                        mx = vals.max()
                        sh = alpha * (vals - mx)
                        smx = sh.max()
                        w_act = np.exp(sh - smx)
                        w_act /= w_act.sum()
                        dx = 2.0 / n
                        grad = np.zeros(n)
                        for wi, idx in zip(w_act, active):
                            if wi < 1e-14:
                                continue
                            k = idx - (n - 1)
                            wm = wi * dx
                            if k >= 0:
                                nv = n - k
                                grad[0:nv] += wm * (1.0 - best_h_new[k:n])
                                grad[k:n] += wm * (-best_h_new[0:nv])
                            else:
                                kk = -k
                                nv = n - kk
                                grad[kk:n] += wm * (1.0 - best_h_new[0:nv])
                                grad[0:nv] += wm * (-best_h_new[kk:n])
                        step = 0.01 / (1 + 0.5 * inner)
                        h_try = best_h_new - step * grad
                        h_try = project_feasible(h_try, n)
                        c_try = compute_c5(h_try, n)
                        if c_try < compute_c5(best_h_new, n) - 1e-16:
                            best_h_new = h_try
                    h_new = best_h_new
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    alt_starts = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(make_beatty_start(n_orig, golden / 2.0, rng))
    alt_starts.append(make_beatty_start(n_orig, 1.0 / np.sqrt(2.0), rng))
    for _ in range(2):
        hb = np.zeros(n_orig)
        idx = rng.choice(n_orig, size=n_orig // 2, replace=False)
        hb[idx] = 1.0
        alt_starts.append(project_feasible(hb, n_orig))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup with progressive alpha
    p1_deadline = t0 + 80
    current_h = best_h
    current_c = best_c
    for alpha in [50, 200, 1000, 5000, 50000, 200000]:
        if time.time() > p1_deadline - 10:
            break
        h_new, c_new = run_lbfgs(current_h, n_orig, p1_deadline,
                                  full_smooth_obj, full_smooth_grad,
                                  (n_orig, alpha, 0.005), maxiter=300)
        if c_new < current_c - 1e-15:
            current_c = c_new
            current_h = h_new
    if current_c < best_c:
        best_c = current_c
        best_h = current_h

    # Phase 2: Active set optimization
    if time.time() < deadline - 500:
        p2_deadline = min(time.time() + 150, deadline - 400)
        h2, c2 = active_set_optimize(best_h, n_orig, p2_deadline,
                                      top_k_list=[3, 5, 8],
                                      alphas=[200000, 1000000],
                                      rounds_per=3)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Iterative reweighting
    if time.time() < deadline - 400:
        p3_deadline = min(time.time() + 200, deadline - 300)
        h3, c3 = iterative_reweighting(best_h, n_orig, p3_deadline,
                                        num_iters=40, eta_base=0.15)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Subgradient with averaging
    if time.time() < deadline - 300:
        p4_deadline = min(time.time() + 150, deadline - 250)
        h4, c4 = subgradient_averaging(best_h, n_orig, p4_deadline,
                                        max_iter=30000, tol_active=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 200:
        p5_deadline = min(time.time() + 150, deadline - 150)
        h5, c5 = grad_pair_search(best_h, n_orig, p5_deadline,
                                   max_rounds=15000, top_candidates=25)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Perturbation + re-optimization
    if time.time() < deadline - 150:
        p6_deadline = min(time.time() + 100, deadline - 100)
        h6, c6 = perturb_and_optimize(best_h, n_orig, p6_deadline, rng,
                                       num_perturbs=4, alpha=500000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Another round of active set + reweighting
    if time.time() < deadline - 120:
        p7_deadline = min(time.time() + 60, deadline - 80)
        h7, c7 = active_set_optimize(best_h, n_orig, p7_deadline,
                                      top_k_list=[3, 5],
                                      alphas=[500000, 2000000],
                                      rounds_per=2)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine local search
    if time.time() < deadline - 80:
        p8_deadline = min(time.time() + 60, deadline - 40)
        h8, c8 = fine_local_search(best_h, n_orig, p8_deadline,
                                    deltas=[0.002, 0.001, 0.0005, 0.0002],
                                    radius=50)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: More gradient-guided search
    if time.time() < deadline - 50:
        p9_deadline = min(time.time() + 35, deadline - 20)
        h9, c9 = grad_pair_search(best_h, n_orig, p9_deadline,
                                   max_rounds=8000, top_candidates=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Final L-BFGS polish with very high alpha
    if time.time() < deadline - 15:
        for alpha in [5000000, 20000000]:
            if time.time() > deadline - 10:
                break
            h10, c10 = run_lbfgs(best_h, n_orig, deadline - 5,
                                  full_smooth_obj, full_smooth_grad,
                                  (n_orig, alpha, 0.0), maxiter=500)
            if c10 < best_c:
                best_c = c10
                best_h = h10

    # Final very fine adjustments
    if time.time() < deadline - 3:
        hf, cf = fine_local_search(best_h, n_orig, deadline - 2,
                                    deltas=[0.0001, 0.00005, 0.00002],
                                    radius=30)
        if cf < best_c:
            best_c = cf
            best_h = hf

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
