import cvxpy as cp

def solve_lp_cut(h0, n, C_target, max_iters=50):
    """Solve the linearized QCQP for a given target C."""
    dx = 2.0 / n
    h = cp.Variable(n)
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0
    ]
    
    for it in range(max_iters):
        # Add linearized constraints at h0
        c_vec = compute_c5_vec(h0, n)
        # Only add constraints for "active" lags (near the max)
        active = np.where(c_vec >= c_vec.max() - 0.01 * c_vec.max())[0]
        
        for idx in active:
            k = idx - (n - 1)
            # Linearize h[j]*(1-h[j+k]) at h0
            if k >= 0:
                nv = n - k
                # Σ_{j=0}^{nv-1} h0[j]*(1-h[j+k]) + h[j]*(1-h0[j+k]) - h0[j]*(1-h0[j+k])
                # = Σ h0[j] - h0[j]*h[j+k] + h[j] - h[j]*h0[j+k] - h0[j] + h0[j]*h0[j+k]
                # = Σ(-h0[j]*h[j+k] + h[j] - h[j]*h0[j+k] + h0[j]*h0[j+k])
                # = Σ(h[j]*(1-h0[j+k]) + h0[j]*(h0[j+k]-h[j+k]))
                lin = (h[0:nv] * (1 - h0[k:n]) + h0[0:nv] * (h0[k:n] - h[k:n]))
            else:
                kk = -k
                nv = n - kk
                lin = (h[kk:n] * (1 - h0[0:nv]) + h0[kk:n] * (h0[0:nv] - h[0:nv]))
            
            constraints.append(cp.sum(lin) * dx <= C_target)
        
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.SCS)  # or use another solver
        except Exception:
            return h0, False
        
        if h.value is None:
            return h0, False
        
        h_new = np.array(h.value).flatten()
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new <= C_target + 1e-6:
            return h_new, True
        
        h0 = h_new
    
    return h0, compute_c5(h0, n) <= C_target + 1e-6

For lag k ≥ 0: overlap(k) = dx * [Σ_{j=0}^{n-1-k} h[j] - Σ_{j=0}^{n-1-k} h[j]*h[j+k]]
For lag k < 0: overlap(k) = dx * [Σ_{j=-k}^{n-1} h[j] - Σ_{j=-k}^{n-1} h[j]*h[j+k]]

The first term depends only on the "prefix sums" of h. The second term is the autocorrelation.

So overlap(k) = dx * (prefix_sum or suffix_sum) - dx * R(k)

where R(k) is the linear autocorrelation.

To minimize max_k overlap(k):
- We want the autocorrelation R(k) to be as large as possible (to minimize the overlap)
- But we also want the prefix/suffix sums to be as small as possible

The prefix sum Σ_{j=0}^{n-1-k} h[j] is maximized when h is concentrated at the beginning, and minimized when it's spread out or concentrated at the end.

This suggests that the optimal function might have a specific "shape" - perhaps it's concentrated in a way that balances the prefix sums against the autocorrelation.

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    if max_shifted > 500:
        val = m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        if w[lag_idx] < 1e-14:
            continue
        k = lag_idx - (n - 1)
        wm = w[lag_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])

def binary_simulated_anneal(h_init, n, deadline, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    
    # Create initial binary
    h = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h_init)
    h[sorted_idx[:n_ones]] = 1.0
    
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c
    
    T = 0.05
    n_iters = 200000
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Pick two positions to swap
        i = rng.randint(n)
        if h[i] == 1.0:
            zeros = np.where(h == 0.0)[0]
            if len(zeros) == 0:
                continue
            j = zeros[rng.randint(len(zeros))]
        else:
            ones = np.where(h == 1.0)[0]
            if len(ones) == 0:
                continue
            j = ones[rng.randint(len(ones))]
        
        if i == j:
            continue
        
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        
        if c_new < c:
            c = c_new
        else:
            T_temp = T * (1 - it / n_iters)
            if T_temp < 1e-10:
                T_temp = 1e-10
            accept = rng.random() < np.exp(-(c_new - c) / T_temp)
            if not accept:
                h[i], h[j] = h[j], h[i]
            else:
                c = c_new
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

        for idx in active:

            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - y_new[k:n]) * dx
                grad[k:n] += (-y_new[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - y_new[0:nv]) * dx
                grad[0:nv] += (-y_new[kk:n]) * dx

            best_c = c_new
