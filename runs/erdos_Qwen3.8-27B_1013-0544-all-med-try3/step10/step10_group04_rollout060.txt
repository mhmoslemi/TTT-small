```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = _compute_grad_from_weights(w, x, n)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _compute_grad_from_weights(w, x, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 500.0), (2000, 3000.0),
                   (5000, 10000.0), (20000, 50000.0), (100000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 300000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c
```

```python
    return h.copy(), compute_c5(h, n)

def flip_delta_c(h, n, pos, delta):
    """Compute change in C vector when h[pos] changes by delta."""
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    center = n - 1
    # k >= 0: corr[center+k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
    # First index (i=pos): valid if pos <= n-1-k, i.e., k <= n-1-pos
    for k in range(0, n - pos):
        d[center + k] += delta * (1.0 - h[pos + k]) * dx
    # Second index (i+k=pos, i=pos-k): valid if pos-k >= 0, i.e., k <= pos
    for k in range(1, pos + 1):
        d[center + k] += -delta * h[pos - k] * dx
    # k < 0, kp = -k: corr[center-kp] = sum_{i=kp}^{n-1} h[i](1-h[i-kp])
    # First index (i=pos): valid if pos >= kp, i.e., kp <= pos
    for kp in range(1, pos + 1):
        d[center - kp] += delta * (1.0 - h[pos - kp]) * dx
    # Second index (i-kp=pos, i=pos+kp): valid if pos+kp <= n-1, i.e., kp <= n-1-pos
    for kp in range(1, n - pos):
        d[center - kp] += -delta * h[pos + kp] * dx
    return d

def binary_local_search(h, n, deadline, max_rounds=200):
    """Local search on nearly-binary structure."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_c_vec = compute_c5_vec(h, n)
    current_max = best_c
    rng = np.random.RandomState(777)
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    if len(ones) == 0 or len(zeros) == 0:
        return h, best_c
    
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        accepted = 0
        n_trials = min(2000, len(ones) * len(zeros) // max(1, len(ones)+len(zeros)))
        n_trials = max(n_trials, 200)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            # Compute delta for flipping a: 1->0, b: 0->1
            d_a = flip_delta_c(h, n, a, -1.0)
            d_b = flip_delta_c(h, n, b, 1.0)
            new_c_vec = best_c_vec + d_a + d_b
            new_max = float(np.max(new_c_vec))
            if new_max < current_max - 1e-15:
                # Accept
                h[a] = 0.0
                h[b] = 1.0
                best_c_vec = new_c_vec
                current_max = new_max
                ones.remove(a)
                zeros.remove(b)
                ones.append(b)
                zeros.append(a)
                accepted += 1
                if current_max < best_c:
                    best_c = current_max
                    # Re-sort occasionally
                    if accepted % 100 == 0:
                        ones.sort()
                        zeros.sort()
        if accepted == 0:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    return h, current_max

def systematic_local_search(h, n, deadline):
    """Try all swaps involving positions near the worst shift."""
    h = project_feasible(h.copy(), n)
    current_max = compute_c5(h, n)
    best_c_vec = compute_c5_vec(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    if len(ones) == 0 or len(zeros) == 0:
        return h, current_max
    
    # Focus on the worst shift
    worst_idx = int(np.argmax(best_c_vec))
    k_worst = worst_idx - (n - 1)
    
    # Only consider swaps where a and b are both "active" at the worst shift
    if k_worst >= 0:
        active_ones = [i for i in ones if i < n - k_worst or i >= k_worst]
        active_zeros = [i for i in zeros if i < n - k_worst or i >= k_worst]
    else:
        kk = -k_worst
        active_ones = [i for i in ones if i >= kk or i < n - kk]
        active_zeros = [i for i in zeros if i >= kk or i < n - kk]
    
    if not active_ones or not active_zeros:
        active_ones = ones
        active_zeros = zeros
    
    # Limit to top candidates
    active_ones = active_ones[:200]
    active_zeros = active_zeros[:200]
    
    for a in active_ones:
        if time.time() > deadline:
            break
        d_a = flip_delta_c(h, n, a, -1.0)
        for b in active_zeros:
            if a == b:
                continue
            d_b = flip_delta_c(h, n, b, 1.0)
            new_max = float(np.max(best_c_vec + d_a + d_b))
            if new_max < current_max - 1e-15:
                h[a] = 0.0
                h[b] = 1.0
                best_c_vec += d_a + d_b
                current_max = new_max
                ones.remove(a)
                zeros.remove(b)
                ones.append(b)
                zeros.append(a)
                break
    return h, current_max

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_optimize(h, n, deadline, n_trials=5, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    if mags is None:
        mags = [0.02, 0.04, 0.01, 0.06, 0.03]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            maxiter = min(10000, max(1000, 4000000 // max(n, 1)))
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 10000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxiter * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = make_thue_morse(500)
        n_base = 500
    
    h = project_feasible(h_init, n_base)
    best_c = compute_c5(h, n_base)
    best_h_overall = h.copy()
    best_n_overall = n_base
    
    # Phase 1: L-BFGS at base resolution
    t1 = min(t0 + 80, deadline - 500)
    if t1 > time.time() + 5:
        h, c = lbfgs_phase(h, n_base, t1, n_configs=4)
        if c < best_c:
            best_c = c
            best_h_overall = h.copy()
    
    # Phase 2: SLSQP polish at base
    t2 = min(time.time() + 40, deadline - 460)
    if t2 > time.time() + 5:
        h, c = slqp_polish(h, n_base, t2, alpha=300000)
        if c < best_c:
            best_c = c
            best_h_overall = h.copy()
    
    # Phase 3: Multi-start at base resolution
    t3 = min(time.time() + 80, deadline - 400)
    if t3 > time.time() + 10:
        starts = [
            make_thue_morse(n_base),
            make_random_binary(n_base, seed + 1),
            make_random_binary(n_base, seed + 2),
            make_sine_pattern(n_base, 5.0),
        ]
        for s in starts:
            if time.time() > t3 - 10:
                break
            s = project_feasible(s, n_base)
            c_s = compute_c5(s, n_base)
            if c_s < best_c + 0.05:  # Only optimize promising starts
                maxiter = min(5000, max(1000, 2000000 // n_base))
                bounds = [(0.0, 1.0)] * n_base
                try:
                    res = minimize(smooth_obj, s, jac=smooth_grad,
                                  args=(n_base, 2000, 3000.0), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxiter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxfun': maxiter * 20,
                                           'maxls': 50})
                    h_s = project_feasible(res.x, n_base)
                    c_s = compute_c5(h_s, n_base)
                    if c_s < best_c:
                        best_c = c_s
                        best_h_overall = h_s.copy()
                except Exception:
                    pass
    
    # Phase 4: Perturbation at base
    t4 = min(time.time() + 60, deadline - 350)
    if t4 > time.time() + 10:
        h, c = perturbation_optimize(best_h_overall, n_base, t4,
                                     n_trials=4, mags=[0.02, 0.04, 0.01, 0.06])
        if c < best_c:
            best_c = c
            best_h_overall = h.copy()
    
    # Phase 5: Upsample to 2x and optimize
    n_up1 = min(2 * n_base, 1200)
    t5 = min(time.time() + 120, deadline - 300)
    if t5 > time.time() + 10:
        h_up = interpolate_to(best_h_overall, n_base, n_up1)
        h_up, c_up = lbfgs_phase(h_up, n_up1, t5, n_configs=3)
        if c_up < best_c:
            best_c = c_up
            best_h_overall = h_up.copy()
            best_n_overall = n_up1
    
    # Phase 6: Upsample to higher and optimize
    n_up2 = min(2 * n_up1, 2000)
    if best_n_overall < n_up2:
        t6_start = time.time()
        t6 = min(t6_start + 150, deadline - 200)
        if t6 > t6_start + 15:
            h_up2 = interpolate_to(best_h_overall, best_n_overall, n_up2)
            h_up2, c_up2 = lbfgs_phase(h_up2, n_up2, t6, n_configs=3)
            if c_up2 < best_c:
                best_c = c_up2
                best_h_overall = h_up2.copy()
                best_n_overall = n_up2
    
    # Phase 7: SLSQP polish at best resolution
    t7 = min(time.time() + 60, deadline - 120)
    if t7 > time.time() + 5:
        alpha_slq = 2000000 if best_n_overall <= 1000 else 500000
        h2, c2 = slqp_polish(best_h_overall, best_n_overall, t7, alpha=alpha_slq)
        if c2 < best_c:
            best_c = c2
            best_h_overall = h2.copy()
    
    # Phase 8: Binary local search
    t8 = min(time.time() + 80, deadline - 60)
    if t8 > time.time() + 5:
        h_bin, c_bin = binary_local_search(best_h_overall, best_n_overall, t8,
                                           max_rounds=100)
        if c_bin < best_c:
            best_c = c_bin
            best_h_overall = h_bin.copy()
    
    # Phase 9: Systematic local search
    t9 = min(time.time() + 40, deadline - 30)
    if t9 > time.time() + 3:
        h_sys, c_sys = systematic_local_search(best_h_overall, best_n_overall, t9)
        if c_sys < best_c:
            best_c = c_sys
            best_h_overall = h_sys.copy()
    
    # Phase 10: Final gradient refinement on nearly-binary solution
    t10 = min(time.time() + 30, deadline - 10)
    if t10 > time.time() + 3:
        n_final = best_n_overall
        bounds = [(0.0, 1.0)] * n_final
        alpha_final = 200000 if n_final <= 1000 else 50000
        try:
            res = minimize(smooth_obj, best_h_overall, jac=smooth_grad,
                          args=(n_final, alpha_final, 100000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 30000, 'maxls': 50})
            h_final = project_feasible(res.x, n_final)
            c_final = compute_c5(h_final, n_final)
            if c_final < best_c:
                best_c = c_final
                best_h_overall = h_final.copy()
        except Exception:
            pass
    
    # Final feasibility check
    n_final = best_n_overall
    best_h_overall = project_feasible(best_h_overall, n_final)
    best_c = compute_c5(best_h_overall, n_final)
    
    return best_h_overall, best_c, n_final
```
