    for restart in range(n_restarts):
        if time.time() > deadline:

        if restart > 0:
            for _ in range(rng.randint(5, 15)):
                oi = np.where(h == 1)[0]

                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)

        cur_max = corr.max()
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            
            if len(zeros_idx) == 0 or len(ones_idx) == 0:
                break
            
            # Compute impact for all zeros
            # impact_i[m] = (1 - h[i+m]) - h[i-m] for valid indices
            # Shape: (n_zeros, 2n-1)
            i_offsets = zeros_idx[:, None] + m_range[None, :]  # (n_zeros, 2n-1)
            i_offsets_neg = zeros_idx[:, None] - m_range[None, :]
            
            impact_i = np.zeros((len(zeros_idx), 2*n-1))
            msk_pos = (i_offsets >= 0) & (i_offsets < n)
            msk_neg = (i_offsets_neg >= 0) & (i_offsets_neg < n)
            impact_i[msk_pos] += 1.0 - h[i_offsets[msk_pos]]
            impact_i[msk_neg] -= h[i_offsets_neg[msk_neg]]
            
            # Compute impact for all ones
            j_offsets = ones_idx[:, None] + m_range[None, :]
            j_offsets_neg = ones_idx[:, None] - m_range[None, :]
            
            impact_j = np.zeros((len(ones_idx), 2*n-1))
            msk_pos = (j_offsets >= 0) & (j_offsets < n)
            msk_neg = (j_offsets_neg >= 0) & (j_offsets_neg < n)
            impact_j[msk_pos] -= 1.0 - h[j_offsets[msk_pos]]
            impact_j[msk_neg] += h[j_offsets_neg[msk_neg]]
            
            # Compute new max for all (i,j) pairs
            # new_corr[i_idx, j_idx, m] = corr[m] + impact_i[i_idx, m] + impact_j[j_idx, m]
            new_max = np.max(
                corr[None, None, :] + impact_i[:, None, :] + impact_j[None, :, :],
                axis=2
            )
            
            # Find the best swap (minimum new_max)
            flat_idx = np.argmin(new_max)
            i_idx = flat_idx // len(ones_idx)
            j_idx = flat_idx % len(ones_idx)
            best_new_max = new_max[i_idx, j_idx]
            
            if best_new_max < cur_max:
                i = zeros_idx[i_idx]
                j = ones_idx[j_idx]
                h[i] = 1.0
                h[j] = 0.0
                corr += impact_i[i_idx] + impact_j[j_idx]
                cur_max = best_new_max
            else:
                break  # No improvement possible
        
        c = cur_max * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

                    jm = j + m_range

                    delta = np.zeros(2*n-1)
                    msk = (im >= 0) & (im < n)

                    msk2 = (imn >= 0) & (imn < n)
                    delta[msk2] -= h[imn[msk2]]
                    msk3 = (jm >= 0) & (jm < n)

            impact_i[msk2] -= h[i_off_neg[msk2]]
            
            # Compute impact for ones
            j_off = ones_idx[:, None] + m_range[None, :]
            j_off_neg = ones_idx[:, None] - m_range[None, :]
            
            impact_j = np.zeros((no, 2*n-1), dtype=np.float64)
            msk = (j_off >= 0) & (j_off < n)
            impact_j[msk] -= 1.0 - h[j_off[msk]]
            msk2 = (j_off_neg >= 0) & (j_off_neg < n)

                chunk_max = tensor.max(axis=2)  # (chunk, no)
                flat_idx = chunk_max.argmin()

                cj = flat_idx % no
                cv = chunk_max[ci, cj]

                cur_max = best_val
            else:
                break
        
        c = cur_max * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
