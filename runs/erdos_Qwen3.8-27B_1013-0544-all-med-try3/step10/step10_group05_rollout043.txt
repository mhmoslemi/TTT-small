            if tot < 1e-15:

            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - n_target):
            h_bin[ones_idx[order[i]]] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(n_target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
    return h_bin

def compute_neighborhood_scores(h, active_lags, n):
    """Compute S(p) = sum over active lags of (h[p+lag] + h[p-lag]) with boundary=0."""
    S = np.zeros(n)
    for lag in active_lags:
        lag = int(lag)
        if lag == 0:
            S += 2.0 * h
        elif lag > 0:
            S[:n-lag] += h[lag:]
            S[lag:] += h[:n-lag]
        else:
            ll = -lag
            S[ll:] += h[:n-ll]
            S[:n-ll] += h[ll:]
    return S

def binary_sa(h_bin, n, deadline, n_iter_max=200000):
    """Simulated annealing on binary solutions."""
    h = h_bin.copy().astype(float)
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    best_h = h.copy()
    cur_c = best_c
    
    T_start = 0.01
    T_end = 0.0001
    n_iter = min(n_iter_max, int((deadline - time.time()) * 200))
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    import random
    rng_ra = random.Random(12345)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / n_iter)
        tol = max(1e-10, 0.002 * cur_c)
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        active_mask = c_vec >= max_val - tol
        active_lags = np.where(active_mask)[0] - (n - 1)
        
        # Compute neighborhood scores
        S = compute_neighborhood_scores(h, active_lags, n)
        
        # Choose move: with some probability, pick randomly; otherwise, pick greedily
        if rng_ra.random() < 0.3:
            # Random move
            a = rng_ra.choice(ones_list)
            b = rng_ra.choice(zeros_list)
        else:
            # Greedy: minimize S[a] - S[b]
            S_ones = np.array([S[i] for i in ones_list])
            S_zeros = np.array([S[i] for i in zeros_list])
            a = ones_list[int(np.argmin(S_ones))]
            b = zeros_list[int(np.argmax(S_zeros))]
        
        # Evaluate the move
        h[a] = 0
        h[b] = 1
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        
        # Accept/reject
        if delta < 0 or rng_ra.random() < np.exp(-delta / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            ones_list.remove(a)
            zeros_list.append(a)
            zeros_list.remove(b)
            ones_list.append(b)
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c
