from scipy.optimize import minimize, linprog
import time, math


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if n == 0:
        return h
    target = 0.5 * n
    for _ in range(200000):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (d / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (d / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def smooth_max_val(c, alpha):
    c = np.asarray(c, dtype=float)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400.0:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)


def smooth_max_grad(c, alpha, x, n):
    m = float(c.max())
    sh = alpha * (c - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idc = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idc
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad


def act_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (h.sum() - 0.5 * n) ** 2


def act_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = float(vals.max())
    sh = alpha * (vals - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idc = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idc
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    if mu > 0.0:
        grad += 2.0 * mu * (h.sum() - 0.5 * n)
    return grad


def full_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - 0.5 * n) ** 2


def full_grad(h, n, alpha, mu):

    grad = smooth_max_grad(c, alpha, h, n)
    if mu > 0.0:
        grad += 2.0 * mu * (h.sum() - 0.5 * n)
    return grad

def inc_step(h, n, keep):
    c = compute_c5_vec(h, n)
    nmax = c.size
    k = max(2, min(int(keep), nmax - 1))
    act = np.argsort(-c)[:k]
    nact = int(act.size)
    A = np.zeros((nact, n + 1))
    b = np.zeros(nact)
    hh = np.clip(h, 0.0, 1.0)
    for r in range(nact):
        mm = int(act[r])
        g = single_grad(hh, n, mm)
        f = float(c[mm])
        rv = np.concatenate([g, np.array([-1.0])])
        A[r, :] = rv
        b[r] = f - float(np.dot(g, hh))
    Aeq = np.zeros((1, n + 1))

import numpy as np
from scipy.optimize import minimize, linprog
import time, math


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(np.correlate(h, 1 - h, mode='full').max() * (2.0 / n))


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    T = 0.5 * float(n)
    for _ in range(200000):
        d = T - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            h = np.clip(h + (d / tot) * room, 0, 1) if tot > 1e-16 else h
        else:
            tot = h.sum()
            h = np.clip(h + (d / tot) * h, 0, 1) if tot > 1e-16 else h
    return h


def smooth_max_logsumexp(vals, alpha):
    m = float(vals.max())
    sh = alpha * (vals - m)
    smx = float(sh.max())
    if smx > 400.0:
        e = np.exp(sh - smx)
        s = max(float(e.sum()), 1e-300)
        return m + (smx + math.log(s)) / alpha
    e = np.exp(sh)
    s = max(float(e.sum()), 1e-300)
    return m + math.log(s) / alpha


def smooth_weights(vals, alpha):
    m = float(vals.max())
    sh = alpha * (vals - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w = w / w.sum()
    return w


def smooth_max_grad(c, alpha, x, n):
    w = smooth_weights(c, alpha)
    out = np.zeros(n)
    dx = 2.0 / n
    cc = n - 1
    for mm in range(len(w)):
        wm = float(w[mm])
        if wm < 1e-15:
            continue
        k = mm - cc
        wm = wm * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                out[0:nv] += wm * (1.0 - x[k:n])
                out[k:n] += -wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                out[kk:n] += wm * (1.0 - x[0:nv])
                out[0:nv] += -wm * x[kk:n]
    return out


def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_logsumexp(c[active], alpha) + mu * (x.sum() - 0.5 * n) ** 2


def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_weights(c[active], alpha)
    out = np.zeros(n)
    dx = 2.0 / n
    cc = n - 1
    for a, gi in zip(active, g):
        if float(gi) < 1e-15:
            continue
        k = int(a) - cc
        wm = float(gi) * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                out[0:nv] += wm * (1.0 - x[k:n])
                out[k:n] += -wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                out[kk:n] += wm * (1.0 - x[0:nv])
                out[0:nv] += -wm * x[kk:n]
    if mu > 0.0:
        out += 2.0 * mu * (x.sum() - 0.5 * n)
    return out


def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_logsumexp(c, alpha) + mu * (x.sum() - 0.5 * n) ** 2


def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    out = smooth_max_grad(c, alpha, x, n)
    if mu > 0.0:
        out += 2.0 * mu * (x.sum() - 0.5 * n)
    return out


def single_grad(h, n, mm):
    c = compute_c5_vec(h, n)
    g = smooth_weights(c[np.array([mm])], 1.0)[0] if False else 1.0
    out = np.zeros(n)
    dx = 2.0 / n
    cc = n - 1
    k = int(mm) - cc
    wm = g * dx
    if k >= 0:
        nv = n - k
        if nv > 0:
            out[0:nv] += wm * (1.0 - h[k:n])
            out[k:n] += -wm * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            out[kk:n] += wm * (1.0 - h[0:nv])
            out[0:nv] += -wm * h[kk:n]
    return out


def make_beatty(n, ap):
    h = np.zeros(n)
    tg = n // 2
    pos = set()
    for k in range(1, 30 * n + 5):
        p = int(math.floor(k * ap + 0.5)) % n
        pos.add(p)
        if len(pos) >= tg:
            break
    for p in pos:
        h[p] = 1.0
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    i = rng.choice(n, size=n // 2, replace=False)
    h[i] = 1.0
    return h


def upsample(h, no, nn):
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nn)


def decimate_to(nn, hold, no):
    i = (np.arange(nn) * (no / float(nn))).round().astype(np.int64) % no
    return project_feasible(np.clip(hold[i], 0, 1), nn)


def run_active_round(h, n, dl, topk, alpha, mu):
    c = compute_c5_vec(h, n)
    mx = float(c.max())
    thr = mx - max(1e-8, 3e-4 * mx)
    act = np.where(c >= thr)[0]
    if act.size == 0:
        act = np.argsort(-c)[:max(2, topk)]
    act = act[np.argsort(-c[act])][:max(2, topk)]
    rem = max(0.0, dl - time.time())
    if rem < 3:
        return h, compute_c5(h, n)
    if n > 1500:
        mxi = max(40, min(90, int(rem * 0.45)))
    elif n > 750:
        mxi = max(60, min(260, int(rem * 1.0)))
    else:
        mxi = max(80, min(430, int(rem * 1.4)))
    bds = [(0.0, 1.0) for _ in range(n)]
    try:
        r = minimize(act_obj, h, jac=act_grad, args=(n, active=act, alpha=alpha, mu=mu),
                     method='L-BFGS-B', bounds=bds, options={'maxiter': mxi, 'ftol': 1e-23,
                                                              'gtol': 1e-17, 'maxfun': mxi * 9,
                                                              'maxls': 25, 'maxcor': 30})
        h2 = project_feasible(r.x, n)
        return h2, compute_c5(h2, n)
    except Exception:
        h2 = project_feasible(h + 1e-5 * np.random.default_rng(0).standard_normal(n), n)
        return h2, compute_c5(h2, n)


def run_full_round(h, n, dl, alpha):
    rem = max(0.0, dl - time.time())
    if rem < 3:
        return h, compute_c5(h, n)
    mxi = max(30, min(160, int(rem * 0.7)))
    bds = [(0.0, 1.0) for _ in range(n)]
    try:
        r = minimize(full_obj, h, jac=full_grad, args=(n, alpha, 0.0), method='L-BFGS-B',
                     bounds=bds, options={'maxiter': mxi, 'ftol': 1e-23, 'gtol': 1e-17,
                                          'maxfun': mxi * 7, 'maxls': 25, 'maxcor': 30})
        return project_feasible(r.x, n), compute_c5(project_feasible(r.x, n), n)
    except Exception:
        return h, compute_c5(h, n)


def l_p_step(h, n, keep):
    c = compute_c5_vec(h, n)
    kp = max(2, min(int(keep), 2 * n - 2))
    idx = np.argsort(-c)[:kp]
    A = np.empty((len(idx), n + 1))
    b = np.empty(len(idx))
    hclip = np.clip(h, 0, 1)
    for r, mm in enumerate(idx):
        g = single_grad(hclip, n, int(mm))
        f = float(c[int(mm)])
        A[r, 0:n] = g
        A[r, n] = -1.0
        b[r] = f - float(g.dot(hclip))
    Aeq = np.zeros((1, n + 1)); Aeq[0, 0:n] = 1.0
    beq = np.array([0.5 * n])
    obj = np.zeros(n + 1); obj[n] = 1.0
    bnd = [(0.0, 1.0) for _ in range(n)] + [(0.0, None)]
    try:
        s = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bnd, method='highs')
    except Exception:
        return hclip, compute_c5(hclip, n), 0
    if not s.success:
        return hclip, compute_c5(hclip, n), 0
    h2 = project_feasible(s.x[:n], n)
    return h2, compute_c5(h2, n), 1


def linmin(h, n, dl, keep):
    h2 = project_feasible(h.copy(), n)
    cb = compute_c5(h2, n)
    keep2 = int(keep)
    imp = 0
    for ki in range(15):
        t = time.time()
        if t > dl - 2 or imp >= 220:
            break
        km = keep2 + (0 if ki < 8 else (ki // 2) + 2)
        if ki > 9:

                break
            h3, cl3, stt = l_p_step(h2, n, km)

            if cl3 < cb - 5e-15:
                cb = cl3
                h2 = project_feasible(h3, n)
                imp += 1
                if imp >= 220:
                    break
    return h2, cb


def polish(h, n, dl, do_full):
    h0 = project_feasible(h.copy(), n)
    cb = compute_c5(h0, n)
    alphas = [3000.0, 8000.0, 18000.0, 42000.0, 100000.0, 240000.0, 550000.0, 1.4e6, 3.2e6, 7.2e6]
    tops = [3, 4, 5, 6, 4, 5, 6, 7, 5, 6]
    keeps = [5, 6, 7, 8, 7, 8, 9, 10]
    i = 0
    lt = 1e300
    while True:

        i += 1
        if (i % 4) == 0:

def fine_pair_search(h, n, dl, radius, deltas):
    bh = project_feasible(h.copy(), n)
    cb = compute_c5(bh, n)
    for d in deltas:
        imp = True
        wp = 0
        while imp and (time.time() < dl - 3) and wp < 34:
            wp += 1
            imp = False
            for i in range(n):
                if time.time() > dl - 3:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if (bh[i] - d >= 0.0) and (bh[j] + d <= 1.0):
                        bh[i] -= d; bh[j] += d
                        c = compute_c5(bh, n)
                        if c < cb - 1e-15:
                            cb = c; imp = True; break
                        bh[i] += d; bh[j] -= d
                    if (bh[i] + d <= 1.0) and (bh[j] - d >= 0.0):
                        bh[i] += d; bh[j] -= d
                        c = compute_c5(bh, n)
                        if c < cb - 1e-15:
                            cb = c; imp = True; break
                        bh[i] -= d; bh[j] += d
                if imp:
                    break
    return bh, compute_c5(bh, n)

def final_pair(nuance, hh, nn, dl):
    for fd in [0.0005, 0.0002, 0.00008, 0.00003, 0.00001]:
        if time.time() > dl - 2:
            break
        hz, cz = fine_pair_search(hh, nn, min(time.time() + max(0.1, (dl - time.time()) / 5.0), dl - 2), 5, [fd])
        if cz < nuance[1] - 1e-16:
            hh = hz; nuance[1] = cz; nuance[0] = hz
            hh[hh < 0.0] = 0.0; hh[hh > 1.0] = 1.0
    return hh
