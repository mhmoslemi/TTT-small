def make_beatty_initial(n):
    """Create a Beatty sequence initialization"""
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    target = n // 2
    positions = []
    for k in range(1, 3 * n):
        pos = int(k * golden) % n
        if pos not in positions:
            positions.append(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    # Fix sum
    h = project_feasible(h, n)

            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            accepted += 1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

def gradient_refine(h, n, deadline, alphas=(5000, 20000, 100000, 500000, 2000000)):
    """L-BFGS on continuous relaxation with increasing alpha"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            return smooth_max(compute_c5_vec(x, n), alpha)
        
        def obj_grad(x):
            c = compute_c5_vec(x, n)
            w = smooth_max_weights(c, alpha)
            return grad_from_weights(w, x, n)
        
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 1500, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxls': 40, 'maxcor': 40})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_c = new_c
                best_h = new_h
        except Exception:
            pass
    
    return best_h, best_c

def targeted_unbinary(h, n, deadline):
    """Un-binarize only the top 2-4 positions to reduce the max"""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for num_unbin in [2, 3, 4]:
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        top_lags = np.argsort(c_vec)[::-1][:3]
        
        # Find positions contributing most to the top lag
        k_star = top_lags[0] - (n - 1)
        if k_star >= 0:
            contrib = np.where((h[:n-k_star] > 0.5) & (h[k_star:] < 0.5))[0]
        else:
            kk = -k_star
            contrib = np.where((h[kk:] > 0.5) & (h[:n-kk] < 0.5))[0]
        
        if len(contrib) < num_unbin:
            continue
        
        # Pick top num_unbin from contrib
        grad = grad_from_weights(smooth_max_weights(c_vec, 100000.0), h, n)
        top_pos = contrib[np.argsort(-grad[contrib])[:num_unbin]]
        
        # Try setting them to 0.5
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            h_try = h.copy()
            for p in top_pos:
                h_try[p] = 1.0 - frac  # reduce by frac
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                h = h_try.copy()
    
    return h, best_c

def make_beatty(n, ratio):
    h = np.zeros(n)
    target = n // 2
    seen = set()
    for k in range(1, 10 * n):
        pos = int(k * ratio) % n
        if pos not in seen:
            seen.add(pos)
            if len(seen) >= target:
                break
    for p in seen:
        h[p] = 1.0
    return project_feasible(h, n)

def upsample(h_small, n_small, n_large):
    h_large = np.zeros(n_large)
    for i in range(n_large):
        idx = min(int(i * n_small / n_large), n_small - 1)
        h_large[i] = h_small[idx]
    return project_feasible(h_large, n_large)

def two_swap_search(h, n, deadline, rng, max_iters=50000):
    """Try 2-1 swaps: move two 1s and bring in two 0s simultaneously"""
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Pick two ones and two zeros
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            continue
        
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            for x in [i1, i2]:
                ones.remove(x)
                ones.append(0)  # placeholder
                ones[-1] = x
            ones = [o for o in ones if h[o] > 0.5]
            ones.extend([j1, j2])
            zeros = [z for z in zeros if h[z] < 0.5]
            zeros.extend([i1, i2])
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = make_beatty(500, (1.0 + np.sqrt(5.0)) / 2.0)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: Quick gradient to smooth out (45s)
    p1_end = t0 + 45
    h1, c1 = gradient_refine(best_h, best_n, p1_end, alphas=(2000, 10000, 50000))
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Binarize and intensive lag-aware local search (350s)
    p2_end = t0 + 395
    h_bin = binarize(best_h, best_n)
    h2, c2 = lag_aware_local_search(h_bin, best_n, p2_end, rng, max_iters=500000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    
    # Phase 3: Exhaustive pair phase (120s)
    p3_end = t0 + 515
    h3, c3 = exhaustive_pair_phase(best_h, best_n, p3_end, batch_size=50)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Simulated annealing (200s)
    p4_end = t0 + 715
    h4, c4 = simulated_annealing_binary(binarize(best_h, best_n), best_n, p4_end, rng)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()
    
    # Phase 5: Post-SA local search (130s)
    p5_end = t0 + 845
    h5, c5 = lag_aware_local_search(binarize(best_h, best_n), best_n, p5_end, rng, max_iters=300000)
    if c5 < best_c:
        best_c = c5
        best_h = h5
    
    # Phase 6: Targeted un-binarize (80s)
    p6_end = t0 + 925
    if time.time() < p6_end:
        h6, c6 = targeted_unbinary(best_h, best_n, p6_end)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Gradient polish (60s)
    p7_end = t0 + 985
    if time.time() < p7_end:
        h7, c7 = gradient_refine(best_h, best_n, p7_end, alphas=(100000, 500000, 2000000, 5000000))
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # Try to binarize and see if binary is better
        h7b = binarize(best_h, best_n)
        c7b = compute_c5(h7b, best_n)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b
    
    # Phase 8: Final exhaustive pair + targeted (100s)
    p8_end = deadline - 1
    if time.time() < p8_end:
        h8, c8 = exhaustive_pair_phase(best_h, best_n, min(p8_end - 30, t0 + 1045), batch_size=80)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        if time.time() < p8_end:
            h8b, c8b = targeted_unbinary(best_h, best_n, p8_end)
            if c8b < best_c:
                best_c = c8b
                best_h = h8b
    
    # Final: ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # One last try: binarize and quick local search
    h_bin_final = binarize(best_h, best_n)
    c_bin_final = compute_c5(h_bin_final, best_n)
    if c_bin_final < best_c and time.time() < deadline - 3:
        h_bin_final, c_bin_final = exhaustive_pair_phase(h_bin_final, best_n, deadline - 1, batch_size=100)
        if c_bin_final < best_c:
            best_c = c_bin_final
            best_h = h_bin_final
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
