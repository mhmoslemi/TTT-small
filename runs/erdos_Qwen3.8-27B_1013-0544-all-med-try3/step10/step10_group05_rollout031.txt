            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for ii, idx in enumerate(active):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

# Key new function: gradient-guided pair search
def grad_guided_pair_search(h, n, deadline, max_iter=20000):
    """
    For each step:
    1. Compute correlations and identify active set
    2. Compute gradient
    3. For the coordinate with largest gradient magnitude, find the best partner
    4. Make the move if it improves
    """
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Compute gradient for active set
        alpha_act = 10000
        vals = c_vec[active]
        m = vals.max()
        sh = alpha_act * (vals - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        
        dx = 2.0 / n
        grad = np.zeros(n)
        idx_center = n - 1
        for ii, idx in enumerate(active):
            wi = w[ii]
            if wi < 1e-12:
                continue
            k = int(idx) - idx_center
            wm = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - best_h[k:n])
                    grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - best_h[0:nv])
                    grad[0:nv] += wm * (-best_h[kk:n])
        
        # Find the coordinate with the most negative gradient (best to increase)
        # and the one with the most positive gradient (best to decrease)
        min_grad = np.argmin(grad)
        max_grad = np.argmax(grad)
        
        if best_h[min_grad] >= 1.0 or best_h[max_grad] <= 0.0:
            # Need to find alternatives
            alternatives_up = np.where(best_h < 0.999)[0]
            alternatives_down = np.where(best_h > 0.001)[0]
            if len(alternatives_up) == 0 or len(alternatives_down) == 0:
                break
            min_grad = alternatives_up[np.argmin(grad[alternatives_up])]
            max_grad = alternatives_down[np.argmax(grad[alternatives_down])]
        
        # Try moving from max_grad to min_grad
        delta_candidates = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        improved = False
        for delta in delta_candidates:
            if best_h[max_grad] >= delta and best_h[min_grad] <= 1.0 - delta:
                best_h[max_grad] -= delta
                best_h[min_grad] += delta
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[max_grad] += delta
                    best_h[min_grad] -= delta
        
        if not improved:
            # Try top few gradient coordinates
            sorted_idx = np.argsort(grad)
            tried = 0
            for i in sorted_idx[:10]:
                if tried >= 20:
                    break
                if best_h[i] <= 0.001:
                    continue
                for j in sorted_idx[-10:]:
                    if j == i or best_h[j] >= 0.999:
                        continue
                    delta = 0.001 * (1.0 - best_h[i])
                    if delta < 1e-6:
                        continue
                    best_h[i] -= delta
                    best_h[j] += delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        tried += 1
                        break
                    else:
                        best_h[i] += delta
                        best_h[j] -= delta
                if time.time() > deadline:
                    break
            if not improved:
                # No improvement found with gradient-guided moves
                # Try random pair
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j and best_h[i] > 0.01 and best_h[j] < 0.99:
                    delta = 0.001
                    best_h[i] -= delta
                    best_h[j] += delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += delta
                        best_h[j] -= delta
        
        if not improved and it > 1000:
            # Stuck, try perturbations
            break
    
    return best_h, best_c

# Fine pair search (similar to before but more targeted)
def fine_pair_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        changed = True
        while changed and time.time() < deadline:
            changed = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-8, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            # For each active shift, try to reduce it
            for idx in active[:5]:
                if time.time() > deadline:
                    break
                k = int(idx) - (n - 1)
                if k >= 0:
                    # c[k] = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
                    # To reduce: increase h[i] where h[i+k] is high, or decrease h[i] where h[i+k] is low
                    # Best: decrease h[i] for i where (1-h[i+k]) is large (i.e., h[i+k] is small)
                    # and h[i] is large
                    candidates_decrease = []
                    for i in range(n - k):
                        score = best_h[i] * (1.0 - best_h[i+k])
                        if best_h[i] > 0.01:
                            candidates_decrease.append((score, i, i+k, -1))
                    candidates_decrease.sort(reverse=True)
                    for score, i, j, direction in candidates_decrease[:20]:
                        if best_h[i] >= d:
                            best_h[i] -= d
                            # Find where to put it: where h[j2] is small and (1-h[j2+shift]) contributes
                            # Just try putting it at j (where h is small)
                            if best_h[j] <= 1.0 - d:
                                best_h[j] += d
                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    changed = True
                                    break
                                else:
                                    best_h[j] -= d
                                    best_h[i] += d
                            else:
                                best_h[i] += d
                        if time.time() > deadline:
                            break
                else:
                    kk = -k
                    candidates_decrease = []
                    for i in range(kk, n):
                        score = best_h[i] * (1.0 - best_h[i-kk])
                        if best_h[i] > 0.01:
                            candidates_decrease.append((score, i, i-kk, -1))
                    candidates_decrease.sort(reverse=True)
                    for score, i, j, direction in candidates_decrease[:20]:
                        if best_h[i] >= d:
                            best_h[i] -= d
                            if best_h[j] <= 1.0 - d:
                                best_h[j] += d
                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    changed = True
                                    break
                                else:
                                    best_h[j] -= d
                                    best_h[i] += d
                            else:
                                best_h[i] += d
                        if time.time() > deadline:
                            break
    
    return best_h, best_c

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param * n) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)
