        return project_feasible(h, n), compute_c5(h, n)

def greedy_2swap(h, n, deadline, rng, top_k=40):
    h = h.copy()
    c_cur = compute_c5(h, n)
    best_c = c_cur
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    stall = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        alpha = 200000.0
        w = lse_weights(c_vec, alpha)
        grad = corr_grad(w, h, n)
        
        pos_mask = np.array([grad[i] > 1e-12 for i in ones])
        neg_mask = np.array([grad[j] < -1e-12 for j in zeros])
        
        if not pos_mask.any() or not neg_mask.any():
            break
        
        pos_idx = [ones[i] for i in range(len(ones)) if pos_mask[i]]
        neg_idx = [zeros[i] for i in range(len(zeros)) if neg_mask[i]]
        
        # Sort by gradient magnitude
        pos_idx.sort(key=lambda i: -grad[i])
        neg_idx.sort(key=lambda i: grad[i])
        
        pos_idx = pos_idx[:top_k]
        neg_idx = neg_idx[:top_k]
        
        improved = False
        for i in pos_idx:
            for j in neg_idx:
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h, n)
                if c_new < c_cur - 1e-15:
                    c_cur = c_new
                    if c_new < best_c:
                        best_c = c_new
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 20:
                break
            # Random swap to escape
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            c_cur = c_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            stall = 0
    
    return h, best_c

def local_search_2swap_3swap(h, n, deadline, rng):
    h = h.copy()
    c_cur = compute_c5(h, n)
    best_c = c_cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    stall = 0
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        alpha = 100000.0
        w = lse_weights(c_vec, alpha)
        grad = corr_grad(w, h, n)
        
        improved = False
        
        # Try 2-swaps
        pos_idx = [i for i in ones if grad[i] > 1e-10]
        neg_idx = [j for j in zeros if grad[j] < -1e-10]
        pos_idx.sort(key=lambda i: -grad[i])
        neg_idx.sort(key=lambda j: grad[j])
        pos_idx = pos_idx[:30]
        neg_idx = neg_idx[:30]
        
        for i in pos_idx:
            if improved:
                break
            for j in neg_idx:
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < c_cur - 1e-15:
                    c_cur = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    ones.remove(i); ones.append(j)
                    zeros.remove(j); zeros.append(i)
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            # Try 3-swaps
            pos_idx3 = [i for i in ones if grad[i] > 1e-10][:15]
            neg_idx3 = [j for j in zeros if grad[j] < -1e-10][:15]
            
            for i in pos_idx3:
                if improved:
                    break
                for j in pos_idx3:
                    if i == j or improved:
                        continue
                    for k in pos_idx3:
                        if k == i or k == j or improved:
                            continue
                        for a in neg_idx3:
                            if improved:
                                break
                            for b in neg_idx3:
                                if a == b or improved:
                                    continue
                                for c in neg_idx3:
                                    if a == c or b == c or improved:
                                        continue
                                    h[i] = 0; h[j] = 0; h[k] = 0
                                    h[a] = 1; h[b] = 1; h[c] = 1
                                    c_new = compute_c5(h, n)
                                    if c_new < c_cur - 1e-15:
                                        c_cur = c_new
                                        if c_new < best_c:
                                            best_c = c_new
                                            best_h = h.copy()
                                        for xx in [i, j, k]:
                                            ones.remove(xx); zeros.append(xx)
                                        for xx in [a, b, c]:
                                            zeros.remove(xx); ones.append(xx)
                                        improved = True
                                    else:
                                        h[i] = 1; h[j] = 1; h[k] = 1
                                        h[a] = 0; h[b] = 0; h[c] = 0
                        if improved: break
                    if improved: break
                if improved: break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                num_rand = rng.randint(1, 5)
                for _ in range(num_rand):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones.remove(i); ones.append(j)
                    zeros.remove(j); zeros.append(i)
                c_cur = compute_c5(h, n)
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h.copy()
                stall = 0
        else:
            stall = 0
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, T0=0.005, Tf=0.00005):
    h = h.copy()
    c_cur = compute_c5(h, n)
    best_c = c_cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    # Estimate number of iterations
    t_test_start = time.time()
    h[i if (i := ones[0]) else 0]  # dummy
    # Actually let's just estimate: each iteration takes ~2ms for n=500
    # In remaining time, we can do about 500 * (deadline - now) iterations
    
    remaining = deadline - time.time()
    est_iters = int(remaining * 400)  # ~2.5ms per iteration
    if est_iters < 1000:
        est_iters = 1000
    
    cooling = (Tf / T0) ** (1.0 / est_iters)
    T = T0
    
    for it in range(est_iters):
        if time.time() > deadline:
            break
        
        T *= cooling
        if T < Tf:
            T = Tf
        
        move_type = rng.random()
        
        if move_type < 0.85:
            # 2-swap
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            i = ones[ii]
            j = zeros[jj]
            h[i], h[j] = h[j], h[i]
            c_new = compute_c5(h, n)
            delta = c_new - c_cur
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                ones.remove(i); ones.append(j)
                zeros.remove(j); zeros.append(i)
            else:
                h[i], h[j] = h[j], h[i]
        else:
            # 3-swap
            if len(ones) < 3 or len(zeros) < 3:
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            i3 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            j3 = zeros[rng.randint(len(zeros))]
            if len(set([i1, i2, i3])) < 3 or len(set([j1, j2, j3])) < 3:
                continue
            h[i1] = 0; h[i2] = 0; h[i3] = 0
            h[j1] = 1; h[j2] = 1; h[j3] = 1
            c_new = compute_c5(h, n)
            delta = c_new - c_cur
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                for xx in [i1, i2, i3]:
                    ones.remove(xx); zeros.append(xx)
                for xx in [j1, j2, j3]:
                    zeros.remove(xx); ones.append(xx)
            else:
                h[i1] = 1; h[i2] = 1; h[i3] = 1
                h[j1] = 0; h[j2] = 0; h[j3] = 0
    
    return best_h, best_c

def fine_coord(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(h, n)
            w = lse_weights(c_vec, 500000.0)
            grad = corr_grad(w, h, n)
            
            pos_idx = np.where((grad > 1e-12) & (h > 1e-6))[0]
            neg_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-6))[0]
            
            if len(pos_idx) == 0 or len(neg_idx) == 0:
                break
            
            # Find best pair
            best_gain = 0
            best_i, best_j = -1, -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0:
                break
            
            amt = min(step, h[best_i], 1.0 - h[best_j])
            if amt < 1e-12:
                break
            h[best_i] -= amt
            h[best_j] += amt
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
            else:
                h[best_i] += amt
                h[best_j] -= amt
                h = project_feasible(h, n)
    
    return h, best_c

def upsample(h_old, n_old, n_new):
    """Upsample h from n_old to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    # Actually np.interp doesn't support period. Let's do it manually.
    # For periodic interpolation:
    h_new = np.zeros(n_new)
    for i in range(n_new):
        pos = x_new[i]
        # Find bracketing old points
        old_idx = int(pos / (2.0 / n_old))
        frac = (pos - old_idx * 2.0 / n_old) / (2.0 / n_old)
        h_new[i] = (1 - frac) * h_old[old_idx % n_old] + frac * h_old[(old_idx + 1) % n_old]
    return h_new

def make_beatty(n, alpha_val):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    k = 1
    while len(positions) < target and k < 10 * n:
        pos = int(k * alpha_val) % n
        positions.add(pos)
        k += 1
    for p in positions:
        h[p] = 1.0
    # Fix sum to exactly n/2
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for idx in zeros[:int(np.ceil(diff))]:
                h[idx] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for idx in ones[:int(np.ceil(-diff))]:
                h[idx] = 0.0
    return project_feasible(h, n)
