    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 300:

To decrease c[k], we want to:

- Increase h[i+k] where h[i+k] is small (the "destination")

The best single pair update is: find (i, j=i+k) that maximizes h[i]*(1-h[j]) and move mass from i to j.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 350:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_weights(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

    vals = c[active]

        if cs < best_c:
            best_c = cs
            best_h = hs.copy()

    # Phase 1: Active-set optimization at original resolution (~350s)
    p1_end = min(t0 + 350, deadline - 500)
    schedules1 = [
        (500, 3, 0.02, 2, 500),
        (2000, 3, 0.01, 3, 800),
        (10000, 4, 0.005, 3, 1000),
        (50000, 5, 0.005, 3, 1200),
        (200000, 6, 0.003, 3, 1200),
        (500000, 8, 0.003, 3, 1200),
        (2000000, 10, 0.002, 3, 1200),
        (5000000, 12, 0.001, 3, 1200),
        (10000000, 15, 0.001, 3, 1200),
        (20000000, 20, 0.0005, 2, 1200),
    ]
    for alpha, top_k, mu, n_rounds, maxit_cap in schedules1:
        if time.time() > p1_end - 40:
            break
        for rnd in range(n_rounds):
            if time.time() > p1_end - 20:
                break
            h1, c1 = active_optimize_round(best_h, n_orig, p1_end - 10, top_k, alpha, mu, maxit_cap)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    # Phase 2: Full smooth max polish at original resolution (~80s)
    p2_end = min(time.time() + 80, deadline - 400)
    if p2_end > time.time() + 20:
        bounds = [(0.0, 1.0)] * n_orig
        for alpha in [5000000, 20000000, 50000000]:
            if time.time() > p2_end - 20:
                break
            rem = p2_end - time.time() - 10
            if rem < 20:
                break
            maxit = max(100, min(int(rem), 800))
            try:
                res = minimize(obj_full, best_h, jac=grad_full,
                              args=(n_orig, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 12,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 3: Upsample to 2x and optimize (~300s)
    n_up = n_orig * 2
    p3_end = min(time.time() + 300, deadline - 200)
    if p3_end > time.time() + 60 and n_up <= 4000:
        h_up = upsample_h(best_h, n_orig, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.003:
            schedules3 = [
                (10000, 4, 0.005, 2, 600),
                (50000, 5, 0.005, 3, 800),
                (200000, 6, 0.003, 3, 800),
                (500000, 8, 0.003, 3, 800),
                (2000000, 10, 0.002, 3, 800),
                (5000000, 12, 0.001, 3, 800),
                (10000000, 15, 0.001, 3, 800),
                (20000000, 20, 0.0005, 2, 800),
            ]
            for alpha, top_k, mu, n_rounds, maxit_cap in schedules3:
                if time.time() > p3_end - 40:
                    break
                for rnd in range(n_rounds):
                    if time.time() > p3_end - 20:
                        break
                    h_up, c_up = active_optimize_round(h_up, n_up, p3_end - 10, top_k, alpha, mu, maxit_cap)
                    if c_up < best_c:
                        best_c = c_up
                        best_h = h_up.copy()
                        best_n = n_up
                    else:
                        h_up = best_h if len(best_h) == n_up else upsample_h(best_h, best_n, n_up)
        # After higher res, if we improved, do more rounds
        if best_n == n_up:
            for alpha, top_k, mu in [(5000000, 10, 0.001), (10000000, 15, 0.0005), (20000000, 20, 0.0003)]:
                if time.time() > p3_end - 20:
                    break
                h_up2, c_up2 = active_optimize_round(best_h, best_n, p3_end - 10, top_k, alpha, mu, 600)
                if c_up2 < best_c:
                    best_c = c_up2
                    best_h = h_up2

    # Phase 4: Gradient-guided pair search at best resolution (~200s)
    p4_end = min(time.time() + 200, deadline - 60)
    if p4_end > time.time() + 30:
        cn = len(best_h)
        h4, c4 = gradient_pair_search(best_h, cn, p4_end, max_deltas=50)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = cn

    # Phase 5: More active-set rounds (tight, focused) (~100s)
    p5_end = min(time.time() + 100, deadline - 30)
    if p5_end > time.time() + 20:
        cn = len(best_h)
        for alpha, top_k, mu in [(10000000, 3, 0.001), (50000000, 4, 0.001),
                                  (100000000, 5, 0.0005), (200000000, 6, 0.0005)]:
            if time.time() > p5_end - 15:
                break
            h5, c5 = active_optimize_round(best_h, cn, p5_end - 10, top_k, alpha, mu, 500)
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = cn

    # Phase 6: Final ultra-fine pair search (~20s)
    p6_end = deadline - 3
    if p6_end > time.time() + 10:
        cn = len(best_h)
        h6, c6 = gradient_pair_search(best_h, cn, p6_end, max_deltas=30)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = cn

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
