import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * max(1.0, target):
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    vals = c[active]

def directed_local_search(h, n, deadline, max_rounds=50):

    For the worst shift k, the overlap is ∫h(x)(1-h(x+k))dx.
    To reduce it, we should:
    - Where h(x) is high AND h(x+k) is low: reduce h(x) (move mass)
    - Where h(x) is low AND h(x+k) is high: increase h(x)
    But we need to maintain the integral constraint.

            break

        k = worst_idx - (n - 1)
        dx = 2.0 / n

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

        p4_end = p4_start + min(remaining, 200)
        h4, c4 = run_active_lbfgs(h_hi, n_hi, p4_end, top_k=5,
                                  alphas=[5000, 100000, 2000000, 50000000],
                                  mu=0.001)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_hi
    
    # Phase 5: Active-set refinement at best (150s)
    p5_end = min(t0 + 800, deadline - 150)
    if time.time() < p5_end:
        h5, c5 = run_active_lbfgs(best_h, best_n, p5_end, top_k=2,
                                  alphas=[5000000, 50000000, 500000000],
                                  mu=0.0001)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Directed local search (100s)
    p6_end = min(t0 + 900, deadline - 80)
    if time.time() < p6_end:
        h6, c6 = directed_search(best_h, best_n, p6_end, max_rounds=300)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP with equality constraint (80s)
    p7_end = min(t0 + 980, deadline - 40)
    if time.time() < p7_end:
        h7, c7 = run_slqp_polish(best_h, best_n, p7_end,
                                 alphas=[1000000, 10000000], maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Directed search again (40s)
    p8_end = deadline - 15
    if time.time() < p8_end:
        h8, c8 = directed_search(best_h, best_n, p8_end, max_rounds=200)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final: mini pair search with remaining time
    if time.time() < deadline - 2:
        h9, c9 = mini_pair_search(best_h, best_n, deadline - 1, num_pairs=8000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
