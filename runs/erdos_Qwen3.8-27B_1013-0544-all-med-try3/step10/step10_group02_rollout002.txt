                     bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-20,
                              'gtol': 1e-16, 'maxfun': maxiter * 10,
                              'maxls': 50, 'maxcor': 60})
        return project_feasible(res.x, n), compute_c5(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run_active_lbfgs(h, n, deadline, top_k, alphas, mu, tol_frac=0.001):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for round_idx in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_indices(c_vec, top_k, tol_frac)
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(200, int(min(remaining * 0.3, 3000)))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def fourier_fit(h, n, m):
    """Fit a Fourier series with m harmonics to h, return coefficients."""
    x = np.linspace(0, 2, n, endpoint=False)
    # Use FFT for efficiency
    H = np.fft.rfft(h)
    # Keep only m+1 positive frequencies (DC + m harmonics)
    # For n samples on [0,2), the frequency k corresponds to cos(kπx), sin(kπx)
    # Actually np.fft.rfft gives frequencies 0, 1, 2, ..., n/2
    # Frequency k in FFT corresponds to cos(2πkx/n * 2) = cos(2πk*2/(n*L/2)...)
    # Hmm, this is getting complicated. Let me just use lstsq.
    
    # Actually, let me use the DCT-like approach:
    # h(x_i) ≈ c0 + sum_{k=1}^{m} [a_k cos(k*pi*x_i) + b_k sin(k*pi*x_i)]
    # where x_i = 2*i/n for i=0,...,n-1
    
    x = 2.0 * np.arange(n) / n
    A = np.ones((n, 2*m + 1))
    for k in range(1, m + 1):
        A[:, 2*k-1] = np.cos(k * np.pi * x)
        A[:, 2*k] = np.sin(k * np.pi * x)
    
    coeffs, _, _, _ = np.linalg.lstsq(A, h, rcond=None)
    return coeffs

def fourier_eval(coeffs, x):
    """Evaluate Fourier series at points x."""
    m = (len(coeffs) - 1) // 2
    result = np.full_like(x, coeffs[0])
    for k in range(1, m + 1):
        result += coeffs[2*k-1] * np.cos(k * np.pi * x)
        result += coeffs[2*k] * np.sin(k * np.pi * x)
    return result

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    tol = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        threshold = c_max - tol * c_max
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 20:
            active = np.argsort(-c_vec[active])[:20]
            active = np.array([active] if isinstance(active[0], np.ndarray) else active)
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Uniform average gradient over active
        grad = np.zeros(n)
        for idx in active:
            grad += overlap_grad_single(h, n, idx)
        grad /= len(active)
        
        # Line search
        best_eta = eta
        best_c_try = best_c + 1
        for trial_eta in [eta, eta*0.5, eta*0.1, eta*2, eta*5]:
            h_try = h - trial_eta * grad
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_try:
                best_c_try = c_try
                best_eta = trial_eta
        
        h = h - best_eta * grad
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
            eta = min(best_eta * 1.1, 0.1)
        else:
            h = best_h.copy()
            eta *= 0.7
    
    return best_h, best_c
