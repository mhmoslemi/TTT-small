```python
def project_feasible(h, n):
```

Alternative: use a **subgradient method** for the max of correlations. The subgradient of max_k c[k] at a point where k* is the argmax is just the gradient of c[k*]. We can use stochastic subgradient descent:
- At each step, pick a random k (or the current argmax)
- Compute the gradient of c[k] with respect to h
- Take a step in the negative gradient direction
- Project back to the feasible set

The gradient of c[k] = Σ_x h[x] * (1-h[x+k]) with respect to h:
∂c[k]/∂h[x] = (1-h[x+k]) - h[x-k]
(same formula as the "gain" we computed earlier!)

This is very nice. So the subgradient method would be:
1. Compute c, find argmax k*
2. Compute grad = (1-h[x+k*]) - h[x-k*] for all x
3. h = h - lr * grad
4. Project to [0,1] and adjust sum

This is O(n) per step and very fast! Let me add this as an additional optimization phase.

Actually, this is essentially what the "focused binary search" does, but in continuous space. The binary search is the discrete version (only 0→1 and 1→0 moves), while the subgradient method allows all continuous moves.

```python
def subgradient_descent(h_init, n, deadline, rng, lr_init=0.1, lr_decay=0.999):
    """Subgradient descent for max correlation minimization"""
    dx = 2.0 / n
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = int(corr.argmax())
        shift = worst_idx - (n - 1)
        
        # Subgradient at worst shift
        h_fwd = shifted_get(h, shift)
        h_bwd = shifted_get(h, -shift)
        grad = (1.0 - h_fwd) - h_bwd
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        else:
            # Reduce learning rate
            lr *= lr_decay
            h = h_new  # Still move, but with smaller effective step
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary_given(h, n, target_ones, rng):
    target = target_ones
    hb = (h >= 0.5).astype(float)
    n_ones = int(hb.sum())
    if n_ones > target:
        ones = np.where(hb == 1)[0]
        perm = rng.permutation(len(ones))
        hb[ones[perm[:n_ones - target]]] = 0.0
    elif n_ones < target:
        zeros = np.where(hb == 0)[0]
        perm = rng.permutation(len(zeros))
        hb[zeros[perm[:target - n_ones]]] = 1.0
    return hb

def shifted_get(h, offset):
    n = len(h)
    result = np.zeros(n)
    if offset >= 0:
        valid_len = n - offset
        if valid_len > 0:
            result[:valid_len] = h[offset:]
    else:
        o = -offset
        valid_len = n - o
        if valid_len > 0:
            result[o:] = h[:valid_len]
    return result

def thue_morse(n):
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def binary_search_v2(h_init, n, deadline, rng, max_iters=200000):
    target = n // 2
    h = to_binary_given(h_init, n, target, rng).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max
    best_h = h.copy()
    no_improve = 0
    dx = 2.0 / n

    for it in range(max_iters):
        if time.time() > deadline:
            break

        worst_idx = int(corr.argmax())
        shift = worst_idx - (n - 1)

        h_fwd = shifted_get(h, shift)
        h_bwd = shifted_get(h, -shift)
        gain = (1.0 - h_fwd) - h_bwd

        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]

        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break

        improved = False
        min_g_zero = gain[zeros_idx].min()
        max_g_one = gain[ones_idx].max()

        if min_g_zero < max_g_one - 1e-10:
            # Try top-k candidate pairs
            zi = zeros_idx[np.argsort(gain[zeros_idx])[:10]]
            oi = ones_idx[np.argsort(-gain[ones_idx])[:10]]
            for bi in zi:
                if improved:
                    break
                for bj in oi:
                    if bi == bj:
                        continue
                    h[bi] = 1.0
                    h[bj] = 0.0
                    new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
                    new_max = new_corr.max()
                    if new_max < cur_max - 1e-12:
                        cur_max = new_max
                        corr = new_corr
                        no_improve = 0
                        if cur_max < best_c:
                            best_c = cur_max
                            best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[bi] = 0.0
                        h[bj] = 1.0

        if not improved:
            no_improve += 1
            # Random swaps to escape
            n_rand = min(30, 10 + no_improve // 5)
            for _ in range(n_rand):
                if time.time() > deadline:
                    break
                ri = rng.choice(zeros_idx)
                rj = rng.choice(ones_idx)
                if ri == rj:
                    continue
                h[ri] = 1.0
                h[rj] = 0.0
                new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
                new_max = new_corr.max()
                if new_max < cur_max - 1e-12:
                    cur_max = new_max
                    corr = new_corr
                    no_improve = 0
                    if cur_max < best_c:
                        best_c = cur_max
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[ri] = 0.0
                    h[rj] = 1.0

        if no_improve > 300:
            break

    return best_h, float(best_c * dx)

def subgradient_descent(h_init, n, deadline, rng, lr_init=0.05):
    dx = 2.0 / n
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    avg_h = h.copy()
    count = 0

    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = int(corr.argmax())
        shift = worst_idx - (n - 1)

        # Randomize: sometimes pick a top-3 shift
        if rng.rand() < 0.3:
            top3 = np.argsort(-corr)[:3]
            worst_idx = top3[rng.randint(3)]
            shift = worst_idx - (n - 1)

        h_fwd = shifted_get(h, shift)
        h_bwd = shifted_get(h, -shift)
        grad = (1.0 - h_fwd) - h_bwd

        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        avg_h = avg_h + (h_new - avg_h) / (count + 1)
        count += 1

        c = compute_c5(h_new, n)
        c_avg = compute_c5(avg_h, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        if c_avg < best_c:
            best_c = c_avg
            best_h = avg_h.copy()

        h = h_new
        lr = max(lr * 0.9995, 1e-6)

        if count % 5000 == 0 and lr < 1e-4:
            break

    return best_h, best_c

def slsqp_polish(h_init, n, deadline, rng, alphas=None):
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 10000000]
    dx = 2.0 / n
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        if remaining < 5:
            break

        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        def cons_fun(x):
            return x.sum() - n / 2.0

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun}]

        maxiter = max(200, int(remaining * 2))
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass

    return best_h, best_c
```
