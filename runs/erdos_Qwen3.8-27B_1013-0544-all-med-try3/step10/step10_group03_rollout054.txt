    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        idx_max = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[idx_max] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process shifts with significant weight
    thresh = 1e-12
    active = np.where(w > thresh)[0]
    for mm in active:
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active_indices, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_indices]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_indices, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_indices]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(active_indices))
        w[0] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_indices):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgrad_shift(h, n, k_idx):
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def run_homotopy_lbfgs(h, n, deadline, alphas, mu_values, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 5:
            break
        mu = mu_values[min(i, len(mu_values) - 1)]
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 2, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_active_set(h, n, deadline, top_k=5, max_rounds=40):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(max_rounds):

            break
        c_vec = compute_c5_vec(h, n)

        threshold = max_val - max(1e-7, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        if len(active) == 0:
            break
        for alpha in [5000, 20000, 100000]:
            if time.time() > deadline - 2:

            try:
                res = minimize(active_obj, h, jac=active_grad,

                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 30000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def run_subgradient(h, n, deadline, n_iters=100000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t = 0.005
    gamma = 0.99995
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        tol = max(1e-6, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        grad = np.zeros(n)
        for idx in active:
            grad += subgrad_shift(h, n, idx)
        grad /= len(active)
        h_new = h - t * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
        t *= gamma
        if t < 1e-9:
            break
    return best_h, best_c

def run_restart_optimize(h, n, deadline, n_restarts=3, alpha=50000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    for r in range(n_restarts):
        if time.time() > deadline - 10:

            h_new = project_feasible(res.x, n)

            if c_new < best_c:

def fine_local_search(h, n, deadline):

    best_c = compute_c5(h, n)

            # Compute gradient at active shifts
            grad = np.zeros(n)

            grad /= len(active)
            # Find best pair to move
            # Move mass from high gradient to low gradient (or vice versa)
            g_sorted = np.argsort(-np.abs(grad)[:100])
            best_improve = 0
            for i in range(min(30, len(g_sorted))):

                xi = g_sorted[i]

                    gi, gj = grad[xi], grad[xj]
                    if abs(gi - gj) < 1e-10:

                    # Try both directions

                        # Decrease h[xi], increase h[xj]

                            h[xj] += step
