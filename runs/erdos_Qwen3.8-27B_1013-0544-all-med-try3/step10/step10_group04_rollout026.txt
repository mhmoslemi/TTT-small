import time
import numpy as np
from scipy.optimize import minimize

_OPT_CACHE = {
    "xref": None,
    "p": None,
    "n": -1,
    "alpha": -1.0,
    "kappa": -1.0,
    "valid": False,

def fft_size(need):
    n = 1
    while n < need:
        n <<= 1
    return n


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.reshape(-1)
    if h.size != n:
        tmp = np.zeros(n, dtype=float)
        m = min(n, h.size)
        tmp[:m] = h[:m]
        h = tmp
    else:
        h = h.copy()

    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)

    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if abs(diff) <= 1e-13 * max(1.0, target):
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            frac = diff / tot
            if frac >= 1.0:
                h[room > 1e-18] = 1.0
            else:
                h += frac * room
                np.clip(h, 0.0, 1.0, out=h)
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            frac = diff / tot
            if frac <= -1.0:
                h[h > 1e-18] = 0.0
            else:
                h += frac * h
                np.clip(h, 0.0, 1.0, out=h)

    s = float(h.sum())
    diff = target - s
    for _ in range(120):
        if abs(diff) <= 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room <= 1e-18:
                break
            add = room if room < diff else diff
            h[idx] += add
            diff -= add
        else:
            idx = int(np.argmin(h))
            room = h[idx]
            if room <= 1e-18:
                break
            rem = room if room < -diff else -diff
            h[idx] -= rem
            diff += rem

    np.clip(h, 0.0, 1.0, out=h)
    return h


def project_fast(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(25):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-13 * max(1.0, target):
            break
        h += diff / float(n)
        np.clip(h, 0.0, 1.0, out=h)
    if abs(target - float(h.sum())) > 1e-12 * max(1.0, target):
        return project_feasible(h, n)
    return h


def c_vec(h, n):
    if n <= 0:
        return np.empty(0, dtype=float)
    h = np.asarray(h, dtype=float)
    if n <= 256:
        c = np.correlate(1.0 - h, h, mode="full")
    else:
        need = 2 * n - 1
        N = fft_size(need)
        a = np.zeros(N, dtype=float)
        b = np.zeros(N, dtype=float)
        a[:n] = 1.0 - h
        b[:n] = h[::-1]
        c = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(b), N)
        c = c[:need]
        c = np.maximum(c, 0.0)
    c = c * (2.0 / float(n))
    return c


def max_c_fast(h, n):
    c = c_vec(h, n)
    if c.size == 0:
        return float("inf")
    return float(np.max(c))


def evaluate_exact(h, n):
    h = project_feasible(h, n)
    c = np.correlate(1.0 - h, h, mode="full")
    c = c * (2.0 / float(n))
    return h, float(np.max(c))


def interpolate_to(h_old, n_old, n_new):
    if n_new <= 0:
        return np.empty(0, dtype=float)
    if n_old <= 0:
        return project_feasible(np.full(n_new, 0.5, dtype=float), n_new)
    if n_old == 1:
        return project_feasible(np.full(n_new, float(h_old[0]), dtype=float), n_new)
    h_old = np.asarray(h_old, dtype=float)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    dxo = 2.0 / float(n_old)
    dxn = 2.0 / float(n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * dxo
    x_new = (np.arange(n_new, dtype=float) + 0.5) * dxn
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def grad_from_weights(h, n, w):
    if n <= 0:
        return np.empty(0, dtype=float)
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    dx = 2.0 / float(n)

    if n == 1:
        w0 = float(w[0]) if w.size > 0 else 1.0
        return w0 * (1.0 - 2.0 * h) * dx

    center = n - 1
    w0 = float(w[center])
    g = w0 * (1.0 - 2.0 * h)

    wp = w[center + 1:]
    wn = w[:center][::-1]

    if wp.size != n - 1 or wn.size != n - 1:
        # Fallback: recompute stable weights if inconsistent lengths appear.
        c = c_vec(h, n)
        m = float(np.max(c))
        sh = 500.0 * (c - m)
        sh -= float(np.max(sh))
        e = np.exp(sh)
        e = e / float(np.sum(e))
        wp = e[center + 1:]
        wn = e[:center][::-1]
        w0 = float(e[center])
        g = w0 * (1.0 - 2.0 * h)

    u = wp + wn
    U = np.zeros(n, dtype=float)
    U[1:] = u

    need = 2 * n - 1
    N = fft_size(need)

    Up = np.zeros(N, dtype=float)
    Up[:n] = U
    fU = np.fft.rfft(Up)

    Hp = np.zeros(N, dtype=float)
    Hp[:n] = h
    fH = np.fft.rfft(Hp)

    y_minus = np.fft.irfft(fH * fU, N)[:n]

    Hrp = np.zeros(N, dtype=float)
    Hrp[:n] = h[::-1]
    fHr = np.fft.rfft(Hrp)

    y_plus_full = np.fft.irfft(fHr * fU, N)[:n]
    y_plus = y_plus_full[::-1]

    pref_p = np.cumsum(wp)
    constA = np.zeros(n, dtype=float)
    constA[:n - 1] = pref_p[::-1]

    constB = np.zeros(n, dtype=float)
    constB[1:] = np.cumsum(wn)

    g += constA + constB - (y_minus + y_plus)
    g *= dx
    return g


def _smooth_weights(h, n, alpha):
    c = c_vec(h, n)
    if c.size == 0:
        return 0.0, np.empty(0, dtype=float)
    m = float(np.max(c))
    sh = alpha * (c - m)
    sh_max = float(np.max(sh))
    e = np.exp(sh - sh_max)
    z = float(np.sum(e))
    p = e / z
    val = m + np.log(z) / float(alpha)
    return val, p


def lbfgs_obj(h, n, alpha, kappa):
    alpha = float(alpha)
    kappa = float(kappa)
    val, p = _smooth_weights(h, n, alpha)
    diff = float(np.sum(h) - 0.5 * float(n))
    if kappa != 0.0:
        val += 0.5 * kappa * diff * diff
    _OPT_CACHE["xref"] = h
    _OPT_CACHE["p"] = p
    _OPT_CACHE["n"] = int(n)
    _OPT_CACHE["alpha"] = alpha
    _OPT_CACHE["kappa"] = kappa
    _OPT_CACHE["valid"] = True
    return float(val)


def lbfgs_grad(h, n, alpha, kappa):
    alpha = float(alpha)
    kappa = float(kappa)
    p = None

    if (
        _OPT_CACHE["valid"]
        and h is _OPT_CACHE["xref"]
        and int(_OPT_CACHE["n"]) == int(n)
        and abs(float(_OPT_CACHE["alpha"]) - alpha) <= 1e-14
        and abs(float(_OPT_CACHE["kappa"]) - kappa) <= 1e-14
        and _OPT_CACHE["p"] is not None
    ):
        p = _OPT_CACHE["p"]
        _OPT_CACHE["valid"] = False

    if p is None:
        _, p = _smooth_weights(h, n, alpha)

    g = grad_from_weights(h, n, p)
    diff = float(np.sum(h) - 0.5 * float(n))
    if kappa != 0.0:
        g = g + kappa * diff
    return g


def slqp_obj(h, n, alpha, kappa):
    return lbfgs_obj(h, n, float(alpha), 0.0)


def slqp_grad(h, n, alpha, kappa):
    return lbfgs_grad(h, n, float(alpha), 0.0)


def slqp_eq(h, n):
    return float(np.sum(h) - 0.5 * float(n))


def slqp_eq_jac(h, n):
    return np.ones(n, dtype=float)


def estimate_eval_time(h, n):
    try:
        h = project_feasible(h, n)
        t0 = time.time()
        for _ in range(3):
            lbfgs_obj(h, n, 100.0, 1.0)
            lbfgs_grad(h, n, 100.0, 1.0)
        dt = (time.time() - t0) / 3.0
        return max(dt, 1e-5)
    except Exception:
        return 0.01


def make_configs(n):
    if n <= 512:
        alphas = [30.0, 100.0, 400.0, 1500.0, 6000.0, 25000.0]
    elif n <= 1024:
        alphas = [50.0, 200.0, 800.0, 3000.0, 12000.0, 60000.0]
    elif n <= 2048:
        alphas = [50.0, 200.0, 1000.0, 5000.0, 20000.0]
    else:
        alphas = [50.0, 300.0, 1500.0, 8000.0]
    mus = [0.01, 0.05, 0.25, 1.0, 5.0, 25.0, 100.0]
    cfgs = []
    for i, a in enumerate(alphas):
        m = mus[min(i, len(mus) - 1)]
        cfgs.append((float(a), float(m)))
    return cfgs


def phase_seconds(n):
    if n <= 512:
        return 80.0
    if n <= 1024:
        return 220.0
    if n <= 2048:
        return 260.0
    if n <= 3072:
        return 210.0
    if n <= 4096:
        return 260.0
    return 130.0


def run_lbfgs_phase(h, n, end_time, configs):
    h = project_feasible(h, n)
    h_cur = h.copy()
    best_h = h.copy()
    best_fast = max_c_fast(h, n)

    if time.time() > end_time - 10.0:
        return best_h, best_fast

    te = estimate_eval_time(h, n)
    bounds = [(0.0, 1.0)] * int(n)

    for i, cfg in enumerate(configs):
        if time.time() > end_time - 10.0:
            break
        alpha, kappa = float(cfg[0]), float(cfg[1])
        rem = max(0.0, end_time - time.time() - 10.0)
        if rem <= 5.0:
            break
        nleft = max(1, len(configs) - i)
        maxfun = int(rem / (nleft * max(te, 1e-5) * 3.0))

        if n > 8192:
            cap = 3000
        elif n > 4096:
            cap = 6000
        elif n > 2048:
            cap = 12000
        else:
            cap = 25000
        maxfun = min(maxfun, cap)
        maxfun = max(maxfun, 50)

        try:
            res = minimize(
                lbfgs_obj,
                h_cur,
                jac=lbfgs_grad,
                args=(int(n), alpha, kappa),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": max(60, 2 * maxfun),
                    "maxfun": maxfun,
                    "ftol": 1e-12,
                    "gtol": 1e-7,
                    "maxls": 30,
                    "maxcor": 50,
                    "iprint": 0,
                },
            )
            if res is not None and hasattr(res, "x"):
                h_new = project_feasible(res.x, int(n))
                c_new = max_c_fast(h_new, int(n))
                if c_new < best_fast:
                    best_fast = float(c_new)
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
                elif c_new < best_fast + 1e-6:
                    h_cur = h_new.copy()
        except Exception:
            pass

    return best_h, best_fast


def slqp_polish(h, n, end_time, alpha, maxiter):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_fast = max_c_fast(h, n)

    if time.time() > end_time - 10.0 or maxiter <= 0:
        return best_h, best_fast

    bounds = [(0.0, 1.0)] * int(n)
    cons = [
        {
            "type": "eq",
            "fun": slqp_eq,
            "jac": slqp_eq_jac,
            "args": (int(n),),
        }
    ]

    try:
        res = minimize(
            slqp_obj,
            h,
            jac=slqp_grad,
            args=(int(n), float(alpha), 0.0),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-12,
                "iprint": 0,
            },
        )
        if res is not None and hasattr(res, "x"):
            h_new = project_feasible(res.x, int(n))
            c_new = max_c_fast(h_new, int(n))
            if c_new < best_fast:
                best_fast = float(c_new)
                best_h = h_new.copy()
    except Exception:
        pass

    return best_h, best_fast


def active_descent(h, n, end_time, seed=123):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_fast = max_c_fast(h, n)
    rng = np.random.RandomState(seed)

    csize = 2 * int(n) - 1
    base_k = [5, 10, 20, 40, 80, 160]
    if csize > 1000:
        base_k.append(max(20, int(0.02 * csize)))
        base_k.append(max(50, int(0.05 * csize)))

    Ks = []
    for k in base_k:
        k = int(k)
        if 1 <= k < csize and k not in Ks:
            Ks.append(k)
    if not Ks:
        Ks = [max(1, csize // 2)]
    if csize <= 1:
        Ks = [1]
    Ks = sorted(Ks)

    steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    no_improve = 0
    it = 0

    while time.time() < end_time - 3.0:
        it += 1
        if it > 250000:
            break

        c = c_vec(h, n)
        mx = float(np.max(c))
        if mx < best_fast:
            best_fast = mx
            best_h = h.copy()

        K = Ks[it % len(Ks)]
        if K >= csize:
            p = np.full(csize, 1.0 / float(csize), dtype=float)
        else:
            idx = np.argpartition(c, -K)[-K:]
            p = np.zeros(csize, dtype=float)
            p[idx] = 1.0 / float(K)

        g = grad_from_weights(h, n, p)
        g = g - float(np.mean(g))
        gnorm = float(np.linalg.norm(g, ord=np.inf))

        if gnorm <= 1e-14:
            no_improve += 1
        else:
            accepted = False
            for st in steps:
                if time.time() > end_time - 2.0:
                    break
                s = st
                if no_improve > 6:
                    s *= 0.25
                eta = s / gnorm
                h_new = project_fast(h - eta * g, n)
                c_new = max_c_fast(h_new, n)
                if c_new < best_fast - 1e-15:
                    h = h_new.copy()
                    best_fast = float(c_new)
                    best_h = h_new.copy()
                    no_improve = 0
                    accepted = True
                    break
            if not accepted:
                no_improve += 1
                if no_improve == 10:
                    r = rng.randn(n)
                    r = r - float(np.mean(r))
                    h = project_fast(h + 0.001 * r, n)
                    c_try = max_c_fast(h, n)
                    if c_try < best_fast:
                        best_fast = float(c_try)
                        best_h = h.copy()
                    no_improve = 0

    return best_h, best_fast


def random_binary(n, seed):
    if n <= 0:
        return np.empty(0, dtype=float)
    rng = np.random.RandomState(seed)
    x = rng.rand(n)
    h = (x >= np.median(x)).astype(float)
    target = 0.5 * float(n)
    cnt = float(h.sum())

    if cnt > target:
        excess = int(round(cnt - target))
        ones = np.where(h == 1.0)[0]
        if ones.size > 0 and excess > 0:
            k = min(excess, ones.size)
            order = ones[np.argsort(x[ones])[:k]]
            h[order] = 0.0
    elif cnt < target:
        need = int(round(target - cnt))
        zeros = np.where(h == 0.0)[0]
        if zeros.size > 0 and need > 0:
            k = min(need, zeros.size)
            order = zeros[np.argsort(-x[zeros])[:k]]
            h[order] = 1.0

    return project_feasible(h, n)


def binary_start_from_h(h, n_target, seed):
    h = interpolate_to(h, len(h), n_target)
    b = (h > 0.5).astype(float)
    target = 0.5 * float(n_target)
    cnt = float(b.sum())

    if cnt > target:
        excess = int(round(cnt - target))
        ones = np.where(b == 1.0)[0]
        if ones.size > 0 and excess > 0:
            k = min(excess, ones.size)
            order = ones[np.argsort(h[ones])[:k]]
            b[order] = 0.0
    elif cnt < target:
        need = int(round(target - cnt))
        zeros = np.where(b == 0.0)[0]
        if zeros.size > 0 and need > 0:
            k = min(need, zeros.size)
            order = zeros[np.argsort(-h[zeros])[:k]]
            b[order] = 1.0

    b = project_feasible(b, n_target)
    # Tiny deterministic perturbation to avoid a perfectly flat binary basin.
    if seed is not None and n_target > 0:
        rng = np.random.RandomState(int(seed))
        b = project_feasible(b + 1e-4 * (rng.rand(n_target) - 0.5), n_target)
    return b


def perturb_start(h, n, mag, seed):
    h = interpolate_to(h, len(h), n)
    rng = np.random.RandomState(seed)
    p = h + float(mag) * (rng.rand(n) - 0.5)
    return project_feasible(p, n)


def symmetrize_h(h, n):
    h = project_feasible(h, n)
    s = 0.5 * (h + h[::-1])
    return project_feasible(s, n)

        cfgs = [(500.0, 0.25), (2000.0, 1.0), (8000.0, 5.0), (30000.0, 25.0)]
        cand_h, _ = run_lbfgs_phase(hsym, n_sym, end, cfgs)
        ex_h, ex_c = evaluate_exact(cand_h, n_sym)
        if ex_c < best_c:
            best_c = float(ex_c)
            best_h = ex_h.copy()
            best_n = int(n_sym)
        if ex_c < best_c + 0.002:
            work_h = ex_h.copy()
            work_n = int(n_sym)

    # Binaryized basin exploration.
    if deadline - time.time() > 90.0:
        n_bin = 2048 if int(best_n) > 1024 else 1024
        hbin = binary_start_from_h(work_h, n_bin, int(seed) + 101)
        end = min(time.time() + 50.0, deadline - 40.0)
        cfgs = [(500.0, 0.25), (2000.0, 1.0), (10000.0, 5.0), (50000.0, 25.0)]
        cand_h, _ = run_lbfgs_phase(hbin, n_bin, end, cfgs)
        ex_h, ex_c = evaluate_exact(cand_h, n_bin)
        if ex_c < best_c:
            best_c = float(ex_c)
            best_h = ex_h.copy()
            best_n = int(n_bin)
        if ex_c < best_c + 0.002:
            work_h = ex_h.copy()
            work_n = int(n_bin)

    # Small near-best perturbation exploration.
    if deadline - time.time() > 80.0:
        n_exp = int(best_n)
        if n_exp > 2048:
            n_exp = 2048
        hpert = perturb_start(best_h, n_exp, 0.02, int(seed) + 7)
        end = min(time.time() + 40.0, deadline - 30.0)
        cfgs = [(1000.0, 1.0), (5000.0, 5.0), (20000.0, 25.0)]
        cand_h, _ = run_lbfgs_phase(hpert, n_exp, end, cfgs)
        ex_h, ex_c = evaluate_exact(cand_h, n_exp)
        if ex_c < best_c:
            best_c = float(ex_c)
            best_h = ex_h.copy()
            best_n = int(n_exp)
        if ex_c < best_c + 0.002:
            work_h = ex_h.copy()
            work_n = int(n_exp)

    # Final active-subgradient polishing.
    if deadline - time.time() > 30.0:
        n_act = int(best_n)
        if n_act > 4096:
            n_act = 4096
        ha = interpolate_to(best_h, best_n, n_act)
        cand_h, _ = active_descent(ha, n_act, deadline - 5.0, seed=int(seed) + 99)
        ex_h, ex_c = evaluate_exact(cand_h, n_act)
        if ex_c < best_c:
            best_c = float(ex_c)
            best_h = ex_h.copy()
            best_n = int(n_act)

        if deadline - time.time() > 45.0 and int(best_n) != n_act and int(best_n) <= 8192:
            hb = interpolate_to(best_h, best_n, int(best_n))
            cand_h2, _ = active_descent(hb, int(best_n), deadline - 5.0, seed=int(seed) + 203)
            ex_h2, ex_c2 = evaluate_exact(cand_h2, int(best_n))
            if ex_c2 < best_c:
                best_c = float(ex_c2)
                best_h = ex_h2.copy()
                best_n = int(best_n)

    best_h = project_feasible(best_h, int(best_n))
    best_h, best_c = evaluate_exact(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
