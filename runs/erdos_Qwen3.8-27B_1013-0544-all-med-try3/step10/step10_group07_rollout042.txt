import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def init_contribs(h, n):
    """Compute contribs matrix of shape (n, 2n-1).
    contribs[k] = the change in corr if h[k] is flipped from 0 to 1."""
    lags = np.arange(-(n-1), n)
    contribs = np.zeros((n, 2*n-1))
    for k in range(n):
        v1 = (k - lags >= 0) & (k - lags < n)
        contribs[k, v1] = 1.0 - h[k - lags[v1]]
        v2 = (k + lags >= 0) & (k + lags < n)
        contribs[k, v2] -= h[k + lags[v2]]
    return contribs

def update_contribs(contribs, n, old_h, new_h):
    """Update contribs after h changes. Only need to update for positions where h changed."""
    # For positions where h changed from 0 to 1:
    # For all k: contribs[k, k-p+n-1] -= 1, contribs[k, p-k+n-1] -= 1
    # For positions where h changed from 1 to 0:
    # For all k: contribs[k, k-p+n-1] += 1, contribs[k, p-k+n-1] += 1
    changed_01 = np.where((old_h == 0) & (new_h == 1))[0]
    changed_10 = np.where((old_h == 1) & (new_h == 0))[0]
    k = np.arange(n)
    for p in changed_01:
        contribs[:, k - p + n - 1] -= 1.0
        contribs[:, p - k + n - 1] -= 1.0
    for p in changed_10:
        contribs[:, k - p + n - 1] += 1.0
        contribs[:, p - k + n - 1] += 1.0
    return contribs

    ones_count = int(h.sum())

        if ones_count > target:
            idx = np.where(h == 1)[0]
            to_remove = rng.choice(idx, ones_count - target, replace=False)
            h[to_remove] = 0
        elif ones_count < target:
            idx = np.where(h == 0)[0]
            to_add = rng.choice(idx, target - ones_count, replace=False)
            h[to_add] = 1

    contribs = init_contribs(h, n) * dx
    cur_c = np.max(corr)
    best_c = cur_c
    best_h = h.copy()

    pass_count = 0
    max_passes = 10000

        pass_count += 1
        improved = False

        ones = np.where(h == 1)[0]

                best_idx = np.argmin(new_cs)
                best_j = ones[best_idx]
                best_new_c = new_cs[best_idx]
                
                accept = False
                if best_new_c < cur_c - 1e-15:
                    accept = True
                elif T > T_min:

                        accept = True

                    h[i] = 1.0
                    h[best_j] = 0.0

                    cur_c = best_new_c

                        best_h = h.copy()
