import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    return grad

                   (20000, 50000.0)]

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-10, 'maxfun': 50000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_passes=10000, random_order=True, seed=42):
    """Coordinate descent: for each i, find optimal h[i] given all others fixed."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    
    n_pass = 0

            break
        n_pass = p
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

                t = old_val + delta
                if 0 <= t <= 1:

            for t in candidates:

                if val < best_f:

                    best_t = t

                                best_t = t

            break
    
    return best_h, best_c5, n_pass
