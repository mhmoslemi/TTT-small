    c = compute_c5_vec(x, n)
    if active_set is not None:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    m = vals.max()
    mean_v = vals.mean()
    var_v = ((vals - mean_v)**2).mean()
    return float(m + var_weight * var_v)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def correlation_grad_single(x, n, corr_idx):
    """Gradient of correlation at given index w.r.t. x."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = corr_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - x[k:n]) * dx
        grad[k:n] += (-x[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - x[0:nv]) * dx
        grad[0:nv] += (-x[kk:n]) * dx
    return grad

def soft_max_obj(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def soft_max_grad(x, n, alpha, active):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-16:
            continue
        grad += wi * correlation_grad_single(x, n, idx)
    return grad

def sum_active_obj(x, n, active):
    """Sum of active correlations (smoother than max)."""
    c = compute_c5_vec(x, n)
    return float(np.sum(c[active]))

def sum_active_grad(x, n, active):
    grad = np.zeros(n)
    for idx in active:
        grad += correlation_grad_single(x, n, idx)
    return grad

def mean_active_obj(x, n, active):
    c = compute_c5_vec(x, n)
    return float(np.mean(c[active]))

def mean_active_grad(x, n, active):
    grad = np.zeros(n)
    ka = float(len(active))
    for idx in active:
        grad += correlation_grad_single(x, n, idx)
    return grad / ka

def var_active_obj(x, n, active, w_max, w_var):
    """w_max * max + w_var * variance of active correlations."""
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    var = np.mean((vals - np.mean(vals))**2)
    return float(w_max * m + w_var * var)

def var_active_grad(x, n, active, w_max, w_var):
    c = compute_c5_vec(x, n)
    vals = c[active]
    ka = len(active)
    mean_v = vals.mean()
    grad = np.zeros(n)
    # Gradient of max (subgradient at argmax)
    idx_max = active[np.argmax(vals)]
    grad += w_max * correlation_grad_single(x, n, idx_max)
    # Gradient of variance
    for i, idx in enumerate(active):
        dval = (2.0 / ka) * (vals[i] - mean_v)
        grad += w_var * dval * correlation_grad_single(x, n, idx)
    return grad

def identify_active(c_vec, top_k, rel_tol=0.005, abs_tol=1e-6):
    max_val = c_vec.max()
    threshold = max_val - max(abs_tol, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c_vec[active])[:top_k]]
    return active

def lbfgs_wrap(obj, grad, h, n, active, deadline, maxiter=20000):
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h
    try:
        res = minimize(obj, h, jac=grad, args=(n, active),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxiter * 4, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_set_optimize(h, n, deadline, phases):
    """Run multiple phases of active-set optimization."""
    best_c = compute_c5(h, n)
    for phase in phases:
        if time.time() > deadline - 5:
            break
        alpha = phase.get('alpha', 10000)
        top_k = phase.get('top_k', 5)
        rel_tol = phase.get('rel_tol', 0.005)
        obj_type = phase.get('obj', 'soft')
        maxiter = phase.get('maxiter', 10000)
        iterations = phase.get('iter', 3)
        
        for _ in range(iterations):
            if time.time() > deadline - 5:
                break
            c_vec = compute_c5_vec(h, n)
            active = identify_active(c_vec, top_k, rel_tol)
            if len(active) == 0:
                break
            
            h_new = lbfgs_wrap(soft_max_obj, soft_max_grad, h, n, active,
                              deadline - 3, maxiter)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
            else:
                # Try sum/mean objective as alternative
                if obj_type == 'sum':
                    h_new2 = lbfgs_wrap(sum_active_obj, sum_active_grad, h, n,
                                       active, deadline - 3, maxiter // 2)
                    c_new2 = compute_c5(h_new2, n)
                    if c_new2 < best_c - 1e-15:
                        best_c = c_new2
                        h = h_new2
                break
    
    return h, best_c

def var_balanced_opt(h, n, deadline):
    """Optimize with variance balancing to equalize top correlations."""
    best_c = compute_c5(h, n)
    for w_var in [0.01, 0.1, 0.5, 1.0, 5.0]:
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - 0.005 * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 10:
            active = active[np.argsort(-c_vec[active])[:10]]
        if len(active) < 2:
            continue
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(var_active_obj, h, jac=var_active_grad,
                          args=(n, active, 1.0, w_var),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': 40000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_2point_search(h, n, deadline, max_trials=50000):
    """Randomized 2-point local search with adaptive step size."""
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(12345)
    interior_idx = np.where((h > 0.005) & (h < 0.995))[0]
    if len(interior_idx) == 0:
        return h, best_c
    
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _ in range(3000):
                if time.time() > deadline:
                    break
                i = interior_idx[rng.integers(len(interior_idx))]
                j = interior_idx[rng.integers(len(interior_idx))]
                if i == j:
                    continue
                for d in [1, -1]:
                    hi = h[i] + d * step
                    hj = h[j] - d * step
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i] += d * step
                    h[j] -= d * step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d * step
                        h[j] += d * step
    return h, best_c

def multistart_refine(h, n, deadline, n_starts=3):
    """Try small perturbations to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(99)
    
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        # Small random perturbation
        h_pert = h.copy()
        mask = rng.random(n) < 0.3
        h_pert[mask] += rng.uniform(-0.01, 0.01, mask.sum())
        h_pert = project_feasible(h_pert, n)
        
        # Quick optimization
        c_vec = compute_c5_vec(h_pert, n)
        active = identify_active(c_vec, 5, 0.01)
        h_new = lbfgs_wrap(soft_max_obj, soft_max_grad, h_pert, n, active,
                          deadline - 20, 5000)
        # Project and repair
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def triangle_wave_init(n, period_frac=0.4):
    """Create a triangle-wave-like initial condition."""
    x = np.linspace(0, 2, n)
    period = 2.0 * period_frac
    h = np.zeros(n)
    for i in range(n):
        t = x[i] % period
        if t < period / 2:
            h[i] = t / (period / 2)
        else:
            h[i] = 1.0 - (t - period / 2) / (period / 2)
    # Scale to satisfy integral constraint
    h = project_feasible(h, n)
    return h
