    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary_search(h, n, deadline, max_sweeps=200):

    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx

    zeros = np.where(h == 0)[0].tolist()
    
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_delta_c = 0
        best_pair = None
        best_new_corr = None
        
        for z in zeros:
            if time.time() > deadline:
                break
            for o in ones:
                # Compute delta for swapping h[z]=0->1, h[o]=1->0
                delta = np.zeros(2 * n - 1)
                va = (z + lags >= 0) & (z + lags < n)
                delta[lags[va] + (n-1)] -= (1.0 - h[z + lags[va]])
                va2 = (z - lags >= 0) & (z - lags < n)
                delta[lags[va2] + (n-1)] += h[z - lags[va2]]
                vb = (o + lags >= 0) & (o + lags < n)
                delta[lags[vb] + (n-1)] += (1.0 - h[o + lags[vb]])
                vb2 = (o - lags >= 0) & (o - lags < n)
                delta[lags[vb2] + (n-1)] -= h[o - lags[vb2]]
                
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c + best_delta_c - 1e-14:  # strictly better
                    best_delta_c = nc - cur_c
                    best_pair = (z, o)
                    best_new_corr = new_corr
                    if best_delta_c < -1e-14:  # significantly better, keep looking
                        pass
                    # Actually for greedy, take the FIRST improvement (first-improvement)
                    # or the BEST improvement (best-improvement)
                    # Best-improvement is better but slower
                    # Let's use first-improvement for speed
                    break
            if best_pair is not None:
                break
        
        if best_pair is None:
            break
        
        z, o = best_pair
        h[z], h[o] = 1.0, 0.0
        corr = best_new_corr
        cur_c = corr.max() * dx
        ones.remove(o)
        ones.append(z)
        zeros.remove(z)
        zeros.append(o)
    
    return h, cur_c

import numpy as np

import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0

def greedy_first_improvement(h, n, deadline, max_sweeps=5000, seed=42):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max() * dx)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    lags = np.arange(-(n - 1), n)
    n_lags = len(lags)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_order = list(range(len(zeros)))
        rng.shuffle(z_order)
        for zi in z_order:
            if time.time() > deadline:
                break
            z = zeros[zi]
            o_order = list(range(len(ones)))
            rng.shuffle(o_order)
            for oi in o_order:
                o = ones[oi]
                va = (z + lags >= 0) & (z + lags < n)
                vb = (o + lags >= 0) & (o + lags < n)
                va2 = (z - lags >= 0) & (z - lags < n)
                vb2 = (o - lags >= 0) & (o - lags < n)
                delta = np.zeros(2 * n - 1)

                    delta[lags[va] + (n - 1)] -= (1.0 - h[z + lags[va]])

                    ones[oi] = z
                    zeros[zi] = o
                    improved = True

            if improved:

            break

def simulated_annealing_binary(h, n, deadline, seed=42, T0=0.003, Tf=1e-9, max_iters=200000):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    lags = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    T = T0
    n_iters = max_iters
    cooling = (Tf / T0) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        va = (a + lags >= 0) & (a + lags < n)
        vb = (b + lags >= 0) & (b + lags < n)
        va2 = (a - lags >= 0) & (a - lags < n)
        vb2 = (b - lags >= 0) & (b - lags < n)
        delta = np.zeros(2 * n - 1)
        if va.any():
            delta[lags[va] + (n - 1)] -= (1.0 - h[a + lags[va]])
        if va2.any():
            delta[lags[va2] + (n - 1)] += h[a - lags[va2]]
        if vb.any():
            delta[lags[vb] + (n - 1)] += (1.0 - h[b + lags[vb]])
        if vb2.any():
            delta[lags[vb2] + (n - 1)] -= h[b - lags[vb2]]
        new_corr = corr + delta
        new_c = float(new_corr.max()) * dx
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-15)):
            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            cur_c = new_c
            ones[ia], zeros[ib] = b, a
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
        T = max(T * cooling, Tf)
    return best_h, best_c

def kick_and_search(h, n, deadline, seed=42, kick_size=5):
    target = n // 2
    rng = np.random.RandomState(seed)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    h_k = h.copy()
    for _ in range(kick_size):
        a = rng.randint(len(ones))
        b = rng.randint(len(zeros))
        oa, oz = h_k[ones[a]], h_k[zeros[b]]
        h_k[ones[a]], h_k[zeros[b]] = oz, oa
    h_g, c_g = greedy_first_improvement(h_k, n, deadline, max_sweeps=2000, seed=seed)
    return h_g, c_g

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def multi_start_binary(n, deadline, seed=42, n_starts=5):
    best_h = None
    best_c = np.inf
    rng = np.random.RandomState(seed)
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        h_rand = (rng.rand(n) < 0.5).astype(float)
        h_rand = to_binary(h_rand, n)
        t_sub = min(time.time() + 20, deadline - 30)
        h_g, c_g = greedy_first_improvement(h_rand, n, t_sub, max_sweeps=500, seed=seed + s)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Convert to binary and do initial greedy
    t1 = t0 + 0.08 * (budget_s - 5)
    hb = to_binary(best_h, n)
    h_g, c_g = greedy_first_improvement(hb, n, t1, max_sweeps=3000, seed=seed)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g.copy()

    # Phase 2: Intensive SA + greedy cycling
    t2 = t0 + 0.65 * (budget_s - 5)
    n_sa_runs = 30
    sa_time_each = (t2 - time.time()) / max(n_sa_runs, 1) * 0.8
    for i in range(n_sa_runs):
        if time.time() > t2 - 10:
            break
        t_sa_end = min(time.time() + max(sa_time_each, 5), t2 - 5)
        h_sa, c_sa = simulated_annealing_binary(best_h, n, t_sa_end, seed=seed + 1000 + i,
                                               T0=0.004 + 0.001 * (i % 5), Tf=1e-9, max_iters=100000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        # Follow up with greedy
        t_gr = min(time.time() + 5, t2 - 3)
        if time.time() < t2 - 3:
            h_g2, c_g2 = greedy_first_improvement(best_h, n, t_gr, max_sweeps=1000, seed=seed + 2000 + i)
            if c_g2 < best_c:
                best_c = c_g2
                best_h = h_g2.copy()

    # Phase 3: Kick and search (ILS)
    t3 = t0 + 0.78 * (budget_s - 5)
    n_kicks = 20
    for i in range(n_kicks):
        if time.time() > t3 - 5:
            break
        kick_sz = 3 + (i % 4) * 2
        t_k = min(time.time() + 8, t3 - 3)
        h_k, c_k = kick_and_search(best_h, n, t_k, seed=seed + 5000 + i, kick_size=kick_sz)
        if c_k < best_c:
            best_c = c_k
            best_h = h_k.copy()

    # Phase 4: Multi-resolution
    t4 = t0 + 0.88 * (budget_s - 5)
    for nh in [200, 300, 350, 400]:
        if time.time() > t4 - 15:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = to_binary(hh, nh)
        t_sub = min(time.time() + 15, t4 - 10)
        h_gn, c_gn = greedy_first_improvement(hh, nh, t_sub, max_sweeps=1000, seed=seed + 7000 + nh)
        if c_gn < best_c:
            best_c = c_gn
            best_h = h_gn.copy()
            n = nh

    # Phase 5: SLSQP polish on best binary
    t5 = dl - 15
    if time.time() < t5:
        h_cont = best_h.astype(float).copy()
        for alpha in [20000, 100000, 500000, 2000000]:
            if time.time() > t5 - 5:
                break
            t_alpha = min(time.time() + 15, t5 - 5)
            h_p = slsqp_smooth(h_cont, n, alpha, 2000, t_alpha)
            c_p = compute_c5(h_p, n)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p.copy()
            h_cont = h_p

    # Phase 6: Final binary + SA polish
    t6 = dl - 3
    if time.time() < t6:
        h_b2 = to_binary(best_h, n)
        h_f, c_f = simulated_annealing_binary(h_b2, n, t6, seed=seed + 9000, T0=0.002, Tf=1e-10, max_iters=80000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
        # Quick greedy then SLSQP
        if time.time() < dl - 1:
            h_gf, c_gf = greedy_first_improvement(h_f, n, dl - 1, max_sweeps=500, seed=seed + 9500)
            if c_gf < best_c:
                best_c = c_gf
                best_h = h_gf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
