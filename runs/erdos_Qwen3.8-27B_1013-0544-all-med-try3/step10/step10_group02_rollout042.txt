            if tot < 1e-15:

    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    val = m + lse / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (-x[k:n])
            grad[k:n] += wm * (1.0 - x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - x[kk:n])
            grad[kk:n] += wm * (-x[0:nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        h8, c8 = run_active_phase(best_h, n, deadline - 1, top_k=3,
            alphas=[100000000, 500000000, 2000000000])
        if c8 < best_c:
            best_c, best_h = c8, h8

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
