        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Add/remove from a single point
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def overlap_grad(h, n, k):
    """Gradient of c_k with respect to h."""
    dx_val = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx_val
        grad[k:n] += (-h[0:nv]) * dx_val
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx_val
        grad[0:nv] += (-h[kk:n]) * dx_val
    return grad

def active_set_obj(h, n, active_idx, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def active_set_grad(h, n, active_idx, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi * dx_val * (1.0 - h[k:n])
            grad[k:n] += wi * dx_val * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * dx_val * (1.0 - h[0:nv])
            grad[0:nv] += wi * dx_val * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_active(h, n, deadline, min_time=10):
    """L-BFGS with re-identified active set."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [5000, 50000, 200000, 1000000, 5000000]
    
    for alpha in alphas:
        if time.time() > deadline - min_time:
            break
        # Identify active set
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Use a wider threshold to capture near-max elements
        threshold = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        # Limit to top 20
        if len(active) > 20:
            active = np.argsort(-c_vec[active])[:20]
        
        maxit = max(200, int((deadline - time.time()) * 0.1))
        try:
            res = minimize(active_set_obj, best_h, jac=active_set_grad,
                          args=(n, active, alpha, 0.001), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def grad_subgradient(h, n, deadline, n_iter=5000):
    """Nesterov-accelerated subgradient method for the active max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    h_curr = h.copy()
    h_prev = h.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Nesterov extrapolation
        gamma = 2.0 / (it + 3)
        h_y = h_curr + gamma * (h_curr - h_prev)
        
        c_vec = compute_c5_vec(h_y, n)
        max_val = c_vec.max()
        # Find all k within tolerance of max
        tol = max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Average gradient over active set
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += overlap_grad(h_y, n, k)
        grad /= len(active)
        
        # Step
        step = 2 * best_c / (np.dot(grad, grad) + 1e-10)
        h_new = h_y - step * grad
        h_new = project_feasible(h_new, n)
        
        h_prev = h_curr.copy()
        h_curr = h_new
        
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
    
    return best_h, best_c

def coord_descent_linesearch(h, n, deadline, max_rounds=200):
    """Coordinate descent with proper line search for pair swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Precompute which indices are near 0 or 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Sort by value: lowest first, highest last
        order = np.argsort(best_h)
        low = order[:n//4]
        high = order[-n//4:]
        
        for i in low:
            if time.time() > deadline:
                break
            for j in high:
                if i == j:
                    continue
                # Try moving mass from i to j (increase h[i], decrease h[j])
                # and from j to i
                for direction in [1, -1]:
                    if direction == 1:
                        lo, hi = i, j
                    else:
                        lo, hi = j, i
                    
                    # Binary search for optimal delta
                    lo_amt = 0.0
                    amt = min(1.0 - best_h[lo], best_h[hi])
                    if amt < 1e-12:
                        continue
                    
                    # Try a few deltas
                    found = False
                    for trial in [amt*0.5, amt*0.25, amt*0.1, amt*0.05, amt*0.01]:
                        if trial < 1e-12:
                            continue
                        best_h[lo] += trial
                        best_h[hi] -= trial
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            found = True
                            break
                        else:
                            best_h[lo] -= trial
                            best_h[hi] += trial
                    
                    if found:
                        # Try to go further
                        for trial in [amt*0.75, amt*0.4, amt*0.15, amt*0.08]:
                            if trial < best_h[hi]:
                                best_h[lo] += trial
                                best_h[hi] -= trial
                                c = compute_c5(best_h, n)
                                if c < best_c - 1e-16:
                                    best_c = c
                                    improved = True
                                else:
                                    best_h[lo] -= trial
                                    best_h[hi] += trial
                                    break
                if improved:
                    break
            if improved:
                break
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30, deltas=None):
    """Very fine pair adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # h[i] += d, h[j] -= d
                    if 0.0 <= best_h[i] + d <= 1.0 and 0.0 <= best_h[j] - d <= 1.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # h[i] -= d, h[j] += d
                    if 0.0 <= best_h[i] - d >= 0.0 and 0.0 <= best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    
    return best_h, best_c

def perturbation_search(h, n, deadline, n_perturbs=5):
    """Try random perturbations and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for _ in range(n_perturbs):
        if time.time() > deadline - 60:
            break
        # Perturb: pick random pairs and swap
        h_pert = best_h.copy()
        n_swaps = n // 50
        for _ in range(n_swaps):
            i, j = rng.choice(n, 2, replace=False)
            amt = min(0.01, h_pert[i], 1.0 - h_pert[j])
            h_pert[i] += amt
            h_pert[j] -= amt
        h_pert = project_feasible(h_pert, n)
        
        # Quick optimize
        h_opt, c_opt = lbfgs_active(h_pert, n, time.time() + 30)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c
