def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            room = 1.0 - h

                break
            h += (diff / tot) * room

            if tot < 1e-14:

            h += (diff / tot) * h

def binary_landscaping(h, n, deadline, rng, alpha=500000, maxiter=2000):
    """Round to binary, then optimize continuously."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    
    h_relaxed = lbfgsb_optimize(h_bin.astype(float), n, alpha, min(deadline, time.time() + 40), maxiter=maxiter)
    return h_relaxed

    for alpha, maxit in zip(alpha_schedule, maxiter_schedule):
        if time.time() > deadline - 200:
            break
        remaining = deadline - 200 - time.time()
        if remaining < 10:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(remaining, 60), maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Subgradient method on top lags
    if time.time() < deadline - 150:
        sg_deadline = min(deadline - 100, time.time() + 80)
        h_opt, c_opt = subgradient_optimize(best_h, n, sg_deadline, top_m=3, initial_step=0.005)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Targeted local search
    if time.time() < deadline - 100:
        ls_deadline = min(deadline - 60, time.time() + 60)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Another round of smooth max at very high alpha
    if time.time() < deadline - 80:
        for alpha in [200000000, 500000000]:
            if time.time() > deadline - 40:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(30, deadline - 40 - time.time()), maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Simulated annealing for diversification
    if time.time() < deadline - 60:
        sa_deadline = min(deadline - 30, time.time() + 50)
        h_opt, c_opt = simulated_annealing(best_h, n, sa_deadline, rng,
                                           alpha_sa=200000, t_init=0.003,
                                           t_min=0.00001, cool=0.99995)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Binary landscaping
    if time.time() < deadline - 50:
        h_bin = binary_landscaping(best_h, n, time.time() + 30, rng, alpha=2000000, maxiter=2000)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
            # Polish
            h_opt = lbfgsb_optimize(best_h, n, 100000000, time.time() + 20, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 7: Try different resolutions
    for n_try in [600, 750]:
        if time.time() > deadline - 60:
            break
        remaining = deadline - 30 - time.time()
        if remaining < 30:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        # Optimize at new resolution
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, time.time() + min(20, remaining * 0.3), maxiter=1000)
        h_opt = lbfgsb_optimize(h_opt, n_try, 50000000, time.time() + min(15, remaining * 0.2), maxiter=1500)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Final intensive local search
    if time.time() < deadline - 15:
        n_final = len(best_h)
        ls_deadline = min(deadline - 5, time.time() + 40)
        h_opt, c_opt = targeted_local_search(best_h, n_final, ls_deadline, top_k=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 9: Final subgradient + local search
    if time.time() < deadline - 10:
        n_final = len(best_h)
        sg_deadline = min(deadline - 5, time.time() + 20)
        h_opt, c_opt = subgradient_optimize(best_h, n_final, sg_deadline, top_m=2, initial_step=0.002)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        
        if time.time() < deadline - 3:
            h_opt, c_opt = targeted_local_search(best_h, n_final, deadline - 2, top_k=10)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 10: Ultra-fine final polish
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 1000000000, deadline - 2, maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
