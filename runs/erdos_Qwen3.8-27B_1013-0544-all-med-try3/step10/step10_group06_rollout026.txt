    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e6):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,

                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:

def directed_descent(h, n, deadline, rng, top_k=3):
    """Use gradient of the worst lag to guide moves, with line search."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_all_lags(h, n)
    center = n - 1
    no_improve = 0

    while time.time() < deadline:
        current_max = c_all.max()
        sorted_idx = np.argsort(c_all)[::-1]
        top_lags = []
        for s in sorted_idx:
            lag = s - center
            if abs(lag) >= 1 and abs(lag) < n:
                top_lags.append(s)
            if len(top_lags) >= top_k:
                break

        # Compute gradient of the worst lag's overlap
        k_star = top_lags[0]
        lag = k_star - center
        m = abs(lag)

        # Gradient of g_k(h) = sum h[i]*(1-h[i+k]) w.r.t. h
        grad_k = np.zeros(n)
        if lag > 0:
            for i in range(lag, n):
                grad_k[i] += (1.0 - h[i - lag])
                grad_k[i - lag] += (-h[i])
        else:
            idx = -lag
            for i in range(n - idx):
                grad_k[i] += (1.0 - h[i + idx])
                grad_k[i + idx] += (-h[i])

        # We want to decrease g_k, so move in direction -grad_k
        # But also respect [0,1] bounds and sum constraint
        direction = -grad_k
        # Zero out where we can't move
        for i in range(n):
            if h[i] <= 1e-10 and direction[i] > 0:
                direction[i] = 0
            if h[i] >= 1.0 - 1e-10 and direction[i] < 0:
                direction[i] = 0

        norm_dir = np.sqrt(np.sum(direction**2))
        if norm_dir < 1e-12:
            no_improve += 1
            if no_improve > 10:
                h = best_h.copy()
                idx = rng.choice(n, size=5, replace=False)
                h[idx] += rng.normal(0, 0.02, 5)
