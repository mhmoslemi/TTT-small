def swap_delta_simple(h, i, j, n):
    """Compute change in raw correlate(h, 1-h) when h[i]:1->0, h[j]:0->1"""
    nm1 = n - 1
    one_minus_h = 1.0 - h
    delta = np.zeros(2*n - 1)
    
    # Term: -(1-h_new)[i+shift] where h_new differs from h only at i and j
    # (1-h_new)[i+shift] = 1 - h_new[i+shift]
    # If i+shift == j: h_new[j] = 1, so 1-1 = 0
    # Else: 1 - h[i+shift]
    # Valid: 0 <= i+shift < n, i.e., -i <= shift <= n-1-i
    # k = shift + nm1, so k in [nm1-i, nm1-i+n-1] = [nm1-i, 2*nm1-i]
    lo_k = max(0, nm1 - i)
    hi_k = min(2*n-2, 2*nm1 - i)
    if lo_k <= hi_k:
        shifts = np.arange(lo_k, hi_k+1) - nm1  # actual shift values
        idx = i + shifts  # = i+shift
        vals = 1.0 - h[idx]
        # Check if any idx == j (i.e., shift == j-i)
        mask_j = (idx == j)
        if mask_j.any():
            vals[mask_j] = 0.0  # 1 - h_new[j] = 1-1 = 0
        delta[lo_k:hi_k+1] -= vals
    
    # Term: +(1-h_new)[j+shift]
    lo_k = max(0, nm1 - j)
    hi_k = min(2*n-2, 2*nm1 - j)
    if lo_k <= hi_k:
        shifts = np.arange(lo_k, hi_k+1) - nm1
        idx = j + shifts
        vals = 1.0 - h[idx]
        mask_i = (idx == i)
        if mask_i.any():
            vals[mask_i] = 1.0  # 1 - h_new[i] = 1-0 = 1
        delta[lo_k:hi_k+1] += vals
    
    # Term: +h[i-shift] (uses OLD h values)
    # Valid: 0 <= i-shift < n, i.e., i-k ≤ shift ≤ i, k in [i-nm1, i+nm1]
    lo_k = max(0, i - nm1)
    hi_k = min(2*n-2, i + nm1)
    if lo_k <= hi_k:
        shifts = np.arange(lo_k, hi_k+1) - nm1
        idx = i - shifts
        delta[lo_k:hi_k+1] += h[idx]
    
    # Term: -h[j-shift] (uses OLD h values)
    lo_k = max(0, j - nm1)
    hi_k = min(2*n-2, j + nm1)
    if lo_k <= hi_k:
        shifts = np.arange(lo_k, hi_k+1) - nm1
        idx = j - shifts
        delta[lo_k:hi_k+1] -= h[idx]
    
    return delta

def swap_delta(h, i, j, n):
    nm1 = n - 1
    o1 = 1.0 - h
    delta = np.zeros(0)
    delta = np.zeros(2 * n - 1)
    lo1 = max(0, nm1 - i)
    hi1 = min(2 * n - 2, 2 * nm1 - i)
    if lo1 <= hi1:
        idx = i + np.arange(lo1, hi1 + 1) - nm1
        vals = o1[idx].copy()
        mj = (idx == j)
        if mj.any():
            vals[mj] = 0.0
        delta[lo1:hi1 + 1] -= vals
    lo2 = max(0, nm1 - j)
    hi2 = min(2 * n - 2, 2 * nm1 - j)
    if lo2 <= hi2:
        idx = j + np.arange(lo2, hi2 + 1) - nm1
        vals = o1[idx].copy()

        if mi.any():

        delta[lo2:hi2 + 1] += vals
    lo3 = max(0, i - nm1)
    hi3 = min(2 * n - 2, i + nm1)

        idx = i - shifts

    lo4 = max(0, j - nm1)

def local_search_exact(h, n, deadline, rng, max_sweeps=5000, sample_per_sweep=8000):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    raw_c = np.correlate(h, 1.0 - h, mode='full')
    best_raw = raw_c.max()
    best_c = best_raw * dx
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    stuck = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones_list = list(ones)
        zeros_list = list(zeros)
        np_pairs = min(len(ones_list), len(zeros_list))
        if np_pairs == 0:
            break
        n_try = min(sample_per_sweep, len(ones_list) * len(zeros_list))
        tried = set()
        found_swap = False
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            if i == j or (i, j) in tried or (j, i) in tried:
                continue
            tried.add((i, j))
            d = swap_delta(h, i, j, n)
            new_raw = raw_c + d
            new_max = new_raw.max()
            if new_max < best_raw - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                raw_c = new_raw
                best_raw = new_max
                best_c = best_raw * dx
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
                zeros.add(i)
                improved = True
                found_swap = True
                break
        if not improved:
            stuck += 1
            if stuck > 50:
                if stuck > 200:
                    break
        else:
            stuck = 0
    return h, best_c

        new_max = new_raw.max()
        de = new_max - cur_raw
        if de <= 0 or (T > 1e-8 and np.log(rng.uniform(1e-15, 1.0)) > de / T):
            h[i] = 0.0
            h[j] = 1.0
            raw_c = new_raw
            cur_raw = new_max
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_c = best_raw * dx
                best_h = h.copy()
    return best_h, best_c

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def fine_coord_descent(h, n, deadline, steps=None, max_rounds=300):
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < max_rounds and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            w = smooth_max_w(c_vec, 200000.0)
            grad = h_grad_from_w(w, best_h, n)
            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def multiresolve(h_src, n_src, n_dst):
    h_dst = np.zeros(n_dst)
    ratio = n_src / n_dst
    for j in range(n_dst):
        i0 = int(j * ratio)
        i1 = int((j + 1) * ratio)
        if i1 > n_src:
            i1 = n_src
        if i1 > i0:
            h_dst[j] = h_src[i0:i1].mean()
        else:
            h_dst[j] = h_src[min(i0, n_src - 1)]
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    random.seed(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    # Phase 1: Quick L-BFGS at parent (40s)
    p1 = t0 + 40
    h1, c1 = run_lbfgs(best_h, best_n, p1, alpha=20000, mu=0.01, maxit=1000)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    # Phase 2: Aggressive binary local search (250s)
    p2 = min(t0 + 290, deadline - 500)
    h_bin = binarize(best_h, best_n)
    h2, c2 = local_search_exact(h_bin, best_n, p2, rng, max_sweeps=8000, sample_per_sweep=10000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    # Phase 3: Simulated annealing (150s)
    p3 = min(t0 + 440, deadline - 350)
    if time.time() < p3:
        h3, c3 = simulated_annealing(best_h, best_n, p3, rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    # Phase 4: More local search after SA (120s)
    p4 = min(t0 + 560, deadline - 280)
    if time.time() < p4:
        h4, c4 = local_search_exact(best_h, best_n, p4, rng, max_sweeps=5000, sample_per_sweep=8000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    # Phase 5: L-BFGS continuous refinement (180s)
    p5 = min(t0 + 740, deadline - 180)
    if time.time() < p5:
        for alpha in [30000, 100000, 500000, 2000000]:
            if time.time() > p5 - 5:
                break
            rem = max(15, p5 - time.time())
            h5, c5 = run_lbfgs(best_h, best_n, time.time() + rem, alpha=alpha, mu=0.001, maxit=1500)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    # Phase 6: Fine coordinate descent (100s)
    p6 = min(t0 + 840, deadline - 100)
    if time.time() < p6:
        h6, c6 = fine_coord_descent(best_h, best_n, p6,
                                    steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                    max_rounds=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    # Phase 7: Final local search + SA (50s)
    p7 = deadline - 15
    if time.time() < p7:
        h7, c7 = local_search_exact(best_h, best_n, min(p7, time.time() + 30), rng, max_sweeps=3000, sample_per_sweep=5000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    if time.time() < deadline - 5:
        h8, c8 = simulated_annealing(best_h, best_n, deadline - 3, rng)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        # Final local search
        if time.time() < deadline - 1:
            h9, c9 = local_search_exact(best_h, best_n, deadline - 1, rng, max_sweeps=2000, sample_per_sweep=5000)
            if c9 < best_c:
                best_c = c9
                best_h = h9
    # Final polish
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
