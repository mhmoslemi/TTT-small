def compute_c5_vec(h, n):

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:

            nv = n - kk

            grad[0:nv] += wm * (-x[kk:n])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(max(np.sum(np.exp(sh - mx)), 1e-300))) / alpha
    else:
        val = m + np.log(max(np.sum(np.exp(sh)), 1e-300)) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(c) / len(c)
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(max(np.sum(np.exp(sh - smx)), 1e-300))) / alpha
    else:
        val = mx + np.log(max(np.sum(np.exp(sh)), 1e-300)) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(active, dtype=float) / len(active)
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)

        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)

                          bounds=bounds,

def targeted_local_search(h, n, deadline):

    best_h = project_feasible(h.copy(), n)

        while improved and time.time() < deadline:

            c_vec = compute_c5_vec(best_h, n)
            c_max = c_vec.max()
            # Focus on the lags near the maximum

            active_lags = np.where(c_vec >= threshold)[0]
            # Limit to top 20 lags

def projected_subgrad(h, n, deadline, n_active=10):
    """Projected subgradient on the exact max (using top-k subgradient)."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    eta = 0.02
    patience = 0
    max_patience = 500
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        # Select active set using softmax weighting
        sh = 5000.0 * (c_vec - c_max)
        weights = np.exp(sh)
        wsum = weights.sum()
        if wsum < 1e-300:
            wsum = 1.0
        weights /= wsum
        
        # Compute subgradient as weighted average over top n_active
        top_idx = np.argsort(-weights)[:n_active]
        top_w = weights[top_idx]
        tw = top_w.sum()
        if tw > 0:
            top_w /= tw
        
        grad = np.zeros(n)
        for wi, idx in zip(top_w, top_idx):
            if wi < 1e-12:
                continue
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += wi * (1.0 - best_h[k:n]) * dx
                grad[k:n] += wi * (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wi * (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += wi * (-best_h[kk:n]) * dx
        
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            patience = 0
            eta = min(eta * 1.01, 0.05)
        else:
            patience += 1
            eta *= 0.97
            if patience > max_patience:
                break
    
    return best_h, best_c

def make_binary_ips(h, n):
    """Create a binary start by keeping top n/2 values."""
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    rem = n / 2.0 - h_bin.sum()
    if abs(rem) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += rem
    return project_feasible(h_bin, n)

def periodic_start(n, period, pattern):
    """Create a periodic binary-like start."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(pattern[i % period])
    # Adjust sum
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def upsample(h_src, n_src, n_dst):
    """Linearly interpolate from n_src to n_dst points."""
    x_src = np.linspace(0, 2.0, n_src)
    x_dst = np.linspace(0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
