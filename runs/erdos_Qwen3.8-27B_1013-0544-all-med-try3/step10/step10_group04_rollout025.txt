import numpy as np
import time
from scipy.optimize import minimize

_DEADLINE = [0.0]
_LAST_X = [None]


class StopOptimization(Exception):
    pass


def _stop_callback(res):
    if time.perf_counter() > _DEADLINE[0]:
        _LAST_X[0] = getattr(res, "x", None)
        raise StopOptimization()


def make_rng(seed):
    return np.random.RandomState(int(seed) % 2147483647)


def even_int(x):
    x = int(x)
    if x < 32:
        x = 32
    if x % 2 != 0:
        x += 1
    return x


def next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_convolve_real(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    la = a.size
    lb = b.size
    n = next_power_of_two(la + lb - 1)
    fa = np.fft.rfft(a, n)
    fb = np.fft.rfft(b, n)
    out = np.fft.irfft(fa * fb, n)
    return out[:la + lb - 1]


def c_values_fft(h):
    h = np.asarray(h, dtype=float)
    v = 1.0 - h[::-1]
    return fft_convolve_real(h, v)


def score_fft(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = c_values_fft(h)
    m = float(np.max(c))
    if m < 0.0 and m > -1e-12:
        m = 0.0
    return m * dx


def score_exact(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(c))
    if m < 0.0 and m > -1e-12:
        m = 0.0
    return m * dx


def adjust_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    for _ in range(12):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            step = min(diff, float(1.0 - h[idx]) + 1e-15)
        else:
            idx = int(np.argmin(h))
            step = max(diff, -float(h[idx]) - 1e-15)
        if step == 0.0:
            break
        h[idx] += step
    return np.clip(h, 0.0, 1.0)


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        x = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, x.size), x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

    target = 0.5 * float(n)

    h = np.clip(x, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) < 1e-13:
        return adjust_sum(h, n)

    if abs(diff) < 0.75:
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s > diff:
                h2 = h + (diff / s) * room
                return adjust_sum(np.clip(h2, 0.0, 1.0), n)
        else:
            s = float(h.sum())
            if s > -diff:
                h2 = h - ((-diff) / s) * h
                return adjust_sum(np.clip(h2, 0.0, 1.0), n)

    y = x
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    if not np.isfinite(lo):
        lo = 0.0
    if not np.isfinite(hi):
        hi = 1.0
    if hi <= lo:
        h = np.full(n, 0.5)
        return adjust_sum(h, n)

    for _ in range(64):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(y - mid, 0.0, 1.0).sum())
        if val > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    h = np.clip(y - lam, 0.0, 1.0)
    return adjust_sum(h, n)


def interpolate_to(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    if n_new == n_old:
        return project_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def downsample_best(h_old, n_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    if n_new >= n_old:
        return interpolate_to(h_old, n_old, n_new)
    if n_old % n_new == 0:
        k = n_old // n_new
        h_new = h_old.reshape(n_new, k).mean(axis=1)
    else:
        h_new = interpolate_to(h_old, n_old, n_new)
    return project_feasible(h_new, n_new)


def random_binary_start(n, seed):
    n = even_int(n)
    rng = make_rng(seed)
    target = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[:target]
    h[idx] = 1.0
    return project_feasible(h, n)

    order = np.argsort(-h)
    out = np.zeros(n, dtype=float)

def logsumexp_stable(a):
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        return 0.0
    m = float(np.max(a))
    return m + float(np.log(np.sum(np.exp(a - m))))


def smooth_obj(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = interpolate_to(x, x.size, n)
    if not np.all(np.isfinite(x)):
        return 1e100

    dx = 2.0 / float(n)
    c = c_values_fft(x) * dx
    m = float(np.max(c))
    if m < 0.0 and m > -1e-12:
        m = 0.0

    sh = alpha * (c - m)
    lse = logsumexp_stable(sh)
    diff = float(x.sum() - 0.5 * n)
    return float(m + lse / alpha + mu * diff * diff)


def smooth_grad(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = interpolate_to(x, x.size, n)

    dx = 2.0 / float(n)
    c = c_values_fft(x) * dx
    m = float(np.max(c))
    if m < 0.0 and m > -1e-12:
        m = 0.0

    sh = alpha * (c - m)
    mx = float(np.max(sh))
    w = np.exp(sh - mx)
    ws = float(w.sum())
    if ws <= 0.0 or not np.isfinite(ws):
        w = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
    else:
        w /= ws

    if n == 1:
        diff = float(x.sum() - 0.5)
        return dx * (1.0 - 2.0 * x) + 2.0 * mu * diff

    center = n - 1
    A = w[center - 1::-1]
    B = w[center + 1:]
    qval = A + B

    prefA = np.empty(n, dtype=float)
    prefB = np.empty(n, dtype=float)
    prefA[0] = 0.0
    prefB[0] = 0.0
    if n > 1:
        prefA[1:] = np.cumsum(A)
        prefB[1:] = np.cumsum(B)

    j = np.arange(n)
    lin = prefA[n - 1 - j] + prefB[j]

    q = np.zeros(2 * n - 1, dtype=float)
    q[0:center] = qval[::-1]
    q[center + 1:] = qval

    qconv = fft_convolve_real(x, q)
    qconv = qconv[center:center + n]

    grad_raw = w[center] * (1.0 - 2.0 * x) + lin - qconv
    diff = float(x.sum() - 0.5 * n)
    grad = dx * grad_raw
    if abs(diff) > 1e-15:
        grad = grad + (2.0 * mu * diff)
    return grad

def lbfgs_optimize(h0, n, deadline):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = score_fft(h0)
    start_h = h0.copy()
    start_c = best_c
    bounds = [(0.0, 1.0)] * n
    configs = config_for_n(n)
    opts = lbfgs_options(n)

    for alpha, mu in configs:
        now = time.perf_counter()
        if now >= deadline - 4.0:
            break

        _DEADLINE[0] = deadline
        _LAST_X[0] = start_h.copy()

        try:
            res = minimize(
                smooth_obj,
                start_h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                callback=_stop_callback,
                options=opts,
            )
            if res is None or not hasattr(res, "x") or res.x is None:
                continue
            cand = project_feasible(res.x, n)
        except StopOptimization:
            if _LAST_X[0] is not None:
                cand = project_feasible(_LAST_X[0], n)
            else:
                cand = best_h
            cc = score_fft(cand)
            if cc < best_c:
                best_c = cc
                best_h = cand.copy()
            return best_h, best_c
        except Exception:
            continue

        cc = score_fft(cand)
        if not np.isfinite(cc):
            continue

        if cc < best_c:
            best_c = cc
            best_h = cand.copy()
            start_h = cand.copy()
            start_c = cc
        elif cc < best_c + 1e-7:
            start_h = cand.copy()
            start_c = cc
        else:
            start_h = best_h.copy()
            start_c = best_c

    return best_h, best_c

def projected_gradient_polish(h, n, deadline, alpha=None):
    h = project_feasible(h, n)
    if alpha is None:
        alpha = polish_alpha(n)
    step0 = polish_step0(n)

    best_h = h.copy()
    best_c = score_fft(h)
    cur_h = h.copy()
    cur_c = best_c
    step = step0
    stall = 0
    mults = (1.0, 0.5, 0.25, 0.125, 0.0625, 2.0)

    while time.perf_counter() < deadline:
        g = smooth_grad(cur_h, n, alpha, 0.0)
        gnorm = float(np.linalg.norm(g))
        if not np.isfinite(gnorm) or gnorm < 1e-13:
            break

        base = step / gnorm
        trial_h = None
        trial_c = cur_c

        for mult in mults:
            if time.perf_counter() >= deadline:
                break
            cand = cur_h - (base * mult) * g
            cand = project_feasible(cand, n)
            cc = score_fft(cand)
            if cc < trial_c - 1e-14:
                trial_c = cc
                trial_h = cand

        if trial_h is not None:
            cur_h = trial_h
            cur_c = trial_c
            stall = 0
            if step < 0.12:
                step = min(0.12, step * 1.12)
            if cur_c < best_c - 1e-13:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            step *= 0.5
            stall += 1
            if stall > 10 and step < 1e-5:
                break

    return best_h, best_c


def make_phase_deadline(now, deadline, fraction, max_sec, reserve=25.0):
    rem = deadline - now
    if rem <= reserve:
        return deadline
    allowed = min(float(max_sec), max(0.5 * reserve, float(fraction) * rem))
    rem_after = rem - allowed
    if rem_after < reserve:
        allowed = max(0.0, rem - reserve)
    return now + min(allowed, rem)

def optimize_n_with_starts(h_global, n_global, n, phase_deadline, seed, parent_h, parent_n):

    now = time.perf_counter()

        h1, c1 = lbfgs_optimize(h_loc, n, first_dl)
        if c1 < best_loc_c - 1e-13:
            best_loc_c = c1
            best_loc_h = h1.copy()

    trials = []
    if parent_h is not None and parent_n > 0:
        trials.append(downsample_best(parent_h, parent_n, n))

    if n <= 160:
        trials.append(random_binary_start(n, seed + 101 * n + 1))
        trials.append(random_binary_start(n, seed + 101 * n + 2))
        trials.append(mirror_start(best_loc_h, n))
        trials.append(complement_start(best_loc_h, n))
        trials.append(perturb_start(best_loc_h, n, 0.06, seed + 101 * n + 3))
        trials.append(threshold_by_rank(best_loc_h, n))
    elif n <= 512:
        trials.append(mirror_start(best_loc_h, n))
        trials.append(complement_start(best_loc_h, n))
        trials.append(perturb_start(best_loc_h, n, 0.04, seed + 101 * n + 3))
        trials.append(perturb_start(best_loc_h, n, 0.09, seed + 101 * n + 4))
        trials.append(random_binary_start(n, seed + 101 * n + 5))
        trials.append(threshold_by_rank(best_loc_h, n))
    else:
        trials.append(mirror_start(best_loc_h, n))
        trials.append(complement_start(best_loc_h, n))
        trials.append(perturb_start(best_loc_h, n, 0.02, seed + 101 * n + 3))
        trials.append(perturb_start(best_loc_h, n, 0.05, seed + 101 * n + 4))
        trials.append(threshold_by_rank(best_loc_h, n))

    if n <= 128:
        max_trials = 5
    elif n <= 256:
        max_trials = 4
    elif n <= 600:
        max_trials = 3
    else:
        max_trials = 2

    trials = trials[:max_trials]

    n_trials = len(trials)
    for i, h0 in enumerate(trials):
        now = time.perf_counter()
        if now >= phase_deadline - 18.0:
            break
        rem = max(0.0, phase_deadline - 18.0 - now)
        per = max(8.0, rem / float(max(1, n_trials - i)))
        td = min(phase_deadline - 12.0 - max(0, (n_trials - i - 1)) * 3.0, now + per)
        if td <= now + 4.0:
            break
        h1, c1 = lbfgs_optimize(h0, n, td)
        if c1 < best_loc_c - 1e-13:
            best_loc_c = c1
            best_loc_h = h1.copy()

    if time.perf_counter() < phase_deadline - 4.0:
        h_pol, c_pol = projected_gradient_polish(best_loc_h, n, phase_deadline - 0.5)
        if c_pol < best_loc_c - 1e-13:
            best_loc_c = c_pol
            best_loc_h = h_pol.copy()

    c_exact = score_exact(best_loc_h)
    return best_loc_h, c_exact


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    budget_s = float(budget_s)
    if budget_s < 20.0:
        budget_s = 20.0

    deadline = t0 + max(10.0, budget_s) - 8.0
    seed = int(seed)

    try:
        h_par = np.asarray(initial_h_values, dtype=float).ravel()
        if h_par.size < 8:
            raise ValueError("bad parent")
    except Exception:
        h_par = random_binary_start(400, seed + 17)

    n_par = h_par.size
    if n_par > 2200:
        n_par_new = 1600
        h_par = interpolate_to(h_par, n_par, n_par_new)
        n_par = n_par_new
    if n_par % 2 != 0:
        n_par_new = n_par + 1
        h_par = interpolate_to(h_par, n_par, n_par_new)
        n_par = n_par_new

    parent_h = project_feasible(h_par, n_par)
    parent_n = n_par

    best_h = parent_h.copy()
    best_n = parent_n
    best_c = score_exact(best_h)

    reserve = 25.0

    # Parent/base resolution phase.
    now = time.perf_counter()
    if now < deadline - reserve - 25.0:
        end = make_phase_deadline(now, deadline, 0.24, 420.0, reserve)
        if end > now + 20.0:
            h_try, c_try = optimize_n_with_starts(best_h, best_n, best_n, end, seed, None, 0)
            if c_try < best_c:
                best_c = c_try
                best_n = len(h_try)
                best_h = h_try

    # Mid-low resolution search.
    now = time.perf_counter()
    if now < deadline - reserve - 25.0:
        n_task = 256
        end = make_phase_deadline(now, deadline, 0.15, 260.0, reserve)
        if end > now + 18.0:
            h_try, c_try = optimize_n_with_starts(best_h, best_n, n_task, end, seed + 5000, parent_h, parent_n)
            if c_try < best_c:
                best_c = c_try
                best_n = len(h_try)
                best_h = h_try

    # Low resolution search.
    now = time.perf_counter()
    if now < deadline - reserve - 25.0:
        n_task = 128
        if n_task != best_n:
            end = make_phase_deadline(now, deadline, 0.11, 220.0, reserve)
            if end > now + 18.0:
                h_try, c_try = optimize_n_with_starts(best_h, best_n, n_task, end, seed + 9000, parent_h, parent_n)
                if c_try < best_c:
                    best_c = c_try
                    best_n = len(h_try)
                    best_h = h_try

    # Higher resolution from the current best.
    now = time.perf_counter()
    if now < deadline - reserve - 30.0:
        n_high = choose_high_n(best_n)
        end = make_phase_deadline(now, deadline, 0.34, 520.0, reserve)
        if end > now + 25.0:
            h_try, c_try = optimize_n_with_starts(best_h, best_n, n_high, end, seed + 13000, parent_h, parent_n)
            if c_try < best_c:
                best_c = c_try
                best_n = len(h_try)
                best_h = h_try

    # Very low resolution opportunistic pass.
    now = time.perf_counter()
    if now < deadline - reserve - 25.0:
        n_task = 64
        if n_task != best_n:
            end = make_phase_deadline(now, deadline, 0.05, 120.0, reserve)
            if end > now + 12.0:
                h_try, c_try = optimize_n_with_starts(best_h, best_n, n_task, end, seed + 20000, parent_h, parent_n)
                if c_try < best_c:
                    best_c = c_try
                    best_n = len(h_try)
                    best_h = h_try

    # Final local polish on the best candidate.
    now = time.perf_counter()
    if now < deadline - 4.0:
        h_fin, c_fin = projected_gradient_polish(best_h, best_n, deadline - 0.5)
        c_fin = score_exact(h_fin)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
            best_n = len(h_fin)

    best_h = project_feasible(best_h, best_n)
    best_c = score_exact(best_h)
    return best_h, float(best_c), int(len(best_h))
