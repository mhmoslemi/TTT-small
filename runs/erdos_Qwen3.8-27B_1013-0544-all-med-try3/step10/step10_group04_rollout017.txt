```python
    # Compute the linear coefficients for f_k(t) where t = h[i]
    # f_k(t) = a_k + b_k * t
    # b_k = sum of contributions: (1-h[i+k]) if i+k valid + (-h[i-k]) if i-k valid
    # a_k = f_k(h) - h[i] * b_k
    
    b = np.zeros(2*n-1)
    dx = 2.0 / n
    center = n - 1
    
    # For shift k (0-indexed from -(n-1) to n-1, stored at index k+n-1):
    # f_k = sum_j h[j](1-h[j+k])*dx
    # Contribution of h[i] to f_k:
    #   - term j=i: h[i]*(1-h[i+k])*dx if 0 ≤ i+k < n
    #   - term j=i-k: h[i-k]*(1-h[i])*dx if 0 ≤ i-k < n
    
    for k in range(-(n-1), n):
        ki = k + center  # index in correlation array
        b_k = 0.0
        if 0 <= i + k < n:
            b_k += (1.0 - h[i+k]) * dx
        if k != 0 and 0 <= i - k < n:
            b_k += (-h[i-k]) * dx
        b[ki] = b_k
    
    # a_k = f_k(h) - h[i]*b_k
    a = c_vec - h[i] * b
    
    # Minimize max_k (a_k + b_k * t) over t ∈ [0,1]
    # This is a 1D convex piecewise linear minimization
    # Use scipy for simplicity
    from scipy.optimize import minimize_scalar
    def f_1d(t):
        return float(np.max(a + b * t))
    
    res = minimize_scalar(f_1d, bounds=(0, 1), method='bounded',
                          options={'xatol': 1e-12})
    return res.x
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500.0:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def grad_for_shift(h, n, k):
    """Gradient of f_k = sum_j h[j](1-h[j+k])*dx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def subgradient_method(h, n, deadline, top_k=3, lr_init=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    c_vec = compute_c5_vec(h, n)
    for i in range(500000):
        if time.time() > deadline:
            break
        # Find top-k shifts
        top_idx = np.argpartition(-c_vec, top_k - 1)[:top_k]
        # Average gradient of top-k shifts
        grad_avg = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            grad_avg += grad_for_shift(h, n, k)
        grad_avg /= top_k
        # Step
        h_new = project_feasible(h - lr * grad_avg, n)
        c_new = compute_c5_vec(h_new, n)
        if c_new.max() < best_c:
            best_c = float(c_new.max())
            best_h = h_new.copy()
            c_vec = c_new
        else:
            # Try smaller step
            h_new2 = project_feasible(h - lr * 0.1 * grad_avg, n)
            c_new2 = compute_c5_vec(h_new2, n)
            if c_new2.max() < best_c:
                best_c = float(c_new2.max())
                best_h = h_new2.copy()
                h = h_new2
                c_vec = c_new2
                lr = max(lr * 0.95, 1e-7)
            else:
                lr = max(lr * 0.8, 1e-7)
                if lr < 1e-7:
                    break
        if i % 100 == 0:

    return best_h, best_c

        except Exception:

    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500.0:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def obj_jac(x, na, al):
        return softmax_grad(x, na, al, 0.0)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), jac=lambda x, na, al: obj_jac(x, na, al),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500.0:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def grad_for_shift(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def subgradient_method(h, n, deadline, top_k=3, lr_init=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    c_vec = compute_c5_vec(h, n)
    stall_count = 0
    for i in range(1000000):
        if time.time() > deadline:
            break
        top_idx = np.argpartition(-c_vec, min(top_k, 2*n-2))[:top_k]
        grad_avg = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            grad_avg += grad_for_shift(h, n, k)
        grad_avg /= top_k
        h_new = project_feasible(h - lr * grad_avg, n)
        c_new = compute_c5_vec(h_new, n)
        c_new_max = float(c_new.max())
        if c_new_max < best_c - 1e-15:
            best_c = c_new_max
            best_h = h_new.copy()
            c_vec = c_new
            h = h_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                lr *= 0.5
                stall_count = 0
                if lr < 1e-8:
                    break
            h_new2 = project_feasible(h - lr * 0.01 * grad_avg, n)
            c_new2 = compute_c5_vec(h_new2, n)
            if float(c_new2.max()) < best_c:
                best_c = float(c_new2.max())
                best_h = h_new2.copy()
                c_vec = c_new2
                h = h_new2

                lr *= 0.95
                if lr < 1e-8:
                    break

    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': 1000000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=20000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500.0:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

            h_new = project_feasible(res.x, n)

            if c_new < best_c:
```
