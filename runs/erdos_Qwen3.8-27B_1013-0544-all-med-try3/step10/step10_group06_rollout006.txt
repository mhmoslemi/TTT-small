import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = w / ws
    return w

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag]) * dx
            grad[0:n - lag] += w[k] * (-h[lag:n]) * dx
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n]) * dx
            grad[idx:n] += w[k] * (-h[0:n - idx]) * dx
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 80,
                                'iprint': -1, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
