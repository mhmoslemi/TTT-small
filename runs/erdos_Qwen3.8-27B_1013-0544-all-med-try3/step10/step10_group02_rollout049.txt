import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        logsum = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        logsum = np.log(np.sum(np.exp(sh))) / alpha
    val = m + logsum
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def focused_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        logsum = (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        logsum = np.log(np.sum(np.exp(sh))) / alpha
    val = mx + logsum
    # Penalty for non-active shifts exceeding current max
    all_max = c.max()
    if all_max > mx:
        val += 10.0 * (all_max - mx)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def focused_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    # Add gradient for non-active max penalty
    all_mv = c.max()
    if all_mv > mx:
        non_active_max_idx = np.argmax(c)
        if non_active_max_idx not in active:
            k = non_active_max_idx - (n - 1)
            wm = 10.0 * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, maxit_list=None, label=""):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-19,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 60, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_focused(h, n, deadline, top_k_list, alphas):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for tk in top_k_list:
        for alpha in alphas:
            remaining = deadline - time.time()
            if remaining < 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > tk:
                active = np.argsort(-c_vec[active])[:tk]
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            maxit = max(500, int(remaining * 2))
            try:
                res = minimize(focused_obj, best_h, jac=focused_grad,
                              args=(n, active, alpha, 0.002), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-19,
                                       'gtol': 1e-16, 'maxfun': maxit * 10,
                                       'maxls': 60, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.002, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try +d to i, -d from j
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        h_new = project_feasible(h, n)
                        c = compute_c5(h_new, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_new
                            improved = True
                            h = h_new
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try -d from i, +d to j
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        h_new = project_feasible(h, n)
                        c = compute_c5(h_new, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_new
                            improved = True
                            h = h_new
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def binary_sa(h, n, deadline, n_iter=50000, seed=123):
    """Simulated annealing on binary structure (with one fractional entry)"""
    rng = np.random.RandomState(seed)
    h_bin = project_feasible(h.copy(), n)
    # Make it more binary
    sorted_idx = np.argsort(-h_bin)
    n_ones = n // 2
    h_cur = np.zeros(n)
    h_cur[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_cur.sum()
    if remaining > 0:
        h_cur[sorted_idx[n_ones]] = remaining
    elif remaining < 0:
        h_cur[sorted_idx[n_ones - 1]] += remaining
    h_cur = project_feasible(h_cur, n)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    T = 0.001
    T_end = 1e-6
    decay = (T_end / T) ** (1.0 / max(n_iter, 1))
    ones_pos = np.where(h_cur >= 0.5)[0]
    zeros_pos = np.where(h_cur < 0.5)[0]
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]
        if i == j:
            continue
        # Swap: set i to 0, j to 1
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        h_try = project_feasible(h_cur, n)
        c_try = compute_c5(h_try, n)
        delta = c_try - compute_c5(h_cur, n)
        if delta < 0 or rng.rand() < np.exp(-max(delta, -1) / max(T, 1e-10)):
            h_cur = h_try
            ones_pos = np.where(h_cur >= 0.5)[0]
            zeros_pos = np.where(h_cur < 0.5)[0]
            c_cur = c_try
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
        T *= decay
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    """Structure-preserving upsampling using cubic-like interpolation"""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    # Add small smoothing to preserve structure
    h_new = project_feasible(h_new, n_new)
    return h_new

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    def sq_jac(x, nn):
        return np.ones(nn)
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
