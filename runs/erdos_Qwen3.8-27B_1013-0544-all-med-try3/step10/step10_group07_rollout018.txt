    """Project h onto {x: sum(x)=n/2, 0<=x<=1}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    # Bisection on tau: find tau s.t. sum(clip(h-tau, 0, 1)) = target

        if s > target:

                    a, b = oi[rng.randint(len(oi))], zi[rng.randint(len(zi))]
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        
        cur_c = float(corr.max())
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            peak_idx = int(np.argmax(corr))
            k = peak_idx - (n - 1)
            
            if k >= 0:
                va = np.arange(0, n - k)
            else:
                va = np.arange(-k, n)
            cz = va[(h[va] == 0) & (h[va + k] == 1)]
            
            if len(cz) > 0:
                perm = rng.permutation(len(cz))
                for pi in perm[:60]:
                    i = cz[pi]
                    jt = i + k
                    for _ in range(8):
                        j = ones[rng.randint(len(ones))]
                        if j == jt or j == i:
                            continue
                        delta = np.zeros(2 * n - 1)
                        im = i + shifts
                        m1 = (im >= 0) & (im < n)
                        delta[m1] += 1.0 - h[im[m1]]
                        m2 = (i - shifts >= 0) & (i - shifts < n)
                        delta[m2] -= h[i - shifts[m2]]
                        jm = j + shifts
                        m3 = (jm >= 0) & (jm < n)
                        delta[m3] -= 1.0 - h[jm[m3]]
                        m4 = (j - shifts >= 0) & (j - shifts < n)
                        delta[m4] += h[j - shifts[m4]]
                        nc = float((corr + delta).max())
                        if nc < cur_c - 1e-12:
                            h[i], h[j] = 1.0, 0.0
                            corr = corr + delta
                            cur_c = nc
                            ones.remove(j)
                            ones.append(i)
                            zeros.remove(i)
                            zeros.append(j)
                            improved = True
                            break
                    if improved:
                        break
            
            if not improved:
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                delta = np.zeros(2 * n - 1)
                im = i + shifts
                m1 = (im >= 0) & (im < n)
                delta[m1] += 1.0 - h[im[m1]]
                m2 = (i - shifts >= 0) & (i - shifts < n)
                delta[m2] -= h[i - shifts[m2]]
                jm = j + shifts
                m3 = (jm >= 0) & (jm < n)
                delta[m3] -= 1.0 - h[jm[m3]]
                m4 = (j - shifts >= 0) & (j - shifts < n)
                delta[m4] += h[j - shifts[m4]]
                nc = float((corr + delta).max())
                if nc < cur_c - 1e-12:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
        
        if cur_c * dx < best_c:
            best_c = cur_c * dx
            best_h = h.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: Binary local search (20% of budget)
    p1 = t0 + 0.20 * (budget_s - 3.0)
    hb, cb = binary_search_fast(best_h, n0, p1, max_sweeps=3000, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Gradient descent with annealed alpha (50% of budget)
    p2 = t0 + 0.65 * (budget_s - 3.0)
    # Start from a slightly smoothed version of the binary solution
    h_smooth = project_feasible(0.5 + 0.5 * (2.0 * best_h - 1.0) + 0.03 * rng.randn(n0), n0)
    alphas = [3.0, 8.0, 20.0, 50.0, 150.0, 500.0, 1500.0, 5000.0]
    iters = [150, 150, 200, 200, 200, 200, 150, 100]
    gh, gc = gradient_descent(h_smooth, n0, p2, alphas, iters, seed=seed)
    if gc < best_c:
        best_c = gc
        best_h = gh.copy()
    
    # Also run gradient descent from the binary solution directly
    if time.time() < p2 - 10:
        gh2, gc2 = gradient_descent(best_h.copy(), n0, p2 - 10, [10.0, 50.0, 200.0, 1000.0], [150, 150, 150, 100], seed=seed + 1)
        if gc2 < best_c:
            best_c = gc2
            best_h = gh2.copy()
    
    # Phase 3: Binary search from gradient result + multi-resolution (20%)
    p3 = t0 + 0.83 * (budget_s - 3.0)
    # Binary polish of current best
    hb3, cb3 = binary_search_fast(best_h, n0, t0 + 0.75 * (budget_s - 3.0), max_sweeps=500, n_restarts=2, seed=seed + 10)
    if cb3 < best_c:
        best_c = cb3
        best_h = hb3.copy()
    
    # Multi-resolution
    if time.time() < p3 - 20:
        for nh in [200, 300, 350, 400]:
            if time.time() > p3 - 15:
                break
            if nh == n0:
                continue
            xo = np.linspace(0, 2, n0, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh, ch = gradient_descent(hh, nh, min(time.time() + 40, p3 - 10), [10.0, 50.0, 200.0], [100, 100, 80], seed=seed + 100)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n0 = nh
    
    # Phase 4: Final polish (12%)
    p4 = deadline
    if time.time() < p4 - 15:
        h_fin = project_feasible(best_h + 0.01 * rng.randn(len(best_h)), len(best_h))
        nf = len(best_h)
        hf, cf = gradient_descent(h_fin, nf, p4 - 10, [50.0, 200.0, 1000.0, 5000.0], [100, 100, 80, 60], seed=seed + 200)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    if time.time() < deadline - 5:
        hb4, cb4 = binary_search_fast(best_h, len(best_h), deadline - 2, max_sweeps=300, n_restarts=2, seed=seed + 300)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
