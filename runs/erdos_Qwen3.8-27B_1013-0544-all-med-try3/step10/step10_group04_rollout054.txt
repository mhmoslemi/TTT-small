import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_grad(x, n, k_top, mu):

    c = compute_c5_vec(x, n)
    idx = np.argsort(-c)[:k_top]
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    w_sum = 0.0
    for i in idx:
        w = c[i] ** 2  # weight by square of value
        w_sum += w
    for i in idx:
        k = i - center
        wm = (c[i]**2 / w_sum) * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk

                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 100.0), (1000, 500.0), (5000, 2000.0),
                   (10000, 5000.0), (50000, 20000.0)]
    for alpha, mu in configs:

            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new.copy()

            contrib = np.where((h[-k_w:] > 0.5) & (h[:-k_w] < 0.5))[0]
        else:
            contrib = np.array([], dtype=int)
        if len(contrib) == 0:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = contrib[rng.randint(len(contrib))]
            zeros = np.where(h < 0.5)[0]
            if i in zeros:
                continue
            j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        if k_w >= 0:
            nv = n - k_w
            c_new_k = float(np.sum(h[:nv] * (1 - h[k_w:])) * (2.0 / n))
        elif k_w < 0:
            nv = n + k_w
            c_new_k = float(np.sum(h[-k_w:] * (1 - h[:-k_w])) * (2.0 / n))
        else:
            c_new_k = 0.0
        if c_new_k < c_vec[imax] * 0.95:
            c_vec = compute_c5_vec(h, n)
        else:
            c_local = compute_c5_vec(h, n)
            if c_local.max() > best_c + 0.005:
                h[i], h[j] = h[j], h[i]
                continue
            c_vec = c_local
    return best_h, best_c

def upscale(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 150000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst_idx = int(np.argmax(c_vec))
        k_w = worst_idx - (n - 1)
        if k_w >= 0:
            pairs = [(i, i + k_w) for i in range(n - k_w)]
        elif k_w < 0:
            pairs = [(i, i + k_w) for i in range(-k_w, n)]
        else:
            pairs = []
        improved = False
        for (i, j) in pairs[:200]:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.003, -0.003, 0.001, -0.001, 0.005, -0.005]:
                nv = best_h[i] + delta
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                best_h[i] = nv
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = best_h[i] - delta if delta > 0 else best_h[i] + delta
                    best_h[i] = max(0.0, min(1.0, best_h[i]))
            if improved:
                break
        if not improved and rnd > 30:
            break
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = thue_morse_gen(512)
        n_base = 512

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: Explore structured and random candidates (50s)
    t1 = min(t0 + 50, deadline - 950)
    if t1 > time.time() + 5:
        rng = np.random.RandomState(seed + 100)
        candidates = [(best_c, best_h.copy(), n_base)]
        for ns in [256, 512, 1024]:
            if time.time() > t1:
                break
            tm = thue_morse_gen(ns)
            if tm.sum() != ns / 2:
                ones_idx = np.where(tm == 1)[0]
                zeros_idx = np.where(tm == 0)[0]
                diff = int(ns / 2 - tm.sum())
                if diff > 0:
                    pick = rng.choice(zeros_idx, min(diff, len(zeros_idx)), replace=False)
                    tm[pick] = 1
                elif diff < 0:
                    pick = rng.choice(ones_idx, min(-diff, len(ones_idx)), replace=False)
                    tm[pick] = 0
            tm = project_feasible(tm, ns)
            c_tm = compute_c5(tm, ns)
            candidates.append((c_tm, tm, ns))
        for _ in range(300):
            if time.time() > t1:
                break
            hr = (rng.rand(n_base) < 0.5).astype(float)
            if hr.sum() != n_base / 2:
                ones = np.where(hr == 1)[0]
                zeros = np.where(hr == 0)[0]
                diff = int(n_base / 2 - hr.sum())
                if diff > 0:
                    p = rng.choice(zeros, min(diff, len(zeros)), replace=False)
                    hr[p] = 1.0
                else:
                    p = rng.choice(ones, min(-diff, len(ones)), replace=False)
                    hr[p] = 0.0
            c_r = compute_c5(hr, n_base)
            candidates.append((c_r, hr, n_base))
        candidates.sort(key=lambda x: x[0])
        if candidates[0][0] < best_c:
            best_c = candidates[0][0]
            best_h = candidates[0][1].copy()

    # Phase 2: Discrete swap search at current resolution (70s)
    t2 = min(t0 + 120, deadline - 900)
    if t2 > time.time() + 10:
        n_cur = len(best_h)
        h_01 = (best_h > 0.5).astype(float)
        if h_01.sum() != n_cur / 2:
            if h_01.sum() < n_cur / 2:
                frac = np.where(best_h < 0.5)[0]
                rem = int(n_cur / 2 - h_01.sum())
                idx = np.argsort(-best_h[frac])[:rem]
                h_01[frac[idx]] = 1.0
            else:
                frac = np.where(best_h > 0.5)[0]
                rem = int(h_01.sum() - n_cur / 2)
                idx = np.argsort(best_h[frac])[:rem]
                h_01[frac[idx]] = 0.0
        h_01 = project_feasible(h_01, n_cur)
        h_ls, c_ls = discrete_swap_search(h_01, n_cur, t2, max_iters=200000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 3: L-BFGS at base resolution (180s)
    t3 = min(t0 + 300, deadline - 750)
    if t3 > time.time() + 10:
        n_opt = len(best_h)
        best_h = project_feasible(best_h, n_opt)
        best_h, best_c = run_lbfgs(best_h, n_opt, t3,
            configs=[(200, 100.0), (1000, 500.0), (5000, 2000.0),
                     (20000, 10000.0), (80000, 40000.0)])

    # Phase 4: SLSQP with exact gradient (60s)
    t4 = min(t0 + 360, deadline - 700)
    if t4 > time.time() + 10:
        h_s, c_s = slsqp_exact(best_h, len(best_h), t4, mu=10000.0, maxiter=3000)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 5: Upscale to n=1000, L-BFGS (180s)
    n1 = 1000
    if time.time() < deadline - 250:
        h1000 = upscale(best_h, len(best_h), n1)
        t5 = min(time.time() + 100, deadline - 150)
        if t5 > time.time() + 10:
            h1000, c1000 = run_lbfgs(h1000, n1, t5,
                configs=[(500, 200.0), (3000, 1000.0), (10000, 5000.0),
                         (50000, 20000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        else:
            best_h = h1000
            best_c = min(best_c, compute_c5(best_h, n1))

    # Phase 6: Active descent at n=1000 (100s)
    t6 = min(time.time() + 80, deadline - 130)
    if t6 > time.time() + 10:
        best_h, best_c = active_descent(best_h, len(best_h), t6, k_top=50, max_iters=60000)

    # Phase 7: SLSQP at n=1000 (50s)
    t7 = min(time.time() + 40, deadline - 100)
    if t7 > time.time() + 10:
        h_s2, c_s2 = slsqp_exact(best_h, len(best_h), t7, mu=30000.0, maxiter=2000)
        if c_s2 < best_c:
            best_c = c_s2
            best_h = h_s2

    # Phase 8: Upscale to n=2000 (150s)
    n2 = 2000
    if time.time() < deadline - 100:
        h2000 = upscale(best_h, len(best_h), n2)
        t8 = min(time.time() + 70, deadline - 50)
        if t8 > time.time() + 10:
            h2000, c2000 = run_lbfgs(h2000, n2, t8,
                configs=[(2000, 1000.0), (10000, 5000.0), (50000, 25000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        else:
            best_h = h2000
            best_c = min(best_c, compute_c5(best_h, n2))

    # Phase 9: Active descent at final resolution (60s)
    t9 = min(time.time() + 40, deadline - 30)
    if t9 > time.time() + 10:
        best_h, best_c = active_descent(best_h, len(best_h), t9, k_top=60, max_iters=40000)

    # Phase 10: Final targeted refinement (30s)
    t10 = min(time.time() + 20, deadline - 5)
    if t10 > time.time() + 5:
        best_h, best_c = targeted_refine(best_h, len(best_h), t10)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
