def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
            h = np.clip(h, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
            h = np.clip(h, 0, 1)
    return h

def smooth_obj_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_func(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-13:
            continue
        k = idx - center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wi * (1.0 - x[k:])
                grad[k:] += wi * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wi * (1.0 - x[:nv])
                grad[:nv] += wi * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    if configs is None:
        configs = [(100, 10.0), (500, 50.0), (2000, 200.0),
                   (5000, 1000.0), (20000, 5000.0), (50000, 20000.0),
                   (100000, 50000.0), (200000, 100000.0)]
    
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj_func, best_h, jac=smooth_grad_func,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 200000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=200000, maxiter=5000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    try:

                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            step = diff / tot
            h += step * room
            h = np.minimum(h, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            step = diff / tot
            h *= (1.0 + step)
            h = np.maximum(h, 0.0)
    h[np.abs(h - 0.5) < 1e-15] = 0.5  # hmm, this doesn't help
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()
    w = np.exp(sh - mx)
    ws = w.sum()
    w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-13:
            continue
        k = idx - center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wi * (1.0 - x[k:])
                grad[k:] += -wi * x[:nv]
        else:
            kk = -k

                grad[kk:] += wi * (1.0 - x[:nv])

def make_thue_morse(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        if i % 2 == 0:
            t[i] = t[i // 2]
        else:
            t[i] = 1 - t[i // 2]
    return t.astype(float)

    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 if phase < 0.5 else 0.0
    return h
