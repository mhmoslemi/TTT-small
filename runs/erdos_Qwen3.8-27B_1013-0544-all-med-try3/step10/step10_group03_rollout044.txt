import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_obj(x, n, top_idx, alpha):

    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()

def subgradient_proj(h, n, deadline, n_iter=5000, base_lr=0.001):
    """Subgradient method with projection onto feasible set."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Find active shifts (within 1% of max)

        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        else:
            h = h_new  # still move, but don't update best

def pair_search(h, n, deadline, radius=40):
    """Local pair-exchange search."""
    best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        iters = 0
        while improved and iters < 50:
            improved = False
            iters += 1
            # Focus on positions with high values or low values
            h_sorted = np.argsort(np.abs(h - 0.5))
            candidates = h_sorted[-min(100, n):]
            np.random.shuffle(candidates)
            
            for i in candidates:
                if time.time() > deadline:
                    return best_h if 'best_h' in dir() else h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from j to i
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        h[i] = min(max(h[i], 0.0), 1.0)
                        h[j] = min(max(h[j], 0.0), 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def multiresolution_cycle(h, n, deadline):
    """Downsample, optimize, upsample cycle."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50:
            continue
        if time.time() > deadline - 10:
            break
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        t_c = min(40, (deadline - time.time()) * 0.3)
        if t_c > 10:
            h_c, c_c = multi_alpha_lbfgs(h_c, n_coarse, time.time() + t_c,
                alpha_list=[500, 5000, 50000])
        
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000):
    """SLSQP with equality constraint."""
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def eq_constraint(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def perturbation_restart(h, n, deadline, n_restarts=3):
    """Random perturbations to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for r in range(n_restarts):
        if time.time() > deadline - 5:
            break
        # Random perturbation
        h_p = h + rng.normal(0, 0.02, n)
        h_p = project_feasible(h_p, n)
        
        # Quick optimize
        t_sub = min(20, (deadline - time.time()) / (n_restarts - r))
        if t_sub > 5:
            h_p, c_p = multi_alpha_lbfgs(h_p, n, time.time() + t_sub,
                alpha_list=[1000, 10000, 100000])
            h_p, c_p = top_shifts_lbfgs(h_p, n, time.time() + t_sub * 0.5, top_k=3)
        
        c_final = compute_c5(h_p, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_p.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    np.random.seed(seed)
    
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS multi-alpha at original resolution
    if time.time() < deadline - 400:
        t1 = min(time.time() + 150, deadline - 400)
        h, c1 = multi_alpha_lbfgs(h, n, t1,
            alpha_list=[100, 500, 2000, 10000, 50000, 200000])
        if c1 < best_c:
            best_c = c1; best_h = h
    
    # Phase 2: Top-shifts focused
    if time.time() < deadline - 300:
        t2 = min(time.time() + 100, deadline - 300)
        h, c2 = top_shifts_lbfgs(h, n, t2, top_k=5, tolerance=0.003)
        if c2 < best_c:
            best_c = c2; best_h = h
    
    # Phase 3: Subgradient projection
    if time.time() < deadline - 250:
        t3 = min(time.time() + 60, deadline - 250)
        h, c3 = subgradient_proj(h, n, t3, n_iter=3000, base_lr=0.0005)
        if c3 < best_c:
            best_c = c3; best_h = h
    
    # Phase 4: Multi-resolution cycle
    if time.time() < deadline - 200:
        t4 = min(time.time() + 80, deadline - 200)
        h, c4 = multiresolution_cycle(h, n, t4)
        if c4 < best_c:
            best_c = c4; best_h = h
    
    # Phase 5: Binary push and relax
    if time.time() < deadline - 150:
        t5 = min(time.time() + 50, deadline - 150)
        h_bin, c5 = binary_push_and_relax(h, n, t5)
        if c5 < best_c:
            best_c = c5; best_h = h_bin
    
    # Phase 6: Top-shifts with tighter tolerance
    if time.time() < deadline - 120:
        t6 = min(time.time() + 50, deadline - 120)
        h, c6 = top_shifts_lbfgs(h, n, t6, top_k=3, tolerance=0.001)
        if c6 < best_c:
            best_c = c6; best_h = h
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 80:
        t7 = min(time.time() + 40, deadline - 80)
        h, c7 = slqp_polish(h, n, t7, alpha=500000)
        if c7 < best_c:
            best_c = c7; best_h = h
    
    # Phase 8: Local pair search
    if time.time() < deadline - 50:
        t8 = min(time.time() + 30, deadline - 50)
        h, c8 = pair_search(h, n, t8, radius=30)
        if c8 < best_c:
            best_c = c8; best_h = h
    
    # Phase 9: Perturbation restart
    if time.time() < deadline - 30:
        t9 = min(time.time() + 20, deadline - 30)
        h, c9 = perturbation_restart(h, n, t9, n_restarts=2)
        if c9 < best_c:
            best_c = c9; best_h = h
    
    # Phase 10: Final top-shifts + SLSQP
    if time.time() < deadline - 5:
        t10a = min(time.time() + 15, deadline - 3)
        h, c10a = top_shifts_lbfgs(h, n, t10a, top_k=3, tolerance=0.0005)
        if c10a < best_c:
            best_c = c10a; best_h = h
        if time.time() < deadline - 2:
            h, c10b = slqp_polish(h, n, deadline - 1, alpha=1000000)
            if c10b < best_c:
                best_c = c10b; best_h = h
    
    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
