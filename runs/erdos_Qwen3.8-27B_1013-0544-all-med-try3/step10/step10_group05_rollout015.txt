import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 50:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 200 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def coord_sweep(h, n, n_grid=200):
    h = h.copy()
    c_full = np.correlate(h, 1 - h, mode='full')
    total_len = 2 * n - 1
    for i in range(n):
        old_t = h[i]
        if old_t <= 0.0 or old_t >= 1.0:
            continue
        lo = max(0, n - 1 - i)
        hi = min(total_len - 1, total_len - 1 - i)
        aff_len = hi - lo + 1
        k_vals = np.arange(lo, hi + 1) - (n - 1)
        if lo > 0 and hi < total_len - 1:
            unaff = max(c_full[:lo].max(), c_full[hi + 1:].max())
        elif lo > 0:
            unaff = c_full[:lo].max()
        elif hi < total_len - 1:
            unaff = c_full[hi + 1:].max()
        else:
            unaff = 0.0
        c_aff = c_full[lo:hi + 1].copy()
        coeff = np.zeros(aff_len)
        quad_mask = np.zeros(aff_len, dtype=bool)
        for idx in range(aff_len):
            k = k_vals[idx]
            if k == 0:
                quad_mask[idx] = True
            else:
                cv = 0.0
                if 0 <= i + k < n:
                    cv += 1.0 - h[i + k]
                if 0 <= i - k < n:
                    cv -= h[i - k]
                coeff[idx] = cv
        ts = np.linspace(0.0, 1.0, n_grid)
        delta_t = (ts[:, None] - old_t) * coeff[None, :]
        new_vals = c_aff[None, :] + delta_t
        qi = np.where(quad_mask)[0]
        if len(qi) > 0:
            qi0 = qi[0]
            new_vals[:, qi0] = c_aff[qi0] + ts - ts * ts - old_t + old_t * old_t
        aff_max = new_vals.max(axis=1)
        total_max = np.maximum(aff_max, unaff)
        min_idx = np.argmin(total_max)
        best_t = ts[min_idx]
        best_val = total_max[min_idx]
        cur_best = c_full.max()
        if best_val < cur_best - 1e-13:
            h[i] = best_t
            delta = best_t - old_t
            for idx in range(aff_len):
                if quad_mask[idx]:
                    c_full[lo + idx] = c_aff[idx] + best_t - best_t * best_t - old_t + old_t * old_t
                else:
                    c_full[lo + idx] = c_aff[idx] + delta * coeff[idx]
    return h, c_full

    cur_h = h.copy()

        if c_new_val < best_c - 1e-14:

            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 80:
                cur_h = best_h.copy()
                no_improve_count = 0
        sweep += 1
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, top_k=8, alpha=5000000.0, mu=0.001):
