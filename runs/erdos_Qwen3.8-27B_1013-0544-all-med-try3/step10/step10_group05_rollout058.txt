def optimal_swap_delta(c_vec, G, i, j, h, n):
    """Find optimal δ for swap h[i]+=δ, h[j]-=δ."""
    d = G[:, i] - G[:, j]  # (2n-1,)
    a = c_vec  # (2n-1,)
    
    # Feasible range for δ
    delta_min = max(0, -h[i])  # h[i]+δ >= 0 -> δ >= -h[i], but h[i]>=0 so δ>=0... wait
    # h[i] + δ <= 1 -> δ <= 1 - h[i]
    # h[j] - δ >= 0 -> δ <= h[j]
    # h[j] - δ <= 1 -> δ >= h[j] - 1, but h[j]<=1 so this is always OK
    delta_lo = 0.0  # We'll only consider δ >= 0 (for this direction)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_hi <= 0:
        return 0.0
    
    # Active constraints
    max_val = c_vec.max()
    tol = max(1e-8, 0.01 * max_val)
    active = np.where(c_vec >= max_val - tol)[0]
    
    # Candidate δ values
    candidates = [0.0, delta_hi]
    
    # Intersection points between pairs of active constraints
    for k1 in active:
        for k2 in active:
            if k1 == k2:
                continue
            if abs(d[k1] - d[k2]) < 1e-15:
                continue
            delta_candidate = (a[k2] - a[k1]) / (d[k1] - d[k2])
            if delta_lo < delta_candidate < delta_hi:
                candidates.append(delta_candidate)
    
    # Also add intersection of active with non-active (for the "knee")
    # Actually, let's also check where each active constraint crosses the current max
    for k in active:
        if abs(d[k]) > 1e-15:
            # Where does a[k] + d[k]*δ = max_val?
            delta_cross = (max_val - a[k]) / d[k]
            if delta_lo < delta_cross < delta_hi:
                candidates.append(delta_cross)
    
    # Evaluate f(δ) at all candidates
    best_delta = 0.0
    best_val = c_vec.max()
    for delta_c in candidates:
        new_c = c_vec + delta_c * d
        val = new_c.max()
        if val < best_val:
            best_val = val
            best_delta = delta_c
    
    return best_delta

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-15:
