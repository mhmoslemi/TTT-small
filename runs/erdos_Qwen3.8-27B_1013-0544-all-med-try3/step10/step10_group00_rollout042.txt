def project_feasible(h, n):

def swap_delta_vec(h, corr, i, j, n):

    Returns an array of the same length as corr containing the deltas."""

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    def lbfgs_obj(x, n, alpha, mu):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val

    def lbfgs_grad(x, n, alpha, mu):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = h_grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 5,
                               'maxls': 30, 'maxcor': 30})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def upsample(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new

        h3, c3 = targeted_local_search(h_bin, best_n, p3_deadline, rng, max_rounds=100000)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: Simulated annealing (150s)
    p4_deadline = min(t0 + 490, deadline - 300)
    if time.time() < p4_deadline:
        h4, c4 = simulated_annealing_binary(h_bin, best_n, p4_deadline, rng,
                                             t_start=3.0, t_end=0.05, max_iters=500000)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 5: Try different resolutions
    # Try n_parent+50 or n_parent+100
    candidate_ns = []
    if best_n != 550:
        candidate_ns.append(550)
    if best_n != 600:
        candidate_ns.append(600)
    
    p5_deadline = min(t0 + 750, deadline - 200)
    for n_try in candidate_ns:
        if time.time() > p5_deadline - 60:
            break
        # Upsample
        h_try = upsample(best_h, best_n, n_try)
        h_try_bin = binarize(h_try, n_try)
        sub_deadline = min(p5_deadline, time.time() + 120)
        h_try2, c_try = greedy_local_search_fast(h_try_bin, n_try, sub_deadline, rng, max_iters=2000000)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try2.copy()
            best_n = n_try
    
    # Phase 6: More greedy search on best (100s)
    p6_deadline = min(t0 + 850, deadline - 150)
    if time.time() < p6_deadline:
        h_bin_best = binarize(best_h, best_n)
        h6, c6 = greedy_local_search_fast(h_bin_best, best_n, p6_deadline, rng, max_iters=3000000)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    
    # Phase 7: L-BFGS fractional refinement (80s)
    p7_deadline = min(t0 + 930, deadline - 100)
    if time.time() < p7_deadline:
        for alpha in [100000, 500000]:
            if time.time() > p7_deadline - 10:
                break
            h7, c7 = run_lbfgs(best_h, best_n, min(p7_deadline, time.time() + 35),
                              alpha=alpha, mu=0.001, maxit=1000)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Fine coordinate descent (50s)
    p8_deadline = min(t0 + 980, deadline - 50)
    if time.time() < p8_deadline:
        h8, c8 = fine_coord_descent(best_h, best_n, p8_deadline,
                                   steps=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                   max_rounds=200)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Another round of binary greedy on best binarized (50s)
    p9_deadline = min(t0 + 1030, deadline - 25)
    if time.time() < p9_deadline:
        h9_bin = binarize(best_h, best_n)
        h9, c9 = greedy_local_search_fast(h9_bin, best_n, p9_deadline, rng, max_iters=2000000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final L-BFGS polish (30s)
    if time.time() < deadline - 5:
        h10, c10 = run_lbfgs(best_h, best_n, deadline - 3, alpha=1000000, mu=0.0005, maxit=500)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
