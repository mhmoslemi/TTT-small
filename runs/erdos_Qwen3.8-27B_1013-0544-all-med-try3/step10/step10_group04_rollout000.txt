def binary_sa(h, n, t_start, t_end, n_iters, deadline):
    """Simulated annealing on binary structure."""
    h_b = (h > 0.5).astype(float)
    # Fix to have exactly n/2 ones
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    # Adjust to get exactly n/2
    if len(ones) > n//2:
        # Remove excess ones
        excess = len(ones) - n//2
        for i in np.random.choice(ones, excess, replace=False):
            h_b[i] = 0
    elif len(ones) < n//2:
        need = n//2 - len(ones)
        for i in np.random.choice(zeros, need, replace=False):
            h_b[i] = 1
    
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    current_c = best_c
    rng = np.random.RandomState(42)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Swap a random 1 with a random 0
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i] = 0
        h_b[j] = 1
        new_c = compute_c5(h_b, n)
        
        T = t_start * (t_end / t_start) ** (it / n_iters)
        if new_c < current_c or rng.rand() < np.exp(-(new_c - current_c) / T):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_b.copy()
        else:
            h_b[i] = 1
            h_b[j] = 0
    
    return best_h, best_c

                grad[0:nv] += wm * (-x[kk:n])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_grad_topk(x, n, alpha, topk=100):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    # Sort indices by weight, only process top-k
    idx_order = np.argsort(w)[::-1][:topk]
    for idx in idx_order:
        wi = w[idx]
        if wi < 1e-14:
            break
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_phase(h, n, deadline, alphas, topk=100, maxiter_per=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        t_start = time.time()
        max_it = maxiter_per
        # Reduce maxiter based on time
        elapsed = time.time() - t_start
        remaining = deadline - 30 - time.time()
        if elapsed > 0 and elapsed < remaining:
            est_per_iter = elapsed / max(1, min(int(elapsed * 10), max_it))
            max_it = min(max_it, max(100, int(remaining / max(est_per_iter, 0.001))))
        try:
            res = minimize(smooth_obj, best_h,
                          jac=lambda x: smooth_grad_topk(x, n, alpha, topk),
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_it, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': max_it * 3, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_sa(h, n, deadline, n_iters=50000, seed=999):
    """Simulated annealing on binary structure."""
    h_b = (h > 0.5).astype(float)
    # Fix to have exactly n//2 ones
    while h_b.sum() > n / 2:
        ones = np.where(h_b == 1)[0]
        h_b[ones[0]] = 0
    while h_b.sum() < n / 2:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[0]] = 1
    rng = np.random.RandomState(seed)
    best_h = h_b.copy()
    best_c = compute_c5(h_b, n)
    current_c = best_c
    t_start = 0.005
    t_end = 0.00001
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        else:
            pass
        # Recompute ones/zeros periodically
        if it % 1000 == 0:
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_c = compute_c5(h_b, n)
        T = t_start * (t_end / t_start) ** (it / float(n_iters))
        accept = new_c < current_c or (rng.rand() < np.exp(-(new_c - current_c) / max(T, 1e-10)))
        if accept:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
    # Interpolate back to continuous
    return best_h, best_c

def sharpen(h, n, strength=0.3):
    """Push values toward 0 or 1, maintaining the integral constraint."""
    h_s = h + strength * (2.0 * h - 1.0)
    h_s = np.clip(h_s, 0.0, 1.0)
    return project_feasible(h_s, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def targeted_refine(h, n, deadline, n_rounds=200):
    """Target the worst shifts specifically."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Get top 5 worst shifts
        top5 = np.argsort(c_vec)[::-1][:5]
        improved = False
        for idx in top5:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv <= 0:
                    continue
                # Try nudging h[i] down and h[i+k] up
                # Find the pair with max contribution
                contrib = best_h[0:nv] * (1.0 - best_h[k:n])
                max_i = int(np.argmax(contrib))
                # Try to reduce: decrease h[max_i] or increase h[max_i+k]
                for delta in [0.001, -0.001, 0.0005, -0.0005]:
                    h_try = best_h.copy()
                    h_try[max_i] += delta
                    h_try[max_i + k] -= delta
                    if h_try[max_i] < 0 or h_try[max_i] > 1 or h_try[max_i+k] < 0 or h_try[max_i+k] > 1:
                        continue
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-15:
                        best_c = c_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
    return best_h, best_c
