def project_feasible(h, n):

            if tot < 1e-15:

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

    return float(val)

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_optimize(h, n, deadline, topk=5):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for _ in range(8):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        elif len(active) > topk:
            top_idx = active[np.argsort(-c_vec[active])][:topk]
            active = top_idx
        for alpha in [5000, 50000, 200000, 1000000]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(topk_obj, h, jac=topk_grad,
                              args=(n, active, alpha, 0.05),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 4000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 40000, 'maxls': 80})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def gradient_local_search(h, n, deadline, base_step=0.003):
    best_c = compute_c5(h, n)
    step_mult = 1.0
    stuck_count = 0
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        total_grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                total_grad[0:nv] += (1.0 - h[k:n]) * dx
                total_grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                total_grad[kk:n] += (1.0 - h[0:nv]) * dx
                total_grad[0:nv] += (-h[kk:n]) * dx
        total_grad /= len(active)
        grad_mag = np.abs(total_grad)
        top_indices = np.argsort(-grad_mag)[:60]
        improved = False
        step = base_step * step_mult
        for i_idx in range(len(top_indices)):
            if time.time() > deadline:
                break
            i = top_indices[i_idx]
            for j_idx in range(len(top_indices)):
                if i_idx == j_idx:
                    continue
                j = top_indices[j_idx]
                if total_grad[i] > total_grad[j]:
                    hi, hj = i, j
                else:
                    hi, hj = j, i
                for s in [step, step * 0.5, step * 0.2, step * 0.05]:
                    if h[hi] - s >= 0 and h[hj] + s <= 1:
                        h[hi] -= s
                        h[hj] += s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[hi] += s
                            h[hj] -= s
                    if h[hi] + s <= 1 and h[hj] - s >= 0:
                        h[hi] += s
                        h[hj] -= s
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[hi] -= s
                            h[hj] += s
        if improved:
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 5:
                step_mult *= 0.4
                stuck_count = 0
            if step_mult < 1e-5:
                break
    return h, best_c

def fine_pair_search(h, n, deadline, window=30):
    best_c = compute_c5(h, n)
    for step in [0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    if h[i] + step <= 1 and h[j] - s >= 0 if False else (h[i] + step <= 1 and h[j] - step >= 0):
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if time.time() > deadline:
                        return h, best_c
    return h, best_c

def perturb_reoptimize(h, n, deadline, n_trials=5, noise_scale=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for trial in range(n_trials):
        if time.time() > deadline - 30:
            break
        h_pert = h + np.random.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        cands = [(h_pert, c_pert)]
        cands.append((h, compute_c5(h, n)))
        for cc, hh in cands:
            t_sub = min(25, deadline - time.time() - 20)
            if t_sub < 5:
                break
            hh, cc = lbfgs_round(hh, n, 10000, 100.0, time.time() + t_sub, maxiter=2000)
            if cc < best_c:
                best_c = cc
                best_h = hh
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
        h = best_h
        noise_scale *= 0.6
    return best_h, best_c

def fourier_init(n, K=15, seed=42):
    rng = np.random.RandomState(seed)
    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = rng.uniform(0.05, 0.15) * (1.0 / k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    # Phase 1: Quick L-BFGS on initial
    if time.time() < deadline - 600:
        t1 = min(time.time() + 60, deadline - 600)
        best_h, best_c = lbfgs_round(best_h, n_orig, 500, 50.0, t1, maxiter=3000)
    # Phase 2: Work at n=500
    n = 500
    if n_orig != n:
        x_old = np.linspace(0, 2.0, n_orig, endpoint=False)
        x_new = np.linspace(0, 2.0, n, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n)
    else:
        h_new = best_h.copy()
    best_c_n = compute_c5(h_new, n)
    if best_c_n > best_c:
        best_h_local = best_h.copy()
        h_current = best_h.copy()
        n_current = n_orig
    else:
        best_h_local = h_new.copy()
        h_current = h_new.copy()
        n_current = n
        best_c = best_c_n
    # Phase 3: Multi-alpha L-BFGS at current resolution
    if time.time() < deadline - 500:
        t3 = min(time.time() + 120, deadline - 500)
        for alpha, mu in [(200, 20.0), (1000, 100.0), (5000, 500.0),
                          (20000, 2000.0), (100000, 10000.0), (500000, 50000.0)]:
            if time.time() > t3:
                break
            h_current, c = lbfgs_round(h_current, n_current, alpha, mu, t3, maxiter=3000)
            if c < best_c:
                best_c = c
                best_h_local = h_current.copy()
    # Phase 4: Top-shift focused
    if time.time() < deadline - 300:
        t4 = min(time.time() + 100, deadline - 300)
        h_current, c = topk_optimize(h_current, n_current, t4, topk=5)
        if c < best_c:
            best_c = c
            best_h_local = h_current.copy()
    # Phase 5: Gradient local search
    if time.time() < deadline - 200:
        t5 = min(time.time() + 80, deadline - 200)
        h_current, c = gradient_local_search(h_current, n_current, t5, base_step=0.003)
        if c < best_c:
            best_c = c
            best_h_local = h_current.copy()
    # Phase 6: Perturb and re-optimize to escape local optima
    if time.time() < deadline - 120:
        t6 = min(time.time() + 80, deadline - 120)
        h_pert, c_pert = perturb_reoptimize(best_h_local, n_current, t6, n_trials=4, noise_scale=0.015)
        if c_pert < best_c:
            best_c = c_pert
            best_h_local = h_pert
    # Phase 7: Another L-BFGS pass with high alpha
    if time.time() < deadline - 80:
        t7 = min(time.time() + 40, deadline - 80)
        for alpha, mu in [(50000, 5000.0), (200000, 50000.0), (1000000, 200000.0)]:
            if time.time() > t7:
                break
            h_tmp, c = lbfgs_round(best_h_local, n_current, alpha, mu, t7, maxiter=2000)
            if c < best_c:
                best_c = c
                best_h_local = h_tmp
    # Phase 8: Top-shift final round
    if time.time() < deadline - 50:
        t8 = min(time.time() + 30, deadline - 50)
        h_tmp, c = topk_optimize(best_h_local, n_current, t8, topk=3)
        if c < best_c:
            best_c = c
            best_h_local = h_tmp
    # Phase 9: Fine local search
    if time.time() < deadline - 20:
        t9 = min(time.time() + 25, deadline - 20)
        h_tmp, c = gradient_local_search(best_h_local, n_current, t9, base_step=0.001)
        if c < best_c:
            best_c = c
            best_h_local = h_tmp
    # Phase 10: Fine pair search
    if time.time() < deadline - 5:
        t10 = deadline - 2
        h_tmp, c = fine_pair_search(best_h_local, n_current, t10, window=20)
        if c < best_c:
            best_c = c
            best_h_local = h_tmp
    # Final project and verify
    best_h_local = project_feasible(best_h_local, n_current)
    best_c = compute_c5(best_h_local, n_current)
    return best_h_local, best_c, n_current
