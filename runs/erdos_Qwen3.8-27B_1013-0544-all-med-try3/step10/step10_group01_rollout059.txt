import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-17:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-17:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-17:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - n1
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_set(c_vec, top_k, rel_tol=0.001):
    max_val = c_vec.max()
    threshold = max_val - max(1e-10, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def lbfgs_topk(h, n, deadline, active, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(topk_obj, h, jac=topk_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxit * 12,
                               'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def lbfgs_full(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(full_smooth_obj, h, jac=full_smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxit * 12,
                               'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def grad_coord_descent(h, n, deadline, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            n_pos = min(len(pos_idx), 30)
            n_neg = min(len(neg_idx), 30)
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for ii in range(n_pos):
                gi = grad[pos_idx[ii]]
                for jj in range(n_neg):
                    j = neg_idx[jj]
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = pos_idx[ii]
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-17:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        old_i = best_h[i]
                        old_j = best_h[j]
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        old_i = best_h[i]
                        old_j = best_h[j]
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] = old_i
                            best_h[j] = old_j
    return best_h, best_c

def polyak_subgrad(h, n, deadline, c_target, max_iters=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if c_target <= best_c:
        c_target = best_c + 0.001
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        n1 = n - 1
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - n1
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        f_val = max_val
        eta = (c_target - f_val) / (gnorm * gnorm + 1e-20)
        if eta <= 0:
            c_target = f_val + 0.0005
            eta = 0.01
        eta = min(eta, 0.05)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-17:
            best_c = c_new
            best_h = h_new
            c_target = min(c_target, best_c + 0.0005)
    return best_h, best_c

def perturb_and_reopt(h, n, deadline, rng, n_tries=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_tries):
        if time.time() > deadline:
            break
        h_pert = h.copy()
        idx = rng.choice(n, size=n // 50 + 1, replace=False)
        h_pert[idx] += rng.uniform(-0.05, 0.05, size=len(idx))
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.01:
            c_vec = compute_c5_vec(h_pert, n)
            active = get_active_set(c_vec, 8, 0.002)
            alpha = 200000
            h_res, c_res = lbfgs_topk(h_pert, n, deadline, active, alpha, 0.01, 500)
            if c_res < best_c:
                best_c = c_res
                best_h = h_res
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()

    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    golden = (1.0 + np.sqrt(5.0)) / 2.0

        make_beatty_start(n, golden / 2.0),
        make_beatty_start(n, np.sqrt(2.0) / 2.0),
        make_beatty_start(n, (3.0 - np.sqrt(5.0)) / 2.0),

            if c1 < best_c - 1e-15:

                best_h = h1

    # Phase 2: Top-k active set optimization (multiple rounds)

            (5, 200000, 0.01, 4),
            (8, 500000, 0.005, 3),

            active = get_active_set(c_vec, top_k, 0.0005)

                if time.time() > p5_end - 3:

                h5, c5 = lbfgs_topk(best_h, n, p5_end, active, alpha, 0.0005, maxit)
                if c5 < best_c - 1e-15:

    # Phase 6: Fine pair search
    if time.time() < deadline - 50:
        p6_end = min(time.time() + 40, deadline - 35)

            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation and re-optimization
    if time.time() < deadline - 30:
        p7_end = min(time.time() + 25, deadline - 20)
        h7, c7 = perturb_and_reopt(best_h, n, p7_end, rng, n_tries=4)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final gradient descent
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 15, deadline - 12)
        h8, c8 = grad_coord_descent(best_h, n, p8_end,
            step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            max_steps=40000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final active set polish
    if time.time() < deadline - 12:
        c_vec = compute_c5_vec(best_h, n)
        active = get_active_set(c_vec, 5, 0.0003)
        for alpha in [5000000, 20000000]:
            if time.time() > deadline - 8:
                break
            h9, c9 = lbfgs_topk(best_h, n, deadline - 6, active, alpha, 0.0001, 300)
            if c9 < best_c - 1e-15:
                best_c = c9
                best_h = h9

    # Phase 10: Very fine pair search
    if time.time() < deadline - 5:
        h10, c10 = fine_pair_search(best_h, n, deadline - 3, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
