Let me implement paired coordinate descent:

For pair (j, k) with current values h[j]=u, h[k]=v:
- Set c = u + v (must maintain this)
- For each shift s, compute the contribution of h[j] and h[k] to c_s
- Express c_s as A_s + B_j_s * h[j] + B_k_s * h[k] = A_s + B_j_s * t + B_k_s * (c - t) = (A_s + B_k_s*c) + (B_j_s - B_k_s) * t

- Minimize max_s C_s(t) over t ∈ [max(0,c-1), min(1,c)]

This is the same 1D minimax! And we can find the optimal t.

This is much better than single coordinate descent because it maintains the constraint.

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_coefficients(h, idx, n):
    """Get (a[k], b[k]) for the coordinate at index idx.
    Returns arrays a and b of length 2n-1 where c_k = a[k] + b[k]*h[idx]"""
    a = np.zeros(2*n-1)
    b = np.zeros(2*n-1)
    j = idx
    
    # Forward: i=j, shift s, k = s+n-1. Valid: 0 <= j+s < n => s in [-j, n-1-j]
    s_fwd = np.arange(-j, n-j)
    k_fwd = s_fwd + n - 1
    idx_fwd = j + s_fwd
    b[k_fwd] += (1.0 - h[idx_fwd])
    
    # Backward: i=j-s, shift s, k = s+n-1. Valid: 0 <= j-s < n => s in [j-n+1, j]
    s_bwd = np.arange(j-n+1, j+1)
    k_bwd = s_bwd + n - 1
    idx_bwd = j - s_bwd
    a[k_bwd] += h[idx_bwd]
    b[k_bwd] -= h[idx_bwd]
    
    return a, b

def paired_coordinate_step(h, j, k, n, n_grid=500):
    """Find optimal (h[j], h[k]) for fixed sum, minimizing max correlation."""
    c_val = h[j] + h[k]
    a_j, b_j = get_coefficients(h, j, n)
    a_k, b_k = get_coefficients(h, k, n)
    
    # Current correlation
    corr = np.correlate(h, 1-h, mode='full')

        maxes2 = vals2.max(axis=1)
        best_t = ts2[np.argmin(maxes2)]
    
    new_j = float(np.clip(best_t, t_lo, t_hi))
    new_k = float(np.clip(c_val - best_t, t_lo, t_hi))
    return new_j, new_k

            if j == k:

            old_j, old_k = h[j], h[k]
            new_j, new_k = paired_coordinate_step(h, j, k, n)
            h[j], h[k] = new_j, new_k
            # Check improvement

            improved = True
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # Revert to best
            h = best_h.copy()
            if not improved:
                break
    
    return best_h, best_c
