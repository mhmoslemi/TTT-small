    best_c = compute_c5(h, n)
    lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        grad, c_vec = active_gradient(h, n, top_k, alpha)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        h_trial = h - lr * grad
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-18:
            h = h_trial
            best_c = c_trial
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h, best_c

def active_lbfgs(h, n, deadline, top_k, alpha, mu):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-7, 0.001 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    def obj_act(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        vals = c[active]
        return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2
    def grad_act(x):
        c_raw = np.correlate(x, 1.0 - x, mode='full')
        vals_raw = c_raw[active]
        m = vals_raw.max()
        sh = alpha * (vals_raw - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w_s = w.sum()
        if w_s < 1e-30:
            w = np.ones(len(w)) / len(w)
        else:
            w = w / w_s
        dx = 2.0 / n
        idx_center = n - 1
        g = np.zeros(n)
        for wi, idx in zip(w, active):
            if wi < 1e-15:
                continue
            k = int(idx) - idx_center
            kk = abs(k)
            nv = n - kk
            if nv > 0:
                g[kk:n] += wi * (1.0 - x[0:nv])
                g[0:nv] += wi * (-x[kk:n])
        g *= dx
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = max(200, min(int(rem * 4), 5000))
    try:
        res = minimize(obj_act, h, jac=grad_act, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxfun': maxit * 10,
                                'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def fine_coordinate_descent(h, n, deadline, radius=50, deltas=None):
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 3:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            active_shifts = np.where(c_vec >= max_val - tol)[0]
            idx_center = n - 1
            moved = set()
            for s_idx in active_shifts[:15]:
                if time.time() > deadline - 5:
                    break
                k = int(s_idx) - idx_center
                kk = abs(k)
                if kk == 0:
                    continue
                for i in range(n - kk):
                    if (i, i + kk) in moved or (i + kk, i) in moved:
                        continue
                    a, b = i, i + kk
                    if best_h[a] >= d and best_h[b] <= 1.0 - d:
                        best_h[a] -= d
                        best_h[b] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-18:
                            best_c = c
                            improved = True
                            moved.add((a, b))
                        else:
                            best_h[a] += d
                            best_h[b] -= d
                    if best_h[b] >= d and best_h[a] <= 1.0 - d:
                        best_h[a] += d
                        best_h[b] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-18:
                            best_c = c
                            improved = True
                            moved.add((b, a))
                        else:
                            best_h[a] -= d
                            best_h[b] += d
    return best_h, best_c

def global_pair_search(h, n, deadline, delta=0.001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    idx = np.argsort(-best_h)
    jdx = np.argsort(best_h)
    for _ in range(200):
        if time.time() > deadline - 5:
            break
        improved = False
        for a in idx[:min(50, n // 2)]:
            if time.time() > deadline - 10:
                break
            if best_h[a] < delta:
                continue
            for b in jdx[:min(50, n // 2)]:
                if a == b or best_h[b] > 1.0 - delta:
                    continue
                best_h[a] -= delta
                best_h[b] += delta
                c = compute_c5(best_h, n)
                if c < best_c - 1e-18:
                    best_c = c
                    improved = True
                    idx = np.argsort(-best_h)
                    jdx = np.argsort(best_h)
                else:
                    best_h[a] += delta
                    best_h[b] -= delta
        if not improved:
            delta *= 0.5
            if delta < 1e-7:
                break
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_structured(n, pattern_type=0):
    h = np.zeros(n)
    n_ones = n // 2
    if pattern_type == 0:
        for i in range(n):
            if (i * 7) % n < n // 2:
                h[i] = 1.0
    elif pattern_type == 1:
        period = n // 2
        for i in range(n):
            if i % (period + 1) < period // 2:
                h[i] = 1.0
    elif pattern_type == 2:
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n):
            frac = (i / n * golden * 10) % 1.0
            if frac < 0.5:
                h[i] = 1.0
    return project_feasible(h, n)

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def perturb_reopt(h, n, deadline, rng, n_trials=5, max_step=0.02):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    per_trial = max(10, (deadline - time.time() - 10) / n_trials)
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        t_sub = deadline - 10 - (n_trials - 1 - t) * per_trial
        t_sub = max(time.time() + 5, t_sub)
        h_p = best_h + rng.uniform(-max_step, max_step, size=n)
        h_p = project_feasible(h_p, n)
        c_p = compute_c5(h_p, n)
        if c_p < best_c + 0.01:
            h_new, c_new = active_lbfgs(h_p, n, t_sub, top_k=40000, alpha=50000, mu=0.005)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    alt_starts = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(make_beatty(n_orig, golden / 2.0))
    alt_starts.append(make_beatty(n_orig, np.sqrt(2.0) / 2.0))
    alt_starts.append(make_beatty(n_orig, 0.4312 / 1.0))
    alt_starts.append(make_beatty(n_orig, 0.381966))
    alt_starts.append(make_random_binary(n_orig, rng))
    alt_starts.append(make_random_binary(n_orig, rng))
    alt_starts.append(make_structured(n_orig, 0))
    alt_starts.append(make_structured(n_orig, 1))
    for hs in alt_starts:
        cs = compute_c5(hs, n_orig)
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()

    # Phase 1: Quick optimization at original resolution
    p1_end = min(t0 + 80, deadline - 600)
    if p1_end > t0 + 15:
        for alpha in [500, 5000, 50000]:
            if time.time() > p1_end - 30:
                break
            h_new, c_new = proj_grad_descent(best_h, best_n, p1_end - 30,
                                             top_k=30, alpha=alpha, max_iter=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        h_new, c_new = active_lbfgs(best_h, best_n, p1_end - 5, top_k=100, alpha=20000, mu=0.005)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Upsample to 2x and optimize
    n_up = best_n * 2
    p2_end = min(time.time() + 300, deadline - 400)
    if p2_end > time.time() + 60 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.005:
            for alpha in [2000, 20000, 200000]:
                if time.time() > p2_end - 80:
                    break
                h_new, c_new = proj_grad_descent(h_up, n_up, p2_end - 80,
                                                 top_k=50, alpha=alpha, max_iter=3000)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new
            if best_n == n_up:
                h_new, c_new = active_lbfgs(h_up, n_up, p2_end - 20,
                                            top_k=200, alpha=50000, mu=0.003)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new
                h_new, c_new = active_lbfgs(h_up, n_up, p2_end - 5,
                                            top_k=100, alpha=5000, mu=0.01)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new

    # Phase 3: Try 3x or 4x upsample from best
    cur_n = len(best_h)
    n_up2 = cur_n * 2
    p3_end = min(time.time() + 200, deadline - 250)
    if p3_end > time.time() + 50 and n_up2 <= 4000:
        h_up2 = upsample_h(best_h, cur_n, n_up2)
        c_up2 = compute_c5(h_up2, n_up2)
        if c_up2 <= best_c + 0.003:
            h_new, c_new = proj_grad_descent(h_up2, n_up2, p3_end - 60,
                                             top_k=60, alpha=30000, max_iter=4000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                best_n = n_up2
                h_up2 = h_new
            if best_n == n_up2:
                h_new, c_new = active_lbfgs(h_up2, n_up2, p3_end - 10,
                                            top_k=300, alpha=100000, mu=0.002)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up2
                    h_up2 = h_new

    # Phase 4: Multi-restart with perturbation at best resolution
    p4_end = min(time.time() + 150, deadline - 200)
    if p4_end > time.time() + 30:
        cur_n = len(best_h)
        h_new, c_new = perturb_reopt(best_h, cur_n, p4_end, rng,
                                     n_trials=6, max_step=0.015)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Fine coordinate descent
    p5_end = min(time.time() + 150, deadline - 100)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        h_new, c_new = fine_coordinate_descent(best_h, cur_n, p5_end,
                                               radius=60,
                                               deltas=[0.002, 0.001, 0.0005,
                                                       0.0002, 0.0001, 0.00005])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 6: Another round of projected gradient + fine descent
    p6_end = min(time.time() + 100, deadline - 80)
    if p6_end > time.time() + 15:
        cur_n = len(best_h)
        h_new, c_new = proj_grad_descent(best_h, cur_n, p6_end - 40,
                                         top_k=30, alpha=100000, max_iter=3000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = fine_coordinate_descent(best_h, cur_n, p6_end,
                                               radius=40,
                                               deltas=[0.0005, 0.0002, 0.0001,
                                                       0.00005, 0.00002, 0.00001])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 7: Global pair search for large moves
    p7_end = min(time.time() + 60, deadline - 40)
    if p7_end > time.time() + 10:
        cur_n = len(best_h)
        h_new, c_new = global_pair_search(best_h, cur_n, p7_end, delta=0.005)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h_new, c_new = fine_coordinate_descent(best_h, cur_n, min(time.time() + 20, deadline - 5),
                                                   radius=40,
                                                   deltas=[0.0002, 0.0001, 0.00005])
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 8: Ultra-fine final polish
    p8_end = deadline - 2
    if p8_end > time.time() + 10:
        cur_n = len(best_h)
        h_new, c_new = fine_coordinate_descent(best_h, cur_n, p8_end,
                                               radius=30,
                                               deltas=[0.0001, 0.00005, 0.00002,
                                                       0.00001, 0.000005])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final: also try evaluating at original n with downsampled version
    if best_n != n_orig:
        h_down = downsample_h(best_h, best_n, n_orig)
        c_down = compute_c5(h_down, n_orig)
        if c_down < best_c:
            best_c = c_down
            best_h = h_down
            best_n = n_orig

    # Also try: take best structure, re-optimize at n_orig with L-BFGS
    if best_n != n_orig and time.time() < deadline - 15:
        h_try = downsample_h(best_h, best_n, n_orig)
        h_try, c_try = active_lbfgs(h_try, n_orig, deadline - 3,
                                    top_k=200, alpha=50000, mu=0.005)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            best_n = n_orig

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
