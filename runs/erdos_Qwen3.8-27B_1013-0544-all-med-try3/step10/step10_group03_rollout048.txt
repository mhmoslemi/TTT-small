    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_smoothed_topk(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smoothed_topk(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgs_global(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 5.0), (200, 20.0), (1000, 100.0),
               (5000, 500.0), (20000, 5000.0), (100000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(obj_smoothed_full, h, jac=grad_smoothed_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_lbfgs_topk(h, n, deadline, top_k=8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(15):
        if time.time() > deadline - 3:
            break
        top_idx, _ = get_top_shifts(h, n, top_k)
        
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(obj_smoothed_topk, h, jac=grad_smoothed_topk,
                              args=(n, top_idx, alpha, 50.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 10000, 'maxls': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def phase_subgradient(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.02
    stall_count = 0
    
    while time.time() < deadline - 2:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        # Get top 5 active shifts
        top5 = np.argsort(-c_vec)[:5]
        # Average gradient of top 5
        grad = np.zeros(n)
        for idx in top5:
            grad += grad_shift_single(h, n, idx)
        grad /= 5.0
        
        # Project gradient to satisfy sum constraint
        grad -= grad.mean()
        
        # Step in negative gradient direction
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                step *= 0.5
                if step < 1e-8:
                    break
                stall_count = 0
    return best_h, best_c

def phase_fine_search(h, n, deadline, window=40):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.004, 0.001, 0.0004, 0.0001]
    
    for d in deltas:
        if time.time() > deadline - 2:
            break
        improved = True
        n_ev = 0
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return best_h, best_c
                if n_ev > 50000:
                    break
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    n_ev += 1
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        # Re-start with current best
        h = best_h.copy()
    return best_h, best_c

def phase_gradient_fine(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for base_step in step_sizes:
        if time.time() > deadline - 2:
            break
        for _range in range(200):
            if time.time() > deadline - 2:
                return best_h, best_c
            c_vec = compute_c5_vec(h, n)
            top5 = np.argsort(-c_vec)[:5]
            grad = np.zeros(n)
            for idx in top5:
                grad += grad_shift_single(h, n, idx)
            grad /= 5.0
            grad -= grad.mean()
            
            # Find best pair move
            g_pos = np.argsort(-grad)[:30]
            g_neg = np.argsort(grad)[:30]
            found = False
            for i in g_pos:
                if found:
                    break
                for j in g_neg:
                    if i == j:
                        continue
                    step = base_step
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            found = True
                            break
                        else:
                            h[i] += step
                            h[j] -= step
                    if h[i] + step <= 1 and h[j] - step >= 0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            found = True
                            break
                        else:
                            h[i] -= step
                            h[j] += step
            if not found:
                h = best_h.copy()
                break
    return best_h, best_c

def phase_multistart(h, n, deadline, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(42)
    
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        # Perturb
        eps = 0.05 * (1 - s * 0.2)
        h_pert = h + rng.randn(n) * eps
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS from perturbed start
        t_sub = min(80, deadline - time.time() - 10)
        if t_sub < 30:
            continue
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_smoothed_topk, h_pert, jac=grad_smoothed_topk,
                          args=(n, get_top_shifts(h_pert, n, 8)[0], 50000, 50.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 8000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, h_init)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Global L-BFGS (150s)
    t1 = min(t0 + 150, deadline - 600)
    best_h, best_c = phase_lbfgs_global(best_h, n, t1)
    
    # Phase 2: Top-k focused L-BFGS (200s)
    if time.time() < deadline - 400:
        t2 = min(time.time() + 200, deadline - 400)
        h2, c2 = phase_lbfgs_topk(best_h, n, t2, top_k=8)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Subgradient polish (80s)
    if time.time() < deadline - 250:
        t3 = min(time.time() + 80, deadline - 250)
        h3, c3 = phase_subgradient(best_h, n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: More top-k L-BFGS with tighter parameters (120s)
    if time.time() < deadline - 200:
        t4 = min(time.time() + 120, deadline - 200)
        h4, c4 = phase_lbfgs_topk(best_h, n, t4, top_k=5)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Gradient-guided fine search (100s)
    if time.time() < deadline - 120:
        t5 = min(time.time() + 100, deadline - 120)
        h5, c5 = phase_gradient_fine(best_h, n, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Fine pair search (100s)
    if time.time() < deadline - 50:
        t6 = min(time.time() + 100, deadline - 50)
        h6, c6 = phase_fine_search(best_h, n, t6, window=30)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final subgradient (30s)
    if time.time() < deadline - 10:
        t7 = min(time.time() + 30, deadline - 3)
        h7, c7 = phase_subgradient(best_h, n, t7)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
