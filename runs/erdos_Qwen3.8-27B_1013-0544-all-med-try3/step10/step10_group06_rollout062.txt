```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        return float(m + np.log(w.sum()) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(len(c))
    return w / s

def compute_smg_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n-lag])
            grad[0:n-lag] -= w[k] * h[lag:n]
        else:
            idx = -lag
            grad[0:n-idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] -= w[k] * h[0:n-idx]
    return grad * dx

def lbfgsb_optimize(h, n, alpha, timeout_s, maxiter=5000, pen_coeff=1e8):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    
    def obj_and_grad(x):
        if time.time() > t_start + timeout_s:
            x = np.clip(x, 0.0, 1.0)
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += pen_coeff * diff * diff
        grad = compute_smg_grad(x, n, alpha)
        grad += 2.0 * pen_coeff * diff * np.ones(n)
        return obj_val, grad
    
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1, 'maxfun': maxiter * 10})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def sa_exact_max(h, n, deadline, rng, t_init=0.01, t_min=0.00005, cool=0.9999, max_no_improve=50000):
    """Simulated annealing using exact max as energy, with swap moves."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    no_improve = 0
    n_half = n // 2
    
    while time.time() < deadline and T > t_min:
        # Choose move type
        r = rng.random()
        if r < 0.7:
            # Random swap
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
        else:
            # Structured: swap at a high-overlap lag
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-5:]
            k_star = top_k[rng.randint(5)]
            lag = k_star - (n - 1)
            if lag == 0:
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j:
                    T *= cool
                    continue
            else:
                m = abs(lag)
                if m >= n:
                    i = rng.randint(0, n)
                    j = rng.randint(0, n)
                    if i == j:
                        T *= cool
                        continue
                i = rng.randint(0, n - m)
                j = i + m
        
        # Compute old energy (exact max)
        c_old = compute_all_lags(h, n)
        e_old = c_old.max()
        
        # Try the swap
        h[i], h[j] = h[j], h[i]
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = c_new.max()
        
        de = e_new - e_old
        if de < -1e-15:
            accept = True
        elif T > t_min:
            ratio = de / T
            if ratio < 50:
                accept = rng.random() < np.exp(-ratio)
            else:
                accept = False
        else:
            accept = False
        
        if accept:
            if e_new < best_c - 1e-14:
                best_c = float(e_new)
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
            no_improve += 1
        
        T *= cool
        
        if no_improve > max_no_improve:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            no_improve = 0
            T = max(T, 0.003)
    
    return best_h, best_c

def binary_hill_climbing(h, n, deadline, rng, max_rounds=200):
    """Hill climbing on the binary projection."""
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    
    best_bin = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    one_set = np.where(h_bin == 1)[0]
    zero_set = np.where(h_bin == 0)[0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        c_all = compute_all_lags(h_bin, n)
        # Focus on top lags
        top_lags = np.argsort(c_all)[-10:]
        
        for i_idx in rng.permutation(len(one_set)):
            if time.time() > deadline:
                break
            i = one_set[i_idx]
            for j_idx in rng.permutation(min(20, len(zero_set))):
                j = zero_set[j_idx]
                # Try swap
                h_bin[i] = 0
                h_bin[j] = 1
                c_test = compute_c5(h_bin, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_bin = h_bin.copy()
                    one_set = np.where(h_bin == 1)[0]
                    zero_set = np.where(h_bin == 0)[0]
                    improved = True
                    break
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        if not improved:
            break
    
    return best_bin, best_c

def coordinate_descent(h, n, deadline, rng, n_sweeps=5):
    """Cycle through coordinates, optimizing each one given all others."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            # Optimize h[i] given all others
            # c_k = sum_j h[j](1-h[j+k]) 
            # The terms involving h[i]:
            # For lag 0: h[i](1-h[i]) = h[i] - h[i]^2, derivative = 1-2h[i]
            # For lag k>0 where i appears:
            #   as h[i]: (1-h[i-k]) or (1-h[i+k])
            #   as (1-h[i]): h[i-k] or h[i+k]
            
            # Simpler: just try a few values
            old_val = h[i]
            # The optimal h[i] given others is in [0,1]
            # The contribution to the objective (smooth max) is complex
            # So just do a small local search
            candidates = [old_val]
            for frac in [0.1, 0.3, 0.7, 0.9]:
                nv = old_val * (1 - frac)
                candidates.append(nv)
            candidates.append(0.0)
            candidates.append(1.0)
            
            best_val = old_val
            best_c_i = best_c  # Use current best as reference
            # Actually, we should compute the full C5 for each candidate
            # But that's expensive. Let's just do a few.
            
            # Try moving h[i] slightly
            for delta in [0.01, -0.01, 0.05, -0.05]:
                nv = np.clip(old_val + delta, 0.0, 1.0)
                h[i] = nv
                # Rebalance sum
                diff = target_sum - h.sum()
                if abs(diff) > 1e-10:
                    # Distribute the difference to other coordinates
                    others = np.arange(n) != i
                    if diff > 0:
                        room = 1.0 - h[others]
                        tot = room.sum()
                        if tot > 1e-10:
                            h[others] += (diff / tot) * room
                    else:
                        tot = h[others].sum()
                        if tot > 1e-10:
                            h[others] += (diff / tot) * h[others]
                    h = np.clip(h, 0.0, 1.0)
                
                c_test = compute_c5(h, n)
                if c_test < best_c_i - 1e-14:
                    best_c_i = c_test
                    best_val = nv
                h[i] = old_val  # revert for next candidate
            
            if best_val != old_val:
                h[i] = best_val
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
    
    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        return float(m + np.log(w.sum()) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return w / s

def compute_smg_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n-lag])
            grad[0:n-lag] -= w[k] * h[lag:n]
        else:
            idx = -lag
            grad[0:n-idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] -= w[k] * h[0:n-idx]
    return grad * dx

def lbfgsb_optimize(h, n, alpha, timeout_s, maxiter=5000, pen_coeff=1e8):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += pen_coeff * diff * diff
        grad = compute_smg_grad(x, n, alpha)
        grad += 2.0 * pen_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1, 'maxfun': maxiter * 8})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

def weighted_topk_objective(h, n, k_top=5, alpha=1000):
    """Objective using weighted sum of top-k lags."""
    c = compute_all_lags(h, n)
    top_k = np.argsort(c)[-k_top:]
    w = np.zeros(len(c))
    w[top_k] = 1.0
    w /= w.sum()
    return float(w @ c)

def topk_refine(h, n, deadline, rng, k_top=5):
    """Refine by minimizing weighted top-k lags using subgradient steps."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.002

    for it in range(200000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-k_top:]
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0
        w /= w.sum()

        # Subgradient of weighted sum
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in top_idx:
            lag = k - (n-1)
            if lag == 0:
                grad[w[k]] * (1.0 - 2.0*h)
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - h[0:n-lag])
                grad[0:n-lag] -= w[k] * h[lag:n]
            else:
                idx = -lag
                grad[0:n-idx] += w[k] * (1.0 - h[idx:n])
                grad[idx:n] -= w[k] * h[0:n-idx]
        grad *= dx

        h -= lr * grad
        h = project_feasible(h, n)
        c_val = compute_c5(h, n)
        if c_val < best_c - 1e-14:
            best_c = c_val
            best_h = h.copy()
            lr = min(lr * 1.01, 0.01)
        else:
            lr *= 0.95

        # Periodic restart
        if it % 5000 == 0 and lr < 0.0001:
            h = best_h + rng.normal(0, 0.003, n)
            h = project_feasible(h, n)
            lr = 0.002

    return best_h, best_c

            best_c = c_opt
            best_h = h_opt
        h_opt2, c_opt2 = topk_refine(best_h, n, tk_dead, rng, k_top=3)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2

    # Phase 6: Multiple perturbation + re-opt (120s)
    if time.time() < deadline - 150:
        for trial in range(4):
            if time.time() > deadline - 80:
                break
            pert_size = 0.015 + 0.005 * trial
            h_pert = best_h + rng.normal(0, pert_size, n)
            h_pert = project_feasible(h_pert, n)
            rem = deadline - 80 - time.time()
            if rem < 10:
                break
            h_opt = lbfgsb_optimize(h_pert, n, 5000000, min(20, rem*0.5), maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # Quick SA on the perturbed result
            if time.time() < deadline - 60:
                h_sa, c_sa = sa_continuous(h_opt, n, min(deadline-60, time.time()+20), rng,
                                            t_init=0.002, t_min=0.00003, cool=0.99995)
                if c_sa < best_c:
                    best_c = c_sa
                    best_h = h_sa

    # Phase 7: Higher resolution (150s)
    if time.time() < deadline - 120:
        n_try = 1000
        h_interp = expand_to_resolution(best_h, n, n_try)
        rem = deadline - 80 - time.time()
        if rem > 30:
            h_opt = lbfgsb_optimize(h_interp, n_try, 500000, min(25, rem*0.35), maxiter=2000)
            c_opt = compute_c5(h_opt, n_try)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_try
            rem2 = deadline - 60 - time.time()
            if rem2 > 15:
                h_opt2 = lbfgsb_optimize(best_h, n, 50000000, min(20, rem2*0.5), maxiter=3000)
                c_opt2 = compute_c5(h_opt2, n)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 8: Final SA + binary cycle (80s)
    if time.time() < deadline - 50:
        n_f = len(best_h)
        h_opt, c_opt = sa_continuous(best_h, n_f, min(deadline-40, time.time()+40), rng,
                                      t_init=0.003, t_min=0.00003, cool=0.99995)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        # Quick binary
        if time.time() < deadline - 20:
            h_bin2, c_bin2 = binary_hill_climb(best_h, n_f, min(deadline-15, time.time()+20), rng, rounds=100)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = h_bin2.astype(float)
                h_rel = lbfgsb_optimize(best_h, n_f, 100000000, max(5, deadline-10-time.time()), maxiter=2000)
                c_rel = compute_c5(h_rel, n_f)
                if c_rel < best_c:
                    best_c = c_rel
                    best_h = h_rel

    # Phase 9: Final polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        rem = deadline - 3 - time.time()
        if rem > 5:
            h_opt = lbfgsb_optimize(best_h, n_f, 500000000, rem, maxiter=5000)
            c_opt = compute_c5(h_opt, n_f)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
