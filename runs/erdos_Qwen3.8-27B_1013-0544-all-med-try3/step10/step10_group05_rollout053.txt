Δc(k*) ≈ δ * (dc/dh[j] - dc/dh[i])

We want this to be negative, so we want dc/dh[j] < dc/dh[i].

So the best move is: find i with the largest dc/dh[i] and j with the smallest dc/dh[j], subject to feasibility.

This is O(n) per step! Much better than O(n²).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    # Project to [0,1]^n with sum = n/2
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-boundary elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_max_val(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_weights(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def c5_gradient(c_vec, h, n, weights=None):
    """Compute gradient of smooth max of c_vec w.r.t. h."""
    if weights is None:
        weights = smooth_max_weights(c_vec, 1000)
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx, w in enumerate(weights):
        if w < 1e-12:
            continue
        k = idx - center
        wm = w * dx
        if k >= 0:
            if k < n:
                grad[0:n-k] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:n-k])
        else:
            kk = -k
            if kk < n:
                grad[kk:n] += wm * (1.0 - h[0:n-kk])
                grad[0:n-kk] += wm * (-h[kk:n])
    return grad

def full_obj(x, n, alpha, mu):

    return smooth_max_val(c, alpha) + mu * (x.sum() - n/2.0)**2

    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha)
    grad = c5_gradient(c, x, n, w)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

    x_new = np.linspace(0, 2, n_new, endpoint=False)

    no_improve = 0

            no_improve = 0
            if delta < 1e-10:
                break
    
    return best_h, best_c
