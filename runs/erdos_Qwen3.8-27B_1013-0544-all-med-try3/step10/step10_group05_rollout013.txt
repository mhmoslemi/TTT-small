def compute_c5_vec(h, n):
    """Compute all correlation values. Returns array of length 2n-1."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    """Log-sum-exp approximation of max."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_weights(c, alpha):
    """Compute softmax weights for the smooth max."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

    """Gradient of c[k] w.r.t. h, where k_shift is the actual shift (can be negative)."""
    grad = np.zeros(n)
    if k_shift >= 0:
        nv = n - k_shift
        # c = sum_{i=0}^{nv-1} h[i+k_shift] * (1-h[i])
        # d/dh[j]: 
        #   if j >= k_shift: h[j] appears in h[i+k_shift] with i=j-k_shift, contribution (1-h[j-k_shift])
        #   if j < nv: h[j] appears in (1-h[i]) with i=j, contribution -h[j+k_shift]
        # So for j in [0, nv): contribution from (1-h[j]) is -h[j+k_shift]
        # For j in [k_shift, n): contribution from h[j] is (1-h[j-k_shift])
        grad[:nv] += -h[k_shift:k_shift+nv]  # from (1-h[i]) term
        grad[k_shift:n] += (1.0 - h[:nv])  # from h[i+k_shift] term
    else:
        kk = -k_shift
        nv = n - kk
        # c = sum_{i=0}^{nv-1} h[i] * (1-h[i+kk])
        # d/dh[j]:
        #   if j < nv: h[j] appears with contribution (1-h[j+kk])
        #   if j >= kk: h[j] appears in (1-h[i+kk]) with i=j-kk, contribution -h[j-kk]
        grad[:nv] += (1.0 - h[kk:n])  # from h[i] term
        grad[kk:n] += -h[:nv]  # from (1-h[i+kk]) term
    return grad * dx

    c = compute_c5_vec(h, n)
    c_active = c[active]
    w = smooth_max_weights(c_active, alpha)

        k = int(idx) - center
        grad += wi * grad_c5_single(h, n, k, dx)
    return grad, c.max()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w


def grad_c_shift(h, n, k):
    """Gradient of shift-k correlation (before *dx) w.r.t. h."""
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] += -h[k:k + nv]
            grad[k:n] += (1.0 - h[:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[:nv] += (1.0 - h[kk:n])
            grad[kk:n] += -h[:nv]
    return grad

    c = compute_c5_vec(x, n)
    ca = c[active]
    val = smooth_max_val(ca, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val


def grad_act(x, n, alpha, mu, active):
    c = compute_c5_vec(x, n)
    ca = c[active]
    w = smooth_max_weights(ca, alpha)
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-13:

        k = int(idx) - center
        g += wi * grad_c_shift(x, n, k) * dx
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val


def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for wi in range(len(c)):

        k = wi - center

def get_active(c, n, top_k, frac=0.01):
    max_val = c.max()
    threshold = max_val - max(1e-8, frac * max_val)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c[active])[:top_k]]
    if len(active) < 3:
        active = c.argsort()[-top_k:]
    return active

                if time.time() > p2_end - 15:
                    break
                rem = p2_end - time.time()
                if rem < 15:
                    break
                mu_val = max(0.0005, 500.0 / a)
                hk, ck = lgbf_full(h_work, n_work, time.time() + rem * 0.85, a, mu_val)
                if ck < c_work:
                    c_work = ck
                    h_work = hk
            # Active set refinement
            for top_k in [10, 20, 40, 80]:
                if time.time() > p2_end - 15:
                    break
                rk = time.time() + (p2_end - time.time()) * 0.5
                hk, ck = lgbf_active(h_work, n_work, rk, 500000, 0.0005, top_k)
                if ck < c_work:
                    c_work = ck
                    h_work = hk
            if c_work < best_c:
                best_c = c_work
                best_h = h_work.copy()
                best_n = n_work

    # Phase 3: Try upsampling to n=1500 (150s)
    n_up2 = 1500
    p3_end = min(time.time() + 150, deadline - 300)
    if p3_end > time.time() + 40 and best_n < n_up2:
        h_up = upsample_h(best_h, n_up2)
        c_up = compute_c5(h_up, n_up2)
        if c_up <= best_c + 0.001:
            for a in [5000, 50000, 500000, 3000000]:
                if time.time() > p3_end - 15:
                    break
                rem = p3_end - time.time()
                if rem < 20:
                    break
                mu_val = max(0.0003, 500.0 / a)
                hk, ck = lgbf_full(h_up, n_up2, time.time() + rem * 0.7, a, mu_val)
                if ck < c_up:
                    c_up = ck
                    h_up = hk
            for top_k in [20, 50, 100]:
                if time.time() > p3_end - 15:
                    break
                rk = time.time() + (p3_end - time.time()) * 0.5
                hk, ck = lgbf_active(h_up, n_up2, rk, 2000000, 0.0003, top_k)
                if ck < c_up:
                    c_up = ck
                    h_up = hk
            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()
                best_n = n_up2

    # Phase 4: Try n=2000 (120s) if time allows
    n_up3 = 2000
    p4_end = min(time.time() + 120, deadline - 150)
    if p4_end > time.time() + 40 and best_n < n_up3:
        h_up = upsample_h(best_h, n_up3)
        c_up = compute_c5(h_up, n_up3)
        if c_up <= best_c + 0.001:
            for a in [10000, 100000, 1000000]:
                if time.time() > p4_end - 20:
                    break
                rem = p4_end - time.time()
                if rem < 25:
                    break
                mu_val = max(0.0002, 500.0 / a)
                hk, ck = lgbf_full(h_up, n_up3, time.time() + rem * 0.7, a, mu_val)
                if ck < c_up:
                    c_up = ck
                    h_up = hk
            for top_k in [30, 60, 120]:
                if time.time() > p4_end - 15:
                    break
                rk = time.time() + (p4_end - time.time()) * 0.5
                hk, ck = lgbf_active(h_up, n_up3, rk, 5000000, 0.0002, top_k)
                if ck < c_up:
                    c_up = ck
                    h_up = hk
            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()
                best_n = n_up3

    # Phase 5: Subgradient + L-BFGS at best resolution (80s)
    cur_n = len(best_h)
    p5_end = min(time.time() + 80, deadline - 100)
    if p5_end > time.time() + 20:
        n_sg = max(100, int((p5_end - time.time()) * 0.4))
        h_sg, c_sg = subgrad_step(best_h, cur_n, 0.01, p5_end * 0.5, n_sg)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
        h_sg2, c_sg2 = lgbf_active(best_h, cur_n, p5_end, 1000000, 0.0003, 50)
        if c_sg2 < best_c:
            best_c = c_sg2
            best_h = h_sg2

    # Phase 6: Perturbation + re-optimization (60s)
    p6_end = min(time.time() + 60, deadline - 50)
    if p6_end > time.time() + 20:
        h6, c6 = perturb_reopt(best_h, best_n, p6_end, rng, n_trials=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Local refinement (remaining time minus buffer)
    p7_end = deadline - 3
    if p7_end > time.time() + 15:
        cur_n = len(best_h)
        if cur_n <= 1000:
            h7, c7 = local_refine(best_h, cur_n, p7_end)
        else:
            # For higher n, do a lighter refinement
            h7, c7 = local_refine(best_h, cur_n, min(p7_end, time.time() + 40))
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final: one more active set polish
    if time.time() < deadline - 5:
        cur_n = len(best_h)
        h_f, c_f = lgbf_active(best_h, cur_n, deadline - 2, 10000000, 0.0,
                                min(30, cur_n // 10))
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
