        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n

    t6 = min(time.time() + 120, deadline - 60)
    if t6 > time.time() + 30:
        h6000 = upsample(best_h, len(best_h), n_very_high)
        t6a = min(time.time() + 80, deadline - 30)
        h6000, c6000 = run_lbfgs(h6000, n_very_high, t6a,
            configs=[(500, 200.0), (2000, 1000.0), (5000, 3000.0), (20000, 10000.0)])
        if c6000 < best_c:
            best_c = c6000
            best_h = h6000
    
    # Phase 7: Final coordinate refinement (30s)
    t7 = min(time.time() + 30, deadline - 5)
    if t7 > time.time() + 3:
        best_h, best_c = targeted_coordinate_refine(best_h, len(best_h), t7)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
