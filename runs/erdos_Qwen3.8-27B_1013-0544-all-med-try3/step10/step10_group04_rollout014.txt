import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c_vec = np.correlate(h, 1-h, mode='full') * dx
    c_max = c_vec.max()
    # Active shifts
    tol = 1e-6
    active = c_vec > c_max - tol
    if active.sum() > 0:
        w = (c_vec - c_max + tol * active) / active.sum()  # weights for active shifts
        # Actually, for subgradient of max, we use average of gradients of active functions
        # Or just the gradient of the argmax
    else:
        w = np.zeros_like(c_vec)
        w[argmax] = 1
    
    # Compute gradient of weighted sum of c_k
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk

    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, c_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (500, 200.0),
                   (1000, 500.0), (3000, 2000.0), (8000, 10000.0),
                   (20000, 50000.0), (50000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 500000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def subgradient_optimize(h, n, deadline, n_steps=20000, top_k=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    lr = 0.005
    stagnation = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 2000:
                break
        
        if step % 1000 == 500 and step > 0:
            lr *= 0.7
            if lr < 1e-5:
                break
        
        top_idx = np.argsort(-c_vec)[:top_k]
        weights = np.zeros(2 * n - 1)
        spread = c_vec[top_idx[0]] - c_vec[top_idx[-1]] + 1e-10
        for i, idx in enumerate(top_idx):
            weights[idx] = (c_vec[idx] - c_vec[top_idx[-1]] + 0.01)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights[top_idx[0]] = 1.0
        else:
            weights /= wsum
        
        grad = np.zeros(n)
        for idx in top_idx:
            wi = weights[idx]
            if wi < 1e-15:
                continue
            k = idx - center
            wm = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - h[k:n])
                    grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - h[0:nv])
                    grad[0:nv] += wm * (-h[kk:n])
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        h = h_new
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(777)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            orig = h[i]
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.005, -0.005]:
                nv = orig + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h = best_h.copy()
                    h[i] = orig
            if improved:
                h = best_h.copy()
        if not improved and rnd > 30:
            break
    return best_h, best_c

def make_binary_init(n, n_blocks, seed):
    rng = np.random.RandomState(seed)
    block_size = n // n_blocks
    extra = n - block_size * n_blocks
    h = np.zeros(n)
    num_ones_blocks = n_blocks // 2
    block_vals = []
    for b in range(n_blocks):
        sz = block_size + (1 if b < extra else 0)
        block_vals.append(sz)
    # Choose which blocks are 1
    chosen = set(rng.choice(n_blocks, num_ones_blocks, replace=False).tolist())
    for b in range(n_blocks):
        start = sum(block_vals[:b])
        end = start + block_vals[b]
        if b in chosen:
            h[start:end] = 1.0
    return project_feasible(h, n)

def make_symmetric_init(n, n_bumps, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    centers = rng.uniform(0.1, 1.9, n_bumps)
    widths = rng.uniform(0.05, 0.3, n_bumps)
    heights = rng.uniform(0.5, 0.5, n_bumps)
    for c, w, hh in zip(centers, widths, heights):
        x = np.linspace(0, 2, n)
        h += hh * np.exp(-0.5 * ((x - c) / w) ** 2)
        h += hh * np.exp(-0.5 * ((x - (2 - c)) / w) ** 2)
    return project_feasible(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=6, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(99999)
    if mags is None:
        mags = [0.03, 0.06, 0.01, 0.08, 0.02, 0.05]
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 5000, 'ftol': 1e-20, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    np.random.seed(seed)
    
    # Get parent
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(400, 0.5)
        n_base = 400
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick polish at base resolution (30s)
    t1 = min(t0 + 30, deadline - 900)
    if time.time() < t1 - 5:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(200, 50.0), (1000, 500.0), (5000, 5000.0), (20000, 50000.0)])
    
    # Phase 2: Generate multiple initializations at n=1000 (5s)
    n_mid = 1000
    candidates_mid = []
    
    # Candidate 1: interpolated parent
    h_cand1 = interpolate_to(best_h, n_base, n_mid)
    candidates_mid.append(h_cand1)
    
    # Candidate 2: block pattern
    h_cand2 = make_binary_init(n_mid, 20, seed + 1)
    candidates_mid.append(h_cand2)
    
    # Candidate 3: another block pattern
    h_cand3 = make_binary_init(n_mid, 14, seed + 2)
    candidates_mid.append(h_cand3)
    
    # Candidate 4: smooth bumps
    h_cand4 = make_symmetric_init(n_mid, 2, seed + 3)
    candidates_mid.append(h_cand4)
    
    # Score all candidates
    for hc in candidates_mid:
        c = compute_c5(hc, n_mid)
        if c < best_c:
            best_c = c
            best_h = hc.copy()
    
    # Phase 3: Optimize top-2 best candidates at n=1000 (200s)
    scored = [(compute_c5(hc, n_mid), i) for i, hc in enumerate(candidates_mid)]
    scored.sort()
    top_indices = [scored[0][1], scored[1][1]]
    
    t3_end = min(t0 + 250, deadline - 600)
    for ci in top_indices:
        if time.time() > t3_end - 30:
            break
        hc = candidates_mid[ci]
        phase_deadline = min(time.time() + 90, t3_end)
        hc_opt, hc_c = run_lbfgs(hc, n_mid, phase_deadline,
            configs=[(200, 50.0), (1000, 500.0), (5000, 5000.0),
                     (20000, 50000.0), (50000, 200000.0)])
        if hc_c < best_c:
            best_c = hc_c
            best_h = hc_opt.copy()
    
    # Phase 4: Upsample to n=2000 and optimize (300s)
    n_high = 2000
    t4_start = time.time()
    if time.time() < deadline - 350:
        h_high = interpolate_to(best_h, n_base if len(best_h) != n_mid else n_mid, n_high)
        t4_end = min(t4_start + 180, deadline - 170)
        h_high, c_high = run_lbfgs(h_high, n_high, t4_end,
            configs=[(100, 20.0), (500, 200.0), (2000, 2000.0),
                     (5000, 10000.0), (20000, 50000.0), (50000, 200000.0),
                     (100000, 500000.0)])
        if c_high < best_c:
            best_c = c_high
            best_h = h_high.copy()
    
    # Phase 5: SLSQP polish at high resolution (80s)
    if time.time() < deadline - 120:
        t5 = min(time.time() + 70, deadline - 40)
        h5, c5 = slqp_polish(best_h, len(best_h), t5, alpha=10000000, maxiter=5000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Subgradient optimization (80s)
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 20)
        h6, c6 = subgradient_optimize(best_h, len(best_h), t6, n_steps=15000, top_k=10)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Coordinate descent (40s)
    if time.time() < deadline - 20:
        h7, c7 = coordinate_descent(best_h, len(best_h), deadline - 5, n_rounds=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Final SLSQP if time
    if time.time() < deadline - 10:
        h8, c8 = slqp_polish(best_h, len(best_h), deadline - 2, alpha=20000000, maxiter=2000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
