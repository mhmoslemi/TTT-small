def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:

            h = np.clip(h, 0.0, 1.0)

            if tk == active_idx:
                continue
            kk2 = tk - (n - 1)
            w2 = (c_vec[tk] - m + 1.0) / (c_vec.std() + 1e-15)
            w2 = max(0.0, min(0.5, w2))
            if k_worst >= 0:
                nv = n - kk2
            else:
                kk2 = -kk2
                nv = n - kk2
            if kk2 >= 0:
                nv = n - kk2
                if nv > 0:
                    grad_dir[:nv] += w2 * (1.0 - best_h[kk2:])
                    grad_dir[kk2:] += w2 * (-best_h[:nv])
            else:
                nv = n - kk2
                if nv > 0:
                    grad_dir[nv:] += w2 * (1.0 - best_h[:n-nv])
                    grad_dir[:n-nv] += w2 * (-best_h[nv:])
        # Find top coordinates to try
        neg_grad = -grad_dir
        top_coords = np.argsort(-neg_grad)[:n_try]
        improved = False
        for j in top_coords:
            if time.time() > deadline:
                break
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.003, -0.003]:
                nv = best_h[j] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = nv
                # Redistribute to maintain sum
                diff = best_h.sum() - trial.sum()
                others = np.delete(np.arange(n), j)
                ovals = trial[others]
                if diff > 0:
                    room = 1.0 - ovals
                    tot = room.sum()
                    if tot < 1e-14:
                        continue
                    ovals += (diff / tot) * room
                else:
                    tot = ovals.sum()
                    if tot < 1e-14:
                        continue
                    ovals += (diff / tot) * ovals
                ovals = np.clip(ovals, 0, 1)
                trial[others] = ovals
                c_trial = compute_c5(trial, n)
                if c_trial < best_c - 1e-15:
                    best_c = c_trial
                    best_h = trial.copy()
                    improved = True
                    break
        if not improved:
            # Try smaller steps
            for j in top_coords[:50]:
                if time.time() > deadline:
                    break
                for delta in [0.0005, -0.0005, 0.0002, -0.0002]:
                    nv = best_h[j] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    trial = best_h.copy()
                    trial[j] = nv
                    diff = best_h.sum() - trial.sum()
                    others = np.delete(np.arange(n), j)
                    ovals = trial[others]
                    if diff > 0:
                        room = 1.0 - ovals
                        tot = room.sum()
                        if tot < 1e-14:
                            continue
                        ovals += (diff / tot) * room
                    else:
                        tot = ovals.sum()
                        if tot < 1e-14:
                            continue
                        ovals += (diff / tot) * ovals
                    ovals = np.clip(ovals, 0, 1)
                    trial[others] = ovals
                    c_trial = compute_c5(trial, n)
                    if c_trial < best_c - 1e-15:
                        best_c = c_trial
                        best_h = trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved and rnd > 50:
                break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_refine(h, n, deadline, n_trials=6):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(9999)
    bounds = [(0.0, 1.0)] * n
    mags = [0.015, 0.03, 0.02, 0.04, 0.01, 0.025]
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = best_h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0 and diff > 0:
                h_b[i] = 1.0
                diff -= 1.0
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0 and diff < 0:
                h_b[i] = 0.0
                diff += 1.0
    return project_feasible(h_b, n)

def active_set_refine(h, n, deadline):
    """Focus optimization on reducing the top few overlap values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    remaining = deadline - time.time()
    step = min(remaining / 200, 2.0)
    for it in range(int(remaining / step)):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        # Top 3 active lags
        top3 = np.argsort(c_vec)[-3:]
        for idx in top3:
            k = idx - (n - 1)
            # Compute which h values are involved
            if k >= 0:
                involved = set(range(0, n - k)) | set(range(k, n))
            else:
                kk = -k
                involved = set(range(0, n - kk)) | set(range(kk, n))
            # Try adjusting the most "interior" values first
            arr = sorted(involved, key=lambda j: -min(best_h[j], 1 - best_h[j]))
            for j in arr[:30]:
                for delta in [0.003, -0.003, 0.001, -0.001]:
                    nv = best_h[j] + delta
                    if nv < 0 or nv > 1:
                        continue
                    trial = best_h.copy()
                    trial[j] = nv
                    diff_s = (best_h[j] - nv)
                    # Redistribute
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    if diff_s > 0:
                        trial[mask] -= diff_s / mask.sum()
                    else:
                        trial[mask] -= diff_s / mask.sum()
                    trial = np.clip(trial, 0, 1)
                    trial = project_feasible(trial, n)
                    c_t = compute_c5(trial, n)
                    if c_t < best_c - 1e-16:
                        best_c = c_t
                        best_h = trial
                        break
                if best_c < compute_c5(h, n) - 1e-14:
                    break
            if best_c < compute_c5(h, n) - 1e-14:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick L-BFGS at base resolution
    t1 = min(t0 + 60, deadline - 900)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(100, 10.0), (500, 50.0), (2000, 200.0),
                     (5000, 1000.0), (15000, 5000.0)])
    
    # Phase 2: SLSQP at base
    t2 = min(t0 + 100, deadline - 850)
    if t2 > time.time() + 10:
        h2, c2 = slqp_polish(best_h, n_base, t2, alpha=50000, maxiter=2000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n1 = 1000
    t3_start = time.time()
    if t3_start < deadline - 500:
        h1000 = interpolate_to(best_h, n_base, n1)
        c1000 = compute_c5(h1000, n1)
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        t3 = min(t3_start + 200, deadline - 350)
        if t3 > time.time() + 15:
            h1000, c1000 = run_lbfgs(h1000, n1, t3,
                configs=[(200, 20.0), (1000, 100.0), (3000, 500.0),
                         (10000, 2000.0), (30000, 10000.0),
                         (100000, 30000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
    
    # Phase 4: SLSQP at n=1000
    t4 = time.time()
    if t4 < deadline - 300:
        t4_end = min(t4 + 80, deadline - 200)
        h4, c4 = slqp_polish(best_h, len(best_h), t4_end, alpha=100000, maxiter=3000)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Coordinate descent at n=1000
    t5 = time.time()
    if t5 < deadline - 250:
        t5_end = min(t5 + 80, deadline - 150)
        best_h, best_c = coord_descent(best_h, len(best_h), t5_end,
                                       n_rounds=300, n_try=150)
    
    # Phase 6: Perturbation at n=1000
    t6 = time.time()
    if t6 < deadline - 200:
        t6_end = min(t6 + 60, deadline - 120)
        best_h, best_c = perturbation_refine(best_h, len(best_h), t6_end, n_trials=5)
    
    # Phase 7: Upsample to n=2000
    n2 = 2000
    t7_start = time.time()
    if t7_start < deadline - 150:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        c2000 = compute_c5(h2000, n2)
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        else:
            best_h = h2000  # still use higher res for further optimization
        t7 = min(t7_start + 120, deadline - 60)
        if t7 > time.time() + 15:
            h2000, c2000 = run_lbfgs(h2000, n2, t7,
                configs=[(500, 50.0), (2000, 500.0), (8000, 2000.0),
                         (30000, 8000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
    
    # Phase 8: SLSQP at final resolution
    t8 = time.time()
    nf = len(best_h)
    if t8 < deadline - 40:
        t8_end = min(t8 + 50, deadline - 15)
        h8, c8 = slqp_polish(best_h, nf, t8_end, alpha=200000, maxiter=1500)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Final coordinate descent
    t9 = time.time()
    if t9 < deadline - 15:
        best_h, best_c = coord_descent(best_h, len(best_h), deadline - 5,
                                       n_rounds=200, n_try=100)
    
    # Final feasibility
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    
    return best_h, best_c, nf
