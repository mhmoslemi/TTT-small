    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:

            tot = room.sum()

            if tot < 1e-15:

def subgradient_descent(h, n, dx, deadline, n_steps=50000, lr_init=0.01, decay=0.9999):
    """Momentum-based subgradient on the max of correlation values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    mu_mom = 0.9
    lr = lr_init
    idx_center = n - 1
    
    for step in range(n_steps):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Active set: top lags within tolerance
        tol = max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        # Cap active set size
        if len(active) > 30:
            order = np.argsort(-c_vec[active])[:30]
            active = active[order]
        
        # Compute subgradient (average over active)
        grad = np.zeros(n)
        for mm in active:
            k = int(mm) - idx_center
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += dx * (1.0 - h[k:n])
                    grad[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += dx * (1.0 - h[0:nv])
                    grad[0:nv] += dx * (-h[kk:n])
        grad /= len(active)
        
        # Momentum update
        v = mu_mom * v - lr * grad
        h_new = h + v
        h = project_feasible(h_new, n)
        
        # Check improvement
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Decay learning rate
        lr *= decay
        if lr < 1e-10:
            break
    
    return best_h, best_c
