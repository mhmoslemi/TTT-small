def make_beatty(n, frac_ones):
    """Create a Beatty-sequence-like initialization"""
    h = np.zeros(n)
    # Use golden ratio for good distribution
    alpha = (1 + 5**0.5) / 2  # golden ratio
    # Beatty sequence: positions floor(k * alpha) mod n
    seen = set()
    count = 0
    k = 0
    target = n // 2
    while count < target and k < 10 * n:
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
            count += 1
        k += 1
    return h

def lbfgs_obj(x, n, alpha, mu):

    g = h_grad_from_w(w, h, n)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def binary_swap_search(h, n, deadline, rng, max_iters=500000):
    """Aggressive binary local search with single and double swaps"""
    h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    stagnation = 0
    max_stagnation = 5000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Recompute gradient every 20 iterations
        if it % 20 == 0:
            alpha = 100000.0
            w = smooth_max_w(c_vec, alpha)
            grad = h_grad_from_w(w, h, n)
        
        improved = False
        
        # Single swap: guided by gradient
        if n_ones > 0 and n_zeros > 0:
            # Get top candidates
            g_ones = np.array([grad[i] for i in ones])
            g_zeros = np.array([grad[j] for j in zeros])
            
            # Remove from high gradient, add to low gradient
            top_rm = np.argsort(-g_ones)[:min(30, n_ones)]
            top_ad = np.argsort(g_zeros)[:min(30, n_zeros)]
            
            for ti in top_rm:
                if improved:
                    break
                i = ones[ti]
                for tj in top_ad:
                    j = zeros[tj]
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        c_vec = compute_c5_vec(h, n)
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                        improved = True
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
        
        if not improved:
            # Try random single swaps
            for _ in range(10):
                if time.time() > deadline:
                    break
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    c_vec = compute_c5_vec(h, n)
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            # Try double swap
            if n_ones >= 2 and n_zeros >= 2:
                for _ in range(5):
                    if time.time() > deadline:
                        break
                    i1 = ones[rng.randint(n_ones)]
                    i2 = ones[rng.randint(n_ones)]
                    if i1 == i2:
                        continue
                    j1 = zeros[rng.randint(n_zeros)]
                    j2 = zeros[rng.randint(n_zeros)]
                    if j1 == j2:
                        continue
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        c_vec = compute_c5_vec(h, n)
                        ones.remove(i1)
                        ones.remove(i2)
                        ones.extend([j1, j2])
                        zeros.remove(j1)
                        zeros.remove(j2)
                        zeros.extend([i1, i2])
                        improved = True
                        break
                    else:
                        h[i1] = 1.0
                        h[i2] = 1.0
                        h[j1] = 0.0
                        h[j2] = 0.0
        
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            
            # If stuck for too long, do a random perturbation
            if stagnation > max_stagnation:
                num_perturb = rng.randint(3, 10)
                for _ in range(num_perturb):
                    i = ones[rng.randint(n_ones)]
                    j = zeros[rng.randint(n_zeros)]
                    h[i], h[j] = h[j], h[i]
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                best_c = compute_c5(h, n)
                c_vec = compute_c5_vec(h, n)
                stagnation = 0
    
    return h, best_c

    zeros = np.where(h < 0.5)[0].tolist()
    T = T0
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    while time.time() < deadline and T > 0.001:
