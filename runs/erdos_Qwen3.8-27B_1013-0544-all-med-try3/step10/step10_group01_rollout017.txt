    if smx > 400:

    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    # Add small penalty for constraint violation
    val += 0.001 * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha):

    vals = c[active]

    smx = sh.max()

def coord_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    # Try small perturbations
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        while time.time() < deadline:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            # Compute gradient for active set
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx
                    grad[k:n] += (-best_h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                    grad[0:nv] += (-best_h[kk:n]) * dx
            grad /= len(active)
            # Find best pair to swap
            pos_mask = (grad > 1e-8) & (best_h > 1e-8)
            neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            # Try top candidates
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                break  # No improvement at this step size, go to next
    return best_h, best_c

def pair_swap_search(h, n, deadline, deltas=None, top_radius=30):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in ones[::3]:  # Sample every 3rd one
                if time.time() > deadline:
                    break
                for j in zeros[::3]:
                    if i == j:
                        continue
                    # Try moving weight from i to j
                    if best_h[i] >= d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def upsample(h, n_old, n_new):
    """Upsample by duplication"""
    h_new = np.zeros(n_new)
    for i in range(n_old):
        h_new[2*i] = h[i]
        h_new[2*i+1] = h[i]
    return h_new

def make_beatty(n, alpha_param):

            if h.sum() >= n_ones:

            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx
                    grad[k:n] += (-best_h[0:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                    grad[0:nv] += (-best_h[kk:n]) * dx
            grad /= len(active)
            pos_mask = (grad > 1e-8) & (best_h > 1e-8)
            neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                gi = grad[i]
                for j in neg_idx[:40]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                no_improve_count = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                no_improve_count += 1
    return best_h, best_c

def binary_flip_search(h, n, deadline, max_rounds=30):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(best_h > 0.3)[0]
        zeros = np.where(best_h < 0.7)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:min(len(ones), 200)]:
            if time.time() > deadline:
                break
            for j in zeros[:min(len(zeros), 100)]:
                if i == j:
                    continue
                save_i = best_h[i]
                save_j = best_h[j]
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = save_i
                    best_h[j] = save_j
                    best_h = project_feasible(best_h, n)
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        h0 = make_beatty(500, golden / 2.0)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick polish at original resolution (40s)
    if time.time() < deadline - 40:
        p1_end = min(t0 + 40, deadline - 40)
        bounds0 = [(0.0, 1.0)] * n0
        for alpha in [500, 5000, 50000, 200000]:
            if time.time() > p1_end:
                break
            try:
                res = minimize(full_smooth_obj, best_h, jac=full_smooth_grad,
                               args=(n0, alpha, 0.01), method='L-BFGS-B',
                               bounds=bounds0,
                               options={'maxiter': 400, 'ftol': 1e-26,
                                        'gtol': 1e-19, 'maxfun': 4000,
                                        'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n0)
                c_new = compute_c5(h_new, n0)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Quick active set at original resolution
    if time.time() < deadline - 30:
        p1b_end = min(time.time() + 20, deadline - 30)
        h_a, c_a = lbfgs_active_multi(best_h, n0, p1b_end,
                                       alphas=[50000, 500000, 2000000],
                                       top_k_list=[5, 8, 12],
                                       rounds=4, mu=0.001)
        if c_a < best_c:
            best_c = c_a
            best_h = h_a
            best_n = n0

    # Phase 2: Upsample to n=1000 and optimize
    n1 = 1000
    h1 = upsample_dup(best_h, best_n, n1)
    h1 = project_feasible(h1, n1)
    c1 = compute_c5(h1, n1)

    # Also try: if original was already 1000, just use it
    if best_n == n1:
        h1 = best_h.copy()
        c1 = best_c

    # Also generate fresh starts at n1
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts_n1 = [h1]
    starts_n1.append(make_beatty(n1, golden / 2.0))
    starts_n1.append(make_beatty(n1, 1.0 / (1.0 + np.sqrt(2.0))))
    starts_n1.append(make_random_binary(n1, rng))

    # Find best start at n1
    for hs in starts_n1:
        cs = compute_c5(hs, n1)
        if cs < c1:
            c1 = cs
            h1 = hs.copy()

    bounds1 = [(0.0, 1.0)] * n1

    # Phase 3a: Full smooth max L-BFGS at n1 (60s)
    if time.time() < deadline - 300:
        p3a_end = min(time.time() + 60, deadline - 300)
        for alpha in [200, 2000, 10000, 50000]:
            if time.time() > p3a_end:
                break
            try:
                res = minimize(full_smooth_obj, h1, jac=full_smooth_grad,
                               args=(n1, alpha, 0.01), method='L-BFGS-B',
                               bounds=bounds1,
                               options={'maxiter': 400, 'ftol': 1e-26,
                                        'gtol': 1e-19, 'maxfun': 4000,
                                        'maxls': 30, 'maxcor': 50})
                h_new = project_feasible(res.x, n1)
                c_new = compute_c5(h_new, n1)
                if c_new < c1:
                    c1 = c_new
                    h1 = h_new
            except Exception:
                pass
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            best_n = n1

    # Phase 3b: Active set optimization at n1 (200s)
    if time.time() < deadline - 250:
        p3b_end = min(time.time() + 200, deadline - 250)
        h1, c1 = lbfgs_active_multi(h1, n1, p3b_end,
                                     alphas=[20000, 100000, 500000, 2000000, 10000000],
                                     top_k_list=[4, 6, 8, 12, 16, 20],
                                     rounds=10, mu=0.001)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            best_n = n1

    # Phase 3c: Coordinate descent at n1 (120s)
    if time.time() < deadline - 150:
        p3c_end = min(time.time() + 120, deadline - 150)
        h1, c1 = coord_descent_fast(h1, n1, p3c_end,
                                     steps=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            best_n = n1

    # Phase 3d: More active set rounds
    if time.time() < deadline - 100:
        p3d_end = min(time.time() + 80, deadline - 100)
        h1, c1 = lbfgs_active_multi(h1, n1, p3d_end,
                                     alphas=[100000, 1000000, 5000000],
                                     top_k_list=[6, 10, 14],
                                     rounds=6, mu=0.0005)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
            best_n = n1

    # Phase 4: Try polishing other starts at n1
    if time.time() < deadline - 80:
        p4_end = min(time.time() + 60, deadline - 80)
        for hs in starts_n1[1:]:
            if time.time() > p4_end:
                break
            c_hs = compute_c5(hs, n1)
            if c_hs < best_c:
                best_c = c_hs
                best_h = hs.copy()
                best_n = n1
            # Quick optimization
            sub_deadline = min(time.time() + 20, p4_end)
            h_opt, c_opt = lbfgs_active_multi(hs, n1, sub_deadline,
                                               alphas=[100000, 1000000],
                                               top_k_list=[6, 10],
                                               rounds=3, mu=0.001)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n1

    # Phase 5: Fine coordinate descent on best solution
    if time.time() < deadline - 30:
        nt = len(best_h)
        p5_end = min(time.time() + 30, deadline - 25)
        best_h, best_c = coord_descent_fast(best_h, nt, p5_end,
                                             steps=[0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    # Phase 6: Final L-BFGS polish
    if time.time() < deadline - 10:
        nt = len(best_h)
        bounds_t = [(0.0, 1.0)] * nt
        for alpha in [5000000, 20000000]:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(full_smooth_obj, best_h, jac=full_smooth_grad,
                               args=(nt, alpha, 0.01), method='L-BFGS-B',
                               bounds=bounds_t,
                               options={'maxiter': 200, 'ftol': 1e-28,
                                        'gtol': 1e-20, 'maxfun': 2000})
                h_new = project_feasible(res.x, nt)
                c_new = compute_c5(h_new, nt)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Very fine coordinate descent
    if time.time() < deadline - 3:
        nt = len(best_h)
        best_h, best_c = coord_descent_fast(best_h, nt, deadline - 2,
                                             steps=[0.00005, 0.00002, 0.00001, 0.000005])

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
