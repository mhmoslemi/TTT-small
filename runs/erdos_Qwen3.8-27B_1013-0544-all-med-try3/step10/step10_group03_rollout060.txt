    target = n / 2.0

def grad_weighted(h, n, w):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k_idx in range(ncorr):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def softmax_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w_max = (sh == mx)
        w_max /= w_max.sum()
        return w_max
    w /= s
    return w

def smooth_max_value(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def fista_opt(h, n, alpha, lr, niter, deadline, mu=0.05):
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    stale = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        h_y = h_curr + ((t - 1.0) / (t + 1.0)) * (h_curr - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        c_y = compute_c5_vec(h_y, n)
        w_y = softmax_weights(c_y, alpha)
        g_y = grad_weighted(h_y, n, w_y)
        g_y += 2.0 * mu * (h_y.sum() - n / 2.0)
        h_curr = project_feasible(h_y - lr * g_y, n)
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h_curr
        c_val = compute_c5(h_curr, n)
        if c_val < best_c - 1e-15:
            best_c = c_val
            best_h = h_curr.copy()
            stale = 0
        else:
            stale += 1
        if stale > 2000:
            break
    return best_h, best_c

def lbfgs_smooth(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max_value(c, alpha)
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    def grad(x):
        c = compute_c5_vec(x, n)
        w = softmax_weights(c, alpha)
        g = grad_weighted(x, n, w)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def top_shifts_lbfgs(h, n, top_k, alpha, mu, maxiter, deadline):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-6, 0.002 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = active[np.argsort(-c_vec[active])][:top_k]
    if len(active) == 0:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        vals = np.array([c[i] for i in active])
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        if smx > 500:
            val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
        else:
            val = mx + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    def grad(x):
        c = compute_c5_vec(x, n)
        vals = np.array([c[i] for i in active])
        mx = vals.max()
        sh = alpha * (vals - mx)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for wi, idx in zip(w, active):
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fine_pair_search(h, n, deadline, center_radius=30):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - center_radius)
                hi = min(n, i + center_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def random_restart_opt(h, n, deadline, n_restarts=6, seed_val=42):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed_val)
    per_restart = (deadline - time.time()) / n_restarts
    for r in range(n_restarts):
        t_deadline = time.time() + per_restart
        if time.time() > deadline - 5:
            break
        noise_scale = 0.02 + 0.02 * r
        h_pert = h + rng.normal(0, noise_scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        for alpha, lr, ni in [(500, 0.02, 800), (5000, 0.005, 600), (50000, 0.0005, 400)]:
            if time.time() > t_deadline - 2:
                break
            h_pert, c_pert = fista_opt(h_pert, n, alpha, lr, ni, t_deadline - 2)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
    return best_h, best_c

def multiresolution_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 20:
            continue
        t_rem = deadline - time.time()
        if t_rem < 30:
            break
        x_fine = np.linspace(0, 2, n)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        t_c = min(t_rem * 0.3, 60)
        for alpha, mu in [(200, 100.0), (2000, 1000.0)]:
            if time.time() > time.time() + t_c - 5:
                break
            h_c, c_c = lbfgs_smooth(h_c, n_coarse, alpha, mu, 2000, time.time() + min(25, t_c))
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        for alpha, lr, ni in [(1000, 0.01, 500)]:
            if time.time() < deadline - 10:
                h_f, c_f = fista_opt(h_f, n, alpha, lr, ni, deadline - 10)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    n = n_orig
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Quick L-BFGS polish (50s)
    t1 = min(t0 + 50, deadline - 700)
    if t1 > t0 + 5:
        for alpha, mu in [(200, 50.0), (2000, 500.0), (20000, 5000.0)]:
            if time.time() > t1 - 3:
                break
            h_new, c_new = lbfgs_smooth(best_h, n, alpha, mu, 3000, t1 - 3)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 2: FISTA with alpha schedule (100s)
    t2 = min(t0 + 150, deadline - 600)
    if t2 > time.time() + 10:
        for alpha, lr in [(200, 0.05), (2000, 0.01), (20000, 0.002)]:
            if time.time() > t2 - 5:
                break
            niter = int((t2 - 5 - time.time()) * 60)
            niter = max(niter, 200)
            h_new, c_new = fista_opt(best_h, n, alpha, lr, niter, t2 - 5)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 3: Random restarts (250s)
    t3 = min(t0 + 400, deadline - 350)
    if t3 > time.time() + 30:
        best_h, best_c = random_restart_opt(best_h, n, t3, n_restarts=8, seed_val=seed)
    
    # Phase 4: Top-shift focused L-BFGS (100s)
    t4 = min(t0 + 500, deadline - 250)
    if t4 > time.time() + 10:
        for top_k in [3, 5]:
            for alpha in [5000, 50000, 500000]:
                if time.time() > t4 - 5:
                    break
                h_new, c_new = top_shifts_lbfgs(best_h, n, top_k, alpha, 10.0, 2000, t4 - 5)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    
    # Phase 5: FISTA fine-tune (60s)
    t5 = min(t0 + 560, deadline - 180)
    if t5 > time.time() + 10:
        for alpha, lr in [(50000, 0.0005), (500000, 0.00005)]:
            if time.time() > t5 - 5:
                break
            niter = int((t5 - 5 - time.time()) * 80)
            niter = max(niter, 100)
            h_new, c_new = fista_opt(best_h, n, alpha, lr, niter, t5 - 5)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 6: Fine pair search (40s)
    t6 = min(t0 + 600, deadline - 140)
    if t6 > time.time() + 10:
        best_h, best_c = fine_pair_search(best_h, n, t6, center_radius=25)
    
    # Phase 7: Try n=1000 (300s)
    t7_start = time.time()
    t7 = deadline - 10
    if t7 - t7_start > 80:
        n2 = 1000
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n2)
        c_new = compute_c5(h_new, n2)
        
        # L-BFGS at n=1000
        t7a = min(t7_start + 120, t7 - 120)
        if t7a > time.time() + 10:
            for alpha, mu in [(500, 100.0), (5000, 1000.0)]:
                if time.time() > t7a - 5:
                    break
                h_new, c_new = lbfgs_smooth(h_new, n2, alpha, mu, 2000, t7a - 5)
        
        # FISTA at n=1000
        t7b = min(t7_start + 220, t7 - 50)
        if t7b > time.time() + 10:
            for alpha, lr in [(2000, 0.01), (20000, 0.002)]:
                if time.time() > t7b - 5:
                    break
                niter = int((t7b - 5 - time.time()) * 20)
                niter = max(niter, 100)
                h_new, c_new = fista_opt(h_new, n2, alpha, lr, niter, t7b - 5)
        
        # Top-shift at n=1000
        t7c = min(t7_start + 280, t7 - 10)
        if t7c > time.time() + 10:
            for alpha in [10000, 100000]:
                if time.time() > t7c - 5:
                    break
                h_new, c_new = top_shifts_lbfgs(h_new, n2, 4, alpha, 10.0, 1500, t7c - 5)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            n = n2
    
    # Phase 8: Final polish
    if time.time() < deadline - 5:
        h_new, c_new = fine_pair_search(best_h, n, deadline - 3, center_radius=20)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
