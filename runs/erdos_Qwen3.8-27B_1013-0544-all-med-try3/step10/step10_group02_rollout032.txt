        # Adjust to have n/2 ones
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            for i in range(n):
                if h[i] < 1.0:
                    add = min(1.0 - h[i], diff)
                    h[i] += add
                    diff -= add
                    if diff < 1e-10:
                        break
        elif diff < 0:
            for i in range(n):
                if h[i] > 0.0:
                    sub = min(h[i], abs(diff))
                    h[i] -= sub
                    diff += sub
                    if diff > -1e-10:
                        break
    else:
        # p < n, tile the pattern
        h = np.tile(h_p, n // p + 1)[:n].copy()
        h = project_feasible(h, n)
    
    return project_feasible(h, n)

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        h[ones[0]] = 0.5  # to make sum = n/2
    return project_feasible(h, n)

def interp_h(h_from, n_from, n_to):
    """Interpolate h from n_from points to n_to points."""
    x_from = np.linspace(0, 2, n_from)
    x_to = np.linspace(0, 2, n_to)
    h_to = np.interp(x_to, x_from, h_from)
    return project_feasible(h_to, n_to)

def subgradient_with_backtrack(h, n, deadline, eta0=0.005, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = eta0
    dx = 2.0 / n
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        
        if c_max < best_c - 1e-13:
            best_c = c_max
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5000:
                break
        
        # Active set
        tol = max(1e-8, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Backtracking
        accepted = False
        for bt in range(15):
            h_new = h - eta * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_max * (1 + 1e-10):
                h = h_new
                accepted = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                eta = min(eta * 1.01, 0.03)
                break
            eta *= 0.5
        
        if not accepted:
            eta *= 0.8
            if eta < 1e-10:
                break
    
    return best_h, best_c

def coord_descent(h, n, deadline, n_passes=200):
    """Coordinate descent with projection."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Random order for diversity
        order = np.random.permutation(n)
        
        for i in order:
            if time.time() > deadline:
                break
            old = best_h[i]
            best_local = best_c
            best_val = old
            
            # Try candidate values
            cands = []
            # Grid around current
            for delta in np.arange(-0.3, 0.31, 0.05):
                v = old + delta
                if 0 < v < 1:
                    cands.append(v)
            # Extremes
            if old > 0.01:
                cands.append(0.0)
            if old < 0.99:
                cands.append(1.0)
            # Midpoint
            cands.append(0.5)
            
            for val in cands:
                best_h[i] = val
                proj = project_feasible(best_h, n)
                c_new = compute_c5(proj, n)
                if c_new < best_local:
                    best_local = c_new
                    best_val = val
                    best_h = proj.copy()
                best_h[i] = old  # restore (will re-project later)
            
            if best_val != old:
                best_h[i] = best_val
                best_h = project_feasible(best_h, n)
                if best_local < best_c:
                    best_c = best_local
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c

def pair_search_fast(h, n, deadline, max_rounds=100):
    """Fast pair search: move mass between pairs of coordinates."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        for _ in range(2000):
            if time.time() > deadline:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            d = np.random.choice([0.001, 0.005, 0.01, 0.02, 0.05])
            
            # Move from j to i
            if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                best_h[i] += d
                best_h[j] -= d
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] -= d
                    best_h[j] += d
            
            # Move from i to j
            if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                best_h[i] -= d
                best_h[j] += d
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] += d
                    best_h[j] -= d
        
        if not improved:
            break
    
    return best_h, best_c
