def full_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-16:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_smooth_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_smooth_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def constraint_fun(x, n):
    return float(x.sum() - n / 2.0)

def constraint_jac(x, n):
    return np.ones(n)

def slqp_smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slqp_smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)

    mx = sh.max()

def run_lbfgs_anneal(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu, maxiter in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(full_smooth_obj, h, jac=full_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxiter * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_topk_anneal(h, n, deadline, top_k=8, alphas=None):
    if alphas is None:
        alphas = [2000, 10000, 50000, 200000, 1000000, 5000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mu_val = 0.01
    
    for _ in range(30):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(0.0005, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-c_vec)[:top_k]
        elif len(active) > top_k:
            top_active = active[np.argsort(-c_vec[active])][:top_k]
            active = top_active
        
        for alpha in alphas:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(topk_smooth_obj, h, jac=topk_smooth_grad,
                              args=(n, active, alpha, mu_val),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': 50000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def run_slqp_phase(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50000, 200000, 1000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(slqp_smooth_obj, h, jac=slqp_smooth_grad,
                          args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_local_search(h, n, deadline, max_radius=40):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        iters = 0
        while improved and iters < 200:
            improved = False
            iters += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - max_radius)
                hi = min(n, i + max_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = np.clip(h, 0.0, 1.0)
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, n)
    return h, best_c

def coord_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    target = n / 2.0
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in steps:
        if time.time() > deadline:
            break
        for _cpass in range(3):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                # Try increasing h[i] and decreasing another point
                if h[i] + d <= 1.0:
                    h[i] += d
                    # Compensate by decreasing the point with highest value
                    others = np.where((h > d + 1e-15) & (np.arange(n) != i))[0]
                    if len(others) > 0:
                        comp_idx = others[np.argmax(h[others])]
                        h[comp_idx] -= d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[comp_idx] += d
                            h = np.clip(h, 0.0, 1.0)
                if h[i] - d >= 0.0:
                    h[i] -= d
                    others = np.where((h < 1.0 - d - 1e-15) & (np.arange(n) != i))[0]
                    if len(others) > 0:
                        comp_idx = others[np.argmin(h[others])]
                        h[comp_idx] += d
                        h = np.clip(h, 0.0, 1.0)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[comp_idx] -= d
                            h = np.clip(h, 0.0, 1.0)
            if not improved:
                break
            h = project_feasible(h, n)
    return h, best_c

def gradient_driven_local(h, n, deadline):
    best_c = compute_c5(h, n)
    base_step = 0.005
    alpha = 50000
    
    for _outer in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(0.0005, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        
        # Average gradient over active shifts
        grad = np.zeros(n)
        for idx in active:
            grad += np.correlate(h, 1 - h, mode='full')[idx]  # placeholder
        # Use the actual gradient computation
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            dx = 2.0 / n
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Find best 2-point move
        sorted_idx = np.argsort(-np.abs(grad))[:80]
        found = False
        for ii in range(len(sorted_idx)):
            if time.time() > deadline:
                return h, best_c
            i = sorted_idx[ii]
            for jj in range(len(sorted_idx)):
                if ii == jj:
                    continue
                j = sorted_idx[jj]
                gi, gj = grad[i], grad[j]
                # We want to decrease the objective
                # If gi > 0, decreasing h[i] helps; if gi < 0, increasing h[i] helps
                for direction in [1, -1]:
                    for mult in [1.0, 0.5, 0.25, 0.125]:
                        step = base_step * mult
                        if direction == 1:
                            ni = h[i] - step
                            nj = h[j] + step
                        else:
                            ni = h[i] + step
                            nj = h[j] - step
                        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                            continue
                        h[i] = ni
                        h[j] = nj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                        else:
                            h[i] = ni if direction == 1 else h[i]
                            # Restore
                            h[i] = h[i] + (step if direction == 1 else -step)
                            h[j] = h[j] - (step if direction == 1 else -step)
                    if found:
                        break
            if found:
                break
        if not found:
            base_step *= 0.5
            if base_step < 1e-7:
                break
    return h, best_c

def multiresolution_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_fine = np.linspace(0, 2, n)
    
    for n_coarse in [max(50, n // 4), max(50, n // 2)]:
        if time.time() > deadline - 10:
            break
        x_coarse = np.linspace(0, 2, n_coarse)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        t_c = min(25, (deadline - time.time()) * 0.3)
        if t_c > 10:
            cfgs = [(500, 20.0, 2000), (2000, 200.0, 3000), (10000, 2000.0, 5000)]
            h_c, c_c = run_lbfgs_anneal(h_c, n_coarse, time.time() + t_c, cfgs)
        
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

                 (5000, 5000.0, 5000), (20000, 20000.0, 8000)]
        h, best_c = run_lbfgs_anneal(h, n_init, t1, cfgs1)
        best_h = h.copy()

    n_target = 1000
    x_old = np.linspace(0, 2, n_init)
    x_new = np.linspace(0, 2, n_target)
    h_hi = np.interp(x_new, x_old, h)
    h_hi = project_feasible(h_hi, n_target)
    c_hi = compute_c5(h_hi, n_target)
    
    if c_hi < best_c:
        best_c = c_hi
        best_h = h_hi.copy()
        h = h_hi.copy()
        n = n_target
    else:
        # Still use higher resolution for optimization
        h = h_hi.copy()
        n = n_target
        best_c = c_hi
    
    # Phase 3: Aggressive L-BFGS at high resolution
    if time.time() < deadline - 350:
        t3 = min(time.time() + 200, deadline - 350)

                 (20000, 5000.0, 8000), (100000, 50000.0, 10000),
                 (500000, 500000.0, 15000), (2000000, 2000000.0, 20000)]
        h, c3 = run_lbfgs_anneal(h, n, t3, cfgs3)

        h_m, c4 = multiresolution_refine(best_h, n, t4)
        if c4 < best_c:
            best_c = c4
            best_h = h_m.copy()
            h = h_m.copy()

    if time.time() < deadline - 200:
        t5 = min(time.time() + 120, deadline - 200)
        h_t, c5 = run_topk_anneal(best_h, n, t5, top_k=6,

        if c6 < best_c:
            best_c = c6
            best_h = h_s.copy()
            h = h_s.copy()
    
    # Phase 7: Another top-k round with tighter tolerance
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        h_t2, c7 = run_topk_anneal(best_h, n, t7, top_k=4,
                                    alphas=[50000, 200000, 1000000, 5000000])
        if c7 < best_c:
            best_c = c7

    # Phase 9: Final coordinate descent polish
    if time.time() < deadline - 5:
        h_cd, c9 = coord_descent(best_h, n, min(time.time() + 10, deadline - 3))

            best_c = c9

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
