def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_gradient(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    try:
        res = minimize(lbfgs_objective, h, jac=lbfgs_gradient, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-20, 'maxfun': maxit * 10,
                               'maxls': 80, 'maxcor': 60})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

def make_sturmian(n, slope):
    h = np.zeros(n)
    for k in range(n):
        if int((k + 1) * slope) - int(k * slope) == 1:
            h[k] = 1.0
    count = h.sum()
    diff = n / 2.0 - count
    if diff > 0:
        zeros = np.where(h == 0)[0]
        step = max(1, len(zeros) // max(1, int(np.ceil(diff))))
        for i in range(int(np.ceil(diff))):

        for i in range(int(np.ceil(-diff))):

            h3, c3 = partial_unbinarize(best_h_n, n, K=12, deadline=p3_dl, maxit=2000)
            if c3 < best_c_n:
                best_c_n = c3
                best_h_n = h3

        p4_dl = min(p3_dl + 25, deadline - 50)
        if time.time() < p4_dl:
            h4, c4 = coord_descent_fine(best_h_n, n, p4_dl,
                                        steps=[0.001, 0.0003, 0.0001, 0.00003, 0.00001],
                                        max_rounds=150)
            if c4 < best_c_n:
                best_c_n = c4
                best_h_n = h4

        try_update(best_h_n, best_c_n, n)

    # Focused refinement on best solution
    if best_overall_h is not None and time.time() < deadline - 30:
        bh = best_overall_h.copy()
        bn = best_overall_n
        bc = best_overall_c

        rc_dl = min(t0 + 80, deadline - 200)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > rc_dl - 5:
                break
            h_rc, c_rc = run_lbfgs(bh, bn, min(rc_dl, time.time() + 20), alpha=alpha, mu=0.0005, maxit=2000)
            if c_rc < bc:
                bc = c_rc
                bh = h_rc

        bh_bin = binarize(bh, bn)
        b1_dl = min(rc_dl + 40, deadline - 100)
        h_b1, c_b1 = binary_search_1swap(bh_bin, bn, min(b1_dl, time.time() + 50), rng, max_cands=45)
        if c_b1 < bc:
            bc = c_b1
            bh = h_b1
        h_b2, c_b2 = binary_search_2swap(bh_bin, bn, min(b1_dl, time.time() + 30), rng, k_size=14)
        if c_b2 < bc:
            bc = c_b2
            bh = h_b2

        um_dl = min(b1_dl + 30, deadline - 60)
        if time.time() < um_dl:
            for K in [8, 14, 20]:
                if time.time() > um_dl:
                    break
                h_um, c_um = partial_unbinarize(bh, bn, K=K, deadline=um_dl, maxit=3000)
                if c_um < bc:
                    bc = c_um
                    bh = h_um

        cd_dl = min(um_dl + 30, deadline - 20)
        if time.time() < cd_dl:
            h_cd, c_cd = coord_descent_fine(bh, bn, cd_dl,
                                            steps=[0.0003, 0.0001, 0.00003, 0.00001, 0.000003],
                                            max_rounds=250)
            if c_cd < bc:
                bc = c_cd
                bh = h_cd

        # Final Nelder-Mead on sensitive positions
        if time.time() < deadline - 5:
            c_vec = compute_c5_vec(bh, bn)
            alpha_sm = 500000.0
            w_sm = smooth_max_w(c_vec, alpha_sm)
            grad_sm = h_grad_from_w(w_sm, bh, bn)
            sens = np.argsort(-np.abs(grad_sm))[:15]
            h_nm, c_nm = partial_unbinarize(bh, bn, K=15, deadline=deadline - 2, maxit=2000)
            if c_nm < bc:
                bc = c_nm
                bh = h_nm

        # Ultra-fine final pass
        if time.time() < deadline - 2:
            h_final, c_final = coord_descent_fine(bh, bn, deadline - 1,
                                                  steps=[0.00001, 0.000003, 0.000001],
                                                  max_rounds=100)
            if c_final < bc:
                bc = c_final
                bh = h_final

        try_update(bh, bc, bn)

    if best_overall_h is None:
        best_overall_h = np.full(500, 0.5)
        best_overall_n = 500
        best_overall_c = compute_c5(best_overall_h, best_overall_n)

    best_overall_h = project_feasible(best_overall_h, best_overall_n)
    best_overall_c = compute_c5(best_overall_h, best_overall_n)

    return best_overall_h, best_overall_c, best_overall_n
