    """Gradient of correlation at shift_idx w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

            return 1  # non-decreasing

Let me also think about the "active set" more carefully. At the optimum, typically 2-3 shifts are at the maximum. If we know which ones, we can focus on them. But with the exact 1D minimax, we automatically handle all shifts (we minimize the max over ALL shifts, not just active ones). So we don't need to explicitly track the active set for the move itself, but we do need it for the gradient (to prioritize which pairs to try).

One more consideration: the sum constraint. In pair descent, each move (h[i]+t, h[j]-t) preserves the sum exactly. So we don't need to re-project.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
            remaining = diff
            for i in idx:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
                remaining -= add
                if remaining < 1e-15:
                    break
        else:
            idx = np.argsort(h)
            remaining = -diff
            for i in idx:
                sub = min(remaining, h[i])
                h[i] -= sub
                remaining -= sub
                if remaining < 1e-15:
                    break
    return h

def compute_grad_matrix(h, n):
    """G[p, s_idx] = d(c_s)/d(h[p]) for all p in [0,n), s_idx in [0, 2n-2]."""
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((n, m))
    for p in range(n):
        # s = 0, 1, ..., n-1 (positive and zero shifts), index n-1+s
        s = np.arange(n)
        # First term: (1-h[p-s]) valid when 0 <= p-s < n, i.e., s <= p
        t1 = np.zeros(n)
        mask1 = s <= p
        t1[mask1] = 1.0 - h[p - s[mask1]]
        # Second term: -h[p+s] valid when p+s < n, i.e., s < n-p
        t2 = np.zeros(n)
        mask2 = p + s < n
        t2[mask2] = -h[p + s[mask2]]
        G[p, n-1:n] = dx * (t1 + t2)
        
        # s = -1, -2, ..., -(n-1), i.e., r = 1, ..., n-1, index n-1-r
        r = np.arange(1, n)
        t1 = np.zeros(n-1)
        mask1 = p + r < n
        t1[mask1] = 1.0 - h[p + r[mask1]]
        t2 = np.zeros(n-1)
        mask2 = p >= r
        t2[mask2] = -h[p - r[mask2]]
        G[p, 0:n-1] = dx * (t1 + t2)
    return G

def solve_1d_minimax(c_base, d, t_lo, t_hi):
    """Find t in [t_lo, t_hi] minimizing max_k (c_base[k] + d[k]*t)."""
    if t_lo >= t_hi - 1e-15:
        return t_lo
    
    def subgrad_direction(t):
        """Returns -1 (decreasing), 0 (at min), +1 (increasing)."""
        vals = c_base + d * t
        mx = vals.max()
        # Active: vals within tolerance of max
        tol = 1e-12
        mask = vals >= mx - tol
        da = d[mask]
        if da.size == 0:
            return 0
        if da.min() > 1e-14:
            return 1  # increasing
        if da.max() < -1e-14:
            return -1  # decreasing
        return 0  # 0 in subgradient
    
    d_lo = subgrad_direction(t_lo)
    if d_lo >= 0:
        return t_lo
    d_hi = subgrad_direction(t_hi)
    if d_hi <= 0:
        return t_hi
    
    a, b = t_lo, t_hi
    for _ in range(80):
        mid = 0.5 * (a + b)
        d_mid = subgrad_direction(mid)
        if d_mid >= 0:
            b = mid
        else:
            a = mid
        if b - a < 1e-15:
            break
    return 0.5 * (a + b)

def pair_descent(h, n, deadline, max_iters=5000, n_cand_per_side=25):
    """Pair descent with exact 1D minimax."""
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        G = compute_grad_matrix(h, n)
        
        max_val = c_vec.max()
        # Active shifts (within 1% of max)
        threshold = max_val * 0.99
        active = np.where(c_vec >= threshold)[0]
        
        # Gradient = average over active shifts
        grad = G[:, active].mean(axis=1)
        
        # Candidate pairs
        sorted_idx = np.argsort(grad)
        nc = min(n_cand_per_side, n // 5)
        lows = sorted_idx[:nc]
        highs = sorted_idx[-nc:]
        
        best_improvement = 0
        best_i = best_j = -1
        best_t = 0.0
        
        # Try pairs from lows to highs
        for i in lows:
            for j in highs:
                if i == j:
                    continue
                t_lo = max(0.0, h[j] - 1.0, -h[i])
                t_hi = min(1.0 - h[i], h[j])
                if t_hi - t_lo < 1e-13:
                    continue
                d = G[i] - G[j]
                if np.max(np.abs(d)) < 1e-15:
                    continue
                t_opt = solve_1d_minimax(c_vec, d, t_lo, t_hi)
                new_max = np.max(c_vec + d * t_opt)
                imp = max_val - new_max
                if imp > best_improvement:
                    best_improvement = imp
                    best_i, best_j, best_t = i, j, t_opt
        
        # Also try some random pairs (both directions)
        rng = np.random.default_rng(it + 1)
        n_rand = min(50, n_cand_per_side * 4)
        for _ in range(n_rand):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for (ii, jj) in [(i, j), (j, i)]:
                t_lo = max(0.0, h[jj] - 1.0, -h[ii])
                t_hi = min(1.0 - h[ii], h[jj])
                if t_hi - t_lo < 1e-13:
                    continue
                d = G[ii] - G[jj]
                if np.max(np.abs(d)) < 1e-15:
                    continue
                t_opt = solve_1d_minimax(c_vec, d, t_lo, t_hi)
                new_max = np.max(c_vec + d * t_opt)
                imp = max_val - new_max
                if imp > best_improvement:
                    best_improvement = imp
                    best_i, best_j, best_t = ii, jj, t_opt
        
        if best_i < 0 or best_improvement < 1e-14:
            break
        
        h[best_i] += best_t
        h[best_j] -= best_t
        h = np.clip(h, 0, 1)
        
        c_vec = compute_c5_vec(h, n)
        new_max = c_vec.max()
        if new_max < best_c:
            best_c = new_max
    
    return h, best_c

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_smooth(h, n, deadline, alpha, maxiter=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c
