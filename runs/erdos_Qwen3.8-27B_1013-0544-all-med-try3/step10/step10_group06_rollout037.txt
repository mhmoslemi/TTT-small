Actually, I realize there's a subtlety. The subgradient method for minimizing a max of smooth functions is:
- At each point, the subdifferential of max_k f_k is the convex hull of {∇f_k : k in active set}
- The steepest descent direction is the point in the subdifferential with minimum norm

minimize ||∑ λ_k ∇f_k||² subject to ∑λ_k = 1, λ_k ≥ 0

def steepest_subgradient(h, n, K=5):
    """Compute the steepest descent direction for the max of correlations."""
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-K:]

    grads = []
    for k in top_idx:
        lag = k - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if lag == 0:
            g[:] = (1.0 - 2.0 * h) * dx
        elif lag > 0:
            g[lag:n] += (1.0 - h[0:n - lag]) * dx
            g[0:n - lag] += (-h[lag:n]) * dx
        else:
            idx = -lag
            g[0:n - idx] += (1.0 - h[idx:n]) * dx
            g[idx:n] += (-h[0:n - idx]) * dx
        grads.append(g)
    
    grads = np.array(grads)  # shape (K, n)

    # minimize ||grads.T @ lambda||^2 s.t. sum(lambda)=1, lambda>=0
    # This is equivalent to: find lambda that minimizes lambda^T G G^T lambda s.t. 1^T lambda = 1, lambda >= 0
    G = grads  # (K, n)
    Q = G @ G.T  # (K, K)

    # Use scipy
    from scipy.optimize import minimize as sp_min
    def qobj(lam):
        return lam @ Q @ lam
    def qgrad(lam):
        return 2 * Q @ lam
    constraints = [{'type': 'eq', 'fun': lambda lam: lam.sum() - 1, 'jac': lambda lam: np.ones(K)}]
    bounds = [(0, 1)] * K
    lam0 = np.ones(K) / K
    res = sp_min(qobj, lam0, jac=qgrad, method='SLSQP', bounds=bounds, constraints=constraints,
                 options={'maxiter': 100, 'ftol': 1e-12})

    lam = np.clip(lam, 0, 1)
    lam /= lam.sum()

            tot = room.sum()
            if tot < 1e-15:

            h += frac * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            frac = max(diff / tot, -0.4)
            h += frac * h

    for i, alpha in enumerate(alpha_list):
        remaining = deadline - time.time()
        if remaining < 180:
            break
        timeout = min(25 + i * 8, max(15, remaining * 0.25))
        maxit = 400 + i * 250
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + timeout, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient method with steepest direction
    if time.time() < deadline - 250:
        gd_deadline = min(deadline - 200, time.time() + 150)
        h_opt, c_opt = subgradient_method(best_h, n, gd_deadline, top_k=5,
                                           max_iter=200000, momentum=0.25)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Intensive local search
    if time.time() < deadline - 150:
        ls_deadline = min(deadline - 80, time.time() + 180)
        h_opt, c_opt = intensive_local_search(best_h, n, ls_deadline, top_k=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish with very high alpha
    if time.time() < deadline - 120:
        for alpha in [20000000, 100000000, 500000000]:
            if time.time() > deadline - 60:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 30, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Try higher resolution
    for n_try in [800, 1000, 1500]:
        if time.time() > deadline - 150:
            break
        remaining = deadline - 80 - time.time()
        if remaining < 80:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 20000, time.time() + min(30, remaining * 0.25),
                                maxiter=2000)
        h_opt = lbfgsb_optimize(h_opt, n_try, 1000000, time.time() + min(30, remaining * 0.25),
                                maxiter=3000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            h_opt, c_opt = subgradient_method(best_h, n_try,
                                              time.time() + min(25, remaining * 0.15),
                                              top_k=5, max_iter=50000, momentum=0.25)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h_opt, c_opt = intensive_local_search(best_h, n_try,
                                                  time.time() + min(25, remaining * 0.15),
                                                  top_k=8)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Another round of subgradient + local search at best resolution
    if time.time() < deadline - 40:
        n_final = len(best_h)
        h_opt, c_opt = subgradient_method(best_h, n_final,
                                          time.time() + min(50, (deadline - time.time()) * 0.3),
                                          top_k=7, max_iter=80000, momentum=0.2)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final local search
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt, c_opt = intensive_local_search(best_h, n_final, deadline - 3, top_k=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 100000000, deadline - 1, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
