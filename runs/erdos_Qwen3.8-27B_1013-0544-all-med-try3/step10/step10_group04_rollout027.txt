def compute_c5(h, n):

        if abs(diff) < 1e-14:

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(x, n)
    m = c.max()

    dx = 2.0 / n
