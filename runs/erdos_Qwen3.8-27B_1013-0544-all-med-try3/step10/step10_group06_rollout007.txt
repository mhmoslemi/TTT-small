    h = np.clip(h.copy(), 0.0, 1.0)

            break

        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()

        else:
            idx = -lag
            grad[idx:] += w[k] * (1.0 - h[:n-idx])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def full_smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w[k] * (1.0 - h[lag:])
            grad[lag:] += w[k] * (-h[:n - lag])
        else:
            idx = -lag
            grad[idx:] += w[k] * (1.0 - h[:n - idx])
            grad[:n - idx] += w[k] * (-h[idx:])
    return grad * dx

def focused_grad(h, n, lags, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag, w in zip(lags, weights):
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n - lag])
        else:
            idx = -lag
            grad[idx:] += w * (1.0 - h[:n - idx])
            grad[:n - idx] += w * (-h[idx:])
    return grad * dx

def lag_correlation(h, n, lag):
    if lag > 0:
        return np.sum(h[:n - lag] * (1.0 - h[lag:]))
    elif lag < 0:
        m = -lag
        return np.sum(h[m:] * (1.0 - h[:n - m]))
    else:
        return np.sum(h * (1.0 - h))

def lbfgsb_run(h, n, alpha, deadline, maxiter=3000, penalty=1e8):
    target_sum = n / 2.0
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty * diff * diff
        grad = full_smooth_grad(x, n, alpha)
        grad += 2.0 * penalty * diff * np.ones(n)
        return obj_val, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_local_search(h, n, deadline, rng, top_k=8, recompute_every=80):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    min_lr = 1e-6
    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-top_k:]
    it = 0
    
    while time.time() < deadline:
        if it % recompute_every == 0:
            c_all = compute_all_lags(h, n)
            cur_max = c_all.max()
            if cur_max < best_c - 1e-14:
                best_c = float(cur_max)
                best_h = h.copy()
            top_idx = np.argsort(c_all)[-top_k:]
        
        # Pick a random top lag (biased toward worse)
        probs = c_all[top_idx]
        probs = probs ** 3
        total = probs.sum()
        if total < 1e-30:
            total = 1.0
        probs = probs / total
        k_star = top_idx[rng.choice(top_k, p=probs)]
        lag = k_star - (n - 1)
        if lag == 0:
            if c_all[k_star] > 1e-10:
                grad = (1.0 - 2.0 * h) * (2.0 / n)
                h_new = project_feasible(h - lr * grad, n)
                c_new = compute_all_lags(h_new, n)
                if c_new.max() < c_all.max() + 1e-12:
                    h = h_new
            it += 1
            continue
        
        # Try a few random pairs at this lag
        m = abs(lag)
        if m >= n:
            it += 1
            continue
        
        improved_local = False
        for _ in range(3):
            if time.time() > deadline:
                break
            i = rng.randint(0, n - m)
            j = i + m
            hi, hj = h[i], h[j]
            # Compute contribution
            prod = hi * (1.0 - hj)
            if prod < 1e-9:
                continue
            # Move mass from higher to lower
            if hi > hj:
                d = min(lr * 0.5, (hi - hj) * 0.3, 0.08)
            else:
                d = -min(lr * 0.5, (hj - hi) * 0.3, 0.08)
            if abs(d) < 1e-8:
                continue
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            # Quick check: does this lag improve?
            c_lag_new = lag_correlation(h, n, lag)
            c_lag_old_check = prod  # approximate old
            # Full check every so often
            if it % 10 == 0 or c_lag_new < c_all[k_star] - 0.01:
                c_check = compute_all_lags(h, n)
                if c_check.max() <= c_all.max() + 1e-12:
                    improved_local = True
                    c_all = c_check
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    break
            else:
                improved_local = True
                c_all[k_star] = c_lag_new
        
        if not improved_local:
            lr *= 0.95
        else:
            lr = min(lr * 1.01, 0.05)
        
        if lr < min_lr:
            # Perturb and reset
            h = best_h + rng.normal(0, 0.01, n)
            h = project_feasible(h, n)
            lr = 0.03
            best_c = compute_c5(h, n)
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-top_k:]
        
        it += 1
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def targeted_local_search(h, n, deadline, rng):

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    K = 5
    
    while time.time() < deadline:
        c_all = compute_all_lags(h, n)

            if m >= n:

                contribs = h[:n - m] * (1.0 - h[m:])
            else:

            top_pairs = np.argsort(contribs)[-20:]

                else:

                        d = min(d, (hi - hj) * 0.4)
                        if d < 1e-7:

                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_new = compute_all_lags(h, n)
                        if c_new.max() < cur_max - 1e-14:
                            improved = True
                            cur_max = c_new.max()
                            c_all = c_new
                            if cur_max < best_c - 1e-14:
                                best_c = float(cur_max)
                                best_h = h.copy()
                            break
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                elif hj > hi + 1e-6:
                    for d in [0.05, 0.03, 0.01, 0.005]:
                        d = min(d, (hj - hi) * 0.4)

                            c_all = c_new
                            if cur_max < best_c - 1e-14:

                            break
                        else:

                            h = project_feasible(h, n)
        
        if not improved:
            # Micro perturbation
            idx = rng.randint(n)
            h[idx] += rng.uniform(-0.005, 0.005)
            h = project_feasible(h, n)
    
    return best_h, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_perturb(h, n, rng, flip_frac=0.05):
    """Randomly flip some values and re-optimize."""
    h_new = h.copy()
    n_flip = max(1, int(n * flip_frac))
    idx = rng.choice(n, n_flip, replace=False)
    h_new[idx] = 1.0 - h_new[idx]
    return project_feasible(h_new, n)
