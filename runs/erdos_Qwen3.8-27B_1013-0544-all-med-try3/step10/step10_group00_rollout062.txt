import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            frac = h - h_bin
            order = np.argsort(-frac)
            for idx in order[:int(np.ceil(diff))]:
                h_bin[idx] = 1.0
        else:
            frac = h_bin - h
            # actually we need h values where h_bin is 1, sort by h ascending
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def c5_gradient_w(w, h, n):
    """Compute gradient of weighted sum of correlations"""
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

# Fast binary local search
def binary_local_search(h, n, deadline, rng, max_iters=500000, use_annealing=True, T0=0.01, Tmin=0.001):
    h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h > 0.5)[0].astype(int)
    zeros = np.where(h < 0.5)[0].astype(int)
    
    total_iters = max_iters
    for it in range(total_iters):
        if time.time() > deadline:
            break
        
        # Adapt temperature
        if use_annealing:
            frac = it / total_iters
            T = T0 * (Tmin / T0) ** frac
        else:
            T = 0
        
        # Choose move type
        # 80% single swap, 20% pair swap
        if rng.random() < 0.8 and len(ones) > 0 and len(zeros) > 0:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h, n)
            delta = new_c - cur_c
            if delta < -1e-15 or (T > 0 and rng.random() < math.exp(-delta / T) if delta > 0 else True):
                if delta < -1e-15:
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                else:
                    cur_c = new_c
                # Update sets
                idx_i = np.where(ones == i)[0]
                idx_j = np.where(zeros == j)[0]
                if len(idx_i) > 0:
                    ones = np.delete(ones, idx_i[0])
                    zeros = np.delete(zeros, idx_j[0])
                    ones = np.append(ones, j)
                    zeros = np.append(zeros, i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            # Pair swap: move two 1s to two 0 positions
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    continue
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_c = compute_c5(h, n)
                delta = new_c - cur_c
                if delta < -1e-15 or (T > 0 and delta > 0 and rng.random() < math.exp(-delta / T)):
                    if delta < -1e-15:
                        cur_c = new_c
                        if new_c < best_c:

                    else:
                        cur_c = new_c
                    # Update sets (tricky with pairs)
                    ones = np.where(h > 0.5)[0].astype(int)

                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
    
    return h, best_c

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    cnt = h_bin.sum()
    if abs(cnt - target) < 1e-10:
        return h_bin
    if cnt < target:
        # Need more 1s
        diff = int(round(target - cnt))
        frac = h - h_bin
        order = np.argsort(-frac)
        for i in range(diff):
            h_bin[order[i]] = 1.0
    else:
        # Need fewer 1s
        diff = int(round(cnt - target))
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for i in range(diff):
            h_bin[ones[order[i]]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def c5_grad_w(w, h, n):
    dx = 2.0 / n
    nf = len(w)
    grad = np.zeros(n)
    for mm in range(nf):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def thue_morse_init(n):
    h = np.zeros(n)
    target = n // 2
    cnt = 0
    for i in range(n):
        # Thue-Morse: parity of bit count
        t = bin(i).count('1') % 2
        if t == 1:
            h[i] = 1.0
            cnt += 1
            if cnt >= target:
                break
    # Fill remaining with 0s (already 0)
    # Fix if needed
    if h.sum() != target:
        zeros = np.where(h == 0)[0]
        for i in range(int(target - h.sum())):
            h[zeros[i]] = 1.0
    return h

def beauty_seq_init(n, alpha_val):
    h = np.zeros(n)
    target = n // 2
    positions = np.zeros(n, dtype=bool)
    for i in range(n):
        frac = (i * alpha_val) % 1.0
        if frac < 0.5:
            positions[i] = True
    # Adjust to get exactly target ones
    cnt = positions.sum()
    if cnt > target:
        extra = np.where(positions)[0]
        rng = np.random.RandomState(0)
        to_remove = rng.choice(extra, size=cnt - target, replace=False)
        positions[to_remove] = False
    elif cnt < target:
        missing = np.where(~positions)[0]
        rng = np.random.RandomState(0)
        to_add = rng.choice(missing, size=target - cnt, replace=False)
        positions[to_add] = True
    h[positions] = 1.0
    return h

    zeros = list(np.where(h < 0.5)[0])
    n_iters = max_iters
    no_improve = 0
    improve_threshold = 5000
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        frac = (it + 1) / n_iters
        T = T0 * (Tmin / T0) ** frac
        
        # Choose move
        r = rng.random()
        if r < 0.75:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h, n)

def binary_hill_climb(h, n, deadline, rng, max_iters=200000):

def lbfgs_obj(x, n, alpha, mu):
    h = project_feasible(x, n)
    c = compute_c5_vec(h, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    h = project_feasible(x, n)
    c = compute_c5_vec(h, n)
    w = smooth_max_w(c, alpha)
    g = c5_grad_w(w, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 30, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)
