import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    # Project onto {0≤h≤1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        total = h.sum()
        if abs(total - target) < 1e-15:
            break
        # Distribute the difference among elements with room
        diff = target - total
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if mask.sum() == 0:
                break
            scale = diff / room[mask].sum()
            h[mask] = np.minimum(1.0, h[mask] + scale * room[mask])
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            scale = -diff / h[mask].sum()
            h[mask] = np.maximum(0.0, h[mask] - scale * h[mask])
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute evenly among elements not at boundary
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for i in range(len(w)):
        if w[i] < 1e-15:
            continue
        k = i - n_minus_1
        wi_dx = w[i] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wi_dx * (1.0 - h[k:])
            grad[k:] += wi_dx * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wi_dx * (1.0 - h[:nv])
            grad[:nv] += wi_dx * (-h[kk:])
    return grad

def active_set_obj(h, n, active, alpha):

    vals = c[active]
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            tot = room[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * room[mask]
        else:
            mask = h > 1e-12
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    best_h = h.copy()
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, min(i + 40, n)):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    np.random.seed(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Start at n=1000
    n = 1000
    h = upsample(h_init, n_orig, n)
    h = project_feasible(h, n)
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Coarse full L-BFGS (100s)
    t1 = min(t0 + 100, deadline - 600)
    best_h, best_c = run_full_lbfgs(best_h, n, t1,
        alphas=[50, 200, 500], mu=10.0, max_iter=2000)
    
    # Phase 2: Medium active-set L-BFGS (200s)
    t2 = min(t0 + 300, deadline - 400)
    best_h, best_c = run_lbfgs_active(best_h, n, t2,
        alphas=[2000, 5000, 20000],
        fracs=[0.02, 0.01, 0.005],
        mu=5.0, max_iter=3000)
    
    # Phase 3: Fine active-set L-BFGS with tighter focus (250s)
    t3 = min(t0 + 550, deadline - 250)
    best_h, best_c = run_lbfgs_active(best_h, n, t3,
        alphas=[50000, 100000, 200000, 500000],
        fracs=[0.005, 0.003, 0.002, 0.001],
        mu=2.0, max_iter=4000)
    
    # Phase 4: Multi-subgradient polish (100s)
    t4 = min(t0 + 650, deadline - 150)
    best_h, best_c = multi_subgradient(best_h, n, t4, n_iter=3000)
    
    # Phase 5: Single subgradient polish (80s)
    t5 = min(t0 + 730, deadline - 120)
    best_h, best_c = subgradient_polish(best_h, n, t5, n_iter=4000, step_init=0.001)
    
    # Phase 6: Active set L-BFGS with very tight focus (100s)
    t6 = min(t0 + 830, deadline - 80)
    best_h, best_c = run_lbfgs_active(best_h, n, t6,
        alphas=[200000, 500000, 1000000],
        fracs=[0.001, 0.0005, 0.0002],
        mu=1.0, max_iter=3000)
    
    # Phase 7: Perturbation restarts (80s)
    t7 = min(t0 + 910, deadline - 60)
    best_h, best_c = perturbation_restart(best_h, n, t7, n_restarts=8, noise_scale=0.005)
    
    # Phase 8: Final active set + subgradient (60s)
    t8 = min(t0 + 970, deadline - 30)
    best_h, best_c = run_lbfgs_active(best_h, n, t8,
        alphas=[500000, 1000000],
        fracs=[0.0005, 0.0002],
        mu=1.0, max_iter=2000)
    
    # Phase 9: Final subgradient (50s)
    t9 = deadline - 5
    best_h, best_c = subgradient_polish(best_h, n, t9, n_iter=3000, step_init=0.0005)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
