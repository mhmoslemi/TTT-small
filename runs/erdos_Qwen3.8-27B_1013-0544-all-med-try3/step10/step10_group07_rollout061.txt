            grad[p] = dx * s
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_h, best_c

def quadratic_residue_pattern(n):
    """Generate a pattern based on quadratic residues mod nearest prime >= n."""
    def is_prime(m):
        if m < 2:
            return False
        if m < 4:
            return True
        if m % 2 == 0 or m % 3 == 0:
            return False
        i = 5
        while i * i <= m:
            if m % i == 0 or m % (i + 2) == 0:
                return False
            i += 6
        return True
    
    p = n
    while not is_prime(p):
        p += 1
    
    residues = set()
    for x in range(1, (p + 1) // 2 + 1):
        residues.add((x * x) % p)
    
    h = np.zeros(n)
    target = n // 2
    # Map residues to positions
    pos = sorted([r for r in residues if r < n])
    h[pos[:target]] = 1.0
    if h.sum() < target:
        remaining = [i for i in range(n) if h[i] == 0]
        for r in pos[target:]:
            if r < n and h.sum() < target:
                h[r] = 1.0
    if h.sum() != target:
        h = make_binary(h, n)
    return h

def random_pattern(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def period_pattern(n, period, on_count, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for start in range(0, n, period):
        if on_count > 0:
            if start + on_count <= n:
                h[start:start+on_count] = 1.0
            else:
                h[start:n] = 1.0
    return make_binary(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = random_pattern(n0, seed)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Binary local search from initial (35% time)
    p1 = t0 + 0.35 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h, n, p1, max_sweeps=8000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Multiple random restarts with binary search (25% time)
    p2 = t0 + 0.60 * (budget_s - 5)
    rng = np.random.RandomState(seed + 100)
    starts = []
    # Quadratic residue pattern
    starts.append(quadratic_residue_pattern(n))
    # Random patterns
    for s in range(3):
        starts.append(random_pattern(n, seed + 200 + s))
    # Periodic patterns
    for per, on in [(7, 3), (8, 3), (9, 4), (5, 2), (6, 3)]:
        starts.append(period_pattern(n, per, on, seed + 300))
    
    for si, hs in enumerate(starts):
        if time.time() > p2 - 10:
            break
        if si == 0:
            # Try quadratic residue directly
            c_direct = compute_c5(hs, n)
            if c_direct < best_c:
                best_c = c_direct
                best_h = hs.copy()
        sub_dl = min(p2, time.time() + (p2 - t0) * 0.12)
        if time.time() > sub_dl:
            continue
        hb2, cb2 = fast_binary_search(hs, n, sub_dl, max_sweeps=3000, seed=seed + 400 + si)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    
    # Phase 3: SLSQP sub-binary refinement (15% time)
    p3 = t0 + 0.75 * (budget_s - 5)
    if time.time() < p3:
        h_init = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [1000, 5000, 20000]:
            if time.time() > p3 - 10:
                break
            hp = slsqp_smooth(h_init, n, alpha, 500, p3 - 5)
            cp = compute_c5(hp, n)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
                h_init = hp.copy()
    
    # Phase 4: Projected gradient sub-binary (10% time)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        hg, cg = projected_gradient(best_h, n, p4, alpha=5000, lr_init=0.005, max_iter=3000)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    
    # Phase 5: Multi-resolution (5% time)
    p5 = t0 + 0.90 * (budget_s - 5)
    for nh in [248, 252, 256, 240]:
        if time.time() > p5:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h, kind='linear'), nh)
        hb3, cb3 = fast_binary_search(hh, nh, min(p5, time.time() + 15), max_sweeps=2000, seed=seed+500)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
            n = nh
    
    # Phase 6: Final extended binary search (remaining time)
    p6 = dl - 5
    if time.time() < p6:
        hb6, cb6 = fast_binary_search(best_h, n, p6, max_sweeps=5000, seed=seed + 600)
        if cb6 < best_c:
            best_c = cb6
            best_h = hb6.copy()
    
    # Final polish with SLSQP
    if time.time() < dl - 2:
        hp = slsqp_smooth(project_feasible(best_h + 0.01 * rng.randn(n), n), n, 50000, 300, dl - 1)
        cp = compute_c5(hp, n)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
