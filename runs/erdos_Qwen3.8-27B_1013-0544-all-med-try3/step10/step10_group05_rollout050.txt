import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def compute_c5_fft(h, n):
    dx = 2.0 / n
    v = 1.0 - h
    n_fft = _next_pow2(2 * n - 1)
    H = fft(h, n_fft)
    V = fft(v[::-1], n_fft)
    corr = ifft(H * V).real[:2 * n - 1]
    return corr * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

    if n <= 1500:
        return float(np.max(compute_c5_vec(h, n)))
    return float(np.max(compute_c5_fft(h, n)))

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            if tot < 1e-16:

            h += (diff / tot) * room

            vals = h
            tot = vals.sum()
            if tot < 1e-16:

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha

    """Gradient of smooth max over active constraints + penalty"""
    vals = c_full[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)

    dx = 2.0 / n

    idx_center = n - 1

def full_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def full_grad(c_full, h, n, alpha, mu):
    m = c_full.max()
    sh = alpha * (c_full - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    active_mask = w > 1e-14
    for idx in np.where(active_mask)[0]:
        k = int(idx) - idx_center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 30 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_sturmian(n, slope):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 30 * n):
        pos = int(k * slope) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_stripe(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def run_active_opt(h, n, deadline, top_k, alpha, mu=0.001, use_fft=False):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_fft(h, n) if (use_fft and n > 1000) else compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-6, 0.001 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 3:
        return h, compute_c5(h, n)
    maxit = max(100, int(rem * 4))
    maxit = min(maxit, 5000)
    try:
        res = minimize(active_obj, h, jac=lambda x, *a: active_grad(
            compute_c5_vec(x, n), x, n, a[0], a[1], a[2]),
            args=(n, active, alpha, mu), method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-20,
                     'gtol': 1e-14, 'maxfun': maxit * 6,
                     'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def run_full_opt(h, n, deadline, alpha, mu=0.005):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 3:
        return h, compute_c5(h, n)
    maxit = max(100, int(rem * 3))
    maxit = min(maxit, 4000)
    try:
        res = minimize(full_obj, h,
                      jac=lambda x, *a: full_grad(compute_c5_vec(x, n), x, n, a[0], a[1]),
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': maxit * 6,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def coordinate_descent(h, n, deadline, n_sweeps=5):
    """Coordinate descent: try adjusting each h[i] to minimize max correlation"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        if time.time() > deadline - 5:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 5:
                break
            # Try to find optimal value for h[i] given others fixed
            # The correlation at lag k involving h[i]:
            # contributes h[i]*(1-h[i+k]) and h[i+k]*(1-h[i]) etc.
            # Simple: try a few values around current
            old_val = best_h[i]
            # Try: 0, 1, and nearby values
            candidates = [0.0, 1.0, max(0, old_val - 0.01), min(1, old_val + 0.01)]
            candidates = [c for c in candidates if abs(c - old_val) > 1e-10]
            # Also try the value that equalizes the sum constraint
            best_local = old_val
            best_local_c = best_c
            for cv in candidates:
                # Check sum constraint approximately
                new_sum = best_h.sum() - old_val + cv
                if abs(new_sum - n / 2.0) > 0.5:
                    continue
                best_h[i] = cv
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_local_c - 1e-12:
                    best_local_c = c
                    best_local = cv
                else:
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
            if best_local != old_val:
                best_h[i] = best_local
                best_h = project_feasible(best_h, n)
                if best_local_c < best_c:
                    best_c = best_local_c
                    improved = True
        if not improved:
            break
    return best_h, best_c

def pair_search_fine(h, n, deadline, max_iter=50000):
    """Fine pair search: move mass between two positions"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    delta = 0.002
    rng = np.random.RandomState(123)
    iterations = 0
    no_improve = 0
    while time.time() < deadline - 3 and iterations < max_iter:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        iterations += 1
        for d in [delta, -delta]:
            ni, nj = best_h[i] + d, best_h[j] - d
            if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                continue
            best_h[i] = max(0, min(1, ni))
            best_h[j] = max(0, min(1, nj))
            c = compute_c5(best_h, n)
            if c < best_c - 1e-14:
                best_c = c
                no_improve = 0
            else:
                best_h[i] -= d
                best_h[j] += d
                no_improve += 1
        if no_improve > 200:
            delta *= 0.7
            no_improve = 0
            if delta < 1e-7:
                break
    return best_h, best_c

def binary_anneal(n, deadline, rng, n_iter=50000):
    """Simulated annealing on binary sequences"""
    n_ones = n // 2
    positions = set(rng.choice(n, size=n_ones, replace=False).tolist())
    h = np.zeros(n)
    for p in positions:
        h[p] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.1
    T_min = 1e-6
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        if it % 1000 == 0:
            T = max(T * 0.999, T_min)
        # Move: pick a 1-position and a 0-position, swap
        ones_list = list(positions)
        a = ones_list[rng.randint(len(ones_list))]
        zeros = [i for i in range(n) if i not in positions]
        if not zeros:
            continue
        b = zeros[rng.randint(len(zeros))]
        positions.discard(a)
        positions.add(b)
        h[a] = 0.0
        h[b] = 1.0
        c_new = compute_c5(h, n)
        delta_c = c_new - best_c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, T_min)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            positions.discard(b)
            positions.add(a)
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c

def directed_gradient_step(h, n, c_vec, deadline, step_size=0.001, max_steps=1000):
    """Take gradient descent steps focused on reducing max correlation"""
    best_h = h.copy()
    best_c = c_vec.max()
    for _ in range(max_steps):
        if time.time() > deadline - 3:
            break
        max_idx = np.argmax(c_vec)
        k = int(max_idx) - (n - 1)
        # Gradient of c[k] w.r.t. h
        dx = 2.0 / n
        grad_k = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad_k[0:nv] += dx * (1.0 - h[k:n])
            grad_k[k:n] += -dx * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad_k[kk:n] += dx * (1.0 - h[0:nv])
            grad_k[0:nv] += -dx * h[kk:n]
        # Project gradient onto feasible set
        sum_grad = grad_k.sum()
        grad_proj = grad_k - (sum_grad / n)  # remove component along constraint
        # Step
        new_h = best_h - step_size * grad_proj
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_feasible(new_h, n)
        c_new_vec = compute_c5_vec(new_h, n)
        c_new = c_new_vec.max()
        if c_new < best_c - 1e-15:
            best_h = new_h
            best_c = c_new
            c_vec = c_new_vec
        else:
            step_size *= 0.9
            if step_size < 1e-10:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = make_beatty(500, (1.0 + np.sqrt(5.0)) / 4.0)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Try alternative starts
    starts = [
        make_beatty(n_orig, (1.0 + np.sqrt(5.0)) / 4.0),
        make_beatty(n_orig, np.sqrt(2.0) / 2.0),
        make_beatty(n_orig, 0.4312),
        make_beatty(n_orig, 0.3819),
        make_random_binary(n_orig, rng),
        make_random_binary(n_orig, rng),
        make_stripe(n_orig, 5, 0.5),
        make_stripe(n_orig, 7, 0.5),
    ]
    for h_alt in starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS full optimization at n_orig (60s)
    p1_end = min(t0 + 50, deadline - 800)
    if p1_end > t0 + 10:
        for alpha in [5000, 50000, 500000]:
            if time.time() > p1_end - 3:
                break
            h1, c1 = run_full_opt(best_h, best_n, p1_end, alpha, 0.005)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    # Phase 2: Active set optimization at n_orig (80s)
    p2_end = min(time.time() + 80, deadline - 650)
    if p2_end > time.time() + 15:
        configs = [
            (4, 20000, 0.001),
            (6, 50000, 0.0005),
            (8, 100000, 0.0005),
            (10, 200000, 0.0003),
            (15, 500000, 0.0003),
            (20, 1000000, 0.0002),
            (30, 5000000, 0.0002),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 5:
                break
            for _ in range(2):
                if time.time() > p2_end - 5:
                    break
                h2, c2 = run_active_opt(best_h, best_n, p2_end - 3, top_k, alpha, mu)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2

    # Phase 3: Upsample to n=1000 (180s)
    n_up = max(1000, best_n * 2)
    if n_up > 4000:
        n_up = 4000
    p3_end = min(time.time() + 180, deadline - 350)
    if p3_end > time.time() + 40 and n_up <= 4000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.002:
            for alpha in [5000, 50000, 500000, 2000000]:
                if time.time() > p3_end - 20:
                    break
                h3, c3 = run_full_opt(h_up, n_up, p3_end - 5, alpha, 0.003)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
                    best_n = n_up
            # Active set at higher res
            for top_k, alpha, mu in [(6, 500000, 0.0005), (10, 2000000, 0.0003),
                                      (15, 5000000, 0.0002), (20, 10000000, 0.0002)]:
                if time.time() > p3_end - 10:
                    break
                h3a, c3a = run_active_opt(best_h, best_n, p3_end - 5, top_k, alpha, mu, use_fft=(n_up > 1000))
                if c3a < best_c:
                    best_c = c3a
                    best_h = h3a

    # Phase 4: Try n=2000 (150s)
    n_up2 = 2000
    p4_end = min(time.time() + 150, deadline - 250)
    if p4_end > time.time() + 60 and best_n <= 1000:
        h_up2 = upsample_h(best_h, best_n, n_up2)
        c_up2 = compute_c5_fft(h_up2, n_up2).max()
        if c_up2 <= best_c + 0.003:
            for alpha in [10000, 100000, 1000000]:
                if time.time() > p4_end - 30:
                    break
                h4, c4 = run_full_opt(h_up2, n_up2, p4_end - 10, alpha, 0.002)
                if c4 < best_c:
                    best_c = c4
                    best_h = h4
                    best_n = n_up2
            for top_k, alpha in [(8, 2000000), (12, 5000000), (20, 20000000)]:
                if time.time() > p4_end - 15:
                    break
                h4a, c4a = run_active_opt(best_h, best_n, p4_end - 5, top_k, alpha, 0.0002, use_fft=True)
                if c4a < best_c:
                    best_c = c4a
                    best_h = h4a

    # Phase 5: Directed gradient + pair search (120s)
    p5_end = min(time.time() + 120, deadline - 150)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        c_vec = compute_c5_vec(best_h, cur_n)
        h5, c5 = directed_gradient_step(best_h, cur_n, c_vec, p5_end - 30, step_size=0.002)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        h5b, c5b = pair_search_fine(best_h, cur_n, p5_end - 5, max_iter=30000)
        if c5b < best_c:
            best_c = c5b
            best_h = h5b

    # Phase 6: More active set at fine level (100s)
    p6_end = min(time.time() + 100, deadline - 100)
    if p6_end > time.time() + 20:
        cur_n = len(best_h)
        for top_k, alpha, mu in [(4, 5000000, 0.0002), (6, 10000000, 0.0001),
                                  (8, 20000000, 0.0001), (12, 50000000, 0.0001)]:
            if time.time() > p6_end - 10:
                break
            h6, c6 = run_active_opt(best_h, cur_n, p6_end - 5, top_k, alpha, mu)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Binary annealing for exploration (80s)
    p7_end = min(time.time() + 80, deadline - 60)
    if p7_end > time.time() + 30:
        cur_n = len(best_h)
        if cur_n <= 1000:
            h7, c7 = binary_anneal(cur_n, p7_end, rng, n_iter=40000)
            if c7 < best_c + 0.01:
                # Refine the binary solution
                h7r, c7r = run_full_opt(h7, cur_n, p7_end - 3, 50000, 0.01)
                if c7r < best_c:
                    best_c = c7r
                    best_h = h7r

    # Phase 8: Final fine pair search (remaining time)
    p8_end = deadline - 3
    if p8_end > time.time() + 10:
        cur_n = len(best_h)
        h8, c8 = pair_search_fine(best_h, cur_n, p8_end, max_iter=50000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: one more active set pass with very tight tolerance
    if time.time() < deadline - 5:
        cur_n = len(best_h)
        h9, c9 = run_active_opt(best_h, cur_n, deadline - 2, 6, 20000000, 0.0001)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c_final = compute_c5(best_h, final_n)

    # Try evaluate_erdos_solution if available
    try:
        eval_c = evaluate_erdos_solution(best_h)
        if eval_c is not None and eval_c < best_c_final:
            best_c_final = float(eval_c)
    except Exception:
        pass

    return best_h, best_c_final, final_n
