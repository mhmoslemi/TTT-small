import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return w / s

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    penalty_coeff = 1e6
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1, 'maxfun': maxiter * 20})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def worst_lag_gradient_descent(h, n, deadline, K=5, lr_init=0.001, 
                                lr_decay=0.999, n_iter=50000):
    """Gradient descent targeting the K worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        # Get top K lags
        top_K = np.argsort(c_all)[-K:]
        
        # Compute gradient for each top lag
        grad = np.zeros(n)
        for k in top_K:
            lag = k - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += (1.0 - h[0:n - lag])
                grad[0:n - lag] += (-h[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += (1.0 - h[idx:n])
                grad[idx:n] += (-h[0:n - idx])
        
        grad *= (2.0 / n) / K  #(dx/K)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Rejected step, decrease learning rate
            lr *= 0.95
            h = h.copy()  # keep old h
        
        lr = max(lr * lr_decay, 1e-7)
    
    return best_h, best_c

- The "excess" pairs are those where h[i] is large and h[i+m] is small

Wait, actually for the full correlation (both positive and negative lags), lag m and lag -m are related:
- f_m (for m>0) = sum_{i=0}^{n-1-m} h[i](1-h[i+m])
- f_{-m} = sum_{i=m}^{n-1} h[i](1-h[i-m]) = sum_{i=0}^{n-1-m} h[i+m](1-h[i])

So f_m + f_{-m} = sum_{i=0}^{n-1-m} [h[i](1-h[i+m]) + h[i+m](1-h[i])]
= sum_{i=0}^{n-1-m} [h[i] + h[i+m] - 2h[i]h[i+m]]

And f_m - f_{-m} = sum_{i=0}^{n-1-m} [h[i] - h[i+m] - 2h[i]h[i+m] + 2h[i+m]h[i]]

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):

def project_feasible(h, n):

        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:

def worst_lag_gradient_descent(h, n, deadline, K=4, lr_init=0.0003):
    """Gradient descent targeting top K worst lags with line search."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(100000):
        if time.time() > deadline:
            break

        c_all = compute_all_lags(h, n)
        top_K = np.argsort(c_all)[-K:]

        grad = np.zeros(n)
        for k_idx in top_K:
            lag = int(k_idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += (1.0 - h[0:n - lag])
                grad[0:n - lag] += (-h[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += (1.0 - h[idx:n])
                grad[idx:n] += (-h[0:n - idx])

        grad *= (2.0 / n) / K

        # Momentum
        momentum = beta * momentum + (1 - beta) * grad

        # Try step
        h_trial = h - lr * momentum
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)

        if c_trial < best_c - 1e-14:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial.copy()
        else:
            lr *= 0.9
            if lr < 1e-9:
                break
        lr = min(lr * 1.001, lr_init)

    return best_h, best_c
