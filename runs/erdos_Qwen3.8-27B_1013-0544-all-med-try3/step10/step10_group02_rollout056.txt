import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_max_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_max_grad(x, n, active, alpha, mu):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()

        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.012)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=25):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            step = max(1, n // 500)
            for i in range(0, n, step):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def upsample(h_coarse, n_coarse, n_fine):
    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return project_feasible(h_fine, n_fine)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS at original resolution
    p1_end = t0 + 130
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 500, 5000, 50000, 500000, 5000000],
        mu=0.01, maxiter_per=[800, 1200, 1500, 2000, 2500, 3000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active-set at original resolution
    p2_end = min(time.time() + 70, deadline - 500)
    if time.time() < p2_end:
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=1000
    n_work = 1000
    if time.time() < deadline - 350:
        h_work = upsample(best_h, best_n, n_work)
        c_work = compute_c5(h_work, n_work)
        p3_end = min(time.time() + 220, deadline - 300)
        h3, c3 = run_lbfgs(h_work, n_work, p3_end,
            alphas=[20, 200, 2000, 20000, 200000, 2000000, 10000000],
            mu=0.005, maxiter_per=[800, 1200, 1500, 2000, 2500, 3000, 3500])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_work

    # Phase 4: Active-set at working resolution
    p4_end = min(time.time() + 130, deadline - 180)
    if time.time() < p4_end:
        h4, c4 = run_active(best_h, best_n, p4_end, top_k=8,
            alphas=[200000, 1000000, 5000000, 20000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Waterfill refinement
    p5_end = min(time.time() + 80, deadline - 100)
    if time.time() < p5_end:
        h5, c5 = waterfill_refine(best_h, best_n, p5_end, n_iter=120)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Subgradient refinement
    p6_end = min(time.time() + 40, deadline - 50)
    if time.time() < p6_end:
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Second waterfill round
    p7_end = min(time.time() + 30, deadline - 25)
    if time.time() < p7_end:
        h7, c7 = waterfill_refine(best_h, best_n, p7_end, n_iter=60)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search
    p8_end = deadline - 2
    if time.time() < p8_end:
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=20)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
