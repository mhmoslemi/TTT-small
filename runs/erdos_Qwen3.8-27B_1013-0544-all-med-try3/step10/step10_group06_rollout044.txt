                # Actually: reduce h[i] by d, increase h[i+lag] by d
                # Effect on C[lag]: for term i: (h[i]-d)*(1-h[i+lag]+d) - h[i]*(1-h[i+lag])
                # = -d*(1-h[i+lag]) + d*h[i] - d^2 (approx -d*(1-2*h[i+lag]+h[i]) for small d)
                # But also affects other lags. Use simple heuristic:
                # We want to reduce h[i]*(1-h[i+lag]), so if h[i] is high and h[i+lag] is low,
                # moving mass from i to i+lag helps.
                
                pairs = n - m
                for _attempt in range(min(pairs, 200)):
                    if time.time() > deadline:
                        return best_h, best_c
                    i = np.random.randint(0, pairs)
                    j = i + m
                    if h[i] > h[j] + 1e-6:
                        d = min(0.005, (h[i] - h[j]) * 0.3)
                        if d < 1e-8:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > h[i] + 1e-6:
                        d = min(0.005, (h[j] - h[i]) * 0.3)
                        if d < 1e-8:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_h, best_c

def gradient_direction_sa(h, n, deadline, alpha=200000, t_init=0.005, t_min=0.00001, cool=0.9999):
    """SA guided by gradient direction for better moves."""
    rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stall = 0

    while time.time() < deadline and T > t_min:
        grad = smooth_max_grad(h, n, alpha)
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-3:]

        r = rng.random()
        if r < 0.5:
            # Gradient-guided pair
            k_star = top_idx[rng.randint(len(top_idx))]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            # Use gradient to determine direction
            gi = grad[i]
            gj = grad[j]
            if gi < gj - 1e-8:
                # Move mass from i to j (reduce gradient at i)
                d = rng.uniform(0.001, 0.03)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif gj < gi - 1e-8:
                d = -rng.uniform(0.001, 0.03)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                d = rng.choice([-1, 1]) * rng.uniform(0.001, 0.02)
                d = min(max(d, -h[i]*0.3), (1-h[i])*0.3)
        else:
            # Random pair
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            d = rng.choice([-1, 1]) * rng.uniform(0.001, 0.04)
            d = min(max(d, -h[i]*0.3), (1-h[i])*0.3)

        if abs(d) < 1e-9:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha)
        max_new = float(c_new.max())

        de = e_new - e_old
        accept = de < 0 or (T > t_min and rng.random() < np.exp(-min(30.0, de / T)))

        if accept:
            if max_new < best_c - 1e-14:
                best_c = max_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            stall += 1

        T *= cool
        if stall > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            stall = 0
            T = max(T, 0.002)

    return best_h, best_c

def multi_start_search(n, deadline, rng):
    """Try multiple structured starting points."""
    candidates = []

    # Binary pattern: alternating blocks
    block = n // 10
    for phase in range(5):
        h = np.zeros(n)
        for i in range(n):
            blk = (i + phase) // block
            if blk % 2 == 0:
                h[i] = 1.0
        h = project_feasible(h, n)
        candidates.append(h)

    # Sinusoidal
    x = np.linspace(0, 2, n)
    for freq in [2, 3, 4, 5]:
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x + rng.uniform(0, 2*np.pi))
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        candidates.append(h)

    # Random binary
    for _ in range(3):
        h = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
        candidates.append(h)

    best_h = None
    best_c = float('inf')
    for h_cand in candidates:
        c = compute_c5(h_cand, n)
        if c < best_c:
            best_c = c
            best_h = h_cand.copy()
    return best_h, best_c

def polish_high_alpha(h, n, deadline):
    """Final polish with very high alpha L-BFGS-B."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5_000_000, 20_000_000, 100_000_000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + min(30, deadline - time.time() - 5), maxiter=5000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h, best_c = multi_start_search(500, t0 + 10, rng)

    n = len(best_h)

    # Phase 1: L-BFGS-B with increasing alpha (moderate)
    alpha_phases_1 = [
        (10000, 1000, 30),
        (50000, 1500, 35),
        (200000, 2000, 40),
        (1000000, 2500, 50),
    ]
    for alpha, maxit, tl in alpha_phases_1:
        if time.time() > deadline - 300:
            break
        tl_eff = min(tl, deadline - time.time() - 250)
        if tl_eff < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + tl_eff, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Greedy local search
    if time.time() < deadline - 250:
        ls_deadline = min(deadline - 200, time.time() + 80)
        h_opt, c_opt = greedy_local_search(best_h, n, ls_deadline, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: High-alpha L-BFGS-B
    alpha_phases_2 = [
        (5_000_000, 3000, 50),
        (20_000_000, 4000, 60),
        (100_000_000, 4000, 70),
    ]
    for alpha, maxit, tl in alpha_phases_2:
        if time.time() > deadline - 200:
            break
        tl_eff = min(tl, deadline - time.time() - 150)
        if tl_eff < 10:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + tl_eff, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Gradient-guided SA
    if time.time() < deadline - 150:
        sa_deadline = min(deadline - 100, time.time() + 100)
        h_opt, c_opt = gradient_direction_sa(best_h, n, sa_deadline, alpha=200000,
                                             t_init=0.004, t_min=0.00001, cool=0.99992)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Another round of high-alpha polish
    if time.time() < deadline - 80:
        h_opt, c_opt = polish_high_alpha(best_h, n, deadline - 60)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_try in [700, 1000]:
        if time.time() > deadline - 100:
            break
        remaining = deadline - 70 - time.time()
        if remaining < 40:
            break
        h_interp = expand_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 500000, time.time() + min(30, remaining * 0.4), maxiter=2000)
        h_opt = lbfgsb_optimize(h_opt, n_try, 10_000_000, time.time() + min(25, remaining * 0.3), maxiter=3000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try

    # Phase 7: Final greedy + SA at best resolution
    if time.time() < deadline - 40:
        n_final = len(best_h)
        ls_deadline = min(deadline - 25, time.time() + 40)
        h_opt, c_opt = greedy_local_search(best_h, n_final, ls_deadline, top_k=7)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final polish
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 100_000_000, deadline - 5, maxiter=5000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(500, 0.5), 500)
        best_c = compute_c5(best_h, 500)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
