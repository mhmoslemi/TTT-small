        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def make_structured_start(n, mode, rng):

                if t >= target:
                    break

    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Phase 1: Quick L-BFGS to get good continuous starting point (50s)
    p1_end = t0 + 50
    h1, c1 = run_lbfgs(best_h, best_n, p1_end, alpha=30000, mu=0.01, maxit=2000)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    h1b, c1b = run_lbfgs(best_h, best_n, min(p1_end + 20, deadline), alpha=300000, mu=0.001, maxit=2000)
    if c1b < best_c:
        best_c = c1b
        best_h = h1b

    # Phase 2: Also try structured starts at n=500 quickly (20s)
    p2_end = t0 + 70
    for mode in range(4):
        if time.time() > p2_end:
            break
        h_s = make_structured_start(best_n, mode, rng)
        c_s = compute_c5(h_s, best_n)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Phase 3: Main SA on binary (380s)
    p3_end = t0 + 450
    h_bin = binarize(best_h, best_n)
    h3, c3 = sa_binary(h_bin, best_n, p3_end, rng, T0=0.012, T1=1e-9, restarts=4)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Greedy binary improvement after SA (100s)
    p4_end = t0 + 550
    h4, c4 = greedy_binary_improve(best_h, best_n, p4_end, max_iters=100000, rng=rng)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: L-BFGS continuous refinement with increasing alpha (150s)
    p5_end = t0 + 700
    for alpha in [100000, 500000, 2000000, 10000000]:
        if time.time() > p5_end - 10:
            break
        t_sub = min(p5_end, time.time() + 35)
        h5, c5 = run_lbfgs(best_h, best_n, t_sub, alpha=alpha, mu=0.0005, maxit=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Second SA with lower temperature (polishing) (200s)
    p6_end = t0 + 900
    h_bin6 = binarize(best_h, best_n)
    h6, c6 = sa_binary(h_bin6, best_n, p6_end, rng, T0=0.003, T1=1e-10, restarts=3)
    if c6 < best_c:
        best_c = c6
        best_h = h6.copy()

    # Phase 7: Final greedy + L-BFGS + coord descent (70s)
    p7_end = deadline - 2
    h7, c7 = greedy_binary_improve(best_h, best_n, min(p7_end - 30, deadline), max_iters=50000, rng=rng)
    if c7 < best_c:
        best_c = c7
        best_h = h7

    if time.time() < p7_end - 15:
        h7b, c7b = run_lbfgs(best_h, best_n, p7_end - 15, alpha=5000000, mu=0.0001, maxit=1500)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b

    if time.time() < p7_end - 2:
        h7c, c7c = fine_coord_descent(best_h, best_n, p7_end,
                                      steps=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                      max_rounds=200)
        if c7c < best_c:
            best_c = c7c
            best_h = h7c

    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
