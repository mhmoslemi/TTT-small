def compute_c5(h, n):

    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-14:

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n

    k = shift_idx - (n - 1)

        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx

    return grad

            if tot < 1e-15:

def shift_grad_single(h, n, shift_idx):

    grad = np.zeros(n)
    k = shift_idx - (n - 1)

        nv = n - k

    # Phase 4: Anti-periodic ansatz
    if time.time() < deadline - 250:
        t4 = min(time.time() + 80, deadline - 250)
        h_ap, c_ap = run_antiperiodic(best_h, n, t4)
        if c_ap < best_c:
            best_c = c_ap
            best_h = h_ap

    # Phase 5: More active-set polish with very high alpha, top-3
    if time.time() < deadline - 180:
        t5 = min(time.time() + 80, deadline - 180)
        best_h, best_c = run_active_lbfgs(best_h, n, t5,
                                           alphas=[200000, 1000000, 5000000],
                                           top_k=3, mu=0.0)

    # Phase 6: SLSQP polish on top-3
    if time.time() < deadline - 120:
        t6 = min(time.time() + 40, deadline - 120)
        h6, c6 = run_slqp_active(best_h, n, t6, alpha=500000, top_k=3, maxiter=1500)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Multi-restart to escape local optima
    if time.time() < deadline - 80:
        t7 = min(time.time() + 50, deadline - 80)
        best_h, best_c = multi_restart(best_h, n, t7, seed=seed)

    # Phase 8: Final active-set very high alpha
    if time.time() < deadline - 40:
        t8 = min(time.time() + 30, deadline - 40)
        best_h, best_c = run_active_lbfgs(best_h, n, t8,
                                           alphas=[2000000, 10000000],
                                           top_k=3, mu=0.0)

    # Phase 9: Local 2-point search for final micro-improvement
    if time.time() < deadline - 10:
        t9 = min(time.time() + 25, deadline - 10)
        best_h, best_c = local_2point(best_h, n, t9, radius=25)

    # Final SLSQP if time remains
    if time.time() < deadline - 2:
        h_final, c_final = run_slqp_active(best_h, n, deadline - 1,
                                            alpha=5000000, top_k=3, maxiter=1000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
