def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute proportionally
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += (diff / tot) * room
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_w(c, alpha):

    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)

import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= s
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lbfgs_obj(x, n, alpha, mu):
    xc = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(xc, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    xc = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(xc, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, xc, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit=3000):

        res = minimize(lbfgs_obj, h.copy(), jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-26,
                               'gtol': 1e-20, 'maxfun': maxit * 8,
                               'maxls': 40, 'maxcor': 40})

def make_beatty(n, alpha):

    target = n // 2
    positions = set()
    for k in range(1, 10 * n):

        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = 1.0
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = 0.0
    return project_feasible(h, n)

            if time.time() > deadline - 15:
                break
            h_p, c_p = run_lbfgs(best_h, n, min(deadline - 10, time.time() + 25),
                                 alpha=alpha, mu=0.0005, maxit=1500)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

        # Fine coord descent
        h_f, c_f = fine_coord_descent(best_h, n, deadline - 5,
                                      steps=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                      max_rounds=300)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final constraint enforcement
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
