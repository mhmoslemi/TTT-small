def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0

        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:

    return project_feasible(h, n)
