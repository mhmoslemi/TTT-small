    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:

    return m + np.log(np.sum(np.exp(sh))) / alpha

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):

            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    val = smooth_max_val(c, alpha)
    if mu > 0:

    return val

    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:

    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alphas, maxit_each=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad, args=(n, alpha, 0.01),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit_each, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit_each * 5,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def sa_binary_single(h_init, n, deadline, rng, t_init=0.02, t_final=1e-8, num_iters=400000):
    dx = 2.0 / n
    h = binarize(h_init, n).copy()
    n_ones = n // 2
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    log_ratio = math.log(t_final / t_init)
    for it in range(num_iters):
        if time.time() > deadline:
            break
        T = t_init * math.exp(log_ratio * (it / float(num_iters)))
        if T < 1e-12:
            T = 1e-12
        ri = rng.randint(n_ones)
        rj = rng.randint(n - n_ones)
        i = ones[ri]
        j = zeros[rj]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - curr_c
        if delta < -1e-15 or (delta <= 0.5 and rng.random() < math.exp(-delta / T)):
            curr_c = new_c
            ones[ri] = j
            zeros[rj] = i
            if curr_c < best_c - 1e-15:
                best_c = curr_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def sa_multipass(h_init, n, deadline, rng):
    best_h = binarize(h_init, n).copy()
    best_c = compute_c5(best_h, n)
    num_passes = 8
    time_per_pass = max(10.0, (deadline - time.time()) / (num_passes + 1))
    for p in range(num_passes):
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + time_per_pass)
        if p == 0:
            start_h = best_h
        else:
            start_h = best_h.copy()
            ones = np.where(start_h > 0.5)[0]
            zeros = np.where(start_h < 0.5)[0]
            num_perturb = len(ones) // 10
            for _ in range(num_perturb):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                start_h[i], start_h[j] = start_h[j], start_h[i]
        t_init = 0.015 / (p + 1)
        t_final = max(1e-9, 1e-8 / (p + 2))
        ni = max(50000, int(300000 / (p + 1)))
        h_new, c_new = sa_binary_single(start_h, n, sub_deadline, rng,
                                         t_init=t_init, t_final=t_final, num_iters=ni)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def fine_coord_descent(h, n, deadline, steps=None, max_rounds=200):
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < max_rounds and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            w = smooth_max_w(c_vec, 300000.0)
            grad = h_grad_from_w(w, best_h, n)
            pos_mask = (grad > 1e-10) & (best_h > 1e-6)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-6)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old + 1)[:-1] + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new + 1)[:-1] + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_beatty_binary(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = max(diff, -1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Phase 1: Quick L-BFGS at parent resolution (50s)
    p1_deadline = t0 + 50
    h1, c1 = run_lbfgs(best_h, best_n, p1_deadline, alphas=[5000, 20000, 100000], maxit_each=1500)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: SA at n=500 (250s)
    p2_deadline = min(t0 + 300, deadline - 500)
    h2, c2 = sa_multipass(best_h, best_n, p2_deadline, rng)
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = best_n

    # Phase 3: L-BFGS refinement at n=500 (80s)
    p3_deadline = min(t0 + 380, deadline - 420)
    h3, c3 = run_lbfgs(best_h, best_n, p3_deadline, alphas=[50000, 200000, 1000000], maxit_each=2000)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Fine coord descent at n=500 (60s)
    p4_deadline = min(t0 + 440, deadline - 360)
    h4, c4 = fine_coord_descent(best_h, best_n, p4_deadline,
                                steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                max_rounds=150)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: SA at n=750 (250s)
    n750 = 750
    p5_deadline = min(t0 + 690, deadline - 200)
    if time.time() < p5_deadline - 30:
        h750_start = interpolate_h(best_h, best_n, n750)
        h750_bin = binarize(h750_start, n750)
        h5, c5 = sa_multipass(h750_bin, n750, p5_deadline, rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n750

        # Quick L-BFGS at n=750
        if time.time() < p5_deadline + 30:
            p5b_deadline = min(p5_deadline + 30, deadline - 150)
            h5b, c5b = run_lbfgs(best_h, best_n, p5b_deadline, alphas=[50000, 500000], maxit_each=1500)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: SA at n=1000 (200s)
    n1000 = 1000
    p6_deadline = min(t0 + 900, deadline - 100)
    if time.time() < p6_deadline - 30:
        h1000_start = interpolate_h(best_h, best_n, n1000)
        h1000_bin = binarize(h1000_start, n1000)
        h6, c6 = sa_multipass(h1000_bin, n1000, p6_deadline, rng)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n1000

        if time.time() < p6_deadline + 20:
            p6b_deadline = min(p6_deadline + 20, deadline - 80)
            h6b, c6b = run_lbfgs(best_h, best_n, p6b_deadline, alphas=[100000, 1000000], maxit_each=1000)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b

    # Phase 7: Final L-BFGS + fine tuning at best resolution (remaining time)
    p7_deadline = deadline - 5
    if time.time() < p7_deadline - 20:
        h7, c7 = run_lbfgs(best_h, best_n, p7_deadline - 20, alphas=[200000, 2000000], maxit_each=1500)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    if time.time() < p7_deadline - 2:
        h8, c8 = fine_coord_descent(best_h, best_n, p7_deadline,
                                    steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                    max_rounds=100)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
