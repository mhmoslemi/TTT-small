    c = compute_c5_vec(h, n)
    val = smooth_max(c, alpha)

    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, h, n)

def gradient_sa(h_bin, n, deadline, rng, T0=0.02, T1=0.00005, recompute_grad_every=100):
    """Simulated annealing with gradient-biased moves on binary h."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    # Precompute correlation for gradient
    c_vec = compute_c5_vec(h, n)
    alpha = 50000.0
    w = smooth_max_w(c_vec, alpha)
    grad = h_grad_from_w(w, h, n)
    # grad[i] > 0 means decreasing h[i] helps
    # grad[i] < 0 means increasing h[i] helps
    
    it = 0
    recompute = 0
    no_improve = 0
    
    while time.time() < deadline:
        recompute += 1
        if recompute >= recompute_grad_every:
            c_vec = compute_c5_vec(h, n)
            alpha = 50000.0
            w = smooth_max_w(c_vec, alpha)
            grad = h_grad_from_w(w, h, n)
            recompute = 0
        
        # Temperature for this iteration
        T = T0 * (T1 / T0) ** (it / max(1, int(500000)))
        T = max(T, T1)
        
        # Generate move
        use_gradient = (rng.random() < 0.7)
        
        if use_gradient:
            # Biased selection
            # For remove (1→0): prefer high grad[i]
            # For add (0→1): prefer low grad[j] (i.e., grad[j] < 0)
            remove_scores = grad[ones]  # want to remove positions with high grad
            add_scores = grad[zeros]  # want to add positions with low grad (negative)
            
            # softmax sampling
            remove_scores_shifted = remove_scores - remove_scores.max()
            add_scores_shifted = add_scores - add_scores.min()
            
            r_probs = np.exp(remove_scores_shifted / max(0.001, T))
            r_probs /= r_probs.sum()
            a_probs = np.exp(add_scores_shifted / max(0.001, T))
            a_probs /= a_probs.sum()
            
            i_idx = rng.choice(n_ones, p=r_probs)
            j_idx = rng.choice(n_zeros, p=a_probs)
        else:
            i_idx = rng.randint(n_ones)
            j_idx = rng.randint(n_zeros)
        
        i = ones[i_idx]
        j = zeros[j_idx]
        
        # Apply swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - curr_c
        
        # Accept?
        accept = (delta < 0) or (rng.random() < np.exp(-delta / max(T, 1e-10)))
        
        if accept:
            curr_c = new_c
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        it += 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        sh -= smx
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lbfgs_obj_func(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad_func(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj_func, h.copy(), jac=lbfgs_grad_func,
                       args=(n, alpha, mu), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-30,
                                'gtol': 1e-20, 'maxfun': maxit * 5,
                                'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

    recompute = 0
    no_improve = 0
    stuck_count = 0
    total_budget_iters = 5000000

        recompute += 1
        if recompute >= recompute_grad_every:

            alpha_g = 80000.0
            w_g = smooth_max_w(c_vec, alpha_g)
            grad = h_grad_from_w(w_g, h, n)
            recompute = 0

        progress = min(it / total_budget_iters, 1.0)
        T = T0 * (T1 / T0) ** progress

            T = max(T, T0 * 0.3)

                break

            as_ = add_scores - add_scores.min()

            r_probs += 1e-15

            a_probs += 1e-15

            i_idx = rng.choice(n_ones, p=r_probs)

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0

        it += 1

def fine_coord_descent(h, n, deadline, steps=None, max_rounds=300):
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0

            tol = max(1e-10, 0.002 * max_val)
            w = smooth_max_w(c_vec, 300000.0)
            grad = h_grad_from_w(w, best_h, n)

            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)

            if not pos_mask.any() or not neg_mask.any():
                break

            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]

            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j

def nemesis_subset(h, n, positions, deadline, maxit=3000):
    h_full = h.copy()
    pos_arr = np.array(positions)
    x0 = h_full[pos_arr].copy()

    def obj_small(x):

        h_tmp = project_feasible(h_tmp, n)
