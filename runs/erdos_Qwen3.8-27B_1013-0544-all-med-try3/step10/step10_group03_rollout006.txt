    """Subgradient method with Nesterov momentum for the smooth max."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(2 * n - 1):
        if w[ii] < 1e-15:
            continue
        k = ii - (n - 1)
        wi = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi * (1.0 - h[k:n])
            grad[k:n] += wi * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * (1.0 - h[0:nv])
            grad[0:nv] += wi * (-h[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=0.005, maxiter=15000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 50000, 500000, 5000000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        remaining = deadline - time.time() - 10
        per_iter = 0.005
        maxit = min(maxiter, int(remaining / per_iter))
        if maxit < 100:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgradient_nesterov(h, n, deadline, alpha=100000, max_iter=5000):
    """Subgradient with Nesterov acceleration and adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    t_iter = 1.0
    beta = 0.0
    base_lr = 0.002 * (2.0 / n)
    
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        
        grad = subgrad_step(y, n, alpha, 1/alpha)
        
        t_np = (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        beta_new = (t_iter - 1.0) / t_np
        
        y_new = y - base_lr * grad
        y_new = np.clip(y_new, 0.0, 1.0)
        y_new = project_feasible(y_new, n)
        
        h_new = y_new + beta_new * (y_new - y)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        y = h_new
        t_iter = t_np
    
    return best_h, best_c

def pair_coordinate_descent(h, n, deadline, alpha=50000, max_rounds=200):
    """Coordinate descent: minimize over pairs (i,j) to reduce max correlation."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        
        c = compute_c5_vec(h, n)
        max_val = c.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient focused on active shifts
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        # Find top candidates by |gradient|
        idxs = np.argsort(-np.abs(grad))[:80]
        
        improved = False
        for ii in range(len(idxs)):
            if time.time() > deadline - 5:
                break
            i = idxs[ii]
            for jj in range(len(idxs)):
                if ii == jj:
                    continue
                j = idxs[jj]
                gi, gj = grad[i], grad[j]
                
                # Determine direction
                if gi > gj:
                    di = -1.0  # decrease h[i]
                    dj = 1.0   # increase h[j]
                else:
                    di = 1.0
                    dj = -1.0
                
                for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                    new_hi = h[i] + di * step
                    new_hj = h[j] + dj * step
                    if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                        continue
                    h[i] = new_hi
                    h[j] = new_hj
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= di * step
                        h[j] -= dj * step
        
        if not improved:
            # Try smaller steps
            for i in range(min(30, n)):
                if time.time() > deadline - 5:
                    break
                if np.abs(grad[i]) < 1e-6:
                    continue
                for j in range(min(30, n)):
                    if i == j:
                        continue
                    for step in [0.0001, 0.00005, 0.00002]:
                        for di, dj in [(-1, 1), (1, -1)]:
                            if grad[i] * di + grad[j] * dj > 0:
                                continue
                            new_hi = h[i] + di * step
                            new_hj = h[j] + dj * step
                            if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                                continue
                            h[i] = new_hi
                            h[j] = new_hj
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] -= di * step
                                h[j] -= dj * step
            if not improved:
                break
    
    return h, best_c

def randomized_restart(h, n, deadline, n_restarts=5, scale=0.005):
    """Add small random perturbations and re-optimize to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.default_rng(12345)
    
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        h_pert = h.copy() + rng.uniform(-scale, scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        if time.time() > deadline - 15:
            break
        t_sub = min(time.time() + 20, deadline - 10)
        h_opt, c_opt = run_lbfgs(h_pert, n, t_sub,
                                  alphas=[50000, 500000], mu=0.005, maxiter=5000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=2000):
    """SLSQP with hard equality constraint."""
    def obj(x):
        return smooth_obj(x, n, alpha, 0.0)
    def jac(x):
        return smooth_grad(x, n, alpha, 0.0)
    def con(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def top_multi_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_multi_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_shifts_lbfgs(h, n, deadline, top_k=8):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for iteration in range(15):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        if len(active) == 0:
            break
        
        for alpha in [100000, 1000000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(top_multi_obj, h, jac=top_multi_grad,
                              args=(n, list(active), alpha, 0.01),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 20000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    # Phase 1: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, h_init)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 2: Quick L-BFGS warmup
    t_phase2 = min(t0 + 80, deadline - 700)
    h, best_c = run_lbfgs(h, n, t_phase2,
                          alphas=[2000, 20000, 200000], mu=0.01, maxiter=8000)
    best_h = h.copy()
    
    # Phase 3: Aggressive L-BFGS with high alpha
    t_phase3 = min(time.time() + 250, deadline - 400)
    h, c_tmp = run_lbfgs(h, n, t_phase3,
                         alphas=[500000, 5000000, 50000000], mu=0.001, maxiter=20000)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 4: Top-shifts focused L-BFGS
    t_phase4 = min(time.time() + 150, deadline - 250)
    h, c_tmp = top_shifts_lbfgs(h, n, t_phase4, top_k=6)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 5: Subgradient with Nesterov momentum
    t_phase5 = min(time.time() + 200, deadline - 150)
    h, c_tmp = subgradient_nesterov(h, n, t_phase5, alpha=200000, max_iter=3000)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 6: Pair coordinate descent
    t_phase6 = min(time.time() + 150, deadline - 100)
    h, c_tmp = pair_coordinate_descent(h, n, t_phase6, alpha=50000)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 7: SLSQP polish
    t_phase7 = min(time.time() + 60, deadline - 50)
    h, c_tmp = slqp_polish(h, n, t_phase7, alpha=1000000, maxiter=2000)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 8: Randomized restarts
    t_phase8 = min(time.time() + 80, deadline - 30)
    h, c_tmp = randomized_restart(h, n, t_phase8, n_restarts=4, scale=0.003)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Phase 9: Final top-shifts + pair descent
    t_phase9 = min(time.time() + 40, deadline - 10)
    h, c_tmp = top_shifts_lbfgs(h, n, t_phase9, top_k=4)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    t_phase10 = deadline - 3
    h, c_tmp = pair_coordinate_descent(h, n, t_phase10, alpha=100000)
    if c_tmp < best_c:
        best_c = c_tmp
        best_h = h.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    # Also check if original resolution gives better result
    if n_orig != n:
        h_orig = project_feasible(h_init, n_orig)
        c_orig = compute_c5(h_orig, n_orig)
        if c_orig < best_c:
            return h_orig, c_orig, n_orig
    
    return best_h, best_c, n
