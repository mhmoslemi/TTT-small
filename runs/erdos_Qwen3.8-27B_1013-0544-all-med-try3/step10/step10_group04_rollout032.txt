import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):

                grad[k:n] += wm * (-x[0:nv])

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec_raw(h, n):
    """Return correlation vector without dx factor."""
    return np.correlate(h, 1 - h, mode='full')

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:

            nv = n - kk
            if nv > 0:

                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    return lambda x: smooth_obj(x, n, alpha, mu)

def make_smooth_grad(n, alpha, mu):
    return lambda x: smooth_grad(x, n, alpha, mu)

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (10000, 5000.0), (50000, 20000.0), (200000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(make_smooth_obj(n, alpha, mu), h,
                          jac=make_smooth_grad(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def binary_push_smooth(h, n):
    """Push toward binary but keep small fractional parts for optimization."""
    h_b = h.copy()
    # Keep a Gaussian-like transition: values close to 0.5 get pushed to 0 or 1
    threshold_lo = 0.3
    threshold_hi = 0.7
    h_b[h_b < threshold_lo] = 0.0
    h_b[h_b > threshold_hi] = 1.0
    # Linear interpolation in the middle
    mid = (h_b >= threshold_lo) & (h_b <= threshold_hi)
    h_b[mid] = (h[mid] - threshold_lo) / (threshold_hi - threshold_lo)
    h_b[h_b < 0.35] = 0.0
    h_b[h_b > 0.65] = 1.0
    return project_feasible(h_b, n)

    x_old = np.linspace(0, 2, n_old + 1)
    h_old_ext = np.concatenate([h_old, [0.0]])
    x_new = np.linspace(0, 2, n_new + 1)
    h_new = np.interp(x_new[:-1], x_old, h_old_ext)
    return project_feasible(h_new, n_new)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def targeted_refine(h, n, deadline):
    """Coordinate descent: try small perturbations at each point."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_vec = compute_c5_vec(h, n)
    
    for round_num in range(200):
        if time.time() > deadline:
            break
        improved = False
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        # Focus on positions that affect the worst shift
        if k_worst >= 0:
            candidates = list(range(0, n - k_worst)) + list(range(k_worst, n))
        else:
            kk = -k_worst
            candidates = list(range(0, n - kk)) + list(range(kk, n))
        
        # Shuffle for diversity
        rng = np.random.RandomState(round_num)
        rng.shuffle(candidates)
        
        for i in candidates[:min(len(candidates), 200)]:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                old_val = h[i]
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    c_vec = compute_c5_vec(h, n)
                    improved = True
                    break
                else:
                    h[i] = old_val
        if not improved and round_num > 30:
            break
    
    return best_h, best_c

def multi_start_lbfgs(h_init, n, deadline, n_starts=4):
    """Try multiple L-BFGS runs from different perturbed starting points."""
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    rng = np.random.RandomState(99)
    bounds = [(0.0, 1.0)] * n
    
    configs_list = [
        [(200, 100.0), (1000, 500.0), (5000, 2000.0), (20000, 10000.0)],
        [(500, 200.0), (2000, 1000.0), (10000, 5000.0), (50000, 20000.0)],
        [(100, 50.0), (500, 200.0), (3000, 1500.0), (15000, 7500.0)],
        [(300, 150.0), (1500, 500.0), (8000, 4000.0), (40000, 20000.0)],
    ]
    
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        if s == 0:
            h_s = h_init.copy()
            configs = configs_list[0]
        else:
            mag = [0.03, 0.05, 0.08][s % 3]
            h_s = h_init + mag * (rng.rand(n) - 0.5)
            h_s = project_feasible(h_s, n)
            configs = configs_list[s % len(configs_list)]
        
        t_alloc = (deadline - time.time() - 20) / (n_starts - s)
        for alpha, mu in configs:
            if time.time() > deadline - 25:
                break
            try:
                res = minimize(make_smooth_obj(n, alpha, mu), h_s,
                              jac=make_smooth_grad(n, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 20000, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': 50000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h_s = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(600, 0.5)
        n_base = 600
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # === Phase 1: Optimize at base resolution (n_base ~ 500) ===
    # Time: ~100s
    t1 = min(t0 + 100, deadline - 500)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(100, 50.0), (500, 200.0), (2000, 1000.0),
                     (10000, 5000.0), (50000, 20000.0)])
    
    # Quick SLSQP at base
    t1b = min(time.time() + 40, deadline - 480)
    if t1b > time.time() + 5:
        h1b, c1b = slqp_polish(best_h, n_base, t1b, alpha=1000000, maxiter=3000)
        if c1b < best_c:
            best_c = c1b
            best_h = h1b
    
    # === Phase 2: Upsample to n=1000, optimize ===
    n1 = 1000
    t2_start = time.time()
    remaining_after_phase2 = deadline - t2_start
    if remaining_after_phase2 > 200:
        h1000 = interpolate_to(best_h, n_base, n1)
        c1000 = compute_c5(h1000, n1)
        
        # L-BFGS at n=1000
        t2 = min(t2_start + 150, deadline - 300)
        if t2 > time.time() + 10:
            h1000, c1000 = run_lbfgs(h1000, n1, t2,
                configs=[(200, 100.0), (1000, 500.0), (5000, 2000.0),
                         (20000, 10000.0), (100000, 50000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        
        # SLSQP at n=1000
        t3 = min(time.time() + 60, deadline - 250)
        if t3 > time.time() + 5:
            h3, c3 = slqp_polish(best_h, n1, t3, alpha=3000000, maxiter=5000)
            if c3 < best_c:
                best_c = c3
                best_h = h3
    
    # === Phase 3: Binary local search at n=1000 ===
    if time.time() < deadline - 200:
        h_bin, c_bin = fast_binary_local_search(best_h, n1, time.time() + 60,
                                                 max_iters=1000000)
        if c_bin < best_c:
            best_c = c_bin
            # Use binary as new best, but also try to refine it continuously
            h_refine = h_bin.copy()
            h_refine = project_feasible(h_refine, n1)
            best_h = h_refine
    
    # === Phase 4: Multi-start L-BFGS at n=1000 ===
    t4 = min(time.time() + 120, deadline - 150)
    if t4 > time.time() + 30:
        best_h, best_c = multi_start_lbfgs(best_h, n1, t4, n_starts=3)
    
    # === Phase 5: Upsample to n=2000 for final refinement ===
    n2 = 2000
    if time.time() < deadline - 120:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        c2000 = compute_c5(h2000, n2)
        
        t5 = min(time.time() + 60, deadline - 50)
        if t5 > time.time() + 10:
            h2000, c2000 = run_lbfgs(h2000, n2, t5,
                configs=[(500, 200.0), (3000, 1000.0), (15000, 5000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        
        # SLSQP at n=2000
        t6 = min(time.time() + 40, deadline - 20)
        if t6 > time.time() + 5:
            h6, c6 = slqp_polish(best_h, n2, t6, alpha=10000000, maxiter=2000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
        n_final = n2
    else:
        n_final = len(best_h)
    
    # === Phase 6: Final targeted refinement ===
    if time.time() < deadline - 5:
        best_h, best_c = targeted_refine(best_h, len(best_h), deadline - 3)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
