def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_shift_obj(x, n, top_k, alpha=0):
    """Objective: if top_k=0, use smooth max over all shifts.
    If top_k>0, use weighted sum of top_k worst shifts."""
    c = compute_c5_vec(x, n)
    if top_k is None or top_k >= len(c):
        # Smooth max
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    else:
        # Top-K weighted sum
        idx = np.argsort(-c)[:top_k]
        w = c[idx] - c.min() + 1.0  # weights proportional to how bad they are
        w /= w.sum()
        val = np.sum(w * c[idx])
    return val

def topk_shift_grad(x, n, top_k, alpha=0):

    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    
    if top_k is None or top_k >= len(c):
        # Smooth max weights
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400.0:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    return w

    v = lse_max(c, alpha)
    v += mu * (x.sum() - n * 0.5) ** 2
    return v

    w = lse_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    ctr = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - ctr
        wm = wi * dx
        if k >= 0:

                grad[k:] += wm * (-x[:nv])

def lbfgs_fit(h, n, deadline, configs):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        h_cur = project_feasible(h.copy(), n)
        try:
            res = minimize(obj_smax, h_cur, jac=grad_smax,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 50000, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': 500000,
                                    'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-18:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_fit(h, n, deadline, alpha=10000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_eq_fun, 'jac': slqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, project_feasible(h.copy(), n),
                       args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def interp_up(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_reopt(h, n, deadline, n_trials=5, seed=77):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    mags = [0.01, 0.02, 0.015, 0.008, 0.025, 0.005]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        hp = project_feasible(h + mag * (rng.rand(n) - 0.5), n)
        try:
            res = minimize(obj_smax, hp, jac=grad_smax,
                           args=(n, 50000, 20000.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 10000, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-18:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def subgrad_polish(h, n, deadline, n_iter=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c))
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += (1.0 - h[k:])
                grad[k:] += (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += (1.0 - h[:nv])
                grad[:nv] += (-h[kk:])
        grad *= dx
        eta = 0.0005 / (1.0 + 0.005 * it)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-18:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_c
