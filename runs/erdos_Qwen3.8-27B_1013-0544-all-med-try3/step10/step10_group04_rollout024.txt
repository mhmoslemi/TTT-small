```python
import numpy as np
import time
import math
from scipy.optimize import minimize


def c5_vec(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_val(h):
    return float(np.max(c5_vec(h)))


def take_limit(remaining, desired, reserve):
    avail = remaining - reserve
    if avail <= 0.0:
        return 0.0
    if desired < 0.0:
        return 0.0
    if desired < avail:
        return float(desired)
    return float(avail)


def project_box_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    target = float(target)
    if target <= 0.0:
        return np.zeros_like(h)
    if target >= float(h.size):
        return np.ones_like(h)

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)


def enforce_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = float(target)

    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            i = int(np.argmax(room))
            if room[i] <= 1e-15:
                break
            h[i] = min(1.0, h[i] + min(diff, room[i]))
        else:
            i = int(np.argmax(h))
            if h[i] <= 1e-15:
                break
            h[i] = max(0.0, h[i] + min(diff, -h[i]))

    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0.0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmax(h))
        h[i] = min(max(0.0, h[i] + diff), 1.0)

    return h


def resample_h(h, n_new):
    n_new = int(n_new)
    if n_new % 2 == 1:
        n_new += 1
    if n_new <= 2:
        n_new = 4

    n_old = int(len(h))
    if n_new == n_old:
        h2 = h.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
        x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
        h2 = np.interp(x_new, x_old, h)

    h2 = project_box_sum(h2, 0.5 * float(n_new))
    return enforce_sum(h2, 0.5 * float(n_new))


def top_round(h, n):
    n = int(n)
    if n % 2 == 1:
        n += 1
    k = n // 2
    order = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[order[:k]] = 1.0
    return b


def run_length_h(n, p, q, alpha, beta, start_one):
    n = int(n)
    if n % 2 == 1:
        n += 1

    vals = []
    val = 1.0 if start_one else 0.0
    m = 0
    total = 0
    max_len = n + q + 5

    while total < max_len:
        bit = int(math.floor((m + 1) * alpha + beta) - math.floor(m * alpha + beta))
        length = p + bit * (q - p)
        vals.extend([val] * length)
        total += length
        val = 1.0 - val
        m += 1

    h = np.array(vals[:n], dtype=float)
    return top_round(h, n)


def add_idx_coeff(diff, h, idx, sign):
    n = len(h)
    center = n - 1
    pos_start = center + 1

    m = n - idx - 1
    if m > 0:
        diff[pos_start:pos_start + m] += sign * (1.0 - h[idx + 1:idx + 1 + m])

    m2 = idx
    if m2 > 0:
        diff[pos_start:pos_start + m2] -= sign * h[idx - 1:idx - 1 - m2:-1]

    m1 = idx
    if m1 > 0:
        diff[center - m1:center] += sign * (1.0 - h[idx - 1:idx - 1 - m1:-1])[::-1]

    m2 = n - idx - 1
    if m2 > 0:
        diff[center - m2:center] -= sign * h[idx + 1:idx + 1 + m2][::-1]


def pair_coeff_diff(h, j, l):
    n = len(h)
    diff = np.zeros(2 * n - 1, dtype=float)
    add_idx_coeff(diff, h, j, 1.0)
    add_idx_coeff(diff, h, l, -1.0)
    return diff


def pair_eval_max(base_c, diff, dx, a, s, center, x):
    d = x - a
    scale = d * dx
    vals = base_c + scale * diff
    delta0 = -2.0 * (x * x - a * a) + 2.0 * s * d
    vals[center] += dx * delta0
    return float(np.max(vals))


def golden_pair(blo, bhi, base_c, diff, dx, a, s, center, iters, best_x, best_f):
    if bhi - blo <= 1e-14:
        return best_x, best_f

    gr = 0.6180339887498949
    x1 = blo + (1.0 - gr) * (bhi - blo)
    x2 = blo + gr * (bhi - blo)

    f1 = pair_eval_max(base_c, diff, dx, a, s, center, x1)
    f2 = pair_eval_max(base_c, diff, dx, a, s, center, x2)

    if f1 < best_f:
        best_x, best_f = x1, f1
    if f2 < best_f:
        best_x, best_f = x2, f2

    for _ in range(int(iters)):
        if bhi - blo <= 1e-12:
            break
        if f1 < f2:
            bhi = x2
            x2 = x1
            f2 = f1
            x1 = blo + (1.0 - gr) * (bhi - blo)
            f1 = pair_eval_max(base_c, diff, dx, a, s, center, x1)
        else:
            blo = x1
            x1 = x2
            f1 = f2
            x2 = blo + gr * (bhi - blo)
            f2 = pair_eval_max(base_c, diff, dx, a, s, center, x2)

        if f1 < best_f:
            best_x, best_f = x1, f1
        if f2 < best_f:
            best_x, best_f = x2, f2

    return best_x, best_f


def search_pair_update(h, base_c, j, l, grid_points, refine_points):
    if j == l:
        return base_c, float(np.max(base_c))

    n = len(h)
    dx = 2.0 / float(n)
    center = n - 1

    a = float(h[j])
    b = float(h[l])
    s = a + b

    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)

    if hi - lo <= 1e-14:
        return base_c, float(np.max(base_c))

    diff = pair_coeff_diff(h, j, l)
    current_max = float(np.max(base_c))
    best_x = a
    best_f = current_max

    if grid_points > 1:
        xs = np.linspace(lo, hi, int(grid_points))
        for x in xs:
            if abs(float(x) - a) <= 1e-15:
                continue
            f = pair_eval_max(base_c, diff, dx, a, s, center, float(x))
            if f < best_f - 1e-15:
                best_f = f
                best_x = float(x)

    if refine_points > 0 and best_x > lo + 1e-12 and best_x < hi - 1e-12:
        step = (hi - lo) / max(int(grid_points) - 1, 1)
        blo = max(lo, best_x - step)
        bhi = min(hi, best_x + step)
        if bhi - blo > 1e-12:
            best_x, best_f = golden_pair(
                blo, bhi, base_c, diff, dx, a, s, center,
                int(refine_points), best_x, best_f
            )

    if best_f < current_max - 1e-14:
        d = best_x - a
        scale = d * dx
        new_base = base_c + scale * diff
        delta0 = -2.0 * (best_x * best_x - a * a) + 2.0 * s * d
        new_base[center] += dx * delta0

        y = s - best_x
        if y < 0.0:
            y = 0.0
            best_x = s
        elif y > 1.0:
            y = 1.0
            best_x = s - 1.0

        h[j] = best_x
        h[l] = y
        return new_base, float(np.max(new_base))

    return base_c, current_max


def select_target_pairs(h, base_c, max_lags, per_lag):
    n = len(h)
    center = n - 1
    pairs = []

    if len(base_c) == 0:
        return pairs

    max_lags = max(1, int(max_lags))
    per_lag = max(1, int(per_lag))

    order = np.argsort(base_c)[::-1][:max_lags]

    for idx in order:
        idx = int(idx)
        k = idx - center
        if k == 0:
            continue

        if k > 0:
            m = n - k
            if m <= 0:
                continue
            terms = h[:m] * (1.0 - h[k:])
            s = min(per_lag, m)
            if s > 0:
                if s == m:
                    ii = np.arange(m)
                else:
                    ii = np.argpartition(terms, -s)[-s:]
                for pos in ii:
                    i = int(pos)
                    pairs.append((i, i + k))
        else:
            r = -k
            m = n - r
            if m <= 0:
                continue
            terms = h[r:] * (1.0 - h[:m])
            s = min(per_lag, m)
            if s > 0:
                if s == m:
                    ii = np.arange(m)
                else:
                    ii = np.argpartition(terms, -s)[-s:]
                for pos in ii:
                    i = int(r + int(pos))
                    j = int(i - r)
                    pairs.append((i, j))

    seen = set()
    out = []
    for a, b in pairs:
        if a == b:
            continue
        key = (a, b) if a < b else (b, a)
        if key in seen:
            continue
        seen.add(key)
        out.append((a, b))
    return out


def run_pair_descent(h, n, deadline, grid_points=25, refine_points=12, max_restarts=2):
    n = int(n)
    if n <= 2:
        n = 4
    if n % 2 == 1:
        n += 1

    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    h = enforce_sum(h, target)

    base_c = c5_vec(h)
    current_max = float(np.max(base_c))
    best_h = h.copy()
    best_c = current_max

    rng = np.random.RandomState(98765)
    stag = 0
    restarts = 0

    while time.time() < deadline - 5.0:
        improved = False

        target_pairs = select_target_pairs(h, base_c, 4, 8)

        adj_even = [(j, j + 1) for j in range(0, n - 1, 2)]
        adj_odd = [(j, j + 1) for j in range(1, n - 1, 2)]

        rand_pairs = []
        rand_count = max(5, n // 8)
        for _ in range(rand_count):
            jj = int(rng.randint(n))
            ll = int(rng.randint(n))
            if jj != ll:
                rand_pairs.append((jj, ll))

        sweep_pairs = target_pairs + adj_even + adj_odd + rand_pairs

        seen = set()
        for j, l in sweep_pairs:
            if time.time() > deadline - 5.0:
                break
            key = (j, l) if j < l else (l, j)
            if key in seen:
                continue
            seen.add(key)

            base_c, cur = search_pair_update(
                h, base_c, key[0], key[1], grid_points, refine_points
            )
            if cur < current_max - 1e-15:
                current_max = cur
                improved = True

        h = enforce_sum(h, target)
        base_c = c5_vec(h)
        cur = float(np.max(base_c))

        if cur < best_c - 1e-15:
            best_c = cur
            best_h = h.copy()
            stag = 0
        else:
            if improved:
                stag = 1
            else:
                stag += 1

        if stag >= 3:
            if restarts < int(max_restarts):
                mag = 0.035 * (1.0 - 0.25 * float(restarts))
                h2 = h + mag * (rng.rand(n) - 0.5)
                h2 = project_box_sum(h2, target)
                h2 = enforce_sum(h2, target)
                c2 = float(c5_val(h2))
                if c2 < best_c + 0.01 and time.time() < deadline - 60.0:
                    h = h2
                    base_c = c5_vec(h)
                    current_max = float(np.max(base_c))
                    restarts += 1
                    stag = 0
                else:
                    break
            else:
                break

    best_h = enforce_sum(best_h, target)
    c = float(c5_val(best_h))
    if c < best_c:
        best_c = c
        best_h = best_h.copy()
    return best_h, best_c


def active_gradient(h, active, weights):
    n = len(h)
    dx = 2.0 / float(n)
    center = n - 1
    grad = np.zeros(n, dtype=float)

    for i, idx in enumerate(active):
        w = float(weights[i])
        k = int(idx) - center
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * dx
            grad[k:] -= w * dx * h[:n - k]
        else:
            r = -k
            grad[r:] += w * dx
            grad[:n - r] -= w * dx * h[r:]

    return grad


def active_gradient_descent(h, n, deadline, m=5, max_steps=80, eta=3.0):
    n = int(n)
    if n <= 2:
        n = 6
    if n % 2 == 1:
        n += 1

    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    h = enforce_sum(h, target)

    best_h = h.copy()
    best_c = float(c5_val(h))
    cur_c = best_c
    eta_used = float(eta)

    for _ in range(int(max_steps)):
        if time.time() > deadline - 3.0:
            break

        cvec = c5_vec(h)
        cur_c = float(np.max(cvec))
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()

        order = np.argsort(cvec)[::-1][:int(m)]
        vals = cvec[order]
        mx = float(np.max(vals))
        w = np.exp(30.0 * (vals - mx))
        sw = float(w.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.ones(len(order), dtype=float) / max(1, len(order))
        else:
            w = w / sw

        grad = active_gradient(h, order, w)

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = project_box_sum(h - eta_used * factor * grad, target)
            h2 = enforce_sum(h2, target)
            c2 = float(c5_val(h2))
            if c2 < cur_c - 1e-15:
                h = h2
                cur_c = c2
                accepted = True
                if eta_used < 20.0:
                    eta_used *= 1.1
                break

        if not accepted:
            eta_used *= 0.5
            if eta_used < 1e-5:
                break

        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()

    best_h = enforce_sum(best_h, target)
    c = float(c5_val(best_h))
    if c < best_c:
        best_c = c
        best_h = best_h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 15.0

    np.random.seed(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = np.full(500, 0.5, dtype=float)

    if parent.size <= 2:
        parent = np.full(500, 0.5, dtype=float)

    n_parent = int(parent.size)
    if n_parent % 2 == 1:
        parent = resample_h(parent, n_parent + 1)
        n_parent = int(parent.size)

    target_parent = 0.5 * float(n_parent)
    parent = project_box_sum(parent, target_parent)
    parent = enforce_sum(parent, target_parent)

    options = [parent, top_round(parent, n_parent)]

    if n_parent <= 1600:
        options.append(run_length_h(n_parent, 2, 3, 0.6180339887498949, 0.11, True))
        options.append(run_length_h(n_parent, 2, 3, 0.6180339887498949, 0.43, False))
        options.append(run_length_h(n_parent, 3, 4, 0.5, 0.02, True))

    best_h = parent.copy()
    best_n = n_parent
    best_c = float(c5_val(best_h))

    for opt in options[1:]:
        opt = project_box_sum(opt, target_parent)
        opt = enforce_sum(opt, target_parent)
        c = float(c5_val(opt))
        if c < best_c:
            best_c = c
            best_h = opt.copy()

    if budget_s <= 0:
        best_h = enforce_sum(best_h, 0.5 * float(best_n))
        best_c = float(c5_val(best_h))
        return best_h, best_c, int(best_n)

    now = time.time()
    limit = take_limit(deadline - now, 45.0, 700.0)
    if limit > 15.0 and best_n <= 1600:
        h_act, c_act = active_gradient_descent(
            best_h, best_n, now + limit, m=5, max_steps=70, eta=3.0
        )
        if c_act < best_c:
            best_c = c_act
            best_h = h_act.copy()

    now = time.time()
    limit = take_limit(deadline - now, 320.0, 420.0)
    if limit > 40.0:
        h_pair, c_pair = run_pair_descent(
            best_h, best_n, now + limit,
            grid_points=25, refine_points=12, max_restarts=2
        )
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair.copy()

    trial_ns = []
    for r in (0.5, 0.7, 1.3, 1.7):
        m = int(round(n_parent * r))
        if m % 2 == 1:
            m += 1
        if 80 <= m <= 2200 and m != n_parent:
            trial_ns.append(m)
    if best_n != n_parent and best_n not in trial_ns:
        trial_ns.append(best_n)

    trial_ns = sorted(set(trial_ns))[:4]
    attempts = 0

    for n_try in trial_ns:
        attempts += 1
        if attempts > 4:
            break
        if time.time() > deadline - 90.0:
            break

        h_res = resample_h(best_h, best_n, n_try)
        h_bin = top_round(h_res, n_try)
        h_run = run_length_h(n_try, 2, 3, 0.6180339887498949, 0.20, True)

        c_res = float(c5_val(h_res))
        c_bin = float(c5_val(h_bin))
        c_run = float(c5_val(h_run))

        init_h = h_res
        init_c = c_res
        if c_bin < init_c:
            init_h = h_bin
            init_c = c_bin
        if c_run < init_c:
            init_h = h_run
            init_c = c_run

        if init_c < best_c:
            best_c = init_c
            best_h = init_h.copy()
            best_n = n_try

        now = time.time()
        base_limit = 95.0
        if n_try > 900:
            base_limit = 65.0
        elif n_try > 700:
            base_limit = 80.0
        limit = take_limit(deadline - now, base_limit, 80.0)
        if limit < 25.0:
            continue

        grid = 17 if n_try > 750 else 21
        refine = 8
        h_pair, c_pair = run_pair_descent(
            init_h, n_try, now + limit,
            grid_points=grid, refine_points=refine, max_restarts=1
        )
        if c_pair < best_c:
            best_c = c_pair
            best_h = h_pair.copy()
            best_n = n_try

        if time.time() < deadline - 70.0:
            lim2 = take_limit(deadline - time.time(), 25.0, 55.0)
            if lim2 > 10.0:
                h_act, c_act = active_gradient_descent(
                    best_h, best_n, time.time() + lim2,
                    m=5, max_steps=45, eta=3.0
                )
                if c_act < best_c:
                    best_c = c_act
                    best_h = h_act.copy()

    if time.time() < deadline - 30.0:
        limit = take_limit(deadline - time.time(), 220.0, 15.0)
        if limit > 30.0:
            grid = 13 if best_n > 850 else 17
            h_final, c_final = run_pair_descent(
                best_h, best_n, time.time() + limit,
                grid_points=grid, refine_points=6, max_restarts=0
            )
            if c_final < best_c:
                best_c = c_final
                best_h = h_final.copy()

    best_h = enforce_sum(best_h, 0.5 * float(best_n))
    best_c = float(c5_val(best_h))

    h_bin_final = top_round(best_h, best_n)
    c_bin_final = float(c5_val(h_bin_final))
    if c_bin_final < best_c:
        best_h = h_bin_final
        best_c = c_bin_final

    best_h = enforce_sum(best_h, 0.5 * float(best_n))
    best_c = float(c5_val(best_h))

    return best_h, best_c, int(best_n)
```
