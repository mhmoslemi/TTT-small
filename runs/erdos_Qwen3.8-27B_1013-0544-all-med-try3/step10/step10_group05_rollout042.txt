import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            scale = target / tot
            h *= scale
            h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute remaining among non-bound variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

            h[free] += adj

    return h

    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
            h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    ic = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - ic
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 200 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def gradient_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in [1000, 10000, 100000, 1000000, 10000000]:
        if time.time() > deadline - 2:
            break
        stalled = 0
        while time.time() < deadline - 2 and stalled < 150:
            grad = full_grad(best_h, n, alpha, 0.0)
            gmax = np.abs(grad).max()
            if gmax < 1e-14:
                break
            sorted_idx = np.argsort(grad)
            j_best = sorted_idx[0]
            i_best = sorted_idx[-1]
            if i_best == j_best:
                stalled = 151
                break
            if best_h[i_best] < 0.005 or best_h[j_best] > 0.995:
                for j_cand in sorted_idx[:10]:
                    if j_cand != i_best and best_h[j_cand] < 0.995:
                        j_best = j_cand
                        break
                else:
                    for i_cand in sorted_idx[-10:]:
                        if i_cand != j_best and best_h[i_cand] > 0.005:
                            i_best = i_cand
                            break
                    else:
                        stalled += 10
                        continue
            improved = False
            for step in [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
                if best_h[i_best] - step < 0.0 or best_h[j_best] + step > 1.0:
                    continue
                best_h[i_best] -= step
                best_h[j_best] += step
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i_best] += step
                    best_h[j_best] -= step
            if improved:
                stalled = 0
            else:
                stalled += 1
    return best_h, best_c

def local_pair_search(h, n, deadline, top_m=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 500000
    stalled_total = 0
    while time.time() < deadline - 3 and stalled_total < 300:
        grad = full_grad(best_h, n, alpha, 0.0)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        sorted_idx = np.argsort(grad)
        donors = sorted_idx[-top_m:]
        recipients = sorted_idx[:top_m]
        improved = False
        for d in donors:
            if time.time() > deadline - 3:
                break
            for r in recipients:
                if d == r:
                    continue
                if best_h[d] < 0.003 or best_h[r] > 0.997:
                    continue
                for step in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                    if best_h[d] - step < 0.0 or best_h[r] + step > 1.0:
                        continue
                    best_h[d] -= step
                    best_h[r] += step
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[d] += step
                        best_h[r] -= step
                if improved:
                    break
            if improved:
                break
        if improved:
            stalled_total = 0
        else:
            stalled_total += 1
            alpha *= 2
            if alpha > 100000000:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    beatty_alphas = [golden / 2.0, np.sqrt(2.0) / 2.0, 0.4312, 0.3819,
                     (np.sqrt(5.0) - 1.0) / 4.0, 0.4450, 0.3989, 0.4142]
    for a in beatty_alphas:
        h_alt = make_beatty(n_orig, a)
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    for _ in range(4):
        h_rand = np.zeros(n_orig)
        idx = rng.choice(n_orig, size=n_orig // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand = project_feasible(h_rand, n_orig)
        c_rand = compute_c5(h_rand, n_orig)
        if c_rand < best_c:
            best_c = c_rand
            best_h = h_rand.copy()

    # Phase 1: Quick optimization at original resolution
    p1_end = min(t0 + 35, deadline - 900)
    if p1_end > t0 + 10:
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [2000, 20000, 200000]:
            if time.time() > p1_end - 5:
                break
            maxit = max(200, int((p1_end - time.time() - 3) * 3))
            maxit = min(maxit, 800)
            try:
                res = minimize(full_obj, best_h, jac=full_grad,
                              args=(best_n, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 12,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Upsample and optimize at higher resolution
    n_up = best_n * 2
    p2_start = time.time()
    p2_end = min(t0 + 450, deadline - 350)
    if p2_end > p2_start + 60 and n_up <= 4000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        bounds_up = [(0.0, 1.0)] * n_up
        for alpha in [1000, 10000, 100000, 1000000, 10000000]:
            if time.time() > p2_end - 15:
                break
            rem = p2_end - time.time() - 10
            if rem < 20:
                break
            maxit = max(300, int(rem / 5))
            maxit = min(maxit, 2500)
            try:
                res = minimize(full_obj, h_up, jac=full_grad,
                              args=(n_up, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds_up,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 12,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n_up)
                c_new = compute_c5(h_new, n_up)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new
            except Exception:
                pass
        if time.time() < p2_end - 30 and best_n == n_up:
            h_gs, c_gs = local_pair_search(h_up, n_up, p2_end, top_m=8)
            if c_gs < best_c:
                best_c = c_gs
                best_h = h_gs

    # Phase 3: More aggressive optimization at best resolution
    p3_end = min(t0 + 750, deadline - 200)
    if p3_end > time.time() + 30:
        cur_n = best_n
        bounds_cur = [(0.0, 1.0)] * cur_n
        for alpha in [10000, 100000, 1000000, 10000000, 100000000]:
            if time.time() > p3_end - 20:
                break
            rem = p3_end - time.time() - 15
            if rem < 20:
                break
            maxit = max(200, int(rem / 4))
            maxit = min(maxit, 3000)
            try:
                res = minimize(full_obj, best_h, jac=full_grad,
                              args=(cur_n, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds_cur,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 14,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, cur_n)
                c_new = compute_c5(h_new, cur_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 4: Gradient-guided local search
    p4_end = min(t0 + 980, deadline - 100)
    if p4_end > time.time() + 20:
        cur_n = best_n
        h_gs, c_gs = gradient_local_search(best_h, cur_n, p4_end)
        if c_gs < best_c:
            best_c = c_gs
            best_h = h_gs

    # Phase 5: Pair search refinement
    p5_end = deadline - 10
    if p5_end > time.time() + 15:
        cur_n = best_h.__len__()
        h_ps, c_ps = local_pair_search(best_h, cur_n, p5_end, top_m=6)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
