import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)

    v_v = np.zeros(n)

        grad += 10.0 * diff * np.ones(n)

        if w_sum < 1e-15:

            weights[top_idx] = 1.0 / top_k

        # Momentum
        vel = momentum * vel - lr * grad
        h_curr = h_curr + vel
        h_curr = project_feasible(h_curr, n)

        v_hat = v_v / (1.0 - beta2 ** t)
        lr = lr_init / (1.0 + t * 0.00005)
        h_curr = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_curr = project_feasible(h_curr, n)

    return best_h, best_c

def topk_gradient_descent(h_in, n, deadline, top_k=15, lr=0.003, momentum=0.85,
                           sum_penalty=3.0, patience=6000):
    h_curr = h_in.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    vel = np.zeros(n)
    target_sum = n / 2.0
    dx = 2.0 / n
    no_improve = 0

    for it in range(500000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        c_max = c_all.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break

        top_idx = np.argsort(c_all)[-top_k:]
        w_sum = 0.0
        weights = np.zeros(top_k)
        base_val = c_all[top_idx[0]] + 1e-8
        for j, ki in enumerate(top_idx):
            weights[j] = max(1e-6, c_all[ki] - base_val + 1e-6)
        w_sum = weights.sum()
        weights /= w_sum

        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            wk = weights[j] * dx
            lag = k - (n - 1)
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h_curr)
            elif lag > 0:
                grad[lag:n] += wk * (1.0 - h_curr[0:n - lag])
                grad[0:n - lag] += wk * (-h_curr[lag:n])
            else:
                idx = -lag
                grad[:n - idx] += wk * (1.0 - h_curr[idx:n])
                grad[idx:] += wk * (-h_curr[:n - idx])

        diff_sum = h_curr.sum() - target_sum
        grad += sum_penalty * diff_sum * np.ones(n)

        vel = momentum * vel - lr * grad
        h_curr = h_curr + vel
        h_curr = project_feasible(h_curr, n)

    return best_h, best_c

def targeted_swap(h_in, n, deadline, top_k=12, step_sizes=None, patience=300):
    if step_sizes is None:
        step_sizes = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    h_curr = h_in.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    no_improve = 0

    for ss in step_sizes:
        if time.time() > deadline:
            break
        for it in range(100000):
            if time.time() > deadline:
                break
            c_all = compute_all_lags(h_curr, n)
            c_max = c_all.max()
            if c_max < best_c - 1e-14:
                best_c = c_max
                best_h = h_curr.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > patience:
                break

            top_idx = np.argsort(c_all)[-top_k:]
            any_change = False
            for k in top_idx:
                lag = k - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                if m == 0:
                    continue

                if lag > 0:
                    overlap = h_curr[m:n] * (1.0 - h_curr[0:n - m])
                else:
                    overlap = h_curr[0:n - m] * (1.0 - h_curr[m:n])

                i_max = int(np.argmax(overlap))
                if lag > 0:
                    src = i_max + m
                    dst = i_max
                    amt = min(ss, h_curr[src] * 0.25, (1.0 - h_curr[dst]) * 0.25)
                else:
                    src = i_max
                    dst = i_max + m
                    amt = min(ss, h_curr[src] * 0.25, (1.0 - h_curr[dst]) * 0.25)

                if amt > 1e-10:
                    h_curr[src] -= amt
                    h_curr[dst] += amt
                    h_curr = project_feasible(h_curr, n)
                    any_change = True

            if not any_change:
                break
        else:
            continue

    return best_h, best_c

def interpolate(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=True)
    x_dst = np.linspace(0, 2, n_dst, endpoint=True)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def multi_restart(h_base, n, deadline, rng, n_restarts=5, alpha=200000):
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n)

    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        perturb_scale = 0.02 + 0.01 * r
        h_pert = h_base + rng.normal(0, perturb_scale, n)
        h_pert = project_feasible(h_pert, n)

        h_opt = lbfgsb_optimize(h_pert, n, alpha, time.time() + min(20, (deadline - time.time()) / 3),
                                  maxiter=800)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c

def fine_local_search(h_in, n, deadline, rng, top_k=8):
    h_curr = h_in.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    no_improve = 0

    for it in range(200000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h_curr, n)
        c_max = c_all.max()
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break

        top_idx = np.argsort(c_all)[-top_k:]
        k_pick = top_idx[rng.randint(len(top_idx))]
        lag = k_pick - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        if m >= n:
            continue

        i = rng.randint(0, n - m)
        if lag > 0:
            j = i + m
        else:
            j = i + m

        if h_curr[i] > h_curr[j] + 1e-8:
            d = min(0.003 * (1 + rng.random()), (h_curr[i] - h_curr[j]) * 0.2)
            h_curr[i] -= d
            h_curr[j] += d
        elif h_curr[j] > h_curr[i] + 1e-8:
            d = min(0.003 * (1 + rng.random()), (h_curr[j] - h_curr[i]) * 0.2)
            h_curr[i] += d
            h_curr[j] -= d
        else:
            continue

        if abs(d) < 1e-10:
            h_curr = project_feasible(h_curr, n)
            continue

        h_curr = project_feasible(h_curr, n)
        c_test = compute_c5(h_curr, n)
        if c_test >= best_c + 1e-14:
            h_curr = best_h.copy() if no_improve > 200 else h_curr

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3.0

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = None
        n = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n)
        best_c = compute_c5(best_h, n)
    else:
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B multi-alpha sweep at base resolution
    alpha_schedule = [500, 2000, 10000, 50000, 200000, 1000000]
    maxiter_schedule = [400, 400, 500, 500, 600, 600]
    for i, (alpha, maxit) in enumerate(zip(alpha_schedule, maxiter_schedule)):
        if time.time() > deadline - 500:
            break
        time_left = deadline - 500 - time.time()
        if time_left < 5:
            break
        alloc = min(15, time_left / (len(alpha_schedule) - i))
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + alloc, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Adam optimizer at base resolution
    if time.time() < deadline - 350:
        adam_deadline = min(deadline - 250, time.time() + 120)
        h_opt, c_opt = adam_optimize(best_h, n, adam_deadline, alpha_init=5000, lr_init=0.004)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Top-k gradient descent at base resolution
    if time.time() < deadline - 250:
        gk_deadline = min(deadline - 180, time.time() + 100)
        h_opt, c_opt = topk_gradient_descent(best_h, n, gk_deadline, top_k=15,
                                              lr=0.003, momentum=0.85, patience=5000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Targeted swap search
    if time.time() < deadline - 150:
        swap_deadline = min(deadline - 100, time.time() + 60)
        h_opt, c_opt = targeted_swap(best_h, n, swap_deadline, top_k=12)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: L-BFGS-B polish after local search
    if time.time() < deadline - 100:
        h_opt = lbfgsb_optimize(best_h, n, 5000000, time.time() + 20, maxiter=1500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h_opt, c_opt = adam_optimize(best_h, n, min(deadline - 80, time.time() + 30),
                                       alpha_init=50000, lr_init=0.002)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Fine local search with random pair swaps at top lags
    if time.time() < deadline - 80:
        fl_deadline = min(deadline - 40, time.time() + 50)
        h_opt, c_opt = fine_local_search(best_h, n, fl_deadline, rng, top_k=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Try higher resolution (n=750)
    n_h = 750
    if time.time() < deadline - 80:
        h_interp = interpolate(best_h, n, n_h)
        c_interp = compute_c5(h_interp, n_h)
        if time.time() < deadline - 60:
            h_opt = lbfgsb_optimize(h_interp, n_h, 20000, time.time() + 25, maxiter=1000)
            c_opt = compute_c5(h_opt, n_h)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_h
        if time.time() < deadline - 50:
            h_opt, c_opt = adam_optimize(best_h, len(best_h), min(deadline - 30, time.time() + 30),
                                           alpha_init=20000, lr_init=0.003)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = len(best_h)
        if time.time() < deadline - 40:
            h_opt, c_opt = topk_gradient_descent(best_h, len(best_h),
                                                   min(deadline - 20, time.time() + 30),
                                                   top_k=20, lr=0.002, patience=3000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = len(best_h)

    # Phase 8: Try even higher resolution (n=1000) if time permits
    n_h2 = 1000
    if time.time() < deadline - 60 and len(best_h) < n_h2:
        h_interp2 = interpolate(best_h, len(best_h), n_h2)
        if time.time() < deadline - 50:
            h_opt = lbfgsb_optimize(h_interp2, n_h2, 50000, time.time() + 20, maxiter=800)
            c_opt = compute_c5(h_opt, n_h2)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_h2
        if time.time() < deadline - 30:
            h_opt, c_opt = adam_optimize(best_h, len(best_h), min(deadline - 15, time.time() + 20),
                                           alpha_init=50000, lr_init=0.002)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 9: Multi-restart at best resolution
    n_final = len(best_h)
    if time.time() < deadline - 30:
        mr_deadline = min(deadline - 15, time.time() + 35)
        h_opt, c_opt = multi_restart(best_h, n_final, mr_deadline, rng,
                                       n_restarts=3, alpha=500000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 10: Final polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 10000000, deadline - 2, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
