    dx_val = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx_val
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj_f(x, n, k, alpha, mu):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[::-1][:k]
    top_vals = c[idx]
    m = top_vals[0]
    sh = alpha * (top_vals - m)
    val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad_f(x, n, k, alpha, mu):
    c = compute_c5_vec(x, n)
    idx_sorted = np.argsort(c)[::-1][:k]
    top_vals = c[idx_sorted]
    m = top_vals[0]
    sh = alpha * (top_vals - m)
    w = np.exp(sh)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for j in range(k):
        if w[j] < 1e-15:
            continue
        idx_val = idx_sorted[j]
        k_shift = idx_val - center
        wm = w[j] * dx_val
        if k_shift >= 0:
            nv = n - k_shift
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k_shift:n])
                grad[k_shift:n] += wm * (-x[0:nv])
        else:
            kk = -k_shift
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def constraint_eq_f(x, n):
    return float(x.sum() - n / 2.0)

def constraint_eq_jac(x, n):
    return np.ones(n)

def slqp_polish(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq_f, 'jac': constraint_eq_jac, 'args': (n,)}]
    
    def obj_slsqp(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    try:
        res = minimize(obj_slsqp, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_local_search(h, n, deadline, rng, max_rounds=40, n_swaps_per_round=500):
    h_bin = (h > 0.5).astype(float)
    target = n / 2.0
    diff = target - h_bin.sum()
    if diff > 0:
        order = np.argsort(-h)[:int(np.ceil(diff))]
        h_bin[order] = 1.0
    elif diff < 0:
        order = np.argsort(h)[:int(np.ceil(-diff))]
        h_bin[order] = 0.0
    h_bin = project_feasible(h_bin, n)
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 30:
            break
        ones = np.where(best_h >= 0.999)[0]
        zeros = np.where(best_h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        n_attempts = min(n_swaps_per_round, len(ones) * len(zeros) // 10)
        for _ in range(n_attempts):
            if time.time() > deadline - 30:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            best_h[i] = 0.0
            best_h[j] = 1.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
                ones = np.where(best_h >= 0.999)[0]
                zeros = np.where(best_h <= 0.001)[0]
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
        if not improved and rnd > 3:
            break
    return best_h, best_c

def coordinate_descent_binary(h, n, deadline, rounds=15):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for rnd in range(rounds):
        if time.time() > deadline - 20:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 20:
                break
            old_val = best_h[i]
            # Determine the binary target
            if old_val > 0.5:
                new_val = 0.0
            else:
                new_val = 1.0
            if abs(new_val - old_val) < 1e-6:
                continue
            # Need a swap partner
            # Find a partner
            if old_val > 0.5:
                # Turning 1->0, need to turn some 0->1
                zeros = np.where(best_h < 0.01)[0]
                if len(zeros) == 0:
                    break
                # Try the most impactful zero
                j_candidates = zeros[:min(10, len(zeros))]
            else:
                ones = np.where(best_h > 0.99)[0]
                if len(ones) == 0:
                    break
                j_candidates = ones[:min(10, len(ones))]
            
            for j in j_candidates:
                if j == i:
                    continue
                best_h[i] = new_val
                best_h[j] = 1.0 - j_candidates[0] * 0 + (0.0 if old_val > 0.5 else 1.0) if False else (0.0 if old_val > 0.5 else 1.0)
                # Actually set properly
                best_h[i] = 0.0 if old_val > 0.5 else 1.0
                best_h[j] = 1.0 if old_val > 0.5 else 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = old_val
                    best_h[j] = 1.0 - old_val if old_val > 0.5 else 0.0
                    # Restore
                    best_h[j] = (0.0 if old_val > 0.5 else 1.0)
        
        if not improved:
            break
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def pairwise_swap_cd(h, n, deadline, rounds=10):
    """Cleaner coordinate descent: random pairwise swaps."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(99)
    
    for rnd in range(rounds):
        if time.time() > deadline - 15:
            break
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = 300
        for _ in range(n_tries):
            if time.time() > deadline - 15:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h <= 0.5)[0]
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
        if not improved:
            break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, x_old * 0 + h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Standard L-BFGS at base resolution (120s)
    t1 = min(t0 + 120, deadline - 700)
    if t1 > time.time() + 10:
        bounds = [(0.0, 1.0)] * n_base
        for alpha, mu in [(200, 50.0), (1000, 500.0), (3000, 2000.0),
                          (8000, 10000.0), (20000, 50000.0), (50000, 100000.0)]:
            if time.time() > t1 - 5:
                break
            try:
                res = minimize(smooth_obj, best_h, jac=smooth_grad,
                              args=(n_base, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 15000, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': 150000, 'maxls': 50})
                h_new = project_feasible(res.x, n_base)
                c_new = compute_c5(h_new, n_base)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Phase 2: Top-k focused optimization (150s)
    t2 = min(t0 + 270, deadline - 550)
    if t2 > time.time() + 10:
        bounds = [(0.0, 1.0)] * n_base
        topk_configs = [(20, 5000, 200.0), (10, 10000, 500.0),
                        (15, 20000, 1000.0), (10, 30000, 2000.0),
                        (8, 50000, 5000.0), (20, 40000, 2000.0)]
        for k, alpha, mu in topk_configs:
            if time.time() > t2 - 5:
                break
            try:
                res = minimize(topk_obj_f, best_h, jac=topk_grad_f,
                              args=(n_base, k, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 8000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': 80000, 'maxls': 50})
                h_new = project_feasible(res.x, n_base)
                c_new = compute_c5(h_new, n_base)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Phase 3: SLSQP polish (80s)
    t3 = min(t0 + 350, deadline - 470)
    if t3 > time.time() + 10:
        h3, c3 = slqp_polish(best_h, n_base, t3, alpha=5000000, maxiter=4000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Binary local search with restarts (120s)
    t4 = min(t0 + 470, deadline - 350)
    if t4 > time.time() + 10:
        # First try direct binary search
        h_bin, c_bin = binary_local_search(best_h, n_base, t4, rng,
                                           max_rounds=30, n_swaps_per_round=400)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        # Restart with perturbation
        if time.time() < t4 - 30:
            h_pert = best_h + 0.05 * (rng.rand(n_base) - 0.5)
            h_pert = project_feasible(h_pert, n_base)
            h_bin2, c_bin2 = binary_local_search(h_pert, n_base, t4 - 10, rng,
                                                  max_rounds=20, n_swaps_per_round=300)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = h_bin2
    
    # Phase 5: Pairwise swap refinement (60s)
    t5 = min(t0 + 530, deadline - 290)
    if t5 > time.time() + 10:
        best_h, best_c = pairwise_swap_cd(best_h, n_base, t5, rounds=15)
    
    # Phase 6: Another round of top-k + SLSQP (80s)
    t6 = min(t0 + 610, deadline - 210)
    if t6 > time.time() + 10:
        bounds = [(0.0, 1.0)] * n_base
        for k, alpha, mu in [(10, 30000, 2000.0), (8, 50000, 5000.0)]:
            if time.time() > t6 - 20:

            try:
                res = minimize(topk_obj_f, best_h, jac=topk_grad_f,
                              args=(n_base, k, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 6000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': 60000})
                h_new = project_feasible(res.x, n_base)
                c_new = compute_c5(h_new, n_base)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
        if time.time() < t6 - 5:
            h6, c6 = slqp_polish(best_h, n_base, t6 - 2, alpha=10000000, maxiter=2000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Upsample to n=1000 and optimize (200s)
    n_hi = 1000
    if time.time() < deadline - 200:
        h_hi = interpolate_to(best_h, n_base, n_hi)
        t7 = min(time.time() + 100, deadline - 100)
        if t7 > time.time() + 10:
            bounds_hi = [(0.0, 1.0)] * n_hi
            for alpha, mu in [(3000, 2000.0), (10000, 10000.0), (30000, 50000.0)]:
                if time.time() > t7 - 5:
                    break
                try:
                    res = minimize(smooth_obj, h_hi, jac=smooth_grad,
                                  args=(n_hi, alpha, mu), method='L-BFGS-B',
                                  bounds=bounds_hi,
                                  options={'maxiter': 8000, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxfun': 80000})
                    h_new = project_feasible(res.x, n_hi)
                    c_new = compute_c5(h_new, n_hi)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        n_final = n_hi
                    else:
                        n_final = n_hi
                except Exception:
                    pass
            
            # Top-k at high res
            if time.time() < t7 - 5:
                for k, alpha, mu in [(15, 20000, 1000.0), (10, 50000, 5000.0)]:
                    if time.time() > t7 - 3:
                        break
                    try:
                        res = minimize(topk_obj_f, best_h, jac=topk_grad_f,
                                      args=(len(best_h), k, alpha, mu), method='L-BFGS-B',
                                      bounds=[(0.0, 1.0)] * len(best_h),
                                      options={'maxiter': 5000, 'ftol': 1e-20,
                                               'gtol': 1e-15, 'maxfun': 50000})
                        h_new = project_feasible(res.x, len(best_h))
                        c_new = compute_c5(h_new, len(best_h))
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h_new.copy()
                    except Exception:
                        pass
        
        # SLSQP at high res
        if time.time() < deadline - 60:
            t8 = min(time.time() + 50, deadline - 30)
            if t8 > time.time() + 5:
                h8, c8 = slqp_polish(best_h, len(best_h), t8, alpha=15000000, maxiter=2000)
                if c8 < best_c:
                    best_c = c8
                    best_h = h8
        
        # Binary search at high res
        if time.time() < deadline - 30:
            h_bi, c_bi = binary_local_search(best_h, len(best_h), deadline - 10, rng,
                                              max_rounds=10, n_swaps_per_round=200)
            if c_bi < best_c:

                best_h = h_bi
    
    # Phase 8: Final polish
    n_final = len(best_h)
    if time.time() < deadline - 10:
        best_h, best_c = pairwise_swap_cd(best_h, n_final, deadline - 5, rounds=5)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
