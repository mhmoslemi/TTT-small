```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h -= (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (500, 200.0),
                   (1000, 1000.0), (3000, 5000.0), (10000, 20000.0),
                   (30000, 50000.0), (100000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_sweeps=500):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(77)
    c_vec = np.correlate(h, 1 - h, mode='full')
    center = n - 1
    n2m1 = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        n_changed = 0
        order = rng.permutation(n)
        for i in order:
            hi = h[i]
            g = np.zeros(n2m1)
            g[center] = 1.0 - 2.0 * hi
            if i > 0:
                kmax1 = min(i, n - 1)
                idx_l = np.arange(i - 1, i - 1 - kmax1, -1)
                g[center + 1:center + kmax1 + 1] += (1.0 - h[idx_l])
                g[center - kmax1:center] -= h[idx_l]
            if i < n - 1:
                kmax2 = n - 1 - i
                idx_r = np.arange(i + 1, i + 1 + kmax2)
                g[center + 1:center + kmax2 + 1] -= h[idx_r]
                g[center - kmax2:center] += (1.0 - h[idx_r])
            lo_u = -hi
            hi_u = 1.0 - hi
            if lo_u > hi_u:
                continue
            for _ in range(50):
                mid_u = 0.5 * (lo_u + hi_u)
                vals = c_vec + g * mid_u
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-13
                if not np.any(active):
                    active = vals == f_mid
                min_g = np.min(g[active])
                max_g = np.max(g[active])
                if max_g < -1e-15:
                    lo_u = mid_u
                elif min_g > 1e-15:
                    hi_u = mid_u
                else:
                    break
            t_opt = hi + 0.5 * (lo_u + hi_u)
            t_opt = max(0.0, min(1.0, t_opt))
            delta = t_opt - hi
            if abs(delta) > 1e-12:
                h[i] = t_opt
                c_vec += g * delta
                n_changed += 1
        h = project_feasible(h, n)
        c_vec = np.correlate(h, 1 - h, mode='full')
        c_now = float(np.max(c_vec)) * (2.0 / n)
        if c_now < best_c - 1e-15:
            best_c = c_now
            best_h = h.copy()
        elif not (c_now < best_c - 1e-15):
            h = best_h.copy()
            c_vec = np.correlate(h, 1 - h, mode='full')
        if n_changed == 0 and sweep > 10:
            break
        if n_changed < 5 and sweep > 50:
            break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = np.correlate(x, 1 - x, mode='full')
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def eq_fun(x, na):
        return float(x.sum() - na / 2.0)
    def eq_jac(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for rnd in range(200):
        if time.time() > deadline:
            break
        c_vec = np.correlate(h, 1 - h, mode='full')
        worst = int(np.argmax(c_vec))
        presented = 0
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            if present(i):
                continue
            for delta in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                old = h[i]
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    presented += 1
                    break
                else:
                    h[i] = old
            if presented > 5:
                break
        if presented == 0 and rnd > 5:
            break
    return best_h, best_c

    return False

def perturbation_search(h, n, deadline, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    mags = [0.02, 0.04, 0.01, 0.06, 0.03, 0.08, 0.015, 0.05]
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 3000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 5000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c
```
