def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    if h.shape!=(n,): h=np.resize(h,n)
    h[~np.isfinite(h)] = 0.5
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    if abs(h.sum()-target) < 1e-12:
        return h
    lo=float(np.min(h)-1.0)
    hi=float(np.max(h)+1.0)
    for _ in range(80):
       mid=(lo+hi)*0.5
       s=np.sum(np.clip(h-mid,0.0,1.0))
       if s > target: lo=mid
       else: hi=mid
    tau=(lo+hi)*0.5
    x=np.clip(h-tau,0.0,1.0)
    for _ in range(50):
       diff=target - float(np.sum(x))
       if abs(diff) < 1e-13: break
       mask=(x>1e-12)&(x<1.0-1e-12)
       cnt=int(np.count_nonzero(mask))
       if cnt == 0: break
       step=diff / cnt
       if step > 0.0:
          room = 1.0 - x[mask]
          max_step = float(np.min(room))
          step = min(step, max_step)
       else:
          floor = float(np.min(x[mask]))

    F=np.fft.fft(x,N)
    A=np.fft.ifft(F*np.conjugate(F),N).real[:n]

    c=np.empty(2*n-1, dtype=float)

       c[n:]=pref[n_minus_s]-A[s]
       c[:n-1]=(H-pref[s]-A[s])[::-1]

    m=float(np.max(c))
    sh=alpha*(c-m)
    smax=float(np.max(sh))

import numpy as np
import time
from scipy.optimize import minimize

_PRE_CACHE = {}
_OBJ_SCALE = 500.0


def get_pre(n):
    val = _PRE_CACHE.get(n)
    if val is not None:
        return val
    N = 1
    while N < 2 * n:
        N <<= 1
    if n > 1:
        s = np.arange(1, n, dtype=int)
        n_minus_s = n - s
        kneg_idx = N - s
    else:
        s = np.empty(0, dtype=int)
        n_minus_s = np.empty(0, dtype=int)
        kneg_idx = np.empty(0, dtype=int)
    val = (N, s, n_minus_s, kneg_idx)
    _PRE_CACHE[n] = val
    return val


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)

    target = 0.5 * float(n)
    if abs(float(np.sum(h)) - target) < 1e-13:
        return h

    lo = float(np.min(h) - 1.0)
    hi = float(np.max(h) + 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    x = np.clip(h - tau, 0.0, 1.0)

    for _ in range(50):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        step = diff / cnt
        if step > 0.0:
            room = 1.0 - x[mask]
            step = min(step, float(np.min(room)))
        else:
            floor = float(np.min(x[mask]))
            step = max(step, -floor)
        x[mask] += step

    x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            x[mask] += diff / cnt
    x = np.clip(x, 0.0, 1.0)
    return x


def resample_h(h, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    h = np.asarray(h, dtype=float).copy()
    if n_old <= 0 or n_new <= 0:
        return np.full(n_new, 0.5)
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, n_new)


def compute_c_vec_rfft(x, n):
    N, s, n_minus_s, _ = get_pre(n)
    F = np.fft.rfft(x, N)
    A = np.fft.irfft(F * np.conjugate(F), N).real[:n]
    H = float(np.sum(x))

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    if n > 1:
        pref[1:] = np.cumsum(x)

    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1] = H - A[0]
    if n > 1:
        c[n:] = pref[n_minus_s] - A[s]
        c[:n - 1] = (H - pref[s] - A[s])[::-1]
    c *= 2.0 / n
    return c


def compute_c5_fast(h, n):
    return float(np.max(compute_c_vec_rfft(h, n)))


def compute_c5_exact(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def compute_c_vec_fft_complex(x, n):
    N, s, n_minus_s, _ = get_pre(n)
    F = np.fft.fft(x, N)
    A = np.fft.ifft(F * np.conjugate(F), N).real[:n]
    H = float(np.sum(x))

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    if n > 1:
        pref[1:] = np.cumsum(x)

    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1] = H - A[0]
    if n > 1:
        c[n:] = pref[n_minus_s] - A[s]
        c[:n - 1] = (H - pref[s] - A[s])[::-1]
    c *= 2.0 / n
    return c


def smooth_full_obj(x, n, alpha, scale, mu):
    c = compute_c_vec_fft_complex(x, n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    smax = float(np.max(sh))
    ew = np.exp(sh - smax)
    lse = smax + float(np.log(float(np.sum(ew))))
    H = float(np.sum(x))
    val = scale * (m + lse / alpha)
    if mu:
        val += scale * mu * (H - 0.5 * float(n)) ** 2
    return float(val)


def smooth_full_grad(x, n, alpha, scale, mu):
    N, s, n_minus_s, kneg_idx = get_pre(n)
    F = np.fft.fft(x, N)
    A = np.fft.ifft(F * np.conjugate(F), N).real[:n]
    H = float(np.sum(x))
    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    if n > 1:
        pref[1:] = np.cumsum(x)

    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1] = H - A[0]
    if n > 1:
        c[n:] = pref[n_minus_s] - A[s]
        c[:n - 1] = (H - pref[s] - A[s])[::-1]
    c *= dx

    m = float(np.max(c))
    sh = alpha * (c - m)
    smax = float(np.max(sh))
    ew = np.exp(sh - smax)
    sw = float(np.sum(ew))
    w = ew / sw

    P = np.zeros(n, dtype=float)
    Nneg = np.zeros(n, dtype=float)
    if n > 1:
        P[1:] = w[n:]
        Nneg[1:] = w[:n - 1][::-1]

    T = P + Nneg
    Pcum = np.cumsum(P)
    Ncum = np.cumsum(Nneg)

    base = w[n - 1] * (1.0 - 2.0 * x) + Pcum[::-1] + Ncum

    if n > 1:
        Kpos = np.zeros(N, dtype=float)
        Kneg = np.zeros(N, dtype=float)
        Kpos[1:n] = T[1:]
        Kneg[kneg_idx] = T[1:]

        Fk_pos = np.fft.fft(Kpos)
        Fk_neg = np.fft.fft(Kneg)

        plus = np.fft.ifft(F * Fk_neg).real[:n]
        minus = np.fft.ifft(F * Fk_pos).real[:n]
        base -= plus
        base -= minus

    grad = (scale * dx) * base
    if mu:
        grad += (2.0 * scale * mu) * (H - target)
    return grad

                jac=smooth_full_grad,
                args=(int(n), float(alpha), _OBJ_SCALE, float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(mi),
                    "ftol": 1e-14,
                    "gtol": 1e-10,
                    "maxfun": int(max(500, 4 * mi)),
                    "maxls": 50,
                },
            )
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break

        h_new = project_feasible(res.x, n)
        c_new = compute_c5_fast(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            cur = h_new
        elif c_new < best_c + 1e-4:
            cur = h_new
        else:
            cur = best_h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5_fast(best_h, n)
    return best_h, best_c


def try_all_cyclic_shifts(h, n):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5_fast(h, n)

    if n <= 4096:
        steps = list(range(1, n))
    else:
        step = int(n // 4096) + 1
        steps = list(range(step, n, step))

    for r in steps:
        hr = np.roll(h, r)
        cr = compute_c5_fast(hr, n)
        if cr < best_c - 1e-15:
            best_c = cr
            best_h = hr.copy()

    hr = h[::-1].copy()
    cr = compute_c5_fast(hr, n)
    if cr < best_c - 1e-15:
        best_c = cr
        best_h = hr.copy()

    return best_h, best_c


def random_perturb(h, n, amp, seed):
    rng = np.random.default_rng(int(seed))
    z = rng.normal(size=n)
    z -= z.mean()
    norm = float(np.linalg.norm(z))
    if norm > 0.0:
        z *= amp / norm
    return project_feasible(h + z, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 25.0
    seed = int(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 500
        h0 = np.full(n0, 0.5)

    if n0 < 2 or not np.all(np.isfinite(h0)):
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_exact = compute_c5_exact(h0, n0)

    if time.time() < deadline - 80.0 and n0 <= 2048:
        h_sh, c_sh = try_all_cyclic_shifts(h0, n0)
        c_sh_ex = compute_c5_exact(h_sh, n0)
        if c_sh_ex < best_exact:
            best_h = h_sh
            best_n = n0
            best_exact = c_sh_ex

    t1 = min(t0 + 180.0, deadline - 450.0)
    if t1 > time.time() + 30.0:
        h1, _ = run_full_smooth(best_h, n0, t1)
        c1_ex = compute_c5_exact(h1, n0)
        if c1_ex < best_exact:
            best_h = h1
            best_n = n0
            best_exact = c1_ex

        if time.time() < deadline - 100.0 and best_n <= 2048:
            h1_sh, _ = try_all_cyclic_shifts(best_h, best_n)
            c1_sh_ex = compute_c5_exact(h1_sh, best_n)
            if c1_sh_ex < best_exact:
                best_h = h1_sh
                best_exact = c1_sh_ex

    n_high = 2048
    if best_n > n_high:
        n_high = best_n
    if n_high < 256:
        n_high = 256

    high_cands = []
    h_high = resample_h(best_h, best_n, n_high)
    high_cands.append(h_high)
    high_cands.append(resample_h(h0, n0, n_high))

    if time.time() < deadline - 550.0:
        high_cands.append(random_perturb(h_high, n_high, 0.02, seed))
        high_cands.append(random_perturb(h_high, n_high, 0.05, seed + 1))
        high_cands.append(random_perturb(h0, n0, 0.03, seed + 2))

    best_high_h = None
    best_high_fast = 1.0e9
    for hc in high_cands:
        hc = project_feasible(hc, n_high)
        cc = compute_c5_fast(hc, n_high)
        if cc < best_high_fast:
            best_high_fast = cc
            best_high_h = hc.copy()

    if best_high_h is not None and time.time() < deadline - 300.0:
        h_high_sh, c_high_sh = try_all_cyclic_shifts(best_high_h, n_high)
        if c_high_sh < best_high_fast:
            best_high_h = h_high_sh
            best_high_fast = c_high_sh

    t2 = deadline - 150.0
    if best_high_h is not None and t2 > time.time() + 50.0:
        h2, _ = run_full_smooth(best_high_h, n_high, t2)
        c2_ex = compute_c5_exact(h2, n_high)
        if c2_ex < best_exact:
            best_h = h2
            best_n = n_high
            best_exact = c2_ex

        if time.time() < deadline - 90.0 and best_n <= 2048:
            h2_sh, _ = try_all_cyclic_shifts(best_h, best_n)
            c2_sh_ex = compute_c5_exact(h2_sh, best_n)
            if c2_sh_ex < best_exact:
                best_h = h2_sh
                best_exact = c2_sh_ex

    if time.time() < deadline - 50.0:
        if best_n <= 800:
            final_configs = [
                (3000, 0.5, 250),
                (10000, 0.3, 400),
                (50000, 0.3, 600),
                (200000, 0.3, 900),
            ]
        else:
            final_configs = [
                (3000, 0.5, 200),
                (10000, 0.3, 350),
                (50000, 0.3, 550),
                (200000, 0.3, 800),
            ]
        hf, _ = run_full_smooth(best_h, best_n, deadline - 5.0, configs=final_configs)
        cf_ex = compute_c5_exact(hf, best_n)
        if cf_ex < best_exact:
            best_h = hf
            best_exact = cf_ex

    if time.time() < deadline - 25.0:
        scan_ns = [512, 640, 768, 896, 1024, 1280, 1536, 1792, 2048]
        if best_n not in scan_ns:
            scan_ns.append(best_n)
        if n0 not in scan_ns:
            scan_ns.append(n0)

        for nn in scan_ns:
            nn = int(nn)
            if nn < 16 or nn > 4096:
                continue
            if nn == best_n:
                continue
            try:
                hc = resample_h(best_h, best_n, nn)
                c_ex = compute_c5_exact(hc, nn)
                if c_ex < best_exact:
                    best_h = hc
                    best_n = nn
                    best_exact = c_ex
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5_exact(best_h, best_n)

    if not np.isfinite(final_c) or final_c > 0.9:
        best_h = h0
        best_n = n0
        best_h = project_feasible(best_h, best_n)
        final_c = compute_c5_exact(best_h, best_n)

    return best_h, float(final_c), int(best_n)
