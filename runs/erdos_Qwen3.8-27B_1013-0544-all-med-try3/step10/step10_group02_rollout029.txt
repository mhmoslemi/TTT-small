def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_for_shift(h, k, n):
    """Compute gradient of overlap at shift k."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return grad
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return grad
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

    grads = []

        k = idx - (n - 1)
        grads.append(compute_grad_for_shift(h, k, n))
    return np.array(grads)  # shape (n_active, n)

import time

    return np.correlate(h, 1 - h, mode='full') * dx

            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    """Gradient of overlap at shift k (where k = idx - (n-1))."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:

        if nv <= 0:
            return grad

            return grad
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

        active = np.where(c_vec >= threshold)[0]
    else:
        active = np.argsort(-c_vec)[:top_k]

    One step of the LP-based linearization method.
    Returns (h_new, c_new_val) where c_new_val is the LP's predicted max.

    top_shifts = np.argsort(-c_vec)[:top_k_lp]

    vals = []

    """Find minimum-norm subgradient direction for max_k f_k."""
    # Compute gradients for active shifts
    grads = []
    for idx in active_idx:
        k = idx - (n - 1)
        grads.append(compute_grad_k(h, k, n))
    G = np.array(grads)  # (m, n)
    m = G.shape[0]
    
    if m == 1:
        return -G[0], 1.0
    
    # Find w >= 0, sum(w) = 1, minimizing ||G^T w||^2
    # = minimizing w^T (G G^T) w
    # This is a QP: min w^T Q w s.t. sum(w) = 1, w >= 0
    # where Q = G @ G.T
    Q = G @ G.T  # (m, m)
    
    # Simple approach: use projected gradient or a small QP
    # For small m, we can use scipy.optimize.minimize
    from scipy.optimize import minimize as sc_min
    
    def q_obj(w):
        return w @ Q @ w
    
    def q_jac(w):
        return 2.0 * Q @ w
    
    w0 = np.ones(m) / m
    bounds = [(0, 1)] * m
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1, 'jac': lambda w: np.ones(m)}]
    
    res = sc_min(q_obj, w0, jac=q_jac, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
    
    if res.status == 0:
        w = res.x
    else:
        w = w0
    
    direction = -G.T @ w
    norm = np.linalg.norm(direction)
    return direction, norm

    m = c.max()
    sh = alpha * (c - m)

        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _bilevel_obj(w):
    return w[0]

def _bilevel_jac(w):
    return np.array([1.0])

def _sum_con_fun(w):
    return w.sum() - 1.0

def _sum_con_jac(w):
    return np.ones_like(w)

def min_norm_subgrad(h, n, active_idx):
    grads = []
    for idx in active_idx:
        k = idx - (n - 1)
        grads.append(compute_grad_k(h, k, n))
    G = np.array(grads)
    m = G.shape[0]
    if m == 1:
        return -G[0]
    Q = G @ G.T
    w0 = np.ones(m) / m
    
    def qfun(w):
        return float(w @ Q @ w)
    
    def qjac(w):
        return 2.0 * (Q @ w)
    
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(qfun, w0, jac=qjac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
        if res.status == 0:
            w = res.x
        else:
            w = w0
    except Exception:
        w = w0
    return -(G.T @ w)

def lbfgs_optimize(h, n, deadline, alphas=None, mu=0.01, maxiter_per=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 8:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(400, int(remaining * 2))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 5,
                                   'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def lp_linearize_step(h, n, top_k_lp):
    c_vec = compute_c5_vec(h, n)
    top_shifts = np.argsort(-c_vec)[:top_k_lp]
    grads = []
    vals = []
    for idx in top_shifts:
        k = int(idx) - (n - 1)
        grads.append(compute_grad_k(h, k, n))
        vals.append(c_vec[idx])
    G = np.array(grads)
    v = np.array(vals)
    m = len(top_shifts)
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = G @ h - v
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-np.inf, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        if result.status == 0:
            return result.x[:n], result.x[n], top_shifts
    except Exception:
        pass
    return h, c_vec.max(), top_shifts

def lp_method(h, n, deadline, top_k_vals=None, max_iters=1000):
    if top_k_vals is None:
        top_k_vals = [15, 20, 30, 50]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        used_top_k = top_k_vals[it % len(top_k_vals)]
        h_lp, t_lp, _ = lp_linearize_step(best_h, n, used_top_k)
        d = h_lp - best_h
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-11:
            stall_count += 1
            if stall_count > 5:
                break
            continue
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]:
            h_cand = project_feasible(best_h + alpha * d, n)
            c_cand = compute_c5(h_cand, n)
            if c_cand < best_c - 1e-14:
                best_h = h_cand
                best_c = c_cand
                improved = True
                stall_count = 0
                break
        if not improved:
            stall_count += 1
            if stall_count > 15:
                break
    return best_h, best_c

def subgrad_method(h, n, deadline, max_iters=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.008
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 25:
            active = np.argsort(-c_vec)[:25]
        direction = min_norm_subgrad(best_h, n, active)
        norm = np.linalg.norm(direction)
        if norm < 1e-10:
            break
        direction = direction / norm
        h_cand = project_feasible(best_h - eta * direction, n)
        c_cand = compute_c5(h_cand, n)
        if c_cand < best_c - 1e-14:
            best_h = h_cand
            best_c = c_cand
            eta = min(eta * 1.01, 0.04)
        else:
            eta *= 0.97
    return best_h, best_c

def pair_refine(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=50000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slqp_cons, 'jac': _slqp_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def _slqp_cons(x, n):
    return x.sum() - n / 2.0

def _slqp_jac(x, n):
    return np.ones(n)

def make_periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(min(ones_per_period, period)):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 1e-12:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < -1e-12:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def upsample(h, n_from, n_to):
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    h_up = np.interp(x_to, x_from, h)
    return project_feasible(h_up, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        n_orig = 500
        h0 = np.full(n_orig, 0.5)

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick L-BFGS at original resolution (40s)
    p1_end = t0 + 40
    h1, c1 = lbfgs_optimize(best_h, best_n, p1_end,
        alphas=[100, 500, 2000, 10000, 50000],
        mu=0.01, maxiter_per=[1500, 2000, 2000, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: LP-based method at original resolution (100s)
    if time.time() < deadline - 200:
        p2_end = min(time.time() + 100, deadline - 200)
        h2, c2 = lp_method(best_h, best_n, p2_end,
                          top_k_vals=[15, 20, 30], max_iters=500)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient refinement (60s)
    if time.time() < deadline - 150:
        p3_end = min(time.time() + 60, deadline - 150)
        h3, c3 = subgrad_method(best_h, best_n, p3_end, max_iters=3000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to n=1000 and optimize (250s)
    n_fine = 1000
    if time.time() < deadline - 200:
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        
        p4a_end = min(time.time() + 80, deadline - 180)
        h4, c4 = lbfgs_optimize(h_fine, n_fine, p4a_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000],
            mu=0.005, maxiter_per=[1000, 1500, 2000, 2000, 2500, 2500])
        
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_fine
        
        if time.time() < deadline - 120:
            p4b_end = min(time.time() + 100, deadline - 120)
            h4b, c4b = lp_method(best_h, best_n, p4b_end,
                                top_k_vals=[20, 30, 40, 50], max_iters=800)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b
                best_n = n_fine
        
        if time.time() < deadline - 80:
            p4c_end = min(time.time() + 50, deadline - 80)
            h4c, c4c = subgrad_method(best_h, best_n, p4c_end, max_iters=2000)
            if c4c < best_c:
                best_c = c4c
                best_h = h4c
                best_n = n_fine

    # Phase 5: Alternative start at n=1000 (periodic pattern) (100s)
    if time.time() < deadline - 120:
        for (period, ones) in [(4, 2), (5, 2), (6, 3), (7, 3)]:
            h_alt = make_periodic_start(n_fine, period, ones)
            c_alt = compute_c5(h_alt, n_fine)
            p5_end = min(time.time() + 25, deadline - 100)
            h5, c5 = lbfgs_optimize(h_alt, n_fine, p5_end,
                alphas=[200, 2000, 20000, 200000],
                mu=0.01, maxiter_per=[1000, 1500, 1500, 2000])
            if c5 < best_c:
                best_c = c5
                best_h = h5
                best_n = n_fine
            if time.time() > deadline - 80:
                break

    # Phase 6: LP at best resolution (60s)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 55, deadline - 50)
        h6, c6 = lp_method(best_h, best_n, p6_end,
                          top_k_vals=[20, 30, 40], max_iters=400)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish (40s)
    if time.time() < deadline - 40:
        for alpha in [50000, 200000]:
            if time.time() > deadline - 30:
                break
            p7_end = min(time.time() + 18, deadline - 25)
            h7, c7 = slqp_polish(best_h, best_n, p7_end, alpha=alpha, maxiter=800)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Pair refinement (30s)
    if time.time() < deadline - 25:
        p8_end = min(time.time() + 25, deadline - 8)
        h8, c8 = pair_refine(best_h, best_n, p8_end, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final subgradient (remaining)
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = subgrad_method(best_h, best_n, p9_end, max_iters=2000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
