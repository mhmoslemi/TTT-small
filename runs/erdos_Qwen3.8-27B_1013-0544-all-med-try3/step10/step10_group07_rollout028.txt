```python
    dx = 2.0 / n
    target = n / 2.0
    p = np.ones(2*n-1) / (2*n-1)  # initial distribution over shifts
    
    for iter in range(iterations):
        # Compute all correlations
        corr = np.correlate(h, 1-h, mode='full') * dx
        
        # Update distribution
        p = np.exp(alpha * corr)
        p /= p.sum()
        
        # Sample a shift (or use expectation)
        # For efficiency, use weighted average gradient
        grad = np.zeros(n)
        for k_idx in range(2*n-1):
            k = k_idx - (n-1)
            # gradient of corr[k] w.r.t. h
            # corr[k] = sum_i h[i] * (1-h[i+k]) for valid i
            # d/dh[j] = sum over valid i of [delta(i,j)*(1-h[i+k]) + delta(i+k,j)*(-h[i])]
            # = (1-h[j+k]) - h[j-k] (for valid indices)
            g = np.zeros(n)
            # term 1: (1 - h[j+k]) for j where j+k is valid
            for j in range(n):
                jp = j + k
                jm = j - k
                val = 0.0
                if 0 <= jp < n:
                    val += (1 - h[jp])
                if 0 <= jm < n:
                    val -= h[jm]
                g[j] = val
            grad += p[k_idx] * g * dx
        
        # Gradient step
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        if compute_c5(h_new, n) < compute_c5(h, n):
            h = h_new
    
    return h
```

Let me now think about the continuous relaxation. The key insight is that the optimal solution might not be binary. By allowing h to take values in [0,1], we have more degrees of freedom and might achieve a lower C₅.

The SLSQP approach with softmax smoothing is one way, but it's slow. A faster approach is:

**Projected Gradient Descent with Smoothing:**
1. Maintain h (continuous, in [0,1], sum = n/2)
2. Compute corr = correlate(h, 1-h) * dx
3. Compute the smoothed max: m = logsumexp(alpha * corr) / alpha (for large alpha, approximates max)
4. Compute gradient: ∇h m = alpha * softmax(alpha * corr) ⊙ ∇h corr
5. Take step: h ← h - eta * ∇h m
6. Project: clip to [0,1] and adjust sum to n/2
7. Repeat

The gradient ∇h corr[k] for a fixed k:
As computed earlier: (1-h[i+k]) - h[i-k] (for valid indices, times dx)

The softmax-weighted gradient:
∇h m = dx * sum_k softmax_k * [(1-h[i+k])*valid(i+k) - h[i-k]*valid(i-k)]

This is a weighted sum of O(n) vectors, each of length n. Total: O(n²) per iteration.

For n=500: 250K per iteration. Can do thousands of iterations quickly.

This should be effective for the continuous relaxation phase.

```python
def binary_local_search(h, n, deadline, max_sweeps=10000, seed=42):

    target = n // 2  # must be integer
    # Ensure h is binary with exactly target ones
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = corr.max() * dx
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Compute gradient for all possible moves (vectorized)
        # For each zero position i and one position j:
        # delta[k] = dx * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
        
        # This is hard to vectorize directly. Let's use a different approach.
        
        # Approach: for each zero i, compute the effect of adding a 1 there
        # effect_add[i][k] = dx * [(1-h[i+k]) - h[i-k]]
        # For each one j, compute the effect of removing a 1 there
        # effect_rem[j][k] = -dx * [(1-h[j+k]) - h[j-k]]
        # Total effect of swap (i,j) = effect_add[i] + effect_rem[j]
        
        # We want to find (i,j) that minimizes max(corr + effect_add[i] + effect_rem[j])
        
        # This is equivalent to: minimize max_k [corr[k] + effect_add[i][k] + effect_rem[j][k]]
        
        # For a fixed i, the best j is the one that minimizes the max.
        # This is still O(n) per (i,j) pair.
        
        # Alternative: use a greedy heuristic
        # For each i in zeros:
        #   Compute effect_add[i]
        #   Find the k* that maximizes corr + effect_add[i]
        #   For that k*, find j that minimizes effect_rem[j][k*]
        #   Check if the full swap improves
        
        improved = False
        z_list = sorted(zeros)
        rng.shuffle(z_list)  # random order to avoid bias
        
        for i in z_list:
            if time.time() > deadline:
                break
            
            # Compute effect of adding 1 at position i
            eff_i = np.zeros(2*n - 1)
            # (1-h[i+k]) for valid k where i+k in [0,n-1], k != 0
            k_vals = np.arange(max(-i, -(n-1)), min(n-1-i, n-1) + 1)
            valid_k = k_vals[k_vals != 0]
            for k in valid_k:
                eff_i[k + n - 1] += (1 - h[i+k]) * dx
            # -h[i-k] for valid k where i-k in [0,n-1], k != 0
            k_vals2 = np.arange(max(i-(n-1), -(n-1)), min(i, n-1) + 1)
            valid_k2 = k_vals2[k_vals2 != 0]
            for k in valid_k2:
                eff_i[k + n - 1] -= h[i-k] * dx
            
            # For each j in ones, check if swap (i, j) improves
            # effect of removing 1 at j: -eff_j where eff_j is the "add" effect
            # Combined: eff_i - eff_j (where eff_j is the add effect at j)
            # New corr: corr + eff_i - eff_j
            # New max: (corr + eff_i - eff_j).max()
            
            o_list = sorted(ones)
            rng.shuffle(o_list)
            
            for j in o_list:
                if time.time() > deadline:
                    break
                
                # Compute eff_j (add effect at j)
                eff_j = np.zeros(2*n - 1)
                k_vals_j = np.arange(max(-j, -(n-1)), min(n-1-j, n-1) + 1)
                valid_kj = k_vals_j[k_vals_j != 0]
                for k in valid_kj:
                    eff_j[k + n - 1] += (1 - h[j+k]) * dx
                k_vals_j2 = np.arange(max(j-(n-1), -(n-1)), min(j, n-1) + 1)
                valid_kj2 = k_vals_j2[k_vals_j2 != 0]
                for k in valid_kj2:
                    eff_j[k + n - 1] -= h[j-k] * dx
                
                # Combined effect
                combined = corr + eff_i - eff_j
                new_max = combined.max() * dx
                
                if new_max < cur_max - 1e-15:
                    # Accept
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = combined
                    cur_max = new_max
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break  # restart with new i (or continue)
            
            if improved:
                break
    
        if not improved:
            break
    
    return h, cur_max
```
