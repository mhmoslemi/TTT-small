Now let me think about the local search more carefully. The key insight is that for the worst lag δ, we want to reduce the overlap Σ h[i]*(1-h[i+δ])*dx. The most direct way is to move mass from a position where h is high to a position δ away where h is low (or vice versa).

A systematic local search:
1. Compute all lags, find the top K worst lags
2. For each worst lag δ:
   a. For each pair (i, i+δ) where h[i] > h[i+δ]:
      - Try moving a small ε from h[i] to h[i+δ]
      - If this reduces the overall max, accept
   b. For each pair (i, i+δ) where h[i] < h[i+δ]:
      - Try moving a small ε from h[i+δ] to h[i]
      - If this reduces the overall max, accept

The issue is that step 2a/2b is O(n) for each lag, and we have K lags, so it's O(K*n) per round. With K=5 and n=1000, that's 5000 evaluations per round. Each evaluation is O(n²) = O(10⁶) for computing all lags. So one round is 5×10⁹ operations, which is too slow.

- For δ' = δ: -ε*(1-h[i+δ]) + ε*(1-h[i]) ... wait, this is getting complicated because changing h[i] affects all lags, not just δ.

Let me think differently. The change in overlap at lag k when we change h[j] by δh_j is:
Δoverlap(k) = δh_j * (contribution of h[j] to overlap(k))

The contribution of h[j] to overlap(k) (where lag = k - (n-1)):
- If lag = 0: (1 - 2*h[j])
- If lag > 0 and j < n-lag: (1 - h[j+lag])
- If lag > 0 and j ≥ lag: -h[j-lag] (plus the first term if j < n-lag)
- If lag < 0: similar with |lag|

So when we change h[i] by -ε and h[j] by +ε (where j = i+δ), the change in overlap(k) is:
Δoverlap(k) = -ε * contrib(k, i) + ε * contrib(k, j)

For the max to decrease, we need the overlap at the critical lag(s) to decrease. If there's a single critical lag δ*, then:
Δoverlap(δ*) = -ε * contrib(δ*, i) + ε * contrib(δ*, j)

For this to be negative, we need contrib(δ*, i) > contrib(δ*, j).

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def interp_res(h_src, n_src, n_dst):
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx

def lbfgs_optimize(h, n, alpha, t_limit, maxiter=2000):
    target_sum = n / 2.0
    pen = 1e6

    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        f = smooth_max_val(c, alpha)
        d = x.sum() - target_sum
        f += pen * d * d
        g = smooth_max_grad(x, n, alpha)
        g = g + 2.0 * pen * d
        return f, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds,
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-20,
                                         'gtol': 1e-14, 'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def local_search(h, n, deadline, rng, K=5, max_pairs=40):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stuck = 0

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        improved = False

        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = abs(k_idx - (n - 1))
            if lag == 0 or lag >= n:
                continue
            left = h[0:n - lag]
            right = h[lag:n]
            diffs = left - right

            pos_order = np.argsort(-diffs)[:max_pairs]
            for pos in pos_order:
                if time.time() > deadline:
                    break
                if diffs[pos] < 1e-8:
                    continue
                i = pos
                j = pos + lag
                amt = min(0.008, diffs[pos] * 0.3)
                if amt < 1e-8:
                    continue
                h[i] -= amt
                h[j] += amt
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += amt
                    h[j] -= amt
                    h = project_feasible(h, n)

            neg_order = np.argsort(diffs)[:max_pairs]
            for pos in neg_order:
                if time.time() > deadline:
                    break
                if -diffs[pos] < 1e-8:
                    continue
                i = pos
                j = pos + lag
                amt = min(0.008, (-diffs[pos]) * 0.3)
                if amt < 1e-8:
                    continue
                h[i] += amt
                h[j] -= amt
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= amt
                    h[j] += amt
                    h = project_feasible(h, n)

        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 100:
                h = best_h + rng.normal(0, 0.004, n)
                h = project_feasible(h, n)
                stuck = 0

    return best_h, best_c

def fine_local_search(h, n, deadline, rng, K=8, max_pairs=80):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stuck = 0

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        improved = False

        for k_idx in top_idx:
            if time.time() > deadline:
                break
            lag = abs(k_idx - (n - 1))
            if lag == 0 or lag >= n:
                continue
            left = h[0:n - lag]
            right = h[lag:n]
            diffs = left - right

            for direction in [1, -1]:
                if direction == 1:
                    order = np.argsort(-diffs)[:max_pairs]
                else:
                    order = np.argsort(diffs)[:max_pairs]

                for pos in order:
                    if time.time() > deadline:
                        break
                    if direction == 1:
                        if diffs[pos] < 1e-9:
                            continue
                        amt = min(0.003, diffs[pos] * 0.2)
                    else:
                        if -diffs[pos] < 1e-9:
                            continue
                        amt = min(0.003, (-diffs[pos]) * 0.2)
                    if amt < 1e-9:
                        continue
                    i = pos
                    j = pos + lag
                    if direction == 1:
                        h[i] -= amt
                        h[j] += amt
                    else:
                        h[i] += amt
                        h[j] -= amt
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        if direction == 1:
                            h[i] += amt
                            h[j] -= amt
                        else:
                            h[i] -= amt
                            h[j] += amt
                        h = project_feasible(h, n)

        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                h = best_h + rng.normal(0, 0.002, n)
                h = project_feasible(h, n)
                stuck = 0

    return best_h, best_c

def make_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    N = 1000
    candidates = []

    h1000 = interp_res(best_h, len(best_h), N)
    candidates.append(h1000)

    h_bin = make_binary(h1000, N)
    candidates.append(h_bin)

    h_smooth = h1000.copy()
    for _ in range(3):
        h_smooth = np.convolve(h_smooth, np.ones(7) / 7.0, mode='same')
    h_smooth = project_feasible(h_smooth, N)
    candidates.append(h_smooth)

    h_pert = h1000 + rng.normal(0, 0.02, N)
    h_pert = project_feasible(h_pert, N)
    candidates.append(h_pert)

    for ci, h_start in enumerate(candidates):
        if time.time() > deadline - 150:
            break
        h_work = h_start.copy()

        for alpha, maxit in [(500, 800), (3000, 1000), (15000, 1200),
                             (75000, 1500), (300000, 1500), (1500000, 2000)]:
            if time.time() > deadline - 120:
                break
            t_lim = min(time.time() + 40, deadline - 100)
            h_work = lbfgs_optimize(h_work, N, alpha, t_lim, maxiter=maxit)

        if time.time() < deadline - 80:
            ls_deadline = min(time.time() + 60, deadline - 50)
            h_work, c_ls = local_search(h_work, N, ls_deadline, rng)

        c_val = compute_c5(h_work, N)
        if c_val < best_c:
            best_c = c_val
            best_h = h_work.copy()

    if time.time() < deadline - 100:
        h_work = best_h.copy()
        h_work = lbfgs_optimize(h_work, N, 5000000, time.time() + 20, maxiter=2000)
        c_val = compute_c5(h_work, N)
        if c_val < best_c:
            best_c = c_val
            best_h = h_work.copy()

    if time.time() < deadline - 80:
        h_work = best_h.copy()
        f_ls_deadline = min(time.time() + 60, deadline - 30)
        h_work, c_ls = fine_local_search(h_work, N, f_ls_deadline, rng)
        c_val = compute_c5(h_work, N)
        if c_val < best_c:
            best_c = c_val
            best_h = h_work.copy()

    if time.time() < deadline - 40:
        h_work = best_h.copy()
        for alpha in [10000000, 50000000]:
            if time.time() > deadline - 15:
                break
            h_work = lbfgs_optimize(h_work, N, alpha, time.time() + 15, maxiter=2000)
        c_val = compute_c5(h_work, N)
        if c_val < best_c:
            best_c = c_val
            best_h = h_work.copy()

    if time.time() < deadline - 20:
        h_work = best_h.copy()
        f2_deadline = min(time.time() + 12, deadline - 5)
        h_work, c_ls = fine_local_search(h_work, N, f2_deadline, rng)
        c_val = compute_c5(h_work, N)
        if c_val < best_c:
            best_c = c_val
            best_h = h_work.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
