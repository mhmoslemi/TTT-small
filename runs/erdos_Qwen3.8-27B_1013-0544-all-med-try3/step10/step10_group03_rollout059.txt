        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += dx * (1.0 - h[k:n])
                grad[k:n] += dx * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += dx * (1.0 - h[0:nv])
                grad[0:nv] += dx * (-h[kk:n])
        grad /= len(active)
        velocity = beta * velocity + step * grad
        h_new = h - velocity
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            step = min(step * 1.02, 1.5 / n)
        else:
            no_improve += 1
            if no_improve > 300:
                step *= 0.7
                beta = max(beta * 0.995, 0.5)
                no_improve = 0
                if step < 1e-12 / n:
                    break
    return best_h, best_c

def perturb_and_optimize(h, n, deadline, n_trials=10, seed_val=42):
    rng = np.random.default_rng(seed_val)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for trial in range(n_trials):
        if time.time() > deadline - 40:
            break
        scale = 0.03 + 0.06 * np.exp(0.3 * trial)
        noise = rng.normal(0, scale, n)
        h_pert = project_feasible(h + noise, n)
        t_end = min(time.time() + 35, deadline - 10)
        h_opt, c_opt = lbfgs_pass(h_pert, n, t_end, alphas=[500, 5000, 50000], mu=50.0)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def freq_perturb(h, n, deadline, n_trials=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = np.linspace(0, 2, n)
    for trial in range(n_trials):
        if time.time() > deadline - 30:
            break
        freq = (trial + 1) * 2.5
        amp = 0.04 + 0.02 * trial
        phase = trial * 1.7
        noise = amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        h_pert = project_feasible(h + noise, n)
        t_end = min(time.time() + 25, deadline - 5)
        h_opt, c_opt = lbfgs_pass(h_pert, n, t_end, alphas=[2000, 20000], mu=20.0)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def fine_pair(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.001, 0.0005, 0.0002]
    for d in deltas:
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def slqp_pass(h, n, deadline, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_sum, 'jac': c_sum_jac, 'args': (n,)}]
    try:
        if time.time() > deadline - 5:
            return h.copy(), compute_c5(h, n)
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_push(h, n, deadline):
    """Push towards binary, then re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in [0.3, 0.4, 0.45, 0.55, 0.6]:
        if time.time() > deadline - 40:
            break
        h_bin = (h > t).astype(float)
        h_bin = project_feasible(h_bin, n)
        h_opt, c_opt = lbfgs_pass(h_bin, n, time.time() + 30, alphas=[1000, 10000], mu=10.0)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500

    # Work at n=500 (parent resolution) and n=1000
    n = 500
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Quick L-BFGS at n=500 (60s)
    t1 = min(t0 + 60, deadline - 700)
    h, c = lbfgs_pass(h, n, t1, alphas=[50, 200, 1000, 5000, 20000], mu=10.0)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Top-k focused at n=500 (80s)
    if time.time() < deadline - 600:
        t2 = min(time.time() + 80, deadline - 600)
        h2, c2 = topk_optimize(h, n, t2, max_rounds=10, top_k=10)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2

    # Phase 3: Subgradient momentum at n=500 (80s)
    if time.time() < deadline - 500:
        t3 = min(time.time() + 80, deadline - 500)
        h3, c3 = subgrad_momentum(h, n, t3, n_iter=6000, top_k=15)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
            h = h3

    # Phase 4: Binary push (80s)
    if time.time() < deadline - 400:
        t4 = min(time.time() + 80, deadline - 400)
        h4, c4 = binary_push(best_h, n, t4)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
            h = h4

    # Phase 5: Random perturbation search (120s)
    if time.time() < deadline - 250:
        t5 = min(time.time() + 120, deadline - 250)
        h5, c5 = perturb_and_optimize(best_h, n, t5, n_trials=8, seed_val=seed)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
            h = h5

    # Phase 6: Frequency perturbation (60s)
    if time.time() < deadline - 180:
        t6 = min(time.time() + 60, deadline - 180)
        h6, c6 = freq_perturb(best_h, n, t6, n_trials=4)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
            h = h6

    # Phase 7: Refocus top-k on updated solution (100s)
    if time.time() < deadline - 120:
        t7 = min(time.time() + 100, deadline - 120)
        h7, c7 = topk_optimize(best_h, n, t7, max_rounds=12, top_k=6)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
            h = h7

    # Phase 8: Subgradient polish (60s)
    if time.time() < deadline - 60:
        t8 = min(time.time() + 60, deadline - 60)
        h8, c8 = subgrad_momentum(best_h, n, t8, n_iter=10000, top_k=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Phase 9: Fine pair search (40s)
    if time.time() < deadline - 25:
        t9 = min(time.time() + 40, deadline - 20)
        h9, c9 = fine_pair(best_h, n, t9, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()

    # Phase 10: Final SLSQP (20s)
    if time.time() < deadline - 10:
        h10, c10 = slqp_pass(best_h, n, deadline - 5, alpha=500000, maxiter=1500)
        if c10 < best_c:
            best_c = c10
            best_h = h10.copy()

    # Also try n=1000
    n2 = 1000
    x_fine = np.linspace(0, 2, n2)
    x_500 = np.linspace(0, 2, n)
    h_fine = np.interp(x_fine, x_500, best_h)
    h_fine = project_feasible(h_fine, n2)
    c_fine = compute_c5(h_fine, n2)
    # Only use if remaining time allows a quick pass
    if c_fine < best_c and time.time() < deadline - 80:
        t_fine = min(time.time() + 70, deadline - 15)
        h_f, c_f = lbfgs_pass(h_fine, n2, t_fine, alphas=[1000, 10000], mu=10.0)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
            n = n2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
