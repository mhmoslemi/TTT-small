def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, n, k):
    """Compute gradient of f_k with respect to h, where k is the shift index in [0, 2n-2]"""
    grad = np.zeros(n)
    dx = 2.0 / n

        #   (1-h[j-shift]) * dx (from the first factor)
        #   minus contribution from second factor: -h[j] * dx for i where i=j, i.e., j < nv

        # For j in [shift, n-1]: h[j] appears as h[i+shift] with i = j-shift
        #   contribution: dx * (1-h[j-shift]) [only if j-shift < nv, i.e., j < shift+nv = n, always true]
        # For j in [0, nv-1]: h[j] appears as h[i] with i=j
        #   contribution: -dx * h[j+shift]
        # So:

def push_down_step(h, n, c_vec, top_k=10, step_size=0.01):

    m = c_vec.max()
    threshold = m - max(1e-10, 0.005 * m)

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def overlap_grad_single(h, n, k_idx):
    grad = np.zeros(n)
    dx = 2.0 / n
    shift = k_idx - (n - 1)
    if shift >= 0:

        grad[shift:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[shift:n]) * dx
    else:

        grad[0:nv] += (1.0 - h[kk:n]) * dx
        grad[kk:n] += (-h[0:nv]) * dx
    return grad

            else:
                # No improvement, try a small random perturbation
                if it % 500 == 499:
                    pert = 0.005 * np.random.randn(n)
                    pert -= pert.mean()
                    h_try = project_feasible(best_h + pert, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h_try
    
    return best_h, best_c

def upsample_h(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_val, frac=0.5):
    h = np.zeros(n)
    x = np.arange(n, dtype=float)
    fp = (x * alpha_val) % 1.0
    h[fp < frac] = 1.0
    return project_feasible(h, n)

def make_sinusoidal(n, amplitude, freq, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq * x + phase)
    return project_feasible(h, n)

def multi_start_optimize(h_start, n, deadline, alphas, maxit_per_alpha=500):
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        rem = max(50, int(deadline - time.time()))
        maxit = min(maxit_per_alpha, rem)
        h_new = lbfgs_optimize(best_h, n, alpha, 0.01, maxit, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def local_pair_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_trials = min(500, max(100, int((deadline - time.time()) * 5)))
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i, j = rng.choice(n, 2, replace=False)
            delta = 0.005 * 10.0 ** rng.uniform(-1.0, 1.0)
            if rng.rand() < 0.5:
                di = delta
                dj = -delta
            else:
                di = -delta
                dj = delta
            hi = best_h[i] + di
            hj = best_h[j] + dj
            if hi < -1e-10 or hi > 1 + 1e-10 or hj < -1e-10 or hj > 1 + 1e-10:
                continue
            best_h[i] = np.clip(hi, 0, 1)
            best_h[j] = np.clip(hj, 0, 1)
            best_h = project_feasible(best_h, n)
            c = compute_c5(best_h, n)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
            else:
                best_h[i] = np.clip(best_h[i] - di, 0, 1)
                best_h[j] = np.clip(best_h[j] - dj, 0, 1)
                best_h = project_feasible(best_h, n)
        if not improved and rnd > 20:
            # Try larger perturbation
            pert = 0.01 * np.random.randn(n)
            pert -= pert.mean()
            h_try = project_feasible(best_h + pert, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = make_beatty(500, (1 + np.sqrt(5)) / 2.0 - 1.0)
        n_init = 500
    
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    
    # Phase 1: Quick L-BFGS at initial resolution
    p1_end = t0 + 80
    h1, c1 = multi_start_optimize(h_init, n_init, p1_end,
                                   alphas=[50, 500, 5000, 50000, 200000],
                                   maxit_per_alpha=300)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Targeted gradient descent at initial resolution
    p2_end = t0 + 250
    h2, c2 = targeted_gradient_descent(best_h, n_init, p2_end, max_iter=30000)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    
    # Phase 3: Try multiple structured starts at n_init
    p3_end = t0 + 350
    golden = (np.sqrt(5) - 1) / 2.0
    starts = [
        make_beatty(n_init, golden),
        make_beatty(n_init, np.sqrt(2) - 1.0),
        make_beatty(n_init, np.sqrt(3) - 1.0),
        make_sinusoidal(n_init, 0.3, 1.5),
        make_sinusoidal(n_init, 0.3, 2.5),
        make_sinusoidal(n_init, 0.4, 1.0),
    ]
    for s in starts:
        if time.time() > p3_end - 30:
            break
        cs = compute_c5(s, n_init)
        if cs < best_c + 0.01:
            h_s, c_s = lbfgs_optimize(s, n_init, 10000, 0.01, 300, p3_end)
            c_s = compute_c5(h_s, n_init)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s
    
    # Phase 4: Upsample to higher resolution and optimize
    n_hi = n_init * 2
    p4_end = min(t0 + 650, deadline - 350)
    if time.time() < p4_end - 100:
        h_hi = upsample_h(best_h, n_init, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        # Also try structured starts at higher res
        hi_starts = [
            make_beatty(n_hi, golden),
            make_sinusoidal(n_hi, 0.3, 1.5),
        ]
        for s in hi_starts:
            cs = compute_c5(s, n_hi)
            if cs < c_hi and cs < best_c + 0.005:
                h_hi = s
                c_hi = cs
        
        h_hi, c_hi = multi_start_optimize(h_hi, n_hi, min(p4_end, deadline - 200),
                                           alphas=[5000, 50000, 200000, 1000000],
                                           maxit_per_alpha=400)
        if time.time() < deadline - 150:
            h_hi, c_hi = targeted_gradient_descent(h_hi, n_hi, min(deadline - 150, p4_end + 50), max_iter=15000)
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
    
    # Phase 5: Refine best at its resolution
    n_best = len(best_h)
    p5_end = min(t0 + 850, deadline - 200)
    if time.time() < p5_end - 20:
        h5, c5 = lbfgs_optimize(best_h, n_best, 500000, 0.005, 500, p5_end)
        c5 = compute_c5(h5, n_best)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < deadline - 100:
            h5b, c5b = targeted_gradient_descent(best_h, n_best, min(deadline - 100, p5_end + 50), max_iter=20000)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
    
    # Phase 6: Local pair search
    p6_end = deadline - 10
    n_best = len(best_h)
    if time.time() < p6_end - 5:
        h6, c6 = local_pair_search(best_h, n_best, p6_end, max_rounds=100)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Final L-BFGS polish
    if time.time() < deadline - 3:
        n_best = len(best_h)
        h7 = lbfgs_optimize(best_h, n_best, 2000000, 0.001, 200, deadline)
        c7 = compute_c5(h7, n_best)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
