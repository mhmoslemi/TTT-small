from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

    vals = c[active]
    v = lse_val(vals, alpha)
    if mu > 0:

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0

        w /= s
    return w

            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n

    # Phase 1: Quick L-BFGS warmup with full smooth max
    p1_end = min(t0 + 60, deadline - 900)
    for alpha in [100, 1000, 10000, 100000]:
        if time.time() > p1_end:
            break
        h1, c1 = lbfgs_round(best_h, n, p1_end, alpha=alpha, top_k=None, mu=0.05, maxit=1500)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Intensive active-set L-BFGS (the main workhorse)
    p2_end = min(t0 + 350, deadline - 400)
    configs2 = [
        (15, 50000, 2000, 0.005),
        (25, 200000, 2500, 0.005),
        (40, 1000000, 2500, 0.003),
        (60, 5000000, 2000, 0.003),
        (40, 20000000, 2000, 0.002),
        (60, 100000000, 1500, 0.002),
        (80, 500000000, 1200, 0.001),
    ]
    for top_k, alpha, maxit, mu in configs2:
        if time.time() > p2_end:
            break
        h2, c2 = lbfgs_round(best_h, n, p2_end, alpha=alpha, top_k=top_k, mu=mu, maxit=maxit)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient minimax for robustness
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 180, deadline - 300)
        h3, c3 = subgradient_phase(best_h, n, p3_end, max_iter=400000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Second round of active-set L-BFGS with very high alpha
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 150, deadline - 220)
        configs4 = [
            (30, 5000000, 2000, 0.001),
            (50, 50000000, 2000, 0.001),
            (30, 500000000, 1500, 0.0005),
            (100, 1000000000, 1000, 0.0005),
        ]
        for top_k, alpha, maxit, mu in configs4:
            if time.time() > p4_end:
                break
            h4, c4 = lbfgs_round(best_h, n, p4_end, alpha=alpha, top_k=top_k, mu=mu, maxit=maxit)
            if c4 < best_c:
                best_c = c4
                best_h = h4

    # Phase 5: Binary swap search for structural changes
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 100, deadline - 180)
        h5, c5 = binary_swap_search(best_h, n, p5_end, max_rounds=2000, radius=5)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Perturbation + re-optimization
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 80, deadline - 130)
        h6, c6 = perturb_and_optimize(best_h, n, p6_end, seed=seed + 100, num_perturbs=4)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Higher resolution attempt
    if time.time() < deadline - 120:
        p7_end = min(time.time() + 120, deadline - 100)
        n_high = min(2 * n, 1000)
        h_high = upsample_h(best_h, n, n_high)
        c_high_init = compute_c5(h_high, n_high)
        for alpha in [10000, 100000, 1000000, 10000000]:
            if time.time() > p7_end:
                break
            hh, ch = lbfgs_round(h_high, n_high, p7_end, alpha=alpha, top_k=30, mu=0.01, maxit=800)
            if ch < c_high_init:
                h_high = hh
                c_high_init = ch
        if c_high_init < best_c:
            best_c = c_high_init
            best_h = h_high
            best_n = n_high

    # Phase 8: Fine coordinate tuning
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 120, deadline - 15)
        h8, c8 = fine_tune(best_h, best_n, p8_end,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            radius=10)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final L-BFGS polish
    if time.time() < deadline - 5:
        h9, c9 = lbfgs_round(best_h, best_n, deadline - 2, alpha=1000000000, top_k=50, mu=0.0001, maxit=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final very fine adjustment
    if time.time() < deadline - 1:
        h10, c10 = fine_tune(best_h, best_n, deadline - 0.5,
            deltas=[0.00001, 0.000005, 0.000002], radius=5)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
