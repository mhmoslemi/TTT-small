    h_new = h - step_size * grad

        grad = get_gradient(best_h, n, np.array([worst_idx]), np.array([1.0]))
        
        # Find best (i, j) pair: i has most positive grad, j has most negative grad
        # But constraint: h[i] >= d, h[j] <= 1-d
        d_candidates = [0.05, 0.02, 0.01, 0.005, 0.002]
        
        improved = False
        for d in d_candidates:
            # i: maximize grad[i] subject to h[i] >= d
            valid_i = np.where(grad > 0.01 * d and best_h >= d)[0]
            valid_j = np.where(grad < -0.01 * d and best_h <= 1 - d)[0]
            
            if len(valid_i) == 0 or len(valid_j) == 0:
                continue
            
            # Try top few from each
            i_candidates = valid_i[np.argsort(-grad[valid_i])[:30]]
            j_candidates = valid_j[np.argsort(grad[valid_j])[:30]]
            
            best_pair = None
            best_new_max = cur_max
            
            for i in i_candidates[:10]:
                if best_h[i] < d:
                    continue
                for j in j_candidates[:10]:
                    if i == j or best_h[j] > 1 - d:
                        continue
                    dc = fast_delta_c(best_h, n, i, j) * d
                    new_max = (c + dc).max()
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_pair = (i, j, d)
            
            if best_pair is not None:
                i, j, dd = best_pair
                best_h[i] -= dd
                best_h[j] += dd
                best_h = np.clip(best_h, 0, 1)
                improved = True
                break
        
        if not improved:
            # Try random pair near the worst shift
            if k >= 0:
                offset = k
                for trial in range(20):
                    ii = rng_global[rng_global.size]
                    jj = min(n - 1, ii + offset) if ii + offset < n else rng_global[rng_global.size]
                    if ii == jj:
                        continue
                    d = min(0.01, best_h[ii], 1 - best_h[jj])
                    if d < 0.001:
                        continue
                    dc = fast_delta_c(best_h, n, ii, jj) * d
                    if (c + dc).max() < best_c - 1e-14:
                        best_h[ii] -= d
                        best_h[jj] += d
                        best_h = np.clip(best_h, 0, 1)
                        best_c = (c + dc).max()
                        improved = True
                        break
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global, rng_global2
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    rng_global = rng.permutation(100000)
    rng_global2 = rng.permutation(100000)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    n = n_orig
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n
    
    # Phase 1: Quick L-BFGS warmup with full smooth max
    p1_end = min(t0 + 80, deadline - 800)
    if p1_end > t0 + 15:
        bounds = [(0.0, 1.0)] * n
        for alpha in [5000, 50000, 500000]:
            if time.time() > p1_end - 5:
                break
            rem = p1_end - time.time() - 3
            if rem < 8:
                break
            maxit = max(200, int(rem * 4))
            maxit = min(maxit, 2000)
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                              args=(n, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: Intensive active-set L-BFGS-B
    p2_end = min(time.time() + 250, deadline - 500)
    if p2_end > time.time() + 30:
        bounds = [(0.0, 1.0)] * n
        configs = [
            (3, 20000, 0.002),
            (5, 50000, 0.001),
            (5, 200000, 0.001),
            (7, 500000, 0.0005),
            (10, 2000000, 0.0005),
            (10, 5000000, 0.0003),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 10:
                break
            for _ in range(3):
                if time.time() > p2_end - 8:
                    break
                c_vec = compute_c5_vec(best_h, n)
                max_val = c_vec.max()
                threshold = max_val - max(1e-6, 0.001 * max_val)
                active = np.where(c_vec >= threshold)[0]
                if len(active) > top_k:
                    active = np.argsort(-c_vec[active])[:top_k]
                if len(active) == 0:
                    continue
                rem = p2_end - time.time() - 5
                if rem < 5:
                    break
                maxit = max(100, int(rem * 3))
                maxit = min(maxit, 2000)
                try:
                    res = minimize(active_smooth_obj, best_h, jac=active_smooth_grad,
                                  args=(n, active, alpha, mu), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-24,
                                           'gtol': 1e-18, 'maxfun': maxit * 6,
                                           'maxls': 30, 'maxcor': 30})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                except Exception:
                    pass
    
    # Phase 3: Targeted swap search (key innovation)
    p3_end = min(time.time() + 200, deadline - 300)
    if p3_end > time.time() + 20:
        h3, c3 = targeted_swap(best_h, n, p3_end, n_rounds=30000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Another gradient round after swaps
    p4_end = min(time.time() + 100, deadline - 200)
    if p4_end > time.time() + 10:
        bounds = [(0.0, 1.0)] * n
        for alpha in [100000, 1000000, 5000000]:
            if time.time() > p4_end - 5:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-6, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > 5:
                active = np.argsort(-c_vec[active])[:5]
            if len(active) == 0:
                continue
            rem = p4_end - time.time() - 3
            if rem < 5:
                break
            maxit = max(100, int(rem * 3))
            maxit = min(maxit, 1500)
            try:
                res = minimize(active_smooth_obj, best_h, jac=active_smooth_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-24,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 5: Upsample to n*2 and optimize
    n_up = n * 2
    p5_end = min(time.time() + 250, deadline - 100)
    if p5_end > time.time() + 60 and n_up <= 2000:
        h_up = upsample_h(best_h, n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.005:
            bounds_up = [(0.0, 1.0)] * n_up
            for alpha in [5000, 50000, 500000, 5000000]:
                if time.time() > p5_end - 15:
                    break
                rem = p5_end - time.time() - 5
                if rem < 8:
                    break
                c_vec = compute_c5_vec(h_up, n_up)
                max_val = c_vec.max()
                threshold = max_val - max(1e-6, 0.001 * max_val)
                active = np.where(c_vec >= threshold)[0]
                if len(active) > 8:
                    active = np.argsort(-c_vec[active])[:8]
                if len(active) == 0:
                    continue
                maxit = max(100, int(rem * 2))
                maxit = min(maxit, 1500)
                try:
                    res = minimize(active_smooth_obj, h_up, jac=active_smooth_grad,
                                  args=(n_up, active, alpha, 0.003), method='L-BFGS-B',
                                  bounds=bounds_up,
                                  options={'maxiter': maxit, 'ftol': 1e-24,
                                           'gtol': 1e-18, 'maxfun': maxit * 5,
                                           'maxls': 25, 'maxcor': 25})
                    h_new = project_feasible(res.x, n_up)
                    c_new = compute_c5(h_new, n_up)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                        best_n = n_up
                        h_up = h_new
                except Exception:
                    pass
    
    # Phase 6: Final targeted swaps at best resolution
    p6_end = min(time.time() + 120, deadline - 15)
    if p6_end > time.time() + 10:
        cur_n = len(best_h)
        h6, c6 = targeted_swap(best_h, cur_n, p6_end, n_rounds=50000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = cur_n
    
    # Phase 7: Final fine gradient
    p7_end = deadline - 3
    if p7_end > time.time() + 8:
        cur_n = len(best_h)
        bounds_f = [(0.0, 1.0)] * cur_n
        c_vec = compute_c5_vec(best_h, cur_n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 5:
            active = np.argsort(-c_vec[active])[:5]
        if len(active) > 0:
            rem = p7_end - time.time() - 2
            if rem > 3:
                maxit = max(50, int(rem * 4))
                maxit = min(maxit, 1000)
                try:
                    res = minimize(active_smooth_obj, best_h, jac=active_smooth_grad,
                                  args=(cur_n, active, 10000000, 0.001), method='L-BFGS-B',
                                  bounds=bounds_f,
                                  options={'maxiter': maxit, 'ftol': 1e-24,
                                           'gtol': 1e-18, 'maxfun': maxit * 4,
                                           'maxls': 20, 'maxcor': 20})
                    h_new = project_feasible(res.x, cur_n)
                    c_new = compute_c5(h_new, cur_n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                        best_n = cur_n
                except Exception:
                    pass
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
