                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)


def binary_sa(h_bin, n, deadline, n_iter=200000):
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    cur_h = h_bin.copy()
    cur_c = best_c
    ones = np.where(cur_h > 0.5)[0]
    zeros = np.where(cur_h < 0.5)[0]
    n_ones = n // 2
    t_start = time.time()
    elapsed_per_iter = 1.0 / max(1, n_iter)
    total_time = max(1, deadline - time.time())
    t0 = total_time
    t_end = 0.05
    total_iters = max(1000, n_iter)
    for i in range(total_iters):
        if time.time() > deadline:
            break
        T = t_end + (t0 - t_end) * (t_end / t0) ** (i / total_iters)
        i_pos = ones[np.random.randint(len(ones))]
        j_pos = zeros[np.random.randint(len(zeros))]
        if i_pos == j_pos:
            continue
        cur_h[i_pos], cur_h[j_pos] = cur_h[j_pos], cur_h[i_pos]
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i_pos], cur_h[j_pos] = cur_h[j_pos], cur_h[i_pos]
    return best_h, best_c


def binary_iterated_local(h_bin, n, deadline, n_rounds=20):
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    n_ones = n // 2
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        cur_h = best_h.copy()
        cur_c = best_c
        improved = True
        stalling = 0
        while improved and time.time() < deadline - 5:
            improved = False
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            n_swaps = min(500, len(ones) * len(zeros))
            best_delta = 0
            best_i = -1
            best_j = -1
            for _ in range(n_swaps):
                i_pos = ones[np.random.randint(len(ones))]
                j_pos = zeros[np.random.randint(len(zeros))]
                if i_pos == j_pos:
                    continue
                cur_h[i_pos], cur_h[j_pos] = cur_h[j_pos], cur_h[i_pos]
                c = compute_c5(cur_h, n)
                cur_h[i_pos], cur_h[j_pos] = cur_h[j_pos], cur_h[i_pos]
                if c < cur_c - best_delta:
                    best_delta = cur_c - c
                    best_i = i_pos
                    best_j = j_pos
            if best_i >= 0:
                cur_h[best_i], cur_h[best_j] = cur_h[best_j], cur_h[best_i]
                cur_c -= best_delta
                improved = True
                stalling = 0
            else:
                stalling += 1
                if stalling > 3:
                    break
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
        else:
            rng = np.random.RandomState(r)
            h_pert = best_h.copy()
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
            for _ in range(5):
                i_pos = ones[rng.randint(len(ones))]
                j_pos = zeros[rng.randint(len(zeros))]
                if i_pos != j_pos:
                    h_pert[i_pos], h_pert[j_pos] = h_pert[j_pos], h_pert[i_pos]
            # Run SA from perturbed state with limited time
            sub_deadline = min(time.time() + 30, deadline - 10)
            h_sa, c_sa = binary_sa(h_pert, n, sub_deadline, n_iter=50000)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
        if time.time() > deadline - 30:
            break
    return best_h, best_c


def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(100):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        found = False
        if k_worst >= 0:
            affected = np.random.choice(n - k_worst, size=min(50, n - k_worst), replace=False)
            affected2 = affected + k_worst
        else:
            kk = -k_worst
            affected = np.random.choice(n - kk, size=min(50, n - kk), replace=False) + kk
            affected2 = affected - kk
        all_affected = np.unique(np.concatenate([affected, affected2]))
        all_affected = all_affected[(all_affected >= 0) & (all_affected < n)]
        for i in all_affected:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = best_h[i]
            if found:
                h = best_h.copy()
        if not found and round_num > 20:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: Quick L-BFGS at base resolution
    t1 = min(t0 + 60, deadline - 900)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(100, 50.0), (500, 500.0), (2000, 5000.0)])

    # Phase 2: SLSQP polish at base
    t2 = min(t0 + 120, deadline - 800)
    if t2 > time.time() + 5:
        h2, c2 = slqp_polish(best_h, n_base, t2, alpha=500000, maxiter=3000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Binary local search at base
    t3 = min(t0 + 250, deadline - 600)
    if t3 > time.time() + 30:
        h_bin = to_binary_optimal(best_h, n_base)
        h_bin, c_bin = binary_iterated_local(h_bin, n_base, t3, n_rounds=5)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin

    # Phase 4: Upsample to n=1000 and optimize
    n1 = 1000
    if time.time() < deadline - 300:
        h1000 = interpolate_to(best_h, len(best_h), n1)
        t4 = min(time.time() + 120, deadline - 180)
        if t4 > time.time() + 10:
            h1000, c1000 = run_lbfgs(h1000, n1, t4,
                configs=[(100, 50.0), (500, 500.0), (2000, 5000.0), (5000, 20000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000

        # SLSQP at n=1000
        t4b = min(time.time() + 60, deadline - 120)
        if t4b > time.time() + 10:
            h_sb, c_sb = slqp_polish(best_h, n1, t4b, alpha=2000000, maxiter=3000)
            if c_sb < best_c:
                best_c = c_sb
                best_h = h_sb

    # Phase 5: Binary SA at n=1000
    if time.time() < deadline - 200:
        h_bin1k = to_binary_optimal(best_h, len(best_h))
        t5 = min(time.time() + 100, deadline - 100)
        h_bin1k, c_bin1k = binary_sa(h_bin1k, len(h_bin1k), t5, n_iter=300000)
        if c_bin1k < best_c:
            best_c = c_bin1k
            best_h = h_bin1k

        # Iterated local on the SA result
        t5b = min(time.time() + 60, deadline - 60)
        if t5b > time.time() + 20:
            h_bin1k, c_bin1k2 = binary_iterated_local(h_bin1k, len(h_bin1k), t5b, n_rounds=10)
            if c_bin1k2 < best_c:
                best_c = c_bin1k2
                best_h = h_bin1k

    # Phase 6: Continuous refinement of best binary solution
    if time.time() < deadline - 80:
        # Slightly relax the binary solution
        h_relaxed = best_h + 0.05 * (np.random.rand(len(best_h)) - 0.5)
        h_relaxed = project_feasible(h_relaxed, len(best_h))
        t6 = min(time.time() + 50, deadline - 40)
        if t6 > time.time() + 10:
            h6, c6 = run_lbfgs(h_relaxed, len(h_relaxed), t6,
                configs=[(1000, 2000.0), (5000, 20000.0), (20000, 100000.0)])
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Targeted refinement
    if time.time() < deadline - 20:
        best_h, best_c = targeted_refine(best_h, len(best_h), deadline - 5)

    # Phase 8: Final SLSQP if time permits
    if time.time() < deadline - 15:
        h_f, c_f = slqp_polish(best_h, len(best_h), deadline - 5,
                              alpha=10000000, maxiter=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
