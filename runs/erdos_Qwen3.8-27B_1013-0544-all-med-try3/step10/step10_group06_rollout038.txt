import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        log_sum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return float(m + log_sum / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return w / w_sum

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        elif -n < lag < 0:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, maxiter=3000, max_time=None):

        call_count[0] += 1
        if max_time is not None and time.time() - t_start > max_time:
            exceeded[0] = True
        x = np.clip(x, 0.0, 1.0)
        c_val = compute_all_lags(x, n)
        obj_val = smooth_max_value(c_val, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2e6 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, jac=True,

            if time.time() > deadline:
                break
            lag = k_idx - (n - 1)
            if lag == 0 or abs(lag) >= n:
                continue
            m = abs(lag)
            # Try all pairs at this lag with adaptive step
            for step in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-7:
                        d = min(step, (h[i] - h[j]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-14:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            rounds = 0
                            c_all = compute_all_lags(h, n)
                            top_idx = np.argsort(c_all)[-6:]
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > h[i] + 1e-7:
                        d = min(step, (h[j] - h[i]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-14:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                            rounds = 0
                            c_all = compute_all_lags(h, n)
                            top_idx = np.argsort(c_all)[-6:]
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
        
        if not improved:
            break
        rounds += 1
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick L-BFGS-B polish at parent resolution
    n = len(best_h)
    alphas_p1 = [5000, 50000, 500000]
    for alpha in alphas_p1:
        if time.time() > deadline - 500:
            break
        rem = min(30, deadline - 500 - time.time())
        if rem < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, maxiter=1500, max_time=rem)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Upscale to n=1000 and optimize
    n_hires = 1000
    if time.time() < deadline - 350:
        h_up = expand_to_resolution(best_h, len(best_h), n_hires)
        c_up = compute_c5(h_up, n_hires)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
        else:
            # Keep the better one but still work at high res
            pass
        
        h_work = h_up.copy()
        alphas_p2 = [5000, 50000, 200000, 1000000]
        for alpha in alphas_p2:
            if time.time() > deadline - 250:
                break
            rem = min(40, deadline - 250 - time.time())
            if rem < 5:
                break
            h_opt = lbfgsb_optimize(h_work, n_hires, alpha, maxiter=2000, max_time=rem)
            c_opt = compute_c5(h_opt, n_hires)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h_work = h_opt.copy()

    # Phase 3: SA at the current best resolution
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        sa_deadline = min(deadline - 80, time.time() + 250)
        h_opt, c_opt = sa_with_restarts(best_h, n_cur, sa_deadline, rng, n_restarts=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Post-SA L-BFGS-B polish
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        for alpha in [1000000, 10000000]:
            if time.time() > deadline - 30:
                break
            rem = min(25, deadline - 30 - time.time())
            if rem < 3:
                break
            h_opt = lbfgsb_optimize(best_h, n_cur, alpha, maxiter=2000, max_time=rem)
            c_opt = compute_c5(h_opt, n_cur)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Targeted local search
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        ls_deadline = min(deadline - 5, time.time() + 80)
        h_opt, c_opt = targeted_ls(best_h, n_cur, ls_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try n=2000 if time permits
    if time.time() < deadline - 60:
        h_up2 = expand_to_resolution(best_h, len(best_h), 2000)
        h_work2 = h_up2.copy()
        for alpha in [10000, 100000, 500000]:
            if time.time() > deadline - 20:
                break
            rem = min(15, deadline - 20 - time.time())
            if rem < 3:
                break
            h_opt = lbfgsb_optimize(h_work2, 2000, alpha, maxiter=1000, max_time=rem)
            c_opt = compute_c5(h_opt, 2000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h_work2 = h_opt.copy()

    # Phase 7: Final targeted LS
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_opt, c_opt = targeted_ls(best_h, n_cur, deadline - 2, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_cur, 10000000, maxiter=1000,
                                max_time=max(1, deadline - time.time() - 1))
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
