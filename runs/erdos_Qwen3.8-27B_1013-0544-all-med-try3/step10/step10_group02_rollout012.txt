import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smo_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def smo_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def act_grad(x, n, active, alpha, mu):

    mx = c[active].max()

    grad = np.zeros(n)

        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def periodic_init(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < period * duty:
            h[i] = 1.0
    return project_feasible(h, n)

def rand_bin_init(n, seed):

    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0

def do_lbfgs_phase(h, n, deadline, alphas, mu, maxiters, bounds):

def do_slqp(h, n, deadline, alpha, maxit):
    try:

        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        def cfun(x):
            return x.sum() - n / 2.0
        def cjac(x):
            return np.ones(n)
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],

def do_subgrad(h, n, deadline, max_iter=5000):
    h_cur = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(h_cur, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0008 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h_cur[k:]) * dx
                grad[k:] += (-h_cur[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h_cur[:nv]) * dx
                grad[:nv] += (-h_cur[kk:]) * dx
        grad /= len(active)
        h_try = h_cur - eta * grad
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
            eta = min(eta * 1.008, 0.015)
        else:
            eta *= 0.975
        h_cur = h_try
    return best_h, best_c
