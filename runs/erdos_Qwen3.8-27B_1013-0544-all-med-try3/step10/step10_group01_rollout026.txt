import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    c_max = c.max()
    sh = alpha * (c - c_max)
    sh = np.clip(sh, -100, 100)
    w = np.exp(sh)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)

            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new, c_max

def grad_guided_search(h, n, deadline, step_sizes=None, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:

                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:

                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            
            # Find best pair (i, j) to transfer

            best_j = -1
            for i in pos_sorted:

                    if g > best_gain:

                        best_i = i

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def compute_weighted_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(weights)):
        if weights[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = weights[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def phase_subgradient(h, n, deadline, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if alphas is None:

        if time.time() > deadline - 5:
            break
        eta = 0.08
        stagnation = 0
        max_stagnation = 2000
        while time.time() < deadline - 3 and stagnation < max_stagnation:
            c_vec = compute_c5_vec(best_h, n)
            c_max = c_vec.max()
            sh = alpha * (c_vec - c_max)
            sh = np.clip(sh, -80, 80)
            w = np.exp(sh)
            w /= w.sum()
            grad = compute_weighted_grad(best_h, n, w)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            eta_curr = eta * (0.5 + 0.5 * (1.0 / (1.0 + stagnation * 0.01)))
            h_new = best_h - eta_curr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                stagnation = 0
            else:
                eta *= 0.95
                stagnation += 1

    return best_h, best_c

            h_new = project_feasible(res.x, n)

                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:

        except Exception:

    return best_h, best_c

def phase_grad_coord(h, n, deadline, step_sizes=None, max_steps=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    steps_done = 0
    for step in step_sizes:
        stall = 0
        while time.time() < deadline and steps_done < max_steps and stall < 5000:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-6) & (best_h > 1e-6)
            neg_mask = (grad < -1e-6) & (best_h < 1.0 - 1e-6)
            if not pos_mask.any() or not neg_mask.any():
                stall += 1000
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            pos_sorted = pos_idx[np.argsort(-grad[pos_idx])][:50]
            neg_sorted = neg_idx[np.argsort(grad[neg_idx])][:50]
            best_gain = -1
            best_i = -1
            best_j = -1
            for i in pos_sorted:
                gi = grad[i]
                for j in neg_sorted:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain <= 0:
                stall += 1000
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-14:
                stall += 1
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stall = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stall += 1
    return best_h, best_c

def phase_binary_sa(h, n, deadline, rng, n_iter=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones_set = set(np.where(best_h > 0.5)[0].tolist())
    zeros_set = set(np.where(best_h < 0.5)[0].tolist())
    T = 0.01
    T_min = 0.0001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            T = max(T_min, T * 0.98)
        i = rng.choice(list(ones_set)) if ones_set else 0
        j = rng.choice(list(zeros_set)) if zeros_set else 0
        if i == j:
            continue
        best_h[i] = 0.0
        best_h[j] = 1.0
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if c_new < best_c:
                best_c = c_new
            ones_set.remove(i)
            ones_set.add(j)
            zeros_set.remove(j)
            zeros_set.add(i)
        else:
            best_h[i] = 1.0
            best_h[j] = 0.0
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def phase_fine_pairs(h, n, deadline, rng, deltas=None):

    best_c = compute_c5(h, n)
    if deltas is None:

    for d in deltas:
        n_passes = 0
        while time.time() < deadline and n_passes < 5:
            improved = False
            n_trials = 3000
            for _ in range(n_trials):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                if best_h[j] - d >= 0.0 and best_h[i] + d <= 1.0:
                    best_h[j] -= d
                    best_h[i] += d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved = True
                    else:
                        best_h[j] += d
                        best_h[i] -= d
            n_passes += 1
            if not improved:
                break
    return best_h, best_c
