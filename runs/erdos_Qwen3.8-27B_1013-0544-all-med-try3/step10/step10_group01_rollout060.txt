import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_stable(x):
    m = x.max()
    if m > 700:
        return m + np.log(np.sum(np.exp(x - m)))
    elif m < -700:
        return m + np.log(np.sum(np.exp(x - m)))
    else:
        return np.log(np.sum(np.exp(x)))

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    return float(m + logsumexp_stable(sh) / alpha)

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for mm in active:
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    return float(mx + logsumexp_stable(sh) / alpha + mu * (x.sum() - n/2.0)**2)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def m_seq_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2
    return val

def m_seq_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = remaining
    return project_feasible(h, n)

def make_msequence_start(n):
    # Generate m-sequence and adapt to our problem
    m = int(np.floor(np.log2(n)))
    if m < 4:
        m = 4
    seq_len = (1 << m) - 1
    # Find a primitive polynomial for GF(2^m)
    primitive_polys = {
        4: 0b10011, 5: 0b100111, 6: 0b1001011,
        7: 0b10001101, 8: 0b100111001, 9: 0b1000101111,
        10: 0b10000101111, 11: 0b100001010011,
        12: 0b100011100101, 13: 0b1000011001101,
    }
    if m in primitive_polys:
        poly = primitive_polys[m]
    else:
        poly = 0b10011
    # Generate m-sequence
    reg = [1] + [0] * (m - 1)
    seq = []
    for _ in range(2 * seq_len):
        seq.append(reg[0])
        feedback = 0
        for i in range(m):
            bit = (poly >> i) & 1
            if bit:
                feedback ^= reg[i]
        reg = [feedback] + reg[:-1]
    seq = np.array(seq[:seq_len], dtype=float)
    # Adapt to our n
    if n == seq_len:
        h = seq.copy()
    elif n > seq_len:
        h = np.zeros(n)
        reps = int(np.ceil(n / seq_len))
        long_seq = np.tile(seq, reps)[:n]
        h = long_seq
    else:
        h = seq[:n].copy()
    # Project to have exactly n/2 ones
    h = h.round()
    h = project_feasible(h, n)
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def interpolate_to_n(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.5/n_src, 2.0 - 0.5/n_src, n_src)
    x_dst = np.linspace(0.5/n_dst, 2.0 - 0.5/n_dst, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_per_alpha=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_per_alpha is not None and i < len(maxit_per_alpha):
            maxit = maxit_per_alpha[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(m_seq_obj, best_h, jac=m_seq_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 15,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif abs(c_new - best_c) < 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active_set(h, n, deadline, max_rounds=30):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

                    h_new = project_feasible(res.x, n)

def grad_search(h, n, deadline, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                  0.00005, 0.00002, 0.00001, 0.000005]
    steps_done = 0
    for step in step_sizes:
        stall = 0
        while time.time() < deadline and steps_done < max_steps and stall < 500:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            # Find best pair: decrease where grad>0, increase where grad<0
            pos_mask = (grad > 1e-12) & (best_h > 1e-12)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            # Sample for speed
            if len(pos_idx) > 30:
                pos_idx = pos_idx[rng_sample(len(pos_idx), min(30, len(pos_idx)))]
            if len(neg_idx) > 30:
                neg_idx = neg_idx[rng_sample(len(neg_idx), min(30, len(neg_idx)))]
            best_gain = -1e10
            best_i, best_j = -1, -1
            for i in pos_idx:
                gi = grad[i]
                for j in neg_idx:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i, best_j = i, j
            if best_i < 0 or best_gain < 1e-12:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stall = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stall += 1
    return best_h, best_c

def rng_sample(n, k):
    if k >= n:
        return np.arange(n)
    return np.random.choice(n, size=k, replace=False)

def pair_search(h, n, deadline, radius=30, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c

def bit_flip_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        mask = np.random.permutation(n)
        for i in mask:
            if time.time() > deadline:
                break
            old_val = best_h[i]
            # Try flipping to 1 or 0
            candidates = []
            if old_val < 0.999:
                candidates.append(1.0)
            if old_val > 0.001:
                candidates.append(0.0)
            for new_val in candidates:
                best_h[i] = new_val
                # Adjust one other element to maintain sum
                diff = old_val - new_val
                if abs(diff) < 1e-15:
                    continue
                # Find a position to compensate
                if diff > 0:
                    # Need to decrease somewhere else
                    comp_mask = (best_h > 1e-10) & (np.arange(n) != i)
                    comp_idx = np.where(comp_mask)[0]
                    if len(comp_idx) == 0:
                        best_h[i] = old_val
                        continue
                    j = comp_idx[np.random.randint(len(comp_idx))]
                    amt = min(diff, best_h[j])
                    best_h[j] -= amt
                    best_h[i] = old_val
                    best_h[i] = new_val
                    best_h = project_feasible(best_h, n)
                else:
                    # Need to increase somewhere else
                    comp_mask = (best_h < 1.0 - 1e-10) & (np.arange(n) != i)
                    comp_idx = np.where(comp_mask)[0]
                    if len(comp_idx) == 0:
                        best_h[i] = old_val
                        continue
                    j = comp_idx[np.random.randint(len(comp_idx))]
                    amt = min(-diff, 1.0 - best_h[j])
                    best_h[j] += amt
                    best_h[i] = old_val
                    best_h[i] = new_val
                    best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h = project_feasible(best_h, n)
                    # Restore
                    if j is not None:
                        best_h[j] = best_h[j]  # will be fixed by project
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
