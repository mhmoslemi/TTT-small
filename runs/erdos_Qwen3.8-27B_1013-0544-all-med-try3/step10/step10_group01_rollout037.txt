    #   if lag < 0: magnitude = -lag, use C
    #   if lag == 0: no effect
    
    effects = np.zeros((len(active), n))  # effects[r, j] = change in c[active[r]] from swapping any i with j

        # C[i, mag-1] = h[i-mag] + h[i+mag] (OOB=0)

        lo = np.zeros(n)
        # h[i-mag] for i=0,...,n-1: valid for i >= mag
        hi[mag:] = h[:n-mag]
        # h[i+mag] for i=0,...,n-1: valid for i <= n-1-mag
        lo[:n-mag] = h[mag:]

After swap (i: 1→0, j: 0→1):
- sum(A) changes by: -1 if i>=k, +1 if j>=k
- dot(A, B) = sum_{m=k}^{n-1} h[m]*h[m-k]
  Changes involve:
  - The term at m=i (if i>=k): was h[i]*h[i-k] = h[i-k], now 0. Change: -h[i-k]
  - The term at m=j (if j>=k): was h[j]*h[j-k] = 0, now h[j-k]. Change: +h[j-k]
  - The term at m=i+k (if i+k<n, i.e., the B-side position i): was h[i+k]*h[i] = h[i+k], now 0. Change: -h[i+k]
  - The term at m=j+k (if j+k<n): was h[j+k]*h[j] = 0, now h[j+k]. Change: +h[j+k]

So Δf(k) = (Δsum) - (Δdot)
Δsum = -1*(i>=k) + 1*(j>=k)
Δdot = -h[i-k]*(i>=k) + h[j-k]*(j>=k) - h[i+k]*(i+k<n) + h[j+k]*(j+k<n)

Δf(k) = -1*(i>=k) + (j>=k) + h[i-k]*(i>=k) - h[j-k]*(j>=k) + h[i+k]*(i+k<n) - h[j+k]*(j+k<n)
= (h[i-k]-1)*(i>=k) + (1-h[j-k])*(j>=k) + h[i+k]*(i+k<n) - h[j+k]*(j+k<n)
