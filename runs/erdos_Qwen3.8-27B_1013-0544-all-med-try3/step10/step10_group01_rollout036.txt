    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:

    sh = alpha * (c - m)
    smx = sh.max()

        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) == 0:
            break
        
        dx = 2.0 / n
        
        # Compute scores for all positions
        score = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            if s >= 0:
                nv = n - s
                score[0:nv] += (1 - best_h[s:n]) * dx
                score[nv:n] -= best_h[0:nv] * dx
            else:
                ss = -s
                nv = n - ss
                score[0:nv] -= best_h[ss:n] * dx
                score[nv:n] += (1 - best_h[0:nv]) * dx
        
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For ones: want HIGH score (removing them reduces OV most)
        # For zeros: want LOW score (adding them increases OV least)
        ones_scores = score[ones]
        zeros_scores = score[zeros]
        
        top_ones = ones[np.argsort(-ones_scores)[:top_k]]
        top_zeros = zeros[np.argsort(zeros_scores)[:top_k]]
        
        improved = False
        best_swap_c = best_c
        
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                c_new = compute_c5(best_h, n)
                if c_new < best_swap_c:
                    best_swap_c = c_new
                    best_i, best_j = i, j
                best_h[i] = 1.0
                best_h[j] = 0.0
        
        if best_swap_c < best_c - 1e-14:
            best_h[best_i] = 0.0
            best_h[best_j] = 1.0
            best_c = best_swap_c
            improved = True
        else:
            # Try random swaps to escape
            escaped = False
            for _ in range(10):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                if i == j:
                    continue
                best_h[i] = 0.0
                best_h[j] = 1.0
                c_new = compute_c5(best_h, n)
                if c_new < best_c + 0.0005:
                    best_c = c_new
                    escaped = True
                    break
                best_h[i] = 1.0
                best_h[j] = 0.0
            if not escaped:
                break
    
    return best_h, best_c
```

def compute_swap_delta(h, n, i, j):
    """Compute ΔOV vector for swap i(1→0), j(0→1). O(n) time."""
    dx = 2.0 / n
    delta = np.zeros(2*n - 1)
    
    # Effect of removing i (h[i] 1→0):
    # For shift s where i+s valid: -(1-h[i+s])*dx
    # For shift s where i-s valid: +h[i-s]*dx
    for s in range(-(n-1), n):
        k = n - 1 + s
        d = 0.0
        if 0 <= i + s < n:
            d -= (1 - h[i+s]) * dx
        if 0 <= i - s < n:
            d += h[i-s] * dx
        # Effect of adding j (h[j] 0→1):
        if 0 <= j + s < n:
            d += (1 - h[j+s]) * dx
        if 0 <= j - s < n:
            d -= h[j-s] * dx
        delta[k] = d
    
    return delta
```
