import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad *= dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_shifts_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def top_shifts_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi * (1.0 - x[k:n])
            grad[k:n] += wi * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wi * (1.0 - x[0:nv])
            grad[0:nv] += wi * (-x[kk:n])

    grad += 2.0 * mu * (x.sum() - n / 2.0)

def run_lbfgs(h, n, deadline, configs=None, bounds=None):

    best_c = compute_c5(h, n)
    if bounds is None:
        bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(500, 100.0), (2000, 500.0), (5000, 2000.0),

                   (100000, 200000.0), (200000, 1000000.0)]

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)

                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
    return h, best_c

def gradient_line_search(h, n, deadline, base_step=0.01):
    best_c = compute_c5(h, n)
    steps = [base_step, base_step*0.5, base_step*0.25, base_step*0.1, base_step*0.05]
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n])
                grad[k:n] += (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv])
                grad[0:nv] += (-h[kk:n])
        grad /= len(active)
        
        # Top gradient positions
        g_abs = np.abs(grad)
        top_idx = np.argsort(-g_abs)[:30]
        found = False
        for i_pos in range(len(top_idx)):
            if time.time() > deadline:
                return h, best_c
            i = top_idx[i_pos]
            for j_pos in range(len(top_idx)):
                if i_pos == j_pos:
                    continue
                j = top_idx[j_pos]
                for step in steps:
                    if grad[i] > 0:
                        if h[i] - step < 0 or h[j] + step > 1:
                            continue
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found = True
                        else:
                            h[i] += step
                            h[j] -= step
                    elif grad[i] < 0:
                        if h[i] + step > 1 or h[j] - step < 0:
                            continue
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found = True
                        else:
                            h[i] -= step
                            h[j] += step
        if not found:
            base_step *= 0.5
            if base_step < 1e-5:
                break
    return h, best_c

def binary_round(h, n):
    """Round h to binary (exactly n/2 ones) and evaluate."""
    h_b = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(h_b)
    h_b[:] = 0.0
    h_b[idx[n//2:]] = 1.0
    # Randomize ties
    return h_b, compute_c5(h_b, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def multi_resolution_optimize(h, n, deadline):
    """Optimize at multiple scales: coarse -> fine"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    scales = []
    for div in [4, 2]:
        n_c = n // div
        if n_c >= 50:
            scales.append(n_c)
    scales.append(n)
    
    for n_s in scales:
        if time.time() > deadline - 30:
            break
        if n_s == n:
            h_s = best_h
        else:
            h_s = downsample(best_h, len(best_h), n_s)
        
        if time.time() > deadline - 30:
            break
        t_alloc = min(60, (deadline - time.time()))
        if t_alloc > 15:
            h_s, c_s = run_lbfgs(h_s, n_s, time.time() + t_alloc,
                configs=[(1000, 100.0), (5000, 500.0), (20000, 2000.0),
                         (50000, 10000.0), (200000, 50000.0)])
        
        if n_s == n:
            best_h, best_c = h_s, c_s
        else:
            h_up = upsample(h_s, n_s, n)
            c_up = compute_c5(h_up, n)
            if c_up < best_c:
                best_h = h_up
                best_c = c_up
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick refinement at original resolution
    t1 = min(t0 + 60, deadline - 500)
    if time.time() < t1:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            configs=[(500, 50.0), (2000, 200.0), (10000, 1000.0),
                     (50000, 5000.0), (200000, 20000.0)])
    
    # Phase 2: Multi-resolution optimization (go up to n=1000)
    n_work = max(n_orig, 1000)
    if n_work > n_orig:
        h_work = upsample(best_h, n_orig, n_work)
    else:
        h_work = best_h.copy()
    c_work = compute_c5(h_work, n_work)
    if c_work > best_c:
        # Keep original if upsampling worsened
        h_work = np.full(n_work, 0.5)
        # Actually just use the upsampled one but note the value
        h_work = upsample(best_h, n_orig, n_work)
        c_work = compute_c5(h_work, n_work)
    
    best_h = h_work.copy()
    best_c = max(best_c, c_work)  # track the best we can achieve
    
    # Multi-res at n_work
    t2 = min(t0 + 180, deadline - 400)
    if time.time() < t2:
        best_h, best_c = multi_resolution_optimize(h_work, n_work, t2)
    
    # Phase 3: Aggressive top-shift optimization
    t3 = min(t0 + 350, deadline - 250)
    if time.time() < t3:
        best_h, best_c = run_top_shifts(best_h, n_work, t3, top_k=10)
    
    # Phase 4: More L-BFGS with high alpha
    t4 = min(t0 + 500, deadline - 200)
    if time.time() < t4:
        best_h, best_c = run_lbfgs(best_h, n_work, t4,
            configs=[(50000, 2000.0), (200000, 10000.0),
                     (500000, 50000.0), (1000000, 200000.0)])
    
    # Phase 5: Block perturb + re-optimize
    t5 = min(t0 + 650, deadline - 150)
    if time.time() < t5:
        best_h, best_c = block_perturb_optimize(best_h, n_work, t5, rng)
    
    # Phase 6: Top-shift again
    t6 = min(t0 + 780, deadline - 100)
    if time.time() < t6:
        best_h, best_c = run_top_shifts(best_h, n_work, t6, top_k=5)
    
    # Phase 7: Gradient line search
    t7 = min(t0 + 880, deadline - 50)
    if time.time() < t7:
        best_h, best_c = gradient_line_search(best_h, n_work, t7, base_step=0.005)
    
    # Phase 8: Fine pair search
    t8 = min(t0 + 980, deadline - 20)
    if time.time() < t8:
        best_h, best_c = fine_pair_optimize(best_h, n_work, t8, max_rad=15)
    
    # Phase 9: Final top-shift polish
    if time.time() < deadline - 3:
        best_h, best_c = run_top_shifts(best_h, n_work, deadline - 2, top_k=3)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n_work)
    best_c = compute_c5(best_h, n_work)
    
    # Also try binary rounding
    h_bin, c_bin = binary_round(best_h, n_work)
    if c_bin < best_c:
        best_h = h_bin
        best_c = c_bin
    
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
