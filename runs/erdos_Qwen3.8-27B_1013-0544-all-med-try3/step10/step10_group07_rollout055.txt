    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    """Project to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

            if s < 1e-15:

            # d/dh[i]: (1-h[i+s]) for i in [0, n-1-s], -h[i-s] for i in [s, n-1]
            if n - 1 - s >= 0:
                grad[0:n-s] += w * (1.0 - h[s:n])
            if s > 0:
                grad[s:n] -= w * h[0:n-s]
            elif s == 0:
                # h[i]*(1-h[i]), derivative is 1-2h[i]
                grad += w * (1.0 - 2.0 * h)
        else:
            s2 = -s  # positive
            # q_{-s2} = sum_{i=s2}^{n-1} h[i]*(1-h[i-s2])
            # = sum_{j=0}^{n-1-s2} h[j+s2]*(1-h[j])
            # d/dh[i]: for i in [s2, n-1] as first factor: (1-h[i-s2])
            #           for i in [0, n-1-s2] as second factor: -h[i+s2]
            grad[s2:n] += w * (1.0 - h[0:n-s2])
            grad[0:n-s2] -= w * h[s2:n]
    return grad

def subgradient_method(h, n, deadline, max_iters=50000):

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    eta = 0.1  # initial step size
    patience = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute current objective
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_c = corr.max()
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

            eta *= 0.8
            patience = 0
            if eta < 1e-6:
                break
        
        # Compute gradient
        grad = compute_gradient_at_max(h, n, top_k=5)
        grad_norm = np.sqrt((grad**2).sum())
        if grad_norm < 1e-10:

            h_new = h - step * grad
            h_new = project_h(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c + 1e-12:
                h = h_new
                break
        else:
            h = h - eta * grad
            h = project_h(h, n)
        
        # Adaptive eta
        if cur_c - c_new > 1e-8:
            eta = min(eta * 1.05, 1.0)
        else:
            eta = max(eta * 0.95, 1e-5)
    
    return best_h, best_c

    except Exception:

            for i in zeros_list[:]:

                    break
                best_j = -1
                best_nc = cur_c
                for j in ones_list:
                    delta = np.zeros(2 * n - 1)
                    im = i + m_range
                    imn = i - m_range
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    jm = j + m_range
                    jmn = j - m_range
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < best_nc:
                        best_nc = nc
                        best_j = j
                if best_j >= 0 and best_nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[best_j] = 0.0
                    delta = np.zeros(2 * n - 1)
                    im = i + m_range
                    imn = i - m_range
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    jm = best_j + m_range
                    jmn = best_j - m_range
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (best_j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - best_j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    corr = corr + delta
                    cur_c = best_nc
                    ones_list.remove(best_j)
                    ones_list.append(i)
                    zeros_list.remove(i)
                    zeros_list.append(best_j)
                    improved = True
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

def coord_descent(h, n, deadline, max_passes=200):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved_any = False
        idx_order = np.random.RandomState(p + 42).permutation(n)
        for i in idx_order:
            if time.time() > deadline:
                break
            old_val = h[i]
            # Find best value for h[i]
            # q_s as function of h[i]=t: at most quadratic
            # Just try a few candidate values
            candidates = [0.0, 1.0, 0.25, 0.5, 0.75, old_val + 0.1, old_val - 0.1]
            candidates = [c for c in candidates if 0 <= c <= 1]
            if old_val not in candidates:
                candidates.append(old_val)
            best_t = old_val
            best_cand_c = compute_c5(h, n)
            for t in candidates:
                if abs(t - old_val) < 1e-6:
                    continue
                h[i] = t
                h = project_h(h, n)
                c = compute_c5(h, n)
                if c < best_cand_c - 1e-14:
                    best_cand_c = c
                    best_t = t
                h[i] = old_val
            if abs(best_t - old_val) > 1e-6:
                h[i] = best_t
                h = project_h(h, n)
                improved_any = True
                if best_cand_c < best_c:
                    best_c = best_cand_c
                    best_h = h.copy()
        if not improved_any:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0

    # Phase 1: Quick SLSQP from parent
    p1 = t0 + 0.06 * (budget_s - 8)
    h = best_h.copy()
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n0, alpha, 1500, p1)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Soft-max gradient descent (main workhorse)
    p2 = t0 + 0.35 * (budget_s - 8)
    if time.time() < p2:
        h2, c2 = gradient_descend(best_h, n0, p2, max_iters=150000, gamma_init=0.8, gamma_min=0.03)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: SLSQP polish
    p3 = t0 + 0.45 * (budget_s - 8)
    if time.time() < p3:
        h3 = project_h(best_h + 0.02 * rng.randn(n0), n0)
        for alpha in [100000, 1000000, 10000000]:
            if time.time() > p3:
                break
            h3 = slsqp_optimize(h3, n0, alpha, 2500, p3)
            c3 = compute_c5(h3, n0)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()

    # Phase 4: Binary search then smooth
    p4 = t0 + 0.62 * (budget_s - 8)
    if time.time() < p4:
        hb, cb = fast_binary_search(best_h, n0, p4, max_sweeps=1500, n_restarts=4, seed=seed)
        # SLSQP from binary
        hp = project_h(hb + 0.05 * rng.randn(n0), n0)
        for alpha in [10000, 100000, 2000000]:
            if time.time() > p4:
                break
            hp = slsqp_optimize(hp, n0, alpha, 2000, p4)
            cp = compute_c5(hp, n0)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
                n_best = n0

    # Phase 5: Multi-resolution exploration
    p5 = t0 + 0.82 * (budget_s - 8)
    for nh in [200, 300, 400, 500, 150]:
        if time.time() > p5 - 20:
            break
        if nh == n_best:
            continue
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_h(hh, nh)
        # Gradient descent
        hh, ch = gradient_descend(hh, nh, min(p5 - 15, time.time() + 35), max_iters=30000, gamma_init=0.5, gamma_min=0.05)
        # SLSQP
        if time.time() < p5 - 10:
            hh2 = slsqp_optimize(hh, nh, 500000, 1000, p5 - 10)
            ch2 = compute_c5(hh2, nh)
            if ch2 < ch:
                hh, ch = hh2, ch2
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n_best = nh

    # Phase 6: Final gradient + SLSQP
    if time.time() < dl - 20:
        h6, c6 = gradient_descend(best_h, n_best, dl - 40, max_iters=50000, gamma_init=0.3, gamma_min=0.02)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < dl - 10:
            h6b = slsqp_optimize(best_h, n_best, 50000000, 3000, dl - 10)
            c6b = compute_c5(h6b, n_best)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b.copy()

    # Phase 7: Very short final SLSQP
    if time.time() < dl - 3:
        h7 = slsqp_optimize(best_h, n_best, 100000000, 1000, dl - 2)
        c7 = compute_c5(h7, n_best)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()

    best_h = project_h(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
