    sm = s.max()
    if sm > 400:
        val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_lse_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[top_idx]
    m = vals.max()
    s = alpha * (vals - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    np1 = n - 1
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:
            continue
        d = idx - np1
        wm = wi * dx
        if d >= 0:
            nv = n - d
            grad[0:nv] += wm * (1.0 - x[d:n])
            grad[d:n] += wm * (-x[0:nv])
        else:
            dd = -d
            nv = n - dd
            grad[dd:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[dd:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, t_end, alphas, mus, maxiter_list):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for i, (a, mu) in enumerate(zip(alphas, mus)):
        if time.time() > t_end - 3:
            break
        mi = maxiter_list[i] if i < len(maxiter_list) else 3000
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, a, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mi, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 20000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def topk_opt(h, n, t_end, k=6, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 50000, 200000, 1000000]
    for _ in range(20):
        if time.time() > t_end - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        thresh = max_val - max(1e-6, 0.002 * max_val)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > k:
            top_k = np.argsort(-c_vec[active])
            active = active[top_k[:k]]
        if len(active) == 0:
            break
        for a in alphas:
            if time.time() > t_end - 2:
                break
            try:
                res = minimize(topk_lse_obj, h, jac=topk_lse_grad,
                              args=(n, active, a, 1.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': 15000, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def fourier_perturb(h, n, modes, amplitude, seed=0):
    rng = np.random.RandomState(seed)
    h_new = h.copy()
    x = np.arange(n)
    for m, amp in zip(modes, amplitude):
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp * np.sin(m * np.pi * x / n + phase)
    return project_feasible(h_new, n)

def coord_descent(h, n, t_end, n_iter_max=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    c_vec = compute_c5_vec(h, n)
    np1 = n - 1
    active_all = np.argsort(-c_vec)[:30]
    for _ in range(n_iter_max):
        if time.time() > t_end:
            break
        improved = False
        for d_idx in range(len(steps)):
            d = steps[d_idx]
            for ai in active_all:
                if time.time() > t_end:
                    return best_h if 'best_h' in dir() else h, best_c
                a = ai - np1
                nv = n - abs(a)
                if a >= 0:
                    i_range = range(nv)
                else:
                    i_range = range(nv)
                # Find best i modification
                idx_list = list(range(max(0, min(active_all) - 10, n), min(n, max(active_all) + 10)))
                for i in idx_list:
                    if i >= n:
                        continue
                    gi = 0
                    for ai2 in active_all:
                        aa = ai2 - np1
                        if aa >= 0:
                            if i < n - aa:
                                gi += (1.0 - h[i + aa])
                            if i >= aa:
                                gi += (-h[i - aa])
                        else:
                            dd2 = -aa
                            if i >= dd2:
                                gi += (1.0 - h[i - dd2])
                            if i < n - dd2:
                                gi += (-h[dd2 + i - dd2]) if False else 0
                    # Simplified: just try ±d at position i
                    if h[i] + d <= 1.0:
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                    if h[i] - d >= 0.0:
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                    if improved:
                        c_vec = compute_c5_vec(h, n)
                        active_all = np.argsort(-c_vec)[:30]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and d_idx == len(steps) - 1:
            break
    return h, best_c

def pair_descent(h, n, t_end, radius=40):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    np1 = n - 1
    top_positions = set()
    for idx in np.argsort(-c_vec)[:10]:
        d = idx - np1
        top_positions.update(range(max(0, n - abs(d)), min(n, n)))
    top_list = sorted(top_positions)
    if len(top_list) < 10:
        top_list = list(range(n // 3, 2 * n // 3))
    
    deltas = [0.003, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
    for d in deltas:
        if time.time() > t_end:
            break
        improved_any = True
        while improved_any:
            if time.time() > t_end:
                return h, best_c
            improved_any = False
            for i in top_list:
                if time.time() > t_end:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if improved_any:
                        break
                if improved_any:
                    break
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_orig = 300

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    best_n = n_orig

    # Phase 1: Quick polish at original resolution
    t1 = min(t0 + 50, deadline - 800)
    if t1 > t0 + 5:
        best_h, best_c = lbfgs_run(best_h, best_n, t1,
            alphas=[200, 2000, 20000],
            mus=[10.0, 100.0, 1000.0],
            maxiter_list=[3000, 3000, 3000])

    # Phase 2: Try n=1000
    n_hi = 1000
    h_hi = upsample(best_h, best_n, n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    t2 = min(t0 + 150, deadline - 600)
    if t2 > time.time() + 10:
        h_hi, c_hi = lbfgs_run(h_hi, n_hi, t2,
            alphas=[500, 5000, 50000],
            mus=[50.0, 500.0, 5000.0],
            maxiter_list=[2000, 2000, 2000])
    if c_hi < best_c - 1e-6:
        best_h = h_hi
        best_c = c_hi
        best_n = n_hi

    # Phase 3: Back to n=500 for main optimization (or keep at n_hi if better)
    n_main = best_n
    h_main = best_h.copy()
    c_main = best_c

    # Phase 4: Multi-start with structured perturbations
    t4_start = time.time()
    t4_end = min(t4_start + 300, deadline - 250)
    
    perturbations = []
    # Fourier perturbations at different frequencies
    for m, amp, s in [(1, 0.02, 1), (2, 0.015, 2), (3, 0.01, 3), (5, 0.008, 4), (8, 0.005, 5)]:
        h_pert = fourier_perturb(h_main, n_main, [m], [amp], seed=s)
        perturbations.append(h_pert)
    # Random smooth perturbation
    rng = np.random.RandomState(seed)
    noise = np.cumsum(rng.randn(n_main)) * 0.005
    noise -= noise.mean()
    h_rand = project_feasible(h_main + noise, n_main)
    perturbations.append(h_rand)
    # Original
    perturbations.append(h_main)

    for p_i, h_p in enumerate(perturbations):
        if time.time() > t4_end - 30:
            break
        per_budget = min(45, (t4_end - time.time()) / max(1, len(perturbations) - p_i))
        t_per_end = time.time() + per_budget
        if t_per_end > t4_end:
            t_per_end = t4_end
        if per_budget < 10:
            break
        h_opt, c_opt = lbfgs_run(h_p, n_main, t_per_end,
            alphas=[1000, 10000, 100000],
            mus=[100.0, 500.0, 5000.0],
            maxiter_list=[1500, 1500, 1500])
        if c_opt < c_main - 1e-8:
            c_main = c_opt
            h_main = h_opt
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_main.copy()

    # Phase 5: Top-k focused optimization
    t5_end = min(time.time() + 150, deadline - 100)
    if time.time() < t5_end - 10:
        h_main, c_main = topk_opt(h_main, n_main, t5_end, k=5,
            alphas=[10000, 100000, 500000, 2000000])
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 6: Another round of high-alpha L-BFGS on best
    t6_end = min(time.time() + 100, deadline - 60)
    if time.time() < t6_end - 10:
        h_main, c_main = lbfgs_run(h_main, n_main, t6_end,
            alphas=[50000, 200000, 1000000, 5000000],
            mus=[5000.0, 20000.0, 100000.0, 500000.0],
            maxiter_list=[2000, 2000, 2000, 2000])
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 7: Pair descent for fine-tuning
    t7_end = min(time.time() + 80, deadline - 15)
    if time.time() < t7_end - 5:
        h_main, c_main = pair_descent(h_main, n_main, t7_end, radius=30)
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Phase 8: Final top-k squeeze
    t8_end = min(time.time() + 40, deadline - 5)
    if time.time() < t8_end - 3:
        h_main, c_main = topk_opt(h_main, n_main, t8_end, k=3,
            alphas=[500000, 2000000, 10000000])
        if c_main < best_c:
            best_c = c_main
            best_h = h_main.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
