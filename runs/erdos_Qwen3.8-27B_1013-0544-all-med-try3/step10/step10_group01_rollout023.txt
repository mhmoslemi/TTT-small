import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    h_b = np.zeros(n)
    # Sort by h value, assign top n/2 to 1
    indices = np.argsort(-h)
    h_b[indices[:n//2]] = 1.0
    return h_b

def overlap_at_k(h, k, n):
    """Compute overlap at a single shift k (positive or negative)"""
    dx_val = 2.0 / n
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return 0.0
        return float(np.sum(h[:nv] * (1 - h[k:]))) * dx_val
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return 0.0
        return float(np.sum(h[kk:] * (1 - h[:nv]))) * dx_val

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def binary_local_search(h_b, n, deadline, max_iter=2000000):
    """Greedy binary local search using the s[p] trick"""
    h_b = h_b.copy()
    dx_val = 2.0 / n
    ov = np.correlate(h_b, 1 - h_b, mode='full')  # without dx for speed
    best_ov_max = ov.max()
    iters = 0
    
    while time.time() < deadline and iters < max_iter:
        iters += 1
        # Find max overlap location
        max_idx = np.argmax(ov)
        k = max_idx - (n - 1)
        if k == 0:
            break
        
        # Compute s[p] = h[p+k] + h[p-k] with range checks
        s = np.zeros(n)
        if k > 0:
            s[:n-k] += h_b[k:]       # h[p+k] for p in [0, n-k)
            s[k:] += h_b[:n-k]       # h[p-k] for p in [k, n)
        elif k < 0:
            kk = -k
            s[kk:] += h_b[:n-kk]     # h[p-kk] for p in [kk, n)
            s[:n-kk] += h_b[kk:]     # h[p+kk] for p in [0, n-kk)
        
        # Find best 1 with smallest s, best 0 with largest s
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For the swap to help at shift k, we need s[i] < s[j]
        # i is a 1 (goes to 0), j is a 0 (goes to 1)
        best_i = ones[np.argmin(s[ones])]
        best_j = zeros[np.argmax(s[zeros])]
        
        if s[best_i] >= s[best_j]:
            # No improving swap at this shift; try simulated annealing move
            # Or break
            if iters % 1000 == 0:
                # Try random swap
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_b[i] = 0
                h_b[j] = 1
                new_ov = np.correlate(h_b, 1 - h_b, mode='full')
                if new_ov.max() < best_ov_max:
                    best_ov_max = new_ov.max()
                    ov = new_ov
                else:
                    h_b[i] = 1
                    h_b[j] = 0
            break
        
        # Perform swap
        h_b[best_i] = 0
        h_b[best_j] = 1
        ov = np.correlate(h_b, 1 - h_b, mode='full')
        if ov.max() < best_ov_max:
            best_ov_max = ov.max()
        else:
            # Revert
            h_b[best_i] = 1
            h_b[best_j] = 0
            break
    
    return h_b, best_ov_max * dx_val

def simulated_annealing(h_b, n, deadline, total_iters=100000):
    h_b = h_b.copy()
    dx_val = 2.0 / n
    ov = np.correlate(h_b, 1 - h_b, mode='full')
    current_max = ov.max()
    best_max = current_max
    best_h = h_b.copy()
    
    T_start = 0.5
    T_end = 0.01
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / total_iters)
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h_b[i] = 0
        h_b[j] = 1
        new_ov = np.correlate(h_b, 1 - h_b, mode='full')
        new_max = new_ov.max()
        delta = new_max - current_max
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            current_max = new_max
            ov = new_ov
            # Update sets
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h_b.copy()
        else:
            h_b[i] = 1
            h_b[j] = 0
    
    return best_h, best_max * dx_val

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    indices = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[indices[:n // 2]] = 1.0
    return h_b

def thue_morse_start(n):
    h = np.zeros(n)
    n_ones = n // 2
    tm = np.array([bin(k).count('1') & 1 for k in range(n)])
    # Try both parities, pick the better one
    h1 = np.zeros(n)
    h1[np.where(tm == 0)[0][:n_ones]] = 1.0
    h2 = np.zeros(n)
    h2[np.where(tm == 1)[0][:n_ones]] = 1.0
    c1 = compute_c5(h1, n)
    c2 = compute_c5(h2, n)
    return h1 if c1 <= c2 else h2

def beatty_start(n, alpha):
    h = np.zeros(n)
    n_ones = n // 2
    count = 0
    k = 1
    positions = set()
    while count < n_ones and k < 20 * n:
        p = int(k * alpha) % n
        if p not in positions:
            h[p] = 1.0
            positions.add(p)
            count += 1
        k += 1
    return h

def binary_greedy_search(h_b, n, deadline, max_iter=500000):
    """Fast greedy binary local search"""
    h_b = h_b.copy()
    dx_val = 2.0 / n
    ov = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_max = ov.max()
    stalled = 0
    iters = 0
    
    while time.time() < deadline and iters < max_iter:
        iters += 1
        max_idx = int(np.argmax(ov))
        k = max_idx - (n - 1)
        if k == 0:
            break
        
        # s[p] = contribution to "why is overlap high at k"
        # For k > 0: s[p] = h[p+k] (for p < n-k) + h[p-k] (for p >= k)
        s = np.zeros(n)
        if k > 0:
            s[:n-k] = h_b[k:].copy()
            s[k:] += h_b[:n-k]
        else:
            kk = -k
            s[kk:] = h_b[:n-kk].copy()
            s[:n-kk] += h_b[kk:]
        
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Best swap: min s among ones, max s among zeros
        min_s_one = np.min(s[ones])
        max_s_zero = np.max(s[zeros])
        
        if min_s_one >= max_s_zero:
            stalled += 1
            if stalled > 50:
                break
            # Try a random perturbation
            for _ in range(10):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_b[i] = 0
                h_b[j] = 1
                new_ov = np.correlate(h_b, 1.0 - h_b, mode='full')
                if new_ov.max() < best_max:
                    best_max = new_ov.max()
                    ov = new_ov
                    stalled = 0
                    break
                h_b[i] = 1
                h_b[j] = 0
            if stalled > 200:
                break
            continue
        
        best_i = ones[np.argmin(s[ones])]
        best_j = zeros[np.argmax(s[zeros])]
        
        h_b[best_i] = 0
        h_b[best_j] = 1
        ov = np.correlate(h_b, 1.0 - h_b, mode='full')
        new_max = ov.max()
        if new_max < best_max:
            best_max = new_max
            stalled = 0
        else:
            h_b[best_i] = 1
            h_b[best_j] = 0
            stalled += 1
            if stalled > 50:
                break
    
    return h_b, best_max * dx_val

def binary_simulated_annealing(h_b, n, deadline, total_iters=200000):

    dx_val = 2.0 / n
    ov = np.correlate(h_b, 1.0 - h_b, mode='full')

def smooth_max_val(c, alpha):

    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_grad_full(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

def fractional_lbfgs(h, n, deadline, alphas=None, mu=0.001):

def fine_local_search(h, n, deadline, radius=30):
    """Very fine local search with small pair moves"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    
                    # Try moving mass from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):

def smooth_max_val(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    smx = float(sh.max())
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad_full(c, alpha, h, n):
    m = float(c.max())
    sh = alpha * (c - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = smooth_max_grad_full(c, alpha, x, n)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def fractional_lbfgs(h, n, deadline, alphas=None, mu=0.001):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 25:
            break
        maxit = max(200, int(rem * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def targeted_obj(x, n, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    sh = 50000.0 * (vals - m)
    smx = float(sh.max())
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / 50000.0
    else:
        val = m + np.log(np.sum(np.exp(sh))) / 50000.0
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def targeted_grad(x, n, active, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = float(vals.max())
    sh = 50000.0 * (vals - m)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w /= np.sum(w)
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - x[k:n])
            g[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - x[0:nv])
            g[0:nv] += wm * (-x[kk:n])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def targeted_fractional(h, n, deadline, max_k=12):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for k_active in range(3, max_k + 1):
        rem = deadline - time.time()
        if rem < 20:
            break
        ov = compute_c5_vec(best_h, n)
        top_indices = np.argsort(-ov)[:k_active]
        max_val = ov[top_indices[0]]
        thr = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(ov >= thr)[0]
        if len(active) == 0:
            active = top_indices
        maxit = max(200, int(rem * 4))
        for mu in [0.001, 0.0001]:
            if time.time() > deadline - 10:
                break
            try:
                res = minimize(targeted_obj, best_h, jac=targeted_grad,
                              args=(n, active, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def gradient_local_search(h, n, deadline, step_sizes=None, max_steps=100000):
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps_done = 0
    for step in step_sizes:
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = float(c_vec.max())
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = int(idx) - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(len(active), 1)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30, deltas=None):
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Binary optimization (200s)
    p1_end = min(t0 + 200, deadline - 500)

    starts = []
    starts.append(to_binary(h0, n))
    starts.append(thue_morse_start(n))
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts.append(beatty_start(n, golden / 2.0))
    starts.append(beatty_start(n, np.sqrt(2.0) / 2.0))
    starts.append(beatty_start(n, np.sqrt(3.0) / 2.0))
    starts.append(beatty_start(n, np.e / 3.0))
    for _ in range(3):
        idx = np.random.choice(n, size=n // 2, replace=False)
        hr = np.zeros(n)
        hr[idx] = 1.0
        starts.append(hr)

    best_bin_c = np.inf
    best_bin_h = starts[0]
    for hb in starts:
        c = compute_c5(hb, n)
        if c < best_bin_c:
            best_bin_c = c
            best_bin_h = hb.copy()

    if time.time() < p1_end:
        h_bin, c_bin = binary_greedy_search(best_bin_h, n, p1_end)
        if c_bin < best_bin_c:
            best_bin_c = c_bin
            best_bin_h = h_bin

    # Phase 1b: Simulated annealing (80s)
    if time.time() < deadline - 400:
        sa_end = min(time.time() + 80, deadline - 380)
        h_sa, c_sa = binary_simulated_annealing(best_bin_h, n, sa_end, total_iters=400000)
        if c_sa < best_bin_c:
            best_bin_c = c_sa
            best_bin_h = h_sa

    # Phase 1c: More greedy after SA (60s)
    if time.time() < deadline - 350:
        p1c_end = min(time.time() + 60, deadline - 320)
        h_bin2, c_bin2 = binary_greedy_search(best_bin_h, n, p1c_end)
        if c_bin2 < best_bin_c:
            best_bin_c = c_bin2
            best_bin_h = h_bin2

    # Phase 2: Fractional optimization from best binary (300s)
    start_frac = best_bin_h.copy()
    p2_end = min(time.time() + 250, deadline - 250)
    h_f1, c_f1 = fractional_lbfgs(start_frac, n, p2_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000], mu=0.001)
    if c_f1 < best_c:
        best_c = c_f1
        best_h = h_f1

    # Phase 2b: Targeted active set (120s)
    if time.time() < deadline - 150:
        p2b_end = min(time.time() + 120, deadline - 120)
        h_t, c_t = targeted_fractional(best_h, n, p2b_end, max_k=10)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 3: Gradient local search (100s)
    if time.time() < deadline - 120:
        p3_end = min(time.time() + 100, deadline - 80)
        h_g, c_g = gradient_local_search(best_h, n, p3_end,
            step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=80000)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g

    # Phase 4: Fine pair search (80s)
    if time.time() < deadline - 60:
        p4_end = min(time.time() + 80, deadline - 35)
        h_pf, c_pf = fine_pair_search(best_h, n, p4_end, radius=25,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c_pf < best_c:
            best_c = c_pf
            best_h = h_pf

    # Phase 5: Final fractional polish (30s)
    if time.time() < deadline - 15:
        p5_end = min(time.time() + 30, deadline - 5)
        h_p, c_p = fractional_lbfgs(best_h, n, p5_end,
            alphas=[2000000, 10000000, 50000000], mu=0.0001)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    # Phase 6: Final very fine search
    if time.time() < deadline - 3:
        h_ff, c_ff = fine_pair_search(best_h, n, deadline, radius=15,
            deltas=[0.00001, 0.000005, 0.000002, 0.000001])
        if c_ff < best_c:
            best_c = c_ff
            best_h = h_ff

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
