    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def simulated_annealing_binary(h_bin, n, deadline, rng, t_init=0.8, t_end=0.02, total_steps=None):
    ones = set(np.where(h_bin > 0.5)[0])
    zeros = set(np.where(h_bin < 0.5)[0])
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    cur_c = best_c
    cur_h = h_bin.copy()
    no_improve = 0
    best_no_improve = 20000

    step = 0
    if total_steps is None:
        elapsed_est = deadline - time.time()
        total_steps = max(10000, int(elapsed_est * 8000))

    while time.time() < deadline:
        step += 1
        frac = step / total_steps
        t = t_init * (t_end / t_init) ** frac

        if no_improve > best_no_improve and frac > 0.3:
            best_no_improve = int(best_no_improve * 1.3)
            t = max(t, 0.15)

        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h, n)

        delta = new_c - cur_c
        if delta < -1e-15 or (t > 1e-12 and rng.random() < np.exp(-max(delta, 0) / t)):
            cur_c = new_c
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1

        if step % 50000 == 0 and no_improve > 80000:
            for _ in range(10):
                ii = list(ones)[rng.randint(len(ones))]
                jj = list(zeros)[rng.randint(len(zeros))]
                cur_h[ii] = 0.0
                cur_h[jj] = 1.0
                nc = compute_c5(cur_h, n)
                if nc < cur_c:
                    cur_c = nc
                    ones.discard(ii)
                    ones.add(jj)
                    zeros.discard(jj)
                    zeros.add(ii)
                    if nc < best_c:
                        best_c = nc
                        best_h = cur_h.copy()
                        no_improve = 0
                else:
                    cur_h[ii] = 1.0
                    cur_h[jj] = 0.0

    return best_h, best_c

def greedy_binary_refine(h_bin, n, deadline, rng, max_rounds=5000):
    ones = set(np.where(h_bin > 0.5)[0])
    zeros = set(np.where(h_bin < 0.5)[0])
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        alpha = 500000.0
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad_from_w(w, best_h, n)

        improved = False
        pos_ones = [i for i in ones if grad[i] > 1e-10]
        neg_zeros = [j for j in zeros if grad[j] < -1e-10]

        if pos_ones and neg_zeros:
            pos_ones.sort(key=lambda x: grad[x], reverse=True)
            neg_zeros.sort(key=lambda x: grad[x])
            for ii in pos_ones[:40]:
                if improved:
                    break
                for jj in neg_zeros[:40]:
                    best_h[ii] = 0.0
                    best_h[jj] = 1.0
                    nc = compute_c5(best_h, n)
                    if nc < best_c - 1e-14:
                        best_c = nc
                        ones.discard(ii)
                        ones.add(jj)
                        zeros.discard(jj)
                        zeros.add(ii)
                        improved = True
                        break
                    else:
                        best_h[ii] = 1.0
                        best_h[jj] = 0.0

        if not improved and rnd % 20 == 0:
            for _ in range(5):
                if time.time() > deadline:
                    break
                ii = list(ones)[rng.randint(len(ones))]
                jj = list(zeros)[rng.randint(len(zeros))]
                best_h[ii] = 0.0
                best_h[jj] = 1.0
                nc = compute_c5(best_h, n)
                if nc < best_c - 1e-14:
                    best_c = nc
                    ones.discard(ii)
                    ones.add(jj)
                    zeros.discard(jj)
                    zeros.add(ii)
                    improved = True
                else:
                    best_h[ii] = 1.0
                    best_h[jj] = 0.0

    return best_h, best_c

def continuous_refine(h, n, deadline):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    phases = [
        (5000, 0.01, 2000),
        (50000, 0.005, 2000),
        (200000, 0.003, 2000),
        (1000000, 0.001, 3000),
        (5000000, 0.001, 3000),
    ]

    for alpha, mu, maxit in phases:
        if time.time() > deadline - 3:
            break
        time_slice = min(40, deadline - time.time() - 3)
        if time_slice < 5:
            break
        sub_deadline = time.time() + time_slice
        h_new, c_new = run_lbfgs(best_h, n, sub_deadline, alpha=alpha, mu=mu, maxit=maxit)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    return best_h, best_c

def active_set_descent(h, n, deadline, max_rounds=500):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]

        w = smooth_max_w(c_vec, 500000.0)
        grad = h_grad_from_w(w, best_h, n)

        pos_mask = (grad > 1e-12) & (best_h > 1e-6)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-6)

        if not pos_mask.any() or not neg_mask.any():
            break

        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]

        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos_idx[:50]:
            for j in neg_idx[:50]:
                g = grad[i] - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j

        if best_i < 0:
            break

        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        found = False
        for s in steps:
            amt = min(s, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                continue
            trial = best_h.copy()
            trial[best_i] -= amt
            trial[best_j] += amt
            trial = project_feasible(trial, n)
            nc = compute_c5(trial, n)
            if nc < best_c - 1e-14:
                best_h = trial
                best_c = nc
                found = True
                break

        if not found:
            break

    return best_h, best_c

def multi_resolution_search(h_base, n_base, deadline, rng):
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n_base)
    best_n = n_base

    test_ns = [400, 500, 600, 700, 800]
    if n_base in test_ns:
        test_ns.remove(n_base)
    test_ns.sort(key=lambda nn: abs(nn - n_base))

    time_per = max(20, (deadline - time.time()) / max(len(test_ns), 1))

    for nn in test_ns[:3]:
        if time.time() > deadline - 10:
            break
        sub_deadline = time.time() + time_per
        if nn == n_base:
            h_tmp = h_base.copy()
        else:
            h_interp = np.interp(np.linspace(0, 2, nn), np.linspace(0, 2, n_base), h_base)
            h_tmp = project_feasible(h_interp, nn)
        h_bin = binarize(h_tmp, nn)
        h2, c2 = simulated_annealing_binary(h_bin, nn, sub_deadline, rng,
                                            t_init=0.5, t_end=0.05,
                                            total_steps=max(5000, int(time_per * 5000)))
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = nn

    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Phase 1: Quick L-BFGS (40s)
    p1_end = t0 + 40
    h1, c1 = run_lbfgs(best_h, best_n, p1_end, alpha=20000, mu=0.01, maxit=2000)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Binarize + Simulated Annealing (450s)
    p2_end = t0 + 490
    h_bin = binarize(best_h, best_n)
    h2, c2 = simulated_annealing_binary(h_bin, best_n, p2_end, rng,
                                         t_init=0.8, t_end=0.02)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    else:
        pass

    # Phase 3: Greedy binary refinement (150s)
    p3_end = t0 + 640
    h_bin2 = binarize(best_h, best_n)
    h3, c3 = greedy_binary_refine(h_bin2, best_n, p3_end, rng, max_rounds=3000)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: More SA with different params (120s)
    p4_end = t0 + 760
    if time.time() < p4_end:
        h_bin3 = binarize(best_h, best_n)
        h4, c4 = simulated_annealing_binary(h_bin3, best_n, p4_end, rng,
                                             t_init=0.4, t_end=0.01,
                                             total_steps=max(5000, int((p4_end - time.time()) * 10000)))
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Continuous refinement (L-BFGS) (150s)
    p5_end = t0 + 910
    if time.time() < p5_end:
        h5, c5 = continuous_refine(best_h, best_n, p5_end)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active-set coordinate descent (80s)
    p6_end = t0 + 990
    if time.time() < p6_end:
        h6, c6 = active_set_descent(best_h, best_n, p6_end, max_rounds=2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final binary SA + continuous (80s)
    p7_end = deadline - 3
    if time.time() < p7_end:
        h_bin4 = binarize(best_h, best_n)
        half = (p7_end - time.time()) / 2
        h7, c7 = simulated_annealing_binary(h_bin4, best_n, time.time() + half, rng,
                                             t_init=0.3, t_end=0.005,
                                             total_steps=max(3000, int(half * 10000)))
        if c7 < best_c:
            best_c = c7
            best_h = h7
            p7_end2 = deadline - 3
            if time.time() < p7_end2:
                h7b, c7b = continuous_refine(best_h, best_n, p7_end2)
                if c7b < best_c:
                    best_c = c7b
                    best_h = h7b

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
