            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (target / tot)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_stripe(n):
    """Alternating blocks pattern"""
    h = np.zeros(n)
    h[0::4] = 1.0
    h[1::4] = 1.0
    h[2::4] = 0.0
    h[3::4] = 0.0
    return project_feasible(h, n)

def make_sine(h_template, n):
    """Sine-modulated pattern"""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    return project_feasible(h, n)

def run_lbfgs_phase(h, n, deadline, alpha_list, mu, maxit_per_alpha, maxfun_mult=8):
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 10:
            break
        rem = deadline - time.time() - 5
        if rem < 5:
            break
        maxit = min(maxit_per_alpha, max(100, int(rem * 2.5)))
        try:
            res = minimize(full_obj, h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-20, 'maxfun': maxit * maxfun_mult,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            c_cur = compute_c5(h, n)
            if c_new < c_cur:
                h = h_new
        except Exception:
            pass
    return h, compute_c5(h, n)

def run_active_phase(h, n, deadline, top_k_list, alpha_list, mu_list, rounds=2):
    bounds = [(0.0, 1.0)] * n
    cur_c = compute_c5(h, n)
    for top_k, alpha, mu in zip(top_k_list, alpha_list, mu_list):
        if time.time() > deadline - 15:
            break
        for _ in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-6, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                order = np.argsort(-c_vec[active])
                active = active[order[:top_k]]
            if len(active) == 0:
                continue
            rem = deadline - time.time() - 5
            if rem < 5:
                break
            maxit = min(2000, max(100, int(rem * 3)))
            try:
                res = minimize(active_obj, h, jac=active_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-20, 'maxfun': maxit * 10,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c - 1e-15:
                    cur_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, cur_c

def fine_pair_search(h, n, deadline, radius, deltas, max_evals_per_delta):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline - 5:
            break
        improved = True
        evals = 0
        while improved and time.time() < deadline - 5:
            improved = False
            evals = 0
            for i in range(n):
                if time.time() > deadline - 5 or evals > max_evals_per_delta:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if evals >= max_evals_per_delta:
                        break
                    # Try h[i] -= d, h[j] += d
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        evals += 1
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try h[i] += d, h[j] -= d
                    if best_h[i] <= 1.0 - d and best_h[j] >= d:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        evals += 1
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def gradient_step_refine(h, n, deadline, n_steps=500, lr=0.001):
    """Simple projected gradient descent with decreasing step size"""
    cur_h = h.copy()
    cur_c = compute_c5(h, n)
    alpha_smooth = 500000
    for i in range(n_steps):
        if time.time() > deadline - 5:
            break
        lr_i = lr * (1 - i / n_steps) * 2
        if lr_i < 1e-7:
            break
        c = compute_c5_vec(cur_h, n)
        g = smooth_max_grad(c, alpha_smooth, cur_h, n)
        step = cur_h - lr_i * g
        step = np.clip(step, 0.0, 1.0)
        step = project_feasible(step, n)
        c_new = compute_c5(step, n)
        if c_new < cur_c - 1e-16:
            cur_c = c_new
            cur_h = step
        else:
            lr *= 0.7
    return cur_h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate diverse starting points
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    beatty_params = [golden / 2.0, np.sqrt(2.0) / 2.0, 0.4312, 0.382,
                     0.318, 1.0/3.0, 0.414, np.sqrt(3.0)/3.0, 0.366,
                     golden / 3.0 + 0.1, 1.0/7.0 + 0.3]
    alt_starts = []
    for p in beatty_params:
        alt_starts.append(make_beatty(n_orig, p))
    alt_starts.append(make_random_binary(n_orig, rng))
    alt_starts.append(make_random_binary(n_orig, rng))
    alt_starts.append(make_stripe(n_orig))
    alt_starts.append(make_sine(None, n_orig))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup (multi-alpha)
    p1_end = min(t0 + 90, deadline - 800)
    if p1_end > t0 + 15:
        h1, c1 = run_lbfgs_phase(best_h, best_n, p1_end,
                                  [500, 5000, 50000, 500000], 0.01, 1500)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization rounds
    p2_end = min(time.time() + 250, deadline - 400)
    if p2_end > time.time() + 30:
        cur_n = len(best_h)
        h2, c2 = run_active_phase(best_h, cur_n, p2_end,
                                   [4, 8, 12, 20, 30],
                                   [20000, 200000, 500000, 2000000, 5000000],
                                   [0.002, 0.001, 0.0005, 0.0005, 0.0003],
                                   rounds=2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try higher resolution (n=1000)
    n_up = best_n * 2
    p3_start = time.time()
    p3_end = min(p3_start + 250, deadline - 300)
    if p3_end > p3_start + 60 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.002:
            h_up, c_up = run_lbfgs_phase(h_up, n_up, min(p3_end - 60, time.time() + 120),
                                          [5000, 50000, 500000, 2000000], 0.005, 1000)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n_up
            # Active set at higher res
            if time.time() < p3_end - 20:
                h_up, c_up = run_active_phase(best_h, len(best_h), p3_end - 10,
                                               [6, 12, 20],
                                               [500000, 2000000, 5000000],
                                               [0.001, 0.0005, 0.0003],
                                               rounds=2)
                if c_up < best_c:
                    best_c = c_up
                    best_h = h_up

    # Phase 4: Another round of active set at current resolution
    p4_end = min(time.time() + 120, deadline - 200)
    if p4_end > time.time() + 20:
        cur_n = len(best_h)
        h4, c4 = run_active_phase(best_h, cur_n, p4_end,
                                   [4, 6, 10],
                                   [5000000, 10000000, 20000000],
                                   [0.0005, 0.0003, 0.0002],
                                   rounds=3)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient step refinement
    p5_end = min(time.time() + 80, deadline - 150)
    if p5_end > time.time() + 15:
        cur_n = len(best_h)
        h5, c5 = gradient_step_refine(best_h, cur_n, p5_end,
                                       n_steps=300, lr=0.0005)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine pair search (coarse)
    p6_end = min(time.time() + 100, deadline - 80)
    if p6_end > time.time() + 15:
        cur_n = len(best_h)
        h6, c6 = fine_pair_search(best_h, cur_n, p6_end,
                                   radius=50,
                                   deltas=[0.005, 0.002, 0.001],
                                   max_evals_per_delta=20000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final active set polish
    p7_end = min(time.time() + 60, deadline - 30)
    if p7_end > time.time() + 10:
        cur_n = len(best_h)
        h7, c7 = run_active_phase(best_h, cur_n, p7_end,
                                   [3, 5, 8],
                                   [10000000, 20000000, 50000000],
                                   [0.0003, 0.0002, 0.0001],
                                   rounds=3)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Ultra-fine pair search
    p8_end = deadline - 3
    if p8_end > time.time() + 10:
        cur_n = len(best_h)
        h8, c8 = fine_pair_search(best_h, cur_n, p8_end,
                                   radius=30,
                                   deltas=[0.001, 0.0005, 0.0002, 0.0001],
                                   max_evals_per_delta=15000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
