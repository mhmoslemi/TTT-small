    #    lag=0 (m=n-1): t*(1-t)*dx -> B+=dx, C+=-dx

    # 2. j+lag=i, i.e., j=i-lag: valid if 0<=i-lag<n and lag!=0

                C[m] += -dx

        vals = A + B * t + C * t * t

                            options={'xatol': 1e-10})
        t_opt = res.x
    else:
        t_opt = grid[idx]
    
    return t_opt

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def coord_descent_sweep(h, n, mu=0.5, n_grid=300, time_limit=None, order=None):
    dx = 2.0 / n
    target = n / 2.0
    c_vec = np.correlate(h, 1 - h, mode='full') * dx
    h = h.copy()
    N2 = 2 * n - 1
    
    if order is None:
        range_n = range(n)
    else:
        range_n = order
    
    for i in range_n:
        if time_limit is not None and time.time() > time_limit:
            break
        
        B = np.zeros(N2)
        C = np.zeros(N2)
        
        # Term 1: j=i, contributes to c[m] for m where 0<=i+(n-1-m)<n
        m_lo = max(0, n - 1 - i)
        m_hi = min(N2 - 1, 2 * n - 2 - i)
        
        if m_lo <= m_hi:
            m_range = np.arange(m_lo, m_hi + 1)
            lags = (n - 1) - m_range
            ips = i + lags
            
            nz_mask = (lags != 0)
            if nz_mask.any():
                B[m_range[nz_mask]] += (1.0 - h[ips[nz_mask]]) * dx
            
            z_mask = (lags == 0)
            if z_mask.any():
                m0 = m_range[z_mask][0]
                B[m0] += dx
                C[m0] -= dx
        
        # Term 2: j+lag=i, j=i-(n-1-m), exclude lag=0
        m_lo2 = max(0, n - 1 - i)
        m_hi2 = min(N2 - 1, 2 * n - 2 - i)
        
        if m_lo2 <= m_hi2:
            m_range2 = np.arange(m_lo2, m_hi2 + 1)
            inc = (m_range2 != (n - 1))
            if inc.any():
                js = i - (n - 1 - m_range2[inc])
                B[m_range2[inc]] -= h[js] * dx
        
        t_cur = h[i]
        A = c_vec - B * t_cur - C * t_cur * t_cur
        S = h.sum() - t_cur
        
        t_grid = np.linspace(0.0, 1.0, n_grid)
        vals = (A[None, :] + B[None, :] * t_grid[:, None] +
                C[None, :] * (t_grid[:, None]) ** 2).max(axis=1)
        if mu > 0:
            vals += mu * (S + t_grid - target) ** 2
        
        idx = vals.argmin()
        t_lo = t_grid[max(0, idx - 1)]
        t_hi = t_grid[min(n_grid - 1, idx + 1)]
        
        if t_hi - t_lo > 1e-14:
            def f_1d_coord(t):
                v = (A + B * t + C * t * t).max()
                if mu > 0:
                    v += mu * (S + t - target) ** 2
                return v
            try:
                res = minimize_scalar(f_1d_coord, bounds=(float(t_lo), float(t_hi)),
                                    method='bounded', options={'xatol': 1e-10})
                t_opt = float(res.x)
            except Exception:
                t_opt = float(t_grid[idx])
        else:
            t_opt = float(t_grid[idx])
        
        dt = t_opt - t_cur
        if abs(dt) > 1e-15:
            c_vec += B * dt + C * (t_opt * t_opt - t_cur * t_cur)
            h[i] = t_opt
    
    return h, c_vec

def purge_binary(h, n, threshold=0.001):
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    return project_feasible(h, n)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def lm_smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad_full(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            maxit = max(200, int(rem * 2))

                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)

def fine_pair_search(h, n, deadline, radius=80, deltas=None):

        deltas = [0.001, 0.0005, 0.0002, 0.0001]
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if time.time() > deadline:
                        break
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[i] <= 1.0 - d and best_h[j] >= d:

                        best_h[j] -= d
                        c = compute_c5(best_h, n)

                            improved = True
                        else:

    return best_h, best_c

            order = rng.permutation(cur_n)
            h_cd, _ = coord_descent_sweep(best_h, cur_n, mu=0.0, n_grid=500,
                                         time_limit=p5_end - 20, order=order)
            h_cd = project_feasible(h_cd, cur_n)
            c_cd = compute_c5(h_cd, cur_n)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
        
        # Small mu
        for sweep in range(8):
            if time.time() > p5_end - 10:
                break
            h_cd, _ = coord_descent_sweep(best_h, cur_n, mu=0.02, n_grid=500,
                                         time_limit=p5_end - 10)
            h_cd = project_feasible(h_cd, cur_n)
            c_cd = compute_c5(h_cd, cur_n)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
        
        # Active set polish
        if time.time() < p5_end - 15:
            for top_k, alpha, mu in [(4, 100000, 0.01), (8, 1000000, 0.005),
                                      (12, 5000000, 0.003)]:
                if time.time() > p5_end - 10:
                    break
                h_a, c_a = active_lcg(best_h, cur_n, p5_end - 5, top_k, alpha, mu)
                if c_a < best_c:
                    best_c = c_a
                    best_h = h_a
    
    # Phase 6: Fine pair search
    p6_end = min(time.time() + 100, deadline - 20)
    cur_n = len(best_h)
    if p6_end > time.time() + 15:
        h6, c6 = fine_pair_search(best_h, cur_n, p6_end - 5, radius=50,
                                  deltas=[0.0005, 0.0002, 0.0001, 0.00005])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final CD polish
    if time.time() < deadline - 5:
        cur_n = len(best_h)
        h_cd, _ = coord_descent_sweep(best_h, cur_n, mu=0.0, n_grid=600,
                                     time_limit=deadline - 3)
        h_cd = project_feasible(h_cd, cur_n)
        c_cd = compute_c5(h_cd, cur_n)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Final
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
