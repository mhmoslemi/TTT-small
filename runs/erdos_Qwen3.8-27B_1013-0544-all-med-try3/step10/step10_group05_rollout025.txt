def project_feasible(h,n):

    h = np.array(h,float)
    if not np.isfinite all: h=np.full(n,0.5)

       s=h.sum()
       d=target-s
       if abs(d) < 1e-12*max(1,target): break
       if d>0:
          room = 1-h; tot=room.sum()
          if tot <= 1e-15: break
          h += (d/tot)*room
       else:
          tot=s
          if tot <=1e-15: break
          h += (d/tot)*h
    h=np.clip(h,0,1)
    # final correction on non-bound
    d=target-h.sum()
    if abs(d) > 1e-12:
       free=(h>1e-12)&(h<1-1e-12)
       nf=free.sum()
       if nf>0:
          h[free] += d/nf
          h=np.clip(h,0,1)

          # adjust max/min

          else: idx=int(np.argmax(h)); h[idx]=max(0,h[idx]+d)

def _smooth_values(c, alpha):

import numpy as np
from scipy.optimize import minimize
import time


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    if h.shape[0] != n:
        if h.size == n:
            h = h.reshape(n)
        else:
            h = np.resize(h, n)

    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)

    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (d / tot) * h

    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        d = target - float(h.sum())
        if abs(d) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = int(free.sum())
        if nf > 0:
            h[free] += d / float(nf)
        else:
            if d > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + d)
        h = np.clip(h, 0.0, 1.0)

    return h


def upsample_h(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, int(n_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, int(n_new), endpoint=False)
    h_new = np.interp(x_new, x_old, np.asarray(h, dtype=float))
    return project_feasible(h_new, n_new)


def make_default_initial(n, rng):
    h = rng.uniform(0.0, 1.0, size=int(n))
    return project_feasible(h, n)


def compute_c_raw_fft(h, n):
    h = np.asarray(h, dtype=float)
    need = 2 * int(n) - 1
    L = next_power_of_two(need)

    H = np.fft.rfft(h, L)
    ac = np.fft.irfft(H * np.conjugate(H), L)[:n]
    ac = np.real(ac)

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    s = prefix[-1]

    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1:] = prefix[n:0:-1] - ac
    c[n - 2::-1] = (s - prefix[1:]) - ac[1:]
    np.maximum(c, 0.0, out=c)
    return c


def compute_c5_fast(h, n):
    c = compute_c_raw_fft(h, n)
    return float(c.max() * (2.0 / float(n)))


def compute_c5_exact(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def smooth_raw_stats(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e100, np.zeros_like(c, dtype=float)

    if alpha <= 1e-15:
        w = np.full(c.shape, 1.0 / float(c.size), dtype=float)
        return m, w

    sh = alpha * (c - m)
    np.minimum(sh, 0.0, out=sh)
    np.maximum(sh, -700.0, out=sh)

    e = np.exp(sh)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        val = m
        if not np.isfinite(val):
            val = 1e100
        return val, w

    w = e / s
    val = m + np.log(s) / alpha
    if not np.isfinite(val):
        val = 1e100
    return float(val), w


def grad_fft_weights(h, n, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    grad = np.zeros(n, dtype=float)

    center = n - 1
    p0 = float(w[center])
    if p0 > 0.0:
        grad += p0 * (1.0 - 2.0 * h)

    p_pos = np.zeros(n, dtype=float)
    p_neg = np.zeros(n, dtype=float)
    p_pos[1:] = w[n:]
    p_neg[1:] = w[n - 2::-1]

    pos_sum = float(p_pos.sum())
    if pos_sum > 1e-300:
        pos_lin = np.zeros(n, dtype=float)
        pos_lin[:-1] = np.cumsum(p_pos[1:])[::-1]
        grad += pos_lin

    neg_sum = float(p_neg.sum())
    if neg_sum > 1e-300:
        grad += np.cumsum(p_neg)

    A = p_pos + p_neg
    if float(A.sum()) > 1e-300:
        L = next_power_of_two(2 * n - 1)
        H = np.fft.rfft(h, L)

        Aarr = np.zeros(L, dtype=float)
        Aarr[1:n] = A[1:]
        Afft = np.fft.rfft(Aarr, L)

        conv1 = np.fft.irfft(H * np.conjugate(Afft), L)[:n]
        conv2 = np.fft.irfft(H * Afft, L)[:n]
        grad -= conv1 + conv2

    return grad


def grad_from_weights(h, n, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)

    active = np.where(w > 1e-13)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(w))], dtype=int)

    direct_limit = max(64, int(n) // 64)
    if active.size <= direct_limit:
        grad = np.zeros(n, dtype=float)
        center = n - 1
        for idx in active:
            wi = float(w[idx])
            if wi <= 0.0:
                continue
            k = int(idx) - center
            if k == 0:
                grad += wi * (1.0 - 2.0 * h)
            else:
                r = abs(k)
                m = n - r
                if k > 0:
                    grad[:m] += wi * (1.0 - h[r:])
                    grad[r:] += -wi * h[:m]
                else:
                    grad[:m] += -wi * h[r:]
                    grad[r:] += wi * (1.0 - h[:m])
        return grad

    return grad_fft_weights(h, n, w)


def obj_raw(x, n, alpha_c, mu):
    h = np.asarray(x, dtype=float)
    if h.shape[0] != int(n) or not np.all(np.isfinite(h)):
        return 1e100

    c = compute_c_raw_fft(h, n)
    alpha = float(alpha_c) * (2.0 / float(n))
    val, _ = smooth_raw_stats(c, alpha)

    if mu != 0.0:
        d = float(h.sum()) - 0.5 * float(n)
        val += float(mu) * d * d

    if not np.isfinite(val):
        return 1e100
    return float(val)


def grad_raw(x, n, alpha_c, mu):
    h = np.asarray(x, dtype=float)
    if h.shape[0] != int(n) or not np.all(np.isfinite(h)):
        return np.zeros(int(n), dtype=float)

    c = compute_c_raw_fft(h, n)
    alpha = float(alpha_c) * (2.0 / float(n))
    _, w = smooth_raw_stats(c, alpha)

    grad = grad_from_weights(h, n, w)
    if mu != 0.0:
        grad += 2.0 * float(mu) * (float(h.sum()) - 0.5 * float(n))

    return grad


def alternative_start_candidates(h, n, rng):
    h = project_feasible(h, n)
    cands = [
        h.copy(),
        np.ascontiguousarray(h[::-1]).copy(),
        project_feasible(1.0 - h, n),
        project_feasible(1.0 - np.ascontiguousarray(h[::-1]), n),
    ]

    shifts = [
        n // 8,
        -n // 8,
        n // 4,
        -n // 4,
        n // 5,
        -n // 5,
        5 * n // 8,
    ]
    for s in shifts:
        cands.append(np.roll(h, int(s)).copy())
        cands.append(np.roll(np.ascontiguousarray(h[::-1]), int(s)).copy())

    for scale in (0.008, 0.02):
        cand = h + rng.normal(0.0, scale, size=n)
        cand -= cand.mean()
        cands.append(project_feasible(cand, n))

    return cands


def choose_best_start(h, n, rng):
    best_h = project_feasible(h, n)
    best_c = compute_c5_fast(best_h, n)

    for cand in alternative_start_candidates(best_h, n, rng):
        cand = project_feasible(cand, n)
        c = compute_c5_fast(cand, n)
        if c < best_c - 1e-14:
            best_h = cand
            best_c = c

    return best_h, best_c


def active_subgradient(h, n, tol_c, top_k):
    c = compute_c_raw_fft(h, n)
    max_raw = float(c.max())
    dx = 2.0 / float(n)
    thr = max_raw - max(1e-10, float(tol_c) * (0.5 * n))

    active = np.where(c >= thr)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=int)
    elif active.size > top_k:
        vals = c[active]
        part = np.argpartition(-vals, top_k - 1)[:top_k]
        active = active[part]

    w = np.zeros(c.shape, dtype=float)
    w[active] = 1.0 / float(active.size)
    grad = grad_from_weights(h, n, w)
    return grad, max_raw * dx


def projected_gradient_polish(h, n, end, tol_c=1e-5, top_k=96):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5_fast(best_h, n)

    no_improve = 0
    while time.time() < end - 8.0:
        grad, _ = active_subgradient(best_h, n, tol_c, top_k)
        grad -= float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break

        d = -grad / gnorm
        max_step = 0.045

        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            max_step = min(max_step, float(np.min((1.0 - best_h[pos]) / d[pos])))
        if np.any(neg):
            max_step = min(max_step, float(np.min(best_h[neg] / (-d[neg]))))

        if not np.isfinite(max_step) or max_step <= 1e-13:
            break

        improved = False
        fracs = (1.0, 0.7, 0.45, 0.25, 0.12, 0.06, 0.03, 0.012)
        for frac in fracs:
            if time.time() > end:
                break
            hp = project_feasible(best_h + frac * max_step * d, n)
            c = compute_c5_fast(hp, n)
            if c < best_c - 1e-14:
                best_h = hp
                best_c = c
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_h, best_c


def lbfgs_once(x0, n, alpha_c, mu, maxit, maxfun, bounds):
    try:
        return minimize(
            obj_raw,
            x0,
            jac=grad_raw,
            args=(n, alpha_c, mu),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxit),
                "ftol": 1e-14,
                "gtol": 1e-8,
                "maxfun": int(maxfun),
                "maxls": 40,
                "maxcor": 40,
            },
        )
    except Exception:
        return None


def objective_for_round(h, n, exact_threshold_n=2048):
    if int(n) <= exact_threshold_n:
        try:
            return compute_c5_exact(h, n)
        except Exception:
            return compute_c5_fast(h, n)
    return compute_c5_fast(h, n)


def optimize_at(h_in, n, end, rng):
    n = int(n)
    if n < 8:
        return h_in, compute_c5_fast(h_in, n)

    best_h, best_c = choose_best_start(h_in, n, rng)
    bounds = [(0.0, 1.0)] * n

    alphas = [1.0e3, 1.0e4, 1.0e5, 1.0e6, 1.0e7]
    if n >= 4096:
        alphas = [1.0e3, 1.0e4, 1.0e5, 1.0e6]
    if n >= 8192:
        alphas = [1.0e4, 1.0e5, 1.0e6]

    for i, alpha_c in enumerate(alphas):
        rem = float(end - time.time())
        if rem < 45.0:
            break

        mu = 0.10 if i < 2 else 0.03
        maxit = int(min(1200, max(120, int(rem * 2.0))))
        if n >= 4096:
            maxit = int(min(500, max(90, int(rem * 0.8))))
            maxfun = int(max(400, maxit * 4))
        elif n >= 8192:
            maxit = int(min(260, max(60, int(rem * 0.45))))
            maxfun = int(max(300, maxit * 4))
        else:
            maxfun = int(max(1000, maxit * 6))

        if maxit < 20:
            break

        res = lbfgs_once(best_h, n, alpha_c, mu, maxit, maxfun, bounds)
        if res is not None and hasattr(res, "x"):
            h2 = project_feasible(res.x, n)
            c2 = objective_for_round(h2, n)
            if c2 < best_c - 1e-15:
                best_h = h2
                best_c = c2

    rem = float(end - time.time())
    if rem > 150.0 and n <= 4096:
        for scale in (0.006, 0.003):
            rem = float(end - time.time())
            if rem < 90.0:
                break
            hp = project_feasible(best_h + rng.normal(0.0, scale, size=n), n)
            cp = compute_c5_fast(hp, n)
            if cp < best_c + 0.002:
                alpha_c = 1.0e5 if rem > 180.0 else 1.0e4
                maxit = int(min(420, max(80, int(rem * 0.55))))
                maxfun = int(max(500, maxit * 4))
                res = lbfgs_once(hp, n, alpha_c, 0.05, maxit, maxfun, bounds)
                if res is not None and hasattr(res, "x"):
                    h2 = project_feasible(res.x, n)
                    c2 = objective_for_round(h2, n)
                    if c2 < best_c - 1e-15:
                        best_h = h2
                        best_c = c2

    rem = float(end - time.time())
    if rem > 50.0:
        pg_end = time.time() + max(25.0, min(180.0, rem * 0.45))
        if pg_end > end:
            pg_end = end
        h_pg, c_pg = projected_gradient_polish(best_h, n, pg_end, tol_c=1e-5, top_k=96)
        if c_pg < best_c - 1e-14:
            c_pg_exact = objective_for_round(h_pg, n)
            if c_pg_exact < best_c - 1e-15:
                best_h = h_pg
                best_c = c_pg_exact

    return best_h, best_c


def update_best(h, n, deadline, best_h, best_c, best_n):
    n = int(n)
    if n < 8:
        return best_h, best_c, best_n

    h = project_feasible(h, n)
    use_exact = (n <= 4096) or (time.time() < deadline - 25.0 and n <= 10000)
    try:
        if use_exact:
            c = compute_c5_exact(h, n)
        else:
            c = compute_c5_fast(h, n) + 1e-12
    except Exception:
        c = compute_c5_fast(h, n) + 1e-12

    if c < best_c - 1e-15:
        return h, float(c), n
    return best_h, best_c, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    if budget_s > 20.0:
        deadline = t0 + budget_s - 6.0
    else:
        deadline = t0 + max(budget_s - 0.5, 2.0)

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 1024
        h0 = make_default_initial(n0, rng)

    if n0 < 8:
        n0 = 1024
        h0 = make_default_initial(n0, rng)

    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = compute_c5_exact(best_h, best_n) if best_n <= 5000 else compute_c5_fast(best_h, best_n)

    if best_n <= 2048 and deadline - time.time() > 550.0:
        parent_end = min(deadline - 700.0, t0 + 75.0)
        if parent_end - time.time() > 30.0:
            h_opt, _ = optimize_at(best_h, best_n, parent_end, rng)
            best_h, best_c, best_n = update_best(h_opt, best_n, deadline, best_h, best_c, best_n)

    stage_plan = [
        (1024, 230.0),
        (2048, 250.0),
        (4096, 255.0),
        (8192, 160.0),
    ]

    for target_n, dur in stage_plan:
        if time.time() > deadline - 130.0:
            break
        if int(target_n) <= best_n:
            continue

        start_h = upsample_h(best_h, best_n, target_n)
        early_candidates = [start_h]
        for s in (target_n // 4, -target_n // 4, target_n // 3):
            early_candidates.append(np.roll(start_h, int(s)).copy())
        for cand in early_candidates:
            cand = project_feasible(cand, target_n)
            c = compute_c5_fast(cand, target_n)
            if compute_c5_fast(start_h, target_n) is not None and c < compute_c5_fast(start_h, target_n) - 1e-12:
                start_h = cand
                break

        stage_end = min(deadline - 110.0, time.time() + dur)
        if stage_end - time.time() < 45.0:
            continue

        h_opt, _ = optimize_at(start_h, target_n, stage_end, rng)
        best_h, best_c, best_n = update_best(h_opt, target_n, deadline, best_h, best_c, best_n)

    if time.time() < deadline - 45.0 and best_n <= 4096:
        final_end = min(deadline - 20.0, time.time() + max(30.0, min(90.0, deadline - time.time())))
        if final_end - time.time() > 25.0:
            h_opt, _ = optimize_at(best_h, best_n, final_end, rng)
            best_h, best_c, best_n = update_best(h_opt, best_n, deadline, best_h, best_c, best_n)

    if time.time() < deadline - 15.0:
        pg_end = min(deadline - 2.0, time.time() + max(15.0, (deadline - time.time()) * 0.35))
        h_pg, c_pg = projected_gradient_polish(best_h, best_n, pg_end, tol_c=5e-6, top_k=128)
        best_h, best_c, best_n = update_best(h_pg, best_n, deadline, best_h, best_c, best_n)

    best_h = project_feasible(best_h, best_n)

    if time.time() < deadline - 1.5 and best_n <= 8192:
        try:
            c_final = compute_c5_exact(best_h, best_n)
        except Exception:
            c_final = compute_c5_fast(best_h, best_n) + 1e-11
    else:
        c_final = compute_c5_fast(best_h, best_n) + 1e-11

    return best_h, float(c_final), int(best_n)
