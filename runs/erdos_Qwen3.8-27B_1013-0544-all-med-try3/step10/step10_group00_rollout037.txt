    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_swap(h, n, i, j):
    """Evaluate the objective after swapping h[i] (1→0) and h[j] (0→1)"""
    # c[m] = sum_i h[i] * (1-h[i+m-(n-1)])
    # Change: h[i] goes 1→0, h[j] goes 0→1
    # For the first change (h[i]: 1→0):
    #   c[m] -= (1-h[i+m-(n-1)]) for valid m
    # For the second change (h[j]: 0→1):
    #   c[m] += (1-h[j+m-(n-1)]) for valid m
    # But we need to be careful about the interaction (if i+m-(n-1) == j for some m)
    
    # Actually, the cleanest way is:
    # c_new[m] = sum_i h'[i] * (1-h'[i+m-(n-1)])
    # where h' differs from h only at positions i and j
    
    # The difference c_new - c_old:
    # At position i: was h[i]*(1-h'[i+m-(n-1)]) = 1*(1-h'[i+m-(n-1)])
    #               now h'[i]*(1-h'[i+m-(n-1)]) = 0*(...) = 0
    #               change = -(1-h'[i+m-(n-1)])
    #               But h'[i+m-(n-1)] might be different from h[i+m-(n-1)] if i+m-(n-1) == j
    # At position j: was h[j]*(1-h'[j+m-(n-1)]) = 0*(...) = 0
    #               now h'[j]*(1-h'[j+m-(n-1)]) = 1*(1-h'[j+m-(n-1)])
    #               change = +(1-h'[j+m-(n-1)])
    
    # This is getting complicated. Let me just recompute the full correlation.
    # For n=500, it's fast enough.
    h[j] = 1.0
    h[i] = 0.0
    c = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    h[j] = 0.0
    return c.max() * (2.0 / n)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, max_time, alpha, mu, maxit):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() - t_start > max_time:
            raise TimeoutError()
        return lbfgs_obj(x, n, alpha, mu)
    def grad(x):
        if time.time() - t_start > max_time:
            raise TimeoutError()
        return lbfgs_grad(x, n, alpha, mu)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

        best_improvement = 0
        best_move = None
        
        for i in interior[:50]:  # Limit to 50 for speed
            # Find the best partner j and amount δ
            # Try a few candidates for j
            for j in [(i + n//2) % n, (i + 1) % n, (i - 1) % n, rng.randint(n)]:
                if j == i:
                    continue
                # Transfer δ from i to j (or vice versa)
                # h[i] -= δ, h[j] += δ
                # Optimal δ: minimize max_k c_vec(h[i]-δ, h[j]+δ)
                
                # Grid search over δ
                max_delta = min(h[i], 1.0 - h[j])
                for delta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    if delta > max_delta:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - best_improvement:
                        best_improvement = best_c - c_new
                        best_move = (i, j, delta)
                    h[i] += delta
                    h[j] -= delta
        
        if best_move is None:
            break
        
        i, j, delta = best_move
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0, 1)

        T = t0_temp * np.exp(-(time.time() - start_time) / tau)
        
        i = rng.randint(n)
        j = rng.randint(n)
        attempts = 0
        while i == j and attempts < 10:
            j = rng.randint(n)
            attempts += 1
        if i == j:
            continue
        
        delta = rng.random() * 0.05
        if rng.random() < 0.5:
            amt = min(delta, h[i], 1.0 - h[j])
        else:
            amt = min(delta, h[j], 1.0 - h[i])
            i, j = j, i
        
        if amt < 1e-8:
            continue
        
        h[i] -= amt
        h[j] += amt
        
        new_c = compute_c5(h, n)
        delta_c = new_c - cur_c
        
        if delta_c < 0 or (T > 1e-10 and rng.random() < np.exp(-delta_c / T)):
            cur_c = new_c
