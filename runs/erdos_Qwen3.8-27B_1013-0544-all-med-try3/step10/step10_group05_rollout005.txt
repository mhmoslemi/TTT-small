However, if h is binary, then h[i](1-h[i+k]) is 1 only when h[i]=1 and h[i+k]=0, which IS linear in the binary variables. So for binary h, it becomes an integer program / combinatorial problem.

For the continuous relaxation, it's a quadratic program, which is non-convex (since h(1-h') is not convex in (h, h')).

Let me think about another approach: subgradient method with projection.

Actually, let me just focus on making the L-BFGS + local search work better. The key improvements:

1. Use a very high alpha (like 10^6 or higher) for the smooth max, so it approximates the true max very closely
2. Update the active set after every few iterations
3. Use a tighter constraint (mu larger)
4. Do more iterations of the combinatorial search
5. Try to binarize the solution at the end

Let me write a focused implementation:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    a = alpha
    sh = a * (c - m)
    sm = sh.max()
    if sm > 400:
        return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / a
    else:
        return m + np.log(np.sum(np.exp(sh))) / a

def smooth_max_weights(c, alpha):
    a = alpha
    m = c.max()
    sh = a * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    return w

def c5_gradient(x, n, w, dx):
    """w is weights for each correlation index (length 2n-1)"""
    grad = np.zeros(n)
    nc = n - 1
    for mm in range(len(w)):
        wm = w[mm] * dx
        if wm < 1e-15:
            continue
        k = mm - nc
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    return grad

def obj_and_grad_full(x, n, alpha, mu):
    dx = 2.0 / n
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    w = smooth_max_weights(c, alpha)
    g = c5_gradient(x, n, w, dx)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return val, g

def make_sturmian(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = sorted(set(int(k * alpha_param) % n for k in range(1, 200 * n)))
    for i, pos in enumerate(positions[:n_ones]):
        h[pos] = 1.0
    if h.sum() != n_ones:
        # Fix
        need = n_ones - int(h.sum())
        if need > 0:
            zeros = np.where(h == 0)[0]
            h[zeros[:need]] = 1.0
        elif need < 0:
            ones = np.where(h == 1)[0]
            h[ones[-need:]] = 0.0
    return h

def lgbfs_round(h, n, deadline, alpha, mu, top_k=None, maxiter=2000):
    """One round of L-BFGS-B optimization."""
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    
    if top_k is not None:
        # Active set
        threshold = max_val - 0.01 * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        
        def obj(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            v = smooth_max_val(vals, alpha)
            s = x.sum() - n / 2.0
            return v + mu * s * s
        
        def objg(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            v = smooth_max_val(vals, alpha)
            m = vals.max()
            sh = alpha * (vals - m)
            smx = sh.max()
            w_act = np.exp(sh - smx)
            w_act /= w_act.sum()
            # Map back to full weights
            w_full = np.zeros(len(c))
            w_full[active] = w_act
            g = c5_gradient(x, n, w_full, 2.0/n)
            s = x.sum() - n / 2.0
            g += 2.0 * mu * s
            return v + mu * s * s, g
        
        res = minimize(objg, h, jac=True, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxiter * 10,
                               'maxls': 50, 'maxcor': 50})
    else:
        def obj(x):
            return obj_and_grad_full(x, n, alpha, mu)[0]
        
        res = minimize(obj_and_grad_full, h, jac=True, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxiter * 10,
                               'maxls': 50, 'maxcor': 50})
    
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def local_swap_search(h, n, deadline, n_rounds=200):
    """Targeted local search: for the worst lag, try swaps that reduce it."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Find top few worst lags
        top_lags = np.argsort(-c_vec)[:5]
        
        improved = False
        for li in range(len(top_lags)):
            if time.time() > deadline:
                break
            mm = int(top_lags[li])
            k = mm - (n - 1)
            
            if k >= 0:
                # c[k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
                # To reduce: find i where h[i]=1, h[i+k]=0 and try to swap
                # Also: find j where h[j]=0, h[j+k]=1 and try to fix
                n_pairs = n - k
                # Find the most "wasteful" pairs
                waste = best_h[:n_pairs] * (1.0 - best_h[k:])
                top_waste_idx = np.argsort(-waste)[:50]
                for ti in top_waste_idx:
                    if time.time() > deadline:
                        break
                    i = int(ti)
                    # Option 1: reduce h[i], increase h[i+k]
                    for d in [1.0, 0.5, 0.25]:
                        if best_h[i] >= d - 1e-12 and best_h[i+k] + d <= 1 + 1e-12:
                            best_h[i] -= d
                            best_h[i+k] += d
                            best_h = np.clip(best_h, 0, 1)
                            best_h = project_feasible(best_h, n)
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                                break
                            else:
                                best_h[i] += d
                                best_h[i+k] -= d
                                best_h = np.clip(best_h, 0, 1)
                        if improved:
                            break
                    if improved:
                        break
            else:
                kk = -k
                n_pairs = n - kk
                waste = best_h[kk:] * (1.0 - best_h[:n_pairs])
                top_waste_idx = np.argsort(-waste)[:50]
                for ti in top_waste_idx:
                    if time.time() > deadline:
                        break
                    i = int(ti) + kk
                    for d in [1.0, 0.5, 0.25]:
                        if best_h[i] >= d - 1e-12 and best_h[i-kk] + d <= 1 + 1e-12:
                            best_h[i] -= d
                            best_h[i-kk] += d
                            best_h = np.clip(best_h, 0, 1)
                            best_h = project_feasible(best_h, n)
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-16:
                                best_c = c_new
                                improved = True
                                break
                            else:
                                best_h[i] += d
                                best_h[i-kk] -= d
                                best_h = np.clip(best_h, 0, 1)
                        if improved:
                            break
                    if improved:
                        break
        if not improved:
            break
    
    return best_h, best_c
