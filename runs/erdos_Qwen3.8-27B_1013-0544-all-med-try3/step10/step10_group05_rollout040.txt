from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_for_coord(h, i, n):
    """Compute B_k for all k, for coordinate i.
    Returns array of length 2n-1 where B[k+n-1] is the coefficient
    of h[i] in I(k)."""
    dx = 2.0 / n
    B = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        if k >= 0:
            b = 0.0
            if k <= i:
                b += (1.0 - h[i - k])
            if i <= n - 1 - k:
                b -= h[i + k]
            B[k + n - 1] = b * dx
        else:
            kk = -k
            b = 0.0
            if i <= n - 1 - kk:
                b += (1.0 - h[i + kk])
            if kk <= i:
                b -= h[i - kk]
            B[k + n - 1] = b * dx
    return B

def cd_solve_1d(I_vals, B, h_i, mu, sum_h, target):
    """Solve 1D subproblem: min_{delta} max_k(I(k) + B_k*delta) + mu*(sum_h + delta - target)^2
    subject to h_i + delta in [0,1].
    """
    lo = -h_i
    hi = 1.0 - h_i
    
    for _ in range(80):
        if hi - lo < 1e-15:
            break
        mid = (lo + hi) / 2
        # Compute right derivative of the objective at mid
        vals = I_vals + B * mid
        max_val = vals.max()
        # Right derivative of max is the max B among active constraints
        active = vals >= max_val - 1e-14
        slope_max = B[active].max()
        # Derivative of penalty term
        penalty_der = 2.0 * mu * (sum_h + mid - target)
        total_der = slope_max + penalty_der
        if total_der > 0:
            hi = mid
        else:
            lo = mid
    
    return h_i + (lo + hi) / 2

def coordinate_descent(h, n, max_sweeps=50, mu=0.1, order='sequential'):
    """Run coordinate descent with penalty for sum constraint."""
    dx = 2.0 / n
    I_vals = np.correlate(h, 1 - h, mode='full') * dx
    target = n / 2.0
    best_c = I_vals.max()
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        if order == 'random':
            idx_order = np.random.permutation(n)
        else:
            idx_order = range(n)
        
        for i in idx_order:
            B = compute_B_for_coord(h, i, n)
            t_new = cd_solve_1d(I_vals, B, h[i], mu, h.sum(), target)
            delta = t_new - h[i]
            if abs(delta) > 1e-12:
                h[i] = t_new
                I_vals += B * delta
                improved = True
        
        # Project to maintain feasibility
        h = project_feasible(h, n)
        I_vals = np.correlate(h, 1 - h, mode='full') * dx
        c = I_vals.max()
        if c < best_c:
            best_c = c
        
        if not improved:
            break
    
    return h, best_c

    Shape: (2n-1, n)

        idx = k + n - 1

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Try alternative starts at original resolution
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n_orig, golden / 2.0),
        make_beatty(n_orig, np.sqrt(2.0) / 2.0),
        make_beatty(n_orig, 0.4312),
    ]
    for _ in range(2):
        alt_starts.append(make_random_binary(n_orig, rng))
    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Choose target resolution based on time budget
    if budget_s >= 800:
        n_target = 2000
    elif budget_s >= 500:
        n_target = 1500
    else:
        n_target = 1000
    
    # Upsample to higher resolution
    h_up = upsample_h(best_h, best_n, n_target)
    best_h = h_up
    best_n = n_target
    best_c = compute_c5(best_h, best_n)
    
    # Phase 1: Coordinate descent (main optimization workhorse)
    p1_end = min(t0 + int(budget_s * 0.45), deadline - int(budget_s * 0.45))
    h_cd = best_h.copy()
    
    # Multiple CD rounds with random restarts for diversification
    cd_configs = [
        (40, False),   # 40 sweeps, sequential
        (40, True),    # 40 sweeps, random order
        (30, False),
        (30, True),
        (30, False),
    ]
    
    for sweeps, rand in cd_configs:
        if time.time() > p1_end - 30:
            break
        h_cd, c_cd = coordinate_descent(h_cd, best_n, max_sweeps=sweeps,
                                         random_order=rand, rng=rng)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd.copy()
        elif c_cd > best_c + 0.001:
            # CD got worse, reset
            h_cd = best_h.copy()
    
    # Continue CD for more sweeps
    if time.time() < p1_end - 20:
        for _ in range(5):
            if time.time() > p1_end - 15:
                break
            h_cd, c_cd = coordinate_descent(best_h, best_n, max_sweeps=40,
                                             random_order=True, rng=rng)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd.copy()
    
    # Phase 2: L-BFGS active set optimization
    p2_end = min(time.time() + int(budget_s * 0.2), deadline - int(budget_s * 0.3))
    if p2_end > time.time() + 30:
        cur_n = best_n
        configs = [
            (4, 50000, 0.002),
            (6, 200000, 0.001),
            (8, 500000, 0.001),
            (10, 2000000, 0.0005),
            (12, 5000000, 0.0003),
            (16, 10000000, 0.0003),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 20:
                break
            for _ in range(2):
                if time.time() > p2_end - 10:
                    break
                h2, c2 = run_active_round(best_h, cur_n, p2_end - 5, top_k, alpha, mu)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                elif c2 < best_c + 0.0005:
                    # After L-BFGS, do a few more CD sweeps to polish
                    if time.time() < p2_end - 15:
                        h2b, c2b = coordinate_descent(h2, cur_n, max_sweeps=15,
                                                       random_order=True, rng=rng)
                        if c2b < best_c:
                            best_c = c2b
                            best_h = h2b
    # Phase 3: Directed pair search
    p3_end = min(time.time() + int(budget_s * 0.12), deadline - int(budget_s * 0.15))
    if p3_end > time.time() + 20:
        cur_n = len(best_h)
        h3, c3 = directed_pair_search(best_h, cur_n, p3_end, delta=0.002)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        # Follow up with CD
        if time.time() < p3_end - 10 and c3 <= best_c + 0.001:
            h3b, c3b = coordinate_descent(best_h, cur_n, max_sweeps=20,
                                           random_order=True, rng=rng)
            if c3b < best_c:
                best_c = c3b
                best_h = h3b
    
    # Phase 4: More CD + L-BFGS interleaving
    p4_end = min(time.time() + int(budget_s * 0.1), deadline - int(budget_s * 0.1))
    if p4_end > time.time() + 20:
        for _ in range(3):
            if time.time() > p4_end - 20:
                break
            h4, c4 = coordinate_descent(best_h, best_n, max_sweeps=30,
                                         random_order=True, rng=rng)
            if c4 < best_c:
                best_c = c4
                best_h = h4
            h4b, c4b = run_active_round(best_h, best_n, p4_end - 10, 8, 5000000, 0.0003)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b
    
    # Phase 5: Fine pair search
    p5_end = min(time.time() + int(budget_s * 0.08), deadline - 30)
    if p5_end > time.time() + 15:
        cur_n = len(best_h)
        h5, c5 = fine_pair_search(best_h, cur_n, p5_end, radius=30,
            deltas=[0.001, 0.0005, 0.0002, 0.0001])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Final CD polish
    p6_end = deadline - 3
    if p6_end > time.time() + 10:
        cur_n = len(best_h)
        h6, c6 = coordinate_descent(best_h, cur_n, max_sweeps=25,
                                     random_order=True, rng=rng)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Ultra-fine pair search
    p7_end = deadline - 1
    if p7_end > time.time() + 5:
        cur_n = len(best_h)
        h7, c7 = fine_pair_search(best_h, cur_n, p7_end, radius=20,
            deltas=[0.0001, 0.00005, 0.00002])
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
