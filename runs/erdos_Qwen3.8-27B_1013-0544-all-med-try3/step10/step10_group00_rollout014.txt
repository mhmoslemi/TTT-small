def make_beatty(n, alpha):
    """Create initial binary solution using Beatty sequence."""
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    # Fix sum if needed
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for idx in zeros[:int(np.ceil(diff))]:
                h[idx] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for idx in ones[:int(np.ceil(-diff))]:
                h[idx] = 0.0
    return h

def multistart_sa(h_init, n, deadline, rng, num_starts=3, sa_time_frac=0.6):

    best_n = n_parent
    
    resolutions = [n_parent, 500, 400, 600, 700]
    resolutions = list(dict.fromkeys([r for r in resolutions if 100 <= r <= 2000 and r % 2 == 0]))
    
    # Phase 1: L-BFGS on parent (80s)
    p1_deadline = min(t0 + 80, deadline - 300)
    h1, c1 = run_lbfgs(best_h, best_n, p1_deadline, alpha=50000, mu=0.01, maxit=3000)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    h1b, c1b = run_lbfgs(best_h, best_n, min(p1_deadline + 20, deadline - 280), alpha=200000, mu=0.001, maxit=2000)
    if c1b < best_c:
        best_c = c1b
        best_h = h1b
    
    # Phase 2: SA + Greedy at parent resolution (250s)
    p2_deadline = min(t0 + 330, deadline - 200)
    h_bin = binarize(best_h, best_n)
    
    h2a, c2a = sa_binary(h_bin, best_n, min(p2_deadline, time.time() + 120), rng,
                         T0=10.0, T_end=0.1)
    if time.time() < p2_deadline - 5:
        h2b, c2b = greedy_binary_search(h2a, best_n, p2_deadline, max_iters=100000, rng=rng)
        if c2b < c2a:
            h2a, c2a = h2b, c2b
    
    if c2a < best_c:
        best_c = c2a
        best_h = h2a.copy()
    
    # Phase 3: Multi-start SA (200s)
    p3_deadline = min(t0 + 530, deadline - 150)
    if time.time() < p3_deadline:
        h_init3 = binarize(best_h, best_n)
        h3, c3 = multistart_sa(h_init3, best_n, p3_deadline, rng, num_starts=3, sa_time_frac=0.65)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    
    # Phase 4: Try other resolutions (150s)
    p4_deadline = min(t0 + 680, deadline - 120)
    for n_try in resolutions:
        if time.time() > p4_deadline - 30:
            break
        if n_try == best_n:
            continue
        
        # Initialize for this resolution
        h_try = make_random(n_try, rng)
        # Also try beatty
        h_beatty = make_beatty(n_try, 0.6180339887)
        
        # Quick SA
        sa_d = min(p4_deadline, time.time() + 40)
        h_sa, c_sa = sa_binary(h_try, n_try, sa_d, rng, T0=10.0, T_end=0.2)
        
        h_sa_b, c_sa_b = sa_binary(h_beatty, n_try, min(p4_deadline, time.time() + 20), rng,
                                   T0=5.0, T_end=0.2)
        if c_sa_b < c_sa:
            h_sa, c_sa = h_sa_b, c_sa_b
        
        # Quick greedy
        if time.time() < p4_deadline - 5:
            h_g, c_g = greedy_binary_search(h_sa, n_try, min(p4_deadline, time.time() + 20),
                                            max_iters=50000, rng=rng)
            if c_g < c_sa:
                h_sa, c_sa = h_g, c_g
        
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
            best_n = n_try
    
    # Phase 5: Fine-tune non-binary positions (100s)
    p5_deadline = min(t0 + 780, deadline - 60)
    if time.time() < p5_deadline:
        h5, c5 = fine_tune_nonbinary(binarize(best_h, best_n), best_n, p5_deadline, num_positions=20)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: L-BFGS full refinement (80s)
    p6_deadline = min(t0 + 860, deadline - 40)
    if time.time() < p6_deadline:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > p6_deadline - 5:
                break
            h6, c6 = run_lbfgs(best_h, best_n, min(p6_deadline, time.time() + 25),
                              alpha=alpha, mu=0.0005, maxit=2000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Iterative sensitivity fine-tune (60s)
    p7_deadline = min(t0 + 920, deadline - 30)
    if time.time() < p7_deadline:
        h7, c7 = sensitivity_fine_tune(best_h, best_n, p7_deadline)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: One more round of SA + greedy if time (50s)
    p8_deadline = min(t0 + 970, deadline - 20)
    if time.time() < p8_deadline - 10:
        h_bin8 = binarize(best_h, best_n)
        h8a, c8a = sa_binary(h_bin8, best_n, min(p8_deadline, time.time() + 25), rng,
                             T0=3.0, T_end=0.05)
        if time.time() < p8_deadline - 3:
            h8b, c8b = greedy_binary_search(h8a, best_n, p8_deadline, max_iters=50000, rng=rng)
            if c8b < c8a:
                h8a, c8a = h8b, c8a
            if c8a < best_c:
                best_c = c8a
                best_h = h8a
    
    # Phase 9: Final ultra-fine L-BFGS (30s)
    if time.time() < deadline - 5:
        h9, c9 = run_lbfgs(best_h, best_n, deadline - 2, alpha=5000000, mu=0.0001, maxit=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Phase 10: Final sensitivity tune
    if time.time() < deadline - 2:
        h10, c10 = sensitivity_fine_tune(best_h, best_n, deadline - 1)
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
