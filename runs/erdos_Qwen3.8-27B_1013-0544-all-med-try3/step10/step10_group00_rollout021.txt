def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:

            if tot < 1e-14:
                break
            h += (diff / tot) * h

    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:

            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0

            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def swap_grad_delta(h, i, j, w, n):
    """Compute gradient-based estimate of improvement from swap i(1->0), j(0->1)"""
    # grad[i] > 0 means removing i helps; grad[j] < 0 means adding j helps
    # Improvement ≈ -grad[i] + grad[j] (want this negative)

        if c_tweak < best_c:
            best_c = c_tweak
            best_h = h_tweak
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

def combinatorial_search(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    ones_set = set(ones)
    zeros_set = set(zeros)
    
    no_improve = 0
    max_no_improve = 300
    
    while time.time() < deadline:
        c_vec = np.correlate(h, 1-h, mode='full')
        max_val = c_vec.max()
        dx = 2.0 / n
        c_val = max_val * dx
        
        # Compute weights
        alpha = 200000.0
        sh = alpha * (c_vec - max_val)
        sh_max = sh.max()
        w = np.exp(sh - sh_max)
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for m in range(2*n-1):
            if w[m] < 1e-12:
                continue
            offset = n - 1 - m
            k_min = max(0, -offset)
            k_max = min(n, n - offset) if offset > 0 else n
            # Actually: valid k where 0<=k<n and 0<=k+offset<n
            if offset >= 0:
                k_min = max(0, -offset)
                k_max = min(n, n - offset)
            else:
                k_min = max(0, -offset)
                k_max = min(n, n - offset)
            # Simplify: k_min = max(0, -offset), k_max = min(n, n-offset)
            # This ensures 0 <= k < n and 0 <= k+offset < n
            
            for k in range(k_min, k_max):
                pi = k + offset  # partner index
                grad[k] += w[m] * (1.0 - h[pi])
                grad[pi] += w[m] * (-h[k])
        
        # Select candidates
        one_grads = [(grad[i], i) for i in ones]
        zero_grads = [(grad[j], j) for j in zeros]
        one_grads.sort(reverse=True)
        zero_grads.sort()
        
        top_ones = [i for _, i in one_grads[:40]]
        top_zeros = [j for _, j in zero_grads[:40]]
        
        # Find best swap among top candidates
        best_gain = 0
        best_i = -1
        best_j = -1
        best_new_c = c_val
        
        for i in top_ones:
            if one_grads[[idx for idx, vi in enumerate(one_grads) if vi == i][0]][0] < 1e-8:
                continue
            for j in top_zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.correlate(h, 1-h, mode='full').max() * dx
                h[i] = 1.0
                h[j] = 0.0
                if new_max < best_new_c - 1e-12:
                    best_new_c = new_max
                    best_gain = c_val - new_max
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            best_c = best_new_c
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Shake: random swaps
                num_shake = rng.randint(3, 8)
                for _ in range(num_shake):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                # Evaluate after shake
                new_c = compute_c5(h, n)
                if new_c < best_c:
                    best_c = new_c
                # Re-optimize after shake
                max_no_improve = 300
                no_improve = 0
    
    return h, best_c

I need to speed up the evaluation. Options:
1. **Reduce n**: Use n=300 instead of n=500. Then correlate is O(90K) per call, 1600 calls = 144M ops = ~1.4s per iteration. Still slow.

def eval_swap_incr(h, i, j, c_vec, n):
    """Compute new c_vec after swapping h[i]:1->0, h[j]:0->1. Uses incremental update."""
    new_c = c_vec.copy()
    for m in range(2*n-1):
        offset = n - 1 - m
        delta = 0.0
        # Change at i (1->0):
        pi = i + offset  # i as "first" factor, partner
        if 0 <= pi < n:
            delta -= (1.0 - h[pi])
        qi = i - offset  # i as "second" factor partner, k = i - offset
        if 0 <= qi < n:
            delta -= h[qi]
        # Change at j (0->1):
        pj = j + offset
        if 0 <= pj < n:
            delta += (1.0 - h[pj])
        qj = j - offset
        if 0 <= qj < n:
            delta -= h[qj]
        new_c[m] += delta
    return new_c
