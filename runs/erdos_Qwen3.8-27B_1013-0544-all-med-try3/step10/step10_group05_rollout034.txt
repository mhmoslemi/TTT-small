    For k < 0: sum_{i=-k}^{n-1} h[i]*(1-h[i+k]) = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])  [where k is negative]
    
    More simply, for correlate index m, lag = m - (n-1):
    c[m] = sum_i h[i] * (1-h[i + lag]) where valid indices are used.
    
    When we decrease h[a] by d and increase h[b] by d:
    For each valid i in the sum:
    - If i == a: term changes from h[a]*(1-h[a+lag]) to (h[a]-d)*(1-h[a+lag])
      change = -d*(1-h[a+lag])
    - If i+lag == a (i.e., i = a-lag): term changes from h[a-lag]*(1-h[a]) to h[a-lag]*(1-(h[a]-d))
      change = +d*h[a-lag]
    - If i == b: term changes from h[b]*(1-h[b+lag]) to (h[b]+d)*(1-h[b+lag])
      change = +d*(1-h[b+lag])
    - If i+lag == b (i.e., i = b-lag): term changes from h[b-lag]*(1-h[b]) to h[b-lag]*(1-(h[b]+d))
      change = -d*h[b-lag]
    
    Total change at lag = d * [(1-h[a+lag]) + h[a-lag] - (1-h[b+lag]) - h[b-lag]]
    = d * [h[a+lag] + h[a-lag] - h[b+lag] - h[b-lag]]
    (with out-of-bounds terms = 0)
    """
    changes = np.zeros(len(active_lags))
    for idx, m in enumerate(active_lags):
        lag = m - (n - 1)
        # h[a+lag] if valid
        val_a_plus = h[a + lag] if (0 <= a + lag < n) else 0.0
        # h[a-lag] if valid
        val_a_minus = h[a - lag] if (0 <= a - lag < n) else 0.0
        # h[b+lag] if valid
        val_b_plus = h[b + lag] if (0 <= b + lag < n) else 0.0
        # h[b-lag] if valid
        val_b_minus = h[b - lag] if (0 <= b - lag < n) else 0.0
        
        # But we also need to check that the terms are actually in the valid range of the sum
        # For lag >= 0: valid i range is [0, n-1-lag]
        #   i = a is valid if 0 <= a <= n-1-lag
        #   i+lag = a => i = a-lag is valid if 0 <= a-lag <= n-1-lag
        # For lag < 0: let l = -lag, valid i range is [l, n-1] = [-lag, n-1]
        #   i = a is valid if -lag <= a <= n-1
        #   i+lag = a => i = a-lag = a+|lag| is valid if -lag <= a+|lag| <= n-1 => 0 <= a <= n-1-|lag| = n-1+lag
        
        # Actually, let me just check: is the term h[i]*(1-h[i+lag]) included in the sum?
        # It's included iff 0 <= i < n AND 0 <= i+lag < n
        # So for i = a: included iff 0 <= a < n AND 0 <= a+lag < n (always true for a since 0<=a<n, need 0<=a+lag<n)
        # For i+lag = a (i = a-lag): included iff 0 <= a-lag < n AND 0 <= a < n (the second is always true, need 0<=a-lag<n)
        
        # Similarly for b.
        
        # So the changes are:
        # From h[a] as first factor: valid if 0 <= a+lag < n (and 0<=a<n which is true)
        # From h[a] as second factor (i=a-lag): valid if 0 <= a-lag < n (and 0<=a<n which is true)
        # From h[b] as first factor: valid if 0 <= b+lag < n
        # From h[b] as second factor (i=b-lag): valid if 0 <= b-lag < n
        
        c = 0.0
        if 0 <= a + lag < n:
            c += (1.0 - h[a + lag])  # this is (1-h[a+lag]), the change when h[a] decreases by d
        if 0 <= a - lag < n:
            c += h[a - lag]  # this is h[a-lag], the change when h[a] decreases by d (affects 1-h[a] term)
        if 0 <= b + lag < n:
            c -= (1.0 - h[b + lag])  # negative because h[b] increases
        if 0 <= b - lag < n:
            c -= h[b - lag]
        
        changes[idx] = c  # multiplied by d when used
    
    return changes

        #              (where lag = m - (n-1))

        n_zeros = len(zeros)

            a_contrib = np.zeros(K)
            for idx, m in enumerate(active_list):
                lag = m - (n - 1)
                val = 0.0
                if 0 <= a + lag < n:
                    val += h[a + lag]
                if 0 <= a - lag < n:
                    val += h[a - lag]

                # New c at active lag idx = c[m] + (a_contrib[idx] - b_contrib[idx]) * dx
                # We want max over active lags to decrease

                # If even the best case doesn't improve, skip
                # b_contrib range: each term is 0 or 1, so b_contrib ∈ [0, 2]
                # a_contrib ∈ [0, 2]
                # Change at lag idx = (a_contrib[idx] - b_contrib[idx]) * dx
                # For the max to decrease, we need the change at the max lag to be negative
                # i.e., a_contrib[max_idx] < b_contrib[max_idx]

                if 0 <= b - lag_max < n:
                    b_at_max += h[b - lag_max]

                # Full evaluation
                b_contrib = np.zeros(K)
                for idx, m in enumerate(active_list):
                    lag = m - (n - 1)
                    val = 0.0
                    if 0 <= b + lag < n:
                        val += h[b + lag]
                    if 0 <= b - lag < n:
                        val += h[b - lag]
                    b_contrib[idx] = val
                
                new_c = c_vec[active_list] + (a_contrib - b_contrib) * dx
                new_max = new_c.max()
                # Also check non-active lags (they might become the new max)
                # This is expensive, so we'll be optimistic and check later
                # For now, just check active lags
                
                if new_max < current_max - tol:
                    improvement = current_max - new_max
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_swap = (a, b)
        
        if best_swap is None:
            # Try with a looser criterion
            # Maybe we should look at reducing variance among top lags
            break
        
        a, b = best_swap
        h[a] = 0.0
        h[b] = 1.0
        ones.remove(a)
        zeros.remove(b)
        ones.append(b)
        zeros.append(a)
        
        # Update c_vec
        c_vec = compute_c5_vec(h, n)  # Full recompute for safety
        current_max = c_vec.max()
    
    return h, current_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def make_binary_from_continuous(h, n):
    """Create a binary vector with exactly n//2 ones, prioritizing high h values."""
    n_ones = n // 2
    h_binary = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_binary[sorted_idx[:n_ones]] = 1.0
    return h_binary

def precompute_contrib(h, n, active_indices):
    """
    Precompute contrib[p, i] = h[p + lag_i] + h[p - lag_i] for all positions p
    and all active lag indices.
    Out-of-bounds terms are 0.
    """
    n_active = len(active_indices)
    contrib = np.zeros((n, n_active))
    lags = active_indices - (n - 1)
    for i, lag in enumerate(lags):
        # h[p + lag] for valid p
        if lag >= 0:
            # p + lag < n => p < n - lag
            # p >= 0 (always)
            if n - lag > 0:
                contrib[:n - lag, i] = h[lag:n]
            # h[p - lag]: p - lag >= 0 => p >= lag, p < n (always)
            if lag < n:
                contrib[lag:, i] += h[:n - lag]
        else:
            l = -lag
            # h[p + lag] = h[p - l]: p >= l
            if n - l > 0:
                contrib[l:, i] = h[:n - l]
            # h[p - lag] = h[p + l]: p < n - l
            if n - l > 0:
                contrib[:n - l, i] += h[l:]
    return contrib

    Fast vectorized local search on binary vectors.
    Swaps (1->0 at a, 0->1 at b) to reduce max overlap.

    current_max = float(c_vec.max())
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, current_max
    
    dx = 2.0 / n
    no_improve_count = 0
    
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Find active lags (within tolerance of max)

        active = np.where(c_vec >= current_max - active_tol)[0]
        
        if len(active) == 0:
            break
        
        # Limit active set for speed
        if len(active) > 40:
            idx_sorted = np.argsort(-c_vec[active])
            active = active[idx_sorted[:40]]
        
        # Precompute contributions
        contrib = precompute_contrib(h, n, active)
        # contrib[a, i] = h[a+lag_i] + h[a-lag_i]
        # contrib[b, i] = h[b+lag_i] + h[b-lag_i]
        # Change at active lag i for swap (a,b) = (contrib[a,i] - contrib[b,i]) * dx
        
        active_c = c_vec[active]  # current values at active lags
        
        # For each 1-position a, compute the best 0-position b to swap with
        # changes[a, b, i] = (contrib[a,i] - contrib[b,i]) * dx
        # new_c[b, i] = active_c[i] + changes[a, b, i]
        # new_max[b] = max_i new_c[b, i]
        # We want argmax_b (current_max - new_max[b])
        
        best_total_improvement = 0
        best_a = None
        best_b = None
        
        ones_list = ones.tolist()
        zeros_arr = zeros
        n_zeros = len(zeros_arr)
        
        # Vectorized: for each a, compute new_max for all b simultaneously
        # contrib_zeros = contrib[zeros_arr, :]  shape (n_zeros, K)
        contrib_zeros = contrib[zeros_arr, :]
        
        for a in ones_list:
            contrib_a = contrib[a, :]  # shape (K,)
            # change for all b at all active lags: shape (n_zeros, K)
            changes = (contrib_a[None, :] - contrib_zeros) * dx
            new_c = active_c[None, :] + changes  # shape (n_zeros, K)
            new_maxes = new_c.max(axis=1)  # shape (n_zeros,)
            improvements = current_max - new_maxes
            best_idx = np.argmax(improvements)
            if improvements[best_idx] > best_total_improvement:
                best_total_improvement = improvements[best_idx]
                best_a = a
                best_b = int(zeros_arr[best_idx])
        
        if best_total_improvement < tol:
            no_improve_count += 1
            if no_improve_count > 5:
                break
            # Try with slightly looser active set
            active_tol = max(1e-8, 0.01 * current_max)
            active = np.where(c_vec >= current_max - active_tol)[0]
            if len(active) > 50:
                idx_sorted = np.argsort(-c_vec[active])
                active = active[idx_sorted[:50]]
            if len(active) == 0:
                break
            contrib = precompute_contrib(h, n, active)
            active_c = c_vec[active]
            contrib_zeros = contrib[zeros_arr, :]
            
            best_total_improvement = 0
            for a in ones_list:
                contrib_a = contrib[a, :]
                changes = (contrib_a[None, :] - contrib_zeros) * dx
                new_c = active_c[None, :] + changes
                new_maxes = new_c.max(axis=1)
                improvements = current_max - new_maxes
                best_idx = np.argmax(improvements)
                if improvements[best_idx] > best_total_improvement:
                    best_total_improvement = improvements[best_idx]
                    best_a = a
                    best_b = int(zeros_arr[best_idx])
            
            if best_total_improvement < tol:
                break
        
        # Perform the swap
        h[best_a] = 0.0
        h[best_b] = 1.0
        ones = np.delete(ones, np.where(ones == best_a)[0][0])
        ones = np.append(ones, best_b)
        zeros = np.delete(zeros, np.where(zeros == best_b)[0][0])
        zeros = np.append(zeros, best_a)
        
        # Full recompute (fast in numpy for n=500)
        c_vec = compute_c5_vec(h, n)
        current_max = float(c_vec.max())
        no_improve_count = 0
    
    return h, current_max

def continuous_local_search(h, n, deadline, max_iters=200, step_sizes=None):
    """
    Continuous local search using incremental updates.
    Tries small transfers between pairs of positions.
    """
    h = h.copy()
    c_vec = compute_c5_vec(h, n)
    current_max = float(c_vec.max())
    
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    dx = 2.0 / n
    
    for d in step_sizes:
        if time.time() > deadline:
            break
        
        improved = True
        iter_count = 0
        while improved and iter_count < max_iters:
            improved = False
            iter_count += 1
            if time.time() > deadline:
                break
            
            c_vec = compute_c5_vec(h, n)
            current_max = float(c_vec.max())
            active_tol = max(1e-9, 0.003 * current_max)
            active = np.where(c_vec >= current_max - active_tol)[0]
            if len(active) == 0:
                break
            if len(active) > 30:
                idx_s = np.argsort(-c_vec[active])
                active = active[idx_s[:30]]
            
            active_c = c_vec[active]
            lags = active - (n - 1)
            
            # For each position, compute the "gradient-like" contribution
            # grad_p[i] = (1-h[p+lag_i]) - h[p-lag_i] for active lag i
            # Moving d from a to b: Δg_i = d * [grad_b[i] - grad_a[i]]
            # We want to reduce max(active_c + d*(grad_b - grad_a))
            
            # Precompute grad for all positions
            # grad[p, i] = (1 - h[p+lag_i]) - h[p-lag_i] (with bounds)
            K = len(active)
            grad = np.zeros((n, K))
            for i, lag in enumerate(lags):
                if lag >= 0:
                    if n - lag > 0:
                        grad[:n - lag, i] = (1.0 - h[lag:n])
                    if lag < n:
                        grad[lag:, i] -= h[:n - lag]
                else:
                    l = -lag
                    if n - l > 0:
                        grad[l:, i] = (1.0 - h[:n - l])
                    if n - l > 0:
                        grad[:n - l, i] -= h[l:]
            
            # For swap (a: decrease by d, b: increase by d):
            # Δg_i = d * (grad[b,i] - grad[a,i])
            # new_max = max(active_c + d * (grad[b,:] - grad[a,:]))
            
            # Find position with max value (candidate to decrease) and min value (candidate to increase)
            # Only consider pairs where h[a] >= d and h[b] <= 1-d
            
            # Try top candidates
            high_candidates = np.where(h >= d)[0]
            low_candidates = np.where(h <= 1.0 - d)[0]
            
            if len(high_candidates) == 0 or len(low_candidates) == 0:
                break
            
            # Limit candidates for speed
            if len(high_candidates) > 60:
                high_candidates = high_candidates[np.argsort(-h[high_candidates])[:60]]
            if len(low_candidates) > 60:
                low_candidates = low_candidates[np.argsort(h[low_candidates])[:60]]
            
            grad_low = grad[low_candidates, :]  # (n_low, K)
            
            best_improvement = 0
            best_a = None
            best_b = None
            
            for a in high_candidates:
                grad_a = grad[a, :]  # (K,)
                diff = grad_low - grad_a[None, :]  # (n_low, K)
                new_c = active_c[None, :] + d * diff  # (n_low, K)
                new_maxes = new_c.max(axis=1)
                improvements = current_max - new_maxes
                best_idx = np.argmax(improvements)
                if improvements[best_idx] > best_improvement:
                    best_improvement = improvements[best_idx]
                    best_a = a
                    best_b = low_candidates[best_idx]
            
            if best_improvement > 1e-14:
                h[best_a] -= d
                h[best_b] += d
                h = np.clip(h, 0, 1)
                improved = True
        
        # Re-project
        h = project_feasible(h, n)
        current_max = compute_c5(h, n)
    
    return h, current_max

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_active(h, n, active, alpha, mu, deadline, maxit=2000):
    """L-BFGS-B on active set with smooth max."""
    def obj(x):
        c = compute_c5_vec(x, n)
        vals = c[active]
        return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2
    
    def grad(x):
        c = compute_c5_vec(x, n)
        vals = c[active]
        m = vals.max()
        sh = alpha * (vals - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        idx_center = n - 1
        for wi, idx in zip(w, active):
            if wi < 1e-15:
                continue
            k = int(idx) - idx_center
            wm = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-24,
                                'gtol': 1e-18, 'maxfun': maxit * 6,
                                'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def make_structured_binary(n, pattern_type, rng):
    """Create structured binary patterns."""
    h = np.zeros(n)
    n_ones = n // 2
    if pattern_type == 0:  # Alternating
        h[::2] = 1.0
        if h.sum() > n_ones:
            # Remove excess
            excess = int(h.sum() - n_ones)
            idx = np.where(h == 1)[0]
            to_remove = rng.choice(idx, size=excess, replace=False)
            h[to_remove] = 0
            to_add = np.where(h == 0)[0]
            to_add_idx = rng.choice(to_add, size=excess, replace=False)
            h[to_add_idx] = 1
        elif h.sum() < n_ones:
            deficit = int(n_ones - h.sum())
            to_add = np.where(h == 0)[0]
            to_add_idx = rng.choice(to_add, size=deficit, replace=False)
            h[to_add_idx] = 1
    elif pattern_type == 1:  # Block pattern
        block = n // 4
        h[0:block] = 1.0
        h[2*block:3*block] = 1.0
        if h.sum() != n_ones:
            h = make_random_binary(n, rng)
    else:  # Random
        h = make_random_binary(n, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(1000, 0.5)
        n = 1000
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Phase 1: Binary local search from parent (key innovation)
    p1_end = min(t0 + 250, deadline - 400)
    h_binary = make_binary_from_continuous(h0, n)
    h_bin, c_bin = vectorized_binary_local_search(h_binary, n, p1_end, max_rounds=10000)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Phase 2: Continuous refinement from best binary
    p2_end = min(time.time() + 100, deadline - 300)
    if p2_end > time.time() + 20:
        h_cont = best_h.copy()
        h_cont, c_cont = continuous_local_search(h_cont, n, p2_end, max_iters=100,
                                                   step_sizes=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont
    
    # Phase 3: L-BFGS active set refinement
    p3_end = min(time.time() + 120, deadline - 200)
    if p3_end > time.time() + 15:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - 0.005 * max_val)[0]
        if len(active) > 20:
            active = active[np.argsort(-c_vec[active])[:20]]
        
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p3_end - 10:
                break
            h_lb, c_lb = lbfgs_active(best_h, n, active, alpha, 0.01, p3_end - 5, maxit=1500)
            if c_lb < best_c:
                best_c = c_lb
                best_h = h_lb
                # Update active set
                c_vec = compute_c5_vec(best_h, n)
                max_val = c_vec.max()
                active = np.where(c_vec >= max_val - 0.005 * max_val)[0]
                if len(active) > 20:
                    active = active[np.argsort(-c_vec[active])[:20]]
    
    # Phase 4: More binary local search rounds (different active set widths)
    p4_end = min(time.time() + 200, deadline - 200)
    if p4_end > time.time() + 30:
        # Re-binarize the current best
        h_bin2 = make_binary_from_continuous(best_h, n)
        h_bin2, c_bin2 = vectorized_binary_local_search(h_bin2, n, p4_end, max_rounds=10000)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
        
        # Continuous refinement again
        if time.time() < deadline - 150:
            h_cont2 = best_h.copy()
            h_cont2, c_cont2 = continuous_local_search(h_cont2, n, min(time.time() + 80, deadline - 150),
                                                        max_iters=150,
                                                        step_sizes=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            if c_cont2 < best_c:
                best_c = c_cont2
                best_h = h_cont2
    
    # Phase 5: Multi-start binary search
    p5_end = min(time.time() + 150, deadline - 100)
    if p5_end > time.time() + 30:
        for start_type in range(4):
            if time.time() > p5_end - 20:
                break
            if start_type == 0:
                h_start = make_random_binary(n, rng)
            elif start_type == 1:
                h_start = make_structured_binary(n, 0, rng)
            elif start_type == 2:
                # Perturbed parent binary
                h_start = make_binary_from_continuous(best_h, n)
                # Randomly flip a few pairs
                for _ in range(20):
                    ones = np.where(h_start > 0.5)[0]
                    zeros = np.where(h_start < 0.5)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        a = rng.choice(ones)
                        b = rng.choice(zeros)
                        h_start[a] = 0
                        h_start[b] = 1
            else:
                # Golden ratio pattern
                phi = (1.0 + np.sqrt(5.0)) / 2.0
                h_start = np.zeros(n)
                positions = set()
                for k in range(1, 20 * n):
                    pos = int(k * n / phi) % n
                    positions.add(pos)
                    if len(positions) >= n // 2:
                        break
                for pos in positions:
                    h_start[pos] = 1.0
                if h_start.sum() != n // 2:
                    h_start = make_random_binary(n, rng)
            
            h_try, c_try = vectorized_binary_local_search(h_start, n, min(time.time() + 30, p5_end - 10),
                                                           max_rounds=3000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
    
    # Phase 6: Final continuous fine-tuning
    p6_end = min(time.time() + 80, deadline - 30)
    if p6_end > time.time() + 15:
        h_final = best_h.copy()
        h_final, c_final = continuous_local_search(h_final, n, p6_end, max_iters=200,
                                                     step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Phase 7: Final L-BFGS polish
    p7_end = deadline - 2
    if p7_end > time.time() + 10:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        active = np.where(c_vec >= max_val - 0.001 * max_val)[0]
        if len(active) > 10:
            active = active[np.argsort(-c_vec[active])[:10]]
        if len(active) > 0:
            h_pol, c_pol = lbfgs_active(best_h, n, active, 500000, 0.005, p7_end, maxit=1000)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol
    
    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
