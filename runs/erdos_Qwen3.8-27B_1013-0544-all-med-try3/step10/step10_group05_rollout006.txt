def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])

    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    vals = c[active]

                break
            for _ in range(3):
                if time.time() > p5_end - 8:
                    break
                h5, c5 = run_active_round(best_h, n, p5_end, top_k, alpha, mu)
                if c5 < best_c:
                    best_c = c5
                    best_h = h5

    # Phase 6: Fine coordinate polish (100s)
    p6_end = min(time.time() + 100, deadline - 50)
    if p6_end > time.time() + 15:
        h6, c6 = coord_polish(best_h, n, p6_end, [0.001, 0.0005, 0.0002, 0.0001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final gradient refinement + active set (remaining time)
    p7_end = deadline - 3
    if p7_end > time.time() + 10:
        h7, c7 = gradient_directed_refine(best_h, n, p7_end, n_steps=50000, base_step=0.002)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # One more active set round
        if time.time() < p7_end - 10:
            h7, c7 = run_active_round(best_h, n, p7_end, 10, 50000000, 0.0001)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
