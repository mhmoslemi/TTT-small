import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cc = c - c.max()
    mx = cc.max()
    if mx > 500:
        val = c.max() + (mx + np.log(np.sum(np.exp(cc - mx)))) / alpha
    else:
        val = c.max() + np.log(np.sum(np.exp(cc))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cc = c - c.max()
    mx = cc.max()
    w = np.exp(cc - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)

def subgradient_step(h, n, step_size, active_count=3):
    c = compute_c5_vec(h, n)
    max_val = c.max()
    # Find active shifts (top active_count)
    top = np.argsort(-c)[:active_count]
    avg_c = c[top].mean()
    # Compute average gradient over active shifts
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in top:
        k = m - (n - 1)
        if k >= 0:

            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx

            grad[kk:n] += (1.0 - h[0:nv]) * dx

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def compute_c5_active(h, n, active_indices):
    """Compute C5 only at active shift indices for speed."""
    dx = 2.0 / n
    vals = []
    for m in active_indices:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            vals.append(float(np.sum(h[0:nv] * (1.0 - h[k:n])) * dx))
        else:
            kk = -k
            nv = n - kk
            vals.append(float(np.sum(h[kk:n] * (1.0 - h[0:nv])) * dx))
    return np.array(vals)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) > 0:
                h[idx[0]] += diff
                h = np.clip(h, 0.0, 1.0)
        else:

    return h

    """Gradient of c_m with respect to h, where m is correlate index."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = m - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

    cc = c - c.max()
    mx = cc.max()
    if mx > 300:
        val = c.max() + (mx + np.log(np.sum(np.exp(cc - mx)))) / alpha
    else:
        val = c.max() + np.log(np.sum(np.exp(cc))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def multiplicative_weights_game(h, n, deadline, eta=30.0, max_iter=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Work with n positive shifts (k=0 to n-1), indices n-1 to 2n-2 in correlate
    p = np.ones(n) / n
    bounds = [(0.0, 1.0)] * n

        if time.time() > deadline - 2:
            break

        p_active = p.copy()
        
        def inner_obj(x):
            c = compute_c5_vec(x, n)
            c_pos = c[n-1:2*n-1]
            return float(np.dot(p_active, c_pos))
        
        def inner_grad(x):
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for k in range(n):
                if p_active[k] < 1e-12:
                    continue
                pk_dx = p_active[k] * dx_val
                nv = n - k
                grad[0:nv] += pk_dx * (1.0 - x[k:n])
                grad[k:n] += pk_dx * (-x[0:nv])
            return grad
        
        try:
            res = minimize(inner_obj, h, jac=inner_grad, method='L-BFGS-B',

        h = h_inner.copy()

    h_f = np.interp(x_fine, x_coarse, h_c)
    h_f = project_feasible(h_f, n)
    return h_f

def random_restarts(h, n, deadline, num_restarts=3, noise_scale=0.05):
    """Try small random perturbations to escape local minima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    
    for r in range(num_restarts):
        if time.time() > deadline - 30:
            break
        h_pert = h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Short L-BFGS from perturbation
        t_short = min(40, (deadline - time.time()) * 0.3)
        if t_short > 15:
            h_opt, c_opt = run_lbfgs_full(h_pert, n, time.time() + t_short,
                                           configs=[(2000, 500.0), (10000, 5000.0)])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

        if c3 < best_c:

            best_h = h3.copy()

    # Phase 4: Higher alpha L-BFGS (150s)
