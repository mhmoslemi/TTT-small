        for _ in range(3):
            if time.time() > deadline:
                break
            k_star = top_idx[rng.randint(K)]
            lag2 = k_star - (n - 1)
            if lag2 == 0:
                continue
            m2 = abs(lag2)
            if m2 >= n:
                continue
            i = rng.randint(0, n - m2)
            j = i + m2
            if h[i] > h[j] + 1e-7:
                d = min(0.03, (h[i] - h[j]) * 0.2)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved_this_round = True
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
            elif h[j] > h[i] + 1e-7:
                d = min(0.03, (h[j] - h[i]) * 0.2)
                if d < 1e-10:
                    continue
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved_this_round = True
                else:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)

        if improved_this_round:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_h, best_c

def multi_start_sa(h, n, deadline, rng, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    attempts = n_restarts
    per_attempt = (deadline - time.time()) / max(attempts, 1)

    for attempt in range(attempts):
        if time.time() > deadline:
            break
        t_attempt_end = time.time() + per_attempt * 0.9

        if attempt == 0:
            h_cur = best_h.copy()
        else:
            # Perturb from best
            noise_scale = 0.02 + 0.01 * attempt
            h_cur = best_h + rng.normal(0, noise_scale, n)
            h_cur = project_feasible(h_cur, n)
            # Quick local polish
            h_cur = targeted_local_search(h_cur, n, min(time.time() + 10, t_attempt_end - 10), rng, K=3, max_rounds=3000)
            c_cur = compute_c5(h_cur, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()

        # SA
        h_cur = h_cur.copy()
        c_cur = compute_c5(h_cur, n)
        T = 0.005
        T_min = 0.00001
        cool = 0.99992
        best_local = h_cur.copy()
        best_local_c = c_cur
        stuck = 0

        while time.time() < t_attempt_end and T > T_min:
            r = rng.random()
            if r < 0.5:
                i = rng.randint(0, n)
                j = rng.randint(0, n)
            else:
                c_all = compute_all_lags(h_cur, n)
                top_k = np.argsort(c_all)[-3:]
                k_star = top_k[rng.randint(3)]
                lag = k_star - (n - 1)
                if lag == 0 or abs(lag) >= n:
                    T *= cool
                    continue
                i = rng.randint(0, n - abs(lag))
                j = i + abs(lag)

            if i == j:
                T *= cool
                continue

            max_step = min(0.1, 0.005 + T * 5)
            if h[i] > h[j]:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i]:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue

            if abs(d) < 1e-10:
                T *= cool
                continue

            c_old = compute_c5(h_cur, n)
            h_cur[i] -= d
            h_cur[j] += d
            h_cur = project_feasible(h_cur, n)
            c_new = compute_c5(h_cur, n)

            if c_new <= c_old + T:
                if c_new < best_local_c:
                    best_local_c = c_new
                    best_local = h_cur.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                h_cur[i] += d
                h_cur[j] -= d
                h_cur = project_feasible(h_cur, n)
                stuck += 1

            T *= cool
            if stuck > 10000:
                h_cur = best_local + rng.normal(0, 0.008, n)
                h_cur = project_feasible(h_cur, n)
                T = max(T, 0.003)
                stuck = 0

        if best_local_c < best_c:
            best_c = best_local_c
            best_h = best_local.copy()
        h_cur = best_local

    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def fine_grained_descent(h, n, deadline, rng):
    """Very fine-grained coordinate descent on the maximum lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for phase in range(50):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        max_k = np.argmax(c_all)
        max_lag = max_k - (n - 1)
        if max_lag == 0:
            break
        m = abs(max_lag)
        if m >= n:
            break

        # For this lag, try all possible small transfers
        improved = False
        step_size = 0.002 * (1.0 - phase / 50.0) + 0.0003

        if max_lag > 0:
            for i in range(0, n - m, max(1, (n - m) // 500)):
                if time.time() > deadline:
                    break
                if h[i] <= h[i + m] + 1e-8:
                    continue
                d = min(step_size, (h[i] - h[i + m]) * 0.3)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[i + m] += d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += d
                    h[i + m] -= d
                    h = project_feasible(h, n)
        else:
            idx = -max_lag
            for i in range(0, n - idx, max(1, (n - idx) // 500)):
                if time.time() > deadline:
                    break
                if h[i + idx] <= h[i] + 1e-8:
                    continue
                d = min(step_size, (h[i + idx] - h[i]) * 0.3)
                if d < 1e-10:
                    continue
                h[i + idx] -= d
                h[i] += d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                else:
                    h[i + idx] += d
                    h[i] -= d
                    h = project_feasible(h, n)

        if not improved:
            # Try random small perturbations
            for _ in range(100):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j:
                    continue
                if h[i] > h[j] + 1e-7:
                    d = min(step_size * 2, (h[i] - h[j]) * 0.2)
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-7:
                    d = min(step_size * 2, (h[j] - h[i]) * 0.2)
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
            if not improved:
                break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: Quick L-BFGS-B with smooth-max at current resolution (45s)
    if time.time() < deadline - 900:
        for alpha in [10000, 100000, 1000000]:
            if time.time() > deadline - 900:
                break
            t_phase = min(15, deadline - 900 - time.time())
            if t_phase < 3:
                break
            h_opt = lbfgsb_smooth_max(best_h, n, alpha, time.time() + t_phase, maxiter=1500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 2: Top-k sum optimization (60s)
    if time.time() < deadline - 800:
        for k_val in [3, 5, 8, 12]:
            if time.time() > deadline - 800:
                break
            t_phase = min(15, deadline - 800 - time.time())
            if t_phase < 3:
                break
            h_opt = lbfgsb_top_k(best_h, n, k_val, time.time() + t_phase, maxiter=1500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: Targeted local search (200s)
    if time.time() < deadline - 500:
        t_phase = min(200, deadline - 500 - time.time())
        h_opt, c_opt = targeted_local_search(best_h, n, time.time() + t_phase, rng, K=5, max_rounds=30000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish after local search (30s)
    if time.time() < deadline - 400:
        t_phase = min(20, deadline - 400 - time.time())
        if t_phase > 3:
            h_opt = lbfgsb_smooth_max(best_h, n, 5000000, time.time() + t_phase, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Multi-start simulated annealing (250s)
    if time.time() < deadline - 300:
        t_phase = min(250, deadline - 300 - time.time())
        h_opt, c_opt = multi_start_sa(best_h, n, time.time() + t_phase, rng, n_restarts=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Another targeted local search round (150s)
    if time.time() < deadline - 200:
        t_phase = min(150, deadline - 200 - time.time())
        h_opt, c_opt = targeted_local_search(best_h, n, time.time() + t_phase, rng, K=5, max_rounds=40000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Fine-grained descent (100s)
    if time.time() < deadline - 120:
        t_phase = min(100, deadline - 120 - time.time())
        h_opt, c_opt = fine_grained_descent(best_h, n, time.time() + t_phase, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Try higher resolution n=1000 (120s)
    if time.time() < deadline - 60:
        n_high = 1000
        h_high = expand_to_resolution(best_h, n, n_high)
        t_phase = min(40, deadline - 40 - time.time())
        if t_phase > 10:
            h_high = lbfgsb_smooth_max(h_high, n_high, 50000, time.time() + t_phase, maxiter=2000)
            c_high = compute_c5(h_high, n_high)
            if c_high < best_c:
                best_c = c_high
                best_h = h_high
                n = n_high

        if time.time() < deadline - 20 and len(best_h) == n_high:
            h_opt, c_opt = targeted_local_search(best_h, n_high, deadline - 10, rng, K=4, max_rounds=20000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 9: Final fine-grained polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt, c_opt = fine_grained_descent(best_h, n_final, deadline - 2, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
