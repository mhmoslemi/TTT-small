def simulated_annealing(h_init, n, deadline):
    h = h_init.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    
    best_h = h.copy()
    best_c = cur_c
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = 0.01  # initial temperature (in units of C₅)
    T_end = 1e-6
    alpha = 0.9999  # cooling rate
    
    while time.time() < deadline and T > T_end:
        # Choose a random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        delta = compute_swap_delta(h, i, j, n)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        delta_c = new_c - cur_c
        
        # Metropolis accept
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            # Accept
            h[i], h[j] = 0, 1
            corr = new_corr
            cur_c = new_c
            ones = np.where(h == 1)[0]  # expensive!
            zeros = np.where(h == 0)[0]  # expensive!
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= alpha
    
    return best_h, best_c
```

  → delta[k] -= (1-h[i+k-(n-1)]) for valid k
- In the sum, the term where p+k-(n-1)=i, i.e., p = i-k+(n-1): h[p]*(1-h[i]) changes from h[p]*(1-1)=0 to h[p]*(1-0)=h[p]
  → delta[k] += h[p] = h[i-k+(n-1)] for valid p, k

Change due to h[j]: 0→1:
- In the sum, the term p=j: h[j]*(1-h[j+k-(n-1)]) changes from 0 to 1*(1-h[j+k-(n-1)])
  → delta[k] += (1-h[j+k-(n-1)]) for valid k
- In the sum, the term where p+k-(n-1)=j, i.e., p = j-k+(n-1): h[p]*(1-h[j]) changes from h[p]*(1-0)=h[p] to h[p]*(1-1)=0
  → delta[k] -= h[p] = h[j-k+(n-1)] for valid p, k

So total delta[k] = -(1-h[i+k-(n-1)]) + h[i-k+(n-1)] + (1-h[j+k-(n-1)]) - h[j-k+(n-1)]
(with appropriate bounds: 0 ≤ i+k-(n-1) < n, 0 ≤ i-k+(n-1) < n, etc.)

This can be computed as:

- term2: +h[...] at a specific index offset → affects n positions
- term3: +(1-h[...]) at a specific index offset from j → affects n positions

def compute_delta(h, i, j, n):
    delta = np.zeros(2*n-1)
    # Effect of h[i]: 1->0
    # corr index k corresponds to shift k-(n-1)
    # a part: h[i] goes 1->0, affects corr[k] where the "a" index is i
    #   shift = k-(n-1), need 0 <= i+shift < n, i.e., n-1-i <= k <= 2n-2-i
    idx_a = np.arange(n-1-i, 2n-1-i)  # k values where i+shift is valid
    delta[idx_a] -= 1.0 - h[i + idx_a - (n-1)]
    # b part: 1-h[i] goes 0->1, affects corr[k] where the "b" index is i
    #   shift = k-(n-1), need 0 <= k-shift < n and k-shift+shift = i, so k = i+shift, shift = k-i... 
    #   Actually: p+k-(n-1) = i → p = i-k+(n-1). Need 0 <= p < n.
    #   k = i - p + (n-1) for p in [0,n-1], so k in [i-(n-1)+n-1... wait
    #   k = i - p + n - 1 for p = 0,...,n-1 → k = n-1, n-2, ..., n-1-n+1+i... 
    #   k ranges from i-(n-1) to i, intersected with [0, 2n-2]
    idx_b = np.arange(max(0, i-n+1), min(2*n-1, i+1))
    # For k in idx_b: p = i-k+n-1, and delta[k] += h[p]
    p_vals = i - idx_b + (n-1)
    delta[idx_b] += h[p_vals]
    
    # Effect of h[j]: 0->1
    # a part: h[j] goes 0->1
    idx_a = np.arange(n-1-j, 2n-1-j)
    delta[idx_a] += 1.0 - h[j + idx_a - (n-1)]
    # b part: 1-h[j] goes 1->0
    idx_b = np.arange(max(0, j-n+1), min(2*n-1, j+1))
    p_vals = j - idx_b + (n-1)
    delta[idx_b] -= h[p_vals]
    
    return delta
```
