    if abs(diff) > 1e-10:

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_from_w(w, h, n):
    """Compute gradient of smooth max w.r.t. h"""
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)

        k = mm - (n - 1)

            grad[0:nv] += wm * (1.0 - h[k:n])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def greedy_local_search_binary(h, n, deadline, max_rounds=100000):
    """Greedy best-improvement local search on binary solution."""
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        alpha = 50000.0
        w = smooth_max_w(c_vec, alpha)
        grad = grad_from_w(w, h, n)
        
        # Find best swap
        best_gain = 1e-15
        best_i = -1
        best_j = -1
        
        # Try top candidates
        pos_candidates = sorted(ones, key=lambda x: -grad[x])[:50]
        neg_candidates = sorted(zeros, key=lambda x: grad[x])[:50]
        
        for i in pos_candidates:
            if grad[i] <= 0:
                break
            for j in neg_candidates:
                if grad[j] >= 0:
                    break
                gain = grad[i] - grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            break
    
    return h, best_c

def simulated_annealing_binary(h, n, deadline, rng, T0=None, T1=None, max_iters=2000000):
    """Simulated annealing on binary solution."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h < 0.5)[0])
    
    if T0 is None:
        T0 = best_c * 0.05
    if T1 is None:
        T1 = best_c * 0.0001
    
    # Geometric cooling
    ratio = (T1 / T0) ** (1.0 / max_iters)
    T = T0
    
    # Tabu list
    tabu_len = 100
    tabu = {}  # position -> iteration when it becomes untabu
    it = 0
    
    while time.time() < deadline and it < max_iters:
        # Compute gradient for guidance
        if it % 10 == 0:
            c_vec = compute_c5_vec(h, n)
            w = smooth_max_w(c_vec, 50000.0)
            grad = grad_from_w(w, h, n)
        
        # Choose move: gradient-guided with randomness
        # Remove candidate: from ones with positive grad
        one_list = list(ones)
        zero_list = list(zeros)
        
        if not one_list or not zero_list:
            break
        
        # Weighted selection based on gradient
        # Positions with high positive grad are good to remove
        # Positions with low (negative) grad are good to add
        
        # Sample with some randomness
        n_remove_candidates = min(20, len(one_list))
        n_add_candidates = min(20, len(zero_list))
        
        # Sort by gradient (for guidance)
        remove_sorted = sorted(one_list, key=lambda x: -grad[x])[:n_remove_candidates]
        add_sorted = sorted(zero_list, key=lambda x: grad[x])[:n_add_candidates]
        
        # Pick from candidates with some randomness
        r = rng.random()
        if r < 0.7:
            # Guided: pick from top candidates
            i = remove_sorted[rng.randint(len(remove_sorted))]
        else:
            # Random: pick from all
            i = one_list[rng.randint(len(one_list))]
        
        r = rng.random()
        if r < 0.7:
            j = add_sorted[rng.randint(len(add_sorted))]
        else:
            j = zero_list[rng.randint(len(zero_list))]
        
        if i == j:
            continue
        
        # Check tabu
        if i in tabu and tabu[i] > it:
            continue
        if j in tabu and tabu[j] > it:
            continue
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        
        # Accept?
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            # Accept
            cur_c = c_new
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            
            # Update tabu
            tabu[i] = it + tabu_len
            tabu[j] = it + tabu_len
            
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
        
        T *= ratio
        it += 1
    
    return best_h, best_c

def thue_morse(n):
    """Generate Thue-Morse binary sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        # popcount
        pc = bin(i).count('1')
        if pc % 2 == 1:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    target = n // 2
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        # Turn off some
        for idx in ones[target:]:
            h[idx] = 0.0
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - len(ones)]:
            h[idx] = 1.0
    return h

def quadratic_residues(n):
    """Generate binary sequence based on quadratic residues."""
    h = np.zeros(n)
    target = n // 2
    # Find a prime p near n
    from math import isqrt
    p = n
    while p > 1:
        is_prime = True
        for f in range(2, isqrt(p) + 1):
            if p % f == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1
    
    # Quadratic residues mod p
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    
    # Map to sequence
    pos = 0
    for r in sorted(residues):
        if pos < n:
            h[pos] = 1.0
            pos += 1
    
    # Adjust to exactly n/2
    target = n // 2
    cur_ones = int(h.sum())
    if cur_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0.0
    elif cur_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - cur_ones]:
            h[idx] = 1.0
    return h

def run_lbfgs_continuous(h, n, deadline, alpha, mu, maxit=2000):
    """Run L-BFGS on continuous h."""
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max_val(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    
    def grad(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 5})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def continuous_refinement(h, n, deadline, rng):
    """Un-binarize sensitive positions and optimize continuously."""
    h = h.copy()
    best_c = compute_c5(h, n)
    
    # Find sensitive positions
    c_vec = compute_c5_vec(h, n)
    w = smooth_max_w(c_vec, 100000.0)
    grad = grad_from_w(w, h, n)
    
    # Pick positions with high |grad|
    sens_idx = np.argsort(-np.abs(grad))
    
    # Try different subsets
    for num_sens in [10, 20, 30]:
        if time.time() > deadline:
            break
        positions = sens_idx[:num_sens]
        
        # Create a relaxed solution: allow these positions to be continuous
        h_relaxed = h.copy()
        # Set these to 0.5 (midpoint) to give optimizer freedom
        for p in positions:
            h_relaxed[p] = 0.5
        
        # Optimize only these positions (fix others)
        x0 = h_relaxed[positions].copy()
        
        def obj_sub(x):
            h_tmp = h.copy()
            h_tmp[positions] = np.clip(x, 0, 1)
            h_tmp = project_feasible(h_tmp, n)
            return compute_c5(h_tmp, n)
        
        try:
            res = minimize(obj_sub, x0, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-12})
            h_new = h.copy()
            h_new[positions] = np.clip(res.x, 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    
    return h, best_c

```python
def grad_from_w_fast(w, h, n):
    """Vectorized gradient computation."""
    dx = 2.0 / n
    # w has length 2n-1
    # For shift k >= 0 (index n-1+k in w):
    #   grad[0:n-k] += w[n-1+k]*dx * (1-h[k:n])
    #   grad[k:n] += w[n-1+k]*dx * (-h[0:n-k])
    # For shift -k (index n-1-k in w):
    #   grad[k:n] += w[n-1-k]*dx * (1-h[0:n-k])
    #   grad[0:n-k] += w[n-1-k]*dx * (-h[k:n])
    
    grad = np.zeros(n)
    
    # Process positive shifts k=0,1,...,n-1
    for k in range(n):
        idx = n - 1 + k  # index in w
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        nv = n - k
        grad[0:nv] += wk * (1.0 - h[k:n])
        grad[k:n] += wk * (-h[0:nv])
    
    # Process negative shifts k=1,...,n-1
    for k in range(1, n):
        idx = n - 1 - k  # index in w
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        nv = n - k
        grad[k:n] += wk * (1.0 - h[0:nv])
        grad[0:nv] += wk * (-h[k:n])
    
    return grad
```

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_from_w(w, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(n):
        idx = n - 1 + d
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        nv = n - d
        grad[0:nv] += wk * (1.0 - h[d:n])
        grad[d:n] += wk * (-h[0:nv])
    for d in range(1, n):
        idx = n - 1 - d
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        nv = n - d
        grad[d:n] += wk * (1.0 - h[0:nv])
        grad[0:nv] += wk * (-h[d:n])
    return grad

def greedy_local_search(h, n, deadline, max_rounds=100000):

            break
        c_vec = compute_c5_vec(h, n)
        w = smooth_max_w(c_vec, 50000.0)
        grad = grad_from_w(w, h, n)
        
        best_gain = 1e-14
        best_i = -1
