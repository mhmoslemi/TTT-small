def build_G(h,n):
    L=2*n-1
    G=np.empty((L,n), float)
    # k=0 center
    G[n-1] = 1.0 - 2.0*h
    for r in range(1,n):

        rowp[r:] = 1.0

        rown[:n-r] -= h[r:]
        rown[r:] -= h[:n-r]
        G[n-1-r] = rown

def project_feasible(h,n):
    target=n*0.5
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    s=h.sum()
    diff=target-s
    if diff>0:
       room=1.0-h
       sr=room.sum()
       if sr>1e-15:
          h += (diff/sr)*room
       else: # impossible target>n, clip?
          h[:] = min(1.0,target/n)
    elif diff<0:
       sh=s
       if sh>1e-15:
          h += (diff/sh)*h
       else:
          h[:] = 0.0
    h=np.clip(h,0.0,1.0)
    # exact fix
    diff=target-h.sum()
    if abs(diff)>1e-14:
       # find an index not at bound
       idx=np.where((h>1e-12)&(h<1.0-1e-12))[0]
       if idx.size>0:
          i=idx[0]
          h[i]+=diff
          h[i]=np.clip(h[i],0.0,1.0)
       else:
          # choose max room if diff>0 else min positive
          if diff>0:
             i=int(np.argmax(1.0-h)); h[i]=min(1.0,h[i]+diff)
          else:
             i=int(np.argmin(h)); h[i]=max(0.0,h[i]+diff)
    return h

import time
import numpy as np
from scipy.optimize import linprog

_LAG_ORIENT = None


def build_G_with(h, n, fwd):
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    G[n - 1] = 1.0 - 2.0 * h

    for r in range(1, n):
        pidx = n - 1 + r
        nidx = n - 1 - r

        if fwd:
            G[pidx, r:] = 1.0
            G[nidx, :n - r] = 1.0
        else:
            G[pidx, :n - r] = 1.0
            G[nidx, r:] = 1.0

        G[pidx, :n - r] -= h[r:]
        G[pidx, r:] -= h[:n - r]
        G[nidx, :n - r] -= h[r:]
        G[nidx, r:] -= h[:n - r]

    return G


def _gradient_orientation_ok(fwd, n=5, trials=4):
    rng = np.random.RandomState(20240101)
    eps = 1e-5
    tol = 1e-5

    for _ in range(trials):
        h = rng.rand(n)
        G = build_G_with(h, n, fwd)
        idxs = (0, 1, 2, n - 1, 2 * n - 2)

        for idx in idxs:
            for j in range(n):
                hp = h.copy()
                hm = h.copy()
                hp[j] += eps
                hm[j] -= eps

                cp = np.correlate(hp, 1.0 - hp, mode="full")[idx]
                cm = np.correlate(hm, 1.0 - hm, mode="full")[idx]
                fd = (cp - cm) / (2.0 * eps)

                if abs(fd - G[idx, j]) > tol:
                    return False

    return True


def get_orientation():
    global _LAG_ORIENT
    if _LAG_ORIENT is None:
        if _gradient_orientation_ok(True):
            _LAG_ORIENT = True
        elif _gradient_orientation_ok(False):
            _LAG_ORIENT = False
        else:
            _LAG_ORIENT = True
    return _LAG_ORIENT


def build_G(h, n):
    return build_G_with(h, n, get_orientation())


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    s = h.sum()
    diff = target - s

    if diff > 0.0:
        room = 1.0 - h
        sr = room.sum()
        if sr > 1e-15:
            h += (diff / sr) * room
        else:
            h[:] = min(1.0, target / max(1, n))
    elif diff < 0.0:
        sh = s
        if sh > 1e-15:
            h += (diff / sh) * h
        else:
            h[:] = 0.0

    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size > 0:
            i = idx[0]
            h[i] += diff
            h[i] = min(1.0, max(0.0, h[i]))
        else:
            if diff > 0.0:
                i = int(np.argmax(1.0 - h))
                h[i] = min(1.0, h[i] + diff)
            else:
                i = int(np.argmin(h))
                h[i] = max(0.0, h[i] + diff)

    return h


def c5_array(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return c, float(np.max(c) * dx)


def c5_value(h, n):
    return c5_array(h, n)[1]


def lp_direction(h, G, c_arr, maxc, S, n):
    L = c_arr.size
    gaps = np.maximum(0.0, maxc - c_arr)

    A_ub = np.empty((L, n + 1), dtype=float)
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=float)

    lower = -h
    lower = np.maximum(lower, -S)
    lower = np.minimum(lower, 0.0)

    upper = 1.0 - h
    upper = np.minimum(upper, S)
    upper = np.maximum(upper, 0.0)

    bounds = np.column_stack((lower, upper))
    bounds = np.vstack((bounds, np.array([[-1.0, 1.0]])))

    obj = np.zeros(n + 1, dtype=float)
    obj[n] = 1.0

    try:
        res = linprog(
            c=obj,
            A_ub=A_ub,
            b_ub=gaps,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None or res.status != 0:
        return None

    s = float(res.x[n])
    if s >= -1e-8:
        return None

    d = np.asarray(res.x[:n], dtype=float).copy()
    d[np.abs(d) < 1e-13] = 0.0

    ds = d.sum()
    if abs(ds) > 1e-12:
        d -= ds / float(n)
        if d.min() < -S - 1e-8 or d.max() > S + 1e-8:
            d = np.clip(d, -S, S)
            ds = d.sum()
            d -= ds / float(n)

    if np.max(np.abs(d)) < 1e-10:
        return None

    return d


def lp_polish(h, n, deadline, max_iters=100, step_sizes=None):
    h = project_feasible(h, int(n))
    n = int(n)
    c_arr, C = c5_array(h, n)

    if step_sizes is None:
        step_sizes = (0.08, 0.04, 0.02, 0.01, 0.005, 0.002)

    alphas = (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005)

    for _ in range(max_iters):
        if time.time() > deadline:
            break

        G = build_G(h, n)
        improved = False

        for S in step_sizes:
            if time.time() > deadline:
                break

            d = lp_direction(h, G, c_arr, C, S, n)
            if d is None:
                continue

            best_h = None
            best_c_arr = None
            best_C = C

            for a in alphas:
                if time.time() > deadline:
                    break

                hn = h + a * d
                if hn.min() < -1e-8 or hn.max() > 1.0 + 1e-8:
                    continue

                hn = np.clip(hn, 0.0, 1.0)
                hn = project_feasible(hn, n)
                cn, Cn = c5_array(hn, n)

                if Cn < best_C - 1e-12:
                    best_C = Cn
                    best_h = hn
                    best_c_arr = cn

            if best_h is not None:
                h = best_h
                c_arr = best_c_arr
                C = best_C
                improved = True
                break

        if not improved:
            break

    return h, C


def random_balanced(n, rng):
    n = int(n)
    target = 0.5 * n
    h = np.zeros(n, dtype=float)

    m = int(target)
    if m > 0 and m <= n:
        idx = rng.choice(n, size=m, replace=False)
        h[idx] = 1.0

    frac = target - m
    if frac > 1e-12:
        zeros = np.where(h == 0.0)[0]
        if zeros.size > 0:
            h[zeros[0]] = frac

    return project_feasible(h, n)


def binary_threshold(h, n):
    n = int(n)
    target = 0.5 * n
    h2 = np.zeros(n, dtype=float)
    order = np.argsort(h)[::-1]

    m = int(target)
    if 0 < m < n:
        h2[order[:m]] = 1.0
    elif m == n:
        h2[:] = 1.0

    frac = target - m
    if frac > 1e-12 and m < n:
        h2[order[m]] = frac

    return project_feasible(h2, n)


def upsample_h(h, n, nh):
    n = int(n)
    nh = int(nh)
    if nh <= 0:
        return h.copy()
    if nh == n:
        return h.copy()

    xo = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    xn = (np.arange(nh, dtype=float) + 0.5) * (2.0 / nh)
    hh = np.interp(xn, xo, h)
    return project_feasible(hh, nh)


def random_improve(h, n, deadline, seed, max_evals=1500):
    n = int(n)
    h = project_feasible(h, n)
    C = c5_value(h, n)
    rng = np.random.RandomState(seed)

    if n < 2:
        return h, C

    if time.time() >= deadline:
        return h, C

    half = max(1, min(n, max(3, n // 10)))
    order = np.argsort(h)
    low = order[:half]
    high = order[-half:]

    evals = 0
    while time.time() < deadline and evals < max_evals:
        i = int(rng.choice(high))
        j = int(rng.choice(low))
        if i == j or abs(h[i] - h[j]) < 1e-10:
            evals += 1
            continue

        hn = h.copy()
        hn[i], hn[j] = h[j], h[i]
        Cn = c5_value(hn, n)
        evals += 1

        if Cn < C - 1e-12:
            h = hn
            C = Cn
            order = np.argsort(h)
            low = order[:half]
            high = order[-half:]

    return h, C

            try:
                h2, c2 = lp_polish(cand, best_n, cand_deadline, max_iters=40)
                if c2 < best_c - 1e-13:
                    best_h, best_c = h2, c2
            except Exception:
                pass

        if time.time() < remaining_end - 20.0:
            if abs(best_n / 2.0 - round(best_n / 2.0)) < 1e-12:
                cand = binary_threshold(best_h, best_n)
                cand_deadline = min(remaining_end, time.time() + 30.0)
                try:
                    h2, c2 = lp_polish(cand, best_n, cand_deadline, max_iters=40)
                    if c2 < best_c - 1e-13:
                        best_h, best_c = h2, c2
                except Exception:
                    pass

        for shift in (1, 3, 7, 13):
            if time.time() > remaining_end - 10.0:
                break
            cand = np.roll(best_h, shift).copy()
            cand_deadline = min(remaining_end, time.time() + 15.0)
            try:
                h2, c2 = lp_polish(cand, best_n, cand_deadline, max_iters=20)
                if c2 < best_c - 1e-13:
                    best_h, best_c = h2, c2
            except Exception:
                pass

    # 3. Try neighboring/higher resolutions obtained from the best step function.
    if time.time() < deadline - 120.0:
        high_ns = []
        for target in (300, 350, 400, 480):
            if target > best_n and target <= 600:
                high_ns.append(target)

        if best_n * 1.2 > best_n + 10:
            cand = int(1.2 * best_n)
            if cand % 2:
                cand += 1
            if cand > best_n and cand <= 600 and cand not in high_ns:
                high_ns.append(cand)

        for nh in high_ns[:2]:
            if time.time() > deadline - 80.0:
                break
            cand = upsample_h(best_h, best_n, nh)
            cand_deadline = min(deadline - 50.0, time.time() + 95.0)
            try:
                h2, c2 = lp_polish(cand, nh, cand_deadline, max_iters=80)
                if c2 < best_c - 1e-13:
                    best_h, best_c, best_n = h2, c2, nh
            except Exception:
                pass

            # A second low-resolution resampling is sometimes useful.
            if cad_deadline_available = (time.time() < deadline - 70.0):
                cand = upsample_h(best_h, best_n, nh + 20 if nh + 20 <= 600 else nh)
                cand_deadline = min(deadline - 60.0, time.time() + 45.0)
                try:
                    h2, c2 = lp_polish(cand, nh + 20 if nh + 20 <= 600 else nh,
                                       cand_deadline, max_iters=40)
                    if c2 < best_c - 1e-13:
                        best_h, best_c, best_n = h2, c2, nh + 20 if nh + 20 <= 600 else nh
                except Exception:
                    pass

    # 4. A short exact swap pass can move out of very flat first-order minima.
    if time.time() < deadline - 35.0:
        try:
            h2, c2 = random_improve(best_h, best_n, min(deadline - 25.0, time.time() + 30.0),
                                    seed + 12345, max_evals=1500)
            if c2 < best_c - 1e-13:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # 5. Final full-length polish.
    if time.time() < deadline - 4.0:
        try:
            h2, c2 = lp_polish(best_h, best_n, deadline - 2.0, max_iters=1000)
            if c2 < best_c - 1e-13:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Final safety project and exact recomputation.
    best_h = project_feasible(best_h, best_n)
    best_c = c5_value(best_h, best_n)

    return best_h, float(best_c), int(best_n)
