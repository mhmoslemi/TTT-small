Actually, wait. I want to add one more idea: **the "oscillation" heuristic**.

The key insight is that for the optimal h, adjacent points should have different "phases" (one high, one low) to minimize the overlap at small shifts. But at larger shifts, the pattern should repeat to maintain low overlap.

This suggests a structure like: h alternates between high and low values, but with a period that's optimized for the worst-case shift.

For a purely alternating pattern (period 2), the overlap at shift k is:
- k odd: h(x) and h(x+k) are in opposite phases, so h(x)(1-h(x+k)) ≈ h(x)*h(x) = h(x)^2 (if h is 0/1 alternating, this is 1)
- k even: h(x) and h(x+k) are in same phase, so h(x)(1-h(x+k)) ≈ 0

So the maximum would be at odd k, giving C₅ ≈ 0.5 (for binary alternating).

For a period-3 pattern (1,1,0,1,1,0,...):
- The overlap at various shifts would be more balanced.

In general, the optimal period p should satisfy that for each shift k, the fraction of positions where h=1 and h(x+k)=0 is as small as possible.

This is related to the theory of difference sets and balanced sequences. The optimal binary sequences for minimizing maximum cross-correlation are related to m-sequences and other pseudo-random sequences.

But for our continuous problem, the optimal h is not binary. It has intermediate values that "smooth out" the worst shifts.

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_grad_active(h, n, alpha, active_idx=None):
    """Compute gradient of smoothed max using only active shifts."""
    c = compute_c5_vec(h, n)
    m = c.max()
    dx = 2.0 / n
    
    if active_idx is None:
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        active_mask = w > 1e-12
    else:
        w = np.zeros(2*n-1)
        w[active_idx] = 1.0 / len(active_idx)
        active_mask = w > 0
    
    grad = np.zeros(n)
    for idx in np.where(active_mask)[0]:
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:n-k])
        elif -n < k < 0:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] += wm * (-h[kk:])
    return c, m, grad, w

def smooth_obj_with_active(h, n, alpha, mu):
    """Objective with cached active set for speed."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_active(h, n, alpha, mu):

    m = c.max()

def slqp_polish_v2(h, n, deadline, alpha=500000, maxiter=2000):
    """SLSQP polish."""
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    
    bounds = [(0.0, 1.0)] * n

    dx = 2.0 / n

    no_improve = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        c = compute_c5_vec(h, n)
        c_max = c.max()
        
        # Find active set (top M shifts)
        top_M = np.argsort(c)[-M:]
        
        # Compute gradient as weighted average of active shift gradients
        # Use exponential weights based on how close to max
        weights = np.exp(100 * (c[top_M] - c_max))
        weights /= weights.sum()

        for i, idx in enumerate(top_M):
            k = idx - (n - 1)
            wm = weights[i] * dx

            elif 0 < k < n:

            elif -n < k < 0:
                kk = -k
                grad[kk:] += wm * (1.0 - h[:n-kk])
                grad[:n-kk] += wm * (-h[kk:])

        gnorm = np.linalg.norm(grad)

            if no_improve > 100:

        else:

    return best_h, best_c
