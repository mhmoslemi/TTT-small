        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx, top_k, c_all

def grad_descent_step(h, n, step_size, k_top=5):
    """One gradient descent step on top-k max, with projection."""
    grad, top_k, c_all = topk_lag_gradient(h, n, k_top)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad_vec(h, n, alpha):
    """Vectorized gradient of smooth max."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weights(c, alpha)
    
    # Only process lags with significant weight
    active = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        wk = w[k_idx]
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad_vec(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def gradient_local_search(h, n, deadline, rng, k_top=3, max_rounds=50000):
    """Local search guided by gradient of top-k lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    current_step_idx = 0
    rounds_no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = np.correlate(h, 1 - h, mode='full')
        max_c = c_all.max() * (2.0 / n)
        
        if max_c < best_c - 1e-14:
            best_c = max_c
            best_h = h.copy()
            rounds_no_improve = 0
        else:
            rounds_no_improve += 1
        
        if rounds_no_improve > 5000:
            if current_step_idx < len(step_sizes) - 1:
                current_step_idx += 1
            else:
                # Try a random perturbation
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                rounds_no_improve = 0
                current_step_idx = 0
                continue
        
        step_size = step_sizes[current_step_idx]
        
        # Get gradient of top-k
        grad, top_k, _ = topk_lag_gradient(h, n, k_top)
        
        # Try gradient step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < compute_c5(h, n) - 1e-14:
            h = h_new
        else:
            # Try random perturbation in the gradient direction with noise
            h_pert = h + rng.normal(0, step_size * 0.5, n) - step_size * 0.3 * grad
            h_pert = project_feasible(h_pert, n)
            c_pert = compute_c5(h_pert, n)
            if c_pert < compute_c5(h, n) - 1e-14:
                h = h_pert
    
    return best_h, best_c

def targeted_lag_search(h, n, deadline, rng, max_rounds=100000):
    """Search that specifically targets the worst lag pairs."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_all = np.correlate(h, 1 - h, mode='full')
        top_idx = np.argsort(c_all)[-5:]
        
        improved = False
        for k_star in top_idx:
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Find the pair (i, i+m) with maximum contribution to this lag
            if lag > 0:
                # c = sum h[i]*(1-h[i+m])
                # To decrease c, we want to either decrease h[i] or increase h[i+m]
                contribs = h[:n-m] * (1 - h[m:n])
                # Also: -h[i+m]*(1-h[i]) from the other part... 
                # Actually the full contribution at this lag is just sum of h[i]*(1-h[i+m])
                # To minimize: decrease h[i] where h[i] is large and h[i+m] is small
                # Or increase h[i+m] where h[i] is large
                i_max = np.argmax(contribs)
            else:
                # lag = -m, c = sum h[i+m]*(1-h[i]) for i=0..n-m-1
                # equivalently sum h[i]*(1-h[i-m]) for i=m..n-1
                contribs = h[m:n] * (1 - h[:n-m])
                i_max = np.argmax(contribs) + m
            
            # Try small adjustments at the worst pair
            step = min(0.01, 0.5 * max(1e-4, abs(h[i_max] - 0.5)))
            
            if lag > 0:
                i = i_max
                j = i_max + m
                if h[i] > h[j]:
                    h[i] -= step
                    h[j] += step
                else:
                    h[i] += step
                    h[j] -= step
            else:
                i = i_max
                j = i_max + m
                if h[i] > h[j]:
                    h[i] -= step
                    h[j] += step
                else:
                    h[i] += step
                    h[j] -= step
            
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            
            if c_test < best_c - 1e-14:
                best_c = c_test
                best_h = h.copy()
                improved = True
            else:
                # Revert
                if lag > 0:
                    if h[i] < h[j] + step:  # we moved h[i] down
                        h[i] += step
                        h[j] -= step
                    else:
                        h[i] -= step
                        h[j] += step
                else:
                    if h[i] < h[j] + step:
                        h[i] += step
                        h[j] -= step
                    else:
                        h[i] -= step
                        h[j] += step
                h = project_feasible(h, n)
        
        if not improved and rnd % 1000 == 0:
            # Random restart exploration
            h = best_h + rng.normal(0, 0.003, n)
            h = project_feasible(h, n)
    
    return best_h, best_c
