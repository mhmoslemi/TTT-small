Key differences from previous:
1. Use a "power sum" approximation instead of log-sum-exp: f_p = (sum c_k^p)^(1/p) for p=100 or 200. This has a simpler gradient.
   Actually, for negative values this is problematic. Since c_k >= 0 (h is in [0,1], so h(1-h) >= 0), this should be fine.
   
   Actually wait: c_k = sum_j h[j](1-h[j+k]). Since h in [0,1], each term is in [0,1], so c_k in [0, n] (before dx scaling). After dx scaling, c_k in [0, 2]. So all values are non-negative. Power sum works.

   f_p = (sum_k (c_k)^p)^(1/p)
   df_p/dh_i = (1/p) * (sum c_k^p)^(1/p - 1) * sum_k p * c_k^(p-1) * dc_k/dh_i
   = (sum c_k^p)^(-1+1/p) * sum_k c_k^(p-1) * dc_k/dh_i

   This is a weighted average of the dc_k/dh_i, weighted by c_k^(p-1). For large p, this concentrates on the maximum. The gradient is simpler than log-sum-exp.

2. **Use SLSQP with the power-sum objective** as the main optimizer (it handles constraints natively).

3. **Nelder-Mead on a reduced set**: Identify the "active" coordinates (those with h in (0,1)) and do a local search on a subset.

4. **Alternating: optimize the active set, then re-threshold**.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                h += diff / n
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                h -= diff / n
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def power_sum_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    # Power sum approximation of max
    # All c_k >= 0
    cp = c ** p
    s = cp.sum()
    if s < 1e-300:
        val = 0.0
    else:
        val = s ** (1.0 / p)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def power_sum_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    # Ensure non-negative for power
    c_safe = np.maximum(c, 1e-300)
    cp1 = c_safe ** (p - 1)
    s = (c_safe ** p).sum()
    if s < 1e-300:
        return np.zeros(n) + 2*mu*(x.sum()-n/2.0)
    w = cp1 / s ** (1 - 1.0/p)  # weights for gradient
    # Actually: df/dx_i = (sum c_k^p)^(-1+1/p) * sum_k c_k^(p-1) * dc_k/dx_i
    # (sum c_k^p)^(-1+1/p) = s^(-1+1/p) = 1/s^(1-1/p)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        k = idx - center
        wk = w[idx]
        if wk < 1e-15:
            continue
        wm = wk * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        k = idx - center
        wm = w[idx] * dx
        if wm < 1e-15:
            continue
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 100.0), (2000, 1000.0),
                   (5000, 5000.0), (10000, 20000.0), (30000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': 200000, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def subgradient_step(h, n, step_size, alpha=10000):
    """Take a subgradient step on the max objective."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        k = idx - center
        wm = w[idx] * dx
        if wm < 1e-15:
            continue
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def binary_round(h, n):
    """Round to binary, maintaining sum constraint."""
    h_bin = np.zeros(n)
    h_bin[h > 0.5] = 1.0
    target = n / 2.0
    # Adjust
    diff = target - h_bin.sum()
    if abs(diff) < 1e-10:
        return h_bin
    if diff > 0:
        # Need more 1s
        frac = h[h < 0.5]  # candidates to set to 1
        order = np.argsort(-frac)
        for i in order:
            if h_bin[i] < 1.0:
                h_bin[i] = 1.0
                diff -= 1.0
                if abs(diff) < 1e-10:
                    break
    else:
        frac = h[h > 0.5]
        order = np.argsort(frac)
        for i in order:
            if h_bin[i] > 0.0:
                h_bin[i] = 0.0
                diff += 1.0
                if abs(diff) < 1e-10:
                    break
    return h_bin

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Hmm, the discretization uses n points over [0,2] with dx = 2/n
    # So points are at 0, dx, 2dx, ..., (n-1)dx
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def slqp_optimize(h, n, deadline, alpha=1000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, na: x.sum() - na/2.0,
             'jac': lambda x, na: np.ones(na), 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def coord_descent(h, n, deadline, n_rounds=20):
    """Coordinate descent: try adjusting each coordinate in [0,1]."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            # Try a few values
            h[i] = best_h[i]
            for val in [0.0, 1.0, best_h[i] - 0.1, best_h[i] + 0.1,
                        np.clip(best_h[i] - 0.05, 0, 1), np.clip(best_h[i] + 0.05, 0, 1)]:
                h[i] = val
                # Check sum constraint (approximately)

                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    
    # Fix sum constraint
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def subgradient_descent(h, n, deadline, n_iters=5000):
    """Subgradient descent with adaptive step size."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.01
    decay = 0.999
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        h = subgradient_step(h, n, step, alpha=5000)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            step *= 1.01  # increase step slightly on improvement
        else:
            step *= decay  # decrease step
        
        # Periodically project to binary and re-optimize
        if it % 500 == 499:
            h_bin = binary_round(h, n)
            try:
                res = minimize(smooth_max_obj, h_bin, jac=smooth_max_grad,
                              args=(n, 5000, 100.0), method='L-BFGS-B',

                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c
