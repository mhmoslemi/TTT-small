import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Find indices with room in the right direction
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):

def act_grad(x, n, active, alpha, mu):

def upsample_h(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_kronecker(n, alpha_param, offset=0.0):

        active = np.where(c_vec >= threshold)[0]

        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - idx_center
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - best_h[k:n])
                    grad[k:n] += (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - best_h[0:nv])
                    grad[0:nv] += (-best_h[kk:n])
        grad *= dx / len(active)
        
        # Polyak-like step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        eta = max(0.001, (c_max - f_star_est) / (grad_norm_sq + 1e-20))
        eta = min(eta, 0.1)  # cap step size
        
        # Take step
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        else:
            # Reduce step size
            eta *= 0.5
    
    return best_h, best_c

def binary_snapping(h, n):
    """Try to snap h to binary while maintaining sum constraint"""
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    n_ones = n // 2
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def local_pair_search(h, n, deadline, radius, deltas):
    """Fine local pair search"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, max(1, n // 2000)):  # subsample for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]

def lag_gradient(x, n, lag):

    grad = np.zeros(n)
    k = lag
    if k >= 0:
        nv = n - k

        nv = n - kk

            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad *= (2.0 / n)
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    idx_center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 100 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_kronecker(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    fracs = np.sort(np.array([(k * alpha_param) % 1.0 for k in range(1, n + 1)]))
    threshold = fracs[int(n_ones) - 1] if n_ones > 0 else 1.0
    for k in range(1, n + 1):
        if (k * alpha_param) % 1.0 <= threshold:
            h[int((k * alpha_param) % 1.0 * n) % n] = 1.0
    h = np.zeros(n)
    n_ones = n // 2
    fracs = [(k * alpha_param) % 1.0 for k in range(1, n + 1)]
    fracs_sorted = sorted(range(n), key=lambda i: fracs[i])
    positions = fracs_sorted[:n_ones]
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def active_set_optimize(h, n, deadline, top_k, alpha, mu=0.001):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-7, 0.0005 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 5:
        return h, compute_c5(h, n)
    maxit = max(100, int(rem * 2.5))
    maxit = min(maxit, 2500)
    try:
        res = minimize(act_obj, h, jac=act_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def full_optimize(h, n, deadline, alpha, mu=0.01):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 5:
        return h, compute_c5(h, n)
    maxit = max(200, int(rem * 2))
    maxit = min(maxit, 2000)
    try:
        res = minimize(full_obj, h, jac=full_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def subgradient_refine(h, n, deadline, max_iter=5000, momentum=0.5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    idx_center = n - 1
    vel = np.zeros(n)
    f_lower = max(0.2, best_c - 0.05)
    base_eta = 0.05
    decaying_eta = base_eta
    stall_count = 0

    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        if c_max < best_c:
            best_c = c_max
            stall_count = 0
            f_lower = max(0.2, best_c - 0.02)
        else:
            stall_count += 1
            if stall_count > 500:
                decaying_eta *= 0.5
                stall_count = 0
            if decaying_eta < 1e-6:
                break

        threshold = c_max - max(1e-9, 0.001 * c_max)
        active = np.where(c_vec >= threshold)[0]
        n_active = len(active)
        if n_active == 0:
            break

        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - idx_center
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - best_h[k:n])
                    grad[k:n] += (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - best_h[0:nv])
                    grad[0:nv] += (-best_h[kk:n])
        grad *= (2.0 / n) / n_active

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        eta = min(0.1, max(0.001, base_eta * decaying_eta / (1.0 + it * 0.001)))

        vel = momentum * vel - eta * grad
        h_new = best_h + vel
        h_new = project_feasible(h_new, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        else:
            vel = 0.5 * vel

    return best_h, best_c

def directed_local_search(h, n, deadline, n_trials=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    idx_center = n - 1

    for trial in range(n_trials):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.0005 * max_val)
        active = set(int(a) for a in np.where(c_vec >= max_val - tol)[0])

        found_improvement = False
        for idx in list(active)[:min(len(active), 15)]:
            if time.time() > deadline - 5:
                break
            k = idx - idx_center
            if k >= 0:
                nv = n - k
                candidates = [(i, i + k) for i in range(0, nv, max(1, nv // 500))]
            else:
                kk = -k
                nv = n - kk
                candidates = [(i + kk, i) for i in range(0, nv, max(1, nv // 500))]

            for (a, b) in candidates[:200]:
                for delta in [0.001, 0.0005, 0.0002]:
                    if best_h[a] >= delta and best_h[b] <= 1.0 - delta:
                        best_h[a] -= delta
                        best_h[b] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found_improvement = True
                        else:
                            best_h[a] += delta
                            best_h[b] -= delta
        if not found_improvement and trial > 20:
            break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = max(1, n // 3000)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, step):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def binary_snapping(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    n_ones = n // 2
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def perturbation_try(h, n, deadline, rng, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        scale = max(0.005, 0.02 * (1 - t / n_trials))
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.003:
            rem = deadline - time.time() - 10
            if rem > 30:
                h_opt, c_opt = full_optimize(h_pert, n, time.time() + min(rem, 60), 50000, 0.005)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    return best_h, best_c
