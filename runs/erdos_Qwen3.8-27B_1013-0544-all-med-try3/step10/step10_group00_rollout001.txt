                    grad = h_grad_from_w(w, h, n)
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            # Try 2-swap
            if rng.random() < 0.3:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if i1 != i2 and j1 != j2 and i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2:
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        ones.remove(i1); ones.remove(i2)
                        ones.extend([j1, j2])
                        zeros.remove(j1); zeros.remove(j2)
                        zeros.extend([i1, i2])
                        c_vec = compute_c5_vec(h, n)
                        w = smooth_max_w(c_vec, alpha)
                        grad = h_grad_from_w(w, h, n)
                        improved = True
                    else:
                        h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: L-BFGS continuous optimization (60s)
    p1_end = t0 + 60
    for alpha in [5000, 20000, 100000]:
        if time.time() > p1_end:
            break
        h1, c1 = run_lbfgs(best_h, best_n, min(p1_end, time.time()+25), alpha=alpha, mu=0.01, maxit=2000)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Binarize
    best_h_bin = binarize(best_h, best_n)
    c_bin = compute_c5(best_h_bin, best_n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = best_h_bin.copy()
    
    # Phase 2: Intensive simulated annealing (200s)
    p2_end = min(t0 + 260, deadline - 300)
    h2, c2 = binary_sa(best_h, best_n, p2_end, rng, T_init=0.8, T_min=0.001, max_iters=300000)
    if c2 < best_c:
        best_c = c2
