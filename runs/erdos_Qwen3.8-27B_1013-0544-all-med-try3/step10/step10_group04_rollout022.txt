import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    thresh = 1e-13
    indices = np.where(w > thresh)[0]
    for idx in indices:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def interp_to(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    if n_old == 1:
        return project_feasible(np.full(n_new, h_old[0]), n_new)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (800, 200.0),
                   (2000, 1000.0), (8000, 5000.0), (30000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': 200000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_hill_climb(h, n, deadline, max_iters=80000, rng_seed=42):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    ones_count = h_bin.sum()
    if ones_count > target:
        one_idx = np.where(h_bin == 1)[0]
        one_order = one_idx[np.argsort(-h[one_idx])]
        h_bin[one_order[target:]] = 0
    elif ones_count < target:
        zero_idx = np.where(h_bin == 0)[0]
        zero_order = zero_idx[np.argsort(-h[zero_idx])]
        h_bin[zero_order[:target - ones_count]] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    rng = np.random.RandomState(rng_seed)
    accepted = 0
    for it in range(max_iters):
        if time.time() > deadline - 15:
            break
        if it % 2000 == 1999 and accepted == 0:
            break
        ones_pos = np.where(h_bin == 1)[0]
        zeros_pos = np.where(h_bin == 0)[0]
        oi = ones_pos[rng.randint(len(ones_pos))]
        zi = zeros_pos[rng.randint(len(zeros_pos))]
        h_bin[oi] = 0
        h_bin[zi] = 1
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:

            best_h = h_bin.copy()
            accepted += 1

            accepted = 0

    best_h = h.copy()

            if k >= 0:
                overlap_range = range(0, n - k)
            else:
                overlap_range = range(-k, n)

            idxs = np.random.choice(len(overlap_range), size=n_try, replace=False)

                i = overlap_range[ii]

                    nv = h_j + delta

                        found = True

def make_block_pattern(n, n_blocks, rng):
    """Create a pattern with n_blocks: each block is a random-width run of 1s."""
    h = np.zeros(n)
    remaining = n
    target = n // 2
    blocks = n_blocks
    # Distribute ones
    ones_per_block = target // blocks
    rem_ones = target - ones_per_block * blocks
    pos = 0
    for b in range(blocks):
        ones_here = ones_per_block + (1 if b < rem_ones else 0)
        if pos + ones_here > n:
            ones_here = n - pos
        h[pos:pos+ones_here] = 1.0
        pos += ones_here
        gap = 1 + rng.randint(0, 3)
        pos += gap
        if pos >= n:
            break
    h = project_feasible(h, n)
    return h
