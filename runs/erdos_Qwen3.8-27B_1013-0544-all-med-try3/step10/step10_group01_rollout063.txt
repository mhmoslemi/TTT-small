```python
def _next_pow2(n):
    p = 1
    while p < n:
        p *= 2
    return p

def compute_c5_fft(h, n):
    """Compute correlation vector using FFT."""
    v = 1.0 - h
    N = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v, N)
    c_circ = np.fft.irfft(H * np.conj(V), N)
    # full[m] = c_circ[(m - (n-1)) mod N] for m=0..2n-2
    # = c_circ[N-(n-1):N] concatenated with c_circ[0:n]
    full = np.concatenate([c_circ[N-(n-1):], c_circ[:n]])
    return full

def compute_c5_scalar(h, n):
    dx = 2.0 / n
    return float(compute_c5_fft(h, n).max() * dx)

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and softmax weights."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        log_sum = smx + np.log(np.sum(np.exp(sh - smx)))
    else:
        log_sum = np.log(np.sum(np.exp(sh)))
    val = m + log_sum / alpha
    w = np.exp(sh - smx)
    w /= w.sum()
    return float(val), w

def smooth_max_obj(h, n, alpha, mu=0.0):
    dx = 2.0 / n
    c = compute_c5_fft(h, n)
    val, _ = smooth_max_and_weights(c * dx, alpha)
    if mu > 0:
        val += mu * (h.sum() - n/2.0)**2
    return val

def smooth_max_grad(h, n, alpha, mu=0.0):
    dx = 2.0 / n
    c = compute_c5_fft(h, n) * dx
    _, w = smooth_max_and_weights(c, alpha)
    # w has length 2n-1, corresponding to lags -(n-1)..n-1
    # Place w in FFT array
    N = _next_pow2(2 * n - 1)
    w_padded = np.zeros(N)
    w_padded[N-(n-1):] = w[0:n-1]  # lags -(n-1)..-1
    w_padded[0:n] = w[n-1:2*n-1]    # lags 0..n-1

    F_w = np.fft.rfft(w_padded, N)

    F_ones = np.zeros(N//2 + 1)
    F_ones[0] = N  # FFT of ones of length N is N at k=0, 0 elsewhere
    
    grad_fft = (F_ones - 2 * F_h) * np.conj(F_w)
    grad = np.fft.irfft(grad_fft, N)[:n] * dx
    
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n/2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    p = 1
    while p < n:
        p *= 2
    return p

def corr_vec_fft(h, n):
    """FFT-based cross-correlation of h and (1-h), matching np.correlate(h, 1-h, 'full')."""
    v = 1.0 - h
    N = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v, N)
    c_circ = np.fft.irfft(H * np.conj(V), N)
    return np.concatenate([c_circ[N-(n-1):], c_circ[:n]])

def c5_direct(h, n):
    """Direct C5 via np.correlate (for verification)."""
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def c5_fft(h, n):
    dx = 2.0 / n
    return float(corr_vec_fft(h, n).max() * dx)

def smooth_max_val_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        log_sum = smx + np.log(np.sum(np.exp(sh - smx)))
    else:
        log_sum = np.log(np.sum(np.exp(sh)))
    val = float(m + log_sum / alpha)
    w = np.exp(sh - smx)
    w /= w.sum()
    return val, w

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_full(h, n, alpha, mu):
    dx = 2.0 / n
    c = corr_vec_fft(h, n) * dx
    val, _ = smooth_max_val_weights(c, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def grad_full(h, n, alpha, mu):
    dx = 2.0 / n
    c = corr_vec_fft(h, n) * dx
    _, w = smooth_max_val_weights(c, alpha)
    N = _next_pow2(2 * n - 1)
    w_pad = np.zeros(N)
    w_pad[N-(n-1):] = w[:n-1]
    w_pad[:n] = w[n-1:]
    F_w = np.fft.rfft(w_pad, N)
    F_h = np.fft.rfft(h, N)
    F_ones = np.zeros(N // 2 + 1)
    F_ones[0] = N
    term1 = np.fft.irfft((F_ones - F_h) * np.conj(F_w), N)[:n]
    term2 = np.fft.irfft(F_h * np.conj(F_w), N)[:n]
    grad = dx * (term1 - term2)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def obj_topk(h, n, active, alpha, mu):
    dx = 2.0 / n
    c_full = corr_vec_fft(h, n) * dx
    c_act = c_full[active]
    val, _ = smooth_max_val_weights(c_act, alpha)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def grad_topk(h, n, active, alpha, mu):
    dx = 2.0 / n
    c_full = corr_vec_fft(h, n) * dx
    c_act = c_full[active]
    _, w = smooth_max_val_weights(c_act, alpha)
    N = _next_pow2(2 * n - 1)
    w_pad = np.zeros(N)
    w_full = np.zeros(2 * n - 1)
    w_full[active] = w
    w_pad[N-(n-1):] = w_full[:n-1]
    w_pad[:n] = w_full[n-1:]
    F_w = np.fft.rfft(w_pad, N)
    F_h = np.fft.rfft(h, N)
    F_ones = np.zeros(N // 2 + 1)
    F_ones[0] = N
    term1 = np.fft.irfft((F_ones - F_h) * np.conj(F_w), N)[:n]
    term2 = np.fft.irfft(F_h * np.conj(F_w), N)[:n]
    grad = dx * (term1 - term2)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def beatty_start(n, alpha_param, rng):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = c5_fft(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(200, int(remaining * 5))
        try:
            res = minimize(obj_full, best_h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = c5_fft(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active(h, n, deadline, top_k_list=None):

    best_c = c5_fft(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if top_k_list is None:

            break
        for rnd in range(6):

            active = np.where(c_vec >= threshold)[0]

                h_new = project_feasible(res.x, n)

                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:

                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = c5_fft(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stuck = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stuck += 1
                if stuck > 500:
                    break
    return best_h, best_c

def fine_pair(h, n, deadline, radius=40, deltas=None):
    best_h = h.copy()
    best_c = c5_fft(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n - 1, i + radius)
                for j in range(lo, hi + 1):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = c5_fft(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = c5_fft(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def swap_search(h, n, deadline, max_rounds=30):
    best_h = h.copy()
    best_c = c5_fft(h, n)
    rng = np.random.RandomState(123)
    ones_set = set(np.where(best_h > 0.5)[0].tolist())
    zeros_set = set(np.where(best_h < 0.5)[0].tolist())
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones_list = list(ones_set)
        zeros_list = list(zeros_set)
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        for i in ones_list:
            if time.time() > deadline:
                break
            for j in zeros_list[:max(1, len(zeros_list) // 5)]:
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = c5_fft(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones_set.discard(i)
                    ones_set.add(j)
                    zeros_set.discard(j)
                    zeros_set.add(i)
                    improved = True
                    break
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_h = project_feasible(best_h, n)
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = c5_fft(h0, n0)
    best_n = n0

    # Generate alternative starts at n0
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = []
    alt_starts.append(beatty_start(n0, golden / 2.0, rng))
    alt_starts.append(beatty_start(n0, np.sqrt(2.0) / 2.0, rng))
    alt_starts.append(beatty_start(n0, np.sqrt(3.0) / 2.0, rng))
    alt_starts.append(random_binary(n0, rng))
    alt_starts.append(random_binary(n0, rng))

    for h_alt in alt_starts:
        c_alt = c5_fft(h_alt, n0)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup at n0
    p1_end = t0 + min(180, budget_s * 0.2)
    alphas = [20, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    maxits = [800, 1000, 1200, 1500, 1800, 2000, 2500, 3000, 3500]
    h1, c1 = phase_lbfgs(best_h, best_n, p1_end, alphas, mu=0.01, maxit_list=maxits)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization at n0
    if time.time() < deadline - 200:
        p2_end = min(time.time() + 150, deadline - 180)
        h2, c2 = phase_active(best_h, best_n, p2_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient-guided coordinate descent
    if time.time() < deadline - 150:
        p3_end = min(time.time() + 120, deadline - 130)
        h3, c3 = grad_guided(best_h, best_n, p3_end,
            step_sizes=[0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=150000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Upsample to 2*n0 and optimize
    n2 = 2 * n0
    if time.time() < deadline - 250:
        h_up = upsample(best_h, n0, n2)
        # Also try fresh Beatty at n2
        h_beatty2 = beatty_start(n2, golden / 2.0, rng)
        h_beatty2b = beatty_start(n2, np.sqrt(2.0) / 2.0, rng)
        h_rand2 = random_binary(n2, rng)

        cands = [h_up, h_beatty2, h_beatty2b, h_rand2]
        cands_c = [c5_fft(hc, n2) for hc in cands]
        start_idx = int(np.argmin(cands_c))
        local_best_h = cands[start_idx].copy()
        local_best_c = cands_c[start_idx]

        p4a_end = min(time.time() + 100, deadline - 180)
        h4a, c4a = phase_lbfgs(local_best_h, n2, p4a_end,
            alphas=[50, 500, 5000, 50000, 500000, 2000000],
            mu=0.01, maxit_list=[500, 800, 1000, 1500, 2000, 2500])
        if c4a < local_best_c:
            local_best_c = c4a
            local_best_h = h4a

        if time.time() < deadline - 100:
            p4b_end = min(time.time() + 60, deadline - 80)
            h4b, c4b = phase_active(local_best_h, n2, p4b_end,
                top_k_list=[(6, 300000), (10, 1000000), (16, 5000000)])
            if c4b < local_best_c:
                local_best_c = c4b
                local_best_h = h4b

        if c4b if 'c4b' in dir() else c4a < best_c:
            pass
        local_best_c_final = c5_fft(local_best_h, n2)
        if local_best_c_final < best_c:
            best_c = local_best_c_final
            best_h = local_best_h
            best_n = n2

    # Phase 5: Gradient-guided at current best
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 80, deadline - 80)
        h5, c5 = grad_guided(best_h, best_n, p5_end,
            step_sizes=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_steps=100000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine pair search
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 50, deadline - 45)
        h6, c6 = fine_pair(best_h, best_n, p6_end, radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Binary swap search
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 30, deadline - 25)
        h7, c7 = swap_search(best_h, best_n, p7_end, max_rounds=20)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final fine tuning
    if time.time() < deadline - 10:
        p8_end = deadline - 3
        h8, c8 = fine_pair(best_h, best_n, p8_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: one more active-set pass with very high alpha
    if time.time() < deadline - 8:
        p9_end = deadline - 3
        h9, c9 = phase_active(best_h, best_n, p9_end,
            top_k_list=[(4, 5000000), (8, 20000000), (12, 50000000)])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = c5_fft(best_h, final_n)

    # Verify constraint
    assert abs(best_h.sum() * (2.0 / final_n) - 1.0) < 1e-8, f"Constraint violated: {best_h.sum() * (2.0/final_n)}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bounds violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, final_n)

    return best_h, float(best_c), final_n
```
