    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in active_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()

        k = idx - (n - 1)

            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active(c_vec, top_k=3, rel_tol=0.002):
    mv = c_vec.max()
    thr = mv - max(1e-10, rel_tol * mv)
    idx = np.where(c_vec >= thr)[0]
    if len(idx) > top_k:
        idx = idx[np.argsort(-c_vec[idx])[:top_k]]
    return list(idx)

def lbfgs_step(x0, n, act, alpha, mu, niter, dl):
    try:
        r = minimize(obj_active, x0, jac=grad_active,
                     args=(n, act, alpha, mu),
                     method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': niter, 'ftol': 1e-20,
                              'gtol': 1e-16, 'maxfun': niter * 10,
                              'maxls': 50})
        return project_feasible(r.x, n)
    except Exception:
        return x0.copy()

def pair_search(h, n, dl):
    bc = compute_c5(h, n)
    for d in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
        if time.time() > dl:
            break
        imp = True
        while imp and time.time() < dl:
            imp = False
            for i in range(n):
                if time.time() > dl:
                    return h, bc
                jlo = max(0, i - 40)
                jhi = min(n, i + 41)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < bc - 1e-16:
                            bc = c; imp = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < bc - 1e-16:
                            bc = c; imp = True
                        else:
                            h[i] += d; h[j] -= d
    return h, bc

def grad_pairs(h, n, dl):
    bc = compute_c5(h, n)
    for st in [0.003, 0.001, 0.0003, 0.0001, 5e-5]:
        if time.time() > dl:
            break
        imp = True
        while imp and time.time() < dl:
            imp = False
            cv = compute_c5_vec(h, n)
            act = get_active(cv, 3, 0.005)
            g = np.zeros(n)
            for a in act:
                g += shift_grad_single(h, n, a)
            g /= len(act)
            ordr = np.argsort(np.abs(g))
            for a in range(len(ordr)):
                if time.time() > dl:
                    return h, bc
                for b in range(a + 1, min(a + 150, len(ordr))):
                    i, j = ordr[a], ordr[b]
                    if g[i] < g[j]:
                        if h[j] - st >= 0 and h[i] + st <= 1:
                            h[j] -= st; h[i] += st
                            c = compute_c5(h, n)
                            if c < bc - 1e-16:
                                bc = c; imp = True
                            else:
                                h[j] += st; h[i] -= st
                    elif g[i] > g[j]:
                        if h[i] - st >= 0 and h[j] + st <= 1:
                            h[i] -= st; h[j] += st
                            c = compute_c5(h, n)
                            if c < bc - 1e-16:
                                bc = c; imp = True
                            else:
                                h[i] += st; h[j] -= st
    return h, bc

def perturb(h, n, bc, dl, sig):
    if time.time() > dl - 15:
        return h, bc
    hp = h + sig * np.random.randn(n)
    hp = project_feasible(hp, n)
    cv = compute_c5_vec(hp, n)
    act = get_active(cv, 3, 0.01)
