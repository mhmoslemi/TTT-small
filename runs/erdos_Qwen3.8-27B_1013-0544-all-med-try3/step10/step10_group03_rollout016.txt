```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pnorm_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-20)
    val = float(np.sum(c_pos ** p) ** (1.0 / p))
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def pnorm_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    c_pos = np.maximum(c, 1e-20)
    sp = np.sum(c_pos ** p)
    if sp < 1e-30:
        g = 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
        return g
    factor = sp ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1)
    g = np.zeros(n)
    n_corr = len(w)
    for j in range(n_corr):
        if w[j] < 1e-16:
            continue
        m = j - (n - 1)
        ww = w[j] * dx * factor
        if m >= 0:
            nv = n - m
            g[0:nv] += ww * (1.0 - x[m:n])
            g[m:n] -= ww * x[0:nv]
        else:
            kk = -m
            nv = n - kk
            g[kk:n] += ww * (1.0 - x[0:nv])
            g[0:nv] -= ww * x[kk:n]
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_pnorm_lbfgs(h, n, deadline, p_list, mu_list):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if time.time() > deadline - 10:
            break
        for mu in mu_list:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(pnorm_obj, best_h, jac=pnorm_grad,
                              args=(n, p, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': 20000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pnorm_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-20)
    val = float(np.sum(c_pos ** p) ** (1.0 / p))
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def pnorm_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    c_pos = np.maximum(c, 1e-20)
    sp = np.sum(c_pos ** p)
    if sp < 1e-30:
        return 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    factor = sp ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1)
    g = np.zeros(n)
    for j in range(len(w)):
        if w[j] < 1e-16:
            continue
        m = j - (n - 1)
        ww = w[j] * dx * factor
        if m >= 0:
            nv = n - m
            g[:nv] += ww * (1.0 - x[m:])
            g[m:] -= ww * x[:nv]
        else:
            kk = -m
            nv = n - kk
            g[kk:] += ww * (1.0 - x[:nv])
            g[:nv] -= ww * x[kk:]
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()

    w /= w.sum()
    dx = 2.0 / n
