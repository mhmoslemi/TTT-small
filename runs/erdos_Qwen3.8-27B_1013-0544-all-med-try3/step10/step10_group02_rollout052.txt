import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_k_obj(x, n, active, alpha, mu=0.0):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:

    if mu > 0:

def top_k_grad(x, n, active, alpha, mu=0.0):

    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

def periodic_pattern(n, period, ones_per_period, offset=0):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    pattern = np.zeros(period)
    # Spread the ones evenly within the period
    positions = np.round(np.arange(ones_per_period) * period / ones_per_period).astype(int) % period
    pattern[positions] = 1.0
    for i in range(n):
        h[i] = pattern[(i + offset) % period]
    return h

def evaluate_periodic_patterns(n, max_period=30):
    """Evaluate many periodic patterns and return the best one."""
    best_val = np.inf
    best_h = None
    for p in range(max_period):
        for m in range(1, p):  # m ones per period p
            for offset in range(p):
                h = periodic_pattern(n, p, m, offset)
                # Adjust to satisfy sum constraint
                target = n / 2.0
                diff = target - h.sum()
                if abs(diff) > 1e-10:
                    # Spread the adjustment
                    h_flat = h.copy()
                    if diff > 0:
                        idx = np.where(h_flat < 1.0)[0]
                        per = diff / max(1, len(idx))
                        h_flat[idx] = np.minimum(1.0, h_flat[idx] + per)
                    else:
                        idx = np.where(h_flat > 0.0)[0]
                        per = diff / max(1, len(idx))
                        h_flat[idx] = np.maximum(0.0, h_flat[idx] + per)
                    h = project_feasible(h_flat, n)
                c = compute_c5(h, n)
                if c < best_val:
                    best_val = c
                    best_h = h.copy()
    return best_h, best_val

def coordinate_descent(h, n, deadline, radius=50, step_sizes=None):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, max(1, n // 200)):  # Sample indices
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                # Try moving mass from i to j (or vice versa)

                    if i == j:
                        continue
                    # Transfer d from j to i
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Transfer d from i to j
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            best_h = np.clip(best_h, 0, 1)
            best_h = project_feasible(best_h, n)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def periodic_pattern(n, period, ones_per_period, offset=0):
    h = np.zeros(n)
    pattern = np.zeros(period)
    if ones_per_period <= 0 or ones_per_period >= period:
        pattern[:] = 0.5
    else:
        positions = np.round(np.arange(ones_per_period) * period / ones_per_period).astype(int) % period
        pattern[positions] = 1.0
    for i in range(n):
        h[i] = pattern[(i + offset) % period]
    return project_feasible(h, n)

def scan_periodic_patterns(n, max_period=35):
    best_val = np.inf

            h = periodic_pattern(n, p, m, 0)
            c = compute_c5(h, n)
            if c < best_val:

                best_h = h.copy()
    return best_h, best_val

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_per=None):

            if c_new < best_c - 1e-15:

            pass
    return best_h, best_c
