def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    """Downsample h from n_old to n_new points (by averaging blocks)."""
    if n_old % n_new == 0:
        block = n_old // n_new
        h_new = h.reshape(n_new, block).mean(axis=1)
    else:
        x_old = np.linspace(0, 2, n_old, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, h)

    best_n = n0
    
    # Phase 1: Aggressive active-set L-BFGS at original resolution
    p1_end = min(t0 + 120, deadline - 400)
    if p1_end > t0 + 30:
        h1, c1 = active_set_lbfgs(best_h, best_n, p1_end, top_k=12,
                                   alphas=[5000, 20000, 100000, 500000, 2000000, 10000000],
                                   rounds=25)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Gradient-guided pair optimization
    p2_end = min(time.time() + 100, deadline - 250)
    if p2_end > time.time() + 20:
        h2, c2 = gradient_pair_optimize(best_h, best_n, p2_end,
                                         step_scales=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                         max_steps=40000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Multi-lag pair optimization
    p3_end = min(time.time() + 80, deadline - 150)
    if p3_end > time.time() + 15:
        h3, c3 = multi_lag_pair_optimize(best_h, best_n, p3_end,
                                          deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                          max_iter=30000, top_lags=4)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try higher resolution (double)
    n_high = min(n0 * 2, 1200)
    if time.time() < deadline - 200 and n_high > n0:
        h_up = upsample_h(best_h, best_n, n_high)
        c_up = compute_c5(h_up, n_high)
        
        # Optimize at higher resolution
        p4_end = min(time.time() + 150, deadline - 80)
        if p4_end > time.time() + 30:
            h4, c4 = active_set_lbfgs(h_up, n_high, p4_end, top_k=12,
                                       alphas=[5000, 20000, 100000, 500000, 2000000],
                                       rounds=20)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = n_high
    
    # Phase 5: Fine pair search at current best
    p5_end = min(time.time() + 80, deadline - 60)
    if p5_end > time.time() + 15:
        h5, c5 = fine_grid_pair(best_h, best_n, p5_end,
                                 radius=max(20, best_n // 20),
                                 deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Active set polish with very high alpha
    p6_end = min(time.time() + 50, deadline - 30)
    if p6_end > time.time() + 10:
        h6, c6 = active_set_lbfgs(best_h, best_n, p6_end, top_k=8,
                                   alphas=[5000000, 20000000, 100000000],
                                   rounds=12)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP with equality constraint (for very tight polish)
    if time.time() < deadline - 20:
        h7, c7 = run_slqp(best_h, best_n, deadline - 5, alpha=10000000, maxiter=300)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Ultra-fine pair adjustments
    if time.time() < deadline - 5:
        h8, c8 = fine_grid_pair(best_h, best_n, deadline - 2,
                                 radius=max(10, best_n // 30),
                                 deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final: also try to improve at original n if we went to higher n
    if best_n > n0 and time.time() < deadline - 3:
        # Downsample back to n0 and see if it helps
        h_down = upsample_h(best_h, best_n, n0)  # This actually interpolates, not downsamples
        c_down = compute_c5(h_down, n0)
        if c_down < best_c:
            best_c = c_down
            best_h = h_down
            best_n = n0
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
