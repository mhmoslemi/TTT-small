import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_grad(h, n, active_set):
    """Compute gradient of average of active set overlaps."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active_set:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad / len(active_set)

def project_grad_descent(h, n, deadline, max_iter=50000):
    """Projected gradient descent on active set average."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_active = 10
    tol_frac = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        threshold = c_max - max(1e-9, tol_frac * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > n_active:
            active = np.array(np.argsort(-c_vec[active])[:n_active])
        
        grad = active_set_grad(best_h, n, active)
        
        # Backtracking line search
        step = 0.01
        for ls in range(20):
            new_h = best_h - step * grad
            new_h = project_feasible(new_h, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_c = new_c
                best_h = new_h
                break
            step *= 0.5
        else:
            # No improvement, try smaller steps
            step *= 0.5
            if step < 1e-8:
                # Try random perturbation
                break
    
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    """Upsample h from n_old to n_new points via linear interpolation and projection."""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(np.clip(h_new, 0, 1), n_new)

def binary_init(n, seed):

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0

def beatty_init(n, alpha_param):
    """Create a Beatty sequence initial solution."""
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 2 * n + n_ones):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return h
