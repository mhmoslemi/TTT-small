    v_hi = 1.0

        v_mid = (v_lo + v_hi) * 0.5
        vals = alpha + beta * v_mid
        max_val = vals.max()
        tol = 1e-13 * (1.0 + abs(max_val))
        active = (vals >= max_val - tol)
        if not active.any():
            active = (vals == max_val)
        ld = float(beta[active].min())
        rd = float(beta[active].max())
        if ld >= 0.0:
            v_hi = v_mid
        elif rd <= 0.0:
            v_lo = v_mid
        else:
            return v_mid
    cands = np.array([0.0, (v_lo + v_hi) * 0.5, 1.0])
    fav = np.array([(alpha + beta * v).max() for v in cands])
    return float(cands[np.argmin(fav)])

def coord_descent_sweep(h, n, rng, deadline):
    c_vec = compute_c5_vec(h, n)
    for i0 in rng.permutation(n):
        if time.time() > deadline:
            break
        m = 2 * n - 1
        alpha = c_vec.copy()
        beta = np.zeros(m)
        v1_lo = n - 1 - i0
        v1_hi = 2 * n - 2 - i0
        if v1_hi >= v1_lo:
            idx1 = np.arange(v1_lo, v1_hi + 1)
            i1ps = i0 + idx1 - n + 1
            h_ip = h[i1ps]
            alpha[idx1] -= h[i0] * (1.0 - h_ip)
            beta[idx1] += (1.0 - h_ip)
        v2_lo = i0
        v2_hi = i0 + n - 1
        if v2_hi >= v2_lo:
            idx2 = np.arange(v2_lo, v2_hi + 1)
            i2ms = i0 - idx2 + n - 1
            h_im = h[i2ms]
            alpha[idx2] -= h_im * (1.0 - h[i0])
            alpha[idx2] += h_im
            beta[idx2] -= h_im
        v_opt = find_min_max_lines(alpha, beta)
        h[i0] = v_opt
    return h

def run_coord_descent(h_init, n, n_sweeps, deadline, proj_every=10):
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        h = coord_descent_sweep(h, n, rng, deadline)
        if (sweep + 1) % proj_every == 0:
            h = project_feasible(h, n)
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
        if (sweep + 1) % 50 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
    h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h_init, n, deadline, alphas, mu=0.005, maxiter_per=None):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    def lb_obj(x):
        return smooth_max_obj(x, n, 50000) + mu * (x.sum() - n / 2.0) ** 2

    def lb_grad(x):
        g = smooth_max_grad(x, n, 50000)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 2))

        def sm_obj_a(x, a=alpha):
            return smooth_max_obj(x, n, a) + mu * (x.sum() - n / 2.0) ** 2

        def sm_grad_a(x, a=alpha):
            g = smooth_max_grad(x, n, a)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g

        try:
            res = minimize(sm_obj_a, best_h, jac=sm_grad_a,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h_init, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h_init, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h_init.copy(), n), compute_c5(h_init, n)

def active_set_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def active_set_grad(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_active_set(h_init, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000]

    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            idx_sorted = np.argsort(-c_vec[active])
            active = active[idx_sorted[:top_k]]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_vec))])

        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(300, int((deadline - time.time()) * 2))

            def as_obj(x, a=alpha, act=active):
                return active_set_obj(x, n, act, a) + 0.003 * (x.sum() - n / 2.0) ** 2

            def as_grad(x, a=alpha, act=active):
                g = active_set_grad(x, n, act, a)
                g += 2.0 * 0.003 * (x.sum() - n / 2.0)
                return g

            try:
                res = minimize(as_obj, best_h, jac=as_grad,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def perturb_and_cd(h, n, deadline, rng, n_sweeps=50, noise_scale=0.01):
    h_pert = h + rng.randn(n) * noise_scale
    h_pert = project_feasible(h_pert, n)
    h_res, c_res = run_coord_descent(h_pert, n, n_sweeps, deadline, proj_every=10)
    return h_res, c_res

def make_periodic_binary(n, period, ones_count):
    h = np.zeros(n)
    for p in range(ones_count):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Coordinate descent at original resolution
    p1_deadline = t0 + 250
    h1, c1 = run_coord_descent(best_h, best_n, 2000, p1_deadline, proj_every=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: L-BFGS polish at original resolution
    if time.time() < deadline - 200:
        p2_deadline = min(time.time() + 100, deadline - 200)
        h2, c2 = run_lbfgs(best_h, best_n, p2_deadline,
                           alphas=[1000, 10000, 50000, 200000, 1000000],
                           mu=0.003,
                           maxiter_per=[1500, 2000, 2500, 3000, 3000])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Active set refinement
    if time.time() < deadline - 200:
        p3_deadline = min(time.time() + 80, deadline - 200)
        h3, c3 = run_active_set(best_h, best_n, p3_deadline, top_k=6,
                                alphas=[20000, 100000, 500000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Coordinate descent again (after L-BFGS changed the landscape)
    if time.time() < deadline - 180:
        p4_deadline = min(time.time() + 80, deadline - 180)
        h4, c4 = run_coord_descent(best_h, best_n, 1500, p4_deadline, proj_every=5)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try higher resolution n=1000
    n_high = 1000
    if time.time() < deadline - 250:
        x_fine = np.linspace(0, 2, n_high)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_high)
        c_fine = compute_c5(h_fine, n_high)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine.copy()
            best_n = n_high

        p5_deadline = min(time.time() + 200, deadline - 150)
        h5, c5 = run_coord_descent(best_h, best_n, 1000, p5_deadline, proj_every=10)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = best_n

    # Phase 6: SLSQP polish at best resolution
    if time.time() < deadline - 80:
        p6_deadline = min(time.time() + 60, deadline - 60)
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 50:
                break
            h6, c6 = run_slqp(best_h, best_n, p6_deadline, alpha=alpha, maxiter=800)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: More coordinate descent after SLSQP
    if time.time() < deadline - 60:
        p7_deadline = min(time.time() + 50, deadline - 40)
        h7, c7 = run_coord_descent(best_h, best_n, 1000, p7_deadline, proj_every=5)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Perturbation + coordinate descent (try to escape local optima)
    if time.time() < deadline - 50:
        for trial in range(5):
            if time.time() > deadline - 35:
                break
            p8_dl = min(time.time() + 8, deadline - 25)
            ns = max(20, int((p8_dl - time.time()) * 50))
            npert = max(1, min(3, int((deadline - time.time()) / 15)))
            scale = 0.005 / (trial + 1)
            h8, c8 = perturb_and_cd(best_h, best_n, p8_dl, rng,
                                    n_sweeps=npert, noise_scale=scale)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Phase 9: Final SLSQP + coordinate descent
    if time.time() < deadline - 15:
        p9_deadline = min(time.time() + 12, deadline - 5)
        h9, c9 = run_slqp(best_h, best_n, p9_deadline, alpha=500000, maxiter=500)
        if c9 < best_c:
            best_c = c9
            best_h = h9
        if time.time() < deadline - 3:
            h9b, c9b = run_coord_descent(best_h, best_n, 300, deadline - 2, proj_every=3)
            if c9b < best_c:
                best_c = c9b
                best_h = h9b

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
