import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

# Soft-max objective and gradient
def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        if n_ones < n:
            h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slp_obj_fn(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiters=None):

    best_c = compute_c5(best_h, n)

                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})

def run_slqp(h, n, deadline, alpha=200000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(slp_obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def slp_iterate(h, n, deadline, top_k=25, n_iters=200):
    """Successively linearized programming: linearize bilinear terms, solve LP/QP."""
    import cvxpy as cp
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = np.argsort(-c_vec)[:top_k]
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for idx in active:
            shift = idx - (n - 1)
            if shift >= 0:
                valid = n - shift
                weights = (1.0 - best_h[shift:n]) * dx
                overlap = h_var[0:valid] @ weights
            else:
                kk = -shift
                valid = n - kk
                weights = (1.0 - best_h[0:valid]) * dx
                overlap = h_var[kk:n] @ weights
            constraints.append(overlap <= t)
        obj = cp.Minimize(t + 0.001 * cp.sum_squares(h_var - best_h))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, rel_tol=1e-10, abs_tol=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=5000)
            except Exception:
                break
        if h_var.value is not None:
            h_new = np.clip(np.array(h_var.value, dtype=float), 0, 1)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif it > n_iters * 0.7:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] += remaining
    return project_feasible(h_bin, n)

def periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: L-BFGS soft-max at original resolution
    p1_end = t0 + 320
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01, maxiters=[2000, 2000, 2500, 3000, 3000, 3500, 4000, 5000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active-set focused
    if time.time() < deadline - 380:
        p2_end = min(time.time() + 140, deadline - 380)
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=8,
            alphas=[10000, 50000, 200000, 1000000], n_rounds=30)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLP (successively linearized programming)
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 120, deadline - 280)
        h3, c3 = slp_iterate(best_h, best_n, p3_end, top_k=30, n_iters=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Shake and refine
    if time.time() < deadline - 220:
        p4_end = min(time.time() + 80, deadline - 200)
        h4, c4 = shake_refine(best_h, best_n, p4_end, n_shakes=10,
                             shake_scale=0.07, rng=rng)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: SLP again after shake
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 60, deadline - 160)
        h5, c5 = slp_iterate(best_h, best_n, p5_end, top_k=30, n_iters=100)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: SLSQP polish
    if time.time() < deadline - 120:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 105:
                break
            p6_end = min(time.time() + 25, deadline - 95)
            h6, c6 = run_slqp(best_h, best_n, p6_end, alpha=alpha, maxiter=800)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Subgradient refinement
    if time.time() < deadline - 70:
        p7_end = min(time.time() + 50, deadline - 55)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=8000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine pair search
    if time.time() < deadline - 50:
        p8_end = min(time.time() + 45, deadline - 25)
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=40)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Upsample to n=1000 and optimize
    n_fine = 1000
    remaining = deadline - time.time()
    if remaining > 100:
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)

        p9_end = deadline - 30
        h9, c9 = run_lbfgs(h_fine, n_fine, p9_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
            mu=0.005, maxiters=[800, 1200, 1500, 1500, 2000, 2000, 2500])

        # SLP at fine resolution
        if time.time() < deadline - 20:
            p9b_end = min(time.time() + 20, deadline - 12)
            h9b, c9b = slp_iterate(h9, n_fine, p9b_end, top_k=20, n_iters=50)
            if c9b < c9:
                h9 = h9b
                c9 = c9b

        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = n_fine

    # Final: quick fine pair + project
    if time.time() < deadline - 5:
        p10_end = deadline - 3
        h10, c10 = fine_pair_search(best_h, best_n, p10_end, radius=20)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
