    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]

    best_h = h.copy()

    rng = np.random.RandomState(seed)
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 7)):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = sorted(ones)
                for j in o_list:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range

                    if 0 <= k1 < 2*n - 1:
                        delta[k1] += 1.0

                    if 0 <= k2 < 2*n - 1:
                        delta[k2] += 1.0

                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:

            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def _obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def _cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(h, n, zi, oj, m_range):
    delta = np.zeros(2 * n - 1)
    im = zi + m_range
    imn = zi - m_range
    jm = oj + m_range
    jmn = oj - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (oj - zi)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (zi - oj)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_sa(h_init, n, deadline, T0=0.3, T_min=0.00001, seed=42,
              n_iter_per_restart=300000, n_restarts=5):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = np.inf
    m_range = np.arange(-(n - 1), n)
    top_n = n // 2

    for restart in range(n_restarts):
        if time.time() > deadline:
            break

        if restart == 0:
            h = np.zeros(n)
            idx = np.argsort(h_init)[::-1]
            h[idx[:top_n]] = 1.0
        else:
            if best_h is not None:
                h = best_h.copy()
            else:
                h = np.zeros(n)
                idx = np.argsort(h_init)[::-1]
                h[idx[:top_n]] = 1.0
            n_perturb = rng.randint(max(1, top_n // 15), max(2, top_n // 4))
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                perm_o = rng.permutation(len(oi))[:n_perturb]
                perm_z = rng.permutation(len(zi))[:n_perturb]
                for p in range(n_perturb):
                    a = oi[perm_o[p]]
                    b = zi[perm_z[p]]
                    h[a], h[b] = 0.0, 1.0
            h = np.where(h >= 0.5, 1.0, 0.0)
            cnt = int(h.sum())
            if cnt > top_n:
                excess = np.where(h == 1)[0]
                to_flip = rng.choice(excess, cnt - top_n, replace=False)
                h[to_flip] = 0.0
            elif cnt < top_n:
                deficit = np.where(h == 0)[0]
                to_flip = rng.choice(deficit, top_n - cnt, replace=False)
                h[to_flip] = 1.0

        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = float(corr.max()) * dx

        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()

        for it in range(n_iter_per_restart):
            if time.time() > deadline:
                break
            frac = it / n_iter_per_restart
            T = T0 * (T_min / T0) ** frac

            ii = rng.randint(len(zeros_list))
            jj = rng.randint(len(ones_list))
            zi_val = zeros_list[ii]
            oj_val = ones_list[jj]

            delta = compute_swap_delta(h, n, zi_val, oj_val, m_range)
            new_corr = corr + delta
            nc = float(new_corr.max()) * dx
            dc = nc - cur_c

            if dc < 0 or (T > 1e-10 and rng.random() < np.exp(-dc / T)):
                h[zi_val] = 1.0
                h[oj_val] = 0.0
                corr = new_corr
                cur_c = nc
                zeros_list[ii] = oj_val
                ones_list[jj] = zi_val
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()

    return best_h, best_c

def binary_greedy(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            current_h = best_h.copy()
            n_perturb = rng.randint(3, 8)
            for _ in range(n_perturb):
                oi_l = np.where(current_h == 1)[0]
                zi_l = np.where(current_h == 0)[0]
                if len(oi_l) > 0 and len(zi_l) > 0:
                    a = rng.choice(oi_l)
                    b = rng.choice(zi_l)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = float(corr.max()) * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = sorted(ones)
                for j in o_list:
                    delta = compute_swap_delta(h, n, i, j, m_range)
                    new_corr = corr + delta
                    nc = float(new_corr.max()) * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    return best_h, best_c

def iterated_local_search(h_init, n, deadline, n_iter_ils=20, perturbation_range=(5, 25), seed=42):
    """Iterated local search: perturb + greedy."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    target = n // 2

    h = np.zeros(n)

    h[idx[:target]] = 1.0

    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c

    for it in range(n_iter_ils):
        if time.time() > deadline:
            break

        # Greedy to converge
        h = best_h.copy()
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())
        converged = False
        for sweep in range(2000):
            if time.time() > deadline:
                converged = True
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    converged = True
                    break
                o_list = sorted(ones)
                for j in o_list:
                    delta = compute_swap_delta(h, n, i, j, m_range)
                    new_corr = np.correlate(h, 1 - h, mode='full').copy() + delta
                    nc = float(new_corr.max()) * dx
                    if nc < cur_c - 1e-14:

                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved or converged:
                    break
            if not improved or converged:
                break

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # Perturb
        n_perturb = rng.randint(perturbation_range[0], perturbation_range[1])
        h_disturbed = best_h.copy()
        oi = np.where(h_disturbed == 1)[0]
        zi = np.where(h_disturbed == 0)[0]
        for _ in range(n_perturb):
            if len(oi) == 0 or len(zi) == 0:
                break
            a = rng.choice(oi)
            b = rng.choice(zi)
            h_disturbed[a], h_disturbed[b] = 0.0, 1.0
        # Re-binarize and fix count
        h_disturbed = np.where(h_disturbed >= 0.5, 1.0, 0.0)
        cnt = int(h_disturbed.sum())
        if cnt > target:
            excess = np.where(h_disturbed == 1)[0]
            to_flip = rng.choice(excess, cnt - target, replace=False)

            h_disturbed[to_flip] = 1.0

        h = h_disturbed
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = float(corr.max()) * dx

    return best_h, best_c

    for nh in [500, 750]:

            continue
        if time.time() > p5 - 50:
            break
        h_struct = structured_init(nh, seed=seed)
        h_mod = best_mod_pattern(nh, seed=seed + 50)
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nh)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_interp = np.zeros(nh)
        idx = np.argsort(h_interp)[::-1]
        h_bin_interp[idx[:nh // 2]] = 1.0

        for h_init in [h_struct, h_mod, h_bin_interp]:
            if time.time() > p5 - 15:
                break
            c_init = compute_c5(h_init, nh)
            if c_init < best_c:
                best_c = c_init
                best_h = h_init.copy()
                best_n = nh
            hsa, csa = binary_sa(h_init, nh, min(time.time() + 40, p5 - 10),
                                 T0=0.3, T_min=0.0001, seed=seed + nh,
                                 n_iter_per_restart=300000, n_restarts=3)
            if csa < best_c:
                best_c = csa
                best_h = hsa.copy()
                best_n = nh
            hg, cg = binary_greedy(hsa, nh, min(time.time() + 20, p5 - 5),
                                   max_sweeps=500, n_restarts=2, seed=seed + nh + 1)
            if cg < best_c:
                best_c = cg
                best_h = hg.copy()
                best_n = nh

    # Phase 6: Final SA + greedy at best resolution (remaining time)
    if time.time() < dl - 40:
        p6a = dl - 40
        hb, cb = binary_sa(best_h, best_n, p6a, T0=0.25, T_min=0.000005,
                           seed=seed+600, n_iter_per_restart=500000, n_restarts=4)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    if time.time() < dl - 5:
        hb, cb = binary_greedy(best_h, best_n, dl, max_sweeps=1000, n_restarts=3, seed=seed+700)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final SLSQP polish
    if time.time() < dl - 2:
        rng_final = np.random.RandomState(seed + 800)
        hp = best_h.copy()
        if (hp == 0).sum() > 0 and (hp == 1).sum() > 0:
            hp = np.clip(hp + 0.03 * rng_final.randn(best_n), 0, 1)
            hp = project_feasible(hp, best_n)
            hp = slsqp_smooth(hp, best_n, 2000000, 2000, dl - 1)
            cp = compute_c5(hp, best_n)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
