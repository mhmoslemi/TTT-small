def compute_deltas(h, n):
    """Precompute delta vectors for all zeros (0->1) and ones (1->0)."""
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    ncorr = 2*n - 1
    offset = n - 1
    
    delta_zeros = np.zeros((len(zeros_idx), ncorr))
    delta_ones = np.zeros((len(ones_idx), ncorr))
    
    for p, i in enumerate(zeros_idx):
        # Setting h[i] from 0 to 1
        # m=i term: +(1-h[i+l]) for l where 0<=i+l<n
        #   l ranges from -i to n-1-i
        #   i+l ranges from 0 to n-1
        lo = max(0, -i)  # l >= -i means i+l >= 0... wait
        # Actually: 0 <= i+l < n means -i <= l < n-i
        l_min = max(-(n-1), -i)  # l >= -i and l >= -(n-1)
        l_max = min(n-1, n-1-i)  # l < n-i means l <= n-1-i, and l <= n-1
        # l from l_min to l_max
        if l_max >= l_min:
            pos = i + np.arange(l_min, l_max+1)  # positions i+l
            delta_zeros[p, np.arange(l_min, l_max+1) + offset] += (1 - h[pos])

        # 0 <= i-l < n means i-n < l <= i, and l != 0
        l_min2 = max(-(n-1), i-n+1)  # i-l >= 0 means l <= i; i-l < n means l > i-n

            l_range = l_range[l_range != 0]
            if len(l_range) > 0:
                pos2 = i - l_range  # positions i-l
                delta_zeros[p, l_range + offset] -= h[pos2]
    
    for q, j in enumerate(ones_idx):
        # Setting h[j] from 1 to 0
        # m=j term: -(1-h[j+l]) for l where 0<=j+l<n
        l_min = max(-(n-1), -j)
        l_max = min(n-1, n-1-j)

    return delta_zeros, delta_ones, zeros_idx, ones_idx

                    new_corr[j - i + offset] += 1
                    new_corr[i - j + offset] += 1
                    nm = new_corr.max()
                    if nm < cur_max - 1e-15:
                        # Accept swap
                        h[i] = 1.0
                        h[j] = 0.0
                        corr = new_corr
                        cur_max = nm
                        improved = True
                        break
                if improved:
                    break
        
            if not improved:
                break

            best_c = cur_max * dx
            best_h = h.copy()

                    dz[p, L + offset] += 1.0 - h[i + L]
                lo2 = max(-(n - 1), i - n + 1)
                hi2 = min(n - 1, i)
                L2 = np.arange(lo2, hi2 + 1)
                L2 = L2[L2 != 0]
                if len(L2) > 0:
                    dz[p, L2 + offset] -= h[i - L2]
            
            # Compute delta vectors for ones (1->0)
            do = np.zeros((no, ncorr), dtype=np.float64)
            for q in range(no):
                j = int(ones_idx[q])
                lo = max(0, -j)
                hi = min(n - 1, n - 1 - j)
                L = np.arange(lo, hi + 1)
                L = L[L != 0]
                if len(L) > 0:
                    do[q, L + offset] -= 1.0 - h[j + L]
                lo2 = max(-(n - 1), j - n + 1)
                hi2 = min(n - 1, j)
                L2 = np.arange(lo2, hi2 + 1)
                L2 = L2[L2 != 0]
                if len(L2) > 0:
                    do[q, L2 + offset] += h[j - L2]
            
            # Vectorized swap search
            k_star = corr.argmax()
            improved = False
            
            for p in range(nz):
                if time.time() > deadline:
                    break
                i = int(zeros_idx[p])
                base = corr + dz[p]
                
                # Build all candidate new correlations
                all_new = base[None, :] + do  # (no, ncorr)
                
                # Add cross corrections: +1 at lag j-i and i-j for each q
                idx1 = ones_idx - i + offset
                idx2 = i - ones_idx + offset
                ar = np.arange(no)
                all_new[ar, idx1] += 1.0
                all_new[ar, idx2] += 1.0
                
                new_maxes = all_new.max(axis=1)
                candidates = np.where(new_maxes < cur_max - 1e-14)[0]
                
                if len(candidates) > 0:
                    q_best = candidates[np.argmin(new_maxes[candidates])]
                    j = int(ones_idx[q_best])
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = all_new[q_best].copy()
                    cur_max = corr.max()
                    improved = True
                    break
            
            if not improved:
                break
        
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    
    return best_h, best_c

def subgradient_optimize(h, n, deadline, n_iter=50000):
    """Subgradient method with projection for the continuous problem."""
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 1.0 / (np.sqrt(n_iter) * n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        m = corr.max()
        k_star = corr.argmax()
        lag = k_star - (n - 1)
        
        # Subgradient of corr_{k*} w.r.t. h
        grad = np.zeros(n)
        for j in range(n):
            # d/dh[j] of sum_i h[i]*(1-h[i+k*])
            # = (1-h[j+k*]) if j+k* valid - h[j-k*] if j-k* valid
            val = 0.0
            if 0 <= j + lag < n:
                val += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                val -= h[j - lag]
            grad[j] = val
        grad *= dx
        
        # Move in negative gradient direction
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step = step * 0.999
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    rng = np.random.RandomState(seed)
    
    # Phase 1: Quick binary search from parent (10s)
    p1 = min(t0 + 15, dl - 100)
    hb, cb = binary_local_search(best_h, n, p1, max_sweeps=300, n_restarts=2, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: SLSQP continuous optimization from binary solution (60s)
    p2 = min(t0 + 75, dl - 80)
    h = best_h.copy() + 0.1 * rng.randn(n)
    h = project_feasible(h, n)
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() > p2:
            break
        h = slsqp_optimize(h, n, alpha, 2000, p2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Binary local search from best (200s)
    p3 = min(t0 + 275, dl - 60)
    hb, cb = binary_local_search(best_h, n, p3, max_sweeps=8000, n_restarts=4, seed=seed + 1)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 4: SLSQP polish (80s)
    p4 = min(t0 + 355, dl - 50)
    if time.time() < p4:
        h = best_h + 0.05 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [10000, 50000, 200000]:
            if time.time() > p4:
                break
            h = slsqp_optimize(h, n, alpha, 3000, p4)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 5: Try different resolutions (180s)
    p5 = min(t0 + 535, dl - 40)
    for nh in [300, 350, 400, 500]:
        if time.time() > p5 - 20:
            break
        if nh == n:
            continue
        if nh % 2 != 0:
            continue
        xo = np.linspace(0, 2, n + 1)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        
        # Quick binary search at new resolution
        hh_b, ch_b = binary_local_search(hh, nh, p5 - 15, max_sweeps=2000, n_restarts=2, seed=seed + nh)
        
        # Quick SLSQP
        if time.time() < p5 - 10:
            hh_s = slsqp_optimize(hh_b + 0.05 * rng.randn(nh), nh, 10000, 1000, p5 - 5)
            ch_s = compute_c5(hh_s, nh)
            if ch_s < min(best_c, ch_b):
                ch_b = ch_s
                hh_b = hh_s
        
        if ch_b < best_c:
            best_c = ch_b
            best_h = hh_b.copy()
            n = nh
    
    # Phase 6: Extended binary search on best (200s)
    p6 = min(t0 + 735, dl - 30)
    hb, cb = binary_local_search(best_h, n, p6, max_sweeps=15000, n_restarts=6, seed=seed + 100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 7: Final SLSQP + binary polish (remaining time)
    if time.time() < dl - 15:
        h = best_h + 0.02 * rng.randn(n)
        h = project_feasible(h, n)
        h = slsqp_optimize(h, n, 500000, 5000, dl - 12)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if time.time() < dl - 5:
            hb, cb = binary_local_search(best_h, n, dl, max_sweeps=5000, n_restarts=3, seed=seed + 200)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
