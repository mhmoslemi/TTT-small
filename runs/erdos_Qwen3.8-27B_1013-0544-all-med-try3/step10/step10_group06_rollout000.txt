                        d = -min(0.005, (h[j] - h[i]) * 0.1)
                    else:
                        continue
                    if abs(d) < 1e-10:
                        continue
                    hi = h[i]
                    hj = h[j]
                    h[i] = hi - d
                    h[j] = hj + d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    h[i] = hi
                    h[j] = hj
                    if c_test < best_c:
                        h[i] = hi - d
                        h[j] = hj + d
                        h = project_feasible(h, n)
                        best_c = c_test
                        best_h = h.copy()
                        improved_this_sweep = True
                        break
        
        if improved_this_sweep:
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                # Perturb and restart
                h = best_h + rng_coord.normal(0, 0.005, n)
                h = project_feasible(h, n)
                stuck = 0
            if stuck > 100:
                break
    
    return best_h, best_c

def interleave_opt(h, n, deadline, rng):
    """Interleave coordinate descent and gradient steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 50000
    cd_interval = 50  # do CD every N gradient steps
    
    total_iters = 0
    while time.time() < deadline:
        block_time = min(30, deadline - time.time())
        if block_time < 2:
            break
        
        # Gradient phase
        h_opt = lbfgsb_opt(h, n, alpha, time.time() + min(block_time * 0.6, 20), maxiter=300)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
            # Increase alpha for sharper max approximation
            alpha = min(alpha * 1.3, 5000000)
        else:
            alpha = max(alpha * 0.8, 5000)
            # Try perturbation
            h = best_h + rng.normal(0, 0.003, n)
            h = project_feasible(h, n)
        
        # Coordinate descent phase
        if time.time() < deadline - 5:
            h_cd, c_cd = coord_descent_fast(best_h, n, min(time.time() + 15, deadline), 
                                             K=3, max_sweeps=3000)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd.copy()
                h = h_cd.copy()
        
        total_iters += 1
        
    return best_h, best_c

def binary_construction(n, rng):
    """Construct a binary sequence with good autocorrelation using greedy method."""
    h = np.zeros(n)
    ones_needed = n // 2
    # Start with evenly spaced
    spacing = n / ones_needed
    for i in range(ones_needed):
        h[int(i * spacing)] = 1.0
    h = project_feasible(h, n)
    
    # Greedy improvement: try flipping pairs to reduce max lag
    best_c = compute_c5(h, n)
    for iteration in range(200):
        c_all = compute_all_lags(h, n)
        worst_k = np.argmax(c_all)
        lag = worst_k - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if m >= n:
            break
        improved = False
        for i in range(n - m):
            j = i + m
            if h[i] > 0.5 and h[j] < 0.5:
                h[i] = 0.0
                h[j] = 1.0
            elif h[j] > 0.5 and h[i] < 0.5:
                h[j] = 0.0
                h[i] = 1.0
            else:
                continue
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                improved = True
            else:
                # Undo
                h[i], h[j] = h[j], h[i] if False else (h[i], h[j])
                # Need to undo properly
                break
        if not improved:
            # Try random pair flips
            a = rng.randint(n)
            b = rng.randint(n)
            if h[a] > 0.5 and h[b] < 0.5:
                h[a], h[b] = h[b], h[a]
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                else:
                    h[a], h[b] = h[b], h[a]
                    h = project_feasible(h, n)
            if not improved:
                continue
    return project_feasible(h, n)

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    n = len(best_h)
    
    # Phase 1: Quick L-BFGS-B to establish baseline
    if time.time() < deadline - 600:
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 500:
                break
            h_opt = lbfgsb_opt(best_h, n, alpha, time.time() + 25, maxiter=500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 2: Fourier-based alternative start
    if time.time() < deadline - 500:
        for attempt in range(3):
            h_f = fourier_init(n, rng, n_terms=12 + attempt * 3)
            c_f = compute_c5(h_f, n)
            # Optimize from Fourier start
            h_ff = lbfgsb_opt(h_f, n, 20000, time.time() + 15, maxiter=400)
            c_ff = compute_c5(h_ff, n)
            if c_ff < best_c:
                best_c = c_ff
                best_h = h_ff
    
    # Phase 3: Main coordinate descent (deterministic, thorough)
    if time.time() < deadline - 300:
        cd_deadline = min(time.time() + 250, deadline - 250)
        h_cd, c_cd = coord_descent_fast(best_h, n, cd_deadline, K=4, max_sweeps=15000)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Phase 4: Interleave L-BFGS-B and coordinate descent
    if time.time() < deadline - 200:
        il_deadline = min(time.time() + 180, deadline - 180)
        h_il, c_il = interleave_opt(best_h, n, il_deadline, rng)
        if c_il < best_c:
            best_c = c_il
            best_h = h_il
    
    # Phase 5: Aggressive local search on top lags
    if time.time() < deadline - 100:
        h_work = best_h.copy()
        ls_deadline = min(time.time() + 80, deadline - 80)
        for _ in range(2000):
            if time.time() > ls_deadline:
                break
            c_all = compute_all_lags(h_work, n)
            # Always attack the single worst lag
            worst_k = np.argmax(c_all)
            lag = worst_k - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            improved = False
            n_pairs = n - m
            # Try all pairs with small step
            for i in range(n_pairs):
                j = i + m
                if h_work[i] > h_work[j] + 1e-7:
                    d = min(0.008, (h_work[i] - h_work[j]) * 0.2)
                elif h_work[j] > h_work[i] + 1e-7:
                    d = -min(0.008, (h_work[j] - h_work[i]) * 0.2)
                else:
                    continue
                if abs(d) < 1e-11:
                    continue
                hi = h_work[i]
                hj = h_work[j]
                h_work[i] = hi - d
                h_work[j] = hj + d
                h_work = project_feasible(h_work, n)
                c_test = compute_c5(h_work, n)
                if c_test < best_c:
                    best_c = c_test
                    best_h = h_work.copy()
                    improved = True
                else:
                    h_work[i] = hi
                    h_work[j] = hj
                    h_work = project_feasible(h_work, n)
                if improved:
                    break
            if not improved:
                # Try next worst lag
                c_all[worst_k] = -1
                worst_k = np.argmax(c_all)
                lag2 = worst_k - (n - 1)
                if lag2 == 0:
                    break
                m2 = abs(lag2)
                if m2 >= n:
                    break
                for i in range(n - m2):
                    j = i + m2
                    if h_work[i] > h_work[j] + 1e-7:
                        d = min(0.004, (h_work[i] - h_work[j]) * 0.15)
                    elif h_work[j] > h_work[i] + 1e-7:
                        d = -min(0.004, (h_work[j] - h_work[i]) * 0.15)
                    else:
                        continue
                    if abs(d) < 1e-11:
                        continue
                    hi = h_work[i]
                    hj = h_work[j]
                    h_work[i] = hi - d
                    h_work[j] = hj + d
                    h_work = project_feasible(h_work, n)
                    c_test = compute_c5(h_work, n)
                    if c_test < best_c:
                        best_c = c_test
                        best_h = h_work.copy()
                        improved = True
                    else:
                        h_work[i] = hi
                        h_work[j] = hj
                        h_work = project_feasible(h_work, n)
                    if improved:
                        break
    
    # Phase 6: Try a different resolution
    if time.time() < deadline - 80:
        for n_try in [n - 50, n + 100, n + 200]:
            if n_try < 200 or n_try > 1500:
                continue
            if time.time() > deadline - 50:
                break
            h_exp = expand_resolution(best_h, n, n_try)
            h_opt = lbfgsb_opt(h_exp, n_try, 50000, time.time() + 20, maxiter=800)
            c_opt = compute_c5(h_opt, n_try)
            # Quick coordinate descent
            h_cd2, c_cd2 = coord_descent_fast(h_opt, n_try, min(time.time() + 15, deadline - 30),
                                               K=3, max_sweeps=2000)
            if c_cd2 < best_c:
                best_c = c_cd2
                best_h = h_cd2

    if time.time() < deadline - 20:

        c_opt = compute_c5(h_opt, n_fin)
        if c_opt < best_c:

            best_h = h_opt
    
    # Phase 8: Final micro coordinate descent
    if time.time() < deadline - 3:
        n_fin = len(best_h)
        h_work = best_h.copy()
        for _ in range(500):

            c_all = compute_all_lags(h_work, n_fin)

            m = abs(lag)

            improved = False
            for i in range(n_fin - m):

                h_work[i] = hi - d
                h_work[j] = hj + d
                h_work = project_feasible(h_work, n_fin)
                c_test = compute_c5(h_work, n_fin)

                    h_work[j] = hj
                    h_work = project_feasible(h_work, n_fin)
                if improved:

            if not improved:
                break
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
