def focused_binary_search(h, n, deadline, max_sweeps=5000, seed=42):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        
        # Find worst lag(s)
        k_max = np.argmax(corr)
        lag = k_max - (n-1)  # actual lag
        
        # Try targeted swaps for worst lag
        candidates = []
        if lag >= 0:
            # h[i] * (1-h[i+lag]) for i in [0, n-1-lag]
            for i in range(n - lag):
                if h[i] == 1 and h[i+lag] == 0:
                    candidates.append((i, i+lag))
        else:
            lag_abs = -lag
            for i in range(lag_abs, n):
                if h[i] == 1 and h[i-lag_abs] == 0:
                    candidates.append((i, i-lag_abs))
        
        # Also try random swaps
        if not candidates:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                for _ in range(min(50, len(ones)*len(zeros))):
                    candidates.append((rng.choice(zeros), rng.choice(ones)))
        
        # Try candidates in random order
        rng.shuffle(candidates)
        for (i, j) in candidates:
            if time.time() > deadline:
                break
            if h[i] != 0 or h[j] != 1:
                continue
            # Compute delta
            delta = compute_delta(h, n, i, j)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = nc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        
        # Also try some random swaps
        if not improved and len(candidates) < 100:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 5 and len(zeros) > 5:
                for _ in range(10):
                    i = rng.choice(zeros)
                    j = rng.choice(ones)
                    if h[i] != 0 or h[j] != 1:
                        continue
                    delta = compute_delta(h, n, i, j)
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        improved = True
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                        break
        
        if not improved:
            # Try accepting a slightly worse move (simulated annealing style)
            # Or just break
            break
    
    return best_h, best_c

    delta = np.zeros(2*n-1)
    # h_new = h with h[i]=1, h[j]=0
    # h_old = h with h[i]=0, h[j]=1
    
    # First factor changes:
    # At position i (0->1): adds (1 - h_new[i + m - (n-1)])
    for m in range(2*n-1):
        idx = i + m - (n-1)
        if 0 <= idx < n:
            val = 1.0 if idx == i else (0.0 if idx == j else h[idx])
            # h_new[idx]: if idx==i, h_new=1; if idx==j, h_new=0; else h[idx]
            h_new_idx = 1.0 if idx == i else (0.0 if idx == j else h[idx])
            delta[m] += 1.0 - h_new_idx
    
    # At position j (1->0): removes (1 - h_old[j + m - (n-1)])
    for m in range(2*n-1):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
