import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _shift_grad_component(x, n, k):
    """Gradient contribution for a single shift k (k = shift in index units, can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return grad
        grad[0:nv] += (1.0 - x[k:n]) * dx
        grad[k:n] += (-x[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return grad
        grad[kk:n] += (1.0 - x[0:nv]) * dx
        grad[0:nv] += (-x[kk:n]) * dx
    return grad

def pnorm_obj(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 0)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def pnorm_grad(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 0)
    cp = np.sum(c_pos ** p)
    if cp < 1e-30:
        return np.zeros(n)
    S = cp ** (1.0 / p - 1.0)
    weights = c_pos ** (p - 1) * S
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = weights[i]
        if abs(wi) < 1e-20:
            continue
        k = i - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def find_active_shifts(h, n, top_k=5, rel_tol=0.003):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    return active

def sq_eq(x, n):
    return float(x.sum() - n / 2.0)

def sq_eq_jac(x, n):
    return np.ones(n)

def smooth_obj_all(x, n, alpha, mu=0.01):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_all(x, n, alpha, mu=0.01):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_pnorm(h, n, deadline, p_list=None, mu=50.0):
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(pnorm_obj, h, jac=pnorm_grad,
                          args=(n, p, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': 40000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active(h, n, deadline, top_k=5, max_rounds=12):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for r in range(max_rounds):
        if time.time() > deadline - 5:
            break
        active = find_active_shifts(h, n, top_k=top_k, rel_tol=0.004)
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(active_obj, h, jac=active_grad,
                              args=(n, active, alpha, 10.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxfun': 30000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_smooth(h, n, deadline, configs=None):
    if configs is None:
        configs = [(50, 5.0), (500, 50.0), (2000, 500.0), (5000, 5000.0), (10000, 50000.0)]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj_all, h, jac=smooth_grad_all,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_slqp(h, n, deadline, alpha=100000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_eq, 'jac': sq_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_all, h, args=(n, alpha, 0.01), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def local_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_vec = compute_c5_vec(h, n)
    top_k = 3
    top_idx = np.argsort(-c_vec)[:top_k]
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for s in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            g_comb = np.zeros(n)
            for idx in top_idx:
                g_comb += _shift_grad_component(h, n, idx - (n - 1))
            g_comb /= top_k
            abs_g = np.abs(g_comb)
            cand = np.argsort(-abs_g)[:25]
            for ii in range(len(cand)):
                if time.time() > deadline:
                    return best_h, best_c
                i = cand[ii]
                for jj in range(len(cand)):
                    if ii == jj:
                        continue
                    j = cand[jj]
                    for d_i, d_j in [(s, -s), (-s, s)]:
                        if 0 <= h[i] + d_i <= 1 and 0 <= h[j] + d_j <= 1:
                            h[i] += d_i
                            h[j] += d_j
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] -= d_i
                                h[j] -= d_j
    return best_h, best_c

def try_binary_snap(h, n):
    """Try thresholding to binary and check if it improves."""
    c_curr = compute_c5(h, n)
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        idx = np.argsort(-h_bin == 1)[:n_ones]
        idx = np.where(h_bin == 1)[0]
        idx = idx[np.argsort(h[idx])]
        for i in range(n_ones - target):
            h_bin[idx[i]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        idx = idx[np.argsort(-h[idx])]
        for i in range(target - n_ones):
            h_bin[idx[i]] = 1
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < c_curr:
        return h_bin, c_bin
    return h, c_curr
