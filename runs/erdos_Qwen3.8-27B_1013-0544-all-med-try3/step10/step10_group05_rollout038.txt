def compute_c5_vec(h, n):

def compute_c5(h, n):

def project_feasible(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        wi = w[mm]
        if wi < 1e-14:

                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])

            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        wi = w[mm]
        if wi < 1e-14:
            continue
        k = mm - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    pattern = [0]
    for _ in range(12):
        pattern = pattern + [1 - x for x in pattern]
    pattern = pattern[:n]
    n_ones = n // 2
    ones_idx = [i for i in range(n) if pattern[i] == 1]
    if len(ones_idx) > n_ones:
        ones_idx = ones_idx[:n_ones]
    elif len(ones_idx) < n_ones:
        extra = [i for i in range(n) if pattern[i] == 0]
        extra.extend([i for i in range(n) if pattern[i] == 1])
        ones_idx.extend(extra[:n_ones - len(ones_idx)])
    for i in ones_idx:
        h[i] = 1.0
    return h

def binary_local_search(h, n, max_time, max_passes=100):
    dx = 2.0 / n
    h = h.copy()
    n_ones = n // 2
    ones_set = set(np.where(h >= 0.5)[0].tolist())
    if len(ones_set) != n_ones:
        h = np.zeros(n)
        idx = np.where(h >= 0.5)[0][:n_ones]
        h[idx] = 1.0
        ones_set = set(idx.tolist())
    
    c_vec = compute_c5_vec(h, n)
    current_max = c_vec.max()
    t_start = time.time()
    
    for pass_num in range(max_passes):
        if time.time() - t_start > max_time:
            break
        improved = False
        ones_list = sorted(ones_set)
        zeros_list = [i for i in range(n) if i not in ones_set]
        
        for ii, i in enumerate(ones_list):
            if time.time() - t_start > max_time:
                return h, compute_c5(h, n)
            # Compute delta for removing i: dc_k = -[(1-h[i+k]) - h[i-k]]*dx
            delta_i = np.zeros(2 * n - 1)
            d_range = range(-(n - 1), n)
            for idx_c, d in enumerate(d_range):
                ip = i + d
                im = i - d
                term = 0.0
                if 0 <= ip < n:
                    term += (1.0 - h[ip])
                if 0 <= im < n:
                    term -= h[im]
                delta_i[idx_c] = -term * dx
            
            for jj, j in enumerate(zeros_list):
                # Compute delta for adding j
                delta_j_val = 0.0
                new_max = 0.0
                # For efficiency, compute max of (c_vec + delta_i + delta_j)
                # delta_j[k] = [(1-h[j+k]) - h[j-k]] * dx
                # Combined: c_vec + delta_i + delta_j
                # We need max of this
                best_k = -1
                best_val = -1
                for idx_c, d in enumerate(d_range):
                    jp = j + d
                    jm = j - d
                    term_j = 0.0
                    if 0 <= jp < n:
                        term_j += (1.0 - h[jp])
                    if 0 <= jm < n:
                        term_j -= h[jm]
                    new_val = c_vec[idx_c] + delta_i[idx_c] + term_j * dx
                    if new_val > best_val:
                        best_val = new_val
                        best_k = idx_c
                
                if best_val < current_max - 1e-15:
                    # Accept swap
                    h[i] = 0.0
                    h[j] = 1.0
                    ones_set.remove(i)
                    ones_set.add(j)
                    c_vec = c_vec + delta_i
                    # Recompute delta_j part
                    for idx_c, d in enumerate(d_range):
                        jp = j + d
                        jm = j - d
                        term_j = 0.0
                        if 0 <= jp < n:
                            term_j += (1.0 - h[jp])
                        if 0 <= jm < n:
                            term_j -= h[jm]
                        c_vec[idx_c] += term_j * dx
                    current_max = c_vec.max()
                    improved = True
                    zeros_list.remove(j)
                    zeros_list.append(i)
                    break
    
        if not improved:
            break
    
    final_c = compute_c5(h, n)
    return h, final_c

def binary_local_search_fast(h, n, max_time, max_passes=50):
    dx = 2.0 / n
    h = h.copy()
    n_ones = n // 2
    h = (h >= 0.5).astype(float)
    
    c_vec = np.correlate(h, 1 - h, mode='full') * dx
    current_max = c_vec.max()
    t_start = time.time()
    
    d_values = np.arange(-(n-1), n)  # shape (2n-1,)
    
    def compute_delta(idx):
        """Compute delta vector when flipping h[idx] (0->1 or 1->0)"""
        d = d_values
        # h[idx+d] for valid indices
        ip = idx + d
        im = idx - d
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        val_p = np.where(valid_p, 1.0 - h[np.clip(ip, 0, n-1)], 0.0)
        val_m = np.where(valid_m, h[np.clip(im, 0, n-1)], 0.0)
        return (val_p - val_m) * dx
    
    for pass_num in range(max_passes):
        if time.time() - t_start > max_time:
            break
        improved = False
        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()
        
        for i in ones_list:
            if time.time() - t_start > max_time:
                break
            delta_remove = -compute_delta(i)  # removing a 1
            
            best_j = -1
            best_new_max = current_max + 1  # threshold
            best_delta_add = None
            
            for j in zeros_list:
                if time.time() - t_start > max_time:
                    best_j = -1
                    break
                delta_add = compute_delta(j)  # adding a 1
                new_c = c_vec + delta_remove + delta_add
                nm = new_c.max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_j = j
                    best_delta_add = delta_add
            
            if best_j >= 0 and best_new_max < current_max - 1e-14:
                h[i] = 0.0
                h[best_j] = 1.0
                c_vec = c_vec + delta_remove + best_delta_add
                current_max = c_vec.max()
                improved = True
                # Update lists
                if best_j in zeros_list:
                    zeros_list.remove(best_j)
                    zeros_list.append(i)
    
    final_c = float(np.correlate(h, 1 - h, mode='full').max() * dx)
    return h, final_c

def vectorized_binary_local_search(h, n, max_time, max_passes=30):
    dx = 2.0 / n
    h = (h >= 0.5).astype(float)
    n_ones = n // 2
    
    c_vec = np.correlate(h, 1 - h, mode='full') * dx
    current_max = c_vec.max()
    t_start = time.time()
    
    d_values = np.arange(-(n-1), n).astype(np.int32)
    
    def compute_delta_vec(idx):
        ip = idx + d_values
        im = idx - d_values
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        hp = np.zeros(2*n-1)
        hm = np.zeros(2*n-1)
        hp[valid_p] = 1.0 - h[ip[valid_p]]
        hm[valid_m] = h[im[valid_m]]
        return (hp - hm) * dx
    
    for pass_num in range(max_passes):
        if time.time() - t_start > max_time:
            break
        improved = False
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        # For each one position, compute the delta of removing it
        # Then for each zero, compute the delta of adding it
        # Find the pair that minimizes the new max
        
        for i in ones_list:
            if time.time() - t_start > max_time:
                break
            if h[i] != 1:
                continue
            delta_remove = -compute_delta_vec(int(i))
            
            # For all zeros, compute the new max efficiently
            best_j = -1
            best_val = current_max + 0.5
            
            for j in zeros_list:
                if time.time() - t_start > max_time:
                    best_j = -1
                    break
                j = int(j)
                if h[j] != 0:
                    continue
                delta_add = compute_delta_vec(j)
                new_max = (c_vec + delta_remove + delta_add).max()
                if new_max < best_val:
                    best_val = new_max
                    best_j = j
            
            if best_j >= 0 and best_val < current_max - 1e-14:
                h[int(i)] = 0.0
                h[best_j] = 1.0
                delta_add_best = compute_delta_vec(best_j)
                # But h changed, so delta_add was computed with old h
                # We need to recompute since h[i] changed
                # Actually delta_remove was computed before the change, so it's correct
                # delta_add was also computed before the change
                # But after changing h[i]=0, h[j]=1, the c_vec needs both changes
                # The deltas are correct because they were computed with the OLD h
                c_vec = c_vec + delta_remove + delta_add_best
                current_max = c_vec.max()
                improved = True
    
        if not improved:
            break
    
    final_c = float(np.correlate(h, 1 - h, mode='full').max() * dx)
    return h, final_c

def run_lbfgs(h, n, deadline, alphas, mu=0.01):
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 30:
            break
        maxit = max(100, int(rem * 2))
        maxit = min(maxit, 2000)
        try:
            res = minimize(full_obj, best_h, jac=full_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxit, 'ftol': 1e-22,
                                  'gtol': 1e-16, 'maxfun': maxit * 6,
                                  'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_round(h, n, deadline, top_k, alpha, mu=0.001):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-7, 0.001 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = max(100, int(rem * 3))
    maxit = min(maxit, 3000)
    try:
        res = minimize(active_obj, h, jac=active_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 8,
                               'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def fine_pair_search(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Generate diverse starts
    starts = [h0]
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts.append(make_beatty(n0, golden / 2.0))
    starts.append(make_beatty(n0, np.sqrt(2.0) / 2.0))
    starts.append(make_beatty(n0, 0.41421356))
    starts.append(make_thue_morse(n0))
    starts.append(make_random_binary(n0, rng))
    starts.append(make_random_binary(n0, rng))
    
    # Quick screening
    for hs in starts:
        c_s = compute_c5(hs, n0)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()
    
    # Phase 1: L-BFGS warmup at n0
    p1_end = min(t0 + 80, deadline - 900)
    if p1_end > t0 + 20:
        best_h, best_c = run_lbfgs(best_h, best_n, p1_end, [5000, 50000, 200000, 500000])
    
    # Phase 2: Active set optimization at n0
    p2_end = min(time.time() + 150, deadline - 700)
    if p2_end > time.time() + 20:
        for top_k, alpha in [(4, 50000), (6, 200000), (8, 500000), (12, 2000000), (16, 5000000)]:
            if time.time() > p2_end - 10:
                break
            h2, c2 = run_active_round(best_h, best_n, p2_end - 5, top_k, alpha, 0.001)
            if c2 < best_c:
                best_c = c2
                best_h = h2
    
    # Phase 3: Binary local search at n0 (NEW)
    p3_end = min(time.time() + 80, deadline - 500)
    if p3_end > time.time() + 15:
        h_bin = (best_h >= 0.5).astype(float)
        # Fix to exactly n/2 ones
        n_ones = best_n // 2
        n_current_ones = int(h_bin.sum())
        if n_current_ones > n_ones:
            # Remove extra ones
            vals = h_bin.copy()
            for i in range(best_n):
                if vals.sum() <= n_ones:
                    break
                vals[i] = 0.0
            h_bin = vals
        elif n_current_ones < n_ones:
            for i in range(best_n):
                if vals.sum() >= n_ones:
                    break
                if h_bin[i] == 0:
                    h_bin[i] = 1.0
        
        h_bin, c_bin = vectorized_binary_local_search(h_bin, best_n, p3_end - 5)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
    
    # Phase 4: After binary search, do a bit of continuous optimization
    p4_end = min(time.time() + 60, deadline - 400)
    if p4_end > time.time() + 15:
        best_h, best_c = run_lbfgs(best_h, best_n, p4_end, [100000, 500000, 2000000])
    
    # Phase 5: Upsample to 2*n and optimize
    n_up = best_n * 2
    p5_end = min(time.time() + 250, deadline - 250)
    if p5_end > time.time() + 50 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        
        # Also try direct Beatty at higher res
        for bp in [golden/2.0, np.sqrt(2.0)/2.0, 0.4312, 0.382]:
            h_b = make_beatty(n_up, bp)
            c_b = compute_c5(h_b, n_up)
            if c_b < c_up:
                h_up = h_b
                c_up = c_b
        
        if c_up <= best_c + 0.005:
            h_up, c_up = run_lbfgs(h_up, n_up, min(time.time() + 120, p5_end - 100),
                                    [5000, 50000, 200000, 1000000])
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n_up
        
        # Active set at higher res
        if time.time() < p5_end - 30 and best_n == n_up:
            for top_k, alpha in [(6, 500000), (10, 5000000), (14, 20000000)]:
                if time.time() > p5_end - 20:
                    break
                h5, c5 = run_active_round(best_h, best_n, p5_end - 10, top_k, alpha, 0.0005)
                if c5 < best_c:
                    best_c = c5
                    best_h = h5
    
    # Phase 6: Binary local search at higher resolution (NEW)
    p6_end = min(time.time() + 150, deadline - 150)
    if p6_end > time.time() + 30:
        cur_n = len(best_h)
        h_bin2 = (best_h >= 0.5).astype(float)
        n_ones2 = cur_n // 2
        n_cur = int(h_bin2.sum())
        if n_cur > n_ones2:
            for i in range(cur_n):
                if h_bin2.sum() <= n_ones2:
                    break
                h_bin2[i] = 0.0
        elif n_cur < n_ones2:
            for i in range(cur_n):
                if h_bin2.sum() >= n_ones2:
                    break
                if h_bin2[i] == 0:
                    h_bin2[i] = 1.0
        
        avail = p6_end - time.time() - 10
        if avail > 10:
            h_bin2, c_bin2 = vectorized_binary_local_search(h_bin2, cur_n, avail)
            if c_bin2 < best_c:
                best_c = c_bin2
                best_h = h_bin2
    
    # Phase 7: Continue continuous optimization after binary search
    p7_end = min(time.time() + 80, deadline - 80)
    if p7_end > time.time() + 15:
        cur_n = len(best_h)
        best_h, best_c = run_lbfgs(best_h, cur_n, p7_end, [200000, 1000000, 5000000])
        # Active set
        if time.time() < p7_end - 20:
            for top_k, alpha in [(8, 5000000), (12, 20000000)]:
                if time.time() > p7_end - 15:
                    break
                h7, c7 = run_active_round(best_h, cur_n, p7_end - 10, top_k, alpha, 0.0003)
                if c7 < best_c:
                    best_c = c7
                    best_h = h7
    
    # Phase 8: Final fine pair search
    p8_end = deadline - 3
    if p8_end > time.time() + 15:
        cur_n = len(best_h)
        deltas = [0.001, 0.0005, 0.0002, 0.0001]
        best_h, best_c = fine_pair_search(best_h, cur_n, p8_end, radius=50, deltas=deltas)
    
    # Final check
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
