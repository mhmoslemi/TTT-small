        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h

            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

def local_pair_search(h, n, deadline, radius=40):
    """Pairwise coordinate descent with adaptive step sizes."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    base_steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in base_steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try increasing h[i], decreasing h[j]
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    # Try decreasing h[i], increasing h[j]
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
    return h, best_c

    y = h.copy()

    best_c = compute_c5(h, n)
    
    t = 1
    while time.time() < deadline and t < 50000:
        # FTL
        gamma = 2.0 / (t + 1.0)
        y_new = (1 - gamma) * y + gamma * h0
        
        # Gradient step at y_new
        c_vec = compute_c5_vec(y_new, n)
        k_star = np.argmax(c_vec)
        # For smoothness, use top-3 shifts
        top3 = np.argsort(-c_vec)[:3]

        for idx in top3:
            g += shift_grad_single(y_new, n, idx)

        if c < best_c:
            best_c = c
            best_h = h0.copy()
        
        y = (1 - gamma) * y + gamma * h0
        t += 1
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv <= 0:
            return grad
        grad[0:nv] = (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv <= 0:
            return grad
        grad[kk:n] = (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

        for alpha, mu, miter in [(200, 5, 2000), (1000, 50, 3000), (5000, 200, 3000)]:
            if time.time() > deadline - 700:
                break
            h1, c1 = do_lbfgs(best_h, n, time.time() + 35, alpha, mu, maxiter=miter)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    # Phase 2: p-norm optimization (different smooth landscape)
    if time.time() < deadline - 550:
        for pn, mu in [(20, 100), (50, 500)]:
            if time.time() > deadline - 550:
                break
            h2, c2 = do_lbfgs(best_h, n, time.time() + 30, alpha=0, mu=mu,
                              maxiter=2000, use_pnorm=True, pn=pn)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Aggressive L-BFGS (high alpha, sharp max approximation)
    if time.time() < deadline - 450:
        for alpha, mu in [(20000, 500), (100000, 2000), (500000, 10000)]:
            if time.time() > deadline - 440:
                break
            h3, c3 = do_lbfgs(best_h, n, time.time() + 35, alpha, mu, maxiter=4000)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Active-shift bundle method
    if time.time() < deadline - 350:
        t4 = min(time.time() + 150, deadline - 350)
        best_h, best_c = bundle_method(best_h, n, t4, alpha=200000)

    # Phase 5: Nesterov subgradient (explores different directions)
    if time.time() < deadline - 280:
        t5 = min(time.time() + 80, deadline - 280)
        h5, c5 = subgradient_nesterov(best_h, n, t5, alpha_init=0.003)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Focused worst-shift optimization (top 2 only)
    if time.time() < deadline - 230:
        t6 = min(time.time() + 80, deadline - 230)
        best_h, best_c = do_active_shift_opt(best_h, n, t6, top_k=2, alpha=500000, maxiter=3000)

    # Phase 7: SLSQP polish
    if time.time() < deadline - 180:
        h7, c7 = slqp_polish(best_h, n, time.time() + 50, alpha=500000, maxiter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Random restarts to escape local minima
    if time.time() < deadline - 130:
        t8 = min(time.time() + 50, deadline - 130)
        best_h, best_c = random_restart_opt(best_h, n, t8, noise_scale=0.015, n_restarts=2)

    # Phase 9: Another aggressive L-BFGS pass
    if time.time() < deadline - 100:
        for alpha in [2000000]:
            if time.time() > deadline - 90:
                break
            h9, c9 = do_lbfgs(best_h, n, time.time() + 25, alpha, 50000, maxiter=3000)
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Phase 10: SLSQP final polish
    if time.time() < deadline - 80:
        h10, c10 = slqp_polish(best_h, n, time.time() + 35, alpha=1000000, maxiter=1500)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Phase 11: Fine local pair search
    if time.time() < deadline - 30:
        t11 = min(time.time() + 25, deadline - 30)
        best_h, best_c = local_pair_search(best_h, n, t11, radius=15)

    # Phase 12: Final SLSQP
    if time.time() < deadline - 3:
        h12, c12 = slqp_polish(best_h, n, max(deadline - 1, time.time() + 3), alpha=1000000, maxiter=1000)
        if c12 < best_c:
            best_c = c12
            best_h = h12

    # Final: project and evaluate
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
