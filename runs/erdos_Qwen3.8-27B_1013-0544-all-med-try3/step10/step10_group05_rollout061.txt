    """Create Beatty sequence: positions floor(k * alpha) mod n for k=1,..."""
    h = np.zeros(n)
    n_ones = n // 2
    # Use the Sturmian word / mechanical word
    # h[i] = 1 if frac(i * alpha_frac) < 0.5 (approximately)
    for i in range(n):

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_vals_w(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    if smx > 500:
        val = m + (smx + np.log(w.sum())) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val), w

def lag_grad_vec(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[0:nv] += (1.0 - h[k:n])
            g[k:n] += (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:n] += (1.0 - h[0:nv])
            g[0:nv] += (-h[kk:n])
    return g

def obj_full(x, n, alpha, mu):
    c_vec = compute_c5_vec(x, n)
    val = smooth_max_val(c_vec, alpha)
    pen = mu * (x.sum() - n / 2.0) ** 2
    return val + pen

def grad_full(x, n, alpha, mu):
    c_vec = compute_c5_vec(x, n)
    _, w = smooth_max_vals_w(c_vec, alpha)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for mm in range(len(c_vec)):
        if w[mm] < 1e-14:
            continue
        k = mm - center
        wm = w[mm] * dx
        grad += wm * lag_grad_vec(x, n, k)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c_vec = compute_c5_vec(x, n)
    vals = c_vec[active]
    val = smooth_max_val(vals, alpha)
    pen = mu * (x.sum() - n / 2.0) ** 2
    return val + pen

def grad_active(x, n, active, alpha, mu):
    c_vec = compute_c5_vec(x, n)
    vals = c_vec[active]
    _, w = smooth_max_vals_w(vals, alpha)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for i, idx in enumerate(active):
        if w[i] < 1e-14:
            continue
        k = int(idx) - center
        wm = w[i] * dx
        grad += wm * lag_grad_vec(x, n, k)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def make_sturmian(n, rho):
    h = np.zeros(n)
    n_ones = n // 2
    count = 0
    for i in range(n):
        if math.floor((i + 1) * rho) - math.floor(i * rho) == 1:
            h[i] = 1.0
            count += 1
    if count > n_ones:
        ones = np.where(h == 1.0)[0]
        for idx in ones[:count - n_ones]:
            h[idx] = 0.0
    elif count < n_ones:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:n_ones - count]:
            h[idx] = 1.0
    return h

def make_periodic(n, pat):
    h = np.zeros(n)
    plen = len(pat)
    for i in range(n):
        h[i] = pat[i % plen]
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def make_block(n, block_sizes, rng):
    h = np.zeros(n)
    pos = 0
    val = 1
    while pos < n:
        b = min(block_sizes, n - pos)
        h[pos:pos+b] = val
        pos += b
        val = 1 - val
    return project_feasible(h, n)

def run_lfgs(h, n, deadline, alpha, mu, top_k=None, maxiter_cap=5000):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time() - 3
    if rem < 5:
        return h, compute_c5(h, n)
    maxit = max(100, min(int(rem * 2), maxiter_cap))
    maxit = min(maxit, 5000)
    if top_k is not None and top_k < 2 * n - 1:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        thresh = c_max - max(1e-8, 0.002 * c_max)
        active = np.where(c_vec >= thresh)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])[:top_k]]
        if len(active) == 0:
            active = np.array([n - 1])
        active_t = tuple(active)
        try:
            res = minimize(obj_active, h, jac=grad_active,
                          args=(n, active_t, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 50, 'maxcor': 50})
        except Exception:
            return h, compute_c5(h, n)
    else:
        try:
            res = minimize(obj_full, h, jac=grad_full,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 12,
                                   'maxls': 50, 'maxcor': 50})
        except Exception:
            return h, compute_c5(h, n)
    h_new = project_feasible(res.x, n)
    c_new = compute_c5(h_new, n)
    return h_new, c_new
