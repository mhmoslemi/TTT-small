Now, when h[p] changes by ε:
- In the factor h[i+offset]: affected when i+offset = p, i.e., i = p-offset
  - This term is h[p]*(1-h[p-offset]), and h[p] changes by ε

  So k ∈ [n-1-p, 2n-2-p] (inclusive on both ends, since offset ≤ p and offset > p-n)
  
  Actually, offset ≤ p means n-1-k ≤ p, k ≥ n-1-p.
  offset > p-n means n-1-k > p-n, k < 2n-1-p.
  So k ∈ [n-1-p, 2n-2-p] (integer k).
  
  But we also need 0 ≤ i = p-offset < n, which is already encoded.

- In the factor (1-h[i]): affected when i = p
  - This term is h[p+offset]*(1-h[p]), and (1-h[p]) changes by -ε
  - So the term changes by -ε*h[p+offset]
  - Valid when 0 ≤ p+offset < n, i.e., offset ∈ [-p, n-1-p]
  - offset = n-1-k, so -p ≤ n-1-k ≤ n-1-p
  - k ≤ n-1+p and k ≥ p
  - So k ∈ [p, n-1+p]

Δcorr[k] = ε*(1-h[p-offset]) * [k ∈ [n-1-p, 2n-2-p]] - ε*h[p+offset] * [k ∈ [p, n-1+p]]
where offset = n-1-k.

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lse_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):

            continue

        if k >= 0:

            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        def obj(x, n=n, alpha=alpha, mu=mu):
            return lse_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

            if c_new < best_c:

    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        # Compute new max without actually modifying h
                        nm = fast_swap_max(c_vec, h, n, i, j, d)
                        if nm < best_c - 1e-15:
                            h[i] += d
                            h[j] -= d
                            best_c = nm
                            c_vec = compute_c5_vec(h, n)
                            improved = True
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        nm = fast_swap_max(c_vec, h, n, j, i, d)
                        if nm < best_c - 1e-15:
                            h[i] -= d
                            h[j] += d
                            best_c = nm
                            c_vec = compute_c5_vec(h, n)
                            improved = True
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def binary_local_search(h, n, deadline, radius=60):
    """Local search on binary-like solutions (threshold at 0.5)."""
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        # Remove lowest-valued ones
        for idx in ones_idx:
            if n_ones == target:
                break
            if h_bin[idx] == 1:
                h_bin[idx] = 0
                n_ones -= 1
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        for idx in zeros_idx:
            if n_ones == target:
                break
            h_bin[idx] = 1
            n_ones += 1
    best_c = compute_c5(h_bin, n)
    c_vec = compute_c5_vec(h_bin, n)
    # Swap 1s and 0s
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        for i in ones:
            if time.time() > deadline:
                return h_bin, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in zeros:
                if j < lo or j >= hi:
                    continue
                # Swap: h[i]=0, h[j]=1
                nm = fast_swap_max(c_vec, h_bin, n, j, i, 1.0)
                if nm < best_c - 1e-14:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    best_c = nm
                    c_vec = compute_c5_vec(h_bin, n)
                    improved = True
                    break
            if time.time() > deadline:
                return h_bin, best_c
        if not improved:
            break
    return h_bin, best_c

def perturbation_escape(h, n, deadline, n_pert=20):
    """Make random perturbations and re-optimize to escape local optima."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(42)
    for trial in range(10):
        if time.time() > deadline - 30:
            break
        h_pert = h.copy()
        # Random perturbation
        idx = rng.choice(n, size=n_pert * 2, replace=False)
        for k in range(n_pert):
            a, b = idx[2*k], idx[2*k+1]
            amt = 0.05
            if h_pert[a] >= amt and h_pert[b] <= 1 - amt:
                h_pert[a] -= amt
                h_pert[b] += amt
        h_pert = project_feasible(h_pert, n)
        # Quick optimize
        t_sub = min(20, deadline - time.time() - 5)
        if t_sub > 10:
            h_opt, c_opt = run_lbfgs(h_pert, n, time.time() + t_sub,
                                       [(500, 100.0), (2000, 1000.0)])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    return best_h, best_c

def multiresolve(h, n, deadline):
    """Try different resolutions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x_fine = np.linspace(0, 2, n)
    for n_try in [400, 600, 800]:
        if time.time() > deadline - 60:
            break
        x_coarse = np.linspace(0, 2, n_try)
        h_t = np.interp(x_coarse, x_fine, h)
        h_t = project_feasible(h_t, n_try)
        c_t = compute_c5(h_t, n_try)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            # Optimize at this resolution
            h_t, c_t = run_lbfgs(h_t, n_try, min(time.time() + 40, deadline - 60),
                                   [(500, 50.0), (2000, 500.0), (5000, 5000.0)])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: L-BFGS polish at original resolution (40s)
    t1 = min(t0 + 40, deadline - 600)
    best_h, best_c = run_lbfgs(best_h, n_orig, t1,
        [(200, 20.0), (1000, 200.0), (5000, 2000.0)])

    # Phase 2: Fast local search (350s)
    if time.time() < deadline - 400:
        t2 = min(time.time() + 350, deadline - 400)
        best_h, best_c = fast_local_search(best_h, n_orig, t2, radius=50)

    # Phase 3: L-BFGS again after local search (30s)
    if time.time() < deadline - 300:
        t3 = min(time.time() + 30, deadline - 300)
        h3, c3 = run_lbfgs(best_h, n_orig, t3,
            [(1000, 500.0), (5000, 5000.0), (10000, 50000.0)])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Another local search pass (200s)
    if time.time() < deadline - 250:
        t4 = min(time.time() + 200, deadline - 250)
        best_h, best_c = fast_local_search(best_h, n_orig, t4, radius=80)

    # Phase 5: Multi-resolution (150s)
    if time.time() < deadline - 200:
        t5 = min(time.time() + 150, deadline - 200)
        h5, c5 = multiresolve(best_h, n_orig, t5)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            n_orig = len(best_h)

    # Phase 6: Binary local search (100s)
    if time.time() < deadline - 150:
        t6 = min(time.time() + 100, deadline - 150)
        h6, c6 = binary_local_search(best_h, n_orig, t6, radius=80)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation escape (80s)
    if time.time() < deadline - 100:
        t7 = min(time.time() + 80, deadline - 100)
        h7, c7 = perturbation_escape(best_h, n_orig, t7, n_pert=15)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final local search (60s)
    if time.time() < deadline - 30:
        t8 = min(time.time() + 60, deadline - 30)
        best_h, best_c = fast_local_search(best_h, n_orig, t8, radius=100)

    # Phase 9: Final L-BFGS polish (20s)
    if time.time() < deadline - 5:
        t9 = deadline - 3
        h9, c9 = run_lbfgs(best_h, n_orig, t9, [(5000, 5000.0), (20000, 50000.0)])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
