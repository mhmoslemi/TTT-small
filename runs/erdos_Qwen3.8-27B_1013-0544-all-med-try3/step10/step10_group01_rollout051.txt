    for mm in range(n_c):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def topm_obj(x, n, m_count, alpha, mu):
    c = compute_c5_vec(x, n)
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:m_count]
    mx = top.max()
    sh = alpha * (top - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topm_grad(x, n, m_count, alpha, mu):
    c = compute_c5_vec(x, n)
    top_idx = np.argsort(-c)[:m_count]
    top = c[top_idx]
    mx = top.max()
    sh = alpha * (top - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lm_obj, h, jac=lm_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run_topm_lbfgs(h, n, deadline, m_count, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(topm_obj, h, jac=topm_grad, args=(n, m_count, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': maxit * 8,
                               'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def sim_anneal_binary(h, n, deadline, max_iters=300000, T0=0.004, T_min=1e-7, cooling=0.9995):
    rng = np.random.RandomState(12345)
    cur_h = np.zeros(n)
    n_ones_target = n // 2
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h <= 0.5)[0]
    cur_h[:n_ones_target] = 0.0
    if len(ones_idx) >= n_ones_target:
        for i in ones_idx[:n_ones_target]:
            cur_h[i] = 1.0
    else:
        need = n_ones_target - len(ones_idx)
        for i in zeros_idx[:need]:
            cur_h[i] = 1.0
    
    ones_list = list(np.where(cur_h == 1.0)[0])
    zeros_list = list(np.where(cur_h == 0.0)[0])
    
    cur_c = compute_c5(cur_h, n)
    best_h = cur_h.copy()
    best_c = cur_c
    
    T = T0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if T <= T_min:
            break
        T *= cooling
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c

def two_opt_binary(h, n, deadline, max_iters=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones_list = list(np.where(best_h > 0.5)[0])
    zeros_list = list(np.where(best_h <= 0.5)[0])
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break
        i1 = ones_list[np.random.randint(len(ones_list))]
        i2 = ones_list[np.random.randint(len(ones_list)-1)]
        if i2 >= len(ones_list): i2 -= 1
        if i1 == i2: continue
        j1 = zeros_list[np.random.randint(len(zeros_list))]
        j2 = zeros_list[np.random.randint(len(zeros_list)-1)]
        if j2 >= len(zeros_list): j2 -= 1
        if j1 == j2: continue
        best_h[i1] = 0.0; best_h[i2] = 0.0
        best_h[j1] = 1.0; best_h[j2] = 1.0
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            ones_list.remove(i1); ones_list.remove(i2)
            ones_list.append(j1); ones_list.append(j2)
            zeros_list.remove(j1); zeros_list.remove(j2)
            zeros_list.append(i1); zeros_list.append(i2)
        else:
            best_h[i1] = 1.0; best_h[i2] = 1.0
            best_h[j1] = 0.0; best_h[j2] = 0.0
    return best_h, best_c

def peak_shave(h, n, deadline, n_peaks=4, max_steps=80000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(777)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        top_idx = np.argsort(-c_vec)[:n_peaks]
        threshold = max(c_vec[top_idx[-1]] - 1e-10, max_val - 0.005)
        active = np.where(c_vec >= threshold)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx
                grad[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx
                grad[:nv] += (-best_h[kk:]) * dx
        grad /= max(len(active), 1)
        
        pos_mask = (grad < -1e-8) & (best_h > 1e-5)
        neg_mask = (grad > 1e-8) & (best_h < 1.0 - 1e-5)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        if len(pos_idx) > 60:
            pos_idx = pos_idx[rng.choice(len(pos_idx), 60, replace=False)]
        if len(neg_idx) > 60:
            neg_idx = neg_idx[rng.choice(len(neg_idx), 60, replace=False)]
        
        found = False
        best_gain = 0
        best_i, best_j = -1, -1
        for i in pos_idx:
            gi = grad[i]
            for j in neg_idx:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i, best_j = i, j
        if best_i < 0 or best_gain < 1e-10:
            break
        found = True
        for st in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            amt = min(st, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_local(h, n, deadline, radius=30, max_rounds=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        for i in range(n):
            if time.time() > deadline:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for d in [0.001, 0.0005, 0.0002, 0.0001]:
                if best_h[i] + d <= 1.0:
                    for j in range(lo, hi):
                        if i == j: continue
                        if best_h[j] - d >= 0.0:
                            best_h[i] += d; best_h[j] -= d
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new; improved = True; break
                            else:
                                best_h[i] -= d; best_h[j] += d
                    if improved: break
                if best_h[i] - d >= 0.0:
                    for j in range(lo, hi):
                        if i == j: continue
                        if best_h[j] + d <= 1.0:
                            best_h[i] -= d; best_h[j] += d
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new; improved = True; break
                            else:
                                best_h[i] += d; best_h[j] -= d
                    if improved: break
            if improved: break
        if not improved:
            break
    return best_h, best_c

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.append(pos)
        if len(positions) >= n_ones:
            break
    if len(positions) < n_ones:
        for k in range(4 * n, 8 * n):
            pos = int(k * alpha_param) % n
            if pos not in positions:
                positions.append(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions[:n_ones]:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_sturmian(n, alpha_param):
    h = np.zeros(n)
    for k in range(n):
        if int((k + 1) * alpha_param) - int(k * alpha_param) == 1:
            h[k] = 1.0
    n_ones = int(h.sum())
    if n_ones > n // 2:
        pos = np.where(h == 1)[0]
        for i in pos[:n_ones - n//2]:
            h[i] = 0
    elif n_ones < n // 2:
        pos = np.where(h == 0)[0]
        for i in pos[:n//2 - n_ones]:
            h[i] = 1
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = make_sturmian(500, 0.5 + 0.01 * (1.0 + np.sqrt(5.0)) / 2.0)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    
    # Phase 0: Try alternative starts (5s)
    alt_candidates = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_candidates.append(make_beatty(n_orig, golden / 2.0 + 0.001))
    alt_candidates.append(make_sturmian(n_orig, 0.5 + 0.005 * np.sqrt(2)))
    alt_candidates.append(make_sturmian(n_orig, 0.5 + 0.003 * golden))
    h_rand = np.zeros(n_orig)
    h_rand[rng.choice(n_orig, n_orig // 2, replace=False)] = 1.0
    alt_candidates.append(project_feasible(h_rand, n_orig))
    
    for h_alt in alt_candidates:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Phase 1: Quick L-BFGS warmup (60s)
    p1_end = min(t0 + 60, deadline - 400)
    if p1_end > time.time() + 5:
        for alpha in [50, 200, 1000, 5000]:
            if time.time() > p1_end:
                break
            h1, c1 = run_lbfgs(best_h, n_orig, p1_end, alpha, 0.01, 500)
            if c1 < best_c:
                best_c = c1
                best_h = h1
    
    # Phase 2: Top-m L-BFGS (100s) - level the top correlations
    p2_end = min(time.time() + 100, deadline - 300)
    if p2_end > time.time() + 5:
        for m_count in [3, 5, 8, 12]:
            if time.time() > p2_end:
                break
            for alpha in [10000, 100000]:
                if time.time() > p2_end:
                    break
                h2, c2 = run_topm_lbfgs(best_h, n_orig, p2_end, m_count, alpha, 0.005, 400)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
    
    # Phase 3: Binary Simulated Annealing (200s) - KEY NEW PHASE
    p3_end = min(time.time() + 200, deadline - 250)
    if p3_end > time.time() + 10:
        h3, c3 = sim_anneal_binary(best_h, n_orig, p3_end,
                                    max_iters=200000, T0=0.005, T_min=1e-7, cooling=0.9995)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
        else:
            # Try 2-opt on the SA result
            if time.time() < p3_end - 20:
                h3b, c3b = two_opt_binary(h3, n_orig, p3_end - 20, max_iters=50000)
                if c3b < best_c:
                    best_c = c3b
                    best_h = h3b.copy()
    
    # Phase 4: L-BFGS refinement on potentially better binary start (120s)
    p4_end = min(time.time() + 120, deadline - 150)
    if p4_end > time.time() + 5:
        # Convert to continuous and smooth
        h4_init = best_h.copy()
        h4_init = 0.7 * h4_init + 0.3 * 0.5
        h4_init = project_feasible(h4_init, n_orig)
        for alpha in [5000, 50000, 500000, 5000000]:
            if time.time() > p4_end:
                break
            h4, c4 = run_lbfgs(h4_init, n_orig, p4_end, alpha, 0.005, 600)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                h4_init = h4.copy()
        # Top-m refinement
        if time.time() < p4_end:
            for m_count in [4, 10]:
                if time.time() > p4_end:
                    break
                h4b, c4b = run_topm_lbfgs(best_h, n_orig, p4_end, m_count, 200000, 0.005, 500)
                if c4b < best_c:
                    best_c = c4b
                    best_h = h4b
    
    # Phase 5: Peak shaving (120s)
    p5_end = min(time.time() + 120, deadline - 100)
    if p5_end > time.time() + 5:
        for n_peaks in [3, 5, 8]:
            if time.time() > p5_end:
                break
            h5, c5 = peak_shave(best_h, n_orig, p5_end, n_peaks=n_peaks, max_steps=30000)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: More L-BFGS with very high alpha (60s)
    p6_end = min(time.time() + 60, deadline - 60)
    if p6_end > time.time() + 5:
        for alpha in [5000000, 20000000, 100000000]:
            if time.time() > p6_end:
                break
            h6, c6 = run_lbfgs(best_h, n_orig, p6_end, alpha, 0.001, 500)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Fine local search (60s)
    p7_end = min(time.time() + 60, deadline - 30)
    if p7_end > time.time() + 3:
        h7, c7 = fine_local(best_h, n_orig, p7_end, radius=25, max_rounds=500)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Try n=1000 upsampling if time allows (100s)
    p8_end = min(time.time() + 100, deadline - 20)
    if p8_end > time.time() + 30:
        n_up = 1000
        h_up = upsample(best_h, n_up)
        c_up = compute_c5(h_up, n_up)
        # Quick optimize at n=1000
        for alpha in [1000, 50000, 1000000]:
            if time.time() > p8_end:
                break
            h_up2, c_up2 = run_lbfgs(h_up, n_up, p8_end, alpha, 0.01, 300)
            if c_up2 < c_up:
                c_up = c_up2
                h_up = h_up2
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
    
    # Phase 9: Final peak shave + L-BFGS polish (remaining)
    p9_end = deadline - 3
    if time.time() < p9_end:
        h9, c9 = peak_shave(best_h, len(best_h), p9_end - 10, n_peaks=5, max_steps=20000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
        if time.time() < p9_end:
            nf = len(best_h)
            h9b, c9b = run_lbfgs(best_h, nf, p9_end, 100000000, 0.001, 200)
            if c9b < best_c:
                best_c = c9b
                best_h = h9b
        # Very fine local
        if time.time() < p9_end:
            h9c, c9c = fine_local(best_h, len(best_h), p9_end, radius=15, max_rounds=200)
            if c9c < best_c:
                best_c = c9c
                best_h = h9c
    
    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
