from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def fw_step(grad, n):
    """Solve min grad·y s.t. 0≤y≤1, sum(y)=n/2. Returns optimal y."""
    # Sort by grad value (ascending)
    order = np.argsort(grad)
    y = np.zeros(n)
    n_target = n // 2
    # Set n_target smallest to 1
    y[order[:n_target]] = 1.0
    # For odd n, the middle one might be fractional
    if n % 2 == 1:
        y[order[n_target // 2]] = 0.5  # Hmm, this isn't quite right
    # Actually for even n, this is exact
    # For odd n: sum should be (n-1)/2, so we need one fractional
    # Let me just handle even n for simplicity
    return y

def fw_line_search(h, y, n, alpha, mu, max_steps=20):
    """Backtracking line search for FW update h + t*(y-h)."""
    c0 = sm_obj(h, n, alpha, mu)
    g = sm_grad(h, n, alpha, mu)
    direction = y - h
    directional_deriv = np.dot(g, direction)
    if directional_deriv >= 0:
        return h
    t = 1.0
    for _ in range(max_steps):
        h_new = h + t * direction
        h_new = np.clip(h_new, 0, 1)
        obj_new = sm_obj(h_new, n, alpha, mu)
        if obj_new <= c0 + 1e-4 * t * directional_deriv:
            return h_new
        t *= 0.5
    return h + t * direction

        h2, c2 = run_fw(best_h, best_n, p2_deadline, n_iter=50000, rel_tol=0.001)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = best_n

    # Phase 3: L-BFGS-B at higher resolution (n=1000)
    if time.time() < deadline - 350:
        n_fine = 1000
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        p3_deadline = min(time.time() + 250, deadline - 300)
        h3, c3 = run_lbfgs(h_fine, n_fine, p3_deadline,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
            mu=0.005)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_fine

    # Phase 4: Try alternative starting points at best_n
    if time.time() < deadline - 200:
        # Binary round
        h_bin = make_binary(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        h_alt = h_bin.copy()
        if c_bin < best_c:
            best_c = c_bin
        p4a_deadline = min(time.time() + 60, deadline - 150)
        h4a, c4a = run_lbfgs(h_alt, best_n, p4a_deadline,
            alphas=[1000, 10000, 100000, 1000000, 5000000],
            mu=0.01)
        if c4a < best_c:
            best_c = c4a
            best_h = h4a
            best_n = best_n

        # Periodic start (period 7, duty 3)
        h_per = period_start(best_n, 7, 3)
        p4b_deadline = min(time.time() + 50, deadline - 100)
        h4b, c4b = run_lbfgs(h_per, best_n, p4b_deadline,
            alphas=[100, 1000, 10000, 100000, 1000000],
            mu=0.01)
        if c4b < best_c:
            best_c = c4b
            best_h = h4b
            best_n = best_n

    # Phase 5: Active-set L-BFGS-B polish at best resolution
    if time.time() < deadline - 120:
        p5_deadline = min(time.time() + 80, deadline - 100)
        h5, c5 = run_active(best_h, best_n, p5_deadline, top_k=6)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = best_n

    # Phase 6: Second FW pass with tighter tolerance
    if time.time() < deadline - 80:
        p6_deadline = min(time.time() + 40, deadline - 60)
        h6, c6 = run_fw(best_h, best_n, p6_deadline, n_iter=20000, rel_tol=0.0005)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n

    # Phase 7: Final L-BFGS-B with very high alpha
    if time.time() < deadline - 30:
        h7, c7 = run_lbfgs(best_h, best_n, deadline - 25,
            alphas=[10000000, 30000000, 50000000],
            mu=0.005)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    # One last check: try to nudge toward binary if it helps
    h_nudge = best_h.copy()
    for threshold in [0.02, 0.01, 0.005]:
        h_nudge[h_nudge < threshold] = 0.0
        h_nudge[h_nudge > 1.0 - threshold] = 1.0
        h_nudge = project_feasible(h_nudge, final_n)
        c_nudge = compute_c5(h_nudge, final_n)
        if c_nudge < best_c:
            best_c = c_nudge
            best_h = h_nudge

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
