def all_lags_fft(h):
    n=len(h)
    total=2*n-1
    N=1
    while N<total:
       N <<=1
    fa=np.fft.rfft(h,N)
    b=1.0-h
    fb=np.fft.rfft(b[::-1],N)
    conv=np.fft.irfft(fa*fb, N)[:total]
    conv = np.real(conv)
    # remove tiny negative
    conv[(conv<0.0)&(conv>-1e-12)] = 0.0
    return conv

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(200):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)

    for _ in range(30):

        want = int(round(0.5 * n))
        b2 = np.zeros(n)
        b2[order[:want]] = 1.0
        h = b2 + rng.uniform(-0.35, 0.35, n)

    return project_feasible(h, n)


def all_lags_fft(h):
    n = int(len(h))
    total = 2 * n - 1
    N = 1
    while N < total:
        N <<= 1

    fa = np.fft.rfft(h, N)
    b = 1.0 - h
    fb = np.fft.rfft(b[::-1], N)
    conv = np.fft.irfft(fa * fb, N)[:total]
    conv = np.real(conv)
    neg = (conv < 0.0) & (conv > -1e-12)
    if np.any(neg):
        conv[neg] = 0.0
    return conv


def c5_fft(h):
    vals = all_lags_fft(h)
    if vals.size == 0:
        return 0.0
    return float(np.max(vals) * (2.0 / float(len(h))))


def c5_exact(h):
    n = int(len(h))
    if n <= 1:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def grad_lag_count(h, m, n):
    g = np.zeros(n, dtype=float)
    if m == 0:
        return 1.0 - 2.0 * h
    if m > 0:
        g[:n - m] += -h[m:]
        g[m:n] += 1.0 - h[:n - m]
    else:
        q = -int(m)
        g[:n - q] += 1.0 - h[q:]
        g[q:n] += -h[:n - q]
    return g


def select_active(vals, n, cur_c, dx, top_k, margin, max_lags):
    n = int(n)
    if n > 900 and max_lags > 80:
        max_lags = 80
    idxs = np.argsort(vals)[::-1]
    active = []
    threshold = (cur_c - margin) / dx if dx != 0.0 else 0.0
    for idx in idxs:
        if len(active) >= max_lags:
            break
        if len(active) < top_k or float(vals[idx]) > threshold:
            active.append(int(idx) - (n - 1))
    return active, idxs


def lp_step(h, n, active_lags, cur_c, lim):
    n = int(n)
    m = len(active_lags)
    if m == 0:
        return None

    vals = all_lags_fft(h)
    dx = 2.0 / float(n)
    target = 0.5 * float(n)

    A = np.zeros((m, n + 1), dtype=float)
    b = np.empty(m, dtype=float)

    for r, lag in enumerate(active_lags):
        idx = int(n) - 1 + int(lag)
        if idx < 0 or idx >= vals.size:
            return None
        g = grad_lag_count(h, lag, n) * dx
        val = float(vals[idx]) * dx
        gh = float(np.dot(g, h))
        # g*w - t <= g*h - val
        A[r, 0] = -1.0
        A[r, 1:] = g
        b[r] = gh - val

    c = np.zeros(n + 1, dtype=float)
    c[0] = 1.0

    lo = np.clip(h - lim, 0.0, 1.0)
    hi = np.clip(h + lim, 0.0, 1.0)

    if float(np.sum(lo)) > target + 1e-12 or float(np.sum(hi)) < target - 1e-12:
        lo = np.zeros(n, dtype=float)
        hi = np.ones(n, dtype=float)

    bounds = [(0.0, max(cur_c + 1e-9, 1e-12))]
    bounds.extend(zip(lo.tolist(), hi.tolist()))

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target], dtype=float)

    res = None
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if res is None or (not res.success):
            res2 = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method="interior-point")
            if res2 is not None and res2.success:
                res = res2
    except Exception:
        res = None

    if res is not None and hasattr(res, "x") and res.x is not None:
        w = np.asarray(res.x[1:], dtype=float).copy()
        w = np.clip(w, 0.0, 1.0)
        w = project_feasible(w, n)
        return w
    return None


def segment_count_fun(a, h0, delta, n):
    ht = h0 + a * delta
    if np.any(ht < 0.0) or np.any(ht > 1.0):
        ht = np.clip(ht, 0.0, 1.0)
    return c5_fft(ht)


def segment_search(h, w, n, best_c):
    n = int(n)
    delta = w - h
    scale = float(np.max(np.abs(delta))) if delta.size else 0.0
    if scale < 1e-12:
        return h, best_c

    best_cfft = float(best_c)
    best_a = 0.0

    alphas = np.linspace(0.0, 1.0, 41)
    for a in alphas:
        if a <= 0.0:
            continue
        cf = segment_count_fun(float(a), h, delta, n)
        if cf < best_cfft - 1e-15:
            best_cfft = cf
            best_a = float(a)

    if best_a > 0.0:
        l = max(0.0, best_a - 0.3)
        r = min(1.0, best_a + 0.3)
        if r - l > 1e-10:
            alphas2 = np.linspace(l, r, 41)
            for a in alphas2:
                cf = segment_count_fun(float(a), h, delta, n)
                if cf < best_cfft - 1e-15:
                    best_cfft = cf
                    best_a = float(a)

            if r - l > 1e-4 and best_a > 0.0:
                span = r - l
                x1 = r - float(math.sqrt(5.0) - 1.0) * span
                x2 = l + float(math.sqrt(5.0) - 1.0) * span
                f1 = segment_count_fun(x1, h, delta, n)
                f2 = segment_count_fun(x2, h, delta, n)
                aa = l
                bb = r
                for _ in range(35):
                    if f1 < f2:
                        bb = x2
                        x2 = x1
                        f2 = f1
                        span = bb - aa
                        x1 = bb - float(math.sqrt(5.0) - 1.0) * span
                        f1 = segment_count_fun(x1, h, delta, n)
                    else:
                        aa = x1
                        x1 = x2
                        f1 = f2
                        span = bb - aa
                        x2 = aa + float(math.sqrt(5.0) - 1.0) * span
                        f2 = segment_count_fun(x2, h, delta, n)
                    if bb - aa < 1e-7:
                        break
                xc = 0.5 * (aa + bb)
                fc = segment_count_fun(xc, h, delta, n)
                if fc < best_cfft - 1e-15:
                    best_cfft = fc
                    best_a = float(xc)

    if best_a <= 0.0:
        return h, best_c

    hnew = h + best_a * delta
    hnew = project_feasible(hnew, n)
    return hnew, best_cfft


def random_direction_w(h, active_lags, n, lim, rng):
    n = int(n)
    if len(active_lags) == 0:
        return None
    d = np.zeros(n, dtype=float)
    used = 0
    for lag in active_lags:
        if used >= 12:
            break
        g = grad_lag_count(h, lag, n)
        wt = float(rng.uniform(0.2, 1.0))
        d -= wt * g
        used += 1

    d += rng.normal(0.0, 1e-3, n)
    d -= d.mean()
    mx = float(np.max(np.abs(d)))
    if mx < 1e-14:
        return None
    w = h + (float(lim) / mx) * d
    return project_feasible(w, n)


def lp_polish(h, n, best_c, deadline, rng):
    n = int(n)
    h = h.copy()
    best_c = float(best_c)
    dx = 2.0 / float(n)

    no_improve = 0
    lim_cycle = [0.05, 0.12, 0.03, 0.25, 0.08]
    active_cycle = [
        (30, 0.004, 140),
        (15, 0.001, 80),
        (60, 0.008, 200),
        (10, 0.0005, 50),
    ]
    ai = 0

    while time.time() < deadline:
        vals = all_lags_fft(h)
        cur_c = float(np.max(vals)) * dx

        ce = c5_exact(h)
        if ce < best_c - 1e-12:
            best_c = ce
            h = project_feasible(h, n)

        top_k, margin, max_lags = active_cycle[ai % len(active_cycle)]
        active, _ = select_active(vals, n, cur_c, dx, top_k, margin, max_lags)
        improved = False

        for lim in lim_cycle:
            w = lp_step(h, n, active, cur_c, lim)
            if w is None:
                continue
            cand, _ = segment_search(h, w, n, best_c)
            if float(np.max(np.abs(cand - h))) > 1e-11:
                ce = c5_exact(cand)
                if ce < best_c - 1e-12:
                    h = project_feasible(cand, n)
                    best_c = float(ce)
                    improved = True
                    break

        if not improved:
            w = random_direction_w(h, active, n, 0.06, rng)
            if w is not None:
                cand, _ = segment_search(h, w, n, best_c)
                if float(np.max(np.abs(cand - h))) > 1e-11:
                    ce = c5_exact(cand)
                    if ce < best_c - 1e-12:
                        h = project_feasible(cand, n)
                        best_c = float(ce)
                        improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                break
        ai += 1

    return h, best_c


def pair_lag_coeff(h, p, q, s, n, lag, base_val):
    n = int(n)
    cand = [p, p - lag, q, q - lag]
    seen = []
    A = 0.0
    B = 0.0
    C = 0.0
    cur_aff = 0.0

    for i0 in cand:
        if i0 in seen:
            continue
        seen.append(i0)
        u = i0 + lag
        if i0 < 0 or i0 >= n or u < 0 or u >= n:
            continue

        hu = float(h[u])
        hv = float(h[i0])
        cur_aff += hu * (1.0 - hv)

        if u == p:
            aU = 1.0
            bU = 0.0
        elif u == q:
            aU = -1.0
            bU = s
        else:
            aU = 0.0
            bU = hu

        if i0 == p:
            aH = 1.0
            bH = 0.0
        elif i0 == q:
            aH = -1.0
            bH = s
        else:
            aH = 0.0
            bH = hv

        aV = -aH
        bV = 1.0 - bH

        A += aU * aV
        B += aU * bV + bU * aV
        C += bU * bV

    const = float(base_val) - cur_aff + C
    return A, B, const


def pair_count_fun(x, A, B, C, other):
    best = float(other)
    for r in range(len(A)):
        v = float(A[r]) * x * x + float(B[r]) * x + float(C[r])
        if v > best:
            best = v
    return best


def golden_min_pair(A, B, C, other, lo, hi):
    lo = float(lo)
    hi = float(hi)
    if hi <= lo:
        x = 0.5 * (lo + hi)
        return x, pair_count_fun(x, A, B, C, other)

    phi = float(math.sqrt(5.0) - 1.0) / 2.0
    span = hi - lo
    x1 = hi - phi * span
    x2 = lo + phi * span
    f1 = pair_count_fun(x1, A, B, C, other)
    f2 = pair_count_fun(x2, A, B, C, other)

    for _ in range(50):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            span = hi - lo
            x1 = hi - phi * span
            f1 = pair_count_fun(x1, A, B, C, other)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            span = hi - lo
            x2 = lo + phi * span
            f2 = pair_count_fun(x2, A, B, C, other)
        if hi - lo < 1e-7:
            break

    x = 0.5 * (lo + hi)
    f = pair_count_fun(x, A, B, C, other)
    fm = min(f, f1, f2)
    if fm < f:
        f = fm
    return x, f


def pair_find_best_x(h, p, q, n, vals, target_count, order):
    n = int(n)
    if p == q:
        return None
    m = abs(p - q)
    if m == 0 or m >= n:
        return None

    aff_idx = {n - 1, n - 1 + m, n - 1 - m}
    other = 0.0
    for idx in order:
        if int(idx) not in aff_idx:
            other = float(vals[idx])
            break

    if other >= float(target_count) - 1e-9:
        return None

    s = float(h[p]) + float(h[q])
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return None

    A = np.empty(3, dtype=float)
    B = np.empty(3, dtype=float)
    C = np.empty(3, dtype=float)

    for k, lag in enumerate((0, m, -m)):
        idx = n - 1 + lag
        A[k], B[k], C[k] = pair_lag_coeff(h, p, q, s, n, lag, vals[idx])

    best_x = float(neglide := 0.0)
    x0 = float(np.clip(h[p], lo, hi))
    best_count = pair_count_fun(x0, A, B, C, other)
    best_x = x0

    for x in np.linspace(lo, hi, 49):
        xf = float(x)
        cf = pair_count_fun(xf, A, B, C, other)
        if cf < best_count - 1e-14:
            best_count = cf
            best_x = xf

    if lo < best_x < hi:
        span = hi - lo
        l = max(lo, best_x - 0.25 * span)
        r = min(hi, best_x + 0.25 * span)
        for x in np.linspace(l, r, 41):
            xf = float(x)
            cf = pair_count_fun(xf, A, B, C, other)
            if cf < best_count - 1e-14:
                best_count = cf
                best_x = xf

        if r - l > 1e-5 and best_count < float(target_count) - 1e-10:
            gx, gf = golden_min_pair(A, B, C, other, l, r)
            if gf < best_count - 1e-14:
                best_count = gf
                best_x = gx

    if best_count < float(target_count) - 1e-11:
        return float(best_x), float(best_count)
    return None


def pair_local_search(h, n, best_c, deadline, rng):
    n = int(n)
    h = h.copy()
    best_c = float(best_c)
    dx = 2.0 / float(n)
    no_improve = 0

    while time.time() < deadline:
        if abs(float(h.sum()) - 0.5 * n) > 1e-9:
            h = project_feasible(h, n)

        vals = all_lags_fft(h)
        cur_count = float(np.max(vals))
        cur_c = cur_count * dx
        ce = c5_exact(h)
        if ce < best_c - 1e-12:
            best_c = float(ce)
            h = project_feasible(h, n)

        target_count = max(cur_count, best_c / dx)
        order = np.argsort(vals)[::-1]

        active_distances = set()
        for idx in order[:18]:
            lag = int(idx) - (n - 1)
            if lag != 0:
                active_distances.add(abs(lag))

        cand = []
        seen = set()

        for mabs in active_distances:
            mabs = int(mabs)
            L = n - mabs
            if L <= 0:
                continue
            step = max(1, L // 30)
            for i in range(0, L, step):
                p = i
                q = i + mabs
                if (p, q) not in seen:
                    seen.add((p, q))
                    cand.append((p, q))
            for _ in range(6):
                i = int(rng.randint(0, L))
                p = i
                q = i + mabs
                if (p, q) not in seen:
                    seen.add((p, q))
                    cand.append((p, q))

        for _ in range(35):
            i = int(rng.randint(0, n))
            j = int(rng.randint(0, n))
            if i == j:
                continue
            if i > j:
                i, j = j, i
            if (i, j) not in seen:
                seen.add((i, j))
                cand.append((i, j))

        cand = cand[:700]
        improved = False

        for p, q in cand:
            res = pair_find_best_x(h, p, q, n, vals, target_count, order)
            if res is None:
                continue
            best_x, bcount = res
            if bcount * dx >= best_c - 1e-12:
                continue

            oldp = float(h[p])
            oldq = float(h[q])
            s = oldp + oldq

            hp = float(np.clip(best_x, 0.0, 1.0))
            hq = float(np.clip(s - hp, 0.0, 1.0))
            h[p] = hp
            h[q] = hq

            pre_c = c5_exact(h)
            if pre_c >= best_c - 1e-12:
                h[p] = oldp
                h[q] = oldq
                continue

            if abs(float(h.sum()) - 0.5 * n) < 1e-8:
                h2 = h.copy()
                c2 = pre_c
            else:
                h2 = project_feasible(h, n)
                c2 = c5_exact(h2)

            if c2 < best_c - 1e-12:
                h = h2
                best_c = float(c2)
                improved = True
                break
            else:
                h[p] = oldp
                h[q] = oldq

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break

    return h, best_c


def add_noise(h, n, rng, sigma):
    n = int(n)
    d = rng.normal(0.0, sigma, n)
    d -= d.mean()
    w = h + d
    return project_feasible(w, n)


def smooth_obj_grad(x):
    n = int(len(x))
    dx = 2.0 / float(n)
    target = 0.5 * float(n)

    vals = all_lags_fft(x) * dx
    alpha = 200000.0

    m = float(vals.max())
    shifted = alpha * (vals - m)
    mx = float(shifted.max())

    if mx > 500.0:
        we = np.exp(shifted - mx)
        obj = m + (mx + float(np.log(np.sum(we)))) / alpha
        w = we / np.sum(we)
    else:
        w = np.exp(shifted)
        obj = m + float(np.log(np.sum(w))) / alpha
        w = w / np.sum(w)

    diff = float(x.sum()) - target
    penalty = 1.0e6
    obj += penalty * diff * diff

    g = np.zeros(n, dtype=float)
    order = np.argsort(w)[::-1]
    limit = min(60, len(order))
    for pos in range(limit):
        idx = int(order[pos])
        ww = float(w[idx])
        if ww < 1e-10:
            break
        lag = idx - (n - 1)
        g += ww * grad_lag_count(x, lag, n)

    g *= dx
    g += 2.0 * penalty * diff * np.ones(n, dtype=float)

    return float(obj), g


def smooth_lbfgs_polish(h, n, best_c, deadline):
    n = int(n)
    h0 = h.copy()
    try:
        maxiter = 90
        if deadline - time.time() > 90.0:
            maxiter = 140
        if deadline - time.time() > 180.0:
            maxiter = 200

        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0))

        res = minimize(smooth_obj_grad, h0, method="L-BFGS-B",
                       bounds=bounds, jac=True,
                       options={"maxiter": int(maxiter),
                                "ftol": 1e-15,
                                "gtol": 1e-12,
                                "iprint": -1})
        if res is not None and res.x is not None:
            h2 = project_feasible(res.x, n)
            c2 = c5_exact(h2)
            if c2 < best_c:
                return h2, float(c2)
    except Exception:
        pass
    return h0, float(best_c)


def polish_solution(h, n, best_c, deadline, rng):
    n = int(n)
    h_work = project_feasible(h, n)
    c_work = c5_exact(h_work)

    h_best = h_work.copy()
    best_c = float(min(best_c, c_work))
    if c_work < best_c + 1e-14:
        best_c = float(c_work)

    no_global = 0

    while time.time() < deadline:
        if deadline - time.time() < 20.0:
            break

        phase_dead = min(deadline - 5.0, time.time() + 150.0)
        h2, c2 = lp_polish(h_work, n, best_c, phase_dead, rng)
        if c2 < best_c - 1e-12:
            h_work = h2
            h_best = h2.copy()
            best_c = float(c2)
            no_global = 0
        else:
            h_work = h2
            no_global += 1

        if time.time() >= deadline - 10.0:
            break

        phase_dead = min(deadline - 3.0, time.time() + 90.0)
        h2, c2 = pair_local_search(h_work, n, best_c, phase_dead, rng)
        if c2 < best_c - 1e-12:
            h_work = h2
            h_best = h2.copy()
            best_c = float(c2)
            no_global = 0
        else:
            h_work = h2
            no_global += 1

        if time.time() >= deadline - 10.0:
            break

        if no_global >= 3 and deadline - time.time() > 80.0:
            h2, c2 = smooth_lbfgs_polish(h_work, n, best_c, deadline - 5.0)
            if c2 < best_c - 1e-12:
                h_work = h2
                h_best = h2.copy()
                best_c = float(c2)
                no_global = 0
            else:
                h_work = h2
                no_global += 1

        if no_global >= 6 and deadline - time.time() > 60.0:
            h_work = add_noise(h_best, n, rng, 0.03)
            no_global = 6

        if no_global > 14:
            break

    if deadline - time.time() > 8.0:
        h2, c2 = pair_local_search(h_work, n, best_c, deadline - 2.0, rng)
        if c2 < best_c - 1e-12:
            h_best = h2
            best_c = float(c2)

    return h_best, best_c


def insert_candidate(entries, c, h, n, max_entries=5):
    try:
        c = float(c)
    except Exception:
        return entries
    if math.isnan(c):
        return entries

    n = int(n)
    entries.append((c, h, n))

    for i in range(len(entries)):
        for j in range(i + 1, len(entries)):
            if entries[j][0] < entries[i][0]:
                tmp = entries[i]
                entries[i] = entries[j]
                entries[j] = tmp

    while len(entries) > max_entries:
        entries.pop()
    return entries


def adjust_even_n(n, max_n=1200):
    n = int(n)
    if n < 2:
        n = 2
    if n % 2 != 0:
        n += 1
    if n > int(max_n):
        m = int(max_n)
        if m % 2 != 0:
            m -= 1
        n = m
    if n < 2:
        n = 2
    return int(n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    if deadline <= t0 + 5.0:
        deadline = t0 + max(1.0, float(budget_s) - 1.0)

    rng = np.random.RandomState(seed)

    try:
        hpar_raw = np.asarray(initial_h_values, dtype=float).copy()
        npar = int(len(hpar_raw))
        parent_available = (npar >= 2)
        if not parent_available:
            raise RuntimeError
        if (np.min(hpar_raw) < -1e-9 or np.max(hpar_raw) > 1.0 + 1e-9 or
                abs(float(np.sum(hpar_raw)) - 0.5 * npar) > 1e-7):
            hpar = project_feasible(hpar_raw, npar)
        else:
            hpar = project_feasible(hpar_raw, npar)
    except Exception:
        hpar = project_feasible(np.full(500, 0.5, dtype=float), 500)
        npar = 500
        parent_available = False

    entries = []

    if npar >= 2:
        cpar = c5_exact(hpar)
        entries = insert_candidate(entries, cpar, hpar, npar, 6)

    if deadline - t0 > 120.0 and parent_available:
        n_list = [160, 200, 240, 280, 320, 360, 400, 450, 500, 560, 640, 750, 900, 1000]
        for nt in n_list:
            if time.time() > deadline - 300.0:
                break
            if nt == npar:
                continue
            try:
                ht = resample_h(hpar, npar, nt)
                ct = c5_exact(ht)
                entries = insert_candidate(entries, ct, ht, nt, 6)
            except Exception:
                pass

    if not parent_available:
        for nt in [300, 500, 700]:
            hr = make_random_h(nt, rng)
            ct = c5_exact(hr)
            entries = insert_candidate(entries, ct, hr, nt, 6)

    if not entries:
        h0 = project_feasible(np.full(500, 0.5, dtype=float), 500)
        c0 = c5_exact(h0)
        entries = [(c0, h0, 500)]

    if parent_available and deadline - t0 > 300.0:
        for _ in range(2):
            if time.time() > deadline - 220.0:
                break
            try:
                hr = add_noise(hpar, npar, rng, 0.05)
                cr = c5_exact(hr)
                if cr < entries[0][0] + 0.04:
                    entries = insert_candidate(entries, cr, hr, npar, 6)
            except Exception:
                pass

    best_h = entries[0][1]
    best_n = int(entries[0][2])
    best_c = float(entries[0][0])

    rem = deadline - time.time()
    if rem > 70.0:
        primary_len = min(rem, max(60.0, 0.55 * rem))
        try:
            h1, c1 = polish_solution(best_h, best_n, best_c, time.time() + primary_len, rng)
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
                best_n = int(len(best_h))
        except Exception:
            pass

    for e in entries:
        if time.time() > deadline - 90.0:
            break
        ce, he, ne = e
        ne = int(ne)
        if ne == best_n:
            continue
        if float(ce) > float(best_c) + 0.02:
            continue
        sub_dead = min(deadline - 40.0, time.time() + 45.0)
        if sub_dead <= time.time() + 10.0:
            continue
        try:
            h2, c2 = polish_solution(he, ne, ce, sub_dead, rng)
            if c2 < best_c:
                best_h = h2
                best_c = float(c2)
                best_n = int(len(best_h))
        except Exception:
            pass

    for factor in (0.8, 1.2):
        if time.time() > deadline - 40.0:
            break
        nt = adjust_even_n(int(round(best_n * factor)))
        if nt == best_n:
            continue
        try:
            ht = resample_h(best_h, best_n, nt)
            ct = c5_exact(ht)
            sub_dead = min(deadline - 20.0, time.time() + 30.0)
            if sub_dead <= time.time() + 10.0:
                continue
            h2, c2 = polish_solution(ht, nt, ct, sub_dead, rng)
            if c2 < best_c:
                best_h = h2
                best_c = float(c2)
                best_n = int(len(best_h))
        except Exception:
            pass

    if deadline - time.time() > 15.0:
        try:
            sub_dead = min(deadline - 2.0, time.time() + 20.0)
            h2, c2 = polish_solution(best_h, best_n, best_c, sub_dead, rng)
            if c2 < best_c:
                best_h = h2
                best_c = float(c2)
                best_n = int(len(best_h))
        except Exception:
            pass

    best_n = int(len(best_h))
    target = 0.5 * float(best_n)

    if (np.min(best_h) < -1e-10 or np.max(best_h) > 1.0 + 1e-10 or
            abs(float(best_h.sum()) - target) > 1e-8):
        best_h = project_feasible(best_h, best_n)

    final_c = c5_exact(best_h)
    best_h = np.asarray(best_h, dtype=float)

    return best_h, float(final_c), int(best_n)
