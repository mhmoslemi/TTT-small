        # Randomly flip 2-8 bits
        n_flips = rng.randint(1, 5)
        ones = np.where(h_trial > 0.7)[0]
        zeros = np.where(h_trial < 0.3)[0]
        
        if len(ones) < n_flips or len(zeros) < n_flips:
            continue
        
        flip_ones = rng.choice(ones, size=n_flips, replace=False)
        flip_zeros = rng.choice(zeros, size=n_flips, replace=False)
        
        h_trial[flip_ones] -= rng.uniform(0.3, 1.0, n_flips)
        h_trial[flip_zeros] += rng.uniform(0.3, 1.0, n_flips)
        h_trial = project_feasible(h_trial, n)
        
        # Short L-BFGS polish
        try:
            res = minimize(softmax_obj, h_trial, jac=softmax_grad,
                          args=(n, 50000, 0.01), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
            h_trial = project_feasible(res.x, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-14:
                best_c = c_trial
                best_h = h_trial.copy()
        except Exception:
            pass
    
    return best_h, best_c

def pair_swap_search(h, n, deadline, radius=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for delta in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + delta <= 1.0 and best_h[j] - delta >= 0.0:
                        best_h[i] += delta
                        best_h[j] -= delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= delta
                            best_h[j] += delta
                    if best_h[i] - delta >= 0.0 and best_h[j] + delta <= 1.0:
                        best_h[i] -= delta
                        best_h[j] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += delta
                            best_h[j] -= delta
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    
    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(200, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_softmax_obj, best_h, jac=active_softmax_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def upsample(h, n, n_new):
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n, n_new):
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def periodic_start(n, period, ones_per_period):
    h = np.zeros(n)
    for p in range(ones_per_period):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 2:
        n_ones = n // 2
    else:
        n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def run_slqp(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # Phase 1: L-BFGS on original resolution with alpha annealing
    p1_end = t0 + 180
    bounds = [(0.0, 1.0)] * best_n
    alphas_p1 = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for i, alpha in enumerate(alphas_p1):
        remaining = p1_end - time.time()
        if remaining < 15:
            break
        maxit = max(500, int(remaining * 4))
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(best_n, alpha, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: Momentum subgradient on exact top-K
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 120, deadline - 300)
        h2, c2 = momentum_subgrad(best_h, best_n, p2_end, top_k=10, n_iter_max=6000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Active-set L-BFGS
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 100, deadline - 200)
        h3, c3 = active_set_lbfgs(best_h, best_n, p3_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try different resolution (upsample to 2n)
    n_up = best_n * 2
    if n_up <= 2000 and time.time() < deadline - 250:
        h_up = upsample(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        bounds_up = [(0.0, 1.0)] * n_up
        
        p4_end = min(time.time() + 200, deadline - 150)
        for alpha in [100, 1000, 10000, 100000, 500000]:
            remaining = p4_end - time.time()
            if remaining < 15:
                break
            maxit = max(300, int(remaining * 3))
            try:
                res = minimize(softmax_obj, h_up, jac=softmax_grad,
                              args=(n_up, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds_up,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_up_new = project_feasible(res.x, n_up)
                c_up_new = compute_c5(h_up_new, n_up)
                if c_up_new < c_up - 1e-15:
                    c_up = c_up_new
                    h_up = h_up_new
            except Exception:
                pass
        
        # Active set on upsampled
        remaining = p4_end - time.time()
        if remaining > 30:
            h_up2, c_up2 = active_set_lbfgs(h_up, n_up, p4_end, top_k=8,
                alphas=[50000, 200000, 1000000])
            if c_up2 < c_up:
                c_up = c_up2
                h_up = h_up2
        
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n_up
    
    # Phase 5: Binary perturbation search
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = binary_perturb_optimize(best_h, best_n, p5_end, n_trials=40)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Try periodic structured start at best_n
    if time.time() < deadline - 150:
        # Try a few periodic patterns
        for period, ones in [(7, 3), (11, 5), (13, 6), (17, 8), (19, 9)]:
            if time.time() > deadline - 120:
                break
            h_per = periodic_start(best_n, period, ones)
            c_per = compute_c5(h_per, best_n)
            if c_per < best_c:
                # Optimize it
                bounds_n = [(0.0, 1.0)] * best_n
                try:
                    res = minimize(softmax_obj, h_per, jac=softmax_grad,
                                  args=(best_n, 10000, 0.01), method='L-BFGS-B',
                                  bounds=bounds_n,
                                  options={'maxiter': 1000, 'ftol': 1e-18,
                                           'gtol': 1e-14})
                    h_per = project_feasible(res.x, best_n)
                    c_per = compute_c5(h_per, best_n)
                except Exception:
                    pass
                if c_per < best_c:
                    best_c = c_per
                    best_h = h_per
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 60:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > deadline - 50:
                break
            p7_end = min(time.time() + 25, deadline - 40)
            h7, c7 = run_slqp(best_h, best_n, p7_end, alpha=alpha, maxiter=800)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Another round of momentum subgradient (finer)
    if time.time() < deadline - 40:
        p8_end = min(time.time() + 30, deadline - 25)
        h8, c8 = momentum_subgrad(best_h, best_n, p8_end, top_k=6, n_iter_max=4000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Pair swap search
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = pair_swap_search(best_h, best_n, p9_end, radius=25)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final: downsample back to n0 if we upsampled, to see if it's better
    if best_n != n0 and best_n > n0:
        h_ds = downsample(best_h, best_n, n0)
        c_ds = compute_c5(h_ds, n0)
        if c_ds < best_c:
            best_c = c_ds
            best_h = h_ds
            best_n = n0
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
