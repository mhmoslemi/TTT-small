    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(100):
        diff = target - h.sum()

            break
        if diff > 0:

            if tot < 1e-14:
                break

            if tot < 1e-14:

    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return float(m + log_sum / alpha)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)
