import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nv = 2 * n - 1
    center = n - 1
    for idx in range(nv):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nn = n - k
            if nn > 0:
                grad[:nn] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nn])
        else:
            kk = -k
            nn = n - kk
            if nn > 0:
                grad[kk:] += wm * (1.0 - x[:nn])
                grad[:nn] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 10.0), (500, 50.0), (2000, 200.0), (8000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 200000, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=5000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def eq_fun(x):
        return float(x.sum() - n / 2.0)
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
                diff -= 1.0
                if abs(diff) < 1e-12:
                    break
    else:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
                diff += 1.0
                if abs(diff) < 1e-12:
                    break
    return project_feasible(h_b, n)

def gradient_flow(h, n, deadline, n_iter=5000, step_size=0.005):
    """Projected gradient descent targeting top-K worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_top = 5
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Focus on top K worst shifts
        top_idx = np.argsort(-c_vec)[:n_top]
        top_c = c_vec[top_idx]
        
        # Compute subgradient for the max (averaged over top-K)
        dx_val = 2.0 / n
        grad = np.zeros(n)
        center = n - 1
        for idx in top_idx:
            k = idx - center
            # Equal weight for each top shift
            wm = dx_val / n_top * top_c[idx] / max(top_c.mean(), 1e-15)
            if k >= 0:
                nn = n - k
                if nn > 0:
                    grad[:nn] += wm * (1.0 - best_h[k:])
                    grad[k:] += wm * (-best_h[:nn])
            else:
                kk = -k
                nn = n - kk
                if nn > 0:
                    grad[kk:] += wm * (1.0 - best_h[:nn])
                    grad[:nn] += wm * (-best_h[kk:])
        
        # Projected gradient step
        h_new = best_h - step_size * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            # Optionally increase step size
        else:
            # Decrease step size
            step_size *= 0.85
            if step_size < 1e-6:
                break
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_start_lbfgs(h, n, deadline, configs, n_starts=3):
    """Run L-BFGS from multiple slightly perturbed starts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(999)
    starts = [h.copy()]
    for _ in range(n_starts - 1):
        h_p = h + 0.005 * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        starts.append(h_p)
    
    time_per = max(10, (deadline - time.time()) / n_starts)
    for s in range(n_starts):
        t_sub = time.time() + time_per
        if t_sub > deadline - 5:
            break
        h_s, c_s = run_lbfgs(starts[s], n, t_sub, configs)
        if c_s < best_c:

            best_h = h_s.copy()
    return best_h, best_c

def targeted_refine(h, n, deadline):
    """Fine coordinate descent on individual h values targeting worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    
    for rnd in range(300):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        # Compute gradient of the worst shift w.r.t. h
        if k_worst >= 0:
            nv = n - k_worst
            # overlap = sum_{i=0}^{nv-1} h[i]*(1-h[i+k])*dx
            # d/dh[j]: if j < nv: (1-h[j+k])*dx; if j>=k and j-k<nv: -h[j-k]*dx
            grad_worst = np.zeros(n)
            if nv > 0:
                grad_worst[:nv] += (1.0 - best_h[k_worst:k_worst+nv]) * dx_val
                grad_worst[k_worst:] += (-best_h[:nv]) * dx_val
        else:
            kk = -k_worst
            nv = n - kk
            if nv > 0:
                grad_worst[kk:] += (1.0 - best_h[:nv]) * dx_val
                grad_worst[:nv] += (-best_h[kk:]) * dx_val
        
        # Find the variable with largest |gradient| and adjust
        abs_grad = np.abs(grad_worst)
        order = np.argsort(-abs_grad)
        found = False
        for j in order[:50]:
            if time.time() > deadline:
                return best_h, best_c
            if abs_grad[j] < 1e-8:
                break
            # Step in direction that reduces the worst shift
            if grad_worst[j] > 0:
                delta = -0.001
            else:
                delta = 0.001
            new_val = best_h[j] + delta
            if new_val < 0 or new_val > 1:
                # Try smaller step
                delta *= -0.5
                new_val = best_h[j] + delta
                if new_val < 0 or new_val > 1:
                    continue
            
            old_val = best_h[j]
            best_h[j] = new_val
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                found = True
                # Try more steps
                for _ in range(20):
                    if time.time() > deadline:
                        break
                    if grad_worst[j] > 0:
                        nv2 = best_h[j] - 0.001
                    else:
                        nv2 = best_h[j] + 0.001
                    if nv2 < 0 or nv2 > 1:
                        break
                    old2 = best_h[j]
                    best_h[j] = nv2
                    c_chk = compute_c5(best_h, n)
                    if c_chk < best_c - 1e-15:
                        best_c = c_chk
                    else:
                        best_h[j] = old2
                        break
                break
            else:
                best_h[j] = old_val
        
        if not found and rnd > 50:
            break
    
    return best_h, best_c
