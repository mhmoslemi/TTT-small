        p2_end = deadline - 600
    active_configs = [
        (3, 50000, 0.001, 1500),
        (4, 100000, 0.001, 1500),
        (5, 200000, 0.0005, 1200),
        (6, 500000, 0.0005, 1200),
        (8, 1000000, 0.0003, 1000),
        (10, 3000000, 0.0003, 1000),
        (12, 5000000, 0.0002, 800),
        (15, 10000000, 0.0002, 800),
    ]
    for top_k, alpha, mu, maxit in active_configs:
        if time.time() > p2_end - 20:
            break
        for _ in range(2):
            if time.time() > p2_end - 10:
                break
            h2, c2 = run_active_lbfgs(best_h, best_n, p2_end - 5, top_k, alpha, mu, maxit)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient descent fine-tuning (100s)
    p3_end = t0 + 470
    if p3_end > deadline - 500:
        p3_end = deadline - 500
    if p3_end > time.time() + 20:
        h3, c3 = gradient_descent_step(best_h, best_n, 100000, 0.001, 2000, p3_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Coordinate descent (200s)
    p4_end = t0 + 670
    if p4_end > deadline - 350:
        p4_end = deadline - 350
    if p4_end > time.time() + 30:
        h4, c4 = coordinate_descent(best_h, best_n, p4_end, max_rounds=100)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Focused pair search (150s)
    p5_end = t0 + 820
    if p5_end > deadline - 200:
        p5_end = deadline - 200
    if p5_end > time.time() + 30:
        h5, c5 = pair_search_focused(best_h, best_n, p5_end,
                                     [0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Global random pair search (100s)
    p6_end = t0 + 920
    if p6_end > deadline - 120:
        p6_end = deadline - 120
    if p6_end > time.time() + 20:
        h6, c6 = global_pair_search(best_h, best_n, p6_end, 50, 0.001)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Perturbation + re-optimize (100s)
    p7_end = t0 + 1020
    if p7_end > deadline - 60:
        p7_end = deadline - 60
    if p7_end > time.time() + 30:
        h7, c7 = perturb_and_reopt(best_h, best_n, p7_end, rng, 8, 50000, 0.005)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final ultra-fine polish (remaining)
    p8_end = deadline - 2
    if p8_end > time.time() + 10:
        h8, c8 = coordinate_descent(best_h, best_n, p8_end, max_rounds=50)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        
        if p8_end > time.time() + 5:
            h8b, c8b = pair_search_focused(best_h, best_n, p8_end,
                                           [0.0005, 0.0002, 0.0001, 0.00005])
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
