import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    h_bin = np.zeros(n)
    # Sort by value, pick top n/2
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    # Fix if needed (shouldn't be needed with this approach)
    assert h_bin.sum() == target
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(1e-308 + np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = h_grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def simulated_annealing_binary(h_bin, n, deadline, T0, T1, max_iters, rng, tabu_len=200, max_swaps=3):
    """Simulated annealing on binary vectors with exactly n/2 ones."""
    h = h_bin.copy()
    current_c = compute_c5(h, n)
    best_c = current_c
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    tabu = {}  # maps (i,j) set to iteration when it becomes untabu
    current_iter = 0
    
    total_iters = max_iters
    for it in range(total_iters):
        if time.time() > deadline:
            break
        
        # Temperature
        frac = it / total_iters
        T = T0 * (T1 / T0) ** frac
        
        # Choose move size
        if rng.random() < 0.6:
            k_swap = 1
        elif rng.random() < 0.8:
            k_swap = 2
        else:
            k_swap = min(max_swaps, 3)
        
        # Select random positions
        i_idx = rng.choice(ones, size=k_swap, replace=False)
        j_idx = rng.choice(zeros, size=k_swap, replace=False)
        
        # Check tabu
        move_key = frozenset(zip(i_idx, j_idx))
        if move_key in tabu and tabu[move_key] > it:
            continue
        
        # Apply move
        for i, j in zip(i_idx, j_idx):
            h[i] = 0.0
            h[j] = 1.0
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        
        # Evaluate
        new_c = compute_c5(h, n)
        delta = new_c - current_c
        
        # Accept?
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            tabu[move_key] = it + tabu_len
            if len(tabu) > 5000:
                # Clean old tabu entries
                tabu = {k: v for k, v in tabu.items() if v > it}
            
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            for i, j in zip(i_idx, j_idx):
                h[j] = 0.0
                h[i] = 1.0
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
        
        current_iter = it
    
    return best_h, best_c

def tabu_search_binary(h_bin, n, deadline, max_iters, rng, tabu_len=500):
    """Tabu search with aspiration criterion."""
    h = h_bin.copy()
    current_c = compute_c5(h, n)
    best_c = current_c
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    tabu_until = {}  # frozenset -> iteration until which it's tabu
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        if no_improve > 500:
            # Restart with perturbation
            for _ in range(10):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                ones.remove(i); ones.append(j)
                zeros.remove(j); zeros.append(i)
            current_c = compute_c5(h, n)
            no_improve = 0
            continue
        
        # Try all single swaps (or a random subset for large n)
        if n <= 600:
            candidates = [(i, j) for i in ones for j in zeros]
            rng.shuffle(candidates)
            candidates = candidates[:min(500, len(candidates))]
        else:
            candidates = []
            for _ in range(1000):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                candidates.append((i, j))
        
        best_move = None
        best_move_c = current_c
        best_move_key = None
        
        for i, j in candidates:
            move_key = frozenset([(i, j)])
            if move_key in tabu_until and tabu_until[move_key] > it:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h, n)
            
            # Aspiration: allow tabu move if it improves best
            if c < best_c or (move_key not in tabu_until or tabu_until[move_key] <= it):
                if c < best_move_c:
                    best_move_c = c
                    best_move = (i, j)
                    best_move_key = move_key
            
            h[j] = 0.0
            h[i] = 1.0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            tabu_until[best_move_key] = it + tabu_len
            current_c = best_move_c
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c

            if h_tmp.sum() >= target:

        # Fill remaining
        if h_tmp.sum() < target:
            remaining = target - int(h_tmp.sum())
            zeros = np.where(h_tmp == 0)[0]
            # Spread remaining ones
            step = max(1, len(zeros) // remaining)
            for idx in range(0, len(zeros), step):
                if h_tmp.sum() >= target:

                h_tmp[zeros[idx]] = 1.0

        while h_tmp.sum() < target:
            h_tmp[np.where(h_tmp == 0)[0][0]] = 1.0

            h_tmp[np.where(h_tmp == 1)[0][-1]] = 0.0

        if c < best_c:

            best_h = h_tmp.copy()

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    order = np.argsort(-h)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(1e-308 + np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w[:] = 0.0
        w[np.argmax(c)] = 1.0
        return w
    w /= s
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxit):

    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val

    def grad_fn(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = h_grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h.copy(), jac=grad_fn, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)

            max_val = c_vec.max()
            w = smooth_max_w(c_vec, 150000.0)
            grad = h_grad_from_w(w, best_h, n)

            pos_mask = (grad > 1e-12) & (best_h > 1e-6)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-6)

            if not pos_mask.any() or not neg_mask.any():
                break

            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]

            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                for j in neg_idx[:40]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j

            if best_i < 0:
                break

            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = target - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for idx in range(min(diff, len(zeros))):
            h[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for idx in range(min(-diff, len(ones))):
            h[ones[idx]] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    random.seed(seed)

    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    best_n = n_init

    # Phase 1: Quick L-BFGS to get a good continuous starting point (35s)
    p1_deadline = t0 + 35
    h1, c1 = run_lbfgs(h_init, n_init, p1_deadline, alpha=50000, mu=0.01, maxit=1500)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Multi-resolution combinatorial search (750s)
    # Try parent resolution and a few others
    res_list = [n_init]
    for nr in [400, 600, 800]:
        if nr not in res_list:
            res_list.append(nr)
    # Sort: parent first, then others
    res_list = [n_init] + [r for r in res_list if r != n_init]

    p2_start = t0 + 35
    p2_end = deadline - 180
    total_p2_time = p2_end - p2_start
    # Allocate more time to parent resolution
    time_weights = [0.45, 0.2, 0.2, 0.15]
    time_weights = time_weights[:len(res_list)]
    total_w = sum(time_weights)
    time_weights = [w / total_w * total_p2_time for w in time_weights]

    for ri, n in enumerate(res_list):
        res_start = p2_start + sum(time_weights[:ri])
        res_end = res_start + time_weights[ri]
        res_end = min(res_end, deadline - 175)

        if time.time() >= res_end - 5:
            continue

        # Initialize at this resolution
        if n == n_init:
            h_res = binarize(h_init, n)
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_old, h_init)
            h_res = binarize(h_interp, n)

        # Also try Beatty sequence as alternative start
        for alpha_val in [0.6180339887498949, 0.5 + 0.5 / np.sqrt(2.0)]:
            h_beatty = make_beatty(n, alpha_val)
            if compute_c5(h_beatty, n) < compute_c5(h_res, n):
                h_res = h_beatty

        # Sub-phase A: Quick greedy (30% of resolution time)
        greedy_end = res_start + time_weights[ri] * 0.30
        h_res, c_res = greedy_improve_binary(h_res, n, greedy_end, max_iters=80000, rng=rng)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
            best_n = n

        # Sub-phase B: Simulated annealing (60% of resolution time)
        sa_end = res_start + time_weights[ri] * 0.90
        sa_end = min(sa_end, res_end - 10)
        if time.time() < sa_end:
            h_sa, c_sa = simulated_annealing_binary(h_res, n, sa_end, rng,
                                                    T0=0.004, T1=0.00002,
                                                    max_swaps=3, greedy_freq=400,
                                                    perturb_every=2500)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
                best_n = n
            h_res = h_sa

        # Sub-phase C: Final greedy polish (10% of resolution time)
        if time.time() < res_end:
            h_fin, c_fin = greedy_improve_binary(h_res, n, res_end, max_iters=30000, rng=rng)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin.copy()
                best_n = n

    # Phase 3: Continuous refinement on best solution (120s)
    p3_deadline = deadline - 60
    if time.time() < p3_deadline:
        h3, c3 = fine_coord_descent(best_h, best_n, p3_deadline, max_rounds=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: L-BFGS refinement (60s)
    p4_deadline = deadline - 20
    if time.time() < p4_deadline:
        for alpha in [200000, 1000000, 5000000]:
            if time.time() > p4_deadline - 8:
                break
            sub_dl = min(p4_deadline, time.time() + 40)
            h4, c4 = run_lbfgs(best_h, best_n, sub_dl, alpha=alpha, mu=0.001, maxit=800)
            if c4 < best_c:
                best_c = c4
                best_h = h4.copy()

    # Phase 5: Final combinatorial polish
    if time.time() < deadline - 3:
        h5, c5 = greedy_improve_binary(binarize(best_h, best_n), best_n, deadline - 2,
                                        max_iters=50000, rng=rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
