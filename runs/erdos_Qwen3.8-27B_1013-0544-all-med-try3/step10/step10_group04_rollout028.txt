```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

```python
def binary_local_search(h, n, max_iter=100000, time_limit=None, t0=None):
    """Local search on binary sequences using swap moves."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Ensure exactly n/2 ones
    n_ones = h_bin.sum()
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove excess ones (those closest to 0.5)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])  # ones with smallest h values first
        for idx in order[:n_ones - target_ones]:
            h_bin[ones[idx]] = 0
    elif n_ones < target_ones:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])  # zeros with largest h values first
        for idx in order[:target_ones - n_ones]:
            h_bin[zeros[idx]] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for it in range(max_iter):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        # Try random swaps
        improved = False
        for _ in range(200):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_bin[i], h_bin[j] = 0, 1
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-15:

                best_h = h_bin.copy()
                improved = True
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            else:
                h_bin[i], h_bin[j] = 1, 0
        if not improved:
            # Try all swaps (expensive but thorough) - only for small n
            if n <= 500:
                found = False
                for i in ones:
                    for j in zeros:
                        h_bin[i], h_bin[j] = 0, 1

                    if found:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = w / s
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / s
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slqp_cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def slqp_cons_jac(x, n):
    return np.ones(n)

def run_lbfgs_multi(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(obj_fn, h, jac=grad_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def run_slqp(h, n, deadline, alpha=500000, maxiter=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_cons_fun, 'jac': slqp_cons_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                return h_new, c_new
    except Exception:
        pass
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_local_search(h, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - target]:
            h_bin[ones_idx[idx]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:target - n_ones]:
            h_bin[zeros_idx[idx]] = 1

    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    c_vec_all = compute_c5_vec(h_bin, n)
    t_start = time.time()
    fail_count = 0

    while time.time() - t_start < time_limit:
        c_max = c_vec_all.max()
        k_star = int(np.argmax(c_vec_all))
        k = k_star - (n - 1)

        if k >= 0:
            wits = np.array([i for i in range(n - k) if h_bin[i] > 0.5 and h_bin[i + k] < 0.5])
        else:
            kk = -k
            wits = np.array([i for i in range(kk, n) if h_bin[i] > 0.5 and h_bin[i - kk] < 0.5])

        if len(wits) == 0:
            ones_idx = np.where(h_bin > 0.5)[0]
            zeros_idx = np.where(h_bin < 0.5)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = int(ones_idx[rng.randint(len(ones_idx))])
            j = int(zeros_idx[rng.randint(len(zeros_idx))])
        else:
            i = int(wits[rng.randint(len(wits))])
            zeros_idx = np.where(h_bin < 0.5)[0]
            if len(zeros_idx) == 0:
                break
            j = int(zeros_idx[rng.randint(len(zeros_idx))])
            if j == i:
                continue

        h_bin[i], h_bin[j] = 0.0, 1.0
        h_bin = project_feasible(h_bin, n)
        new_c = compute_c5(h_bin, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            best_h = h_bin.copy()
            c_vec_all = compute_c5_vec(h_bin, n)
            fail_count = 0
        else:
            h_bin = best_h.copy()
            fail_count += 1
            if fail_count > 8000:
                break

    return best_h, best_c

def make_block_pattern(n, block_len):
    h = np.zeros(n)
    for i in range(n):
        period = 2 * block_len
        if (i % period) < block_len:
            h[i] = 1.0
    return project_feasible(h, n)

def make_residue_pattern(n, modulus, residues, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for i in range(n):
        if i % modulus in residues:
            h[i] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            rng.shuffle(zeros)
            for idx in zeros[:int(diff)]:
                h[idx] = 1.0
        else:
            ones = np.where(h == 1)[0]
            rng.shuffle(ones)
            for idx in ones[:int(-diff)]:
                h[idx] = 0.0
    return project_feasible(h, n)

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:n // 2]] = 1.0
    return project_feasible(h, n)

def perturbation_refine(h, n, deadline, seed=0, n_trials=8, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    if mags is None:
        mags = [0.02, 0.05, 0.08, 0.03, 0.10, 0.015, 0.06, 0.12]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(obj_fn, h_p, jac=grad_fn,
                          args=(n, 3000, 5000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
        h_bin = np.where(h_p > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        try:
            res2 = minimize(obj_fn, h_bin, jac=grad_fn,
                           args=(n, 2000, 3000.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-18, 'maxls': 50})
            h_new2 = project_feasible(res2.x, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c - 1e-14:
                best_c = c_new2
                best_h = h_new2.copy()
        except Exception:
            pass
    return best_h, best_c

    # Final feasibility check

    best_c = compute_c5(best_h, n_final)
```
