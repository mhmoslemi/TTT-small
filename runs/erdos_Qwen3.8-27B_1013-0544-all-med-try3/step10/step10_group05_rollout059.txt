def grad_active(x, n, active, alpha, mu):
    c = c5_vec(x, n)
    # For active subset, recompute smooth max weights
    vals = c[active]
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    ctr = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - ctr
        wm = wi * dx
        if k >= 0:

            if nv > 0:
                g[:nv] += wm * (1.0 - x[k:n])
                g[k:] += wm * (-x[:nv])
        else:
            jk = -k
            nv = n - jk
            if nv > 0:
                g[jk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[jk:n])
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def pair_refine(h, n, deadline, radius, deltas):
    best_h = h.copy()
    best_c = c5_val(best_h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move mass from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Move mass from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(h, n):
    return float(c5_vec(h, n).max())

def proj(h, n):
    t = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        d = t - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def smax(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smax_g(c, alpha, x, n):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    ctr = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - ctr
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - x[k:n])
                g[k:] += wm * (-x[:nv])
        else:
            jk = -k
            nv = n - jk
            if nv > 0:
                g[jk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[jk:n])
    return g

def obj_full(x, n, alpha, mu):
    c = c5_vec(x, n)
    return smax(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def grad_full(x, n, alpha, mu):
    c = c5_vec(x, n)
    g = smax_g(c, alpha, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_topk(x, n, k, mu):
    c = c5_vec(x, n)
    idx = np.argsort(c)[-k:]
    return float(np.sum(c[idx] ** 2)) + mu * (x.sum() - n / 2.0) ** 2

def grad_topk(x, n, k, mu):
    c = c5_vec(x, n)
    idx = np.argsort(c)[-k:]
    ctr = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in idx:
        kk = int(ii) - ctr
        wm = 2.0 * c[ii] * dx
        if kk >= 0:
            nv = n - kk
            if nv > 0:
                g[:nv] += wm * (1.0 - x[kk:n])
                g[kk:] += wm * (-x[:nv])
        else:
            jk = -kk
            nv = n - jk
            if nv > 0:
                g[jk:] += wm * (1.0 - x[:nv])
                g[:nv] += wm * (-x[jk:n])
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_active(x, n, active, alpha, mu):
    c = c5_vec(x, n)
    return smax(c[active], alpha) + mu * (x.sum() - n / 2.0) ** 2

    vals = c[active]
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()

    dx = 2.0 / n

            continue

        wm = wi * dx
        if k >= 0:

            nv = n - jk

                g[:nv] += wm * (-x[jk:n])

        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

            improved = False
            c_vec = c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active_lags = np.where(c_vec >= max_val - tol)[0]
            ctr = n - 1
            n_active = min(len(active_lags), 8)
            if n_active == 0:
                break
            active_lags = np.array(np.argsort(-c_vec[active_lags])[:n_active])
            for lag_idx in active_lags:
                if time.time() > deadline - 3:
                    return best_h, best_c
                kk = int(lag_idx) - ctr
                if kk >= 0:
                    pairs = [(i, i + kk) for i in range(n - kk)]
                else:
                    jk = -kk
                    pairs = [(i + jk, i) for i in range(n - jk)]
                n_tested = 0
                for (a, b) in pairs:
                    if n_tested > 5000:
                        break
                    n_tested += 1
                    if best_h[a] >= d and best_h[b] <= 1.0 - d:
                        best_h[a] -= d
                        best_h[b] += d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[a] += d
                            best_h[b] -= d
                    if best_h[b] >= d and best_h[a] <= 1.0 - d:
                        best_h[b] -= d
                        best_h[a] += d
                        c = c5_val(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[b] += d
                            best_h[a] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = beatty(500, (1.0 + np.sqrt(5.0)) / 2.0 / 2.0)
        n0 = 500

    h0 = proj(h0, n0)

    starts = [h0.copy()]
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts.append(beatty(n0, golden / 2.0))
    starts.append(beatty(n0, np.sqrt(2.0) / 2.0))
    starts.append(beatty(n0, 0.4312))
    starts.append(beatty(n0, 0.382))
    starts.append(rand_bin(n0, rng))
    starts.append(sine_h(n0, 3.0, 0.7))
    starts.append(sine_h(n0, 7.0, 2.1))
    starts.append(sine_h(n0, 11.0, 4.0))

    best_start = starts[0]
    best_sc = c5_val(best_start, n0)
    for s in starts[1:]:
        sc = c5_val(s, n0)
        if sc < best_sc:
            best_sc = sc
            best_start = s.copy()

    best_h = best_start.copy()
    best_c = best_sc
    best_n = n0

    n_work = n0 * 2
    if n_work < 1000:
        n_work = 1000
    h_work = upsample(best_h, best_n, n_work)
    c_work = c5_val(h_work, n_work)

    bounds = [(0.0, 1.0)] * n_work

    # Phase 1: Coarse-to-fine smooth max L-BFGS
    p1_end = min(t0 + 90, deadline - 750)
    if p1_end > time.time() + 15:
        for al in [50, 500, 5000]:
            if time.time() > p1_end - 8:
                break
            rem = p1_end - time.time() - 5
            if rem < 15:
                break
            mi = max(300, min(3000, int(rem * 2)))
            try:
                res = minimize(obj_full, h_work, jac=grad_full,
                              args=(n_work, al, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': mi, 'ftol': 1e-20, 'gtol': 1e-14,
                                       'maxfun': mi * 6, 'maxls': 30, 'maxcor': 30})
                hn = proj(res.x, n_work)
                cn = c5_val(hn, n_work)
                if cn < c_work:
                    c_work = cn
                    h_work = hn
            except Exception:
                pass

    # Phase 2: Top-k squared optimization
    p2_end = min(time.time() + 120, deadline - 550)
    if p2_end > time.time() + 15:
        for k, al in [(5, None), (3, None)]:
            if time.time() > p2_end - 8:
                break
            rem = p2_end - time.time() - 5
            if rem < 15:
                break
            mi = max(300, min(2000, int(rem * 1.5)))
            try:
                res = minimize(obj_topk, h_work, jac=grad_topk,
                              args=(n_work, k, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': mi, 'ftol': 1e-20, 'gtol': 1e-14,
                                       'maxfun': mi * 6, 'maxls': 30, 'maxcor': 30})
                hn = proj(res.x, n_work)
                cn = c5_val(hn, n_work)
                if cn < c_work:
                    c_work = cn
                    h_work = hn
            except Exception:
                pass

    # Phase 3: High-alpha smooth max
    p3_end = min(time.time() + 160, deadline - 400)
    if p3_end > time.time() + 15:
        for al in [20000, 200000, 2000000]:
            if time.time() > p3_end - 8:
                break
            rem = p3_end - time.time() - 5
            if rem < 15:
                break
            mi = max(400, min(4000, int(rem * 2)))
            try:
                res = minimize(obj_full, h_work, jac=grad_full,
                              args=(n_work, al, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': mi, 'ftol': 1e-22, 'gtol': 1e-16,
                                       'maxfun': mi * 8, 'maxls': 40, 'maxcor': 40})
                hn = proj(res.x, n_work)
                cn = c5_val(hn, n_work)
                if cn < c_work:
                    c_work = cn
                    h_work = hn
            except Exception:
                pass

    # Phase 4: Active set rounds
    p4_end = min(time.time() + 180, deadline - 250)
    if p4_end > time.time() + 15:
        for topk, al in [(4, 50000), (6, 200000), (5, 500000), (3, 2000000), (5, 5000000)]:
            if time.time() > p4_end - 10:
                break
            cv = c5_vec(h_work, n_work)
            mv = cv.max()
            thr = mv - max(1e-8, 0.001 * mv)
            act = np.where(cv >= thr)[0]
            if len(act) > topk:
                act = np.array(np.argsort(-cv[act])[:topk])
            if len(act) == 0:
                continue
            rem = p4_end - time.time() - 8
            if rem < 10:
                break
            mi = max(200, min(2500, int(rem * 2)))
            try:
                res = minimize(obj_active, h_work, jac=grad_active,
                              args=(n_work, act, al, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': mi, 'ftol': 1e-22, 'gtol': 1e-16,
                                       'maxfun': mi * 8, 'maxls': 40, 'maxcor': 40})
                hn = proj(res.x, n_work)
                cn = c5_val(hn, n_work)
                if cn < c_work:
                    c_work = cn
                    h_work = hn
            except Exception:
                pass

    # Phase 5: Multi-restart perturbation
    p5_end = min(time.time() + 100, deadline - 200)
    if p5_end > time.time() + 20:
        for t in range(10):
            if time.time() > p5_end - 30:
                break
            amp = 0.015 + 0.005 * t
            hp = h_work + rng.uniform(-amp, amp, size=n_work)
            hp = proj(hp, n_work)
            cp = c5_val(hp, n_work)
            if cp < c_work + 0.004:
                rem = p5_end - time.time() - 20
                if rem < 20:
                    break
                mi = max(200, min(600, int(rem * 0.8)))
                try:
                    res = minimize(obj_full, hp, jac=grad_full,
                                  args=(n_work, 50000, 0.005), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': mi, 'ftol': 1e-22, 'gtol': 1e-16,
                                           'maxfun': mi * 6, 'maxls': 30, 'maxcor': 30})
                    hn = proj(res.x, n_work)
                    cn = c5_val(hn, n_work)
                    if cn < c_work:
                        c_work = cn
                        h_work = hn
                except Exception:
                    pass

    # Phase 6: Targeted pair search
    p6_end = min(time.time() + 120, deadline - 130)
    if p6_end > time.time() + 15:
        h_work, c_work = targeted_pair_refine(h_work, n_work, p6_end,
                                               deltas=[0.002, 0.001, 0.0005, 0.0002])

    # Phase 7: Final active set with very high alpha
    p7_end = min(time.time() + 100, deadline - 60)
    if p7_end > time.time() + 10:
        for topk, al in [(3, 5000000), (4, 10000000), (5, 20000000)]:
            if time.time() > p7_end - 8:
                break
            cv = c5_vec(h_work, n_work)
            mv = cv.max()
            thr = mv - max(1e-9, 0.0005 * mv)
            act = np.where(cv >= thr)[0]
            if len(act) > topk:
                act = np.array(np.argsort(-cv[act])[:topk])
            if len(act) == 0:
                continue
            rem = p7_end - time.time() - 5
            if rem < 8:
                break
            mi = max(150, min(1500, int(rem * 2)))
            try:
                res = minimize(obj_active, h_work, jac=grad_active,
                              args=(n_work, act, al, 0.0005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': mi, 'ftol': 1e-22, 'gtol': 1e-16,
                                       'maxfun': mi * 8, 'maxls': 40, 'maxcor': 40})
                hn = proj(res.x, n_work)
                cn = c5_val(hn, n_work)
                if cn < c_work:
                    c_work = cn
                    h_work = hn
            except Exception:
                pass

    # Phase 8: Ultra-fine pair search
    p8_end = deadline - 3
    if p8_end > time.time() + 10:
        h_work, c_work = targeted_pair_refine(h_work, n_work, p8_end,
                                               deltas=[0.0001, 0.00005, 0.00002, 0.00001])

    # Update overall best
    if c_work < best_c:
        best_c = c_work
        best_h = h_work
        best_n = n_work

    # Final projection and verification
    best_h = proj(best_h, best_n)
    best_c = c5_val(best_h, best_n)

    return best_h, best_c, best_n
