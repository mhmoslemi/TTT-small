import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, maxls=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (5000, 5000.0), (15000, 20000.0), (40000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(softmax_obj, h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': maxls})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def worst_lag_local_search(h, n, deadline, n_rounds=200):
    """Targeted local search focusing on worst lags with pair adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        center = n - 1
        # Get the worst few lags
        top_k = 5
        worst_indices = np.argsort(c_vec)[::-1][:top_k]
        
        improved = False
        for wi in worst_indices:
            k = int(wi) - center
            if k == 0:
                continue
            kk = abs(k)
            if kk >= n:
                continue
            nv = n - kk
            # For this lag, overlap = sum h[i]*(1-h[i+kk])*dx (for k>0)
            # or sum h[i+kk]*(1-h[i])*dx (for k<0)
            # Try to reduce by moving mass
            if k > 0:
                # h[i]*(1-h[i+k]) is the problem term
                # Reduce h[i] where h[i+k] is small, increase h[i+k] where h[i] is large
                for _iter in range(nv // 5):
                    if time.time() > deadline:
                        return best_h, best_c
                    # Find the worst contributor
                    contrib = best_h[0:nv] * (1.0 - best_h[kk:n])
                    max_idx = int(np.argmax(contrib))
                    if contrib[max_idx] < 0.01:
                        break
                    # Try to move mass from max_idx to max_idx+kk
                    i_from = max_idx
                    i_to = max_idx + kk
                    # Transfer epsilon from i_from to i_to
                    for eps in [0.02, 0.01, 0.005, 0.03, 0.015]:
                        if best_h[i_from] < eps or best_h[i_to] > 1.0 - eps:
                            continue
                        new_h = best_h.copy()
                        new_h[i_from] -= eps
                        new_h[i_to] += eps
                        # Also need to maintain sum - this move maintains sum
                        if new_h[i_from] >= -1e-12 and new_h[i_to] <= 1.0 + 1e-12:
                            c_test = compute_c5(new_h, n)
                            if c_test < best_c - 1e-12:
                                best_c = c_test
                                best_h = new_h
                                improved = True
                                break
                    if not improved:
                        break
            else:
                # k < 0: overlap involves h[i+|k|]*(1-h[i])
                kk2 = -k
                for _iter in range(nv // 5):
                    if time.time() > deadline:
                        return best_h, best_c
                    contrib = best_h[kk2:n] * (1.0 - best_h[0:nv])
                    max_idx = int(np.argmax(contrib))
                    if contrib[max_idx] < 0.01:
                        break
                    i_from = max_idx + kk2
                    i_to = max_idx
                    for eps in [0.02, 0.01, 0.005, 0.03, 0.015]:
                        if best_h[i_from] < eps or best_h[i_to] > 1.0 - eps:
                            continue
                        new_h = best_h.copy()
                        new_h[i_from] -= eps
                        new_h[i_to] += eps
                        if new_h[i_from] >= -1e-12 and new_h[i_to] <= 1.0 + 1e-12:
                            c_test = compute_c5(new_h, n)
                            if c_test < best_c - 1e-12:
                                best_c = c_test
                                best_h = new_h
                                improved = True
                                break
                    if not improved:
                        break
        if not improved:
            # Try single-coordinate moves
            single_improved = False
            for _iter in range(3):
                if time.time() > deadline:
                    return best_h, best_c
                c_vec = compute_c5_vec(best_h, n)
                worst_idx = int(np.argmax(c_vec))
                k = worst_idx - center
                kk = abs(k)
                if kk == 0 or kk >= n:
                    break
                nv = n - kk
                for i in range(0, nv, max(1, nv // 30)):
                    if time.time() > deadline:
                        return best_h, best_c
                    if k > 0:
                        j = i + kk
                    else:
                        j = i + kk
                    # Try adjusting both i and j toward each other
                    hi = best_h[i]
                    hj = best_h[j]
                    for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                        ni = hi + delta
                        nj = hj - delta
                        if ni < -1e-10 or ni > 1.0 + 1e-10:
                            continue
                        if nj < -1e-10 or nj > 1.0 + 1e-10:
                            continue
                        new_h = best_h.copy()
                        new_h[i] = ni
                        new_h[j] = nj
                        c_test = compute_c5(new_h, n)
                        if c_test < best_c - 1e-12:
                            best_c = c_test
                            best_h = new_h
                            single_improved = True
                            break
                    if single_improved:
                        break
                if single_improved:
                    break
            if not single_improved and rnd > 30:
                break
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=2000000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return project_feasible(h, n)

        best_h, best_c = multi_start_search(best_h, n_best_res, t6, n_starts=4, seed_offset=seed)
    
    # Phase 7: Final L-BFGS polish (80s)
    t7 = min(t0 + 890, deadline - 50)
    if t7 > time.time() + 15:
        n_cur = len(best_h)
        best_h, best_c = temperature_anneal_lbfgs(best_h, n_cur, t7,
            temps=(10000, 30000, 100000, 200000))
    
    # Phase 8: Final worst-lag local search (60s)
    t8 = min(t0 + 950, deadline - 10)
    if t8 > time.time() + 10:
        best_h, best_c = worst_lag_local_search(best_h, len(best_h), t8, n_rounds=200)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
