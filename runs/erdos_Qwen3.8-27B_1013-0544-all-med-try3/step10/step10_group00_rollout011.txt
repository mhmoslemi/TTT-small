        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)

                break

            i = pos_global[np.argmax(contrib[pos_idx])]
            j = neg_global[rng.randint(len(neg_global))] if len(neg_global) > 1 else neg_global[0]
            
            amt = min(step, h[i], 1.0 - h[j])

                break

            h[j] += amt
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            else:
                h[i] += amt
                h[j] -= amt
