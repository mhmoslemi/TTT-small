    """Compute all 2n-1 overlap values using FFT."""
    dx = 2.0 / n
    h_pad = np.zeros(2 * n)
    h_pad[:n] = h
    H = fft(h_pad)
    autocorr = ifft(H * np.conj(H)).real[:n]
    
    prefix = np.concatenate([[0.0], np.cumsum(h)])
    
    # c for shift +k (k=0,...,n-1): (prefix[n-k] - autocorr[k]) * dx
    k = np.arange(n)
    c_pos = (prefix[n - k] - autocorr) * dx
    
    # c for shift -k (k=1,...,n-1): (prefix[n] - prefix[k] - autocorr[k]) * dx
    k_neg = np.arange(1, n)
    c_neg = (prefix[n] - prefix[k_neg] - autocorr[k_neg]) * dx
    
    # Full array of size 2n-1
    c_full = np.empty(2 * n - 1)
    c_full[n-1:] = c_pos  # indices n-1 to 2n-2
    c_full[:n-1] = c_neg[::-1]  # indices 0 to n-2 (reversed)
    
    return c_full

def smooth_max_obj(x, n, alpha):
    c = compute_c5_fft(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)
