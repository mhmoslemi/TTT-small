def project_onto_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    def project(tau):
        return np.clip(h - tau, 0, 1)
    
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = project(mid).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return project((lo + hi) / 2)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:

            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

    """Project h onto {0<=x<=1, sum(x)=target} using bisection on Lagrange multiplier."""
    lo, hi = -2.0, 2.0

        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
