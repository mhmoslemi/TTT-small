    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m_lag = k - (n - 1)
        if m_lag == 0:
            grad += w[k] * dx * (1.0 - 2.0 * h)
        elif m_lag > 0:
            grad[0:n-m_lag] += w[k] * dx * (1.0 - h[m_lag:n])
            grad[m_lag:n] += w[k] * dx * (-h[0:n-m_lag])
        else:
            p = -m_lag
            grad[p:n] += w[k] * dx * (1.0 - h[0:n-p])
            grad[0:n-p] += w[k] * dx * (-h[p:n])
    return grad

    def obj_and_grad(x):

        penalty = 1e10 * diff * diff
        obj_val += penalty
        grad = compute_smooth_grad(x, n, alpha)
        grad += 2.0 * 1e10 * diff * np.ones(n)
        return obj_val, grad

                                'maxls': 50, 'iprint': -1})

    except Exception:
        return project_feasible(h.copy(), n)

Hmm, this random swap approach might be too slow (each swap requires a full correlation computation which is O(n²)). For n=500, that's 250K operations per swap, and we're doing ~25 swaps per round, 50000 rounds = 2.5 billion operations. That might be too slow.

Let me optimize: instead of computing the full correlation after each swap, I can compute only the change in the top K lags.

For a swap (i: 1→0, j: 0→1), the change in C[k] (for lag m = k-(n-1)):

If m > 0:
C[k] = ∑_{i'=0}^{n-1-m} h[i'](1-h[i'+m])

The terms that change are those involving h[i] or h[j]:
- h[i] as first index (i' = i): changes if i ≤ n-1-m
  Old: 1*(1-h[i+m]), New: 0*(1-h_new[i+m])
  If i+m = j: h_new[i+m] = h_new[j] = 1. New term: 0. Old term: 1*(1-0) = 1. Δ = -1.
  If i+m ≠ j: h_new[i+m] = h[i+m]. New term: 0. Old term: 1*(1-h[i+m]). Δ = -(1-h[i+m]).

- h[i] as second index (i'+m = i, i.e., i' = i-m): changes if i-m ≥ 0 and i-m ≤ n-1-m, i.e., m ≤ i ≤ n-1
  Old: h[i-m]*(1-1) = 0. New: h_new[i-m]*(1-0) = h_new[i-m].
  If i-m = j: h_new[i-m] = h_new[j] = 1. New term: 1. Old term: 0. Δ = +1.
  If i-m ≠ j: h_new[i-m] = h[i-m]. New term: h[i-m]. Old term: 0. Δ = +h[i-m].

- h[j] as first index (i' = j): changes if j ≤ n-1-m
  Old: 0*(1-h[j+m]) = 0. New: 1*(1-h_new[j+m]).
  If j+m = i: h_new[j+m] = h_new[i] = 0. New term: 1*(1-0) = 1. Δ = +1.
  If j+m ≠ i: h_new[j+m] = h[j+m]. New term: 1*(1-h[j+m]). Δ = +(1-h[j+m]).

- h[j] as second index (i'+m = j, i.e., i' = j-m): changes if j-m ≥ 0 and j-m ≤ n-1-m, i.e., m ≤ j ≤ n-1
  Old: h[j-m]*(1-0) = h[j-m]. New: h_new[j-m]*(1-1) = 0.
  If j-m = i: h_new[j-m] = h_new[i] = 0. New term: 0. Old term: 0*(1-0) = 0. Wait, h[j-m] = h[i] = 1 before swap. Old: 1*(1-0) = 1. New: 0. Δ = -1.
  If j-m ≠ i: h_new[j-m] = h[j-m]. Old: h[j-m]. New: 0. Δ = -h[j-m].

So total ΔC[k] for m > 0:
= [Δ from h[i] as first] + [Δ from h[i] as second] + [Δ from h[j] as first] + [Δ from h[j] as second]

This can be computed in O(1) per k! So for K lags, it's O(K) per swap.

def swap_delta_c(h, i, j, k_top, n):
    """Compute change in C[k] for k in k_top when swapping h[i](1->0) and h[j](0->1)."""
    deltas = np.zeros(len(k_top))
    for idx, k in enumerate(k_top):
        m = k - (n - 1)
        if m == 0:
            # For binary, C[k] = 0 at lag 0, so delta is 0
            deltas[idx] = 0
            continue
        if m > 0:
            delta = 0.0
            # h[i] as first: valid if i <= n-1-m
            if i <= n - 1 - m:
                if i + m == j:
                    delta += -1.0
                else:
                    delta += -(1.0 - h[i+m])
            # h[i] as second: valid if m <= i <= n-1
            if m <= i <= n - 1:
                if i - m == j:
                    delta += 1.0
                else:
                    delta += h[i-m]
            # h[j] as first: valid if j <= n-1-m
            if j <= n - 1 - m:
                if j + m == i:
                    delta += 1.0
                else:
                    delta += (1.0 - h[j+m])
            # h[j] as second: valid if m <= j <= n-1
            if m <= j <= n - 1:
                if j - m == i:
                    delta += -1.0
                else:
                    delta += -h[j-m]
            deltas[idx] = delta
        else:
            p = -m  # p > 0
            # C[k] = sum_{i'=p}^{n-1} h[i'](1-h[i'-p])
            # h[i] as first: valid if i >= p
            if i >= p:
                if i - p == j:
                    delta += -1.0
                else:
                    delta += -(1.0 - h[i-p])
            # h[i] as second: valid if i-p >= p, i.e., i >= 2p... 
            # Actually: second index is i'-p, so i'-p = i means i' = i+p
            # Valid if i+p <= n-1, i.e., i <= n-1-p
            if i <= n - 1 - p:
                if i + p == j:
                    delta += 1.0
                else:
                    delta += h[i+p]
            # h[j] as first: valid if j >= p
            if j >= p:
                if j - p == i:
                    delta += 1.0

                    delta += (1.0 - h[j-p])
            # h[j] as second: valid if j <= n-1-p
            if j <= n - 1 - p:
                if j + p == i:
                    delta += -1.0
                else:
                    delta += -h[j+p]
            deltas[idx] = delta
    return deltas

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    best_h = h.copy()

                h[i] += direction * d
                h = project_feasible(h, n)
            continue

        if m_abs >= n:
            continue

        n_moves = 15
        found = False
        for _ in range(n_moves):

                r = rng.random()
                if r < 0.5:

                    if h[i] > 0.05:

                        h[j] += d

                        h[i] -= d
                        h[j] += d

def expand_resolution(h_src, n_dst):
    x_src = np.linspace(0, 2, len(h_src))
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: L-BFGS-B with increasing alpha
    alpha_schedule = [500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 300:
            break
        remaining = deadline - 300 - time.time()
        if remaining < 10:
            break
        t_sub = min(remaining, 45)
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + t_sub, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Binary projection + combinatorial search
    if time.time() < deadline - 250:
        h_bin = to_binary(best_h, n)
        sa_deadline = min(deadline - 180, time.time() + 200)
        h_comb, c_comb = combinatorial_search(h_bin, n, sa_deadline, rng, K=10, max_rounds=80000)
        if c_comb < best_c:
            best_c = c_comb
            best_h = h_comb.copy()
        else:
            # Relax the binary solution back to continuous
            h_relaxed = project_feasible(h_comb, n)
            # Add small noise to break symmetry
            h_relaxed = project_feasible(h_relaxed + rng.normal(0, 0.01, n), n)
            h_opt = lbfgsb_optimize(h_relaxed, n, 100000, min(deadline - 100, time.time() + 30), maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: L-BFGS-B polish on best continuous solution
    if time.time() < deadline - 200:
        best_h = project_feasible(best_h, n)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 80:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 25, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Targeted continuous local search

        h_opt, c_opt = targeted_local_search_cont(best_h, n, min(deadline - 40, time.time() + 80), rng, K=8, n_iters=40000)

            best_c = c_opt

            h_comb2, c_comb2 = combinatorial_search(h_bin2, n_try, min(deadline - 30, time.time() + 40), rng, K=8, max_rounds=30000)

        best_c = compute_c5(best_h, n_init)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
