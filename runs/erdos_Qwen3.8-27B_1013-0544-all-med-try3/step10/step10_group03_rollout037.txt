        if s >= 0:
            nv = n - s
            grad[0:nv] += wm_dx * (1.0 - x[s:n])
            grad[s:n] += wm_dx * (-x[0:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:n] += wm_dx * (1.0 - x[0:nv])
            grad[0:nv] += wm_dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        wm_dx = wi * dx
        if s >= 0:
            nv = n - s
            grad[0:nv] += wm_dx * (1.0 - x[s:n])
            grad[s:n] += wm_dx * (-x[0:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:n] += wm_dx * (1.0 - x[0:nv])
            grad[0:nv] += wm_dx * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def identify_active(h, n, rel_tol=0.0005):
    c = compute_c5_vec(h, n)
    m = c.max()
    threshold = m - max(1e-10, rel_tol * m)
    active = np.where(c >= threshold)[0]
    return active

def run_lbfgs(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                       args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def run_active_lbfgs(h, n, alpha, mu, maxiter, deadline, reidentify_every=500):
    bounds = [(0.0, 1.0)] * n
    active = identify_active(h, n)
    try:
        res = minimize(active_obj, h, jac=active_grad,
                       args=(n, active, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 10, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def line_search_1d(h, n, i, best_c, deadline):
    """Try to improve by moving h[i] in the best direction."""
    orig = h[i]
    best_val = orig
    best_obj = best_c
    # Try both directions with decreasing steps
    for direction in [1.0, -1.0]:
        step = 0.05
        for _ in range(20):
            if time.time() > deadline:
                break
            new_val = orig + direction * step
            if new_val < 0 or new_val > 1:
                step *= 0.5
                continue
            h[i] = new_val
            c = compute_c5(h, n)
            if c < best_obj - 1e-15:
                best_obj = c
                best_val = new_val
            else:
                break
            step *= 0.5
    h[i] = best_val
    return best_obj

def local_search_1d(h, n, best_c, deadline, max_passes=10):
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_c = best_c
            h[i] = np.clip(h[i], 0.0, 1.0)
            # Try small perturbations
            orig = h[i]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                if time.time() > deadline:
                    break
                new_val = orig + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    orig = new_val
                    improved = True
                else:
                    h[i] = orig
                    break
        if not improved:
            break
    return h, best_c

def pair_local_search(h, n, best_c, deadline):
    """Try pair moves: increase h[i], decrease h[j] (or vice versa)."""
    for delta in [0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(0, n, 3):
                if time.time() > deadline:
                    break
                for j in range(i + 1, min(i + 80, n), 2):
                    if time.time() > deadline:
                        break
                    # Try h[i] += delta, h[j] -= delta
                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        h[i] += delta
                        h[j] -= delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    # Try h[i] -= delta, h[j] += delta
                    if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
    return h, best_c

def round_and_reoptimize(h, n, best_c, deadline):
    """Push values close to 0 or 1, then reoptimize."""
    h_r = h.copy()
    h_r[h_r < 0.02] = 0.0
    h_r[h_r > 0.98] = 1.0
    h_r = project_feasible(h_r, n)
    c_r = compute_c5(h_r, n)
    if c_r <= best_c + 0.001:
        h_r = run_lbfgs(h_r, n, 10000, 10.0, 2000, deadline)
        c_r = compute_c5(h_r, n)
        if c_r < best_c:
            return h_r, c_r
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    def update_best(h_new, c_new):
        nonlocal best_h, best_c
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 1: Graduated L-BFGS with smooth max
    alpha_schedule = [100, 500, 2000, 10000, 50000]
    mu_schedule = [100.0, 100.0, 10.0, 10.0, 1.0]
    for idx, (alpha, mu) in enumerate(zip(alpha_schedule, mu_schedule)):
        if time.time() > deadline - 300:
            break
        t_start = time.time()
        remaining = deadline - 300 - t_start
        maxiter = max(200, int(remaining * 50))
        h = run_lbfgs(h, n, alpha, mu, maxiter, deadline - 300)
        c = compute_c5(h, n)
        update_best(h, c)

    # Phase 2: Active-set focused optimization
    if time.time() < deadline - 200:
        active = identify_active(h, n, rel_tol=0.001)
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 200:
                break
            h_tmp = h.copy()
            h_tmp = run_active_lbfgs(h_tmp, n, alpha, 5.0, 3000, deadline - 200)
            c_tmp = compute_c5(h_tmp, n)
            if c_tmp < best_c:
                update_best(h_tmp, c_tmp)
                h = h_tmp
            # Re-identify active set
            active = identify_active(h, n, rel_tol=0.001)

    # Phase 3: Full smooth max with very high alpha
    if time.time() < deadline - 150:
        h = run_lbfgs(h, n, 200000, 1.0, 5000, deadline - 150)
        c = compute_c5(h, n)
        update_best(h, c)

    # Phase 4: 1D local search
    if time.time() < deadline - 100:
        h, c = local_search_1d(h, n, best_c, deadline - 100, max_passes=15)
        update_best(h, c)

    # Phase 5: Pair local search
    if time.time() < deadline - 80:
        h, c = pair_local_search(h, n, best_c, deadline - 80)
        update_best(h, c)

    # Phase 6: Rounding + reoptimize
    if time.time() < deadline - 60:
        h_r, c_r = round_and_reoptimize(h, n, best_c, deadline - 60)
        update_best(h_r, c_r)
        if c_r < best_c:
            h = h_r

    # Phase 7: Final active-set round with re-identification
    if time.time() < deadline - 40:
        active = identify_active(h, n, rel_tol=0.0003)
        for alpha in [500000, 1000000]:
            if time.time() > deadline - 40:
                break
            h_tmp = h.copy()
            h_tmp = run_active_lbfgs(h_tmp, n, alpha, 0.5, 3000, deadline - 40)
            c_tmp = compute_c5(h_tmp, n)
            if c_tmp < best_c:
                update_best(h_tmp, c_tmp)
                h = h_tmp

    # Phase 8: Final 1D search
    if time.time() < deadline - 10:
        h, c = local_search_1d(h, n, best_c, deadline - 10, max_passes=20)
        update_best(h, c)

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
