def compute_c5_vec(h, n):

def compute_c5(h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    vals = c[active]

    smx = sh.max()

            if tot < 1e-15:

        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_active(h, n, deadline, top_k=8, alphas=None, max_rounds=12):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000]
    for round_idx in range(max_rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            idx_sorted = np.argsort(-c_vec[active])
            active = active[idx_sorted[:top_k]]
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 5,
                                       'maxls': 50, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    tol_expand = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def fine_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    radius = 25
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = best_h.copy()
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: SLP at n_orig (alternating primal/dual)
    p1_end = min(t0 + 200, deadline - 500)
    h_slp, c_slp = slp_optimize(best_h, best_n, p1_end, max_iters=400, mode='alternating')
    if c_slp < best_c:
        best_c = c_slp
        best_h = h_slp

    # Phase 2: L-BFGS polish
    if time.time() < deadline - 350:
        p2_end = min(time.time() + 200, deadline - 350)
        h_lb, c_lb = run_lbfgs(best_h, best_n, p2_end,
            alphas=[50, 200, 1000, 5000, 50000, 200000, 1000000],
            mu=0.01, maxiter_per=[1500, 2000, 2000, 2500, 3000, 3000, 3000])
        if c_lb < best_c:
            best_c = c_lb
            best_h = h_lb

    # Phase 3: Active-set focused
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 120, deadline - 250)
        h_act, c_act = run_active(best_h, best_n, p3_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000], max_rounds=10)
        if c_act < best_c:
            best_c = c_act
            best_h = h_act

    # Phase 4: Second SLP round (from improved point)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 150, deadline - 250)
        h_slp2, c_slp2 = slp_optimize(best_h, best_n, p4_end, max_iters=300, mode='primal')
        if c_slp2 < best_c:
            best_c = c_slp2
            best_h = h_slp2
        p4b_end = min(time.time() + 80, deadline - 200)
        h_slp3, c_slp3 = slp_optimize(best_h, best_n, p4b_end, max_iters=200, mode='dual')
        if c_slp3 < best_c:
            best_c = c_slp3
            best_h = h_slp3

    # Phase 5: Subgradient refinement
    if time.time() < deadline - 180:
        p5_end = min(time.time() + 60, deadline - 160)
        h_sg, c_sg = subgrad_refine(best_h, best_n, p5_end, n_iter=5000)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 6: Try n=1000
    n_hi = 1000
    if time.time() < deadline - 200:
        x_fine = np.linspace(0, 2, n_hi)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_hi)
        c_fine = compute_c5(h_fine, n_hi)

        p6a_end = min(time.time() + 120, deadline - 120)
        h_f6, c_f6 = slp_optimize(h_fine, n_hi, p6a_end, max_iters=100, mode='alternating')
        if c_f6 < best_c:
            best_c = c_f6
            best_h = h_f6
            best_n = n_hi

        if time.time() < deadline - 80:
            p6b_end = min(time.time() + 80, deadline - 60)
            h_f7, c_f7 = run_lbfgs(best_h, best_n, p6b_end,
                alphas=[10, 50, 200, 1000, 5000, 20000],
                mu=0.005, maxiter_per=[1000, 1500, 1500, 2000, 2000, 2000])
            if c_f7 < best_c:
                best_c = c_f7
                best_h = h_f7
                best_n = n_hi

    # Phase 7: Final SLP at best resolution
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 40)
        h_s7, c_s7 = slp_optimize(best_h, best_n, p7_end, max_iters=200, mode='alternating')
        if c_s7 < best_c:
            best_c = c_s7
            best_h = h_s7

    # Phase 8: Final active + subgrad polish
    if time.time() < deadline - 25:
        p8_end = min(time.time() + 15, deadline - 10)
        h_a8, c_a8 = run_active(best_h, best_n, p8_end, top_k=4,
            alphas=[50000, 200000, 1000000], max_rounds=4)
        if c_a8 < best_c:
            best_c = c_a8
            best_h = h_a8

    # Phase 9: Fine local search (short)
    if time.time() < deadline - 5:
        h_l9, c_l9 = fine_local_search(best_h, best_n, deadline - 2)
        if c_l9 < best_c:
            best_c = c_l9
            best_h = h_l9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
