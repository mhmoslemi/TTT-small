import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize_fixed(h, n):
    h = np.clip(h, 0, 1)
    h[(h > 0.5) & (h < 1.0)] = 1.0
    h[(h < 0.5) & (h > 0.0)] = 0.0
    n_ones = n // 2
    cur_ones = int(np.sum(h > 0.5))
    if cur_ones > n_ones:
        ones = np.where(h > 0.5)[0]
        remove = rng_choice(ones, cur_ones - n_ones)
        h[remove] = 0.0
    elif cur_ones < n_ones:
        zeros = np.where(h < 0.5)[0]
        add = rng_choice(zeros, n_ones - cur_ones)
        h[add] = 1.0
    return project_feasible(h, n)

def rng_choice(arr, k):
    n_avail = len(arr)
    k = min(k, n_avail)
    idx = np.random.choice(n_avail, size=k, replace=False)
    return arr[idx]

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    if h.sum() < n / 2.0:
        zeros = np.where(h == 0)[0]
        need = int(n / 2.0 - h.sum())
        if len(zeros) >= need:
            h[zeros[:need]] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, seed_val=0):
    local_rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    n_ones = n // 2
    idx = local_rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def active_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def full_smooth_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def full_smooth_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbfgs_warmup(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [10, 50, 200, 1000, 5000, 20000, 100000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        rem = deadline - time.time()
        if rem < 15:
            break
        maxit = min(3000, max(300, int(rem * 5)))
        try:
            res = minimize(full_smooth_obj, best_h, jac=full_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = n
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_opt(h, n, deadline, top_k=6, alphas=None, rounds=20):
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000, 50000000]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(rounds):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(200, int((deadline - time.time()) * 2))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def simulated_annealing(h, n, deadline, n_starts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for start in range(n_starts):
        if time.time() > deadline:
            break
        # Binarize starting point
        if start == 0:
            curr = binarize_fixed(h, n)
        elif start == 1:
            curr = make_beatty_start(n, (1 + np.sqrt(5)) / 4.0)
        else:
            curr = make_random_binary(n, seed_val=100 + start)
        curr_c = compute_c5(curr, n)
        if curr_c >= best_c + 0.01:
            continue
        T0 = 0.003
        T1 = 0.00005
        n_ones = n // 2
        ones = np.where(curr > 0.5)[0].tolist()
        zeros = np.where(curr < 0.5)[0].tolist()
        iter_count = 0
        while time.time() < deadline - (n_starts - 1 - start) * 15:
            T = T0 * (T1 / T0) ** (iter_count / 8000.0)
            if T < T1:
                T = T1
            if not ones or not zeros:
                break
            oi = ones[np.random.randint(len(ones))]
            zj = zeros[np.random.randint(len(zeros))]
            curr[oi] = 0.0
            curr[zj] = 1.0
            new_c = compute_c5(curr, n)
            delta = new_c - curr_c
            if new_c < best_c:
                best_c = new_c
                best_h = curr.copy()
                # Update sets
                ones.remove(oi)
                zeros.remove(zj)
                ones.append(zj)
                zeros.append(oi)
                curr_c = new_c
            elif delta <= 0 or np.random.random() < np.exp(-delta / T):
                ones.remove(oi)
                zeros.remove(zj)
                ones.append(zj)
                zeros.append(oi)
                curr_c = new_c
            else:
                curr[oi] = 1.0
                curr[zj] = 0.0
            iter_count += 1
            if iter_count % 500 == 0 and T < T1 * 1.1:
                break
    return best_h, best_c

def grad_coord_descent(h, n, deadline, step_sizes=None, max_steps=50000):
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    for step in step_sizes:
        steps_done = 0
        while time.time() < deadline and steps_done < max_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            pos_mask = (grad > 5e-9) & (best_h > 1e-8)
            neg_mask = (grad < -5e-9) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i] * 0.9, (1.0 - best_h[best_j]) * 0.9)
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                if steps_done % 200 == 0:
                    break
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=40, deltas=None):
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def upsample(h, n_old, n_new):
    h_new = np.zeros(n_new)
    for i in range(n_new):
        idx_old = min(int(i * n_old / n_new), n_old - 1)
        h_new[i] = h_old[idx_old]
    scale = (n_old / 2.0) / max(1e-15, h_new.sum())
    h_new = np.clip(h_new * scale, 0, 1)
    return project_feasible(h_new, n_new)
