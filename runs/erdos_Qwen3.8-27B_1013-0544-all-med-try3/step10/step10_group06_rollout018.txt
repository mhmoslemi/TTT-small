When I change h[i] from old to new, the change in c_k is:

Wait, more carefully:
c_k = sum_p h[p] * (1 - h[p + k - (n-1)]) * dx

The index p+k-(n-1) must be in [0, n-1] and p must be in [0, n-1].

When h[i] changes:
- It affects c_k as the first factor: p = i, if i+k-(n-1) is valid (in [0,n-1])
  Contribution: δ * (1-h[i+k-(n-1)]) * dx
- It affects c_k as the second factor: p+k-(n-1) = i, i.e., p = i-k+(n-1), if p is valid
  Contribution: -δ * h[i-k+(n-1)] * dx

So: Δc_k = dx * [δ*(1-h[i+k-(n-1)]) - δ*h[i-k+(n-1)]] (for valid indices, 0 for invalid)

Let lag = k - (n-1). Then:
Δc_k = dx * δ * [(1-h[i+lag]) - h[i-lag]] (for valid indices)

Where "valid" means i+lag ∈ [0,n-1] and i-lag ∈ [0,n-1] respectively.

So for a given i and δ, I can compute all Δc_k in O(n) time!

Then new_c_k = old_c_k + Δc_k, and new_max = max_k new_c_k.

This is O(n) per candidate, O(n * num_candidates) per coordinate, O(n² * num_candidates) per sweep.
For n=500, candidates=10: 500 * 500 * 10 = 2.5M per sweep. Very fast!

```python
def coord_step(h, n, i, delta, c_old):
    """Compute new c array when h[i] changes by delta."""
    dx = 2.0 / n
    c_new = c_old.copy()
    # For each lag k (0 to 2n-2), lag_val = k-(n-1)
    # Δc_k = dx * delta * [(1-h[i+lag_val]) - h[i-lag_val]] (valid terms only)
    for lag_val in range(-(n-1), n):
        k = lag_val + (n-1)
        term1 = 0.0
        if 0 <= i + lag_val < n:
            term1 = 1.0 - h[i + lag_val]
        term2 = 0.0
        if 0 <= i - lag_val < n:
            term2 = h[i - lag_val]
        c_new[k] += dx * delta * (term1 - term2)
    return c_new
```

```python
def coord_step_vec(h, n, i, delta, c_old):
    dx = 2.0 / n
    # lags = -(n-1) to (n-1)
    # i + lag and i - lag
    lags = np.arange(-(n-1), n)  # length 2n-1
    i_plus = i + lags
    i_minus = i - lags
    term1 = np.where((i_plus >= 0) & (i_plus < n), 1.0 - h[i_plus[(i_plus >= 0) & (i_plus < n)]], 0.0)
    term2 = np.where((i_minus >= 0) & (i_minus < n), h[i_minus[(i_minus >= 0) & (i_minus < n)]], 0.0)
    return c_old + dx * delta * (term1 - term2)
```

This is O(n) with numpy vectorization, very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def compute_grad_full(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        else:
            m = abs(lag)
            grad[m:n] += w[k] * (1.0 - h[0:n-m])
            grad[0:n-m] -= w[k] * h[m:n]
    return grad * dx

def lbfgs_optimize(x0, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    penalty = 1e6
    
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1-x, mode='full') * (2.0/n)
        obj = smooth_max_val(c, alpha)
        diff = x.sum() - target_sum
        obj += penalty * diff * diff
        g = compute_grad_full(x, n, alpha)
        g = g + 2 * penalty * diff
        return obj, g
    
    try:
        res = minimize(obj_grad, x0, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(x0.copy(), n)

def coord_step_vec(h, n, i, delta, c_old):
    """Fast computation of new c when h[i] changes by delta."""
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    i_plus = i + lags
    i_minus = i - lags
    mask_plus = (i_plus >= 0) & (i_plus < n)
    mask_minus = (i_minus >= 0) & (i_minus < n)
    term1 = np.zeros(2*n-1)
    term2 = np.zeros(2*n-1)
    term1[mask_plus] = 1.0 - h[i_plus[mask_plus]]
    term2[mask_minus] = h[i_minus[mask_minus]]
    return c_old + dx * delta * (term1 - term2)

def coordinate_descent(h, n, deadline, max_sweeps=500, rng=None, step_sizes=None):
    """Fast coordinate descent with exact c_k updates."""
    h = h.copy()
    best_h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    best_c = float(c.max())

        if time.time() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        
        for idx in order:
            i = int(idx)
            old_val = h[i]
            current_max = c.max()
            best_new_val = old_val
            best_new_c = None
            best_new_max = current_max
            
            # Try candidates

            candidates.add(0.0)
            candidates.add(1.0)
            for ss in step_sizes:

                if old_val + ss <= 1:
                    candidates.add(old_val + ss)

                h[i] = best_new_val
                c = best_new_c

        h = project_feasible(h, n)
        c = np.correlate(h, 1-h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h.copy()
