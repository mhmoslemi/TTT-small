                ones.add(j2)
                zeros.discard(j1)
                zeros.discard(j2)
                zeros.add(i1)
                zeros.add(i2)
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i1], h[i2] = 1.0, 1.0
                h[j1], h[j2] = 0.0, 0.0
    return best_h, best_c

def gradient_double_swap_search(h, n, deadline, rng, max_iters=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    alpha = 80000.0
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad_from_w(w, h, n)

        improved = False
        # First try single swaps
        pos_idx = np.where((grad > 1e-10) & (h > 0.5))[0]
        neg_idx = np.where((grad < -1e-10) & (h < 0.5))[0]
        if len(pos_idx) > 0 and len(neg_idx) > 0:
            pos_idx = pos_idx[np.argsort(-grad[pos_idx])][:20]
            neg_idx = neg_idx[np.argsort(grad[neg_idx])][:20]
            for i in pos_idx:
                if improved:
                    break
                for j in neg_idx:
                    h[i], h[j] = 0.0, 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        ones.remove(int(i))
                        ones.append(int(j))
                        zeros.remove(int(j))
                        zeros.append(int(i))
                        improved = True
                        stall = 0
                        break
                    else:
                        h[i], h[j] = 1.0, 0.0

        if not improved:
            # Try double swaps: pick top 5 remove, top 5 add
            if len(pos_idx) >= 2 and len(neg_idx) >= 2:
                p5 = pos_idx[:5]
                n5 = neg_idx[:5]
                found = False
                for a in range(len(p5)):
                    if found:
                        break
                    for b in range(a + 1, len(p5)):
                        if found:
                            break
                        for c1 in range(len(n5)):
                            if found:
                                break
                            for c2 in range(c1 + 1, len(n5)):
                                h[p5[a]] = 0.0
                                h[p5[b]] = 0.0
                                h[n5[c1]] = 1.0
                                h[n5[c2]] = 1.0
                                c = compute_c5(h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    for x in [p5[a], p5[b]]:
                                        ones.remove(int(x))
                                    for x in [n5[c1], n5[c2]]:
                                        ones.append(int(x))
                                    for x in [n5[c1], n5[c2]]:
                                        zeros.remove(int(x))
                                    for x in [p5[a], p5[b]]:
                                        zeros.append(int(x))
                                    improved = True
                                    found = True
                                    stall = 0
                                    break
                                h[p5[a]] = 1.0
                                h[p5[b]] = 1.0
                                h[n5[c1]] = 0.0
                                h[n5[c2]] = 0.0

        if not improved:
            stall += 1
            if stall > 300:
                stall = 0
                # Random perturbation
                for _ in range(rng.randint(5, 15)):
                    if time.time() > deadline:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                    else:
                        h[i], h[j] = 1.0, 0.0

    return h, best_c

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 8,
                               'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def make_quasiperiodic(n, offset=0):
    h = np.zeros(n)
    target = n // 2
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = []
    for i in range(n):
        pos = int(((i + offset) / n * golden) % 1.0 * n)
        positions.append(pos)
    # Use Balong's method: h[i] = 1 if frac(i*golden) < 0.5
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i * golden) % 1.0) < 0.5 else 0.0
    # Fix count
    cnt = int(h.sum())
    if cnt > target:
        excess = cnt - target
        ones = np.where(h > 0.5)[0]
        # Remove least "central" ones
        for i in range(excess):
            idx = ones[rng.randint(len(ones))]
            h[idx] = 0.0
            ones = np.delete(ones, np.where(ones == idx)[0][0])
    elif cnt < target:
        deficit = target - cnt
        zeros = np.where(h < 0.5)[0]
        for i in range(deficit):
            idx = zeros[rng.randint(len(zeros))]
            h[idx] = 1.0
            zeros = np.delete(zeros, np.where(zeros == idx)[0][0])
    return h

def upsample(h, n_from, n_to):
    """Upsample h from n_from to n_to points using linear interpolation"""
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return np.clip(h_new, 0, 1)

def fine_coord_descent(h, n, deadline, max_rounds=500):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for step in steps:
        if time.time() > deadline:
            break
        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            w = smooth_max_w(c_vec, 150000.0)
            grad = h_grad_from_w(w, best_h, n)

            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break

            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]

            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j

            if best_i < 0:
                break

            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                break
    return best_h, best_c

def nemesis_subset(h, n, positions, deadline, maxit=3000):
    h_full = h.copy()
    x0 = h_full[positions].copy()

    def obj_small(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        h_tmp = h_full.copy()
        h_tmp[positions] = x_clipped
        h_tmp = project_feasible(h_tmp, n)
        return compute_c5(h_tmp, n)

    try:
        res = minimize(obj_small, x0, method='Nelder-Mead',
                      options={'maxiter': maxit, 'xatol': 1e-14, 'fatol': 1e-15})
        h_new = h_full.copy()
        h_new[positions] = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h_full, compute_c5(h_full, n)

def sensitive_positions(h, n, num=15):
    c_vec = compute_c5_vec(h, n)
    w = smooth_max_w(c_vec, 500000.0)
    grad = h_grad_from_w(w, h, n)
    idx = np.argsort(-np.abs(grad))[:num]
    return idx.tolist()

def run(seed=42, budget_s=1100, **kwargs):
    global rng
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n = len(h_parent)
    except Exception:
        h_parent = np.full(400, 0.5)
        n = 400

    h_parent = project_feasible(h_parent, n)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n)
    best_n = n

    # Phase 1: Quick L-BFGS (45s) - get good fractional solution
    p1_deadline = t0 + 45
    if time.time() < p1_deadline:
        h1, c1 = run_lbfgs(best_h, n, p1_deadline, alpha=50000, mu=0.01, maxit=1500)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Binarize + gradient-guided binary search (250s)
    p2_deadline = min(t0 + 300, deadline - 500)
    h_bin = binarize(best_h, n)
    h2, c2 = gradient_guided_binary_search(h_bin, n, p2_deadline, rng, max_iters=300000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Gradient double-swap search (200s)
    p3_deadline = min(t0 + 500, deadline - 350)
    if time.time() < p3_deadline:
        h3, c3 = gradient_double_swap_search(best_h, n, p3_deadline, rng, max_iters=50000)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Simulated annealing (250s)
    p4_deadline = min(t0 + 750, deadline - 200)
    if time.time() < p4_deadline:
        h4, c4 = simulated_annealing_binary(best_h, n, p4_deadline, rng,
                                           T0=0.0005, T_min=0.000003, decay=0.9997)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
        # Second SA round with different params
        if time.time() < p4_deadline - 30:
            h4b, c4b = simulated_annealing_binary(best_h, n, p4_deadline, rng,
                                                T0=0.0003, T_min=0.000002, decay=0.9998)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b.copy()

    # Phase 5: Multi-start at different resolutions (150s)
    p5_deadline = min(t0 + 900, deadline - 100)
    if time.time() < p5_deadline:
        # Try n+10, n+20 resolutions
        for n_try in [n + 10, n + 20]:
            if time.time() > p5_deadline - 30:
                break
            if n_try % 2 != 0:
                continue
            # Upsample
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = np.clip(h_interp, 0, 1)
            h_interp = project_feasible(h_interp, n_try)
            h_b = binarize(h_interp, n_try)
            sub_dl = min(p5_deadline, time.time() + 50)
            h_try, c_try = gradient_guided_binary_search(h_b, n_try, sub_dl, rng, max_iters=80000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                best_n = n_try

        # Try quasiperiodic initialization at same n
        if time.time() < p5_deadline - 20:
            qp = np.zeros(n)
            golden = (1.0 + np.sqrt(5.0)) / 2.0
            for i in range(n):
                qp[i] = 1.0 if ((i * golden) % 1.0) < 0.5 else 0.0
            cnt = int(qp.sum())
            if cnt != n // 2:
                if cnt > n // 2:
                    ones_q = np.where(qp > 0.5)[0]
                    for i in range(cnt - n // 2):
                        qp[ones_q[rng.randint(len(ones_q))]] = 0.0
                else:
                    zeros_q = np.where(qp < 0.5)[0]
                    for i in range(n // 2 - cnt):
                        qp[zeros_q[rng.randint(len(zeros_q))]] = 1.0
            sub_dl = min(p5_deadline, time.time() + 30)
            h_qp, c_qp = gradient_guided_binary_search(qp, n, sub_dl, rng, max_iters=40000)
            if c_qp < best_c:
                best_c = c_qp
                best_h = h_qp.copy()

    # Phase 6: Continuous refinement on best binary (80s)
    p6_deadline = min(t0 + 980, deadline - 40)
    if time.time() < p6_deadline:
        h_fr = best_h.copy()
        # Un-binarize slightly
        h_fr = h_fr * 0.95 + 0.05
        h_fr = project_feasible(h_fr, best_n)
        for alpha in [100000, 500000, 2000000]:
            if time.time() > p6_deadline - 5:
                break
            h6, c6 = run_lbfgs(h_fr, best_n, min(p6_deadline, time.time() + 25),
                              alpha=alpha, mu=0.001, maxit=1500)
            if c6 < best_c:
                best_c = c6
                best_h = h6
                h_fr = h6

    # Phase 7: Nelder-Mead on sensitive positions (30s)
    p7_deadline = min(t0 + 1010, deadline - 20)
    if time.time() < p7_deadline:
        for attempt in range(3):
            if time.time() > p7_deadline - 2:
                break
            sens = sensitive_positions(best_h, best_n, num=12)
            h7, c7 = nemesis_subset(best_h, best_n, sens, p7_deadline, maxit=2000)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Final fine coordinate descent (20s)
    if time.time() < deadline - 3:
        h8, c8 = fine_coord_descent(best_h, best_n, deadline - 1, max_rounds=200)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
