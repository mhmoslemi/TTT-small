    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    idx_center = n - 1
    grad = np.zeros(n)
    for ii in range(len(active)):
        if w[ii] < 1e-12:
            continue
        k = int(active[ii]) - idx_center
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    obj = smooth_max(c, alpha)
    if mu > 0:
        obj += mu * (x.sum() - n / 2.0) ** 2
    return obj

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    idx_center = n - 1
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 100 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run_lbfgs(h, n, deadline, alpha, mu, maxiter=2000, top_k=None):
    """Run L-BFGS-B with optional active set."""
    bounds = [(0.0, 1.0)] * n
    if top_k is not None:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([n - 1])
        res = minimize(active_obj, h, jac=active_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxiter * 10,
                               'maxls': 50, 'maxcor': 50})
    else:
        res = minimize(full_obj, h, jac=full_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxiter * 10,
                               'maxls': 50, 'maxcor': 50})
    h_new = project_feasible(res.x, n)
    c_new = compute_c5(h_new, n)
    return h_new, c_new

def subgradient_refine(h, n, deadline, n_iters=5000, step_scale=0.01):
    """Subgradient method for the min-max problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_cur, n)
        c_max_idx = int(np.argmax(c_vec))
        c_max_val = c_vec[c_max_idx]
        k = c_max_idx - (n - 1)
        # Subgradient for this specific lag
        if k >= 0:
            nv = n - k
            if nv > 0:
                sg = np.zeros(n)
                sg[0:nv] = (1.0 - h_cur[k:n]) * dx
                sg[k:n] += (-h_cur[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                sg = np.zeros(n)
                sg[kk:n] = (1.0 - h_cur[0:nv]) * dx
                sg[0:nv] += (-h_cur[kk:n]) * dx
        # Step size decreases
        step = step_scale / (1.0 + 0.01 * it)
        h_cur = h_cur - step * sg
        h_cur = project_feasible(h_cur, n)
        c_cur = compute_c5(h_cur, n)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
    return best_h, best_c

def pair_improve(h, n, deadline, radius, delta, max_rounds=100):
    """Targeted pair search around active constraints."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        improved = False
        for idx in active[:15]:
            if time.time() > deadline:
                break
            k = int(idx) - (n - 1)
            if k >= 0:
                nv = n - k
                if nv <= 0:
                    continue
                for i in range(0, nv, 1):
                    if time.time() > deadline:
                        break
                    p1 = i
                    p2 = i + k
                    if best_h[p1] >= delta and best_h[p2] <= 1.0 - delta:
                        best_h[p1] -= delta
                        best_h[p2] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-18:
                            best_c = c
                            improved = True
                        else:
                            best_h[p1] += delta
                            best_h[p2] -= delta
            else:
                kk = -k
                nv = n - kk
                if nv <= 0:
                    continue
                for i in range(0, nv, 1):
                    if time.time() > deadline:
                        break
                    p1 = i
                    p2 = i + kk
                    if best_h[p2] >= delta and best_h[p1] <= 1.0 - delta:
                        best_h[p2] -= delta
                        best_h[p1] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-18:
                            best_c = c
                            improved = True
                        else:
                            best_h[p2] += delta
                            best_h[p1] -= delta
        if not improved:
            delta *= 0.7
            if delta < 1e-7:
                break
    return best_h, best_c

def single_point_search(h, n, deadline, delta):
    """Try adjusting single points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Focus on points near 0.5 (transition points)
    mask = (best_h > 0.01) & (best_h < 0.99)
    candidates = np.where(mask)[0]
    for pdb in [-delta, -delta/2, delta/2, delta]:
        for i in candidates:
            if time.time() > deadline:
                break
            new_val = best_h[i] + pdb
            if 0.0 < new_val < 1.0:
                best_h[i] = new_val
                h_adj = project_feasible(best_h, n)
                c = compute_c5(h_adj, n)
                if c < best_c - 1e-18:
                    best_c = c
                    best_h = h_adj
                else:
                    best_h[i] = new_val - pdb
    return best_h, best_c

def perturb_reopt(h, n, deadline, rng, n_trials=8, amp=0.02):
    """Perturb and re-optimize."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        h_p = best_h + rng.uniform(-amp, amp, size=n)
        h_p = project_feasible(h_p, n)
        c_p = compute_c5(h_p, n)
        if c_p < best_c + 0.01:
            rem = deadline - time.time() - 10
            if rem < 10:
                break
            maxit = max(200, int(rem / (n_trials - t) * 2))
            maxit = min(maxit, 1500)
            try:
                h_new, c_new = run_lbfgs(h_p, n, deadline - 5, 500000, 0.005, maxit=maxit)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_c = compute_c5(h0, n0)
    best_h = h0.copy()

    # Generate alternative starts at n0
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n0, golden / 2.0),
        make_beatty(n0, np.sqrt(2.0) / 2.0),
        make_beatty(n0, 0.4312),
        make_beatty(n0, 0.3827),
        make_random_binary(n0, rng),
    ]
    for ha in alt_starts:
        ca = compute_c5(ha, n0)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 1: Quick L-BFGS warmup at original resolution (30s)
    p1_end = min(t0 + 30, deadline - 1000)
    if p1_end > t0 + 5:
        for alpha in [5000, 50000, 500000]:
            if time.time() > p1_end:
                break
            try:
                h1, c1 = run_lbfgs(best_h, n0, p1_end, alpha, 0.01, maxiter=500)
                if c1 < best_c:
                    best_c = c1
                    best_h = h1
            except Exception:
                pass

    # Phase 2: Active set rounds at original resolution (60s)
    p2_end = min(time.time() + 60, deadline - 900)
    if p2_end > time.time() + 10:
        for top_k, alpha, mu in [(6, 2000000, 0.001), (10, 5000000, 0.0005),
                                   (15, 10000000, 0.0003), (25, 20000000, 0.0002)]:
            if time.time() > p2_end - 5:
                break
            try:
                h2, c2 = run_lbfgs(best_h, n0, p2_end - 3, alpha, mu,
                                    maxiter=800, top_k=top_k)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
            except Exception:
                pass

    # Phase 3: Upsample to n=1000 and optimize (300s)
    n_up = 1000
    p3_end = min(time.time() + 300, deadline - 500)
    if p3_end > time.time() + 50:
        h_up = upsample_h(best_h, n0, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.002:
            # L-BFGS rounds with increasing sharpness
            for alpha, mu, top_k in [(10000, 0.01, None),
                                       (100000, 0.005, 30),
                                       (500000, 0.003, 40),
                                       (2000000, 0.002, 50),
                                       (5000000, 0.001, 60),
                                       (10000000, 0.001, 80),
                                       (20000000, 0.0005, 100)]:
                if time.time() > p3_end - 30:
                    break
                rem = p3_end - time.time() - 10
                if rem < 30:
                    break
                maxit = max(300, int(rem * 1.5))
                maxit = min(maxit, 2000)
                try:
                    h3, c3 = run_lbfgs(h_up, n_up, p3_end - 5, alpha, mu,
                                        maxiter=maxit, top_k=top_k)
                    if c3 < best_c:
                        best_c = c3
                        best_h = h3
                        best_n = n_up
                    elif c3 < c_up + 0.001:
                        c_up = c3
                        h_up = h3
                except Exception:
                    pass
            # Subgradient refinement at n=1000
            if time.time() < p3_end - 20:
                h_up, c_up = subgradient_refine(h_up, n_up, p3_end - 5,
                                                 n_iters=3000, step_scale=0.005)
                if c_up < best_c:
                    best_c = c_up
                    best_h = h_up.copy()

    # Phase 4: Fine pair search (200s)
    cur_n = len(best_h)
    p4_end = min(time.time() + 200, deadline - 250)
    if p4_end > time.time() + 20:
        for delta in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() > p4_end - 10:
                break
            h4, c4 = pair_improve(best_h, cur_n, p4_end - 5,
                                   radius=50, delta=delta, max_rounds=50)
            if c4 < best_c:
                best_c = c4
                best_h = h4

    # Phase 5: More L-BFGS at current resolution with very sharp max (120s)
    p5_end = min(time.time() + 120, deadline - 150)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        for alpha, mu, top_k in [(50000000, 0.0005, 100),
                                  (100000000, 0.0003, 120),
                                  (200000000, 0.0002, 150),
                                  (500000000, 0.0001, 200)]:
            if time.time() > p5_end - 10:
                break
            try:
                h5, c5 = run_lbfgs(best_h, cur_n, p5_end - 3, alpha, mu,
                                    maxiter=1000, top_k=top_k)
                if c5 < best_c:
                    best_c = c5
                    best_h = h5
            except Exception:
                pass

    # Phase 6: Subgradient refinement (80s)
    p6_end = min(time.time() + 80, deadline - 80)
    if p6_end > time.time() + 15:
        cur_n = len(best_h)
        h6, c6 = subgradient_refine(best_h, cur_n, p6_end,
                                     n_iters=5000, step_scale=0.003)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Ultra-fine pair search (80s)
    p7_end = min(time.time() + 80, deadline - 40)
    if p7_end > time.time() + 15:
        cur_n = len(best_h)
        for delta in [0.0001, 0.00005, 0.00002, 0.00001]:
            if time.time() > p7_end - 5:
                break
            h7, c7 = pair_improve(best_h, cur_n, p7_end - 3,
                                   radius=30, delta=delta, max_rounds=30)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Perturbation + re-opt (40s)
    p8_end = min(time.time() + 40, deadline - 10)
    if p8_end > time.time() + 15:
        cur_n = len(best_h)
        h8, c8 = perturb_reopt(best_h, cur_n, p8_end, rng,
                                n_trials=4, amp=0.005)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: one more subgradient pass if time remains
    if time.time() < deadline - 5:
        cur_n = len(best_h)
        h_final, c_final = subgradient_refine(best_h, cur_n, deadline - 2,
                                               n_iters=2000, step_scale=0.001)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
