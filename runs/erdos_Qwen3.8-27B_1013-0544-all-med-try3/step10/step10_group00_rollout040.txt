def compute_c5_vec(h, n):

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def binarize(h, n):
    target = n / 2.0

    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:

            order = np.argsort(-h[zeros])

            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def exhaustive_swap_search(h, n, deadline, max_passes=200):
    """Exhaustive single-swap local search"""
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i_idx in range(len(ones)):
            if time.time() > deadline:
                break
            i = ones[i_idx]
            for j_idx in range(len(zeros)):

                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:

                    improved = True
                    # Update lists

                    ones[i_idx] = j
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, best_c

    """Simulated annealing on binary solutions"""
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])

        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        
        delta = c_new - cur_c
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_c = c_new
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        T *= cooling
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def exhaustive_swap(h, n, deadline, max_passes=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].astype(int).copy()
    zeros = np.where(h < 0.5)[0].astype(int).copy()
    n1 = len(ones)
    n0 = len(zeros)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(n1):
            if time.time() > deadline:
                break
            for jj in range(n0):
                i = ones[ii]
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    ones[ii] = j
                    zeros[jj] = i
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, best_c

def sim_anneal(h_init, n, deadline, rng, T0=1.0, cool=0.9999):

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)
