import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h onto {h : 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()

    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def compute_grad(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for m in range(n_full):
        if w[m] < 1e-15:
            continue
        s = m - (n - 1)
        wm = w[m] * dx
        if s >= 0:
            nv = n - s
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[s:n])
                grad[s:n] += wm * (-h[0:nv])
        else:
            ss = -s
            nv = n - ss
            if nv > 0:
                grad[ss:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[ss:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = compute_grad(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs_bounded(h, n, alpha, mu, maxit, time_limit):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def obj_wrapper(x):
        if time.time() - t_start > time_limit:
            return smooth_max_val(compute_c5_vec(np.clip(x,0,1), n), alpha)
        return lbfgs_obj(x, n, alpha, mu)
    
    def grad_wrapper(x):
        if time.time() - t_start > time_limit:
            c = compute_c5_vec(np.clip(x,0,1), n)
            w = smooth_max_w(c, alpha)
            return compute_grad(w, np.clip(x,0,1), n)
        return lbfgs_grad(x, n, alpha, mu)
    
    try:
        res = minimize(obj_wrapper, h, jac=grad_wrapper,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def simulated_annealing_continuous(h, n, deadline, rng, n_iter,
                                    alpha_sm, T_start, T_end,
                                    sigma_start, sigma_end,
                                    n_perturb_lo, n_perturb_hi):
    h_work = project_feasible(h.copy(), n)
    best_h = h_work.copy()
    best_c = compute_c5(h_work, n)
    current_energy = smooth_max_val(compute_c5_vec(h_work, n), alpha_sm)
    
    if n_iter <= 1:
        return best_h, best_c
    cool_rate = (T_end / max(T_start, 1e-20)) ** (1.0 / n_iter)
    sig_cool = (sigma_end / max(sigma_start, 1e-20)) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_start * (cool_rate ** it)
        sigma = sigma_start * (sig_cool ** it)
        
        n_p = rng.randint(n_perturb_lo, n_perturb_hi + 1)
        idx = rng.choice(n, size=n_p, replace=False)
        h_new = h_work.copy()
        h_new[idx] += rng.normal(0, sigma, size=n_p)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5_vec(h_new, n)
        e_new = smooth_max_val(c_new, alpha_sm)
        de = e_new - current_energy
        
        if de < 0 or (T > 1e-15 and rng.random() < np.exp(-max(de, -400) / T)):
            h_work = h_new
            current_energy = e_new
            c_exact = float(c_new.max())
            if c_exact < best_c - 1e-14:
                best_c = c_exact
                best_h = h_work.copy()
    
    c_final = compute_c5(h_work, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h_work.copy()
    return best_h, best_c

def coordinate_descent(h, n, deadline, steps, max_rounds_per_step):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        rounds = 0
        while improved and rounds < max_rounds_per_step and time.time() < deadline:
            improved = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            w = smooth_max_w(c_vec, 500000.0)
            grad = compute_grad(w, best_h, n)
            
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            pos_sorted = pos_idx[np.argsort(-grad[pos_idx])][:25]
            neg_sorted = neg_idx[np.argsort(grad[neg_idx])][:25]
            
            best_gain = 0
            best_i, best_j = -1, -1
            for i in pos_sorted:
                for j in neg_sorted:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0:
                break
            
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-13:
                break
            
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] = old_i - amt
            best_h[best_j] = old_j + amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
            else:
                best_h = project_feasible(best_h, n)
                best_h[best_i] = old_i
                best_h[best_j] = old_j
                best_h = project_feasible(best_h, n)
    
    return best_h, best_c

def interpolate_h(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def targeted_optimize(h, n, deadline, top_k, time_per_call):
    c_vec = compute_c5_vec(h, n)
    top_idx = np.argsort(-c_vec)[:top_k]
    wgt = np.zeros(len(c_vec))
    vals = c_vec[top_idx]
    shift = vals - vals.min() + 0.5
    shift /= shift.sum()
    for i, idx in enumerate(top_idx):
        wgt[idx] = shift[i]
    
    dx = 2.0 / n
    
    def tgt_obj(x):
        c = compute_c5_vec(x, n)
        return np.dot(wgt, c) + 0.001 * (x.sum() - n/2.0)**2
    
    def tgt_grad(x):
        grad = np.zeros(n)
        for m in range(len(wgt)):
            if wgt[m] < 1e-15:
                continue
            s = m - (n-1)
            wm = wgt[m] * dx
            if s >= 0:
                nv = n - s
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - x[s:n])
                    grad[s:n] += wm * (-x[0:nv])
            else:
                ss = -s
                nv = n - ss
                if nv > 0:
                    grad[ss:n] += wm * (1.0 - x[0:nv])
                    grad[0:nv] += wm * (-x[ss:n])
        grad += 0.002 * (x.sum() - n/2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(tgt_obj, h, jac=tgt_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 3000, 'ftol': 1e-20,
                               'gtol': 1e-14, 'maxfun': 30000,
                               'maxls': 40, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: L-BFGS with increasing smoothness (0-150s)
    p1_end = min(t0 + 150, deadline - 300)
    alphas_p1 = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas_p1:
        if time.time() > p1_end:
            break
        h1, c1 = run_lbfgs_bounded(best_h, best_n, alpha, 0.01, 2000, 18)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    
    # Phase 2: Simulated annealing in continuous space (150-420s)
    p2_end = min(t0 + 420, deadline - 300)
    if time.time() < p2_end:
        h2, c2 = simulated_annealing_continuous(
            best_h, best_n, p2_end, rng,
            n_iter=120000, alpha_sm=200,
            T_start=0.012, T_end=0.0001,
            sigma_start=0.06, sigma_end=0.002,
            n_perturb_lo=3, n_perturb_hi=8
        )
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: L-BFGS polish (420-520s)
    p3_end = min(t0 + 520, deadline - 250)
    for alpha in [1000000, 5000000, 20000000]:
        if time.time() > p3_end:
            break
        h3, c3 = run_lbfgs_bounded(best_h, best_n, alpha, 0.005, 2500, 25)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Multi-resolution exploration (520-720s)
    p4_end = min(t0 + 720, deadline - 200)
    for n_try in [400, 600, 700]:
        if time.time() > p4_end:
            break
        if n_try == best_n:
            continue
        h_try = interpolate_h(best_h, best_n, n_try)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > p4_end:
                break
            h_t, c_t = run_lbfgs_bounded(h_try, n_try, alpha, 0.01, 2000, 18)
            h_try = h_t
        c_try = compute_c5(h_try, n_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            best_n = n_try
    
    # Phase 5: Targeted optimization (720-850s)
    p5_end = min(t0 + 850, deadline - 150)
    if time.time() < p5_end:
        for tk in [3, 5, 8, 12]:
            if time.time() > p5_end:
                break
            h5, c5 = targeted_optimize(best_h, best_n, p5_end, tk, 20)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: SA round 2 - low temperature (850-980s)
    p6_end = min(t0 + 980, deadline - 50)
    if time.time() < p6_end:
        h6, c6 = simulated_annealing_continuous(
            best_h, best_n, p6_end, rng,
            n_iter=80000, alpha_sm=500,
            T_start=0.004, T_end=0.00005,
            sigma_start=0.015, sigma_end=0.0008,
            n_perturb_lo=2, n_perturb_hi=5
        )
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final L-BFGS + coordinate descent (980-deadline)
    if time.time() < deadline - 40:
        h7, c7 = run_lbfgs_bounded(best_h, best_n, 50000000, 0.001, 3000, 40)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    if time.time() < deadline - 3:
        h8, c8 = coordinate_descent(
            best_h, best_n, deadline - 3,
            steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                   0.00002, 0.00001, 0.000005, 0.000002],
            max_rounds_per_step=300
        )
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
