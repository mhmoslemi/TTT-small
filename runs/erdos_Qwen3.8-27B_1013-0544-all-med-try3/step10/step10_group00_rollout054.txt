import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += (diff / tot) * room
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def make_beatty(n, alpha):
    h = np.zeros(n)
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n // 2:
                break
    for p in positions:
        h[p] = 1.0
    return project_feasible(h, n)

def interpolate_to_n(h_old, n_old, n_new):
    """Interpolate binary/text h to new resolution"""
    dx_old = 2.0 / n_old
    x_old = dx_old * (np.arange(n_old) + 0.5)  # cell centers
    x_new = (2.0 / n_new) * (np.arange(n_new) + 0.5)
    h_new = np.interp(x_new, x_old, h_old, left=0, right=0)

def simulated_annealing(h, n, deadline, rng, max_iters=100000):
    """Simulated annealing on binary sequences"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    T_start = 0.001
    T_end = 0.00001
    n_iters = max_iters
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / n_iters)
        
        # Propose a swap

        j = zeros[rng.randint(len(zeros))]

        new_c = compute_c5(h, n)
        delta = new_c - cur_c

            cur_c = new_c

            zeros = (zeros[zeros != j]).tolist()
            ones.append(j)
            zeros.append(i)
            ones = np.array(ones)

        else:

            if tot < 1e-14:

            h[j] = 0.0
    
    return best_h, best_c

def two_swap_local_search(h, n, deadline, rng, max_iters=20000):
    """2-opt style: swap two ones with two zeros simultaneously"""
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].astype(int).copy()
    zeros = np.where(h < 0.5)[0].astype(int).copy()
    n_ones = len(ones)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        improved = False
        for _ in range(5):
            i1 = ones[rng.randint(n_ones)]
            i2 = ones[rng.randint(n_ones)]
            if i2 == i1:
                continue
            j1 = zeros[rng.randint(n_ones)]
            j2 = zeros[rng.randint(n_ones)]
            if j2 == j1:
                continue
            
            h[i1] = 0.0; h[j1] = 1.0
            h[i2] = 0.0; h[j2] = 1.0
            new_c = compute_c5(h, n)
            
            if new_c < best_c - 1e-15:
                best_c = new_c
                for oi, oj in [(i1, j1), (i2, j2)]:
                    ones = (ones[ones != oi]).copy()
                    zeros = (zeros[zeros != oj]).copy()
                    ones = np.append(ones, oj)
                    zeros = np.append(zeros, oi)
                improved = True
                break
            else:
                h[i1] = 1.0; h[j1] = 0.0
                h[i2] = 1.0; h[j2] = 0.0
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = make_beatty(500, 0.618033988749895)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: Binarize + quick local search at parent res (40s)
    p1_end = min(t0 + 40, deadline)
    h_bin = binarize(best_h, best_n)
    hc, cc = guided_local_search(h_bin, best_n, p1_end, rng, max_iters=20000, guide_iters=1500)
    if cc < best_c:
        best_c = cc
        best_h = hc
    
    # Phase 2: ILS at parent resolution (140s)
    p2_end = min(t0 + 180, deadline)
    hils, cils = iterated_local_search(best_h, best_n, p2_end, rng, num_iterations=6)
    if cils < best_c:
        best_c = cils
        best_h = hils
    
    # Phase 3: Simulated annealing at parent res (100s)
    p3_end = min(t0 + 280, deadline)
    if time.time() < p3_end:
        hsa, csa = simulated_annealing(best_h, best_n, p3_end, rng, max_iters=40000, t_start=0.003, t_end=5e-6)
        if csa < best_c:
            best_c = csa
            best_h = hsa
        # Quick local search after SA
        if time.time() < p3_end + 20:
            hl, cl = guided_local_search(best_h, best_n, p3_end + 20, rng, max_iters=15000, guide_iters=1000)
            if cl < best_c:
                best_c = cl
                best_h = hl
    
    # Phase 4: Higher resolution n=1000 (200s)
    n_hi = 1000
    p4_end = min(t0 + 480, deadline)
    if time.time() < p4_end:
        h_interp = interpolate_to_n(best_h, best_n, n_hi)
        h_hi = binarize(h_interp, n_hi)
        
        # ILS at high res
        sub_dl = min(p4_end - 40, time.time() + (p4_end - time.time()) * 0.5)
        hh, ch = iterated_local_search(h_hi, n_hi, sub_dl, rng, num_iterations=5)
        
        # SA + local search
        sub_dl2 = min(p4_end, time.time() + (p4_end - time.time()) * 0.4)
        hs, cs = simulated_annealing(hh, n_hi, sub_dl2, rng, max_iters=30000, t_start=0.002, t_end=3e-6)
        if time.time() < p4_end:
            hs2, cs2 = guided_local_search(hs, n_hi, p4_end, rng, max_iters=10000, guide_iters=800)
            cs = min(cs, cs2)
            hs = hs if cs == cs else hs2
        if cs < best_c:
            best_c = cs
            best_h = hs
            best_n = n_hi
    
    # Phase 5: Even higher resolution n=1500 (150s)
    n_hi2 = 1500
    p5_end = min(t0 + 660, deadline)
    if time.time() < p5_end:
        h_interp2 = interpolate_to_n(best_h, best_n, n_hi2)
        h_hi2 = binarize(h_interp2, n_hi2)
        
        sub_dl = min(p5_end - 20, time.time() + (p5_end - time.time()) * 0.5)
        hh2, ch2 = iterated_local_search(h_hi2, n_hi2, sub_dl, rng, num_iterations=4)
        if time.time() < p5_end:
            hs3, cs3 = guided_local_search(hh2, n_hi2, p5_end, rng, max_iters=8000, guide_iters=800)
            if cs3 < ch2:
                hh2, ch2 = hs3, cs3
        if ch2 < best_c:
            best_c = ch2
            best_h = hh2
            best_n = n_hi2
    
    # Phase 6: Alternative initializations (100s)
    p6_end = min(t0 + 780, deadline)
    if time.time() < p6_end:
        golden = 0.618033988749895
        alphas = [golden, golden + 0.05, 0.55, 0.75, 0.45]
        for alpha in alphas:
            if time.time() > p6_end - 5:
                break
            h_bat = make_beatty(best_n, alpha)
            sub_dl = min(p6_end - 5, time.time() + 15)
            hb, cb = guided_local_search(h_bat, best_n, sub_dl, rng, max_iters=8000, guide_iters=800)
            if cb < best_c:
                best_c = cb
                best_h = hb
    
    # Phase 7: 2-swap local search on best (60s)
    p7_end = min(t0 + 860, deadline)
    if time.time() < p7_end:
        h2s, c2s = two_swap_local_search(best_h, best_n, p7_end, rng, max_iters=15000)
        if c2s < best_c:
            best_c = c2s
            best_h = h2s
        # Follow up with 1-swap
        if time.time() < p7_end + 30:
            h1s, c1s = guided_local_search(best_h, best_n, p7_end + 30, rng, max_iters=10000, guide_iters=1000)
            if c1s < best_c:
                best_c = c1s
                best_h = h1s
    
    # Phase 8: Final ILS polish (remaining time)
    if time.time() < deadline - 10:
        hfin, cfin = iterated_local_search(best_h, best_n, deadline - 10, rng, num_iterations=4)
        if cfin < best_c:
            best_c = cfin
            best_h = hfin
    
    # Phase 9: Very short SA restart
    if time.time() < deadline - 5:
        hsa2, csa2 = simulated_annealing(best_h, best_n, deadline - 5, rng, max_iters=8000, t_start=0.001, t_end=1e-6)
        if csa2 < best_c:
            best_c = csa2
            best_h = hsa2
        if time.time() < deadline - 1:
            hf, cf = guided_local_search(best_h, best_n, deadline - 1, rng, max_iters=5000, guide_iters=500)
            if cf < best_c:
                best_c = cf
                best_h = hf
    
    # Ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
