            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def run_lbfgs_active(h, n, alpha, active, beta, maxfun=20000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_level, h, jac=grad_level,
                       args=(n, alpha, active, beta),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxfun, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run_lbfgs_full(h, n, alpha, maxfun=10000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_full_smooth, h, jac=grad_full_smooth,
                       args=(n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxfun, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_moves(h, n, deadline, base_delta=0.001, max_iter=200):
    best_c = compute_c5(h, n)
    delta = base_delta
    iterations = 0
    while time.time() < deadline and iterations < max_iter:
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        active, _ = get_active_shifts(c_vec, tol_frac=0.005)
        
        # Compute gradient w.r.t. active shifts
        alpha = 10000
        vals = c_vec[active]
        sh = alpha * (vals - vals.max())
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            if wi < 1e-14:
                continue
            k = int(idx) - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        # Find best pair move
        g_sort = np.argsort(np.abs(grad))
        candidates = g_sort[:200]
        best_impr = 0
        best_i, best_j, best_d = -1, -1, 0
        for ii in range(min(50, len(candidates))):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(min(50, len(candidates))):
                if ii == jj:
                    continue
                j = candidates[jj]
                if grad[i] > grad[j]:
                    d_move = -delta
                else:
                    d_move = delta
                hi = h[i] + d_move
                hj = h[j] - d_move
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h[i] = hi
                    h[j] = hj
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-16:
                        best_impr = best_c - c_new
                        best_c = c_new
                        best_i, best_j, best_d = i, j, d_move
                    else:
                        h[i] -= d_move
                        h[j] += d_move
        
        iterations += 1
        if best_impr < 1e-16:
            delta *= 0.5
            if delta < 1e-8:
                break
        else:
            delta = min(delta * 0.7, 0.01)
    return h, best_c

def coord_moves(h, n, deadline, base_delta=0.0005):
    best_c = compute_c5(h, n)
    delta = base_delta
    stall = 0
    while time.time() < deadline and stall < 100:
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        active, _ = get_active_shifts(c_vec, tol_frac=0.005)
        
        alpha = 10000
        vals = c_vec[active]
        sh = alpha * (vals - vals.max())
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            if wi < 1e-14:
                continue
            k = int(idx) - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        improved = False
        # Try moving variable in steepest descent direction
        neg_g = np.argsort(grad)  # most negative first (steepest descent)
        for idx_i in neg_g[:100]:
            if time.time() > deadline:
                return h, best_c
            d = -np.sign(grad[idx_i]) * delta
            if d == 0:
                d = delta
            hi = h[idx_i] + d
            if 0 <= hi <= 1:
                h[idx_i] = hi
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    improved = True
                else:
                    h[idx_i] -= d
            else:
                d = -d
                hi = h[idx_i] + d
                if 0 <= hi <= 1:
                    h[idx_i] = hi
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        improved = True
                    else:
                        h[idx_i] -= d
            h = project_feasible(h, n)
        
        if not improved:
            stall += 1
            delta *= 0.5
            if delta < 1e-10:
                break
        else:
            stall = 0
    return h, best_c

def perturb_search(h, n, c, deadline, n_trials=10, scale=0.01):
    best_h = h.copy()
    best_c = c
    rng = np.random.RandomState(12345)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        h_p = h.copy()
        idx = rng.choice(n, size=n // 10, replace=False)
        h_p[idx] += rng.uniform(-scale, scale, size=len(idx))
        h_p = project_feasible(h_p, n)
        c_p = compute_c5(h_p, n)
        # Quick L-BFGS from perturbed point
        c_vec = compute_c5_vec(h_p, n)
        active, _ = get_active_shifts(c_vec, tol_frac=0.01)
        h_r, c_r = run_lbfgs_active(h_p, n, 50000, active, 0.0, maxfun=5000)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    return best_h, best_c

def anti_periodic_init(n, seed=42):
    """Create an anti-periodic initialization h(x+1)=1-h(x)."""
    rng = np.random.RandomState(seed)
    h_half = np.zeros(n // 2)
    # Create a piecewise structure on [0,1]
    n_blocks = 6
    block_idx = (np.arange(n // 2) * n_blocks) // (n // 2)
    for b in range(n_blocks):
        mask = block_idx == b
        h_half[mask] = rng.uniform(0.2, 0.8)
    # Make it anti-periodic
    h = np.concatenate([h_half, 1.0 - h_half])
    # Ensure integral constraint: sum(h) = n/2, which is automatic for anti-periodic
    # Actually sum(h) = sum(h_half) + sum(1-h_half) = sum(h_half) + n/2 - sum(h_half) = n/2. Good.
    return project_feasible(np.clip(h, 0, 1), n)

def smooth_transition(h, n):
    """Apply a small Gaussian smoothing to remove high-frequency noise."""
    from scipy.ndimage import gaussian_filter1d
    h_s = gaussian_filter1d(h, sigma=2.0)
    return project_feasible(h_s, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = anti_periodic_init(500, seed)
        n_orig = 500

    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)

    # Phase 1: Quick L-BFGS at original resolution
    if time.time() < deadline - 400:
        alphas_p1 = [200, 1000, 5000]
        for a in alphas_p1:
            if time.time() > deadline - 400:
                break
            c_vec = compute_c5_vec(best_h, n_orig)
            active, _ = get_active_shifts(c_vec, tol_frac=0.02)
            h_tmp, c_tmp = run_lbfgs_active(best_h, n_orig, a, active, 0.0, maxfun=8000)
            if c_tmp < best_c:
                best_c = c_tmp
                best_h = h_tmp

    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_up = np.interp(x_new, x_old, best_h)
    h_up = project_feasible(h_up, n)
    c_up = compute_c5(h_up, n)
    if c_up <= best_c + 0.001:
        best_h = h_up
        best_c = c_up
        n_work = n
    else:
        n_work = n_orig
        best_h = project_feasible(best_h, n_orig)

    # Phase 3: Multi-alpha L-BFGS at working resolution
    alphas_p3 = [500, 2000, 10000, 50000, 200000, 1000000]
    betas_p3 = [0.0, 0.0, 0.0, 0.01, 0.1, 1.0]
    for i, (a, b) in enumerate(zip(alphas_p3, betas_p3)):
        if time.time() > deadline - 200:
            break
        c_vec = compute_c5_vec(best_h, n_work)
        tol = max(0.001, 0.05 / (i + 1))
        active, _ = get_active_shifts(c_vec, tol_frac=tol)
        h_tmp, c_tmp = run_lbfgs_active(best_h, n_work, a, active, b, maxfun=15000)
        if c_tmp < best_c:
            best_c = c_tmp
            best_h = h_tmp

    # Phase 4: Very aggressive active-set optimization
    if time.time() < deadline - 150:
        for a in [500000, 2000000, 10000000]:
            if time.time() > deadline - 150:
                break
            c_vec = compute_c5_vec(best_h, n_work)
            active, _ = get_active_shifts(c_vec, tol_frac=0.003)
            h_tmp, c_tmp = run_lbfgs_active(best_h, n_work, a, active, 0.1, maxfun=10000)
            if c_tmp < best_c:
                best_c = c_tmp
                best_h = h_tmp

    # Phase 5: Pair moves for fine tuning
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        best_h, best_c = pair_moves(best_h, n_work, t5, base_delta=0.002, max_iter=1000)

    # Phase 6: Coordinate moves
    if time.time() < deadline - 60:
        t6 = min(time.time() + 40, deadline - 60)
        best_h, best_c = coord_moves(best_h, n_work, t6, base_delta=0.001)

    # Phase 7: Perturbation restarts
    if time.time() < deadline - 50:
        t7 = min(time.time() + 40, deadline - 50)
        best_h, best_c = perturb_search(best_h, n_work, best_c, t7, n_trials=8, scale=0.005)

    # Phase 8: Final L-BFGS polish
    if time.time() < deadline - 10:
        c_vec = compute_c5_vec(best_h, n_work)
        active, _ = get_active_shifts(c_vec, tol_frac=0.002)
        h_tmp, c_tmp = run_lbfgs_active(best_h, n_work, 5000000, active, 0.01, maxfun=8000)
        if c_tmp < best_c:
            best_c = c_tmp
            best_h = h_tmp

    # Phase 9: Final coordinate polish
    if time.time() < deadline - 3:
        best_h, best_c = coord_moves(best_h, n_work, deadline - 3, base_delta=0.0005)

    # Final projection and evaluation
    best_h = project_feasible(best_h, n_work)
    best_c = compute_c5(best_h, n_work)

    return best_h, best_c, n_work
