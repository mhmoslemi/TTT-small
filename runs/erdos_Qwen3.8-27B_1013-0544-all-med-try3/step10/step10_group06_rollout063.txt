```python
def compute_a_for_j(h, j, n):
    a = np.zeros(2*n-1)
    lag = np.arange(-(n-1), n)  # lag values: -(n-1), ..., n-1
    k = lag + (n-1)  # corresponding k index
    
    # For each lag:
    # First factor: h[j] * (1-h[j+lag]) contributes (1-h[j+lag]) if 0 ≤ j+lag < n
    # Second factor: h[j-lag] * (1-h[j]) contributes -h[j-lag] if 0 ≤ j-lag < n
    # Special case lag=0: h[j]*(1-h[j]), derivative = 1-2h[j]
    
    valid1 = (j + lag >= 0) & (j + lag < n)
    valid2 = (j - lag >= 0) & (j - lag < n)
    
    term1 = np.zeros(2*n-1)
    term2 = np.zeros(2*n-1)
    valid1[lag == 0] = False  # handle lag=0 separately
    valid2[lag == 0] = False
    
    # term1[k] = (1 - h[j+lag[k]]) for lags where j+lag is valid
    idx1 = j + lag[valid1]
    term1[valid1] = (1.0 - h[idx1])
    
    # term2[k] = -h[j-lag[k]] for lags where j-lag is valid
    idx2 = j - lag[valid2]
    term2[valid2] = -h[idx2]
    
    a = term1 + term2
    a[n-1] = 1.0 - 2.0*h[j]  # lag = 0
    return a
```

```python
        c_all = np.correlate(h, 1 - h, mode='full')

        for j in range(n):
            if time.time() > deadline:
                break
            a = compute_a_vec(h, j, n)
            b = c_all - a * h[j]
            d = a + lam_eff / dx
            t_opt = solve_1d_max_lines(b, d)
            delta = t_opt - h[j]
            if abs(delta) < 1e-12:
                continue
            h[j] = t_opt
            c_all += a * delta

        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
        c_all = np.correlate(h, 1 - h, mode='full')

    return best_h, best_c

def pairwise_local_search(h, n, deadline, rng, top_k=8, base_step=0.015):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False
        for _ in range(5000):
            if time.time() > deadline:
                break
            k_star = top_idx[rng.randint(top_k)]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-8:
                d = min(base_step, (h[i] - h[j]) * 0.3)
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    stall = 0
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
            elif h[j] > h[i] + 1e-8:
                d = min(base_step, (h[j] - h[i]) * 0.3)
                if d < 1e-10:
                    continue
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    stall = 0
                else:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
            else:
                continue
        if not improved:
            stall += 1
            base_step *= 0.7
            if base_step < 1e-5 or stall > 20:
                break
        else:
            stall = 0

    return best_h, best_c

def simulated_annealing_focused(h, n, deadline, rng, alpha_sa=200000, t_init=0.005, t_min=0.00005, cool=0.9997):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_stall = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-5:]
        k_star = top_k[rng.randint(5)]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m = abs(lag)
        if m >= n:
            T *= cool
            continue
        i = rng.randint(0, n - m)
        j = i + m
        if h[i] > h[j] + 1e-8:
            d = min(0.03, (h[i] - h[j]) * 0.3, max(0.002, T * 5))
            h[i] -= d
            h[j] += d
        elif h[j] > h[i] + 1e-8:
            d = min(0.03, (h[j] - h[i]) * 0.3, max(0.002, T * 5))
            h[i] += d
            h[j] -= d
        else:
            T *= cool
            continue
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        e_old = smooth_max_value(np.correlate(h, 1 - h, mode='full'), alpha_sa)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            it_stall = 0
        else:
            ratio = (e_old - smooth_max_value(np.correlate(h, 1-h, mode='full'), alpha_sa)) / T
            if ratio > 0 and ratio < 40:
                if rng.random() < np.exp(-ratio):
                    it_stall += 1
                else:
                    c_all2 = compute_all_lags(h, n)
                    h_rollback = best_h.copy()
                    h[i] += d if h[i] < h[j] else -d
                    h[j] -= d if h[i] < h[j] else d
                    h = project_feasible(h, n)
                    it_stall += 1
            else:
                h[i] += d if lag > 0 else -d
                h[j] -= d if lag > 0 else d
                h = project_feasible(h, n)
                it_stall += 1
        T *= cool
        if it_stall > 30000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            it_stall = 0
            T = max(T, 0.003)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 1000

    n = 1000
    if n_init == n:
        h = project_feasible(h_init, n) if h_init is not None else project_feasible(np.full(n, 0.5), n)
    elif h_init is not None:
        x_src = np.linspace(0, 2, n_init)
        x_dst = np.linspace(0, 2, n)
        h = np.interp(x_dst, x_src, h_init)
        h = project_feasible(h, n)
    else:
        h = project_feasible(np.full(n, 0.5), n)

    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B coarse
    alphas_coarse = [500, 2000, 10000, 50000]
    maxiters_coarse = [400, 600, 800, 1000]
    for alpha, maxit in zip(alphas_coarse, maxiters_coarse):
        if time.time() > deadline - 400:
            break
        rem = deadline - 400 - time.time()
        if rem < 5:
            break
        h_new = lbfgsb_opt(best_h, n, alpha, time.time() + min(rem * 0.35, 50), maxiter=maxit)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Coordinate descent
    if time.time() < deadline - 300:
        cd_deadline = min(deadline - 280, time.time() + 120)
        h_cd, c_cd = coordinate_descent(best_h, n, cd_deadline, lam=0.0, max_sweeps=50)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B fine
    alphas_fine = [200000, 1000000, 5000000, 20000000]
    maxiters_fine = [1000, 1500, 2000, 2500]
    for alpha, maxit in zip(alphas_fine, maxiters_fine):
        if time.time() > deadline - 200:
            break
        rem = deadline - 200 - time.time()
        if rem < 5:
            break
        h_new = lbfgsb_opt(best_h, n, alpha, time.time() + min(rem * 0.3, 60), maxiter=maxit)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: More coordinate descent
    if time.time() < deadline - 200:
        cd_deadline2 = min(deadline - 180, time.time() + 100)
        h_cd, c_cd = coordinate_descent(best_h, n, cd_deadline2, lam=0.0, max_sweeps=50)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd

    # Phase 5: Local search
    if time.time() < deadline - 150:
        ls_deadline = min(deadline - 130, time.time() + 150)
        h_ls, c_ls = pairwise_local_search(best_h, n, ls_deadline, rng, top_k=6, base_step=0.02)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 6: SA
    if time.time() < deadline - 120:
        sa_deadline = min(deadline - 100, time.time() + 100)
        h_sa, c_sa = simulated_annealing_focused(best_h, n, sa_deadline, rng,
                                                 alpha_sa=300000, t_init=0.004,
                                                 t_min=0.00005, cool=0.9998)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 7: Final L-BFGS-B
    if time.time() < deadline - 60:
        h_new = lbfgsb_opt(best_h, n, 100000000, deadline - 20, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 8: Final coordinate descent + local search
    if time.time() < deadline - 20:
        h_cd, c_cd = coordinate_descent(best_h, n, deadline - 10, lam=0.0, max_sweeps=30)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
            h_ls, c_ls = pairwise_local_search(best_h, n, deadline - 3, rng, top_k=5, base_step=0.005)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
