def weights_from_c(c, alpha, top_k):
    m = float(c.max())
    if top_k and top_k > 0:
        w = np.zeros(c.shape, dtype=float)
        threshold = m - max(1e-9, 2e-4 * m)
        idx = np.where(c >= threshold)[0]
        if idx.size == 0:
            idx = np.where(c >= m - 1e-12)[0]
        if idx.size > int(top_k):
            idx = idx[np.argsort(-c[idx])[:int(top_k)]]
        if idx.size > 0:
            w[idx] = 1.0 / float(idx.size)
        else:
            w[int(np.argmax(c))] = 1.0
        return w
    # smooth
    if alpha <= 0:
        w = np.ones_like(c) / c.size
        return w
    z = alpha * (c - m)
    smx = float(z.max())

import numpy as np
import time
from scipy.optimize import minimize

_FFT_CACHE = {}


def _fft_params(n):
    n = int(n)
    if n < 2:
        n = 2
    p = _FFT_CACHE.get(n)
    if p is None:
        M = 1 << (2 * n - 2).bit_length()
        p = (M, slice(1, n), slice(M - n + 1, M))
        _FFT_CACHE[n] = p
    return p


def fft_c_full(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    M, pos_slice, neg_slice = _fft_params(n)
    H = np.fft.rfft(h, M)
    G = np.fft.rfft(1.0 - h, M)
    c = np.fft.irfft(H * np.conj(G), M)
    out = np.empty(2 * n - 1, dtype=float)
    out[:n - 1] = c[neg_slice]
    out[n - 1:] = c[0:n]
    np.maximum(out, 0.0, out)
    return out * (2.0 / float(n))


def fast_c_max(h, n):
    return float(fft_c_full(h, n).max())


def direct_c_value(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    return float(c.max())

    target = 0.5 * float(n)

    yhi = float(np.max(y))
    lo = ylo - 1.0
    hi = yhi + 1.0
    mid = 0.5 * (lo + hi)

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(y - mid, 0.0, 1.0)

    for _ in range(20):

        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            room = 1.0 - x
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            if diff >= tot:
                x = x + room
                x = np.clip(x, 0.0, 1.0)
            else:
                x = x + (diff / tot) * room
        else:
            tot = float(x.sum())
            if tot <= 1e-15:
                break
            if -diff >= tot:
                x = x - x
            else:
                x = x + (diff / tot) * x
        x = np.clip(x, 0.0, 1.0)

    diff = target - float(x.sum())
    for _ in range(20):
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            add = min(diff, 1.0 - x[idx])
            if add > 0.0:
                x[idx] += add
        elif diff < 0:
            idx = int(np.argmin(x))
            sub = min(-diff, x[idx])
            if sub > 0.0:
                x[idx] -= sub
        x = np.clip(x, 0.0, 1.0)
        diff = target - float(x.sum())

    return x


def mix_feasible(a, b, t, n):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if t <= 0.0:
        return a.copy()
    if t >= 1.0:
        return b.copy()
    x = a + (b - a) * t
    target = 0.5 * float(n)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(mask):
            x[mask] += diff / float(np.sum(mask))
        else:
            x = x + diff / float(n)
    return np.clip(x, 0.0, 1.0)


def vertex_from_gradient(G, n):
    n = int(n)
    G = np.asarray(G, dtype=float)
    target = 0.5 * float(n)
    S = int(round(target))
    v = np.zeros(n, dtype=float)

    if S <= 0:
        return v
    if S >= n:
        v.fill(1.0)
        return v

    if abs(target - float(S)) < 1e-12:
        if S < n:
            part = np.argpartition(G, S)[:S]
            v[part] = 1.0
        return v

    Sfloor = int(target)
    if Sfloor > 0 and Sfloor < n:
        part = np.argpartition(G, Sfloor)[:Sfloor]
        v[part] = 1.0

    frac = target - float(Sfloor)
    if frac > 1e-12 and frac < 1.0 - 1e-12:
        G2 = G.copy()
        G2[v > 0.0] = np.inf
        idx = int(np.argmin(G2))
        v[idx] = frac
    return v


def smooth_value_weights(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(c.max())
    if alpha <= 0.0:
        w = np.ones(c.shape, dtype=float)
        w /= float(w.size)
        return m, w
    z = alpha * (c - m)
    smx = float(z.max())
    e = np.exp(z - smx)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros(c.shape, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return m, w
    e = e / s
    val = m + (smx + np.log(max(s, 1e-300)) / alpha)
    return float(val), e


def smooth_max_val(c, alpha):
    v, _ = smooth_value_weights(c, alpha)
    return v


def weights_from_c(c, alpha, top_k):
    c = np.asarray(c, dtype=float)
    top_k = int(top_k)
    m = float(c.max())

    if top_k > 0:
        w = np.zeros(c.shape, dtype=float)
        threshold = m - max(1e-9, 2e-4 * m)
        idx = np.where(c >= threshold)[0]
        if idx.size == 0:
            idx = np.where(c >= m - 1e-12)[0]
        if idx.size > top_k:
            idx = idx[np.argsort(-c[idx])[:top_k]]
        if idx.size > 0:
            w[idx] = 1.0 / float(idx.size)
        else:
            w[int(np.argmax(c))] = 1.0
        return w

    if alpha <= 0.0:
        w = np.ones(c.shape, dtype=float)
        w /= float(w.size)
        return w

    _, w = smooth_value_weights(c, alpha)
    return w


def dense_grad_from_weights(h, wfull, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    wfull = np.asarray(wfull, dtype=float)
    M, pos_slice, neg_slice = _fft_params(n)
    idx = n - 1
    w0 = float(wfull[idx])

    if n == 1:
        L = np.full(1, w0, dtype=float)
        S = 2.0 * w0 * h
        return (2.0 / float(n)) * (L - S)

    Wpos = wfull[idx + 1:].copy()
    Wneg = wfull[:idx][::-1].copy()

    pos_suf = np.empty(n, dtype=float)
    pos_suf[-1] = 0.0
    if n > 1:
        pos_suf[:-1] = np.cumsum(Wpos[::-1])[::-1]

    neg_head = np.empty(n, dtype=float)
    neg_head[0] = 0.0
    if n > 1:
        neg_head[1:] = np.cumsum(Wneg)

    L = w0 + pos_suf + neg_head

    F = np.zeros(M, dtype=float)
    F[0] = 2.0 * w0
    Wsum = Wpos + Wneg
    F[pos_slice] = Wsum
    F[neg_slice] = Wsum[::-1]

    S = np.fft.irfft(np.fft.rfft(h, M) * np.fft.rfft(F, M), M)[:n]
    return (2.0 / float(n)) * (L - S)


def gradient_from_c(c, h, n, alpha, top_k):
    w = weights_from_c(c, alpha, top_k)
    return dense_grad_from_weights(h, w, n)


def dense_smooth_obj(x, n, alpha, mu):
    c = fft_c_full(x, n)
    val = smooth_max_val(c, alpha)
    if mu != 0.0:
        diff = float(x.sum()) - 0.5 * float(n)
        val += mu * diff * diff
    return float(val)


def dense_smooth_jac(x, n, alpha, mu):
    c = fft_c_full(x, n)
    _, w = smooth_value_weights(c, alpha)
    g = dense_grad_from_weights(x, w, n)
    if mu != 0.0:
        g = g + 2.0 * mu * (float(x.sum()) - 0.5 * float(n))
    return g


def fw_round(h, n, deadline, alpha, top_k, rng, max_iter=None, no_impact_limit=40):
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    best_c = fast_c_max(h, n)
    noimp = 0
    prev_t = 0.5

    if max_iter is None or int(max_iter) <= 0:
        max_iter = max(30, int((deadline - time.time()) * 2))
    max_iter = int(min(int(max_iter), 600))

    for _ in range(max_iter):
        if time.time() > deadline:
            break

        c = fft_c_full(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-14:
            best_c = cur
            best_h = h.copy()

        G = gradient_from_c(c, h, n, alpha, top_k)
        gmax = float(np.max(np.abs(G))) if G.size else 0.0
        if gmax < 1e-15:
            break

        if noimp > 4:
            G = G + rng.normal(0.0, 1.0, size=n) * gmax * 1e-7

        v = vertex_from_gradient(G, n)
        if np.max(np.abs(v - h)) < 1e-12:
            noimp += 1
            if noimp > no_impact_limit:
                break
            continue

        lin = float(np.dot(G, v - h))
        if lin > -1e-14:
            noimp += 1
            if noimp > no_impact_limit:
                break
            continue

        pts = []
        for p in (1.0, 0.5, prev_t, 0.25, 0.75, 0.1, 0.9, 0.35, 0.65):
            if p <= 0.0 or p > 1.0:
                continue
            dup = False
            for q in pts:
                if abs(p - q) < 1e-6:
                    dup = True
                    break
            if not dup:
                pts.append(p)

        best_t = 0.0
        best_ct = cur
        best_hd = h

        for tt in pts:
            ht = mix_feasible(h, v, tt, n)
            if np.max(np.abs(ht - h)) < 1e-12:
                continue
            ct = float(fft_c_full(ht, n).max())
            if ct < best_ct - 1e-15:
                best_ct = ct
                best_t = tt
                best_hd = ht

        if best_t > 0.0 and best_ct < cur - 1e-15:
            h = best_hd
            prev_t = best_t
            noimp = 0
            if best_ct < best_c:
                best_c = best_ct
                best_h = h.copy()
        else:
            noimp += 1
            if noimp > 2 and noimp % 3 == 0:
                for _ in range(3):
                    tt = float(rng.uniform(0.05, 0.95))
                    ht = mix_feasible(h, v, tt, n)
                    ct = float(fft_c_full(ht, n).max())
                    if ct < cur - 1e-14:
                        h = ht
                        prev_t = tt
                        noimp = 0
                        if ct < best_c:
                            best_c = ct
                            best_h = h.copy()
                        break
            if noimp > no_impact_limit:
                break

    best_h = project_feasible(best_h, n)
    h2 = project_feasible(h, n)
    c2 = fast_c_max(h2, n)
    if c2 < best_c:
        best_h = h2
        best_c = c2
    else:
        best_c = fast_c_max(best_h, n)
    return best_h, best_c


def pg_round(h, n, deadline, alpha, top_k, max_iter=None, no_impact_limit=60):
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    best_c = fast_c_max(h, n)
    noimp = 0

    if max_iter is None or int(max_iter) <= 0:
        max_iter = max(30, int((deadline - time.time()) * 1.5))
    max_iter = int(min(int(max_iter), 500))

    mults = (0.05, 0.2, 0.8, 3.2, 12.8, 51.2, 204.8)

    for _ in range(max_iter):
        if time.time() > deadline:
            break

        c = fft_c_full(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-14:
            best_c = cur
            best_h = h.copy()

        G = gradient_from_c(c, h, n, alpha, top_k)
        gmax = float(np.max(np.abs(G))) if G.size else 0.0
        if gmax < 1e-15:
            break

        base = 1.0 / gmax
        best_ct = cur
        best_hd = h

        for m in mults:
            y = h - (m * base) * G
            ht = project_feasible(y, n)
            if np.max(np.abs(ht - h)) < 1e-12:
                continue
            ct = float(fft_c_full(ht, n).max())
            if ct < best_ct - 1e-15:
                best_ct = ct
                best_hd = ht

        if best_ct < cur - 1e-15:
            h = best_hd
            noimp = 0
            if best_ct < best_c:
                best_c = best_ct
                best_h = h.copy()
        else:
            noimp += 1
            if noimp > no_impact_limit:
                break

    best_h = project_feasible(best_h, n)
    h2 = project_feasible(h, n)
    c2 = fast_c_max(h2, n)
    if c2 < best_c:
        best_h = h2
        best_c = c2
    else:
        best_c = fast_c_max(best_h, n)
    return best_h, best_c


def lbfgs_round(h, n, deadline, alpha, mu, maxiter=None):
    now = time.time()
    if now > deadline - 20:
        return h

    n = int(n)
    if maxiter is None:
        if n <= 512:
            maxiter = 1800
        elif n <= 1024:
            maxiter = 1300
        elif n <= 2048:
            maxiter = 900
        else:
            maxiter = 700
        time_left = max(1.0, deadline - now)
        scale = min(2.0, time_left / 60.0)
        maxiter = int(max(120, min(4000, maxiter * scale)))

    h_in = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(
            dense_smooth_obj,
            h_in,
            jac=dense_smooth_jac,
            args=(n, alpha, mu),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "ftol": 1e-15,
                "gtol": 1e-12,
                "maxfun": max(iter * 20 for iter in (maxiter, 1)),
                "maxls": 25,
                "maxcor": 20,
            },
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass

    return h_in


def top_indices_by_score(arr, k):
    arr = np.asarray(arr, dtype=float)
    m = int(arr.size)
    k = int(k)
    if m == 0 or k <= 0:
        return np.array([], dtype=int)
    if k >= m:
        return np.argsort(-arr)[:k]
    return np.argpartition(-arr, k)[:k]


def gradient_pair_round(h, n, deadline, alpha, top_k, rng, max_iter=None, no_impact_limit=70):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    best_h = h.copy()
    best_c = fast_c_max(h, n)
    noimp = 0
    top_k = max(1, min(int(top_k), n))
    edge_k = max(4, min(12, top_k))

    if max_iter is None or int(max_iter) <= 0:
        max_iter = max(30, int((deadline - time.time()) * 1.2))
    max_iter = int(min(int(max_iter), 800))

    for _ in range(max_iter):
        if time.time() > deadline:
            break

        c = fft_c_full(h, n)
        cur = float(c.max())
        if cur < best_c - 1e-14:
            best_c = cur
            best_h = h.copy()

        G = gradient_from_c(c, h, n, alpha, 20)
        cands = []
        seen = set()

        if top_k > 0:
            if top_k >= n:
                order = np.argsort(G)
                dst_idx = order[:n]
                src_idx = order[-n:][::-1]
            elif n > 2 * top_k:
                dst_idx = np.argpartition(G, top_k)[:top_k]
                src_idx = np.argpartition(G, n - top_k)[n - top_k:]
            else:
                order = np.argsort(G)
                dst_idx = order[:top_k]
                src_idx = order[-top_k:][::-1]

            for s in src_idx:
                si = int(s)
                for d in dst_idx:
                    di = int(d)
                    if si == di:
                        continue
                    pair = (si, di)
                    if pair not in seen:
                        seen.add(pair)
                        cands.append(pair)

        active_k = min(10, 2 * n - 1)
        tol = max(1e-8, 0.003 * cur)
        active_idx = np.where(c >= cur - tol)[0]
        if active_idx.size > active_k:
            active_idx = active_idx[np.argsort(-c[active_idx])[:active_k]]

        for L in active_idx:
            k = int(L) - (n - 1)
            if k == 0:
                continue
            if k > 0:
                diff = h[0:n - k] - h[k:n]
                if diff.size == 0:
                    continue
                if float(diff.max()) <= 1e-8:
                    continue
                part = top_indices_by_score(diff, edge_k)
                for ii in part:
                    if diff[ii] <= 1e-8:
                        continue
                    s = int(ii)
                    d = int(ii + k)
                    pair = (s, d)
                    if pair not in seen:
                        seen.add(pair)
                        cands.append(pair)
            else:
                kk = -k
                diff = h[kk:n] - h[0:n - kk]
                if diff.size == 0:
                    continue
                if float(diff.max()) <= 1e-8:
                    continue
                part = top_indices_by_score(diff, edge_k)
                for ii in part:
                    if diff[ii] <= 1e-8:
                        continue
                    s = int(ii + kk)
                    d = int(ii)
                    pair = (s, d)
                    if pair not in seen:
                        seen.add(pair)
                        cands.append(pair)

        if len(cands) > 400:
            cands = cands[:400]

        improved = False
        for (s, d) in cands:
            if time.time() > deadline:
                break
            sv = float(h[s])
            dv = float(h[d])
            if sv <= 1e-9 or dv >= 1.0 - 1e-9:
                continue
            maxd = min(sv, 1.0 - dv)
            if maxd <= 1e-9:
                continue

            for frac in (0.5, 1.0, 0.25, 0.75):
                delta = maxd * frac
                h[s] = sv - delta
                h[d] = dv + delta

                if h[s] < 0.0:
                    corr = -h[s]
                    h[s] = 0.0
                    h[d] = min(1.0, h[d] + corr)
                elif h[d] > 1.0:
                    corr = h[d] - 1.0
                    h[d] = 1.0
                    h[s] = min(1.0, h[s] + corr)

                ct = float(fft_c_full(h, n).max())
                if ct < best_c - 1e-15:
                    best_c = ct
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[s] = sv
                    h[d] = dv

            if improved:
                break

        if improved:
            noimp = 0
        else:
            noimp += 1
            if noimp <= no_impact_limit and noimp % 5 == 0:
                rk = max(2, min(12, n))
                if n > rk:
                    src_arr = np.argpartition(G, n - rk)[-rk:]
                    dst_arr = np.argpartition(G, rk)[:rk]
                else:
                    src_arr = np.arange(n)
                    dst_arr = np.arange(n)

                for _ in range(30):
                    si = int(src_arr[rng.randint(0, src_arr.size)])
                    di = int(dst_arr[rng.randint(0, dst_arr.size)])
                    if si == di:
                        continue
                    sv = float(h[si])
                    dv = float(h[di])
                    delta = min(1e-4, 0.5 * min(sv, 1.0 - dv))
                    if delta <= 1e-12:
                        continue
                    h[si] = sv - delta
                    h[di] = dv + delta
                    ct = float(fft_c_full(h, n).max())
                    if ct < best_c - 1e-15:
                        best_c = ct
                        best_h = h.copy()
                        improved = True
                        noimp = 0
                        break
                    else:
                        h[si] = sv
                        h[di] = dv

            if not improved and noimp > no_impact_limit:
                break

    best_h = project_feasible(best_h, n)
    h2 = project_feasible(h, n)
    c2 = fast_c_max(h2, n)
    if c2 < best_c:
        best_h = h2
        best_c = c2
    else:
        best_c = fast_c_max(best_h, n)
    return best_h, best_c


def make_random_binary(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)


def make_sturmian(n, alpha, beta=0.0):
    n = int(n)
    idx = np.arange(n, dtype=float)
    vals = np.mod(alpha * idx + beta, 1.0)
    target = int(round(0.5 * float(n)))
    h = np.zeros(n, dtype=float)
    if target > 0:
        part = np.argsort(vals)[:target]
        h[part] = 1.0
    return project_feasible(h, n)


def resize_h(h, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    h = np.asarray(h, dtype=float)
    if n_new == n_old:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, n_new)


def optimize_resolution(h, n, deadline, rng):
    h = project_feasible(h, n)
    n = int(n)
    t_start = time.time()
    rem0 = max(0.0, deadline - t_start)
    if rem0 < 60.0:
        return h

    d1 = min(deadline, t_start + max(25.0, 0.15 * rem0))
    h, _ = fw_round(h, n, d1, 1200.0, 12, rng,
                    max_iter=max(30, int(rem0 * 0.25)))
    if time.time() > deadline:
        return h

    d2 = min(deadline, time.time() + max(30.0, 0.25 * rem0))
    h = lbfgs_round(h, n, d2, 8000.0, 0.25, None)
    if not np.all(np.isfinite(h)):
        h = project_feasible(np.nan_to_num(h), n)
    if time.time() > deadline:
        return h

    d3 = min(deadline, time.time() + max(25.0, 0.15 * rem0))
    h, _ = fw_round(h, n, d3, 35000.0, 0, rng,
                    max_iter=max(20, int(rem0 * 0.2)))
    if time.time() > deadline:
        return h

    d4 = min(deadline, time.time() + max(25.0, 0.15 * rem0))
    h, _ = pg_round(h, n, d4, 1.0, 40,
                    max_iter=max(20, int(rem0 * 0.25)))
    if time.time() > deadline:
        return h

    h, _ = gradient_pair_round(h, n, deadline, 8000.0, 8, rng,
                               max_iter=max(30, int(rem0 * 0.55)))
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    deadline = t0 + max(30.0, budget - 12.0)
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(512, 0.5)

    n0 = int(h0.size)
    if n0 < 64:
        h0 = make_sturmian(512, 0.5 * (np.sqrt(5.0) - 1.0), 0.0)
        n0 = 512

    if n0 > 4096:
        h0 = resize_h(h0, n0, 4096)
        n0 = 4096

    if n0 % 2 == 1:
        h0 = resize_h(h0, n0, n0 + 1)
        n0 += 1

    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_c = direct_c_value(best_h, best_n)

    golden = 0.5 * (np.sqrt(5.0) - 1.0)

    alt_starts = [
        make_sturmian(n0, golden, 0.0),
        make_sturmian(n0, 0.57735026919, 0.25),
        make_sturmian(n0, 0.54037, 0.71),
        make_random_binary(n0, rng),
        make_random_binary(n0, rng),
    ]

    for ha in alt_starts:
        if time.time() > deadline - 60:
            break
        ha = project_feasible(ha, n0)
        ca = direct_c_value(ha, n0)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()
            best_n = n0

    cur_h = best_h.copy()
    cur_n = int(best_n)
    cur_c = float(best_c)

    ncap_min = min(80, int(deadline - time.time()))
    base_end = min(deadline, t0 + max(30.0, min(90.0, 0.12 * max(30.0, budget - 30))))
    if base_end > time.time() + 25 and cur_n <= 1024:
        cur_h = optimize_resolution(cur_h, cur_n, base_end, rng)

        elif cur_n != best_n:

    if time.time() < deadline - 110 and cur_n < 4096:

        final_h = project_feasible(best_h, final_n)
