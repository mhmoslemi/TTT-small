import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topm_smooth_obj(x, n, M, alpha, mu):
    c = compute_c5_vec(x, n)
    if M < c.size:
        top_idx = np.argpartition(c, -M)[-M:]
        c_top = c[top_idx]
    else:
        c_top = c
    c_max = c.max()
    shifted = alpha * (c_top - c_max)
    mx = shifted.max()
    if mx > 500:
        lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(shifted))) / alpha
    val = c_max + lse
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topm_smooth_grad(x, n, M, alpha, mu):
    c = compute_c5_vec(x, n)
    if M < c.size:
        top_idx = np.argpartition(c, -M)[-M:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(c.size)
        c_top = c
    c_max = c.max()
    shifted = alpha * (c_top - c_max)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for j in range(len(top_idx)):
        wm = w[j] * dx
        if wm < 1e-14:
            continue
        k = top_idx[j] - center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[k:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[k:n])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, M, alphas, mus):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for i, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(topm_smooth_obj, best_h, jac=topm_smooth_grad,
                          args=(n, M, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def slqp_optimize(h, n, deadline, M=30, alpha=5000, maxiter=3000):
    def obj_fn(x, na, m, al):
        c = compute_c5_vec(x, na)
        if m < c.size:
            top_idx = np.argpartition(c, -m)[-m:]
            c_top = c[top_idx]
        else:
            c_top = c
        c_max = c.max()
        shifted = al * (c_top - c_max)
        mx = shifted.max()
        if mx > 500:
            lse = (mx + np.log(np.sum(np.exp(shifted - mx)))) / al
        else:
            lse = np.log(np.sum(np.exp(shifted))) / al
        return float(c_max + lse)
    def obj_jac(x, na, m, al):
        c = compute_c5_vec(x, na)
        if m < c.size:
            top_idx = np.argpartition(c, -m)[-m:]
            c_top = c[top_idx]
        else:
            top_idx = np.arange(c.size)
            c_top = c
        c_max = c.max()
        shifted = al * (c_top - c_max)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        dx = 2.0 / na
        grad = np.zeros(na)
        center = na - 1
        for j in range(len(top_idx)):
            wm = w[j] * dx
            if wm < 1e-14:
                continue

            if k >= 0:
                nv = na - k
                if nv > 0:
                    grad[k:na] += wm * (1.0 - x[0:nv])

            else:
                kk = -k
                nv = na - kk
                if nv > 0:
                    grad[kk:na] += wm * (1.0 - x[0:nv])
                    grad[0:nv] += wm * (-x[kk:na])
        return grad
    def eq_fun(x, na):
        return float(x.sum() - na / 2.0)
    def eq_jac(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, jac=obj_jac, args=(n, M, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_round(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:target]] = 1.0
    return h_b
