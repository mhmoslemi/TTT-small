def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    if s > 350:
        return m + (s + np.log(np.sum(np.exp(sh - s)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    w = np.exp(sh - s)
    w /= w.sum()
    return w

def _full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return float(softmax_max(c, alpha) + mu * (x.sum() - n / 2.0) ** 2)

def _full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = softmax_weights(c, alpha)
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    for m_idx in range(len(c)):
        if w[m_idx] < 1e-16:
            continue
        k = m_idx - center
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return float(softmax_max(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2)

def _active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    w = softmax_weights(vals, alpha)
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    for i, a_idx in enumerate(active):
        if w[i] < 1e-16:
            continue
        k = int(a_idx) - center
        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - x[k:])
                grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, kind='cubic')
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, kind='cubic')
    return project_feasible(h_new, n_new)

def make_golden_h(n):
    """Create a Beatty-sequence-based initial point."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 30 * n):
        pos = int(k * (phi - 1.0)) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for p in positions:
        h[p] = 1.0
    # Smooth slightly
    kernel = np.array([0.15, 0.35, 0.35, 0.15])
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    return project_feasible(h_smooth, n)

def make_symmetric_h(n):
    """Create a symmetric structured initial point."""
    h = np.zeros(n)
    half = n // 2
    # Place ones at positions based on golden ratio in [0, half)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for j in range(half):
        pos = int(j * (1.0 - 1.0 / phi)) % half
        h[pos] = 1.0
        h[n - 1 - pos] = 1.0
    return project_feasible(h, n)

def polish(h, n, deadline, alpha_start=10000, alpha_end=5000000, n_passes=4):
    """Multiple L-BFGS-B passes with increasing alpha."""
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in range(n_passes):
        if time.time() > deadline - 10:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (p / max(1, n_passes - 1))
        rem = deadline - time.time() - 5
        if rem < 10:
            break
        maxit = min(1500, max(100, int(rem * 4)))
        try:
            res = minimize(_full_obj, best_h, jac=_full_grad,
                          args=(n, alpha, 0.005), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_round(h, n, deadline, top_k=8, alpha=2000000, mu=0.001):
    """Optimize focusing on top-k active constraints."""
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-8, 0.001 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])[:top_k]
        active = active[order]
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(2000, max(200, int(rem * 3)))
    try:
        res = minimize(_active_obj, h, jac=_active_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 12,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def pair_descent(h, n, deadline, deltas=None, radius=60):
    """Coordinate pair descent to reduce max overlap."""
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        passes = 0
        while improved and passes < 5 and time.time() < deadline - 5:
            improved = False
            passes += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Focus on reducing the worst constraints
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi, max(1, (hi - lo) // 20)):
                    if i == j:
                        continue
                    # Try moving from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try moving from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def targeted_pair(h, n, deadline):
    """Target pairs that directly affect the worst overlap constraints."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    delta = 0.001
    iters = 0
    max_iters = 50000
    while iters < max_iters and time.time() < deadline - 5:
        iters += 1
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-9, 0.0005 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        found_improvement = False
        for a_idx in active[:min(len(active), 8)]:
            if time.time() > deadline - 5:
                break
            k = int(a_idx) - (n - 1)
            if k >= 0:
                # c[k] involves h[i] and h[i+k] for i in [0, n-k)
                # h[i](1-h[i+k]) - to reduce, increase h[i+k] or decrease h[i]
                for i in range(0, n - k, max(1, (n - k) // 50)):
                    a, b = i, i + k
                    if best_h[a] > delta and best_h[b] < 1.0 - delta:
                        best_h[a] -= delta
                        best_h[b] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found_improvement = True
                        else:
                            best_h[a] += delta
                            best_h[b] -= delta
                    elif best_h[b] > delta and best_h[a] < 1.0 - delta:
                        best_h[b] -= delta
                        best_h[a] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found_improvement = True
                        else:
                            best_h[b] += delta
                            best_h[a] -= delta
            else:
                kk = -k
                for i in range(0, n - kk, max(1, (n - kk) // 50)):
                    a, b = i + kk, i
                    if best_h[a] > delta and best_h[b] < 1.0 - delta:
                        best_h[a] -= delta
                        best_h[b] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found_improvement = True
                        else:
                            best_h[a] += delta
                            best_h[b] -= delta
                    elif best_h[b] > delta and best_h[a] < 1.0 - delta:
                        best_h[b] -= delta
                        best_h[a] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            found_improvement = True
                        else:
                            best_h[b] += delta
                            best_h[a] -= delta
        if not found_improvement:
            delta /= 2.0
            if delta < 5e-6:
                break
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def equalize_step(h, n, top_k=6):
    """Try to equalize the top-k overlap values by small adjustments."""
    c_vec = compute_c5_vec(h, n)
    top_idx = np.argsort(-c_vec)[:top_k]
    top_vals = c_vec[top_idx]
    mean_val = top_vals.mean()
    # For each top constraint, try to reduce it and increase others
    # Simple: just do a small L-BFGS step on a penalty that penalizes variance
    # of the top-k values
    h_new = h.copy()
    for _ in range(3):
        # Subgradient step on (c_i - mean)^2 for i in top_k
        grad = np.zeros(n)
        dx = 2.0 / n
        center = n - 1
        for a_idx in top_idx:
            k = int(a_idx) - center
            weight = (c_vec[a_idx] - mean_val) * 2.0
            if abs(weight) < 1e-10:
                continue
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[:nv] += weight * dx * (1.0 - h_new[k:])
                    grad[k:] += weight * dx * (-h_new[:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:] += weight * dx * (1.0 - h_new[:nv])
                    grad[:nv] += weight * dx * (-h_new[kk:])
        # Normalize and take small step
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            step = 0.01 * h_new.sum() / n
            h_new -= step * grad / gn
            h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    c_old = compute_c5(h, n)
    return h_new if c_new <= c_old + 1e-14 else h, min(c_new, c_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = make_golden_h(500)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Try alternative starts at n0
    for gen_func in [make_golden_h, make_symmetric_h]:
        try:
            h_alt = gen_func(n0)
            c_alt = compute_c5(h_alt, n0)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt.copy()
        except Exception:
            pass

    h_rand = np.zeros(n0)
    n_ones = n0 // 2
    idx = rng.choice(n0, size=n_ones, replace=False)
    h_rand[idx] = 1.0
    h_rand = project_feasible(h_rand, n0)
    c_rand = compute_c5(h_rand, n0)
    if c_rand < best_c:
        best_c = c_rand
        best_h = h_rand.copy()

    # Phase 1: Quick polish at original resolution (60s)
    p1_end = min(t0 + 50, deadline - 700)
    if p1_end > t0 + 10:
        best_h, best_c = polish(best_h, best_n, p1_end,
                                alpha_start=5000, alpha_end=1000000, n_passes=3)

    # Phase 2: Upsample to 2x and optimize (300s)
    n_2 = best_n * 2
    p2_end = min(time.time() + 250, deadline - 450)
    if p2_end > time.time() + 30:
        h_2 = upsample(best_h, best_n, n_2)
        c_2 = compute_c5(h_2, n_2)
        cur_h = h_2
        cur_c = c_2
        cur_n = n_2
        # Also try golden start at 2x
        h_g2 = make_golden_h(n_2)
        c_g2 = compute_c5(h_g2, n_2)
        if c_g2 < cur_c:
            cur_h, cur_c, cur_n = h_g2, c_g2, n_2
        # Polish
        cur_h, cur_c = polish(cur_h, cur_n, p2_end - 60,
                              alpha_start=10000, alpha_end=5000000, n_passes=4)
        # Active set rounds
        for top_k, alp in [(4, 2000000), (8, 5000000), (12, 10000000)]:
            if time.time() > p2_end - 20:
                break
            cur_h, cur_c = active_round(cur_h, cur_n, p2_end - 15,
                                        top_k=top_k, alpha=alp, mu=0.0005)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
            best_n = cur_n

    # Phase 3: Try 4x resolution (250s)
    n_4 = best_n
    base_n = n0
    if n0 * 4 <= 2000:
        n_4 = n0 * 4
    p3_end = min(time.time() + 200, deadline - 300)
    if p3_end > time.time() + 40 and n_4 > best_n:
        # Upsample from best current
        h_4 = upsample(best_h, best_n, n_4)
        c_4 = compute_c5(h_4, n_4)
        # Also try golden at 4x
        h_g4 = make_golden_h(n_4)
        c_g4 = compute_c5(h_g4, n_4)
        cur_h = h_4 if c_4 <= c_g4 else h_g4
        cur_c = min(c_4, c_g4)
        cur_n = n_4
        # Polish at 4x (fewer passes due to cost)
        cur_h, cur_c = polish(cur_h, cur_n, p3_end - 40,
                              alpha_start=10000, alpha_end=2000000, n_passes=2)
        # One active round
        if time.time() < p3_end - 20:
            cur_h, cur_c = active_round(cur_h, cur_n, p3_end - 10,
                                        top_k=8, alpha=5000000, mu=0.0005)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
            best_n = cur_n

    # Phase 4: Targeted pair descent (150s)
    p4_end = min(time.time() + 130, deadline - 250)
    if p4_end > time.time() + 20:
        best_h, best_c = targeted_pair(best_h, best_n, p4_end)

    # Phase 5: Active set rounds with increasing alpha (120s)
    p5_end = min(time.time() + 100, deadline - 180)
    if p5_end > time.time() + 20:
        for top_k, alp, mu in [(4, 5000000, 0.0003), (6, 10000000, 0.0002),
                                (8, 20000000, 0.0001), (10, 50000000, 0.0001)]:
            if time.time() > p5_end - 15:
                break
            best_h, best_c = active_round(best_h, best_n, p5_end - 10,
                                          top_k=top_k, alpha=alp, mu=mu)

    # Phase 6: Equalization + fine pair (100s)
    p6_end = min(time.time() + 80, deadline - 120)
    if p6_end > time.time() + 10:
        best_h, _ = equalize_step(best_h, best_n, top_k=8)
        best_c = compute_c5(best_h, best_n)
        if time.time() < p6_end - 15:
            best_h, best_c = pair_descent(best_h, best_n, p6_end - 10,
                                          deltas=[0.0005, 0.0002, 0.0001, 0.00005],
                                          radius=30)

    # Phase 7: Final targeted refinement (80s)
    p7_end = min(time.time() + 60, deadline - 60)
    if p7_end > time.time() + 10:
        best_h, best_c = targeted_pair(best_h, best_n, p7_end)

    # Phase 8: Ultra-fine final push
    p8_end = deadline - 3
    if p8_end > time.time() + 10:
        best_h, best_c = pair_descent(best_h, best_n, p8_end,
                                      deltas=[0.0001, 0.00005, 0.00002, 0.00001],
                                      radius=20)
        # One more active round
        if time.time() < p8_end - 5:
            best_h, best_c = active_round(best_h, best_n, p8_end - 1,
                                          top_k=6, alpha=20000000, mu=0.0001)

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
