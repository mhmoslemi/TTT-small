                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fn(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

        if k >= 0:

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h.copy(), jac=smooth_grad_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-16, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline, candidate_values=None):

    best_h = h.copy()
    best_c = compute_c5(h, n)

    """Snap to binary and adjust."""
    h_b = np.where(h >= 0.5, 1.0, 0.0)

            if h_b[i] == 0:
