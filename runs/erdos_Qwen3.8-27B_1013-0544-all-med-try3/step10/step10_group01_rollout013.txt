            h_cur = h4

    # Phase 5: Fine pair search (150s)
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 130, deadline - 60)
        h5, c5 = fine_pair_search(h_cur, n, p5_end, radius=40,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c5 < best_c - 1e-16:
            best_c = c5
            best_h = h5
            h_cur = h5

    # Phase 6: Another round of active set (100s)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 80, deadline - 40)
        for alpha in [50000000, 200000000]:
            if time.time() > p6_end:
                break
            for rnd in range(15):
                if time.time() > p6_end - 3:
                    break
                c_vec = compute_c5_vec(h_cur, n)
                max_val = c_vec.max()
                threshold = max_val - max(2e-9, 2e-5 * max_val)
                active = np.where(c_vec >= threshold)[0]
                if len(active) > 10:
                    order = np.argsort(-c_vec[active])
                    active = active[order[:10]]
                if len(active) == 0:
                    break
                maxit = max(100, int((p6_end - time.time()) * 2))
                try:
                    res = minimize(active_max_obj, h_cur, jac=active_max_grad,
                                  args=(n, active, alpha, 0.0001), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-14,
                                           'maxfun': maxit * 5, 'maxls': 25, 'maxcor': 35})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        best_h = h_new
                        h_cur = h_new
                except Exception:
                    pass

    # Phase 7: Final directional search (50s)
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 40, deadline - 10)
        h7, c7 = directional_line_search(h_cur, n, p7_end, max_rounds=5000)
        if c7 < best_c - 1e-16:
            best_c = c7
            best_h = h7

    # Phase 8: Ultra-fine pair (last few seconds)
    if time.time() < deadline - 5:
        p8_end = deadline - 3
        h8, c8 = fine_pair_search(best_h, n, p8_end, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c - 1e-16:
            best_c = c8
            best_h = h8

    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
