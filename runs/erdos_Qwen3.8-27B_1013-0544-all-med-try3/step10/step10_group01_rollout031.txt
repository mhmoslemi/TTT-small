import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_logsumexp(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_logsumexp_grad(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def compute_grad_for_shift(h, n, k):
    """Compute gradient of f(shift) with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        p = -k
        nv = n - p
        grad[p:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[p:n]) * dx
    return grad

    m = c.max()

def projected_subgradient_step(h, grad, n, step_size):

    h_new = h - step_size * grad

def beatty_init(n, slope, phase=0.0):
    """Create a Beatty/Sturmian word initialization."""
    ones = n // 2
    h = np.zeros(n)
    positions = set()
    for k in range(1, 3 * n):
        pos = int(k * slope + phase) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= ones:
                break
    for pos in positions:
        h[pos] = 1.0
    # Fix any remaining
    remaining = ones - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if remaining > 0 and len(zeros) > 0:
            h[zeros[0]] = remaining
        elif remaining < 0:
            ones_idx = np.where(h == 1)[0]
            h[ones_idx[0]] = 1.0 + remaining  # might be fractional
    return project_feasible(h, n)

def thue_morse_init(n):
    """Create a Thue-Morse based initialization."""
    # Thue-Morse: t(n) = number of 1s in binary representation of n, mod 2
    # We want n/2 ones in n positions
    h = np.zeros(n)
    ones_target = n // 2
    # Assign 1 where thue_morse[i] = 0 (or 1, whichever gives closer to n/2)
    tm = np.array([bin(i).count('1') % 2 for i in range(n)])
    # tm has approximately n/2 zeros and n/2 ones
    h[tm == 0] = 1.0
    # Adjust to get exactly n/2
    current = h.sum()
    if current > ones_target:
        # Need to turn some 1s to 0s
        excess = current - ones_target
        idx = np.where(h == 1)[0]
        for i in range(int(excess)):
            h[idx[i]] = 0
    elif current < ones_target:
        deficit = ones_target - current
        idx = np.where(h == 0)[0]
        for i in range(int(deficit)):
            h[idx[i]] = 1
    return project_feasible(h, n)

def upsample_h(h_old, n_old, n_new):
    """Upsample h from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def lbfgs_smooth(h, n, deadline, alpha, mu=0.01, maxit=3000):
    """L-BFGS-B with smooth max objective."""
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_logsumexp(c, alpha)
        val += mu * (x.sum() - n/2.0)**2
        return val

        w = smooth_logsumexp_grad(c, alpha)
        dx = 2.0 / n

            if wi < 1e-14:
                continue
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k

                g[k:n] += wi * (-x[0:nv]) * dx

                g[p:n] += wi * (1.0 - x[0:nv]) * dx
                g[0:nv] += wi * (-x[p:n]) * dx

        return g

def coordinate_descent(h, n, deadline, max_rounds=30):
    """Single-variable coordinate descent."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False

        for i in idx:

                break
            # Try setting h[i] to 0 or 1
            old_val = best_h[i]
            for new_val in [0.0, 1.0, 0.25, 0.5, 0.75]:
                if abs(new_val - old_val) < 1e-10:
                    continue
                # Adjust another variable to maintain sum constraint
                # Actually, let's just project after changing

                c = compute_c5(best_h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = lse_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wi_dx = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wi_dx * (1.0 - h[k:n])
            grad[k:n] += wi_dx * (-h[0:nv])
        else:
            p = -k
            nv = n - p
            grad[p:n] += wi_dx * (1.0 - h[0:nv])
            grad[0:nv] += wi_dx * (-h[p:n])
    return grad

def objective_sm(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    val = lse_val(c, alpha)
    val += mu * (h.sum() - n / 2.0) ** 2
    return val

def gradient_sm(h, n, alpha, mu):
    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alpha_list, mu_list, maxit_list):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i in range(len(alpha_list)):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        alpha = alpha_list[i]
        mu = mu_list[min(i, len(mu_list)-1)]
        maxit = maxit_list[min(i, len(maxit_list)-1)]
        maxit = min(maxit, max(200, int(remaining * 3)))
        try:
            res = minimize(objective_sm, best_h, jac=gradient_sm,
                          args=(n, alpha, mu), method='L-BFGS-B',

def pair_exchange_opt(h, n, deadline, max_rounds=30):

    best_c = compute_c5(h, n)

    for rnd in range(max_rounds):

def simulated_annealing(h, n, deadline, T_start=0.5, T_end=0.01, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    cur_c = best_c
    ones_idx = np.where(h_cur > 0.5)[0].tolist()
    zeros_idx = np.where(h_cur <= 0.5)[0].tolist()

    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / float(max_iter))
        if not ones_idx or not zeros_idx:
            break
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        if i == j:
            continue
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c_new = compute_c5(h_cur, n)
        delta = c_new - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / (T * 0.1)):
            cur_c = c_new
            ones_idx.remove(i)
            ones_idx.append(j)
            zeros_idx.remove(j)
            zeros_idx.append(i)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0

    best_c = compute_c5(best_h, n)
    return best_h, best_c

def beatty_seq(n, slope, phase=0.0):
    ones_count = n // 2
    h = np.zeros(n)
    positions = set()
    for k in range(1, 5 * n):
        pos = int(k * slope + phase) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= ones_count:
                break
    for pos in positions:
        h[pos] = 1.0
    rem = ones_count - int(h.sum())
    if rem > 0:
        zeros = np.where(h == 0)[0]
        for i in range(min(rem, len(zeros))):
            h[zeros[i]] = 1.0
    elif rem < 0:
        ones = np.where(h == 1)[0]
        for i in range(min(-rem, len(ones))):
            h[ones[i]] = 0.0
    return project_feasible(h, n)

def thue_morse_seq(n):
    ones_target = n // 2
    tm = np.array([bin(i).count('1') & 1 for i in range(n)])
    h = np.zeros(n)
    h[tm == 0] = 1.0
    cur = int(h.sum())
    if cur > ones_target:
        excess = cur - ones_target
        idx = np.where(h == 1)[0]
        for i in range(excess):
            h[idx[i]] = 0
    elif cur < ones_target:
        deficit = ones_target - cur
        idx = np.where(h == 0)[0]
        for i in range(deficit):
            h[idx[i]] = 1
    return project_feasible(h, n)

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
