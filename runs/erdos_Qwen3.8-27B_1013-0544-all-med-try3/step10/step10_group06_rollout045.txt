    no_improve = 0

    while time.time() < deadline and rounds < max_rounds:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_K:]
        improved = False

        for step_mult in [1.0, 0.5, 0.25, 0.12, 0.06]:
            if time.time() > deadline:
                break
            base_step = 0.008 * step_mult

            for k in range(200):
                if time.time() > deadline:
                    break
                ki = top_idx[rng.randint(top_K)]
                lag = ki - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue

                # Find the pair with largest contribution for this lag
                if lag > 0:
                    contrib = h[0:n-m] * (1.0 - h[m:n])
                else:
                    contrib = h[m:n] * (1.0 - h[0:n-m])

                # Pick from top contributors with some randomness
                if rng.random() < 0.7:
                    # Pick from top 10% of contributors
                    n_top = max(1, (n - m) // 10)
                    top_contrib_idx = np.argpartition(contrib, -n_top)[-n_top:]
                    local_i = top_contrib_idx[rng.randint(n_top)]
                else:
                    local_i = rng.randint(0, n - m)

                if lag > 0:
                    i = local_i
                    j = local_i + m
                else:
                    i = local_i + m
                    j = local_i

                if h[i] <= h[j] + 1e-10:
                    continue

                d = min(base_step, (h[i] - h[j]) * 0.3)
                if d < 1e-10:
                    continue

                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)

                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                else:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)

            if improved:
                h = best_h.copy()
                break

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break

        rounds += 1

    return best_h, best_c

def fine_pair_search(h, n, deadline, rng, top_K=5):
    """Very fine-grained search: try small adjustments at the single worst pair for each top lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for outer in range(200):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_K:]
        improved = False

        for ki in top_idx:
            if time.time() > deadline:
                break
            lag = ki - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            if lag > 0:
                contrib = h[0:n-m] * (1.0 - h[m:n])
            else:
                contrib = h[m:n] * (1.0 - h[0:n-m])

            # Find top 50 contributors
            n_cand = min(50, n - m)
            cand_idx = np.argpartition(contrib, -n_cand)[-n_cand:]

            for local_i in cand_idx:
                if time.time() > deadline:
                    break
                if lag > 0:
                    i = local_i
                    j = local_i + m
                else:
                    i = local_i + m
                    j = local_i

                if h[i] <= h[j] + 1e-12:
                    continue

                # Try several step sizes
                for frac in [0.5, 0.3, 0.15, 0.08, 0.04]:
                    d = (h[i] - h[j]) * frac
                    if d < 1e-12:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                    if improved:
                        break

                if improved:
                    h = best_h.copy()
                    break

            if improved:
                break

        if not improved:
            break

    return best_h, best_c

def binary_pattern(n, period, offset=0, duty=0.5):
    """Generate a binary pattern with given period and duty cycle."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i + offset) % period
        if phase < int(period * duty):
            h[i] = 1.0
    h = project_feasible(h, n)
    return h

def gradient_descent_top(h, n, deadline, alpha, T, lr_init=0.002, lr_decay=0.999, max_iters=100000):
    """Gradient descent focusing on top T lags with step size decay."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = lr_init

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argpartition(c_all, -T)[-T:]

        # Compute averaged gradient over top T lags (uniform weights)
        grad = np.zeros(n)
        for ki in top_idx:
            lag = ki - (n - 1)
            grad += grad_c_lag(h, n, lag) / T

        # Normalized gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Projected gradient step towards reducing max overlap
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)

        c_new = compute_all_lags(h_new, n)
        max_new = float(c_new.max())
        max_old = float(c_all.max())

        if max_new < max_old - 1e-15:
            h = h_new
            if max_new < best_c - 1e-14:
                best_c = max_new
                best_h = h.copy()
        else:
            lr *= 0.95

        if it % 50 == 0:
            lr *= lr_decay

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Quick L-BFGS-B polish at initial resolution
    for alpha, maxit in [(5000, 300), (50000, 400), (500000, 500)]:
        if time.time() > deadline - budget_s * 0.6:
            break
        h_opt = lbfgsb_optimize(best_h, n_init, alpha, time.time() + 30, maxiter=maxit)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Upsample to n=800
    n2 = 800
    h2 = interpolate_h(best_h, n_init, n2)
    c2 = compute_c5(h2, n2)
    n_use = n2
    h_use = h2
    c_use = c2

    if time.time() < deadline - budget_s * 0.4:
        # L-BFGS-B at higher resolution with increasing alpha
        for alpha, maxit in [(1000, 400), (10000, 500), (100000, 600), (1000000, 800)]:
            if time.time() > deadline - budget_s * 0.35:
                break
            remaining = deadline - budget_s * 0.35 - time.time()
            if remaining < 10:
                break
            h_opt = lbfgsb_optimize(h_use, n_use, alpha, time.time() + min(45, remaining * 0.7),
                                    maxiter=maxit)
            c_opt = compute_c5(h_opt, n_use)
            if c_opt < c_use:
                c_use = c_opt
                h_use = h_opt

    # Phase 3: Top-lag gradient descent
    if time.time() < deadline - budget_s * 0.25:
        gd_deadline = min(deadline - budget_s * 0.25, time.time() + 120)
        for T in [3, 5, 8, 12]:
            if time.time() > gd_deadline:
                break
            h_opt, c_opt = gradient_descent_top(h_use, n_use, min(gd_deadline, time.time() + 30),
                                                alpha=5000, T=T, lr_init=0.001,
                                                lr_decay=0.9995)
            if c_opt < c_use:
                c_use = c_opt
                h_use = h_opt

    # Phase 4: Multi-start with binary patterns
    if time.time() < deadline - budget_s * 0.2:
        patterns = []
        for period in [7, 9, 11, 13, 17, 21, 25]:
            for off in [0, 1, 2]:
                patterns.append(binary_pattern(n_use, period, off))
        # Also try the current solution thresholded
        h_bin = (h_use > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n_use)
        patterns.append(h_bin)

        for pat in patterns[:6]:
            if time.time() > deadline - budget_s * 0.2:
                break
            remaining = deadline - budget_s * 0.2 - time.time()
            if remaining < 20:
                break
            h_pat = lbfgsb_optimize(pat, n_use, 50000, time.time() + min(25, remaining * 0.6),
                                    maxiter=800)
            c_pat = compute_c5(h_pat, n_use)
            if c_pat < c_use:
                c_use = c_pat
                h_use = h_pat
                # Polish
                h_pol = lbfgsb_optimize(h_pat, n_use, 1000000, time.time() + 15, maxiter=600)
                c_pol = compute_c5(h_pol, n_use)
                if c_pol < c_use:
                    c_use = c_pol
                    h_use = h_pol

    # Phase 5: Targeted local search
    if time.time() < deadline - budget_s * 0.12:
        ls_deadline = min(deadline - budget_s * 0.12, time.time() + 150)
        h_opt, c_opt = targeted_local_search(h_use, n_use, ls_deadline, rng, top_K=8)
        if c_opt < c_use:
            c_use = c_opt
            h_use = h_opt

    # Phase 6: Fine pair search
    if time.time() < deadline - budget_s * 0.08:
        fp_deadline = min(deadline - budget_s * 0.08, time.time() + 80)
        h_opt, c_opt = fine_pair_search(h_use, n_use, fp_deadline, rng, top_K=4)
        if c_opt < c_use:
            c_use = c_opt
            h_use = h_opt

    # Phase 7: Another round of top-lag gradient descent + L-BFGS-B
    if time.time() < deadline - budget_s * 0.05:
        for alpha in [200000, 2000000]:
            if time.time() > deadline - budget_s * 0.05:
                break
            h_opt = lbfgsb_optimize(h_use, n_use, alpha, time.time() + 25, maxiter=800)
            c_opt = compute_c5(h_opt, n_use)
            if c_opt < c_use:
                c_use = c_opt
                h_use = h_opt

    # Phase 8: Final fine-tuning
    if time.time() < deadline - 5:
        fp_deadline2 = min(deadline - 3, time.time() + 40)
        h_opt, c_opt = fine_pair_search(h_use, n_use, fp_deadline2, rng, top_K=3)
        if c_opt < c_use:
            c_use = c_opt
            h_use = h_opt

        # Final L-BFGS-B with very high alpha
        if time.time() < deadline - 3:
            h_opt = lbfgsb_optimize(h_use, n_use, 50000000, deadline - 2, maxiter=2000)
            c_opt = compute_c5(h_opt, n_use)
            if c_opt < c_use:
                c_use = c_opt
                h_use = h_opt

    # Update global best
    if c_use < best_c:
        best_c = c_use
        best_h = h_use

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
