def compute_c5_vec(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

    sh = alpha * (c - m)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def compute_c5_counts(h, n):

    return np.correlate(h, 1.0 - h, mode='full').astype(int)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):

    for mm in range(n_full):

            continue
        k = mm - (n - 1)

        if k >= 0:

            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def make_beatty(n, alpha, seed=42):

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    # Beatty sequence: positions floor(k * alpha) for k = 0, 1, 2, ...
    positions = []
    seen = set()
    for k in range(n * 4):
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    # Fix if needed
    if h.sum() != target:
        diff = target - int(h.sum())
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for i in zeros[:diff]:
                h[i] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for i in ones[:abs(diff)]:
                h[i] = 0.0
    return h

def make_halton(n, target):
    """Create a Halton sequence binary vector"""
    h = np.zeros(n)
    # Use 1D low-discrepancy sequence
    positions = []
    for k in range(1, n * 3):
        # Van der Corput base 2
        x = 0.0
        f = 0.5
        kk = k
        while kk > 0:
            x += f * (kk % 2)
            kk //= 2
            f *= 0.5

                break
    for p in positions:
        h[p] = 1.0
    return h

    zeros = np.where(h < 0.5)[0].tolist()

        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - current_c

            h[j] = 0.0
