            if no_improve > 300:
                break
            no_improve += 1
    
    return best_h, best_c

def fine_coord_descent(h, n, deadline, max_rounds=300):
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < max_rounds and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.002 * max_val)
            w = smooth_max_w(c_vec, 150000.0)
            grad = h_grad(w, best_h, n)
            
            pos_mask = (grad > 1e-11) & (best_h > 1e-8)
            neg_mask = (grad < -1e-11) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0:
                break
            
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def multistart_binary(h_init, n, deadline, rng, num_starts=3):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    
    for si in range(num_starts):
        if time.time() > deadline:
            break
        h_tmp = binarize(h_init, n).copy()
        ones = np.where(h_tmp > 0.5)[0]
        zeros = np.where(h_tmp < 0.5)[0]
        num_swaps = rng.randint(10, 40)
        for _ in range(num_swaps):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]
        
        sub_dead = min(deadline, time.time() + max(5, (deadline - time.time()) / (num_starts - si)))
        h_new, c_new = binary_local_search(h_tmp, n, sub_dead, rng, use_multibit=True)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    n = n_parent
    h = project_feasible(h_parent, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Phase 1: Quick L-BFGS to get smooth starting point (40s)
    p1_dead = min(t0 + 40, deadline - 800)
    h1, c1 = run_lbfgs(h, n, p1_dead, alpha=5000, mu=0.01, maxit=2000)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    h1b, c1b = run_lbfgs(best_h, n, min(t0 + 55, deadline - 780), alpha=50000, mu=0.001, maxit=1500)
    if c1b < best_c:
        best_c = c1b
        best_h = h1b
    
    # Phase 2: Intensive binary local search (400s)
    p2_dead = min(t0 + 455, deadline - 350)
    h_bin = binarize(best_h, n)
    h2, c2 = binary_local_search(h_bin, n, p2_dead, rng, use_multibit=True)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    
    # Phase 3: Gradient binary search on result (150s)
    p3_dead = min(t0 + 605, deadline - 200)
    h3, c3 = gradient_binary_search(best_h, n, p3_dead, rng, max_iters=200000)
    if c3 < best_c:
        best_c = c3
        best_h = h3
    
    # Phase 4: Multi-start with larger perturbations (120s)
    p4_dead = min(t0 + 725, deadline - 130)
    h4, c4 = multistart_binary(best_h, n, p4_dead, rng, num_starts=3)
    if c4 < best_c:
        best_c = c4
        best_h = h4
    
    # Phase 5: L-BFGS continuous refinement (80s)
    p5_dead = min(t0 + 805, deadline - 60)
    for alpha in [100000, 500000, 2000000]:
        if time.time() > p5_dead - 5:
            break
        h5, c5 = run_lbfgs(best_h, n, min(p5_dead, time.time() + 25), alpha=alpha, mu=0.0001, maxit=1500)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Fine coordinate descent (60s)
    p6_dead = min(t0 + 865, deadline - 30)
    if time.time() < p6_dead:
        h6, c6 = fine_coord_descent(best_h, n, p6_dead, max_rounds=200)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final binary polish (40s)
    p7_dead = min(t0 + 905, deadline - 15)
    if time.time() < p7_dead:
        h7, c7 = binary_local_search(binarize(best_h, n), n, p7_dead, rng, use_multibit=True)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Ultra-fine coord descent
    if time.time() < deadline - 5:
        h8, c8 = fine_coord_descent(best_h, n, deadline - 3, max_rounds=100)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Final L-BFGS with very high alpha
    if time.time() < deadline - 3:
        h9, c9 = run_lbfgs(best_h, n, deadline - 2, alpha=5000000, mu=0.0, maxit=1000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
