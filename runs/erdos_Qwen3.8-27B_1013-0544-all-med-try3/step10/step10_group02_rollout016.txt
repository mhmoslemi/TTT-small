import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) > 0:
                h[idx[:min(len(idx), int(np.ceil(abs(diff))))]] += diff / len(idx) if len(idx) < int(abs(diff))+1 else 0
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    if mx_sh > 500:
        val = m + (mx_sh + np.log(np.sum(np.exp(sh - mx_sh)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx_sh = sh.max()
    w = np.exp(sh - mx_sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_max_obj(x, n, active, alpha, mu=0.0):

    sh = alpha * (vals - mx)

def active_max_grad(x, n, active, alpha, mu=0.0):

    vals = c[active]

def fourier_project(h, n, m):
    """Project h onto first m Fourier modes (periodic on [0,2])"""
    x = np.linspace(0, 2, n, endpoint=False)
    # Compute DCT-like projection
    coeffs = np.zeros(2 * m + 1)
    coeffs[0] = np.mean(h)
    for k in range(1, m + 1):
        coeffs[k] = (2.0 / n) * np.sum(h * np.cos(k * np.pi * x))
        coeffs[m + k] = (2.0 / n) * np.sum(h * np.sin(k * np.pi * x))
    # Reconstruct
    h_f = np.zeros(n)
    h_f += coeffs[0]
    for k in range(1, m + 1):
        h_f += coeffs[k] * np.cos(k * np.pi * x) + coeffs[m + k] * np.sin(k * np.pi * x)
    return h_f

def fourier_fourier_eval(coeffs, n, m):
    """Evaluate h from Fourier coefficients"""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    h += coeffs[0]
    for k in range(1, m + 1):
        h += coeffs[k] * np.cos(k * np.pi * x) + coeffs[m + k] * np.sin(k * np.pi * x)
    return h

def fourier_optimize(h, n, m, deadline, maxiter=2000):
    """Optimize in Fourier subspace"""
    x = np.linspace(0, 2, n, endpoint=False)
    coeffs = np.zeros(2 * m + 1)
    coeffs[0] = np.mean(h)
    for k in range(1, m + 1):
        coeffs[k] = (2.0 / n) * np.sum(h * np.cos(k * np.pi * x))
        coeffs[m + k] = (2.0 / n) * np.sum(h * np.sin(k * np.pi * x))
    
    def f_obj(c):
        h_f = fourier_fourier_eval(c, n, m)
        h_f = np.clip(h_f, 0, 1)
        # Enforce integral
        diff = n / 2.0 - h_f.sum()
        if abs(diff) > 1e-10:
            h_f = h_f + diff / n
            h_f = np.clip(h_f, 0, 1)
        return compute_c5(h_f, n)
    
    best_c = f_obj(coeffs)
    best_coeffs = coeffs.copy()
    
    # Use Nelder-Mead for the low-dimensional problem
    options = {'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12}
    try:
        res = minimize(f_obj, coeffs, method='Nelder-Mead', options=options)
        c_test = f_obj(res.x)
        if c_test < best_c:
            best_c = c_test
            best_coeffs = res.x
    except Exception:
        pass
    
    h_out = fourier_fourier_eval(best_coeffs, n, m)
    h_out = project_feasible(h_out, n)
    c_out = compute_c5(h_out, n)
    return h_out, c_out

def lbfgs_run(h, n, deadline, alphas, mu=0.01, maxiter_per=None):

    best_c = compute_c5(best_h, n)

            break

                          args=(n, alpha, mu), method='L-BFGS-B',

                                   'gtol': 1e-16, 'maxfun': maxit * 8,

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def downsample(h, n_old, n_new):
    """Downsample h from n_old points to n_new points using average pooling"""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)

def upsample(h, n_old, n_new):
    """Upsample h from n_old points to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

            alphas=[5000, 50000, 500000], mu=0.01, maxiter_per=[800, 800, 1000])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Randomized restarts
    if time.time() < deadline - 50:
        for attempt in range(4):
            if time.time() > deadline - 40:
                break
            noise_s = 0.02 * (1.0 - attempt * 0.2)
            h_nr = best_h + rng.randn(best_n) * noise_s
            h_nr = project_feasible(h_nr, best_n)
            h_nr2, c_nr2 = run_active(h_nr, best_n, min(time.time() + 15, deadline - 10),
                top_k=4, n_rounds=4)
            if c_nr2 < best_c:
                best_c = c_nr2
                best_h = h_nr2

    # Phase 9: Final active-set polish
    if time.time() < deadline - 30:
        p9_end = min(time.time() + 20, deadline - 15)
        h9, c9 = run_active(best_h, best_n, p9_end, top_k=3, n_rounds=5)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Phase 10: Fine pair search (most important for last improvements)
    if time.time() < deadline - 5:
        p10_end = deadline - 2
        h10, c10 = fine_pair_search(best_h, best_n, p10_end, radius=25)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final: re-project and verify
    fn = len(best_h)
    best_h = project_feasible(best_h, fn)
    best_c = compute_c5(best_h, fn)
    if best_c < 0:
        best_c = 0.0

    return best_h, best_c, fn
