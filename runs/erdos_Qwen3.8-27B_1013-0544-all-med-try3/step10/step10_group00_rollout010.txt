import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    T_initial = 0.001  # Initial temperature
    T_final = 0.00001  # Final temperature
    
    it = 0
    max_iters = 500000  # Maximum iterations
    
    # Record the sorted correlations for tracking
    while time.time() < deadline and it < max_iters:
        progress = it / max_iters
        T = T_initial * (T_final / T_initial) ** progress
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[j] = 1.0

            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

                best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if diff > 1e-10:

        h[zeros[0]] = 1.0
    elif diff < -1e-10:
