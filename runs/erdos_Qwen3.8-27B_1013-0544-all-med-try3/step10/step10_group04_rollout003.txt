    """One step of simulated annealing: swap two random entries."""
    i = rng.randint(n)
    j = rng.randint(n)
    if i == j:
        return h, False
    # Compute C5 before
    c_old = compute_c5(h, n)
    # Swap
    h[i], h[j] = h[j], h[i]
    c_new = compute_c5(h, n)
    # Accept or reject
    if c_new < c_old or rng.rand() < np.exp(-(c_new - c_old) / T):
        return h, True
    else:
        h[i], h[j] = h[j], h[i]  # undo
        return h, False

def smooth_obj(x, n, alpha, mu):

    dx = 2.0 / n
    m = c.max() * dx

    else:
        val = m + dx * np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m_c = c.max()
    sh = alpha * (c - m_c)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

                grad[0:nv] += wm * (-x[kk:n])

            res2 = minimize(smooth_obj, h_bin, jac=smooth_grad,
                           args=(n, 3000, 5000.0), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 3000, 'ftol': 1e-18, 'maxls': 50})
            h_new2 = project_feasible(res2.x, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick L-BFGS at base resolution (n=~500) - 50s
    t1 = min(t0 + 50, deadline - 950)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(200, 50.0), (1000, 500.0), (5000, 5000.0), (20000, 50000.0)])
    
    # Phase 2: Perturbation at base - 50s
    t2 = min(t0 + 100, deadline - 900)
    if t2 > time.time() + 10:
        hp, cp = perturb_and_optimize(best_h, n_base, t2, n_trials=6, seed_p=100)
        if cp < best_c:
            best_c = cp
            best_h = hp
    
    # Phase 3: Upsample to n=1500, strong L-BFGS - 200s
    n_mid = 1500
    t3_start = max(time.time(), t0 + 105)
    t3_end = min(t0 + 310, deadline - 550)
    if t3_end > t3_start + 30:
        h_mid = interpolate_to(best_h, len(best_h), n_mid)
        h_mid, c_mid = run_lbfgs(h_mid, n_mid, t3_end,
            configs=[(500, 200.0), (2000, 2000.0), (8000, 10000.0),
                     (20000, 50000.0), (50000, 200000.0), (100000, 500000.0)])
        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
        else:
            best_h = h_mid  # still use as starting point
    
    # Phase 4: SLSQP polish at n=1500 - 80s
    t4_end = min(t0 + 390, deadline - 470)
    if t4_end > time.time() + 15:
        hs, cs = slqp_polish(best_h, len(best_h), t4_end)
        if cs < best_c:
            best_c = cs
            best_h = hs
    
    # Phase 5: Swap-based search at n=1500 - 120s
    t5_end = min(t0 + 510, deadline - 350)
    if t5_end > time.time() + 20:
        hs, cs = swap_search(best_h, n_mid, t5_end, n_max_swaps=300000, seed_sw=777)
        if cs < best_c:
            best_c = cs
            best_h = hs
    
    # Phase 6: Upsample to n=2500, L-BFGS - 200s
    n_hi = 2500
    t6_start = max(time.time(), t0 + 515)
    t6_end = min(t0 + 720, deadline - 200)
    if t6_end > t6_start + 30:
        h_hi = interpolate_to(best_h, len(best_h), n_hi)
        h_hi, c_hi = run_lbfgs(h_hi, n_hi, t6_end,
            configs=[(2000, 2000.0), (10000, 10000.0), (30000, 50000.0),
                     (80000, 200000.0), (150000, 500000.0)])
        if c_hi < best_c:
            best_c = c_hi
            best_h = h_hi
        else:
            best_h = h_hi
    
    # Phase 7: SLSQP at high res - 60s
    t7_end = min(t0 + 780, deadline - 120)
    if t7_end > time.time() + 15:
        hs, cs = slqp_polish(best_h, len(best_h), t7_end)
        if cs < best_c:
            best_c = cs
            best_h = hs
    
    # Phase 8: Swap search at high res - 80s
    t8_end = min(t0 + 860, deadline - 60)
    if t8_end > time.time() + 20:
        hs, cs = swap_search(best_h, len(best_h), t8_end, n_max_swaps=150000, seed_sw=888)
        if cs < best_c:
            best_c = cs
            best_h = hs
    
    # Phase 9: Targeted refinement - remaining time
    t9_end = deadline - 3
    if t9_end > time.time() + 5:
        best_h, best_c = targeted_refine(best_h, len(best_h), t9_end)
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
