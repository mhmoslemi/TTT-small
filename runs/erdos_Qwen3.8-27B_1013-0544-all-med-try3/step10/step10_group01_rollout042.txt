def compute_c5(h, n):
    dx = 2.0 / n
    return float((np.correlate(h, 1 - h, mode='full') * dx).max())

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def thue_morse(n):
    """Generate first n bits of Thue-Morse sequence."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def thue_morse_binary(n):
    """Generate Thue-Morse based binary vector with exactly n//2 ones."""
    t = thue_morse(n)
    h = t.astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Remove some 1s (prefer those that help)
        ones_idx = np.where(h == 1)[0]
        # Just remove randomly for now
        to_remove = ones_idx[:n_ones - target]
        h[to_remove] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = zeros_idx[:target - n_ones]
        h[to_add] = 1
    return h

def rudin_shapiro_binary(n):
    """Generate Rudin-Shapiro based binary vector."""
    r = np.zeros(n, dtype=int)
    r[0] = 1
    if n > 1:
        r[1] = 1
    for i in range(2, n):
        if i % 2 == 0:
            r[i] = r[i // 2]
        else:
            r[i] = -r[i // 2]
    h = ((r + 1) // 2).astype(float)  # Map -1→0, 1→1
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        h[ones_idx[:n_ones - target]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        h[zeros_idx[:target - n_ones]] = 1
    return h

def weyl_binary(n, alpha):
    """Generate Weyl sequence based binary vector."""
    # Position of the k-th 1 is floor(k * alpha * n / (n//2)) mod n
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(n_ones):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
        else:
            # Collision, find next available
            p = pos + 1
            while p in positions and p < n:
                p += 1
            if p < n:
                positions.add(p)
            else:
                p = pos - 1
                while p in positions and p >= 0:
                    p -= 1
                if p >= 0:
                    positions.add(p)
    for pos in positions:
        h[pos] = 1.0
    # If we have too few (due to collisions), add more
    while h.sum() < n_ones:
        zero_pos = np.where(h == 0)[0]
        if len(zero_pos) == 0:
            break
        h[zero_pos[0]] = 1.0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

For a binary vector h, the "change in C₅" from swapping h[i]:1→0 and h[j]:0→1 can be computed as:
- For each lag k>0:
  - Old contribution to c_{-k} (1→0 at distance k): if h[i]=1 and h[i+k]=0, this pair was counted. After swap, h[i]=0, so it's not counted anymore. Change: -1.
  - If h[i+k]=1 and h[i]=1, after swap h[i]=0, new pair (0,1) at distance k. This increases c_k (0→1) by 1.
  - Similarly for h[j]: h[j] was 0, now 1.

    - If h[j+k]=0, after swap h[j]=1, new pair (1,0) at distance k, counted in c_{-k}. Change: +1.

def compute_c5(h, n):
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def thue_morse_binary(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = t.astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        rng_tmp = np.random.RandomState(123)
        rng_tmp.shuffle(ones_idx)
        h[ones_idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        rng_tmp = np.random.RandomState(456)
        rng_tmp.shuffle(zeros_idx)
        h[zeros_idx[:target - n_ones]] = 1.0
    return h

def rudin_shapiro_binary(n):
    r = np.zeros(n, dtype=np.int8)
    r[0] = 1
    if n > 1:
        r[1] = 1
    for i in range(2, n):
        if i % 2 == 0:
            r[i] = r[i // 2]
        else:
            r[i] = -r[i // 2]
    h = ((r + 1) // 2).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]

def beatty_binary(n, alpha, seed=42):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    rng = np.random.RandomState(seed)
    k = 1
    max_k = 20 * n
    while len(positions) < n_ones and k < max_k:
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
        k += 1
    for pos in positions:
        h[pos] = 1.0
    if h.sum() < n_ones:
        zeros_idx = np.where(h == 0)[0]
        rng.shuffle(zeros_idx)
        h[zeros_idx[:int(n_ones - h.sum())]] = 1.0
    return h

def weyl_binary(n, alpha, seed=42):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    rng = np.random.RandomState(seed)
    for k in range(n_ones):
        pos = int((k + 0.5) * alpha * n / n_ones) % n
        while pos in positions:
            pos = (pos + 1) % n
        positions.add(pos)
    for pos in positions:
        h[pos] = 1.0
    return h

def round_to_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(-h[ones_idx])
        h_bin[ones_idx[order[:n_ones - target]]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        h_bin[zeros_idx[order[:target - n_ones]]] = 1.0
    return h_bin

def multi_scale_construction(n, alpha1, alpha2):
    """Construct a multi-scale sequence using two irrational frequencies."""
    h = np.zeros(n)
    n_ones = n // 2
    score = np.zeros(n)
    for k in range(n):
        # Score based on how "empty" the neighborhood is
        # Use two quasi-periodic signals
        s1 = np.sin(2 * np.pi * alpha1 * k / n)
        s2 = np.sin(2 * np.pi * alpha2 * k / n)
        score[k] = -s1 * s2  # Prefer where both signals agree
    # Select top n_ones positions
    idx = np.argsort(-score)[:n_ones]
    h[idx] = 1.0
    return h
