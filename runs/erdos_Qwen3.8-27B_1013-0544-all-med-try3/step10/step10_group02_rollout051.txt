import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def penalty_obj(x, n, mu):
    """Minimize max c + penalty for violations above current max."""
    c = compute_c5_vec(x, n)
    m = c.max()
    violations = np.maximum(0, c - m + 0.001)
    val = m + mu * np.sum(violations**2)
    val += 100.0 * (x.sum() - n/2.0)**2
    return float(val)

def penalty_grad(x, n, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Gradient of max term (use uniform over active set for stability)
    tol = 0.001
    active = c >= m - tol
    n_active = max(1, active.sum())
    for idx in np.where(active)[0]:
        k = idx - (n-1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += dx * (1.0 - x[k:n]) / n_active
            grad[k:n] += dx * (-x[0:nv]) / n_active
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += dx * (1.0 - x[0:nv]) / n_active
            grad[0:nv] += dx * (-x[kk:n]) / n_active
    
    # Gradient of penalty term
    for idx in range(2*n-1):
        viol = c[idx] - m + 0.001
        if viol > 0:
            w = 2 * mu * viol
            k = idx - (n-1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += w * dx * (1.0 - x[k:n])
                grad[k:n] += w * dx * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += w * dx * (1.0 - x[0:nv])
                grad[0:nv] += w * dx * (-x[kk:n])
    
    grad += 200.0 * (x.sum() - n/2.0)
    return grad
