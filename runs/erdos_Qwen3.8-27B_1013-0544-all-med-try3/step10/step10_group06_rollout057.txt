```python
def project_box_sum(v, target_sum):
    """Project v onto {0 <= x <= 1, sum(x) = target_sum}"""
    v = v.copy()
    # First project onto box
    v = np.clip(v, 0, 1)
    # Then fix sum with bisection
    lo, hi = -1, 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)
```

This is the standard Euclidean projection onto the intersection of a box and a halfspace (hyperplane).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000, penalty_coeff=1e6):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:

def subgradient_optimize(h, n, deadline, lr_init=0.1, lr_decay=0.9999, n_iters=20000):
    """Subgradient method with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    target_sum = n / 2.0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        c_max = c_all.max()
        if c_max < best_c:
            best_c = float(c_max)
            best_h = h.copy()
        
        # Find near-maximizing lags
        threshold = c_max - 0.05 * c_max
        active = np.where(c_all >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            w = c_all[k_idx] / c_all[active].sum()  # weight by value
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += w * (1.0 - h[0:n - lag])
                grad[0:n - lag] += w * (-h[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w * (1.0 - h[idx:n])
                grad[idx:n] += w * (-h[0:n - idx])
        
        grad *= (2.0 / n)  # dx factor
```

```python
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])
    return grad * dx

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:

            if tot < 1e-14:
                break
            h += (diff / tot) * h

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000, penalty_coeff=1e6):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)

            if h[i] > h[j] + 1e-8:
                d = rng.uniform(0.001, max_step)
                d = min(d, h[i] * 0.5, (1.0 - h[j]) * 0.5)
            elif h[j] > h[i] + 1e-8:
                d = -rng.uniform(0.001, max_step)
                d = max(d, -h[j] * 0.5, -(1.0 - h[i]) * 0.5)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-5:]
            k_star = top_k[rng.randint(5)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m if lag > 0 else i - m
            if h[i] > h[j] + 1e-7:
                d = rng.uniform(0.001, 0.03)
                d = min(d, (h[i] - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-7:
                d = -rng.uniform(0.001, 0.03)
                d = max(d, -(h[j] - h[i]) * 0.3)
            else:
                T *= cool
                continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        c_old = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = c_new.max()

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min * 2:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if max_new < best_c - 1e-15:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool

        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B at original resolution
    n_orig = len(best_h)
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 600:
            break
        h_opt = lbfgsb_optimize(best_h, n_orig, alpha, time.time() + 40, maxiter=800)
        c_opt = compute_c5(h_opt, n_orig)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Interpolate to n=1000
    n_hi = 1000
    h_hi = expand_to_resolution(best_h, len(best_h), n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    if c_hi < best_c:
        best_c = c_hi
        best_h = h_hi.copy()

    # Phase 3: L-BFGS-B at n=1000 with increasing alpha
    for alpha in [10000, 100000, 1000000, 10000000]:
        if time.time() > deadline - 400:
            break
        h_opt = lbfgsb_optimize(h_hi, n_hi, alpha, time.time() + 50, maxiter=1500)
        c_opt = compute_c5(h_opt, n_hi)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        else:
            h_opt = h_hi

    # Re-interpolate if needed
    if len(h_hi) != n_hi:
        h_hi = best_h if len(best_h) == n_hi else expand_to_resolution(best_h, len(best_h), n_hi)
    else:
        h_hi = best_h.copy() if len(best_h) == n_hi else h_hi

    # Phase 4: Subgradient at n=1000
    if time.time() < deadline - 300:
        h_sub, c_sub = subgradient_optimize(h_hi, n_hi, min(deadline - 250, time.time() + 80))
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub.copy()
            h_hi = h_sub.copy()
        else:
            h_hi = best_h.copy() if len(best_h) == n_hi else h_hi

    # Phase 5: Systematic local search at n=1000 (main improvement)
    if time.time() < deadline - 150:
        h_ls, c_ls = systematic_local_search(h_hi, n_hi, min(deadline - 100, time.time() + 300),
                                             top_k=15, max_sweeps=2000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()
            h_hi = h_ls.copy()

    # Phase 6: Fine local search
    if time.time() < deadline - 80:
        h_fs, c_fs = fine_local_search(h_hi, n_hi, min(deadline - 40, time.time() + 120), top_k=8)
        if c_fs < best_c:
            best_c = c_fs
            best_h = h_fs.copy()
            h_hi = h_fs.copy()

    # Phase 7: SA for escape
    if time.time() < deadline - 100:
        h_sa, c_sa = simulated_annealing(h_hi, n_hi, min(deadline - 60, time.time() + 80), rng,
                                         alpha_sa=150000, t_init=0.003, t_min=0.00002, cool=0.99992)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
            h_hi = h_sa.copy()

    # Phase 8: Another systematic local search round
    if time.time() < deadline - 60:
        h_ls2, c_ls2 = systematic_local_search(h_hi, n_hi, min(deadline - 30, time.time() + 60),
                                               top_k=10, max_sweeps=1000)
        if c_ls2 < best_c:
            best_c = c_ls2
            best_h = h_ls2.copy()

    # Phase 9: L-BFGS-B final polish
    if time.time() < deadline - 30:
        n_final = len(best_h)
        h_pol = lbfgsb_optimize(best_h, n_final, 50000000, deadline - 15, maxiter=3000)
        c_pol = compute_c5(h_pol, n_final)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # Phase 10: Final fine search
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_ff, c_ff = fine_local_search(best_h, n_final, deadline - 2, top_k=5)
        if c_ff < best_c:
            best_c = c_ff
            best_h = h_ff

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
```
